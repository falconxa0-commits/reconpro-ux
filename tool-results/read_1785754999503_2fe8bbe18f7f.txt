    92→# ════════════════════════════════════════════════════════════════════════════════

    94→
    95→class FindingDetailModal(ModalScreen):
    96→    """Modal overlay showing full finding details."""
    97→
    98→    def __init__(self, finding: Dict[str, Any]) -> None:
    99→        super().__init__()
   100→        self.finding = finding
   101→
   102→    def compose(self) -> ComposeResult:
   103→        f = self.finding
   104→        sev = f.get("severity", "info").lower()
   105→        sev_color = SEV_STYLES.get(sev, "white")
   106→        title = f.get("title", "Unknown")
   107→        category = f.get("category", "")
   108→        module = f.get("module", "")
   109→        pts = f.get("points_deducted", 0)
   110→        evidence = f.get("evidence", "")
   111→        remediation = f.get("remediation", "No remediation available.")
   112→        description = f.get("description", "")
   113→        asset = f.get("asset", "")
   114→        related_cves = f.get("related_cves", [])
   115→
   116→        lines: List[str] = []
   117→        lines.append(f"[{sev_color}][{sev.upper()}] {title}[/]")
   118→        lines.append("")
   119→        if asset:
   120→            lines.append(f"[dim]Asset:[/] [cyan]{asset}[/]")
   121→        if category:
   122→            lines.append(f"[dim]Category:[/] {category}")
   123→        if module:
   124→            lines.append(f"[dim]Module:[/] {module}")
   125→        lines.append(f"[dim]Points:[/] -{pts}")
   126→        if related_cves:
   127→            cve_ids = ", ".join(
   128→                c.get("cve_id", "") for c in related_cves[:5] if c.get("cve_id")
   129→            )
   130→            if cve_ids:
   131→                lines.append(f"[dim]CVEs:[/] [red]{cve_ids}[/]")
   132→        lines.append("")
   133→        if description:
   134→            lines.append(f"[bold]Description:[/]")
   135→            lines.append(f"[dim]{description}[/]")
   136→            lines.append("")
   137→        if evidence:
   138→            lines.append(f"[bold]Evidence:[/]")
   139→            lines.append(f"[dim]{evidence}[/]")
   140→            lines.append("")
   141→        lines.append(f"[bold]{GREEN}Remediation:[/]")
   142→        lines.append(f"{GREEN}{remediation}[/]")
   143→
   144→        content = "\n".join(lines)
   145→
   146→        with Vertical(classes="modal-container"):
   147→            yield Label(content, classes="modal-content")
   148→            with Horizontal(classes="modal-actions"):
   149→                yield Button("Close", variant="primary", id="close-modal")
   150→
   151→    def on_button_pressed(self, event: Button.Pressed) -> None:
   152→        if event.button.id == "close-modal":
   153→            self.dismiss()

   156→# ════════════════════════════════════════════════════════════════════════════════
   157→# Boot Screen Widget
   158→# ════════════════════════════════════════════════════════════════════════════════
   159→
   160→
   161→class BootScreen(Widget):
   162→    """Full-screen animated boot sequence."""
   163→
   164→    def __init__(self) -> None:
   165→        super().__init__()
   166→        self._step = 0
   167→        self._spinner_idx = 0
   168→        self._module_idx = 0
   169→        self._module_names = list(MODULE_NAMES.values())
   170→        self._done = False
   171→        self._timer: Optional[Timer] = None
   172→
   173→    def compose(self) -> ComposeResult:
   174→        with Vertical(classes="boot-container"):
   175→            yield Static("", id="boot-line-1", classes="boot-title")
   176→            yield Static("", id="boot-line-2", classes="boot-line")
   177→            yield Static("", id="boot-line-3", classes="boot-line")
   178→            yield Static("", id="boot-line-4", classes="boot-line")
   179→
   180→    def on_mount(self) -> None:
   181→        self._timer = self.set_interval(0.04, self._tick)
   182→
   183→    def _tick(self) -> None:
   184→        app = self.app
   185→        if not isinstance(app, NexusApp):
   186→            return
   187→
   188→        line1 = app.query_one("#boot-line-1", Static)
   189→        line2 = app.query_one("#boot-line-2", Static)
   190→        line3 = app.query_one("#boot-line-3", Static)
   191→        line4 = app.query_one("#boot-line-4", Static)
   192→
   193→        if self._step == 0:
   194→            # Letter-by-letter title
   195→            title = "RECONPRO NEXUS"
   196→            chars = title[: self._spinner_idx]
   197→            self._spinner_idx += 1
   198→            line1.update(f"[{CYAN} bold]{chars}[/]")
   199→            if self._spinner_idx > len(title):
   200→                self._spinner_idx = 0
   201→                self._step = 1
   202→                line1.update(f"[{CYAN} bold]{title}[/]  [{DIM_CYAN}]v{VERSION}[/]")
   203→
   204→        elif self._step == 1:
   205→            # Spinner for "Initializing agent..."
   206→            frame = SPINNER_FRAMES[self._spinner_idx % len(SPINNER_FRAMES)]
   207→            self._spinner_idx += 1
   208→            line2.update(f"  [{CYAN}]{frame}[/] [{DIM_CYAN}]Initializing agent...[/]")
   209→            if self._spinner_idx > 30:
   210→                self._spinner_idx = 0
   211→                self._step = 2
   212→                line2.update(f"  [{GREEN}]✓[/] [{DIM_CYAN}]Agent initialized.[/]")
   213→
   214→        elif self._step == 2:
   215→            # Module names appearing one by one
   216→            if self._module_idx < len(self._module_names):
   217→                visible = self._module_names[: self._module_idx + 1]
   218→                self._module_idx += 1
   219→                modules_str = "  ".join(f"[{CYAN}]○[/] [{DIM_CYAN}]{m}[/]" for m in visible)
   220→                remaining = len(self._module_names) - self._module_idx
   221→                if remaining > 0:
   222→                    modules_str += f"  [{DIM_CYAN}]+{remaining} more...[/]"
   223→                frame = SPINNER_FRAMES[self._spinner_idx % len(SPINNER_FRAMES)]
   224→                self._spinner_idx += 1
   225→                line3.update(
   226→                    f"  [{CYAN}]{frame}[/] [{DIM_CYAN}]Loading {len(self._module_names)} modules...[/]\n  {modules_str}"
   227→                )
   228→            else:
   229→                all_mods = "  ".join(
   230→                    f"[{GREEN}]●[/] [{DIM_CYAN}]{m}[/]" for m in self._module_names
   231→                )
   232→                line3.update(
   233→                    f"  [{GREEN}]✓[/] [{DIM_CYAN}]All {len(self._module_names)} modules loaded.[/]\n  {all_mods}"
   234→                )
   235→                self._step = 3
   236→                self._spinner_idx = 0
   237→
   238→        elif self._step == 3:
   239→            # "Ready." with flash
   240→            if self._spinner_idx == 0:
   241→                line4.update(f"[{YELLOW} bold]  ▶  READY.[/]")
   242→            elif self._spinner_idx == 6:
   243→                line4.update(f"[{GREEN} bold]  ▶  READY.[/]")
   244→            elif self._spinner_idx >= 10:
   245→                self._done = True
   246→                if self._timer:
   247→                    self._timer.stop()
   248→                self.app.call_from_thread(self.app._finish_boot)
   249→            self._spinner_idx += 1
   250→

   252→# ════════════════════════════════════════════════════════════════════════════════
   253→# Module Status Cell Widget
   254→# ════════════════════════════════════════════════════════════════════════════════
   255→
   256→
   257→class ModuleCell(Static):
   258→    """Single module status cell in the grid."""
   259→
   260→    def __init__(self, module_id: str, name: str) -> None:
   261→        super().__init__()
   262→        self.module_id = module_id
   263→        self.module_name = name
   264→        self.status = "idle"  # idle | scanning | done | error
   265→        self.finding_count = 0
   266→        self._spinner_idx = 0
   267→        self._timer: Optional[Timer] = None
   268→
   269→    def on_mount(self) -> None:
   270→        self._update_display()
   271→        self._timer = self.set_interval(0.1, self._spin)
   272→
   273→    def _spin(self) -> None:
   274→        if self.status == "scanning":
   275→            self._spinner_idx = (self._spinner_idx + 1) % len(SPINNER_FRAMES)
   276→            self._update_display()
   277→
   278→    def _update_display(self) -> None:
   279→        name_short = self.module_name[:14]
   280→        if self.status == "idle":
   281→            icon = f"[{DIM_CYAN}]○[/]"
   282→            status_text = f"[{DIM_CYAN}]idle[/]"
   283→        elif self.status == "scanning":
   284→            frame = SPINNER_FRAMES[self._spinner_idx]
   285→            icon = f"[{CYAN}]{frame}[/]"
   286→            status_text = f"[{CYAN}]scanning[/]"
   287→        elif self.status == "done":
   288→            icon = f"[{GREEN}]✓[/]"
   289→            status_text = f"[{GREEN}]done ({self.finding_count})[/]"
   290→        elif self.status == "error":
   291→            icon = f"[{RED}]✗[/]"
   292→            status_text = f"[{RED}]error[/]"
   293→        else:
   294→            icon = "○"
   295→            status_text = "idle"
   296→
   297→        self.update(
   298→            f"{icon} [{CYAN} bold]{name_short}[/]\n  {status_text}"
   299→        )
   300→
   301→    def set_status(self, status: str, count: int = 0) -> None:
   302→        self.status = status
   303→        self.finding_count = count
   304→        self._update_display()
   305→
   306→    def on_unmount(self) -> None:
   307→        if self._timer:
   308→            self._timer.stop()
   309→

   312→# Pulsing Status Dot Widget
   313→# ════════════════════════════════════════════════════════════════════════════════
   314→
   315→
   316→class StatusDot(Static):
   317→    """Pulsing dot indicator for scan status."""
   318→
   319→    scanning: reactive[bool] = reactive(False)
   320→
   321→    def __init__(self) -> None:
   322→        super().__init__("")
   323→        self._phase = 0
   324→
   325→    def on_mount(self) -> None:
   326→        self.set_interval(0.3, self._pulse)
   327→
   328→    def watch_scanning(self, old: bool, new: bool) -> None:
   329→        # Update visual state based on scanning status
   330→        if new:
   331→            self.update(f"[{self._get_pulse_color()}]●[/]")
   332→        else:
   333→            self.update(f"[{GREEN}]●[/]")
   334→
   335→    def _get_pulse_color(self) -> str:
   336→        return ["#003322", "#006644", CYAN, "#006644"][self._phase] if self.scanning else GREEN
   337→
   338→    def _pulse(self) -> None:
   339→        self._phase = (self._phase + 1) % 4
   340→        if self.scanning:
   341→            brightness = self._get_pulse_color()
   342→            self.update(f"[{brightness}]●[/]")
   343→        else:
   344→            self.update(f"[{GREEN}]●[/]")

   682→    def on_key(self, event: Key) -> None:
   683→        """Handle key events — boot skip + history navigation."""
   684→        # Skip boot on ANY key press during boot sequence
   685→        if not self._boot_done and not self._boot_skipped:
   686→            self._boot_skipped = True
   687→            try:
   688→                boot = self.query_one("#boot-screen", BootScreen)
   689→                if boot._timer:
   690→                    boot._timer.stop()
   691→            except NoMatches:
   692→                pass
   693→            self._finish_boot()
   694→            return
   695→
   696→        # Command history navigation
   697→        if event.key == "up" and self._history_index > 0:
   698→            try:
   699→                inp = self.query_one("#command-input", Input)
   700→                if inp.has_focus:
   701→                    self._history_index -= 1
   702→                    inp.value = self._command_history[self._history_index]
   703→            except NoMatches:
   704→                pass
   705→        elif event.key == "down":
   706→            try:
   707→                inp = self.query_one("#command-input", Input)
   708→                if inp.has_focus:
   709→                    if self._history_index < len(self._command_history) - 1:
   710→                        self._history_index += 1
   711→                        inp.value = self._command_history[self._history_index]
   712→                    else:
   713→                        self._history_index = len(self._command_history)
   714→                        inp.value = ""
   715→            except NoMatches:
   716→                pass

   902→    def _process_command(self, raw: str) -> None:
   903→        """Parse and execute a user command."""
   904→        self._chat(f"[{CYAN} bold]▸ {raw}[/]")
   905→
   906→        parts = raw.split()
   907→        cmd = parts[0].lower()
   908→        args = parts[1:]
   909→
   910→        if cmd in ("quit", "q", "exit"):
   911→            self.exit()
   912→            return
   913→
   914→        if cmd == "clear":
   915→            self._clear_all_feeds()
   916→            return
   917→
   918→        if cmd == "help":
   919→            self._show_help()
   920→            return
   921→
   922→        if cmd == "scan":
   923→            self._handle_scan(args)
   924→            return
   925→
   926→        if cmd == "audit":
   927→            self._handle_audit(args)
   928→            return
   929→
   930→        if cmd == "dev":
   931→            self._handle_dev(args)
   932→            return
   933→
   934→        if cmd == "doctor":
   935→            self._handle_doctor()
   936→            return
   937→
   938→        if cmd == "blitz":
   939→            self._handle_blitz(args)
   940→            return
   941→
   942→        if cmd == "subdomains":
   943→            self._handle_subdomains(args)
   944→            return
   945→
   946→        if cmd == "agent":
   947→            self._handle_agent(args)
   948→            return
   949→
   950→        if cmd == "swarm":
   951→            self._handle_swarm(args)
   952→            return
   953→
   954→        if cmd == "adversarial":
   955→            self._handle_adversarial(args)
   956→            return
   957→
   958→        if cmd == "graph":
   959→            self._handle_graph()
   960→            return
   961→
   962→        if cmd == "export":
   963→            self._handle_export(args)
   964→            return
   965→
   966→        if cmd == "history":
   967→            self._handle_history()
   968→            return
   969→
   970→        if cmd == "theme":
   971→            # Already handled in on_input_submitted before _process_command
   972→            return
   973→
   974→        self._chat(f"[{RED}]Unknown command: {cmd}[/]")
   975→        self._chat(f"[{DIM_CYAN}]Type [cyan]help[/] for available commands.[/]")

   977→    def _handle_theme(self, args: List[str]) -> None:
   978→        """Handle theme switching command."""
   979→        if not args or args[0].lower() in ("list", "ls"):
   980→            available = Theme.available_themes()
   981→            current = Theme.current_name()
   982→            self._chat(f"[{CYAN} bold]  THEMES[/]")
   983→            for name in available:
   984→                marker = f" [{GREEN}]●[/]" if name == current else "  [{DIM_CYAN}]○[/]"
   985→                display_name = THEMES[name].get("name", name)
   986→                self._chat(f"  {marker} [{CYAN}]{name}[/] [{MUTED}]- {display_name}[/]")

   988→
   989→        name = args[0].lower()
   990→        try:
   991→            Theme.set_theme(name)
   992→            self._chat(f"[{GREEN}]  ✓ Theme changed to [bold]{name}[/]. Restart nexus to apply fully.[/]")

  1042→    def _handle_scan(self, args: List[str]) -> None:
  1043→        if not args:
  1044→            self._chat(f"[{RED}]Usage: scan <target> [with <mod1> <mod2> ...][/]")
  1045→            return
  1046→
  1047→        # Parse "with" clause
  1048→        modules: Optional[List[str]] = None
  1049→        target_parts: List[str] = []
  1050→        found_with = False
  1051→        mod_parts: List[str] = []
  1052→        for a in args:
  1053→            if a.lower() == "with":
  1054→                found_with = True
  1055→                continue
  1056→            if found_with:
  1057→                mod_parts.append(a)
  1058→            else:
  1059→                target_parts.append(a)
  1060→
  1061→        target = " ".join(target_parts)
  1062→        if not target:
  1063→            self._chat(f"[{RED}]No target specified.[/]")
  1064→            return
  1065→
  1066→        if mod_parts:
  1067→            modules = [m.strip().lower() for m in mod_parts]
  1068→            self._chat(f"[{DIM_CYAN}]Scanning [cyan bold]{target}[/] with modules: {', '.join(modules)}[/]")
  1069→        else:
  1070→            self._chat(f"[{DIM_CYAN}]Scanning [cyan bold]{target}[/]...[/]")
  1071→
  1072→        self.current_target = target
  1073→        self._run_scan_worker(target, modules=modules)

  1155→    def _run_scan_worker(
  1156→        self,
  1157→        target: str,
  1158→        modules: Optional[List[str]] = None,
  1159→        is_local: bool = False,
  1160→    ) -> None:
  1161→        """Run a scan in a background thread."""
  1162→        self.call_from_thread(self.is_scanning.set, True)
  1163→
  1164→        # Set modules to scanning
  1165→        if modules:
  1166→            for m in modules:
  1167→                if m in self._module_cells:
  1168→                    self.call_from_thread(self._module_cells[m].set_status, "scanning")
  1169→        else:
  1170→            all_mods = (
  1171→                list(MODULE_NAMES.keys())
  1172→                if is_local
  1173→                else [k for k in MODULE_NAMES if k not in ("host", "dev", "doctor")]
  1174→            )
  1175→            for m in all_mods:
  1176→                if m in self._module_cells:
  1177→                    self.call_from_thread(self._module_cells[m].set_status, "scanning")
  1178→
  1179→        try:
  1180→            from .scanner import scan, audit_scan
  1181→            from .history import save_scan
  1182→
  1183→            self.call_from_thread(
  1184→                self._chat, f"[{DIM_CYAN}]  ⏳ Scan in progress...[/]"
  1185→            )
  1186→
  1187→            if is_local:
  1188→                result = audit_scan(target=target, modules=modules)
  1189→            else:
  1190→                result = scan(target=target, modules=modules)
  1191→
  1192→            # Process findings
  1193→            data = result.to_dict()
  1194→            self.call_from_thread(self._on_scan_complete, data, modules, is_local)
  1195→
  1196→            # Save to history
  1197→            try:
  1198→                save_scan(data, label="nexus")
  1199→            except Exception:
  1200→                pass
  1201→
  1202→        except Exception as e:
  1203→            self.call_from_thread(self._chat, f"[{RED}]  ✗ Scan error: {e}[/]")
  1204→            if modules:
  1205→                for m in modules:
  1206→                    if m in self._module_cells:
  1207→                        self.call_from_thread(self._module_cells[m].set_status, "error")
  1208→            else:
  1209→                self.call_from_thread(self._set_all_modules_status, "error")
  1210→
  1211→        finally:
  1212→            self.call_from_thread(self.is_scanning.set, False)

  1565→    def on_click(self, event: Click) -> None:
  1566→        """Open finding detail modal for the clicked finding.
  1567→
  1568→        Uses the vertical click position to determine which finding index
  1569→        the user clicked on, mapping it to _findings_list.
  1570→        """
  1571→        try:
  1572→            if (
  1573→                isinstance(event.widget, RichLog)
  1574→                and event.widget.id == "findings-feed"
  1575→                and self._findings_list
  1576→            ):
  1577→                # Get the line offset to determine which finding was clicked.
  1578→                # Each finding entry is 1 line in the RichLog.
  1579→                # We use the y coordinate relative to the widget to estimate the index.
  1580→                region = event.widget.region
  1581→                if region is not None:
  1582→                    # Calculate which line was clicked relative to the scroll position
  1583→                    scroll_offset = event.widget.scroll_y
  1584→                    click_y = event.y - region.y
  1585→                    # Each finding takes 1 line; estimate index from click position
  1586→                    # Account for scroll position
  1587→                    line_height = 1
  1588→                    if hasattr(event.widget, "_line_height"):
  1589→                        line_height = event.widget._line_height
  1590→                    clicked_index = int((click_y + scroll_offset * line_height) / line_height)
  1591→                    # Clamp to valid range
  1592→                    clicked_index = max(0, min(clicked_index, len(self._findings_list) - 1))
  1593→                    finding = self._findings_list[clicked_index]
  1594→                else:
  1595→                    finding = self._findings_list[-1]
  1596→                self.push_screen(FindingDetailModal(finding))
  1597→        except (NoMatches, Exception):
  1598→            pass
