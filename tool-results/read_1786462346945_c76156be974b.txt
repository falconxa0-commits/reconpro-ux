    50→def _ensure_dir(path: Path) -> Path:
    51→    """Create directory if it does not exist and return it."""

    56→def _sparkline(values: list) -> str:
    57→    """Create a tiny ASCII sparkline from numeric *values*."""
    58→    if not values:
    59→        return ""
    60→    blocks = ["\u2581", "\u2582", "\u2583", "\u2584", "\u2585", "\u2586", "\u2587", "\u2588"]
    61→    mn, mx = min(values), max(values)
    62→    rng = mx - mn if mx != mn else 1
    63→    return "".join(
    64→        blocks[min(len(blocks) - 1, int((v - mn) / rng * (len(blocks) - 1)))]
    65→        for v in values
    66→    )

    69→def _safe_serialize(obj: Any) -> Any:
    70→    """Make an object JSON-serializable."""
    71→    if isinstance(obj, (str, int, float, bool, type(None))):
    72→        return obj
    73→    if isinstance(obj, datetime):
    74→        return obj.isoformat()
    75→    if isinstance(obj, Path):
    76→        return str(obj)
    77→    if isinstance(obj, (list, tuple)):
    78→        return [_safe_serialize(v) for v in obj]
    79→    if isinstance(obj, dict):
    80→        return {str(k): _safe_serialize(v) for k, v in obj.items()}
    81→    return str(obj)

    86→
    87→@dataclass
    88→class BenchmarkResult:
    89→    """Single benchmark measurement with comparison helpers.
    90→
    91→    Attributes:
    92→        name:       Human-readable benchmark name (e.g. "dict_create_1000")
    93→        category:   Suite category (memory, cpu, io, scanner, module, startup)
    94→        value:      Measured value (time in seconds, memory in bytes, etc.)
    95→        unit:       Unit of measurement ("s", "ms", "bytes", "KB", "ops/s")
    96→        timestamp:  When the benchmark was taken
    97→        iterations: Number of iterations run for this measurement
    98→        context:    Arbitrary extra metadata dict (platform, python version, etc.)
    99→        extra:      Additional numeric measurements (min, max, median, etc.)
   100→    """
   101→
   102→    name: str
   103→    category: str
   104→    value: float
   105→    unit: str = "ms"
   106→    timestamp: datetime = field(default_factory=lambda: datetime.now(timezone.utc))
   107→    iterations: int = 1
   108→    context: Dict[str, Any] = field(default_factory=dict)
   109→    extra: Dict[str, float] = field(default_factory=dict)
   110→
   111→    def to_dict(self) -> Dict[str, Any]:
   112→        """Serialize to a JSON-safe dictionary."""
   113→        return {
   114→            "name": self.name,
   115→            "category": self.category,
   116→            "value": self.value,
   117→            "unit": self.unit,
   118→            "timestamp": self.timestamp.isoformat(),
   119→            "iterations": self.iterations,
   120→            "context": _safe_serialize(self.context),
   121→            "extra": dict(self.extra),
   122→        }
   123→
   124→    @classmethod
   125→    def from_dict(cls, data: Dict[str, Any]) -> "BenchmarkResult":
   126→        """Deserialize from a dictionary."""
   127→        ts = data.get("timestamp")
   128→        if isinstance(ts, str):
   129→            ts = datetime.fromisoformat(ts)
   130→        elif not isinstance(ts, datetime):
   131→            ts = datetime.now(timezone.utc)
   132→        return cls(
   133→            name=data["name"],
   134→            category=data.get("category", "unknown"),
   135→            value=float(data["value"]),
   136→            unit=data.get("unit", "ms"),
   137→            timestamp=ts,
   138→            iterations=data.get("iterations", 1),
   139→            context=data.get("context", {}),
   140→            extra=data.get("extra", {}),
   141→        )
   142→
   143→    def vs_baseline(self, baseline_value: float) -> Dict[str, Any]:
   144→        """Compare this result against a baseline value.
   145→
   146→        Returns a dict with:
   147→            baseline_value, current_value, delta, delta_pct, direction.
   148→        Direction is "improvement", "regression", or "neutral".
   149→        For timing benchmarks, lower is better (regression = increase).
   150→        For throughput, higher is better (regression = decrease).
   151→        """
   152→        if baseline_value == 0:
   153→            delta_pct = 0.0 if self.value == 0 else float("inf")
   154→        else:
   155→            delta_pct = ((self.value - baseline_value) / baseline_value) * 100
   156→
   157→        delta = self.value - baseline_value
   158→
   159→        # Determine direction based on unit conventions.
   160→        # Timing units (s, ms, us, ns): lower is better.
   161→        # Throughput units (ops/s, req/s): higher is better.
   162→        # Memory units (bytes, KB, MB): lower is better.
   163→        throughput_units = {"ops/s", "req/s", "items/s"}
   164→        lower_is_better = self.unit not in throughput_units
   165→
   166→        if lower_is_better:
   167→            if delta > 0:
   168→                direction = "regression"
   169→            elif delta < 0:
   170→                direction = "improvement"
   171→            else:
   172→                direction = "neutral"
   173→        else:
   174→            if delta > 0:
   175→                direction = "improvement"
   176→            elif delta < 0:
   177→                direction = "regression"
   178→            else:
   179→                direction = "neutral"
   180→
   181→        return {
   182→            "name": self.name,
   183→            "category": self.category,
   184→            "baseline_value": baseline_value,
   185→            "current_value": self.value,
   186→            "delta": round(delta, 6),
   187→            "delta_pct": round(delta_pct, 2) if abs(delta_pct) != float("inf") else delta_pct,
   188→            "direction": direction,
   189→            "unit": self.unit,
   190→        }
   191→
   192→    def vs_previous(self, previous: "BenchmarkResult") -> Optional[Dict[str, Any]]:
   193→        """Compare this result against a previous BenchmarkResult.
   194→
   195→        Returns None if the names or units do not match.
   196→        """
   197→        if self.name != previous.name or self.unit != previous.unit:
   198→            return None
   199→        return self.vs_baseline(previous.value)

   202→# ── BenchmarkSuite ──────────────────────────────────────────────────────

   204→
   205→@dataclass
   206→class BenchmarkCase:
   207→    """A single callable benchmark within a suite."""
   208→    name: str
   209→    fn: Callable[..., float]
   210→    category: str = "generic"
   211→    unit: str = "ms"
   212→    iterations: int = 5
   213→    warmup: int = 1
   214→
   215→
   216→class BenchmarkSuite:
   217→    """A named collection of related benchmarks.
   218→
   219→    Parameters:
   220→        name:        Suite identifier (e.g. "memory", "cpu")
   221→        description: Human-readable description of what the suite measures
   222→    """
   223→
   224→    def __init__(self, name: str, description: str = "") -> None:
   225→        self.name: str = name
   226→        self.description: str = description
   227→        self._cases: List[BenchmarkCase] = []
   228→
   229→    def add(
   230→        self,
   231→        name: str,
   232→        fn: Callable[..., float],
   233→        category: str = "",
   234→        unit: str = "ms",
   235→        iterations: int = 5,
   236→        warmup: int = 1,
   237→    ) -> "BenchmarkSuite":
   238→        """Add a benchmark case. Returns self for chaining."""
   239→        self._cases.append(BenchmarkCase(
   240→            name=name,
   241→            fn=fn,
   242→            category=category or self.name,
   243→            unit=unit,
   244→            iterations=iterations,
   245→            warmup=warmup,
   246→        ))
   247→        return self
   248→
   249→    def run(self) -> List[BenchmarkResult]:
   250→        """Execute all benchmark cases and return results.
   251→
   252→        Each case runs ``warmup`` iterations (discarded) followed by
   253→        ``iterations`` measured iterations. Returns the median value.
   254→        Memory is tracked via tracemalloc for the measured iterations.
   255→        """
   256→        results: List[BenchmarkResult] = []
   257→        for case in self._cases:
   258→            result = self._run_case(case)
   259→            if result is not None:
   260→                results.append(result)
   261→        return results
   262→
   263→    def _run_case(self, case: BenchmarkCase) -> Optional[BenchmarkResult]:
   264→        """Run a single benchmark case with warmup and memory tracking."""
   265→        try:
   266→            # Warmup
   267→            for _ in range(case.warmup):
   268→                try:
   269→                    case.fn()
   270→                except Exception:
   271→                    pass  # warmup errors are silently ignored
   272→
   273→            # Measured iterations
   274→            times: List[float] = []
   275→            peak_memory = 0
   276→
   277→            tracemalloc.start()
   278→            for _ in range(case.iterations):
   279→                t0 = time.perf_counter()
   280→                case.fn()
   281→                t1 = time.perf_counter()
   282→                times.append(t1 - t0)
   283→                _, current_peak = tracemalloc.get_traced_memory()
   284→                peak_memory = max(peak_memory, current_peak)
   285→            tracemalloc.stop()
   286→
   287→            times.sort()
   288→            median = times[len(times) // 2]
   289→            mean = sum(times) / len(times)
   290→
   291→            # Convert to appropriate unit
   292→            value, unit = self._convert_unit(median, case.unit)
   293→            extra = {
   294→                "min_s": round(min(times), 9),
   295→                "max_s": round(max(times), 9),
   296→                "mean_s": round(mean, 9),
   297→                "median_s": round(median, 9),
   298→                "peak_memory_bytes": peak_memory,
   299→            }
   300→
   301→            return BenchmarkResult(
   302→                name=case.name,
   303→                category=case.category,
   304→                value=value,
   305→                unit=unit,
   306→                iterations=case.iterations,
   307→                context={
   308→                    "suite": self.name,
   309→                    "python": sys.version.split()[0],
   310→                    "platform": sys.platform,
   311→                },
   312→                extra=extra,
   313→            )
   314→        except Exception as exc:
   315→            logger.warning("Benchmark case '%s' failed: %s", case.name, exc)
   316→            return None
   317→
   318→    @staticmethod
   319→    def _convert_unit(seconds: float, preferred_unit: str) -> Tuple[float, str]:
   320→        """Convert seconds to the preferred unit."""
   321→        if preferred_unit == "ns":
   322→            return round(seconds * 1e9, 2), "ns"
   323→        elif preferred_unit == "us":
   324→            return round(seconds * 1e6, 2), "us"
   325→        elif preferred_unit == "ms":
   326→            return round(seconds * 1000, 4), "ms"
   327→        elif preferred_unit == "s":
   328→            return round(seconds, 6), "s"
   329→        elif preferred_unit == "bytes":
   330→            return round(seconds, 2), "bytes"  # pass-through for memory
   331→        elif preferred_unit == "KB":
   332→            return round(seconds / 1024, 2), "KB"
   333→        else:
   334→            return round(seconds * 1000, 4), "ms"
   335→
   336→    def __len__(self) -> int:
   337→        return len(self._cases)
   338→
   339→    def __repr__(self) -> str:
   340→        return f"BenchmarkSuite(name={self.name!r}, cases={len(self._cases)})"

   343→# ── BenchmarkBaseline ───────────────────────────────────────────────────

   345→
   346→class BenchmarkBaseline:
   347→    """Historical baseline storage and trend tracking.
   348→
   349→    Stores baselines at ``RECONPRO_HOME / "memory" / "benchmark_baselines.json"``.
   350→    Each benchmark name has a list of historical values with timestamps.
   351→
   352→    Features:
   353→    - Create/update baselines from BenchmarkResult lists
   354→    - Compare results against stored baselines
   355→    - Track trends over time
   356→    - Prune old entries to keep storage bounded
   357→    """
   358→
   359→    MAX_HISTORY_PER_BENCHMARK: int = 100
   360→    DEFAULT_BASELINE_FILE: str = "benchmark_baselines.json"
   361→
   362→    def __init__(self, storage_dir: Optional[Path] = None) -> None:
   363→        if storage_dir is None:
   364→            storage_dir = MEMORY_DIR
   365→        self._dir = _ensure_dir(storage_dir)
   366→        self._file: Path = self._dir / self.DEFAULT_BASELINE_FILE
   367→        self._data: Dict[str, List[Dict[str, Any]]] = {}
   368→        self._lock = threading.Lock()
   369→        self._load()
   370→
   371→    def _load(self) -> None:
   372→        """Load baselines from disk."""
   373→        if self._file.exists():
   374→            try:
   375→                with open(self._file, "r", encoding="utf-8") as fh:
   376→                    self._data = json.load(fh)
   377→                # Validate structure
   378→                if not isinstance(self._data, dict):
   379→                    self._data = {}
   380→            except (json.JSONDecodeError, OSError) as exc:
   381→                logger.warning("Failed to load baselines from %s: %s", self._file, exc)
   382→                self._data = {}
   383→
   384→    def _save(self) -> None:
   385→        """Persist baselines to disk."""
   386→        try:
   387→            _ensure_dir(self._dir)
   388→            with open(self._file, "w", encoding="utf-8") as fh:
   389→                json.dump(self._data, fh, indent=2, default=str)
   390→        except OSError as exc:
   391→            logger.error("Failed to save baselines to %s: %s", self._file, exc)
   392→
   393→    def record(self, result: BenchmarkResult) -> None:
   394→        """Record a single benchmark result as a data point.
   395→
   396→        If the benchmark name already has history, the new value is appended.
   397→        Old entries beyond MAX_HISTORY_PER_BENCHMARK are pruned.
   398→        """
   399→        with self._lock:
   400→            entry: Dict[str, Any] = {
   401→                "value": result.value,
   402→                "unit": result.unit,
   403→                "timestamp": result.timestamp.isoformat(),
   404→                "epoch": result.timestamp.timestamp(),
   405→                "iterations": result.iterations,
   406→                "extra": result.extra,
   407→            }
   408→            key = f"{result.category}:{result.name}"
   409→            history = self._data.setdefault(key, [])
   410→            history.append(entry)
   411→            # Prune old entries
   412→            if len(history) > self.MAX_HISTORY_PER_BENCHMARK:
   413→                self._data[key] = history[-self.MAX_HISTORY_PER_BENCHMARK:]
   414→            self._save()
   415→
   416→    def record_batch(self, results: List[BenchmarkResult]) -> None:
   417→        """Record multiple benchmark results at once."""
   418→        for r in results:
   419→            self.record(r)
   420→
   421→    def get_baseline(self, name: str, category: str = "") -> Optional[float]:
   422→        """Get the most recent baseline value for a benchmark.
   423→
   424→        Returns the value of the most recent entry, or None if not found.
   425→        """
   426→        key = f"{category}:{name}" if category else name
   427→        with self._lock:
   428→            history = self._data.get(key, [])
   429→            if not history:
   430→                # Try searching by name alone
   431→                for k, v in self._data.items():
   432→                    if k.endswith(f":{name}") or k == name:
   433→                        return v[-1]["value"] if v else None
   434→                return None
   435→            return history[-1]["value"]
   436→
   437→    def get_history(
   438→        self, name: str, category: str = "", days: int = 90
   439→    ) -> List[Dict[str, Any]]:
   440→        """Get historical values for a benchmark within the last N days."""
   441→        key = f"{category}:{name}" if category else name
   442→        cutoff = time.time() - days * 86400
   443→        with self._lock:
   444→            history = self._data.get(key, [])
   445→            return [h for h in history if h.get("epoch", 0) >= cutoff]
   446→
   447→    def get_trend(
   448→        self, name: str, category: str = "", days: int = 30
   449→    ) -> List[Dict[str, Any]]:
   450→        """Get daily-average trend for a benchmark.
   451→
   452→        Each entry has ``date`` (YYYY-MM-DD) and ``avg_value``.
   453→        """
   454→        history = self.get_history(name, category, days)
   455→        if not history:
   456→            return []
   457→        daily: Dict[str, List[float]] = {}
   458→        for h in history:
   459→            ts = h.get("timestamp", "")
   460→            date_str = ts[:10] if len(ts) >= 10 else ts
   461→            daily.setdefault(date_str, []).append(h["value"])
   462→        trend = []
   463→        for date_str in sorted(daily):
   464→            values = daily[date_str]
   465→            avg = sum(values) / len(values)
   466→            trend.append({
   467→                "date": date_str,
   468→                "avg_value": round(avg, 4),
   469→                "samples": len(values),
   470→            })
   471→        return trend
   472→
   473→    def get_all_baseline_values(self) -> Dict[str, float]:
   474→        """Get the latest baseline value for every recorded benchmark.
   475→
   476→        Returns a dict mapping ``category:name`` -> latest value.
   477→        """
   478→        with self._lock:
   479→            result = {}
   480→            for key, history in self._data.items():
   481→                if history:
   482→                    result[key] = history[-1]["value"]
   483→            return result
   484→
   485→    def compare(self, results: List[BenchmarkResult]) -> List[Dict[str, Any]]:
   486→        """Compare a list of results against stored baselines.
   487→
   488→        Returns a list of comparison dicts (one per result that has a baseline).
   489→        """
   490→        comparisons = []
   491→        for r in results:
   492→            baseline_value = self.get_baseline(r.name, r.category)
   493→            if baseline_value is not None:
   494→                comparisons.append(r.vs_baseline(baseline_value))
   495→        return comparisons
   496→
   497→    def prune(self, days: int = 365) -> int:
   498→        """Remove entries older than *days*.
   499→
   500→        Returns the number of entries removed.
   501→        """
   502→        cutoff = time.time() - days * 86400
   503→        removed = 0
   504→        with self._lock:
   505→            for key in list(self._data.keys()):
   506→                original_len = len(self._data[key])
   507→                self._data[key] = [
   508→                    h for h in self._data[key]
   509→                    if h.get("epoch", 0) >= cutoff
   510→                ]
   511→                removed += original_len - len(self._data[key])
   512→                # Remove empty keys
   513→                if not self._data[key]:
   514→                    del self._data[key]
   515→            if removed > 0:
   516→                self._save()
   517→        return removed
   518→
   519→    def set_baseline(
   520→        self, name: str, category: str, value: float, unit: str = "ms"
   521→    ) -> None:
   522→        """Manually set a baseline value for a benchmark."""
   523→        result = BenchmarkResult(
   524→            name=name, category=category, value=value, unit=unit
   525→        )
   526→        self.record(result)
   527→
   528→    def clear(self) -> None:
   529→        """Remove all baseline data."""
   530→        with self._lock:
   531→            self._data = {}
   532→            self._save()
   533→

   537→
   538→class BenchmarkAutomation:
   539→    """Automated benchmark runner for continuous performance tracking.
   540→
   541→    Orchestrates benchmark suites, manages baselines, detects regressions,
   542→    and generates reports. Stores results at ``MEMORY_DIR / "benchmarks"``.
   543→
   544→    Parameters:
   545→        results_dir:  Override directory for benchmark result files
   546→        baseline_dir: Override directory for baseline storage
   547→    """
   548→
   549→    # Regression threshold: percentage change considered a regression
   550→    REGRESSION_THRESHOLD: float = 20.0
   551→    # Default quick-benchmark iteration counts (smaller for speed)
   552→    QUICK_ITERATIONS: int = 3
   553→    # Default full-benchmark iteration counts
   554→    FULL_ITERATIONS: int = 5
   555→
   556→    def __init__(
   557→        self,
   558→        results_dir: Optional[Path] = None,
   559→        baseline_dir: Optional[Path] = None,
   560→    ) -> None:
   561→        self._results_dir = _ensure_dir(results_dir or (MEMORY_DIR / "benchmarks"))
   562→        self._baseline = BenchmarkBaseline(baseline_dir)
   563→        self._suites: Dict[str, BenchmarkSuite] = {}
   564→        self._scheduler_timer: Optional[threading.Timer] = None
   565→        self._scheduler_stop = threading.Event()
   566→        self._register_default_suites()
   567→
   568→    # ── Suite Registration ────────────────────────────────────────────
   569→
   570→    def _register_default_suites(self) -> None:
   571→        """Register the six predefined benchmark suites."""
   572→        self._suites["memory"] = self._build_memory_suite()
   573→        self._suites["cpu"] = self._build_cpu_suite()
   574→        self._suites["io"] = self._build_io_suite()
   575→        self._suites["scanner"] = self._build_scanner_suite()
   576→        self._suites["module"] = self._build_module_suite()
   577→        self._suites["startup"] = self._build_startup_suite()
   578→
   579→    @staticmethod
   580→    def _build_memory_suite() -> BenchmarkSuite:
   581→        """Memory allocation and usage patterns."""
   582→        suite = BenchmarkSuite("memory", "Memory allocation and usage patterns")
   583→
   584→        # Dict allocation
   585→        for n in (100, 1000, 10000):
   586→            def _bench_dict_alloc(_n=n):
   587→                d = {str(i): i for i in range(_n)}
   588→                return sys.getsizeof(d)
   589→            suite.add(
   590→                f"dict_alloc_{n}",
   591→                _bench_dict_alloc,
   592→                unit="bytes",
   593→                iterations=5,
   594→            )
   595→
   596→        # List allocation

   940→    def run_full_benchmark(self) -> Dict[str, Any]:
   941→        """Execute all registered benchmark suites.
   942→
   943→        Returns a comprehensive results dict with per-suite results,
   944→        comparisons against baselines, and regression detection.
   945→        """
   946→        logger.info("Starting full benchmark run")
   947→        all_results: List[BenchmarkResult] = []
   948→        suite_summaries: List[Dict[str, Any]] = []
   949→        errors: List[str] = []
   950→        start_time = time.perf_counter()
   951→
   952→        for suite_name, suite in self._suites.items():
   953→            logger.debug("Running suite: %s", suite_name)
   954→            try:
   955→                suite_results = suite.run()
   956→                all_results.extend(suite_results)
   957→                suite_summaries.append({
   958→                    "suite": suite_name,
   959→                    "description": suite.description,
   960→                    "benchmarks_run": len(suite_results),
   961→                    "results": [r.to_dict() for r in suite_results],
   962→                })
   963→            except Exception as exc:
   964→                msg = f"Suite '{suite_name}' failed: {exc}"
   965→                logger.error(msg)
   966→                errors.append(msg)
   967→
   968→        elapsed = time.perf_counter() - start_time
   969→
   970→        # Record results as baselines
   971→        self._baseline.record_batch(all_results)
   972→
   973→        # Compare against previous baselines
   974→        comparisons = self._baseline.compare(all_results)
   975→        regressions = [c for c in comparisons if c.get("direction") == "regression"]
   976→
   977→        # Save results file
   978→        results_file = self._save_results(all_results, "full")
   979→
   980→        summary = {
   981→            "type": "full",
   982→            "timestamp": datetime.now(timezone.utc).isoformat(),
   983→            "elapsed_seconds": round(elapsed, 3),
   984→            "total_benchmarks": len(all_results),
   985→            "regressions_detected": len(regressions),
   986→            "results_file": str(results_file),
   987→            "suites": suite_summaries,
   988→            "comparisons": comparisons,
   989→            "regressions": regressions,
   990→            "errors": errors,
   991→            "platform": sys.platform,
   992→            "python_version": sys.version.split()[0],
   993→        }
   994→
   995→        logger.info(
   996→            "Full benchmark complete: %d benchmarks, %d regressions, %.2fs",
   997→            len(all_results), len(regressions), elapsed,
   998→        )
   999→        return summary

  1113→            return self.run_full_benchmark()

  1121→
  1122→        suite = self._suites[component_lower]

  1156→        """
  1157→        return self._baseline.compare(results)

  1159→    def detect_regression(
  1160→        self,
  1161→        results: List[BenchmarkResult],
  1162→        threshold: Optional[float] = None,
  1163→    ) -> List[Dict[str, Any]]:
  1164→        """Detect performance regressions in results.
  1165→
  1166→        A regression is detected when a timing/memory benchmark increases
  1167→        by more than *threshold* percent compared to its baseline, or when
  1168→        a throughput benchmark decreases by more than *threshold* percent.
  1169→
  1170→        Parameters:
  1171→            results:   Benchmark results to check
  1172→            threshold: Percentage change threshold (default: REGRESSION_THRESHOLD)
  1173→
  1174→        Returns:
  1175→            List of regression details, sorted by severity (worst first).
  1176→        """
  1177→        if threshold is None:
  1178→            threshold = self.REGRESSION_THRESHOLD
  1179→
  1180→        comparisons = self.compare_with_baseline(results)
  1181→        regressions = []
  1182→
  1183→        for comp in comparisons:
  1184→            pct = comp.get("delta_pct", 0)
  1185→            if pct == float("inf") or pct == float("-inf"):
  1186→                regressions.append({**comp, "threshold": threshold, "severity": "critical"})
  1187→            elif abs(pct) >= threshold and comp["direction"] == "regression":
  1188→                severity = "critical" if abs(pct) >= threshold * 2 else "high"
  1189→                regressions.append({**comp, "threshold": threshold, "severity": severity})
  1190→
  1191→        # Sort by delta_pct descending (worst regression first)
  1192→        regressions.sort(key=lambda r: abs(r.get("delta_pct", 0)), reverse=True)
  1193→        return regressions

  1313→    # ── Scheduling ────────────────────────────────────────────────────
  1314→
  1315→    def schedule_benchmarks(
  1316→        self,
  1317→        interval: float = 3600.0,
  1318→        benchmark_type: str = "quick",
  1319→        callback: Optional[Callable[[Dict[str, Any]], None]] = None,
  1320→    ) -> threading.Timer:
  1321→        """Schedule periodic benchmark runs.
  1322→
  1323→        Parameters:
  1324→            interval:       Seconds between benchmark runs (default: 1 hour)
  1325→            benchmark_type: "quick", "full", or a component name
  1326→            callback:       Optional callback invoked with results after each run
  1327→
  1328→        Returns:
  1329→            The Timer object (call .cancel() to stop).
  1330→        """
  1331→        self._scheduler_stop.clear()
  1332→        logger.info(
  1333→            "Scheduling %s benchmarks every %.0f seconds",
  1334→            benchmark_type, interval,
  1335→        )
  1336→
  1337→        def _run_and_reschedule() -> None:
  1338→            if self._scheduler_stop.is_set():
  1339→                return
  1340→            try:
  1341→                if benchmark_type == "quick":
  1342→                    results = self.run_quick_benchmark()
  1343→                elif benchmark_type == "full":
  1344→                    results = self.run_full_benchmark()
  1345→                else:
  1346→                    results = self.run_component_benchmark(benchmark_type)
  1347→                if callback is not None:
  1348→                    try:
  1349→                        callback(results)
  1350→                    except Exception as cb_err:
  1351→                        logger.error("Benchmark callback error: %s", cb_err)
  1352→            except Exception as exc:
  1353→                logger.error("Scheduled benchmark error: %s", exc)
  1354→                logger.debug("Traceback: %s", traceback.format_exc())
  1355→            # Reschedule if not stopped
  1356→            if not self._scheduler_stop.is_set():
  1357→                self._scheduler_timer = threading.Timer(interval, _run_and_reschedule)
  1358→                self._scheduler_timer.daemon = True
  1359→                self._scheduler_timer.start()
  1360→
  1361→        timer = threading.Timer(interval, _run_and_reschedule)
  1362→        timer.daemon = True
  1363→        self._scheduler_timer = timer
  1364→        timer.start()
  1365→        return timer
  1366→
  1367→    def stop_scheduled_benchmarks(self) -> None:
  1368→        """Stop any running scheduled benchmark timer."""
  1369→        self._scheduler_stop.set()
  1370→        if self._scheduler_timer is not None:
  1371→            self._scheduler_timer.cancel()
  1372→            self._scheduler_timer = None
  1373→        logger.info("Scheduled benchmarks stopped")
