   118→  id: string;

   127→// Derive threat cities and connections from API threat data
   128→function deriveThreatData(threats: ApiThreat[]): { cities: ThreatCity[]; connections: [number, number][] } {
   129→  if (threats.length === 0) {
   130→    return { cities: [], connections: [] };
   131→  }
   132→
   133→  // Map threats to fallback cities, using as many as we have threats
   134→  const count = Math.min(threats.length, FALLBACK_CITIES.length);
   135→  const cities: ThreatCity[] = FALLBACK_CITIES.slice(0, count).map((city, i) => {
   136→    const t = threats[i];
   137→    // Assign type based on threat severity
   138→    let type: ThreatCity['type'] = 'target';
   139→    if (t.severity === 'critical' || t.severity === 'high') type = 'source';
   140→    else if (t.severity === 'medium') type = 'hotspot';
   141→
   142→    return {
   143→      ...city,
   144→      type,
   145→      threat: t.title.length > 25 ? t.title.slice(0, 25) + '…' : t.title,
   146→    };
   147→  });
   148→
   149→  // Derive connections: pair source-type indices with target-type indices
   150→  const sourceIndices = cities.reduce<number[]>((acc, c, i) => { if (c.type === 'source') acc.push(i); return acc; }, []);
   151→  const targetIndices = cities.reduce<number[]>((acc, c, i) => { if (c.type === 'target' || c.type === 'hotspot') acc.push(i); return acc; }, []);
   152→
   153→  const connections: [number, number][] = [];
   154→  for (const si of sourceIndices) {
   155→    // Connect each source to 1-2 targets
   156→    for (let j = 0; j < Math.min(2, targetIndices.length); j++) {
   157→      const ti = targetIndices[(si + j) % targetIndices.length];
   158→      connections.push([si, ti]);
   159→    }
   160→  }
   161→  // Also connect some hotspots to targets
   162→  const hotspotIndices = cities.reduce<number[]>((acc, c, i) => { if (c.type === 'hotspot') acc.push(i); return acc; }, []);
   163→  for (const hi of hotspotIndices) {
   164→    if (targetIndices.length > 0) {
   165→      const ti = targetIndices[hi % targetIndices.length];
   166→      connections.push([hi, ti]);
   167→    }
   168→  }
   169→
   170→  return { cities, connections };
   171→}

   177→const R = 2; // Globe radius

   179→function latLngToVec3(lat: number, lng: number, r: number): THREE.Vector3 {
   180→  const phi = (90 - lat) * (Math.PI / 180);
   181→  const theta = (lng + 180) * (Math.PI / 180);
   182→  return new THREE.Vector3(
   183→    -(r * Math.sin(phi) * Math.cos(theta)),
   184→    r * Math.cos(phi),
   185→    r * Math.sin(phi) * Math.sin(theta),
   186→  );
   187→}
   188→
   189→function createArcCurve(start: THREE.Vector3, end: THREE.Vector3, altitude: number): THREE.CubicBezierCurve3 {
   190→  const mid = new THREE.Vector3().addVectors(start, end).multiplyScalar(0.5);
   191→  const dist = start.distanceTo(end);
   192→  mid.normalize().multiplyScalar(R + altitude * (dist / R));
   193→
   194→  const s = new THREE.Vector3().lerpVectors(start, mid, 0.3).normalize().multiplyScalar(R + altitude * 0.5);
   195→  const e = new THREE.Vector3().lerpVectors(end, mid, 0.3).normalize().multiplyScalar(R + altitude * 0.5);
   196→
   197→  return new THREE.CubicBezierCurve3(start, s, e, end);
   198→}

   204→function GlobeSphere() {
   205→  const meshRef = useRef<THREE.Mesh>(null);
   206→
   207→  useFrame((_, delta) => {
   208→    if (meshRef.current) meshRef.current.rotation.y += delta * 0.03;

   211→  return (
   212→    <group>
   213→      {/* Inner dark sphere */}
   214→      <mesh ref={meshRef}>
   215→        <sphereGeometry args={[R - 0.01, 64, 64]} />
   216→        <meshBasicMaterial color="#030806" transparent opacity={0.95} />
   217→      </mesh>
   218→
   219→      {/* Wireframe */}
   220→      <mesh rotation={[0, 0, 0]}>
   221→        <sphereGeometry args={[R, 36, 18]} />
   222→        <meshBasicMaterial color="#00ff88" wireframe transparent opacity={0.06} />
   223→      </mesh>
   224→
   225→      {/* Latitude lines */}
   226→      {[-60, -30, 0, 30, 60].map(lat => {
   227→        const phi = (90 - lat) * (Math.PI / 180);
   228→        const ringR = R * Math.sin(phi);
   229→        const y = R * Math.cos(phi);
   230→        return (
   231→          <mesh key={lat} position={[0, y, 0]} rotation={[Math.PI / 2, 0, 0]}>
   232→            <ringGeometry args={[ringR - 0.003, ringR + 0.003, 128]} />
   233→            <meshBasicMaterial color="#00ff88" transparent opacity={lat === 0 ? 0.15 : 0.07} side={THREE.DoubleSide} />
   234→          </mesh>
   235→        );
   236→      })}
   237→
   238→      {/* Longitude lines */}
   239→      {Array.from({ length: 12 }, (_, i) => i * 30).map(lng => {
   240→        const theta = lng * (Math.PI / 180);
   241→        return (
   242→          <mesh key={lng} rotation={[0, theta, 0]}>
   243→            <ringGeometry args={[R - 0.003, R + 0.003, 128]} />
   244→            <meshBasicMaterial color="#00ff88" transparent opacity={lng === 0 ? 0.15 : 0.05} side={THREE.DoubleSide} />
   245→          </mesh>
   246→        );
   247→      })}
   248→    </group>
   249→  );
   250→}

   253→// ATMOSPHERE GLOW (Fresnel effect)
   254→// ═══════════════════════════════════════════════════════════════════════
   255→
   256→function Atmosphere() {
   257→  const material = useMemo(() => {
   258→    return new THREE.ShaderMaterial({
   259→      vertexShader: `
   260→        varying vec3 vNormal;
   261→        varying vec3 vPosition;
   262→        void main() {
   263→          vNormal = normalize(normalMatrix * normal);
   264→          vPosition = (modelViewMatrix * vec4(position, 1.0)).xyz;
   265→          gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0);
   266→        }
   267→      `,
   268→      fragmentShader: `
   269→        varying vec3 vNormal;
   270→        varying vec3 vPosition;
   271→        void main() {
   272→          float intensity = pow(0.65 - dot(vNormal, vec3(0.0, 0.0, 1.0)), 3.0);
   273→          gl_FragColor = vec4(0.0, 1.0, 0.53, 1.0) * intensity * 0.4;
   274→        }
   275→      `,
   276→      blending: THREE.AdditiveBlending,
   277→      side: THREE.BackSide,
   278→      transparent: true,
   279→      depthWrite: false,
   280→    });
   281→  }, []);
   282→
   283→  return (
   284→    <mesh material={material}>
   285→      <sphereGeometry args={[R + 0.15, 64, 64]} />
   286→    </mesh>
   287→  );
   288→}

   375→// ATTACK ARCS — now accepts cities and connections as props
   376→// ═══════════════════════════════════════════════════════════════════════
   377→
   378→function AttackArcs({ cities, connections }: { cities: ThreatCity[]; connections: [number, number][] }) {
   379→  const groupRef = useRef<THREE.Group>(null);
   380→  const arcsRef = useRef<{
   381→    curve: THREE.CubicBezierCurve3;
   382→    progress: number;
   383→    speed: number;
   384→    color: string;
   385→  }[]>([]);
   386→
   387→  // Initialize arcs
   388→  useMemo(() => {
   389→    arcsRef.current = connections.map(([si, ti]) => {
   390→      const start = latLngToVec3(cities[si].lat, cities[si].lng, R + 0.02);
   391→      const end = latLngToVec3(cities[ti].lat, cities[ti].lng, R + 0.02);
   392→      return {
   393→        curve: createArcCurve(start, end, 0.6),
   394→        progress: Math.random(),
   395→        speed: 0.08 + Math.random() * 0.12,
   396→        color: cities[si].type === 'source' ? '#ff3355' : '#ff8844',
   397→      };
   398→    });
   399→  }, [cities, connections]);
   400→
   401→  useFrame((_, delta) => {
   402→    if (groupRef.current) groupRef.current.rotation.y += delta * 0.03;
   403→
   404→    for (const arc of arcsRef.current) {
   405→      arc.progress = (arc.progress + arc.speed * delta) % 1;
   406→    }
   407→
   408→    // Update arc line geometries
   409→    groupRef.current?.children.forEach((child, i) => {
   410→      const arc = arcsRef.current[i];
   411→      if (!arc || !child) return;
   412→
   413→      const line = child as THREE.Line;
   414→      const geo = line.geometry as THREE.BufferGeometry;
   415→      const posAttr = geo.getAttribute('position') as THREE.BufferAttribute;
   416→      const pts = 50;
   417→
   418→      for (let j = 0; j <= pts; j++) {
   419→        const t = (j / pts + arc.progress) % 1;
   420→        const p = arc.curve.getPoint(t);
   421→        posAttr.setXYZ(j, p.x, p.y, p.z);
   422→      }
   423→      posAttr.needsUpdate = true;
   424→    });
   425→  });
   426→
   427→  const arcData = useMemo(() => {
   428→    return connections.map(([si, ti]) => {
   429→      const start = latLngToVec3(cities[si].lat, cities[si].lng, R + 0.02);
   430→      const end = latLngToVec3(cities[ti].lat, cities[ti].lng, R + 0.02);
   431→      const curve = createArcCurve(start, end, 0.6);
   432→      const pts = 51;
   433→      const positions = new Float32Array(pts * 3);
   434→      for (let j = 0; j < pts; j++) {
   435→        const p = curve.getPoint(j / (pts - 1));
   436→        positions[j * 3] = p.x;
   437→        positions[j * 3 + 1] = p.y;
   438→        positions[j * 3 + 2] = p.z;
   439→      }
   440→      return {
   441→        positions,
   442→        color: cities[si].type === 'source' ? '#ff3355' : '#ff8844',
   443→      };
   444→    });
   445→  }, [cities, connections]);
   446→
   447→  return (
   448→    <group ref={groupRef}>
   449→      {arcData.map((arc, i) => (
   450→        <line key={i}>
   451→          <bufferGeometry>
   452→            <bufferAttribute attach="attributes-position" args={[arc.positions, 3]} />
   453→          </bufferGeometry>
   454→          <lineBasicMaterial color={arc.color} transparent opacity={0.3} />
   455→        </line>
   456→      ))}
   457→    </group>
   458→  );
   459→}

   461→// ═══════════════════════════════════════════════════════════════════════
   462→// TRAVELING PACKETS (dots moving along arcs) — now accepts props
   463→// ═══════════════════════════════════════════════════════════════════════
   464→
   465→function TravelingPackets({ cities, connections }: { cities: ThreatCity[]; connections: [number, number][] }) {
   466→  const meshRefs = useRef<THREE.InstancedMesh>(null);
   467→  const dummy = useMemo(() => new THREE.Object3D(), []);
   468→  const packetsRef = useRef<Array<{
   469→    curve: THREE.CubicBezierCurve3;
   470→    progress: number;
   471→    speed: number;
   472→    color: THREE.Color;
   473→  }>>([]);
   474→
   475→  const count = connections.length;
   476→
   477→  // Initialize packets into ref
   478→  useEffect(() => {
   479→    packetsRef.current = connections.map(([si, ti]) => {
   480→      const start = latLngToVec3(cities[si].lat, cities[si].lng, R + 0.02);
   481→      const end = latLngToVec3(cities[ti].lat, cities[ti].lng, R + 0.02);
   482→      return {
   483→        curve: createArcCurve(start, end, 0.6),
   484→        progress: Math.random(),
   485→        speed: 0.06 + Math.random() * 0.1,
   486→        color: new THREE.Color(cities[si].type === 'source' ? '#ff3355' : '#ff8844'),
   487→      };
   488→    });
   489→  }, [cities, connections]);
   490→
   491→  useFrame((_, delta) => {
   492→    if (!meshRefs.current) return;
   493→    const packets = packetsRef.current;
   494→
   495→    for (let i = 0; i < count; i++) {
   496→      const p = packets[i];
   497→      p.progress = (p.progress + p.speed * delta) % 1;
   498→      const pos = p.curve.getPoint(p.progress);
   499→      dummy.position.copy(pos);
   500→      dummy.scale.setScalar(1 + Math.sin(p.progress * Math.PI) * 0.8);
   501→      dummy.updateMatrix();
   502→      meshRefs.current!.setMatrixAt(i, dummy.matrix);
   503→      meshRefs.current!.setColorAt(i, p.color);
   504→    }
   505→    meshRefs.current.instanceMatrix.needsUpdate = true;
   506→    if (meshRefs.current.instanceColor) meshRefs.current.instanceColor.needsUpdate = true;
   507→  });
   508→
   509→  if (count === 0) return null;
   510→
   511→  return (
   512→    <instancedMesh ref={meshRefs} args={[undefined, undefined, count]}>
   513→      <sphereGeometry args={[0.02, 8, 8]} />
   514→      <meshBasicMaterial />
   515→    </instancedMesh>
   516→  );
   517→}

   519→// ═══════════════════════════════════════════════════════════════════════
   520→// ORBITAL RING + SATELLITES
   521→// ═══════════════════════════════════════════════════════════════════════
   522→
   523→function OrbitalRing() {
   524→  const ringRef = useRef<THREE.Mesh>(null);
   525→  const satRef = useRef<THREE.Mesh>(null);
   526→
   527→  useFrame((state) => {
   528→    const t = state.clock.elapsedTime;
   529→    if (ringRef.current) {
   530→      ringRef.current.rotation.x = Math.PI / 2 + Math.sin(t * 0.1) * 0.1;
   531→      ringRef.current.rotation.z = t * 0.02;
   532→    }
   533→    if (satRef.current) {
   534→      const angle = t * 0.5;
   535→      satRef.current.position.set(
   536→        Math.cos(angle) * (R + 0.5),
   537→        Math.sin(t * 0.1) * 0.2,
   538→        Math.sin(angle) * (R + 0.5),
   539→      );
   540→    }
   541→  });
   542→
   543→  return (
   544→    <group>
   545→      <mesh ref={ringRef}>
   546→        <torusGeometry args={[R + 0.5, 0.003, 8, 128]} />
   547→        <meshBasicMaterial color="#00ff88" transparent opacity={0.15} />
   548→      </mesh>
   549→      <mesh ref={satRef}>
   550→        <sphereGeometry args={[0.025, 8, 8]} />
   551→        <meshBasicMaterial color="#00ff88" />
   552→      </mesh>
   553→    </group>
   554→  );
   555→}

   558→// INNER GLOW (core light)
   559→// ═══════════════════════════════════════════════════════════════════════
   560→
   561→function InnerGlow() {
   562→  const lightRef = useRef<THREE.PointLight>(null);
   563→
   564→  useFrame((state) => {
   565→    if (lightRef.current) {
   566→      lightRef.current.intensity = 1.5 + Math.sin(state.clock.elapsedTime * 2) * 0.3;
   567→    }
   568→  });
   569→
   570→  return <pointLight ref={lightRef} color="#00ff88" intensity={1.5} distance={5} decay={2} />;
   571→}

   712→// ═══════════════════════════════════════════════════════════════════════
   713→// MAIN COMPONENT — fetches real threat data
   714→// ═══════════════════════════════════════════════════════════════════════
   715→
   716→export function ThreatGlobe() {
   717→  const [cities, setCities] = useState<ThreatCity[]>([]);
   718→  const [connections, setConnections] = useState<[number, number][]>([]);
   719→  const [noThreats, setNoThreats] = useState(false);
   720→  const [dataReady, setDataReady] = useState(false);
   721→
   722→  useEffect(() => {
   723→    let cancelled = false;
   724→    async function fetchThreats() {
   725→      try {
   726→        const res = await fetch('/api/threats');
   727→        if (!res.ok) return;
   728→        const json = await res.json();
   729→        if (cancelled) return;
   730→
   731→        const threats: ApiThreat[] = json.threats || [];
   732→        const { cities: derivedCities, connections: derivedConnections } = deriveThreatData(threats);
   733→
   734→        if (cancelled) return;
   735→        setCities(derivedCities);
   736→        setConnections(derivedConnections);
   737→        setNoThreats(threats.length === 0);
   738→        setDataReady(true);
   739→      } catch {
   740→        // On error, show empty globe
   741→        if (!cancelled) {
   742→          setNoThreats(true);
   743→          setDataReady(true);
   744→        }
   745→      }
   746→    }
   747→    fetchThreats();
   748→    return () => { cancelled = true; };
   749→  }, []);
   750→
   751→  return (
   752→    <motion.div
   753→      initial={{ opacity: 0 }}
   754→      animate={{ opacity: 1 }}
   755→      transition={{ duration: 1 }}
   756→      className="cyber-card rounded-2xl overflow-hidden relative"
   757→      style={{ height: 'calc(100vh - 140px)', minHeight: '500px' }}
   758→    >
   759→      {/* Canvas */}
   760→      <Canvas
   761→        camera={{ position: [0, 1.5, 5], fov: 45 }}
   762→        gl={{ antialias: true, alpha: true, powerPreference: 'high-performance' }}
   763→        style={{ background: '#030806' }}
   764→        dpr={[1, 2]}
   765→      >
   766→        {dataReady && <Scene cities={cities} connections={connections} />}
   767→        <EffectComposer>
   768→          <Bloom
   769→            intensity={0.8}
   770→            luminanceThreshold={0.1}
   771→            luminanceSmoothing={0.9}
   772→            mipmapBlur
   773→          />
   774→        </EffectComposer>
   775→      </Canvas>
   776→
   777→      {/* HUD Overlay */}
   778→      {dataReady && <HUD cities={cities} connections={connections} noThreats={noThreats} />}
   779→    </motion.div>
   780→  );
