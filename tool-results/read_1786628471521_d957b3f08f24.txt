    17→function tcpProbe(host: string, port: number, timeoutMs: number): Promise<boolean> {
    18→  return new Promise((resolve) => {
    19→    const timer = setTimeout(() => { try { s.destroy(); } catch {} resolve(false); }, timeoutMs);
    20→    const s = net.createConnection({ host, port, timeout: timeoutMs });
    21→    s.on('connect', () => { clearTimeout(timer); try { s.destroy(); } catch {} resolve(true); });
    22→    s.on('timeout', () => { clearTimeout(timer); try { s.destroy(); } catch {} resolve(false); });
    23→    s.on('error', () => { clearTimeout(timer); resolve(false); });
    24→  });
    25→}

    27→function tcpBannerGrab(host: string, port: number, timeoutMs: number, maxBytes = 500): Promise<string> {
    28→  return new Promise((resolve) => {
    29→    let data = '';
    30→    const timer = setTimeout(() => { try { s.destroy(); } catch {} resolve(data); }, timeoutMs);
    31→    const s = net.createConnection({ host, port, timeout: timeoutMs });
    32→    s.on('data', (chunk: Buffer) => {
    33→      data += chunk.toString();
    34→      if (data.length >= maxBytes) { clearTimeout(timer); try { s.destroy(); } catch {} resolve(data.substring(0, maxBytes)); }
    35→    });
    36→    s.on('timeout', () => { clearTimeout(timer); try { s.destroy(); } catch {} resolve(data); });
    37→    s.on('error', () => { clearTimeout(timer); resolve(data); });
    38→    s.on('close', () => { clearTimeout(timer); resolve(data); });
    39→  });
    40→}

    42→async function handleCurl(cmd: string, timeout: number): Promise<string> {
    43→  const isHead = /\b-sI\b|\b-I\b/.test(cmd);
    44→  const followRedirects = /\b-L\b/.test(cmd);
    45→  const timeoutMatch = cmd.match(/--max-time\s+(\d+)/);
    46→  const timeoutSec = timeoutMatch ? parseInt(timeoutMatch[1]) : 8;
    47→  const headers: Record<string, string> = { 'User-Agent': 'Mozilla/5.0' };
    48→  const headerRegex = /-H\s+["']([^"']+)["']/g;
    49→  let m: RegExpExecArray | null;
    50→  while ((m = headerRegex.exec(cmd)) !== null) {
    51→    const ci = m[1].indexOf(':');
    52→    if (ci > 0) headers[m[1].substring(0, ci).trim()] = m[1].substring(ci + 1).trim();
    53→  }
    54→  const urlMatch = cmd.match(/(?:https?:\/\/[^\s"'`]+)/);
    55→  const url = urlMatch ? urlMatch[0] : '';
    56→  if (!url) return '';
    57→
    58→  try {
    59→    if (isHead && !followRedirects) {
    60→      const res = await safeFetch(url, { method: 'HEAD', headers, followRedirects: false, timeout: timeoutSec * 1000 });
    61→      if (!res.ok && res.status === 403) return '';
    62→      let out = `HTTP/1.1 ${res.status}\n`;
    63→      for (const [k, v] of Object.entries(res.headers)) { out += `${k}: ${v}\n`; }
    64→      return out;
    65→    }
    66→    if (isHead && followRedirects) {
    67→      let cur = url; let out = '';
    68→      for (let i = 0; i < 5; i++) {
    69→        const res = await safeFetch(cur, { method: 'HEAD', headers, followRedirects: false, timeout: timeoutSec * 1000 });
    70→        if (!res.ok && res.status === 403) return '';
    71→        out += `HTTP/1.1 ${res.status}\n`;
    72→        for (const [k, v] of Object.entries(res.headers)) { out += `${k}: ${v}\n`; }
    73→        out += '\n';
    74→        if (res.status >= 300 && res.status < 400) {
    75→          const loc = res.headers['location']; if (loc) { cur = new URL(loc, cur).href; continue; }
    76→        }
    77→        break;
    78→      }
    79→      return out;
    80→    }
    81→    const res = await safeFetch(url, { method: 'GET', headers, followRedirects, timeout: timeoutSec * 1000 });
    82→    if (!res.ok && res.status === 403) return '';
    83→    return res.text;
    84→  } catch { return ''; }
    85→}

    87→async function handleDig(cmd: string, timeout: number): Promise<string> {
    88→  if (cmd.includes('axfr')) return '';
    89→  const isPtr = cmd.includes('-x');
    90→  let cleaned = cmd.replace(/\bdig\b/, '').replace(/\+short\b/g, '').replace(/\+time=\d+/g, '').replace(/\+tries=\d+/g, '').replace(/@\S+/g, '').replace(/\+noall\b/g, '').replace(/\+answer\b/g, '').replace(/2>\/>\/dev\/null/g, '').replace(/2>&1/g, '').trim();
    91→  if (isPtr) {
    92→    const pm = cleaned.match(/-x\s+(\S+)/);
    93→    const ip = pm ? pm[1] : '';
    94→    if (!ip) return '';
    95→    try { return (await dns.resolvePtr(ip)).join('\n'); } catch { return ''; }
    96→  }
    97→  const tokens = cleaned.split(/\s+/).filter(Boolean);
    98→  if (tokens.length === 0) return '';
    99→  const recordTypes = ['A','AAAA','NS','CNAME','MX','TXT','PTR','SOA','SRV','ANY'];
   100→  let domain: string; let type = 'A';
   101→  const last = tokens[tokens.length - 1].toUpperCase();
   102→  if (tokens.length >= 2 && recordTypes.includes(last)) {
   103→    type = last; domain = tokens.slice(0, -1).join(' ');
   104→  } else {
   105→    domain = tokens.join(' ');
   106→  }
   107→  if (!domain) return '';
   108→  try {
   109→    let result: string[] = [];
   110→    switch (type) {
   111→      case 'A': result = await dns.resolve4(domain); break;
   112→      case 'NS': result = await dns.resolveNs(domain); break;
   113→      case 'CNAME': result = await dns.resolveCname(domain); break;
   114→      case 'TXT': { const r = await dns.resolveTxt(domain); result = r.map(x => x.join('')); break; }
   115→      case 'PTR': result = await dns.resolvePtr(domain); break;
   116→      case 'ANY': {
   117→        const r = await dns.resolveAny(domain);
   118→        result = r.map(x => { switch (x.type) { case 'A': case 'AAAA': return x.address; case 'MX': return `${x.exchange} ${x.priority}`; case 'TXT': return x.entries?.join('') || ''; case 'NS': case 'CNAME': case 'PTR': return x.value; default: return JSON.stringify(x); } });
   119→        break;
   120→      }
   121→      default: result = await dns.resolve4(domain); break;
   122→    }
   123→    return result.join('\n');
   124→  } catch { return ''; }
   125→}

   127→async function handleOpenssl(cmd: string, timeout: number): Promise<string> {
   128→  const cm = cmd.match(/-connect\s+(\S+:(\d+))/);
   129→  const sm = cmd.match(/-servername\s+(\S+)/);
   130→  const hasTlsextdebug = cmd.includes('-tlsextdebug');
   131→  const hasStatus = cmd.includes('-status');
   132→  let host = '', port = 443;
   133→  if (cm) { const [h, p] = cm[1].split(':'); host = h; port = parseInt(p) || 443; }
   134→  if (!host) return '';
   135→  const servername = sm ? sm[1] : host;
   136→  return new Promise((resolve) => {
   137→    const timer = setTimeout(() => { try { s.destroy(); } catch {} resolve(''); }, timeout);
   138→    const opts: tls.ConnectionOptions = { host, port, servername, rejectUnauthorized: false };
   139→    const s = tls.connect(opts, () => {
   140→      const protocol = s.getProtocol() || '';
   141→      const cipher = s.getCipher();
   142→      const cert = s.getPeerCertificate();
   143→      let out = '';
   144→      out += `    Protocol  : ${protocol}\n`;
   145→      const cipherName = protocol === 'TLSv1.3' ? `ECDHE-RSA-${cipher.name}` : cipher.name;
   146→      out += `    Cipher    : ${cipherName}\n`;
   147→      if (cert) {
   148→        if (cert.valid_from) out += `notBefore=${cert.valid_from}\n`;
   149→        if (cert.valid_to) out += `notAfter=${cert.valid_to}\n`;
   150→      }
   151→      if (hasTlsextdebug) { /* heartbeat not directly exposed in Node.js TLS */ }
   152→      if (hasStatus) { /* OCSP stapling not directly exposed */ }
   153→      clearTimeout(timer); try { s.destroy(); } catch {} resolve(out);
   154→    });
   155→    s.on('timeout', () => { clearTimeout(timer); try { s.destroy(); } catch {} resolve(''); });
   156→    s.on('error', () => { clearTimeout(timer); resolve(''); });
   157→  });
   158→}

   160→async function handleNetcat(cmd: string, timeout: number): Promise<string> {
   161→  const bashMatch = cmd.match(/\/dev\/tcp\/([^/]+)\/(\d+)/);
   162→  if (bashMatch) {
   163→    const open = await tcpProbe(bashMatch[1], parseInt(bashMatch[2]), Math.min(timeout, 3000));
   164→    return open ? 'OPEN' : 'CLOSED';
   165→  }
   166→  const ncMatch = cmd.match(/nc\s+(?:-\w+\s+)*(\S+)\s+(\d+)/);
   167→  if (ncMatch) {
   168→    return tcpBannerGrab(ncMatch[1], parseInt(ncMatch[2]), timeout);
   169→  }
   170→  return '';
   171→}

   173→function applyPipe(input: string, pipeCmd: string): string {
   174→  if (pipeCmd.includes('openssl x509')) {
   175→    return input.split('\n').filter(l => l.includes('notAfter=') || l.includes('notBefore=')).join('\n');
   176→  }
   177→  const headBytes = pipeCmd.match(/head\s+-c\s+(\d+)/);
   178→  if (headBytes) return input.substring(0, parseInt(headBytes[1]));
   179→  const headLines = pipeCmd.match(/head\s+-(\d+)/);
   180→  if (headLines) return input.split('\n').slice(0, parseInt(headLines[1])).join('\n');
   181→  const grepMatch = pipeCmd.match(/^grep\s+(-\w+\s+)?(.+)$/);
   182→  if (grepMatch) {
   183→    const flags = grepMatch[1]?.trim() || '';
   184→    let pattern = grepMatch[2].trim();
   185→    if ((pattern.startsWith('"') && pattern.endsWith('"')) || (pattern.startsWith("'") && pattern.endsWith("'"))) pattern = pattern.slice(1, -1);
   186→    const isOI = flags.includes('o');
   187→    const isI = flags.includes('i');
   188→    try {
   189→      if (isOI) {
   190→        const regex = new RegExp(pattern, isI ? 'gi' : 'g');
   191→        const matches: string[] = []; let match;
   192→        while ((match = regex.exec(input)) !== null) { matches.push(match[0]); if (matches.length > 100) break; }
   193→        return matches.join('\n');
   194→      }
   195→      const regex = new RegExp(pattern, isI ? 'i' : '');
   196→      return input.split('\n').filter(l => regex.test(l)).join('\n');
   197→    } catch { return ''; }
   198→  }
   199→  return input;
   200→}

   202→async function run(cmd: string, timeout = 8000): Promise<string> {
   203→  try {
   204→    if (cmd.includes('/dev/tcp')) {
   205→      const m = cmd.match(/\/dev\/tcp\/([^/]+)\/(\d+)/);
   206→      if (m) { const open = await tcpProbe(m[1], parseInt(m[2]), Math.min(timeout, 3000)); return cmd.includes('OPEN') && cmd.includes('CLOSED') ? (open ? 'OPEN' : 'CLOSED') : ''; }
   207→      return '';
   208→    }
   209→    let effectiveCmd = cmd.replace(/^echo\s*(""\s*)?\|\s*/, '').trim();
   210→    effectiveCmd = effectiveCmd.replace(/\|\|/g, '\x00\x00');
   211→    const segments: string[] = []; let cur = ''; let inQ = false; let qCh = '';
   212→    for (let i = 0; i < effectiveCmd.length; i++) {
   213→      const ch = effectiveCmd[i];
   214→      if (inQ) { cur += ch; if (ch === qCh) inQ = false; }
   215→      else if (ch === '"' || ch === "'") { inQ = true; qCh = ch; cur += ch; }
   216→      else if (ch === '|') { segments.push(cur.trim()); cur = ''; }
   217→      else { cur += ch; }
   218→    }
   219→    segments.push(cur.trim());
   220→    const restored = segments.map(s => s.replace(/\x00\x00/g, '||'));
   221→    const mainCmd = restored[0]; const pipes = restored.slice(1);
   222→    let output = '';
   223→    if (mainCmd.includes('curl')) output = await handleCurl(mainCmd, timeout);
   224→    else if (mainCmd.includes('dig')) output = await handleDig(mainCmd, timeout);
   225→    else if (mainCmd.includes('openssl')) output = await handleOpenssl(mainCmd, timeout);
   226→    else if (mainCmd.includes('nc ') || mainCmd.includes('/dev/tcp')) output = await handleNetcat(mainCmd, timeout);
   227→    else if (mainCmd.includes('whois')) return '';
   228→    else return '';
   229→    for (const pipe of pipes) output = applyPipe(output, pipe);
   230→    return output.trim();
   231→  } catch { return ''; }
   232→}

   341→async function bannerGrab(domain: string, ip: string | null): Promise<{ banners: Array<{ port: number; banner: string; service: string }>; cveMatches: CVEEntry[] }> {
   342→  const target = ip || domain;
   343→  const ports = [21, 22, 25, 80, 110, 143, 443, 445, 993, 995, 1433, 1521, 3306, 3389, 5432, 5900, 6379, 8080, 8443, 8888, 9090, 27017];
   344→  const banners: Array<{ port: number; banner: string; service: string }> = [];
   345→
   346→  for (const port of ports) {
   347→    const banner = await run(
   348→      `echo "" | timeout 4 nc -w3 ${target} ${port} 2>/dev/null | head -c 500`,
   349→      6000
   350→    );
   351→    if (banner && banner.length > 3) {
   352→      const service = detectService(port, banner);
   353→      banners.push({ port, banner: banner.replace(/[\x00-\x1f\x7f]/g, '.').substring(0, 200), service });
   354→    }
   355→  }
   356→
   357→  // HTTP banner via curl
   358→  const httpBanner = await run(`curl -sI --max-time 6 https://${domain} 2>/dev/null | head -20`, 8000);
   359→  if (httpBanner) banners.push({ port: 443, banner: httpBanner.substring(0, 300), service: 'http' });
   360→  const httpBanner2 = await run(`curl -sI --max-time 6 http://${domain} 2>/dev/null | head -20`, 8000);
   361→  if (httpBanner2) banners.push({ port: 80, banner: httpBanner2.substring(0, 300), service: 'http' });
   362→
   363→  // Match banners against CVE database
   364→  const combined = banners.map(b => b.banner).join(' ').toLowerCase();
   365→  const cveMatches = CVE_DATABASE.filter(cve => {
   366→    return cve.affected.some(a => combined.includes(a)) && cve.versionPattern.test(combined);
   367→  });
   368→
   369→  return { banners, cveMatches };
   370→}
   371→
   372→function detectService(port: number, banner: string): string {
   373→  const b = banner.toLowerCase();
   374→  if (port === 22 || b.includes('ssh')) return 'openssh';
   375→  if (port === 21 || b.includes('ftp')) return 'ftp';
   376→  if (port === 25 || b.includes('smtp') || b.includes('postfix') || b.includes('esmtp')) return 'postfix';
   377→  if (port === 80 || port === 8080 || port === 8443 || b.includes('http')) return 'http';
   378→  if (port === 110 || b.includes('pop3')) return 'pop3';
   379→  if (port === 143 || b.includes('imap')) return 'imap';
   380→  if (port === 443 || b.includes('ssl') || b.includes('tls')) return 'https';
   381→  if (port === 3306 || b.includes('mysql')) return 'mysql';
   382→  if (port === 5432 || b.includes('postgresql') || b.includes('postgres')) return 'postgresql';
   383→  if (port === 6379 || b.includes('redis')) return 'redis';
   384→  if (port === 27017 || b.includes('mongodb')) return 'mongodb';
   385→  if (b.includes('nginx')) return 'nginx';
   386→  if (b.includes('apache')) return 'apache';
   387→  if (b.includes('tomcat')) return 'tomcat';
   388→  return 'unknown';
   389→}

   395→async function scanHTTPVulns(domain: string): Promise<Finding[]> {
   396→  const findings: Finding[] = [];
   397→
   398→  // CORS misconfiguration
   399→  const corsResponse = await run(
   400→    `curl -sI --max-time 6 -H "Origin: https://evil.attacker.com" https://${domain} 2>/dev/null | grep -i "access-control-allow-origin"`,
   401→    8000
   402→  );
   403→  if (corsResponse.includes('evil.attacker.com') || corsResponse.includes('*')) {
   404→    findings.push({
   405→      title: 'CRITICAL: CORS Misconfiguration — Reflects Attacker Origin',
   406→      severity: 'critical', category: 'http-vuln',
   407→      description: `Server reflects any Origin header. An attacker at evil.attacker.com can make cross-origin requests and read the response, enabling credential theft and data exfiltration from any authenticated user.`,
   408→      evidence: `curl -H "Origin: https://evil.attacker.com" → ${corsResponse.trim()}`,
   409→      asset: domain,
   410→    });
   411→  }
   412→
   413→  // Open redirect test
   414→  const redirectPaths = ['/redirect?url=https://evil.com', '/login?return=https://evil.com',
   415→    '/logout?redirect=https://evil.com', '/next=https://evil.com', '/goto=https://evil.com',
   416→    '/auth?redirect_uri=https://evil.com', '/callback?redirect=https://evil.com'];
   417→  for (const path of redirectPaths) {
   418→    const redirResult = await run(
   419→      `curl -sI --max-time 4 -L "https://${domain}${path}" 2>/dev/null | grep -i "location"`,
   420→      6000
   421→    );
   422→    if (redirResult.includes('evil.com') || redirResult.includes('evil')) {
   423→      findings.push({
   424→        title: 'Open Redirect Vulnerability Detected',
   425→        severity: 'high', category: 'http-vuln',
   426→        description: `Path ${path} redirects to external domain without validation. Attackers use open redirects for phishing campaigns that appear to come from legitimate domain.`,
   427→        evidence: `GET ${path} → Location: ${redirResult.trim().substring(0, 100)}`,
   428→        asset: `${domain}${path}`,
   429→      });
   430→      break;
   431→    }
   432→  }
   433→
   434→  // Path traversal tests
   435→  const traversalPaths = ['../../../etc/passwd', '/static/../../etc/passwd', '/images/..%2f..%2f..%2fetc/passwd', '/file/..%252f..%252fetc/passwd'];
   436→  for (const path of traversalPaths) {
   437→    const ptResult = await run(
   438→      `curl -s --max-time 4 "https://${domain}${path}" 2>/dev/null | head -c 500`,
   439→      6000
   440→    );
   441→    if (ptResult.includes('root:') || ptResult.includes('nobody:') || ptResult.includes('/bin/bash')) {
   442→      findings.push({
   443→        title: 'CRITICAL: Path Traversal — /etc/passwd Exposed',
   444→        severity: 'critical', category: 'http-vuln',
   445→        description: `Server responds with file system contents. Path traversal allows reading arbitrary files including /etc/passwd, /etc/shadow, and application secrets.`,
   446→        evidence: `GET ${path} → "root:x:0:0:..." found in response`,
   447→        asset: `${domain}${path}`,
   448→      });
   449→      break;
   450→    }
   451→  }
   452→
   453→  // SSRF detection (check if internal IPs are accessible)
   454→  const ssrfPaths = ['/?url=http://169.254.169.254/latest/meta-data/', '/api/fetch?url=http://127.0.0.1:80', '/proxy?url=http://10.0.0.1'];
   455→  for (const path of ssrfPaths) {
   456→    const ssrfResult = await run(
   457→      `curl -s --max-time 4 "https://${domain}${path}" 2>/dev/null | head -c 500`,
   458→      6000
   459→    );
   460→    if (ssrfResult.includes('ami-id') || ssrfResult.includes('instance-id') || ssrfResult.includes('local-hostname') || ssrfResult.includes('127.0.0.1')) {
   461→      findings.push({
   462→        title: 'CRITICAL: SSRF — Internal Network Accessible',
   463→        severity: 'critical', category: 'http-vuln',
   464→        description: `Server makes requests to internal URLs. SSRF can access cloud metadata (AWS/GCP), internal services, and bypass firewalls.`,
   465→        evidence: `GET ${path} → internal response detected`,
   466→        asset: `${domain}${path}`,
   467→      });
   468→      break;
   469→    }
   470→  }
   471→
   472→  // XSS detection (reflected input)
   473→  const xssPayloads = ['<script>alert(1)</script>', '"><img src=x onerror=alert(1)>', "'-alert(1)-'"];
   474→  for (const payload of xssPayloads) {
   475→    const xssResult = await run(
   476→      `curl -s --max-time 5 "https://${domain}/search?q=${encodeURIComponent(payload)}" 2>/dev/null | head -c 2000`,
   477→      7000
   478→    );
   479→    if (xssResult.includes(payload) && !xssResult.includes('alert(1)</script></') === false) {
   480→      const reflected = xssResult.includes('<script>alert(1)</script>') || xssResult.includes('onerror=alert(1)');
   481→      if (reflected) {
   482→        findings.push({
   483→          title: 'Reflected XSS — Input Executed in Response',
   484→          severity: 'high', category: 'http-vuln',
   485→          description: `User input is reflected in page content without sanitization. XSS enables session hijacking, credential theft, and malware delivery.`,
   486→          evidence: `GET /search?q=<payload> → reflected unescaped in HTML`,
   487→          asset: `${domain}/search`,
   488→        });
   489→        break;
   490→      }
   491→    }
   492→  }
   493→
   494→  // Clickjacking (X-Frame-Options)
   495→  const xfo = await run(`curl -sI --max-time 5 https://${domain} 2>/dev/null | grep -i "x-frame-options"`, 7000);
   496→  if (!xfo) {
   497→    const cspFrame = await run(`curl -sI --max-time 5 https://${domain} 2>/dev/null | grep -i "content-security-policy" | grep -i "frame-ancestors"`, 7000);
   498→    if (!cspFrame) {
   499→      findings.push({
   500→        title: 'Clickjacking — No Frame Protection',
   501→        severity: 'medium', category: 'http-vuln',
   502→        description: 'No X-Frame-Options or frame-ancestors CSP directive. Attacker can embed site in iframe to trick users into clicking hidden buttons (account takeover, unauthorized actions).',
   503→        evidence: 'No X-Frame-Options header, no frame-ancestors in CSP',
   504→        asset: domain,
   505→      });
   506→    }
   507→  }
   508→
   509→  // Cookie security analysis
   510→  const setCookies = await run(`curl -sI --max-time 5 https://${domain} 2>/dev/null | grep -i "set-cookie"`, 7000);
   511→  if (setCookies) {
   512→    const insecureCookies: string[] = [];
   513→    if (!setCookies.toLowerCase().includes('secure')) insecureCookies.push('Secure flag missing');
   514→    if (!setCookies.toLowerCase().includes('httponly')) insecureCookies.push('HttpOnly flag missing');
   515→    if (!setCookies.toLowerCase().includes('samesite')) insecureCookies.push('SameSite missing');
   516→
   517→    if (insecureCookies.length > 0) {
   518→      findings.push({
   519→        title: `Insecure Cookie Configuration — ${insecureCookies.length} issues`,
   520→        severity: 'medium', category: 'http-vuln',
   521→        description: `Cookies lack security flags: ${insecureCookies.join(', ')}. Without Secure, cookies sent over HTTP. Without HttpOnly, accessible to JavaScript (XSS target). Without SameSite, vulnerable to CSRF.`,
   522→        evidence: `Set-Cookie: ${setCookies.substring(0, 100)} | Issues: ${insecureCookies.join(', ')}`,
   523→        asset: domain,
   524→      });
   525→    }
   526→  }
   527→
   528→  // Information disclosure — server version
   529→  const serverHeader = await run(`curl -sI --max-time 5 https://${domain} 2>/dev/null | grep -i "^server:"`, 7000);
   530→  if (serverHeader && serverHeader.match(/apache\/\d|nginx\/\d|php\/\d|tomcat|express|gunicorn/i)) {
   531→    findings.push({
   532→      title: 'Information Disclosure — Server Version Leaked',
   533→      severity: 'low', category: 'http-vuln',
   534→      description: `Server header reveals exact software version: ${serverHeader.trim()}. Attackers use version info to find known exploits.`,
   535→      evidence: serverHeader.trim(),
   536→      asset: domain,
   537→    });
   538→  }
   539→
   540→  // CSRF token check
   541→  const csrfCheck = await run(`curl -s --max-time 5 "https://${domain}/login" 2>/dev/null | grep -iE "csrf|_token|authenticity_token|x-csrf" | head -3`, 7000);
   542→  if (!csrfCheck) {
   543→    findings.push({
   544→      title: 'CSRF Protection — No Anti-CSRF Token Detected',
   545→      severity: 'medium', category: 'http-vuln',
   546→      description: 'No CSRF token found in login form. State-changing requests may be forged from other sites, enabling unauthorized actions on behalf of authenticated users.',
   547→      evidence: 'No csrf_token, authenticity_token, or x-csrf-token in login page',
   548→      asset: `${domain}/login`,
   549→    });
   550→  }
   551→
   552→  // Mixed content detection
   553→  const mixedContent = await run(
   554→    `curl -s --max-time 8 https://${domain} 2>/dev/null | grep -oiE 'src="http://[^"]+"|href="http://[^"]+"|url\("http://[^)]+"\)' | head -5`,
   555→    10000
   556→  );
   557→  if (mixedContent) {
   558→    findings.push({
   559→      title: `Mixed Content — ${mixedContent.split('\n').filter(Boolean).length} HTTP Resources on HTTPS Page`,
   560→      severity: 'low', category: 'http-vuln',
   561→      description: 'HTTPS page loads resources via HTTP. Browsers will show warnings, and MITM can inject content into the insecure resources.',
   562→      evidence: mixedContent.trim().substring(0, 150),
   563→      asset: domain,
   564→    });
   565→  }
   566→
   567→  return findings;
   568→}

   571→// 3. SSL/TLS ATTACK VECTOR SCANNER
   572→// ══════════════════════════════════════════════════════════════════════════════

   574→async function scanSSLVulns(domain: string): Promise<Finding[]> {
   575→  const findings: Finding[] = [];
   576→
   577→  // Full SSL scan with all cipher suites
   578→  const sslFull = await run(`echo | openssl s_client -connect ${domain}:443 -servername ${domain} 2>/dev/null`, 10000);
   579→  const sslLines = sslFull.split('\n').map(l => l.trim()).filter(Boolean);
   580→
   581→  // Protocol version checks
   582→  const protocolLine = sslLines.find(l => l.startsWith('Protocol'));
   583→  const proto = protocolLine || '';
   584→
   585→  if (proto.includes('TLSv1.0')) {
   586→    findings.push({
   587→      title: 'BEAST Attack Vulnerable — TLS 1.0 Active',
   588→      severity: 'high', category: 'ssl-vuln',
   589→      description: 'TLS 1.0 is vulnerable to BEAST (Browser Exploit Against SSL/TLS). Block cipher attack can decrypt communication blocks. PCI-DSS prohibits TLS 1.0.',
   590→      evidence: proto,
   591→      asset: `${domain}:443`,
   592→    });
   593→  }
   594→  if (proto.includes('TLSv1.1')) {
   595→    findings.push({
   596→      title: 'Deprecated TLS 1.1 in Use',
   597→      severity: 'medium', category: 'ssl-vuln',
   598→      description: 'TLS 1.1 is deprecated (RFC 8996). Remove support to prevent downgrade attacks.',
   599→      evidence: proto,
   600→      asset: `${domain}:443`,
   601→    });
   602→  }
   603→  if (proto.includes('SSLv3')) {
   604→    findings.push({
   605→      title: 'CRITICAL: POODLE — SSLv3 Detected',
   606→      severity: 'critical', category: 'ssl-vuln',
   607→      description: 'SSLv3 is extremely vulnerable to POODLE (Padding Oracle On Downgraded Legacy Encryption). Complete decryption of encrypted sessions possible.',
   608→      evidence: proto,
   609→      asset: `${domain}:443`,
   610→    });
   611→  }
   612→  if (proto.includes('SSLv2')) {
   613→    findings.push({
   614→      title: 'CRITICAL: DROWN Attack — SSLv2 Detected',
   615→      severity: 'critical', category: 'ssl-vuln',
   616→      description: 'SSLv2 enables DROWN (Decrypting RSA with Obsolete and Weakened eNcryption). Cross-protocol attack can decrypt TLS connections.',
   617→      evidence: proto,
   618→      asset: `${domain}:443`,
   619→    });
   620→  }
   621→
   622→  // Cipher analysis
   623→  const cipherLine = sslLines.find(l => l.includes('Cipher'));
   624→  const cipher = cipherLine || '';
   625→
   626→  // Check for weak ciphers by testing multiple
   627→  const weakCiphers = ['RC4', 'DES', '3DES', 'NULL', 'EXPORT', 'MD5', 'RC2', 'IDEA'];
   628→  const cipherSuite = await run(
   629→    `echo | openssl s_client -connect ${domain}:443 -servername ${domain} -cipher 'ALL:eNULL' 2>/dev/null | grep "Cipher"`,
   630→    10000
   631→  );
   632→
   633→  for (const weak of weakCiphers) {
   634→    if (cipherSuite.toUpperCase().includes(weak)) {
   635→      findings.push({
   636→        title: `CRITICAL: Weak Cipher ${weak} Accepted`,
   637→        severity: 'critical', category: 'ssl-vuln',
   638→        description: `Server accepts ${weak} cipher suite. ${weak === 'RC4' ? 'RC4 has known statistical biases exploitable since 2013.' : weak === 'NULL' ? 'NULL cipher means NO encryption — traffic is plaintext.' : weak === 'EXPORT' ? 'EXPORT-grade ciphers are trivially breakable (40-bit keys).' : `${weak} is cryptographically broken and must be disabled.`}`,
   639→        evidence: `Cipher: ${cipherSuite.trim()}`,
   640→        asset: `${domain}:443`,
   641→      });
   642→      break;
   643→    }
   644→  }
   645→
   646→  if (cipher.includes('CBC') && proto.includes('TLSv1')) {
   647→    findings.push({
   648→      title: 'BEAST Susceptible — CBC Mode + TLS 1.0',
   649→      severity: 'medium', category: 'ssl-vuln',
   650→      description: 'CBC cipher suites with TLS 1.0 are vulnerable to BEAST attack. Use GCM-mode ciphers or upgrade to TLS 1.2+.',
   651→      evidence: `${proto} | ${cipher}`,
   652→      asset: `${domain}:443`,
   653→    });
   654→  }
   655→
   656→  // Heartbleed check (CVE-2014-0160)
   657→  const heartbleedCheck = await run(
   658→    `echo | openssl s_client -connect ${domain}:443 -servername ${domain} -tlsextdebug 2>&1 | grep -i "heartbeat"`,
   659→    8000
   660→  );
   661→  if (heartbleedCheck) {
   662→    findings.push({
   663→      title: 'Heartbleed Vulnerability Possible (CVE-2014-0160)',
   664→      severity: 'critical', category: 'ssl-vuln',
   665→      description: 'Server supports heartbeat extension. If running vulnerable OpenSSL (1.0.1-1.0.1f), up to 64KB of server memory can be leaked per request — including private keys, passwords, session tokens.',
   666→      evidence: `Heartbeat extension: ${heartbleedCheck.trim()}`,
   667→      asset: `${domain}:443`,
   668→    });
   669→  }
   670→
   671→  // Certificate issues
   672→  const certDates = await run(
   673→    `echo | openssl s_client -connect ${domain}:443 -servername ${domain} 2>/dev/null | openssl x509 -noout -dates 2>/dev/null`,
   674→    10000
   675→  );
   676→  if (certDates) {
   677→    const notAfter = certDates.match(/notAfter=(.+)/);
   678→    if (notAfter) {
   679→      const expiry = new Date(notAfter[1].trim());
   680→      const now = new Date();
   681→      const daysLeft = Math.floor((expiry.getTime() - now.getTime()) / 86400000);
   682→      if (daysLeft < 0) {
   683→        findings.push({
   684→          title: 'CRITICAL: SSL Certificate EXPIRED',
   685→          severity: 'critical', category: 'ssl-vuln',
   686→          description: `Certificate expired ${Math.abs(daysLeft)} days ago. Visitors see security warnings. Connections may be intercepted.`,
   687→          evidence: `notAfter: ${notAfter[1].trim()} | Expired ${Math.abs(daysLeft)} days ago`,
   688→          asset: `${domain}:443`,
   689→        });
   690→      } else if (daysLeft < 30) {
   691→        findings.push({
   692→          title: `SSL Certificate Expiring in ${daysLeft} Days`,
   693→          severity: 'high', category: 'ssl-vuln',
   694→          description: `Certificate expires soon. Service disruption and security warnings imminent.`,
   695→          evidence: `notAfter: ${notAfter[1].trim()} | ${daysLeft} days remaining`,
   696→          asset: `${domain}:443`,
   697→        });
   698→      }
   699→    }
   700→  }
   701→
   702→  // OCSP stapling
   703→  const ocsp = await run(
   704→    `echo | openssl s_client -connect ${domain}:443 -servername ${domain} -status 2>/dev/null | grep -i "OCSP"`,
   705→    8000
   706→  );
   707→  if (!ocsp) {
   708→    findings.push({
   709→      title: 'OCSP Stapling Not Enabled',
   710→      severity: 'low', category: 'ssl-vuln',
   711→      description: 'Without OCSP stapling, browsers must contact CA separately to verify certificate revocation, adding latency and privacy exposure.',
   712→      evidence: 'No OCSP response in handshake',
   713→      asset: `${domain}:443`,
   714→    });
   715→  }
   716→
   717→  // Perfect Forward Secrecy
   718→  const pfsCheck = await run(
   719→    `echo | openssl s_client -connect ${domain}:443 -servername ${domain} 2>/dev/null | grep -iE "ECDHE-RSA|ECDHE-ECDSA|DHE-RSA|DHE-DSS"`,
   720→    8000
   721→  );
   722→  if (!pfsCheck) {
   723→    findings.push({
   724→      title: 'Perfect Forward Secrecy — Not Supported',
   725→      severity: 'medium', category: 'ssl-vuln',
   726→      description: 'Without PFS, compromise of server private key allows decryption of ALL past sessions captured via passive surveillance.',
   727→      evidence: 'No ECDHE/DHE cipher negotiated',
   728→      asset: `${domain}:443`,
   729→    });
   730→  }
   731→
   732→  return findings;
   733→}
   734→
   735→// ══════════════════════════════════════════════════════════════════════════════
   736→// 4. DNS VULNERABILITY ASSESSMENT
   737→// ══════════════════════════════════════════════════════════════════════════════
   738→
   739→async function scanDNSVulns(domain: string): Promise<Finding[]> {
   740→  const findings: Finding[] = [];
   741→
   742→  // Zone transfer (AXFR) attempt
   743→  const nsRecords = await run(`dig +short +time=3 ${domain} NS`, 5000);
   744→  const nsList = nsRecords.split('\n').filter(Boolean);
   745→  for (const ns of nsList.slice(0, 3)) {
   746→    const axfrResult = await run(`dig axfr ${domain} @${ns} +time=5 +tries=1 2>/dev/null`, 7000);
   747→    if (axfrResult && axfrResult.includes('ANSWER SECTION') && axfrResult.split('\n').length > 5) {
   748→      findings.push({
   749→        title: 'CRITICAL: DNS Zone Transfer (AXFR) Successful',
   750→        severity: 'critical', category: 'dns-vuln',
   751→        description: `Full zone transfer from ${ns} revealed complete DNS records including internal hostnames, IP addresses, and service records. Complete infrastructure mapping possible.`,
   752→        evidence: `dig axfr ${domain} @${ns} → ${axfrResult.split('\n').length} records transferred`,
   753→        asset: ns,
   754→      });
   755→      break;
   756→    }
   757→  }
   758→
   759→  // Subdomain takeover detection (CNAME to dead services)
   760→  const takeoverSubs = ['staging', 'dev', 'blog', 'cdn', 'assets', 'static', 'mail', 'api', 'app', 'test'];
   761→  const takeoverCandidates: string[] = [];
   762→  for (const sub of takeoverSubs) {
   763→    const cname = await run(`dig +short +time=2 ${sub}.${domain} CNAME`, 4000);
   764→    if (cname) {
   765→      const deadServices = ['herokuapp.com', 'cloudfront.net', 'github.io', 's3.amazonaws.com',
   766→        'shopify.com', 'fastly.net', 'cloudwaysapps.com', 'pantheon.io'];
   767→      for (const dead of deadServices) {
   768→        if (cname.toLowerCase().includes(dead)) {
   769→          const subExists = await run(`curl -sI --max-time 4 https://${sub}.${domain} 2>/dev/null | head -1`, 6000);
   770→          if (subExists && !subExists.includes('200') && !subExists.includes('301') && !subExists.includes('302')) {
   771→            takeoverCandidates.push(`${sub}.${domain} → CNAME: ${cname} (${dead})`);
   772→          }
   773→        }
   774→      }
   775→    }
   776→  }
   777→  if (takeoverCandidates.length > 0) {
   778→    findings.push({
   779→      title: `Subdomain Takeover Risk — ${takeoverCandidates.length} candidate(s)`,
   780→      severity: 'high', category: 'dns-vuln',
   781→      description: `Subdomains point to third-party services via CNAME but may not be properly provisioned. If the third-party resource is deleted, an attacker can claim the subdomain and serve malicious content.`,
   782→      evidence: takeoverCandidates.join(' | '),
   783→      asset: domain,
   784→    });
   785→  }
   786→
   787→  // DNS cache snooping
   788→  const cacheResult = await run(`dig +time=3 ${domain} @8.8.8.8 ANY +noall +answer 2>/dev/null`, 5000);
   789→  if (cacheResult && cacheResult.split('\n').length > 8) {
   790→    findings.push({
   791→      title: 'DNS Cache Snooping Possible — Excessive Records Visible',
   792→      severity: 'medium', category: 'dns-vuln',
   793→      description: `Recursive DNS query reveals extensive record types. DNS cache can be probed to determine which domains are recently visited by the resolver's clients.`,
   794→      evidence: `dig ANY ${domain} → ${cacheResult.split('\n').length} records from cache`,
   795→      asset: domain,
   796→    });
   797→  }
   798→
   799→  // DNS rebinding protection check
   800→  const dnsRebindHeaders = await run(`curl -sI --max-time 5 https://${domain} 2>/dev/null | grep -iE "x-dns-prefetch|x-content-type|dns-prefetch"`, 7000);
   801→  if (!dnsRebindHeaders) {
   802→    findings.push({
   803→      title: 'DNS Rebinding — No Protection Detected',
   804→      severity: 'medium', category: 'dns-vuln',
   805→      description: 'No DNS rebinding protection headers. Attackers can bypass Same-Origin Policy by controlling DNS to oscillate between their server and target IP.',
   806→      evidence: 'No DNS rebinding protection headers found',
   807→      asset: domain,
   808→    });
   809→  }
   810→
   811→  return findings;
   812→}

   814→// ══════════════════════════════════════════════════════════════════════════════
   815→// MAIN POST HANDLER
   816→// ══════════════════════════════════════════════════════════════════════════════
   817→
   818→export async function POST(request: NextRequest) {
   819→  const { allowed } = checkRateLimit(request.headers.get('x-forwarded-for') || 'unknown', 30, 60000);
   820→  if (!allowed) return NextResponse.json({ error: 'Rate limit exceeded' }, { status: 429 });
   821→
   822→  try {
   823→    const body = await request.json();
   824→    const target = body.target?.toString().trim().replace(/^https?:\/\//, '').replace(/\/.*$/, '');
   825→    const sanitized = sanitizeDomain(target);
   826→    if (!sanitized) {
   827→      return NextResponse.json({ error: 'Invalid target domain' }, { status: 400 });
   828→    }
   829→    if (isBlockedDomain(sanitized)) {
   830→      return NextResponse.json({ error: 'Target domain is not permitted' }, { status: 403 });
   831→    }
   832→
   833→    // Resolve IP first
   834→    const ip = await run(`dig +short +time=3 +tries=1 ${sanitized} A`, 5000);
   835→    const mainIp = ip?.split('\n')[0]?.trim() || null;
   836→    if (mainIp && isPrivateIP(mainIp)) {
   837→      return NextResponse.json({ error: 'Target resolves to a private or reserved IP address' }, { status: 403 });
   838→    }
   839→
   840→    // Run all modules in parallel
   841→    const [
   842→      bannerResult,
   843→      httpVulns,
   844→      sslVulns,
   845→      dnsVulns,
   846→    ] = await Promise.all([
   847→      bannerGrab(sanitized, mainIp),
   848→      scanHTTPVulns(sanitized),
   849→      scanSSLVulns(sanitized),
   850→      scanDNSVulns(sanitized),
   851→    ]);
   852→
   853→    // Build response
   854→    const allFindings = [...httpVulns, ...sslVulns, ...dnsVulns];
   855→    const c = allFindings.filter(f => f.severity === 'critical').length;
   856→    const h = allFindings.filter(f => f.severity === 'high').length;
   857→    const m = allFindings.filter(f => f.severity === 'medium').length;
   858→    const exploitSummary = {
   859→      totalCVEs: bannerResult.cveMatches.length,
   860→      criticalCVEs: bannerResult.cveMatches.filter(v => v.severity === 'critical').length,
   861→      weaponized: bannerResult.cveMatches.filter(v => v.exploitAvailable === 'weaponized').length,
   862→      theoretical: bannerResult.cveMatches.filter(v => v.exploitAvailable === 'theoretical').length,
   863→      avgCVSS: bannerResult.cveMatches.length > 0
   864→        ? Math.round(bannerResult.cveMatches.reduce((a, b) => a + b.cvss, 0) / bannerResult.cveMatches.length * 10) / 10
   865→        : 0,
   866→    };
   867→
   868→    return applySecurityHeaders(NextResponse.json({
   869→      success: true,
   870→      scan: {
   871→        target,
   872→        ip: mainIp,
   873→        cveFindings: bannerResult.cveMatches.map(v => ({
   874→          cve: v.id, title: v.title, cvss: v.cvss, severity: v.severity,
   875→          affected: v.affected.join(', '), evidence: `Version pattern match via banner grab`,
   876→          remediation: v.remediation, exploitAvailable: v.exploitAvailable, epss: v.epss,
   877→        })),
   878→        httpVulns: httpVulns.map(f => ({
   879→          ...f, proof: f.evidence,
   880→        })),
   881→        sslVulns: sslVulns.map(f => ({
   882→          ...f, risk: f.severity,
   883→        })),
   884→        dnsVulns,
   885→        banners: bannerResult.banners,
   886→        exploitSummary,
   887→        attackSurface: {
   888→          score: Math.min(100, c * 25 + h * 15 + m * 8 + exploitSummary.weaponized * 20),
   889→          vectors: [
   890→            ...(bannerResult.cveMatches.length > 0 ? ['Known CVE Exploits'] : []),
   891→            ...(httpVulns.some(f => f.severity === 'critical') ? ['Web Application Exploits'] : []),
   892→            ...(sslVulns.some(f => f.severity === 'critical') ? ['Cryptographic Attacks'] : []),
   893→            ...(dnsVulns.some(f => f.severity === 'critical') ? ['DNS Infrastructure Attacks'] : []),
   894→            ...(exploitSummary.weaponized > 0 ? ['Weaponized Exploits Available'] : []),
   895→          ],
   896→        },
   897→      },
   898→    }));
   899→  } catch (error) {
   900→    return safeErrorResponse(error, 500, 'vuln-scan');
   901→  }
   902→}
