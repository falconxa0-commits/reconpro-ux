    74→// ── Helpers ─────────────────────────────────────────────────────────────────
    75→
    76→function getInitials(name: string): string {
    77→  return name
    78→    .split(' ')
    79→    .map((n) => n[0])
    80→    .join('')
    81→    .toUpperCase();
    82→}
    83→
    84→function getRoleColor(role: string): string {
    85→  switch (role) {
    86→    case 'Admin':
    87→      return 'bg-[rgba(168,85,247,0.15)] text-[#888888] border-[rgba(168,85,247,0.3)]';
    88→    case 'Security Lead':
    89→      return 'bg-[rgba(52,211,153,0.15)] text-[#00ff88] border-[rgba(52,211,153,0.3)]';
    90→    case 'Analyst':
    91→      return 'bg-[rgba(88,166,255,0.15)] text-[#44aaff] border-[rgba(88,166,255,0.3)]';
    92→    case 'Viewer':
    93→      return 'bg-[rgba(139,148,158,0.15)] text-[#444444] border-[rgba(139,148,158,0.3)]';
    94→    default:
    95→      return 'bg-[rgba(139,148,158,0.15)] text-[#444444] border-[rgba(139,148,158,0.3)]';
    96→  }
    97→}
    98→
    99→function getStatusColor(status: string): string {
   100→  switch (status) {
   101→    case 'online':
   102→      return 'bg-[#00ff88]';
   103→    case 'away':
   104→      return 'bg-[#d29922]';
   105→    case 'offline':
   106→      return 'bg-[#333333]';
   107→    default:
   108→      return 'bg-[#333333]';
   109→  }
   110→}
   111→
   112→function getAvatarBg(name: string): string {
   113→  const colors = [
   114→    'rgba(52,211,153,0.15)',
   115→    'rgba(88,166,255,0.15)',
   116→    'rgba(168,85,247,0.15)',
   117→    'rgba(244,63,94,0.15)',
   118→    'rgba(210,153,34,0.15)',
   119→    'rgba(121,192,255,0.15)',
   120→    'rgba(255,123,114,0.15)',
   121→    'rgba(187,128,220,0.15)',
   122→  ];
   123→  const idx = name.charCodeAt(0) % colors.length;
   124→  return colors[idx];
   125→}
   126→
   127→function getAvatarTextColor(name: string): string {
   128→  const colors = ['#00ff88', '#44aaff', '#888888', '#ff3355', '#d29922', '#79c0ff', '#ff7b72', '#bb80d4'];
   129→  const idx = name.charCodeAt(0) % colors.length;
   130→  return colors[idx];
   131→}

   154→
   155→export function TeamManagement() {
   156→  const authHeaders = useAuthHeaders();
   157→  const [members, setMembers] = useState<Member[]>([]);
   158→  const [teams, setTeams] = useState<Team[]>([]);
   159→  const [loading, setLoading] = useState(true);
   160→  const [error, setError] = useState('');
   161→  const [searchQuery, setSearchQuery] = useState('');
   162→  const [inviteOpen, setInviteOpen] = useState(false);
   163→  const [inviteEmail, setInviteEmail] = useState('');
   164→  const [inviteName, setInviteName] = useState('');
   165→  const [inviteRole, setInviteRole] = useState('Viewer');
   166→  const [copiedId, setCopiedId] = useState<string | null>(null);
   167→
   168→  // Team add dialog
   169→  const [teamOpen, setTeamOpen] = useState(false);
   170→  const [teamName, setTeamName] = useState('');
   171→  const [teamDesc, setTeamDesc] = useState('');
   172→
   173→  // Role edit dialog
   174→  const [roleEditOpen, setRoleEditOpen] = useState(false);
   175→  const [roleEditId, setRoleEditId] = useState('');
   176→  const [roleEditValue, setRoleEditValue] = useState('');
   177→
   178→  const fetchMembers = useCallback(async () => {
   179→    try {
   180→      const res = await fetch('/api/members', { headers: authHeaders });
   181→      if (!res.ok) throw new Error(`HTTP ${res.status}`);
   182→      const json = await res.json();
   183→      setMembers(json.members || []);
   184→    } catch (err) {
   185→      console.error('Failed to fetch members:', err);
   186→      setError('Failed to load team data. Please try again.');
   187→    }
   188→  }, [authHeaders]);
   189→
   190→  const fetchTeams = useCallback(async () => {
   191→    try {
   192→      const res = await fetch('/api/teams', { headers: authHeaders });
   193→      if (!res.ok) throw new Error(`HTTP ${res.status}`);
   194→      const json = await res.json();
   195→      setTeams(json.teams || []);
   196→    } catch (err) {
   197→      console.error('Failed to fetch teams:', err);
   198→      setError('Failed to load team data. Please try again.');
   199→    }
   200→  }, [authHeaders]);
   201→
   202→  const fetchAllRef = useCallback(async () => {
   203→    setError('');
   204→    await Promise.all([fetchMembers(), fetchTeams()]);

   207→  useEffect(() => {
   208→    let cancelled = false;
   209→    (async () => {
   210→      try {
   211→        await fetchAllRef();
   212→      } finally {
   213→        if (!cancelled) setLoading(false);
   214→      }
   215→    })();
   216→    return () => { cancelled = true; };
   217→  }, [fetchAllRef]);
   218→
   219→  const filteredMembers = members.filter(
   220→    (m) =>
   221→      m.name.toLowerCase().includes(searchQuery.toLowerCase()) ||
   222→      m.email.toLowerCase().includes(searchQuery.toLowerCase()) ||
   223→      m.role.toLowerCase().includes(searchQuery.toLowerCase())
   224→  );
   225→
   226→  const adminCount = members.filter((m) => m.role === 'Admin').length;
   227→  const onlineCount = members.filter((m) => m.status === 'online').length;
   228→
   229→  const handleInvite = async () => {
   230→    if (!inviteName || !inviteEmail) return;
   231→    try {
   232→      await fetch('/api/members', {
   233→        method: 'POST',
   234→        headers: { 'Content-Type': 'application/json', ...authHeaders },
   235→        body: JSON.stringify({ name: inviteName, email: inviteEmail, role: inviteRole.toLowerCase().replace(' ', '_') }),
   236→      });
   237→      setInviteOpen(false);
   238→      setInviteEmail('');
   239→      setInviteName('');
   240→      setInviteRole('Viewer');
   241→      await fetchMembers();
   242→    } catch (err) {
   243→      console.error('Failed to invite member:', err);
   244→    }
   245→  };
   246→
   247→  const handleRoleChange = async () => {
   248→    if (!roleEditId || !roleEditValue) return;
   249→    try {
   250→      await fetch('/api/members', {
   251→        method: 'PATCH',
   252→        headers: { 'Content-Type': 'application/json', ...authHeaders },
   253→        body: JSON.stringify({ id: roleEditId, role: roleEditValue.toLowerCase().replace(' ', '_') }),
   254→      });
   255→      setRoleEditOpen(false);
   256→      await fetchMembers();
   257→    } catch (err) {
   258→      console.error('Failed to update role:', err);
   259→    }
   260→  };
   261→
   262→  const handleDeleteMember = async (id: string) => {
   263→    try {
   264→      await fetch(`/api/members?id=${id}`, { method: 'DELETE' });
   265→      await fetchMembers();
   266→    } catch (err) {
   267→      console.error('Failed to delete member:', err);
   268→    }
   269→  };
   270→
   271→  const handleAddTeam = async () => {
   272→    if (!teamName) return;
   273→    try {
   274→      await fetch('/api/teams', {
   275→        method: 'POST',
   276→        headers: { 'Content-Type': 'application/json', ...authHeaders },
   277→        body: JSON.stringify({ name: teamName, description: teamDesc }),
   278→      });
   279→      setTeamOpen(false);
   280→      setTeamName('');
   281→      setTeamDesc('');
   282→      await fetchTeams();
   283→    } catch (err) {
   284→      console.error('Failed to create team:', err);
   285→    }
   286→  };
   287→
   288→  const handleDeleteTeam = async (id: string) => {
   289→    try {
   290→      await fetch(`/api/teams?id=${id}`, { method: 'DELETE' });
   291→      await fetchTeams();
   292→    } catch (err) {
   293→      console.error('Failed to delete team:', err);
   294→    }
   295→  };
   296→
   297→  const handleCopyId = (id: string) => {
   298→    navigator.clipboard.writeText(id);
   299→    setCopiedId(id);
   300→    setTimeout(() => setCopiedId(null), 2000);
   301→  };
   302→
   303→  const openRoleEdit = (member: Member) => {
   304→    setRoleEditId(member.id);
   305→    setRoleEditValue(member.role);
   306→    setRoleEditOpen(true);
   307→  };
   308→
   309→  if (loading) {
   310→    return (
   311→      <div className="w-full flex items-center justify-center py-20">
   312→        <p className="text-[#444444]">Loading...</p>
   313→      </div>
   314→    );
   315→  }
   316→
   317→  if (error) {
   318→    return (
   319→      <div className="w-full flex flex-col items-center justify-center gap-3 py-20">
   320→        <AlertCircle className="h-8 w-8 text-red-400" />
   321→        <p className="text-sm text-red-400">{error}</p>
   322→        <Button variant="outline" size="sm" onClick={fetchAllRef} className="border-zinc-700 text-white hover:bg-zinc-800">
   323→          <RefreshCw className="mr-2 h-3.5 w-3.5" />
   324→          Retry
   325→        </Button>
   326→      </div>
   327→    );
   328→  }
   329→
   330→  return (
   331→    <motion.div
   332→      variants={containerVariants}
   333→      initial="hidden"
   334→      animate="visible"
   335→      className="w-full space-y-6"
   336→    >
   337→      {/* ── Header ──────────────────────────────────────────────────────── */}
   338→      <motion.div variants={itemVariants} className="flex flex-col sm:flex-row sm:items-center justify-between gap-4">
   339→        <div className="flex items-center gap-3">
   340→          <div className="p-2 rounded-lg bg-[rgba(52,211,153,0.1)] border border-[rgba(52,211,153,0.2)]">
   341→            <Users className="w-5 h-5 text-[#00ff88]" />
   342→          </div>
   343→          <div>
   344→            <h2 className="text-xl font-bold text-[#f0f0f0]">Team Management</h2>
   345→            <p className="text-sm text-[#444444]">Manage members, roles, and team assignments</p>
   346→          </div>
   347→        </div>
   348→        <Dialog open={inviteOpen} onOpenChange={setInviteOpen}>
   349→          <DialogTrigger asChild>
   350→            <Button className="bg-[#00ff88] hover:bg-[#00cc6a] text-[#0a0d14] font-semibold gap-2">
   351→              <UserPlus className="w-4 h-4" />
   352→              Invite Member
   353→            </Button>
   354→          </DialogTrigger>
   355→          <DialogContent className="bg-[#080b14] border-[#21262d] text-[#f0f0f0]">
   356→            <DialogHeader>
   357→              <DialogTitle className="text-[#f0f0f0]">Invite New Member</DialogTitle>
   358→              <DialogDescription className="text-[#444444]">
   359→                Send an invitation to add a new team member.
   360→              </DialogDescription>
   361→            </DialogHeader>
   362→            <div className="space-y-4 py-4">
   363→              <div className="space-y-2">
   364→                <label className="text-sm font-medium text-[#f0f0f0]">Name</label>
   365→                <Input
   366→                  placeholder="John Doe"
   367→                  value={inviteName}
   368→                  onChange={(e) => setInviteName(e.target.value)}
   369→                  className="bg-[#050710] border-[#21262d] text-[#f0f0f0] placeholder:text-[#333333] focus:border-[#00ff88]"
   370→                />
   371→              </div>
   372→              <div className="space-y-2">
   373→                <label className="text-sm font-medium text-[#f0f0f0]">Email Address</label>
   374→                <Input
   375→                  placeholder="colleague@company.com"
   376→                  value={inviteEmail}
   377→                  onChange={(e) => setInviteEmail(e.target.value)}
   378→                  className="bg-[#050710] border-[#21262d] text-[#f0f0f0] placeholder:text-[#333333] focus:border-[#00ff88]"
   379→                />
   380→              </div>
   381→              <div className="space-y-2">
   382→                <label className="text-sm font-medium text-[#f0f0f0]">Role</label>
   383→                <div className="grid grid-cols-2 gap-2">
   384→                  {['Admin', 'Security Lead', 'Analyst', 'Viewer'].map((r) => (
   385→                    <button
   386→                      key={r}
   387→                      onClick={() => setInviteRole(r)}
   388→                      className={`px-3 py-2 rounded-lg text-sm font-medium border transition-all ${
   389→                        inviteRole === r
   390→                          ? 'bg-[rgba(52,211,153,0.15)] border-[rgba(52,211,153,0.4)] text-[#00ff88]'
   391→                          : 'bg-[#050710] border-[#21262d] text-[#444444] hover:border-[#30363d]'
   392→                      }`}
   393→                    >
   394→                      {r}
   395→                    </button>
   396→                  ))}
   397→                </div>
   398→              </div>
   399→            </div>
   400→            <DialogFooter>
   401→              <Button variant="outline" onClick={() => setInviteOpen(false)} className="border-[#21262d] text-[#444444] hover:bg-[#050710]">
   402→                Cancel
   403→              </Button>
   404→              <Button onClick={handleInvite} className="bg-[#00ff88] hover:bg-[#00cc6a] text-[#0a0d14] font-semibold">
   405→                Send Invitation
   406→              </Button>
   407→            </DialogFooter>
   408→          </DialogContent>
   409→        </Dialog>
   410→      </motion.div>
   411→
   412→      {/* ── Stats Row ───────────────────────────────────────────────────── */}
   413→      <motion.div variants={itemVariants} className="grid grid-cols-2 lg:grid-cols-4 gap-4">
   414→        {[
   415→          { label: 'Total Members', value: members.length, icon: Users, color: '#00ff88' },
   416→          { label: 'Active Teams', value: teams.length, icon: Shield, color: '#44aaff' },
   417→          { label: 'Pending Invites', value: 0, icon: AlertCircle, color: '#d29922' },
   418→          { label: 'Admins', value: adminCount, icon: Crown, color: '#888888' },
   419→        ].map((stat) => (
   420→          <motion.div
   421→            key={stat.label}
   422→            whileHover={cardHover}
   423→            className="rounded-xl border border-[#21262d] bg-[#080b14] p-4 transition-shadow hover:shadow-lg"
   424→            style={{ boxShadow: '0 0 0px transparent' }}
   425→          >
   426→            <div className="flex items-center justify-between mb-3">
   427→              <span className="text-xs font-medium text-[#444444] uppercase tracking-wider">{stat.label}</span>
   428→              <stat.icon className="w-4 h-4" style={{ color: stat.color }} />
   429→            </div>
   430→            <p className="text-2xl font-bold text-[#f0f0f0]">{stat.value}</p>
   431→          </motion.div>
   432→        ))}
   433→      </motion.div>
   434→
   435→      {/* ── Tabs ────────────────────────────────────────────────────────── */}
   436→      <motion.div variants={itemVariants}>
   437→        <Tabs defaultValue="members" className="w-full">
   438→          <TabsList className="bg-[#080b14] border border-[#21262d]">
   439→            <TabsTrigger
   440→              value="members"
   441→              className="data-[state=active]:bg-[rgba(52,211,153,0.1)] data-[state=active]:text-[#00ff88] text-[#444444]"
   442→            >
   443→              <Users className="w-4 h-4 mr-1.5" />
   444→              Members
   445→            </TabsTrigger>
   446→            <TabsTrigger
   447→              value="teams"
   448→              className="data-[state=active]:bg-[rgba(52,211,153,0.1)] data-[state=active]:text-[#00ff88] text-[#444444]"
   449→            >
   450→              <Shield className="w-4 h-4 mr-1.5" />
   451→              Teams
   452→            </TabsTrigger>
   453→          </TabsList>
   454→
   455→          {/* ── Members Tab ─────────────────────────────────────────────── */}
   456→          <TabsContent value="members" className="mt-4">
   457→            <motion.div
   458→              className="rounded-xl border border-[#21262d] bg-[#080b14] overflow-hidden"
   459→            >
   460→              {/* Search bar */}
   461→              <div className="p-4 border-b border-[#21262d]">
   462→                <div className="relative">
   463→                  <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-[#333333]" />
   464→                  <Input
   465→                    placeholder="Search members by name, email, or role..."
   466→                    value={searchQuery}
   467→                    onChange={(e) => setSearchQuery(e.target.value)}
   468→                    className="pl-9 bg-[#050710] border-[#21262d] text-[#f0f0f0] placeholder:text-[#333333] focus:border-[#00ff88] h-9"
   469→                  />
   470→                </div>
   471→              </div>
   472→
   473→              {/* Table */}
   474→              <div className="max-h-[400px] overflow-y-auto">
   475→                {members.length === 0 ? (
   476→                  <div className="px-4 py-8 text-center">
   477→                    <Users className="w-8 h-8 text-[#333333] mx-auto mb-3" />
   478→                    <p className="text-sm text-[#444444]">No members yet. Click &quot;Invite Member&quot; to add your first team member.</p>
   479→                  </div>
   480→                ) : (
   481→                <Table>
   482→                  <TableHeader>
   483→                    <TableRow className="border-b border-[#21262d] hover:bg-transparent">
   484→                      <TableHead className="text-[#444444] font-medium text-xs uppercase tracking-wider">Member</TableHead>
   485→                      <TableHead className="text-[#444444] font-medium text-xs uppercase tracking-wider">Role</TableHead>
   486→                      <TableHead className="text-[#444444] font-medium text-xs uppercase tracking-wider hidden md:table-cell">Last Active</TableHead>
   487→                      <TableHead className="text-[#444444] font-medium text-xs uppercase tracking-wider hidden lg:table-cell">Status</TableHead>
   488→                      <TableHead className="text-[#444444] font-medium text-xs uppercase tracking-wider text-right">Actions</TableHead>
   489→                    </TableRow>
   490→                  </TableHeader>
   491→                  <TableBody>
   492→                    <AnimatePresence>
   493→                      {filteredMembers.map((member, idx) => (
   494→                        <motion.tr

   496→                          initial={{ opacity: 0, x: -10 }}
   497→                          animate={{ opacity: 1, x: 0 }}
   498→                          transition={{ delay: idx * 0.04, duration: 0.3 }}
   499→                          className="border-b border-[#161b22] hover:bg-[rgba(52,211,153,0.03)] transition-colors"
   500→                        >
   501→                          <TableCell className="py-3">
   502→                            <div className="flex items-center gap-3">
   503→                              <div className="relative">
   504→                                <div
   505→                                  className="w-9 h-9 rounded-full flex items-center justify-center text-sm font-bold shrink-0"
   506→                                  style={{
   507→                                    backgroundColor: getAvatarBg(member.name),
   508→                                    color: getAvatarTextColor(member.name),
   509→                                  }}
   510→                                >
   511→                                  {getInitials(member.name)}
   512→                                </div>
   513→                                <div
