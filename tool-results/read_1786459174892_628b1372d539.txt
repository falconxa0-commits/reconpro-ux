    43→def _render_summary(result, title: str = "RECONPRO") -> None:
    44→    score = result.total_score
    45→    grade = result.grade
    46→    grade_color = GRADE_COLORS.get(grade, "bold bright_red")
    47→    sc = result.severity_counts
    48→    total = len(result.findings)
    49→    bar_width = 50
    50→    filled = int(score / 100 * bar_width)
    51→    bar = "█" * filled + "░" * (bar_width - filled)
    52→    lines = [
    53→        Text(f"\n  {bar} [bold {grade_color}]{score}/100 ({grade})[/{grade_color}]"),
    54→        Text(f"\n  Target: [cyan]{result.target}[/]"),
    55→        Text(f"  Modules: [dim]{', '.join(result.modules_run)}[/]"),
    56→        Text(""),
    57→        Text(
    58→            f"  Findings: [bold]{total}[/]  "
    59→            f"[bright_red]{sc.get('critical', 0)} critical[/], "
    60→            f"[red]{sc.get('high', 0)} high[/], "
    61→            f"[yellow]{sc.get('medium', 0)} medium[/], "
    62→            f"[green]{sc.get('low', 0)} low[/], "
    63→            f"[dim]{sc.get('info', 0)} info[/]",
    64→        ),
    65→    ]
    66→    if result.vibesec_score is not None:
    67→        lines.append(Text(
    68→            f"\n  VibeSec Score: [bold bright_green]{result.vibesec_score}/100 ({result.vibesec_grade})[/]"
    69→        ))
    70→    console.print(Panel(Group(*lines), border_style=grade_color,
    71→                         title=f"[bold]{title}[/bold]", title_align="left", padding=(1, 2)))

    74→def _render_findings_table(result, show_remediation: bool = False) -> None:
    75→    if not result.findings:
    76→        console.print("\n  [bright_green]No findings. Target is clean.[/]")
    77→        return
    78→    table = Table(title=f"Findings — {len(result.findings)} detected",
    79→                  border_style="bright_white", header_style="bold bright_white")
    80→    table.add_column("Severity", style="bold", width=10)
    81→    table.add_column("Module", style="cyan", width=12)
    82→    table.add_column("Category", style="magenta", width=18)
    83→    table.add_column("Finding", style="white")
    84→    table.add_column("Pts", style="yellow", width=5)
    85→    if show_remediation:
    86→        table.add_column("Fix", style="bright_green", width=50)
    87→    for f in result.findings[:80]:
    88→        sev = f.get("severity", "info")
    89→        color = SEV_COLORS.get(sev, "white")
    90→        row = [
    91→            f"[{color}]{sev.upper()}[/{color}]",
    92→            f.get("module", ""), f.get("category", ""),
    93→            f.get("title", "")[:70], str(f.get("points_deducted", 0)),
    94→        ]
    95→        if show_remediation:
    96→            row.append(f.get("remediation", "")[:50])
    97→        table.add_row(*row)
    98→    console.print(table)

   101→def _render_module_breakdown(result) -> None:
   102→    if not result.module_results:
   103→        return
   104→    t = Table(title="Module Breakdown", border_style="dim", header_style="bold dim")
   105→    t.add_column("Module", style="bold", width=16)
   106→    t.add_column("Findings", style="white", width=10)
   107→    t.add_column("Critical", style="bright_red", width=10)
   108→    t.add_column("High", style="red", width=8)
   109→    t.add_column("Medium", style="yellow", width=8)
   110→    for mod_id, mod_data in result.module_results.items():
   111→        findings = mod_data.get("findings", [])
   112→        t.add_row(mod_id.upper(), str(len(findings)),
   113→                   str(sum(1 for f in findings if f.get("severity") == "critical")),
   114→                   str(sum(1 for f in findings if f.get("severity") == "high")),
   115→                   str(sum(1 for f in findings if f.get("severity") == "medium")))
   116→    console.print(t)

   127→def _render_remediations(result) -> None:
   128→    actionable = [f for f in result.findings
   129→                  if f.get("remediation") and f.get("severity") not in ("info",)
   130→                  and f.get("points_deducted", 0) > 0]
   131→    if not actionable:
   132→        console.print("\n  [bright_green]No actionable fixes needed.[/]")
   133→        return
   134→    console.print("\n[bold bright_white]  Fix Commands:[/bold bright_white]\n")
   135→    seen = set()
   136→    for f in actionable:
   137→        fix = f.get("remediation", "").strip()
   138→        sev = f.get("severity", "info")
   139→        color = SEV_COLORS.get(sev, "white")
   140→        if fix and fix not in seen:
   141→            seen.add(fix)
   142→            console.print(f"  [{color}]{sev.upper():8}[/{color}]  {fix}")
   143→    console.print()

   146→def _output_result(result, args, title: str = "RECONPRO", show_remediation: bool = False) -> None:
   147→    json_output = getattr(args, "json_output", False)
   148→    output_file = getattr(args, "output_file", None)
   149→    if json_output:
   150→        text = json.dumps(result.to_dict(), indent=2, default=str)
   151→        if output_file:
   152→            with open(output_file, "w") as fp:
   153→                fp.write(text)
   154→            console.print(f"  [green]JSON saved: [cyan]{output_file}[/][/]")
   155→        else:
   156→            print(text)
   157→    else:
   158→        _render_summary(result, title=title)
   159→        console.print()
   160→        _render_module_breakdown(result)
   161→        console.print()
   162→        _render_findings_table(result, show_remediation=show_remediation)
   163→        if show_remediation:
   164→            _render_remediations(result)
   165→        _render_badge(result)
   166→        if output_file:
   167→            with open(output_file, "w") as fp:
   168→                fp.write(json.dumps(result.to_dict(), indent=2, default=str))
   169→            console.print(f"\n  [green]JSON saved: [cyan]{output_file}[/][/]")
   170→        # Auto-save to history
   171→        save_scan(result.to_dict())
   172→    console.print()

   186→def _spinner_wrap(msg: str, fn, *a, _json_mode: bool | None = None, **kw):
   187→    """Run a function with a spinner (suppressed in JSON mode)."""
   188→    if _json_mode is None:
   189→        _json_mode = getattr(_cli_args, 'json_output', False) if _cli_args else False
   190→    if _json_mode:
   191→        return fn(*a, **kw)
   192→    with Progress(SpinnerColumn(), TextColumn("[bold]{task.description}[/]"),
   193→                   TimeElapsedColumn(), console=console) as progress:
   194→        task = progress.add_task(msg, total=None)
   195→        result = fn(*a, **kw)
   196→        progress.update(task, completed=True)
   197→    return result

   200→def _print_findings(findings, args, title: str = "SCAN") -> None:
   201→    """Print findings from a module run, respecting JSON mode."""
   202→    if getattr(args, 'json_output', False):
   203→        out = []
   204→        for f in findings:
   205→            out.append({
   206→                "title": f.title,
   207→                "severity": f.severity,
   208→                "category": f.category,
   209→                "module": f.module,
   210→                "description": f.description,
   211→                "evidence": f.evidence,
   212→                "asset": f.asset,
   213→                "points_deducted": f.points_deducted,
   214→                "remediation": f.remediation,
   215→            })
   216→        console.print_json(json.dumps(out, indent=2, default=str))
   217→    else:
   218→        if not findings:
   219→            console.print("  [dim]No findings.[/]")
   220→        else:
   221→            for f in findings:
   222→                c = SEV_COLORS.get(f.severity, "white")
   223→                console.print(f"  [{c}]{f.severity.upper():8}[/{c}]  {f.title}")
   224→                if f.evidence:
   225→                    console.print(f"       [dim]{f.evidence[:200]}[/]")
   226→            console.print(f"  [bold]{len(findings)} finding(s).[/]")
   227→    console.print()
