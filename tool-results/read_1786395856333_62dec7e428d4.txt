   178→# Directed Graph Implementation (pure Python)
   179→# ═══════════════════════════════════════════════════════════════════════════
   180→
   181→
   182→class DiGraph:
   183→    """Lightweight directed graph using adjacency lists.
   184→
   185→    Thread-safe for single-writer scenarios. Supports standard graph operations
   186→    needed for attack path analysis.
   187→    """
   188→
   189→    def __init__(self) -> None:
   190→        self._nodes: Dict[str, GraphNode] = {}
   191→        self._adj: Dict[str, Dict[str, GraphEdge]] = defaultdict(dict)
   192→        self._radj: Dict[str, Dict[str, GraphEdge]] = defaultdict(dict)
   193→
   194→    # -- Mutation -------------------------------------------------------------
   195→
   196→    def add_node(self, node: GraphNode) -> None:
   197→        self._nodes[node.id] = node
   198→
   199→    def add_edge(self, edge: GraphEdge) -> None:
   200→        self.add_node(GraphNode(id=edge.source, label=edge.source, node_type="implicit"))
   201→        self.add_node(GraphNode(id=edge.target, label=edge.target, node_type="implicit"))
   202→        self._adj[edge.source][edge.target] = edge
   203→        self._radj[edge.target][edge.source] = edge
   204→
   205→    # -- Queries --------------------------------------------------------------
   206→
   207→    @property
   208→    def nodes(self) -> Dict[str, GraphNode]:
   209→        return dict(self._nodes)
   210→
   211→    def successors(self, node_id: str) -> List[str]:
   212→        return list(self._adj.get(node_id, {}).keys())
   213→
   214→    def predecessors(self, node_id: str) -> List[str]:
   215→        return list(self._radj.get(node_id, {}).keys())
   216→
   217→    def edges_from(self, node_id: str) -> List[GraphEdge]:
   218→        return list(self._adj.get(node_id, {}).values())
   219→
   220→    def edges_to(self, node_id: str) -> List[GraphEdge]:
   221→        return list(self._radj.get(node_id, {}).values())
   222→
   223→    def get_edge(self, source: str, target: str) -> Optional[GraphEdge]:
   224→        return self._adj.get(source, {}).get(target)
   225→
   226→    def has_node(self, node_id: str) -> bool:
   227→        return node_id in self._nodes
   228→
   229→    def node_count(self) -> int:
   230→        return len(self._nodes)
   231→
   232→    def edge_count(self) -> int:
   233→        return sum(len(targets) for targets in self._adj.values())
   234→
   235→    def all_edges(self) -> List[GraphEdge]:
   236→        edges = []
   237→        for targets in self._adj.values():
   238→            edges.extend(targets.values())
   239→        return edges
   240→
   241→    # -- Traversal ------------------------------------------------------------
   242→
   243→    def bfs_paths(self, start: str, end: str, max_depth: int = 10) -> List[List[str]]:
   244→        """Find all paths from start to end using BFS up to max_depth."""
   245→        if start not in self._nodes or end not in self._nodes:
   246→            return []
   247→        paths: List[List[str]] = []
   248→        queue: deque = deque()
   249→        queue.append(([start], 0))
   250→        while queue:
   251→            path, depth = queue.popleft()
   252→            if depth > max_depth:
   253→                continue
   254→            current = path[-1]
   255→            if current == end and len(path) > 1:
   256→                paths.append(path)
   257→                continue
   258→            for nxt in self.successors(current):
   259→                if nxt not in path:  # Simple cycle prevention
   260→                    queue.append((path + [nxt], depth + 1))
   261→        return paths
   262→
   263→    def all_paths_between(self, start: str, end: str, max_depth: int = 6) -> List[List[str]]:
   264→        """Find all simple paths (DFS) between two nodes."""
   265→        if start not in self._nodes or end not in self._nodes:
   266→            return []
   267→        paths: List[List[str]] = []
   268→        stack: List[Tuple[str, List[str], Set[str]]] = [(start, [start], {start})]
   269→        while stack:
   270→            node, path, visited = stack.pop()
   271→            if len(path) - 1 >= max_depth:
   272→                continue
   273→            for nxt in self.successors(node):
   274→                if nxt == end:
   275→                    paths.append(path + [nxt])
   276→                elif nxt not in visited:
   277→                    stack.append((nxt, path + [nxt], visited | {nxt}))
   278→        return paths
   279→
   280→    def shortest_path(self, start: str, end: str) -> Optional[List[str]]:
   281→        """Find shortest path using BFS."""
   282→        paths = self.bfs_paths(start, end, max_depth=20)
   283→        return min(paths, key=len) if paths else None
   284→
   285→    def ancestors(self, node_id: str, max_depth: int = 10) -> Set[str]:
   286→        """Get all ancestor nodes up to max_depth."""
   287→        visited: Set[str] = set()
   288→        queue = deque([(node_id, 0)])
   289→        while queue:
   290→            n, d = queue.popleft()
   291→            if d >= max_depth:
   292→                continue
   293→            for p in self.predecessors(n):
   294→                if p not in visited:
   295→                    visited.add(p)
   296→                    queue.append((p, d + 1))
   297→        return visited
   298→
   299→    def descendants(self, node_id: str, max_depth: int = 10) -> Set[str]:
   300→        """Get all descendant nodes up to max_depth."""
   301→        visited: Set[str] = set()
   302→        queue = deque([(node_id, 0)])
   303→        while queue:
   304→            n, d = queue.popleft()
   305→            if d >= max_depth:
   306→                continue
   307→            for s in self.successors(n):
   308→                if s not in visited:
   309→                    visited.add(s)
   310→                    queue.append((s, d + 1))
   311→        return visited
   312→
   313→    def in_degree(self, node_id: str) -> int:
   314→        return len(self.predecessors(node_id))
   315→
   316→    def out_degree(self, node_id: str) -> int:
   317→        return len(self.successors(node_id))
   318→
   319→    def betweenness_proxy(self) -> Dict[str, int]:
   320→        """Approximate betweenness centrality by counting shortest-path traversals.
   321→
   322→        For each pair of nodes, count how many shortest paths pass through
   323→        each intermediate node. This is a simplified proxy for betweenness.
   324→        """
   325→        counts: Dict[str, int] = defaultdict(int)
   326→        node_list = list(self._nodes.keys())
   327→        for start in node_list:
   328→            for end in node_list:
   329→                if start == end:
   330→                    continue
   331→                path = self.shortest_path(start, end)
   332→                if path and len(path) > 2:
   333→                    for intermediate in path[1:-1]:
   334→                        counts[intermediate] += 1
   335→        return dict(counts)

   481→# ═══════════════════════════════════════════════════════════════════════════
   482→
   483→
   484→class AttackGraphEngine:
   485→    """Builds and analyzes attack graphs from scan findings.
   486→
   487→    Transforms a flat list of findings into a directed graph of
   488→    relationships, then extracts attack chains, choke points, blast radii,
   489→    high-value assets, and kill chain mappings.
   490→    """
   491→
   492→    MAX_GRAPH_NODES = 5000
   493→    MAX_GRAPH_EDGES = 20000
   494→
   495→    def __init__(self) -> None:
   496→        self.graph = DiGraph()
   497→        self._finding_nodes: Dict[str, str] = {}  # finding hash -> node_id
   498→        self._category_nodes: Dict[str, List[str]] = defaultdict(list)
   499→
   500→    def build_graph(self, findings: List[Dict[str, Any]], target: str = "") -> None:
   501→        """Build the attack graph from scan findings.
   502→
   503→        Creates nodes for each finding and edges based on category relationships.
   504→        Enforces MAX_GRAPH_NODES and MAX_GRAPH_EDGES to prevent memory exhaustion.
   505→        """
   506→        if len(findings) > self.MAX_GRAPH_NODES:
   507→            findings = findings[:self.MAX_GRAPH_NODES]
   508→
   509→        self.graph = DiGraph()
   510→        self._finding_nodes = {}
   511→        self._category_nodes = defaultdict(list)
   512→
   513→        # Create entry point node
   514→        entry_id = "entry:recon"
   515→        self.graph.add_node(GraphNode(
   516→            id=entry_id, node_type="entry_point",
   517→            label=f"Recon: {target or 'target'}",
   518→            severity="info", risk_score=0.0,
   519→            metadata={"phase": "Reconnaissance"},
   520→        ))
   521→
   522→        # Create finding nodes
   523→        for i, f in enumerate(findings):
   524→            cat = _classify_finding(f)
   525→            node_id = f"finding:{i}"
   526→            severity = f.get("severity", "info").lower()
   527→            risk = _SEVERITY_WEIGHTS.get(severity, 1.0)
   528→
   529→            self.graph.add_node(GraphNode(
   530→                id=node_id, node_type="finding",
   531→                label=f.get("title", f"Finding {i}")[:80],
   532→                severity=severity, risk_score=risk,
   533→                metadata={
   534→                    "category": cat,
   535→                    "title": f.get("title", ""),
   536→                    "asset": f.get("asset", ""),
   537→                    "module": f.get("module", ""),
   538→                },
   539→            ))
   540→
   541→            self._finding_nodes[self._finding_hash(f)] = node_id
   542→            self._category_nodes[cat].append(node_id)
   543→
   544→            # Edge from entry to finding
   545→            self.graph.add_edge(GraphEdge(
   546→                source=entry_id, target=node_id,
   547→                edge_type="discovers", weight=0.3, confidence=0.9,
   548→            ))
   549→
   550→        # Create edges based on relationship templates
   551→        self._create_relationship_edges(findings)
   552→
   553→        # Create objective node
   554→        objective_id = "objective:compromise"
   555→        self.graph.add_node(GraphNode(
   556→            id=objective_id, node_type="objective",
   557→            label="Full System Compromise",
   558→            severity="critical", risk_score=10.0,
   559→            metadata={"phase": "Actions on Objectives"},
   560→        ))
   561→
   562→        # Connect high-severity findings to objective
   563→        for node_id in self._category_nodes.get("privilege_escalation", []):
   564→            self.graph.add_edge(GraphEdge(
   565→                source=node_id, target=objective_id,
   566→                edge_type="leads_to", weight=0.9, confidence=0.8,
   567→            ))
   568→        for node_id in self._category_nodes.get("data_exposure", []):
   569→            self.graph.add_edge(GraphEdge(
   570→                source=node_id, target=objective_id,
   571→                edge_type="leads_to", weight=0.7, confidence=0.7,
   572→            ))
   573→
   574→    def analyze(self, findings: List[Dict[str, Any]], target: str = "") -> AttackGraphResult:
   575→        """Full attack graph analysis.
   576→
   577→        Builds the graph, extracts chains, identifies choke points,
   578→        and produces a comprehensive result.
   579→        """
   580→        t0 = time.monotonic()
   581→
   582→        self.build_graph(findings, target)
   583→        result = AttackGraphResult(target=target)
   584→
   585→        # Serialize graph
   586→        result.nodes = [n.to_dict() for n in self.graph.nodes.values()]
   587→        result.edges = [e.to_dict() for e in self.graph.all_edges()]
   588→
   589→        # Extract attack chains
   590→        chains = self._extract_attack_chains()
   591→        result.attack_chains = [c.to_dict() for c in chains]
   592→
   593→        # Identify choke points
   594→        choke_points = self._identify_choke_points()
   595→        result.choke_points = [cp.to_dict() for cp in choke_points]
   596→
   597→        # Identify single points of failure
   598→        spofs = self._identify_single_point_failures()
   599→        result.single_point_failures = spofs
   600→
   601→        # Identify high-value assets
   602→        hvas = self._identify_high_value_assets()
   603→        result.high_value_assets = hvas
   604→
   605→        # Identify critical attack hubs
   606→        hubs = self._identify_attack_hubs()
   607→        result.critical_attack_hubs = hubs
   608→
   609→        # Kill chain mapping
   610→        result.kill_chain_mapping = self._map_kill_chain()
   611→
   612→        result.analysis_duration_ms = round((time.monotonic() - t0) * 1000, 2)
   613→        return result

   615→    def compute_blast_radius(self, node_id: str) -> BlastRadius:
   616→        """Compute the blast radius from a given node."""
   617→        descendants = self.graph.descendants(node_id, max_depth=10)
   618→
   619→        affected_nodes = list(descendants)
   620→        affected_assets: List[str] = []
   621→        sev_spread: Dict[str, int] = defaultdict(int)
   622→        total_risk = 0.0
   623→
   624→        for nid in affected_nodes:
   625→            node = self.graph.nodes.get(nid)
   626→            if node:
   627→                sev_spread[node.severity] = sev_spread.get(node.severity, 0) + 1
   628→                total_risk += node.risk_score
   629→                if node.metadata.get("asset"):
   630→                    affected_assets.append(node.metadata["asset"])
   631→
   632→        return BlastRadius(
   633→            source_node=node_id,
   634→            affected_nodes=affected_nodes,
   635→            affected_assets=list(set(affected_assets)),
   636→            max_depth=max((len(self.graph.shortest_path(node_id, d) or []) - 1) for d in affected_nodes) if affected_nodes else 0,
   637→            severity_spread=dict(sev_spread),
   638→            total_risk=total_risk,
   639→        )

   643→    def _create_relationship_edges(self, findings: List[Dict[str, Any]]) -> None:
   644→        """Create edges between finding nodes based on category relationships."""
   645→        node_list = list(self.graph.nodes.keys())
   646→        finding_nodes = [n for n in node_list if n.startswith("finding:")]
   647→
   648→        # Build category index
   649→        cat_index: Dict[str, List[str]] = defaultdict(list)
   650→        for nid in finding_nodes:
   651→            node = self.graph.nodes.get(nid)
   652→            if node:
   653→                cat = node.metadata.get("category", "misc")
   654→                cat_index[cat].append(nid)
   655→
   656→        def _add_edge_safe(edge: GraphEdge) -> None:
   657→            """Add edge only if under the global edge limit."""
   658→            if self.graph.edge_count() < self.MAX_GRAPH_EDGES:
   659→                self.graph.add_edge(edge)
   660→
   661→        # Apply relationship templates
   662→        for src_cat, tgt_cat, edge_type, weight in _RELATIONSHIP_TEMPLATES:
   663→            src_nodes = cat_index.get(src_cat, [])
   664→            tgt_nodes = cat_index.get(tgt_cat, [])
   665→            for sn in src_nodes:
   666→                for tn in tgt_nodes:
   667→                    if sn != tn:
   668→                        _add_edge_safe(GraphEdge(
   669→                            source=sn, target=tn,
   670→                            edge_type=edge_type, weight=weight,
   671→                            confidence=min(weight, 0.9),
   672→                        ))
   673→
   674→        # Asset-based edges: findings sharing the same asset
   675→        asset_index: Dict[str, List[str]] = defaultdict(list)
   676→        for nid in finding_nodes:
   677→            node = self.graph.nodes.get(nid)
   678→            if node:
   679→                asset = node.metadata.get("asset", "")
   680→                if asset:
   681→                    asset_index[asset].append(nid)
   682→
   683→        for asset, nodes in asset_index.items():
   684→            for i, j in combinations(range(len(nodes)), 2):
   685→                _add_edge_safe(GraphEdge(
   686→                    source=nodes[i], target=nodes[j],
   687→                    edge_type="shared_asset", weight=0.4, confidence=0.7,
   688→                ))
   689→                _add_edge_safe(GraphEdge(
   690→                    source=nodes[j], target=nodes[i],
   691→                    edge_type="shared_asset", weight=0.4, confidence=0.7,
   692→                ))

   694→    def _extract_attack_chains(self) -> List[AttackChain]:
   695→        """Extract attack chains from the graph."""
   696→        chains: List[AttackChain] = []
   697→        entry_id = "entry:recon"
   698→        objective_id = "objective:compromise"
   699→
   700→        if not self.graph.has_node(entry_id) or not self.graph.has_node(objective_id):
   701→            return chains
   702→
   703→        # Find all paths from entry to objective
   704→        paths = self.graph.all_paths_between(entry_id, objective_id, max_depth=8)
   705→
   706→        for i, path in enumerate(paths):
   707→            if len(path) < 3:
   708→                continue  # Skip trivial paths
   709→
   710→            # Build chain metadata
   711→            steps = []
   712→            labels = []
   713→            severities = []
   714→            phases = []
   715→            total_risk = 0.0
   716→
   717→            for nid in path:
   718→                node = self.graph.nodes.get(nid, GraphNode(id=nid, label=nid, node_type="unknown"))
   719→                steps.append(nid)
   720→                labels.append(node.label)
   721→                severities.append(node.severity)
   722→                total_risk += node.risk_score
   723→                cat = node.metadata.get("category", "")
   724→                phase = _CATEGORY_TO_PHASE.get(cat, "")
   725→                if phase:
   726→                    phases.append(phase)
   727→
   728→            # Compute chain metrics
   729→            avg_risk = total_risk / max(len(steps), 1)
   730→            impact = min(1.0, avg_risk / 10.0)
   731→            likelihood = min(1.0, 1.0 / (len(steps) * 0.5))
   732→            confidence = min(1.0, likelihood * impact)
   733→
   734→            if len(steps) >= 6:
   735→                complexity = "very_high"
   736→            elif len(steps) >= 4:
   737→                complexity = "high"
   738→            elif len(steps) >= 3:
   739→                complexity = "medium"
   740→            else:
   741→                complexity = "low"
   742→
   743→            chain_id = hashlib.sha256(
   744→                f"chain:{i}:{','.join(path)}".encode()
   745→            ).hexdigest()[:12]
   746→
   747→            narrative = (
   748→                f"Attack chain with {len(steps)} steps from '{labels[1]}' "
   749→                f"through {len(steps) - 2} intermediate vulnerabilities "
   750→                f"to '{labels[-1]}'. Complexity: {complexity}, "
   751→                f"Estimated impact: {impact:.0%}."
   752→            )
   753→
   754→            chains.append(AttackChain(
   755→                chain_id=chain_id,
   756→                steps=steps,
   757→                step_labels=labels,
   758→                step_severities=severities,
   759→                entry_point=steps[1] if len(steps) > 1 else steps[0],
   760→                objective=labels[-1] if labels else "",
   761→                complexity=complexity,
   762→                likelihood=round(likelihood, 2),
   763→                impact=round(impact, 2),
   764→                blast_radius=len(steps),
   765→                confidence=round(confidence, 2),
   766→                narrative=narrative,
   767→                mitre_phases=list(dict.fromkeys(phases)),  # Deduplicated, ordered
   768→            ))
   769→
   770→        # Sort by confidence
   771→        chains.sort(key=lambda c: c.confidence, reverse=True)
   772→        return chains[:20]  # Cap at 20 chains

   774→    def _identify_choke_points(self) -> List[ChokePoint]:
   775→        """Identify nodes that appear in many attack paths."""
   776→        entry_id = "entry:recon"
   777→        objective_id = "objective:compromise"
   778→
   779→        if not self.graph.has_node(entry_id) or not self.graph.has_node(objective_id):
   780→            return []
   781→
   782→        # Find all paths and count node traversals
   783→        all_paths = self.graph.all_paths_between(entry_id, objective_id, max_depth=6)
   784→        traversal_counts: Dict[str, int] = defaultdict(int)
   785→        for path in all_paths:
   786→            for nid in path[1:-1]:  # Exclude entry and objective
   787→                traversal_counts[nid] += 1
   788→
   789→        if not traversal_counts:
   790→            return []
   791→
   792→        max_traversals = max(traversal_counts.values())
   793→        choke_points: List[ChokePoint] = []
   794→
   795→        for nid, count in traversal_counts.items():
   796→            if count < 2:
   797→                continue
   798→            node = self.graph.nodes.get(nid)
   799→            if not node or node.node_type != "finding":
   800→                continue
   801→
   802→            concentration = count / max_traversals
   803→            if concentration >= 0.3:  # Appears in >30% of paths
   804→                choke_points.append(ChokePoint(
   805→                    node_id=nid,
   806→                    label=node.label,
   807→                    paths_through=count,
   808→                    risk_concentration=concentration,
   809→                    severity=node.severity,
   810→                    recommendation=(
   811→                        f"Remediating '{node.label}' would break {count} attack paths. "
   812→                        f"This is a critical choke point with {concentration:.0%} "
   813→                        f"risk concentration. Prioritize this vulnerability."
   814→                    ),
   815→                ))
   816→
   817→        choke_points.sort(key=lambda cp: cp.risk_concentration, reverse=True)
   818→        return choke_points[:10]
   819→
   820→    def _identify_single_point_failures(self) -> List[Dict[str, Any]]:
   821→        """Identify nodes whose removal would disconnect entry from objective."""
   822→        entry_id = "entry:recon"
   823→        objective_id = "objective:compromise"
   824→
   825→        if not self.graph.has_node(entry_id) or not self.graph.has_node(objective_id):
   826→            return []
   827→
   828→        baseline_paths = self.graph.all_paths_between(entry_id, objective_id, max_depth=6)
   829→        spofs: List[Dict[str, Any]] = []
   830→
   831→        for nid in list(self.graph.nodes.keys()):
   832→            if nid.startswith("entry:") or nid.startswith("objective:"):
   833→                continue
   834→
   835→            # Temporarily check if removing this node breaks all paths
   836→            remaining_paths = [p for p in baseline_paths if nid not in p]
   837→            if not remaining_paths and baseline_paths:
   838→                node = self.graph.nodes.get(nid)
   839→                if node:
   840→                    spofs.append({
   841→                        "node_id": nid,
   842→                        "label": node.label,
   843→                        "severity": node.severity,
   844→                        "category": node.metadata.get("category", "unknown"),
   845→                        "paths_blocked": len(baseline_paths),
   846→                        "description": (
   847→                            f"Removing '{node.label}' would block ALL {len(baseline_paths)} "
   848→                            f"attack paths. This is a single point of failure — "
   849→                            f"both a critical dependency and a high-value remediation target."
   850→                        ),
   851→                    })
   852→
   853→        return spofs[:10]

   855→    def _identify_high_value_assets(self) -> List[Dict[str, Any]]:
   856→        """Identify assets with the most critical/high findings."""
   857→        asset_findings: Dict[str, List[Dict[str, Any]]] = defaultdict(list)
   858→
   859→        for nid, node in self.graph.nodes.items():
   860→            if node.node_type == "finding":
   861→                asset = node.metadata.get("asset", "unknown")
   862→                if asset:
   863→                    asset_findings[asset].append({
   864→                        "node_id": nid,
   865→                        "title": node.label,
   866→                        "severity": node.severity,
   867→                        "risk_score": node.risk_score,
   868→                    })
   869→
   870→        hvas: List[Dict[str, Any]] = []
   871→        for asset, items in asset_findings.items():
   872→            if len(items) < 2:
   873→                continue
   874→            total_risk = sum(i["risk_score"] for i in items)
   875→            critical_count = sum(1 for i in items if i["severity"] == "critical")
   876→            high_count = sum(1 for i in items if i["severity"] == "high")
   877→
   878→            if critical_count > 0 or high_count >= 2:
   879→                hvas.append({
   880→                    "asset": asset,
   881→                    "finding_count": len(items),
   882→                    "total_risk": round(total_risk, 2),
   883→                    "critical_count": critical_count,
   884→                    "high_count": high_count,
   885→                    "risk_level": "critical" if total_risk >= 15 else "high" if total_risk >= 8 else "medium",
   886→                })
   887→
   888→        hvas.sort(key=lambda x: x["total_risk"], reverse=True)
   889→        return hvas[:10]

   891→    def _identify_attack_hubs(self) -> List[Dict[str, Any]]:
   892→        """Identify nodes with highest connectivity (potential attack hubs)."""
   893→        betweenness = self.graph.betweenness_proxy()
   894→        if not betweenness:
   895→            return []
   896→
   897→        max_b = max(betweenness.values())
   898→        hubs: List[Dict[str, Any]] = []
   899→
   900→        for nid, score in sorted(betweenness.items(), key=lambda x: x[1], reverse=True)[:10]:
   901→            if score == 0:
   902→                continue
   903→            node = self.graph.nodes.get(nid)
   904→            if not node or node.node_type != "finding":
   905→                continue
   906→
   907→            hubs.append({
   908→                "node_id": nid,
   909→                "label": node.label,
   910→                "severity": node.severity,
   911→                "betweenness": score,
   912→                "betweenness_ratio": round(score / max_b, 2) if max_b > 0 else 0.0,
   913→                "in_degree": self.graph.in_degree(nid),
   914→                "out_degree": self.graph.out_degree(nid),
   915→                "description": (
   916→                    f"Attack hub with betweenness {score} "
   917→                    f"(top {(score/max_b)*100:.0f}% of all nodes). "
   918→                    f"Connects to {self.graph.out_degree(nid)} downstream targets "
   919→                    f"and receives from {self.graph.in_degree(nid)} sources."
   920→                ),
   921→            })
   922→
   923→        return hubs

   925→    def _map_kill_chain(self) -> Dict[str, Any]:
   926→        """Map findings to Lockheed Martin Kill Chain phases."""
   927→        phase_findings: Dict[str, List[Dict[str, Any]]] = defaultdict(list)
   928→
   929→        for nid, node in self.graph.nodes.items():
   930→            if node.node_type != "finding":
   931→                continue
   932→            cat = node.metadata.get("category", "")
   933→            phase = _CATEGORY_TO_PHASE.get(cat)
   934→            if phase:
   935→                phase_findings[phase].append({
   936→                    "node_id": nid,
   937→                    "title": node.label,
   938→                    "severity": node.severity,
   939→                    "category": cat,
   940→                })
   941→
   942→        phase_coverage = {}
   943→        for phase in KILL_CHAIN_PHASES:
   944→            items = phase_findings.get(phase, [])
   945→            phase_coverage[phase] = {
   946→                "finding_count": len(items),
   947→                "findings": items[:10],  # Cap for readability
   948→                "has_findings": len(items) > 0,
   949→            }
   950→
   951→        # Detect phase gaps
   952→        covered_phases = {p for p, data in phase_coverage.items() if data["has_findings"]}
   953→        gap_phases = [p for p in KILL_CHAIN_PHASES if p not in covered_phases]
   954→
   955→        return {
   956→            "phases": phase_coverage,
   957→            "coverage_percentage": round(len(covered_phases) / len(KILL_CHAIN_PHASES) * 100, 1),
   958→            "covered_phases": sorted(covered_phases),
   959→            "gap_phases": gap_phases,
   960→            "phase_progression": self._assess_phase_progression(phase_findings),
   961→        }
   962→
   963→    def _assess_phase_progression(self, phase_findings: Dict[str, List[Dict[str, Any]]]) -> str:
   964→        """Assess how far an attacker could progress through the kill chain."""
   965→        progression = []
   966→        for phase in KILL_CHAIN_PHASES:
   967→            if phase in phase_findings and phase_findings[phase]:
   968→                progression.append(phase)
   969→
   970→        if not progression:
   971→            return "No kill chain progression detected."
   972→
   973→        if progression == KILL_CHAIN_PHASES:
   974→            return f"FULL CHAIN: Attacker can progress through all {len(progression)} phases."
   975→
   976→        last = progression[-1]
   977→        idx = KILL_CHAIN_PHASES.index(last)
   978→        return (
   979→            f"Attacker can progress from {progression[0]} to {last} "
   980→            f"(phase {idx + 1} of {len(KILL_CHAIN_PHASES)}). "
   981→            f"{len(KILL_CHAIN_PHASES) - idx - 1} phases remain uncovered."
   982→        )
