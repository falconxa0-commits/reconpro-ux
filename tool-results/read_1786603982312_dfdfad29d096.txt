   126→async function bannerGrab(domain: string, ip: string | null): Promise<{ banners: Array<{ port: number; banner: string; service: string }>; cveMatches: CVEEntry[] }> {
   127→  const target = ip || domain;
   128→  const ports = [21, 22, 25, 80, 110, 143, 443, 445, 993, 995, 1433, 1521, 3306, 3389, 5432, 5900, 6379, 8080, 8443, 8888, 9090, 27017];
   129→  const banners: Array<{ port: number; banner: string; service: string }> = [];
   130→
   131→  for (const port of ports) {
   132→    const banner = await run(
   133→      `echo "" | timeout 4 nc -w3 ${target} ${port} 2>/dev/null | head -c 500`,
   134→      6000
   135→    );
   136→    if (banner && banner.length > 3) {
   137→      const service = detectService(port, banner);
   138→      banners.push({ port, banner: banner.replace(/[\x00-\x1f\x7f]/g, '.').substring(0, 200), service });
   139→    }
   140→  }
   141→
   142→  // HTTP banner via curl
   143→  const httpBanner = await run(`curl -sI --max-time 6 https://${domain} 2>/dev/null | head -20`, 8000);
   144→  if (httpBanner) banners.push({ port: 443, banner: httpBanner.substring(0, 300), service: 'http' });
   145→  const httpBanner2 = await run(`curl -sI --max-time 6 http://${domain} 2>/dev/null | head -20`, 8000);
   146→  if (httpBanner2) banners.push({ port: 80, banner: httpBanner2.substring(0, 300), service: 'http' });
   147→
   148→  // Match banners against CVE database
   149→  const combined = banners.map(b => b.banner).join(' ').toLowerCase();
   150→  const cveMatches = CVE_DATABASE.filter(cve => {
   151→    return cve.affected.some(a => combined.includes(a)) && cve.versionPattern.test(combined);
   152→  });
   153→
   154→  return { banners, cveMatches };
   155→}
   156→
   157→function detectService(port: number, banner: string): string {
   158→  const b = banner.toLowerCase();
   159→  if (port === 22 || b.includes('ssh')) return 'openssh';
   160→  if (port === 21 || b.includes('ftp')) return 'ftp';
   161→  if (port === 25 || b.includes('smtp') || b.includes('postfix') || b.includes('esmtp')) return 'postfix';
   162→  if (port === 80 || port === 8080 || port === 8443 || b.includes('http')) return 'http';
   163→  if (port === 110 || b.includes('pop3')) return 'pop3';
   164→  if (port === 143 || b.includes('imap')) return 'imap';
   165→  if (port === 443 || b.includes('ssl') || b.includes('tls')) return 'https';
   166→  if (port === 3306 || b.includes('mysql')) return 'mysql';
   167→  if (port === 5432 || b.includes('postgresql') || b.includes('postgres')) return 'postgresql';
   168→  if (port === 6379 || b.includes('redis')) return 'redis';
   169→  if (port === 27017 || b.includes('mongodb')) return 'mongodb';
   170→  if (b.includes('nginx')) return 'nginx';
   171→  if (b.includes('apache')) return 'apache';
   172→  if (b.includes('tomcat')) return 'tomcat';
   173→  return 'unknown';
   174→}

   238→  // SSRF detection (check if internal IPs are accessible)
   239→  const ssrfPaths = ['/?url=http://169.254.169.254/latest/meta-data/', '/api/fetch?url=http://127.0.0.1:80', '/proxy?url=http://10.0.0.1'];
   240→  for (const path of ssrfPaths) {
   241→    const ssrfResult = await run(
   242→      `curl -s --max-time 4 "https://${domain}${path}" 2>/dev/null | head -c 500`,
   243→      6000
   244→    );
   245→    if (ssrfResult.includes('ami-id') || ssrfResult.includes('instance-id') || ssrfResult.includes('local-hostname') || ssrfResult.includes('127.0.0.1')) {
   246→      findings.push({
   247→        title: 'CRITICAL: SSRF — Internal Network Accessible',
   248→        severity: 'critical', category: 'http-vuln',
   249→        description: `Server makes requests to internal URLs. SSRF can access cloud metadata (AWS/GCP), internal services, and bypass firewalls.`,
   250→        evidence: `GET ${path} → internal response detected`,
   251→        asset: `${domain}${path}`,
   252→      });
   253→      break;
   254→    }
   255→  }
   256→
   257→  // XSS detection (reflected input)
   258→  const xssPayloads = ['<script>alert(1)</script>', '"><img src=x onerror=alert(1)>', "'-alert(1)-'"];
   259→  for (const payload of xssPayloads) {
   260→    const xssResult = await run(
   261→      `curl -s --max-time 5 "https://${domain}/search?q=${encodeURIComponent(payload)}" 2>/dev/null | head -c 2000`,
   262→      7000
   263→    );
   264→    if (xssResult.includes(payload) && !xssResult.includes('alert(1)</script></') === false) {
   265→      const reflected = xssResult.includes('<script>alert(1)</script>') || xssResult.includes('onerror=alert(1)');
   266→      if (reflected) {
   267→        findings.push({
   268→          title: 'Reflected XSS — Input Executed in Response',
   269→          severity: 'high', category: 'http-vuln',
   270→          description: `User input is reflected in page content without sanitization. XSS enables session hijacking, credential theft, and malware delivery.`,
   271→          evidence: `GET /search?q=<payload> → reflected unescaped in HTML`,
   272→          asset: `${domain}/search`,
   273→        });
   274→        break;
   275→      }
   276→    }
   277→  }
   278→
   279→  // Clickjacking (X-Frame-Options)
   280→  const xfo = await run(`curl -sI --max-time 5 https://${domain} 2>/dev/null | grep -i "x-frame-options"`, 7000);
   281→  if (!xfo) {
   282→    const cspFrame = await run(`curl -sI --max-time 5 https://${domain} 2>/dev/null | grep -i "content-security-policy" | grep -i "frame-ancestors"`, 7000);
   283→    if (!cspFrame) {
   284→      findings.push({
   285→        title: 'Clickjacking — No Frame Protection',
   286→        severity: 'medium', category: 'http-vuln',
   287→        description: 'No X-Frame-Options or frame-ancestors CSP directive. Attacker can embed site in iframe to trick users into clicking hidden buttons (account takeover, unauthorized actions).',
   288→        evidence: 'No X-Frame-Options header, no frame-ancestors in CSP',
   289→        asset: domain,
   290→      });
   291→    }
   292→  }
   293→
   294→  // Cookie security analysis
   295→  const setCookies = await run(`curl -sI --max-time 5 https://${domain} 2>/dev/null | grep -i "set-cookie"`, 7000);
   296→  if (setCookies) {
   297→    const insecureCookies: string[] = [];
   298→    if (!setCookies.toLowerCase().includes('secure')) insecureCookies.push('Secure flag missing');
   299→    if (!setCookies.toLowerCase().includes('httponly')) insecureCookies.push('HttpOnly flag missing');
   300→    if (!setCookies.toLowerCase().includes('samesite')) insecureCookies.push('SameSite missing');
   301→
   302→    if (insecureCookies.length > 0) {
   303→      findings.push({
   304→        title: `Insecure Cookie Configuration — ${insecureCookies.length} issues`,
   305→        severity: 'medium', category: 'http-vuln',
   306→        description: `Cookies lack security flags: ${insecureCookies.join(', ')}. Without Secure, cookies sent over HTTP. Without HttpOnly, accessible to JavaScript (XSS target). Without SameSite, vulnerable to CSRF.`,
   307→        evidence: `Set-Cookie: ${setCookies.substring(0, 100)} | Issues: ${insecureCookies.join(', ')}`,
   308→        asset: domain,
   309→      });
   310→    }
   311→  }
   312→
   313→  // Information disclosure — server version
   314→  const serverHeader = await run(`curl -sI --max-time 5 https://${domain} 2>/dev/null | grep -i "^server:"`, 7000);
   315→  if (serverHeader && serverHeader.match(/apache\/\d|nginx\/\d|php\/\d|tomcat|express|gunicorn/i)) {
   316→    findings.push({
   317→      title: 'Information Disclosure — Server Version Leaked',
   318→      severity: 'low', category: 'http-vuln',
   319→      description: `Server header reveals exact software version: ${serverHeader.trim()}. Attackers use version info to find known exploits.`,
   320→      evidence: serverHeader.trim(),
   321→      asset: domain,
   322→    });
   323→  }
   324→
   325→  // CSRF token check
   326→  const csrfCheck = await run(`curl -s --max-time 5 "https://${domain}/login" 2>/dev/null | grep -iE "csrf|_token|authenticity_token|x-csrf" | head -3`, 7000);
   327→  if (!csrfCheck) {
   328→    findings.push({
   329→      title: 'CSRF Protection — No Anti-CSRF Token Detected',
   330→      severity: 'medium', category: 'http-vuln',
   331→      description: 'No CSRF token found in login form. State-changing requests may be forged from other sites, enabling unauthorized actions on behalf of authenticated users.',
   332→      evidence: 'No csrf_token, authenticity_token, or x-csrf-token in login page',
   333→      asset: `${domain}/login`,
   334→    });
   335→  }
   336→
   337→  // Mixed content detection
   338→  const mixedContent = await run(
   339→    `curl -s --max-time 8 https://${domain} 2>/dev/null | grep -oiE 'src="http://[^"]+"|href="http://[^"]+"|url\("http://[^)]+"\)' | head -5`,
   340→    10000
   341→  );
   342→  if (mixedContent) {
   343→    findings.push({
   344→      title: `Mixed Content — ${mixedContent.split('\n').filter(Boolean).length} HTTP Resources on HTTPS Page`,
   345→      severity: 'low', category: 'http-vuln',
   346→      description: 'HTTPS page loads resources via HTTP. Browsers will show warnings, and MITM can inject content into the insecure resources.',
   347→      evidence: mixedContent.trim().substring(0, 150),
   348→      asset: domain,
   349→    });
   350→  }
   351→
   352→  return findings;
   353→}

   356→// 3. SSL/TLS ATTACK VECTOR SCANNER
   357→// ══════════════════════════════════════════════════════════════════════════════
   358→
   359→async function scanSSLVulns(domain: string): Promise<Finding[]> {
   360→  const findings: Finding[] = [];
   361→
   362→  // Full SSL scan with all cipher suites
   363→  const sslFull = await run(`echo | openssl s_client -connect ${domain}:443 -servername ${domain} 2>/dev/null`, 10000);
   364→  const sslLines = sslFull.split('\n').map(l => l.trim()).filter(Boolean);
   365→
   366→  // Protocol version checks
   367→  const protocolLine = sslLines.find(l => l.startsWith('Protocol'));
   368→  const proto = protocolLine || '';
   369→
   370→  if (proto.includes('TLSv1.0')) {
   371→    findings.push({
   372→      title: 'BEAST Attack Vulnerable — TLS 1.0 Active',
   373→      severity: 'high', category: 'ssl-vuln',
   374→      description: 'TLS 1.0 is vulnerable to BEAST (Browser Exploit Against SSL/TLS). Block cipher attack can decrypt communication blocks. PCI-DSS prohibits TLS 1.0.',
   375→      evidence: proto,
   376→      asset: `${domain}:443`,
   377→    });
   378→  }
   379→  if (proto.includes('TLSv1.1')) {
   380→    findings.push({
   381→      title: 'Deprecated TLS 1.1 in Use',
   382→      severity: 'medium', category: 'ssl-vuln',
   383→      description: 'TLS 1.1 is deprecated (RFC 8996). Remove support to prevent downgrade attacks.',
   384→      evidence: proto,
   385→      asset: `${domain}:443`,
   386→    });
   387→  }
   388→  if (proto.includes('SSLv3')) {
   389→    findings.push({
   390→      title: 'CRITICAL: POODLE — SSLv3 Detected',
   391→      severity: 'critical', category: 'ssl-vuln',
   392→      description: 'SSLv3 is extremely vulnerable to POODLE (Padding Oracle On Downgraded Legacy Encryption). Complete decryption of encrypted sessions possible.',
   393→      evidence: proto,
   394→      asset: `${domain}:443`,
   395→    });
   396→  }
   397→  if (proto.includes('SSLv2')) {
   398→    findings.push({
   399→      title: 'CRITICAL: DROWN Attack — SSLv2 Detected',
   400→      severity: 'critical', category: 'ssl-vuln',
   401→      description: 'SSLv2 enables DROWN (Decrypting RSA with Obsolete and Weakened eNcryption). Cross-protocol attack can decrypt TLS connections.',
   402→      evidence: proto,
   403→      asset: `${domain}:443`,
   404→    });
   405→  }
   406→
   407→  // Cipher analysis
   408→  const cipherLine = sslLines.find(l => l.includes('Cipher'));
   409→  const cipher = cipherLine || '';
   410→
   411→  // Check for weak ciphers by testing multiple
   412→  const weakCiphers = ['RC4', 'DES', '3DES', 'NULL', 'EXPORT', 'MD5', 'RC2', 'IDEA'];
   413→  const cipherSuite = await run(
   414→    `echo | openssl s_client -connect ${domain}:443 -servername ${domain} -cipher 'ALL:eNULL' 2>/dev/null | grep "Cipher"`,
   415→    10000
   416→  );
   417→
   418→  for (const weak of weakCiphers) {
   419→    if (cipherSuite.toUpperCase().includes(weak)) {
   420→      findings.push({
   421→        title: `CRITICAL: Weak Cipher ${weak} Accepted`,
   422→        severity: 'critical', category: 'ssl-vuln',
   423→        description: `Server accepts ${weak} cipher suite. ${weak === 'RC4' ? 'RC4 has known statistical biases exploitable since 2013.' : weak === 'NULL' ? 'NULL cipher means NO encryption — traffic is plaintext.' : weak === 'EXPORT' ? 'EXPORT-grade ciphers are trivially breakable (40-bit keys).' : `${weak} is cryptographically broken and must be disabled.`}`,
   424→        evidence: `Cipher: ${cipherSuite.trim()}`,
   425→        asset: `${domain}:443`,
   426→      });
   427→      break;
   428→    }
   429→  }
   430→
   431→  if (cipher.includes('CBC') && proto.includes('TLSv1')) {
   432→    findings.push({
   433→      title: 'BEAST Susceptible — CBC Mode + TLS 1.0',
   434→      severity: 'medium', category: 'ssl-vuln',
   435→      description: 'CBC cipher suites with TLS 1.0 are vulnerable to BEAST attack. Use GCM-mode ciphers or upgrade to TLS 1.2+.',
   436→      evidence: `${proto} | ${cipher}`,
   437→      asset: `${domain}:443`,
   438→    });
   439→  }
   440→
   441→  // Heartbleed check (CVE-2014-0160)
   442→  const heartbleedCheck = await run(
   443→    `echo | openssl s_client -connect ${domain}:443 -servername ${domain} -tlsextdebug 2>&1 | grep -i "heartbeat"`,
   444→    8000
   445→  );
   446→  if (heartbleedCheck) {
   447→    findings.push({
   448→      title: 'Heartbleed Vulnerability Possible (CVE-2014-0160)',
   449→      severity: 'critical', category: 'ssl-vuln',
   450→      description: 'Server supports heartbeat extension. If running vulnerable OpenSSL (1.0.1-1.0.1f), up to 64KB of server memory can be leaked per request — including private keys, passwords, session tokens.',
   451→      evidence: `Heartbeat extension: ${heartbleedCheck.trim()}`,
   452→      asset: `${domain}:443`,
   453→    });
   454→  }
   455→
   456→  // Certificate issues
   457→  const certDates = await run(
   458→    `echo | openssl s_client -connect ${domain}:443 -servername ${domain} 2>/dev/null | openssl x509 -noout -dates 2>/dev/null`,
   459→    10000
   460→  );
   461→  if (certDates) {
   462→    const notAfter = certDates.match(/notAfter=(.+)/);
   463→    if (notAfter) {
   464→      const expiry = new Date(notAfter[1].trim());
   465→      const now = new Date();
   466→      const daysLeft = Math.floor((expiry.getTime() - now.getTime()) / 86400000);
   467→      if (daysLeft < 0) {
   468→        findings.push({
   469→          title: 'CRITICAL: SSL Certificate EXPIRED',
   470→          severity: 'critical', category: 'ssl-vuln',
   471→          description: `Certificate expired ${Math.abs(daysLeft)} days ago. Visitors see security warnings. Connections may be intercepted.`,
   472→          evidence: `notAfter: ${notAfter[1].trim()} | Expired ${Math.abs(daysLeft)} days ago`,
   473→          asset: `${domain}:443`,
   474→        });
   475→      } else if (daysLeft < 30) {
   476→        findings.push({
   477→          title: `SSL Certificate Expiring in ${daysLeft} Days`,
   478→          severity: 'high', category: 'ssl-vuln',
   479→          description: `Certificate expires soon. Service disruption and security warnings imminent.`,
   480→          evidence: `notAfter: ${notAfter[1].trim()} | ${daysLeft} days remaining`,
   481→          asset: `${domain}:443`,
   482→        });
   483→      }
   484→    }
   485→  }
   486→
   487→  // OCSP stapling
   488→  const ocsp = await run(
   489→    `echo | openssl s_client -connect ${domain}:443 -servername ${domain} -status 2>/dev/null | grep -i "OCSP"`,
   490→    8000
   491→  );
   492→  if (!ocsp) {
   493→    findings.push({
   494→      title: 'OCSP Stapling Not Enabled',
   495→      severity: 'low', category: 'ssl-vuln',
   496→      description: 'Without OCSP stapling, browsers must contact CA separately to verify certificate revocation, adding latency and privacy exposure.',
   497→      evidence: 'No OCSP response in handshake',
   498→      asset: `${domain}:443`,
   499→    });
   500→  }
   501→
   502→  // Perfect Forward Secrecy
   503→  const pfsCheck = await run(
   504→    `echo | openssl s_client -connect ${domain}:443 -servername ${domain} 2>/dev/null | grep -iE "ECDHE-RSA|ECDHE-ECDSA|DHE-RSA|DHE-DSS"`,
   505→    8000
   506→  );
   507→  if (!pfsCheck) {
   508→    findings.push({
   509→      title: 'Perfect Forward Secrecy — Not Supported',
   510→      severity: 'medium', category: 'ssl-vuln',
   511→      description: 'Without PFS, compromise of server private key allows decryption of ALL past sessions captured via passive surveillance.',
   512→      evidence: 'No ECDHE/DHE cipher negotiated',
   513→      asset: `${domain}:443`,
   514→    });
   515→  }
   516→
   517→  return findings;
   518→}

   521→// 4. DNS VULNERABILITY ASSESSMENT
   522→// ══════════════════════════════════════════════════════════════════════════════

   524→async function scanDNSVulns(domain: string): Promise<Finding[]> {
   525→  const findings: Finding[] = [];
   526→
   527→  // Zone transfer (AXFR) attempt
   528→  const nsRecords = await run(`dig +short +time=3 ${domain} NS`, 5000);
   529→  const nsList = nsRecords.split('\n').filter(Boolean);
   530→  for (const ns of nsList.slice(0, 3)) {
   531→    const axfrResult = await run(`dig axfr ${domain} @${ns} +time=5 +tries=1 2>/dev/null`, 7000);
   532→    if (axfrResult && axfrResult.includes('ANSWER SECTION') && axfrResult.split('\n').length > 5) {
   533→      findings.push({
   534→        title: 'CRITICAL: DNS Zone Transfer (AXFR) Successful',
   535→        severity: 'critical', category: 'dns-vuln',
   536→        description: `Full zone transfer from ${ns} revealed complete DNS records including internal hostnames, IP addresses, and service records. Complete infrastructure mapping possible.`,
   537→        evidence: `dig axfr ${domain} @${ns} → ${axfrResult.split('\n').length} records transferred`,
   538→        asset: ns,
   539→      });
   540→      break;
   541→    }
   542→  }
   543→
   544→  // Subdomain takeover detection (CNAME to dead services)
   545→  const takeoverSubs = ['staging', 'dev', 'blog', 'cdn', 'assets', 'static', 'mail', 'api', 'app', 'test'];
   546→  const takeoverCandidates: string[] = [];
   547→  for (const sub of takeoverSubs) {
   548→    const cname = await run(`dig +short +time=2 ${sub}.${domain} CNAME`, 4000);
   549→    if (cname) {
   550→      const deadServices = ['herokuapp.com', 'cloudfront.net', 'github.io', 's3.amazonaws.com',
   551→        'shopify.com', 'fastly.net', 'cloudwaysapps.com', 'pantheon.io'];
   552→      for (const dead of deadServices) {
   553→        if (cname.toLowerCase().includes(dead)) {
   554→          const subExists = await run(`curl -sI --max-time 4 https://${sub}.${domain} 2>/dev/null | head -1`, 6000);
   555→          if (subExists && !subExists.includes('200') && !subExists.includes('301') && !subExists.includes('302')) {
   556→            takeoverCandidates.push(`${sub}.${domain} → CNAME: ${cname} (${dead})`);
   557→          }
   558→        }
   559→      }
   560→    }
   561→  }
   562→  if (takeoverCandidates.length > 0) {
   563→    findings.push({
   564→      title: `Subdomain Takeover Risk — ${takeoverCandidates.length} candidate(s)`,
   565→      severity: 'high', category: 'dns-vuln',
   566→      description: `Subdomains point to third-party services via CNAME but may not be properly provisioned. If the third-party resource is deleted, an attacker can claim the subdomain and serve malicious content.`,
   567→      evidence: takeoverCandidates.join(' | '),
   568→      asset: domain,
   569→    });
   570→  }
   571→
   572→  // DNS cache snooping
   573→  const cacheResult = await run(`dig +time=3 ${domain} @8.8.8.8 ANY +noall +answer 2>/dev/null`, 5000);
   574→  if (cacheResult && cacheResult.split('\n').length > 8) {
   575→    findings.push({
   576→      title: 'DNS Cache Snooping Possible — Excessive Records Visible',
   577→      severity: 'medium', category: 'dns-vuln',
   578→      description: `Recursive DNS query reveals extensive record types. DNS cache can be probed to determine which domains are recently visited by the resolver's clients.`,
   579→      evidence: `dig ANY ${domain} → ${cacheResult.split('\n').length} records from cache`,
   580→      asset: domain,
   581→    });
   582→  }
   583→
   584→  // DNS rebinding protection check
   585→  const dnsRebindHeaders = await run(`curl -sI --max-time 5 https://${domain} 2>/dev/null | grep -iE "x-dns-prefetch|x-content-type|dns-prefetch"`, 7000);
   586→  if (!dnsRebindHeaders) {
   587→    findings.push({
   588→      title: 'DNS Rebinding — No Protection Detected',
   589→      severity: 'medium', category: 'dns-vuln',
   590→      description: 'No DNS rebinding protection headers. Attackers can bypass Same-Origin Policy by controlling DNS to oscillate between their server and target IP.',
   591→      evidence: 'No DNS rebinding protection headers found',
   592→      asset: domain,
   593→    });
   594→  }
   595→
   596→  return findings;
   597→}

   599→// ══════════════════════════════════════════════════════════════════════════════
   600→// MAIN POST HANDLER
   601→// ══════════════════════════════════════════════════════════════════════════════
   602→
   603→export async function POST(request: NextRequest) {
   604→  try {
   605→    const body = await request.json();
   606→    const target = body.target?.toString().trim().replace(/^https?:\/\//, '').replace(/\/.*$/, '');
   607→    if (!target || !/^[a-zA-Z0-9][\w.-]+$/.test(target)) {
   608→      return NextResponse.json({ error: 'Invalid target domain' }, { status: 400 });
   609→    }
   610→
   611→    // Resolve IP first
   612→    const ip = await run(`dig +short +time=3 +tries=1 ${target} A`, 5000);
   613→    const mainIp = ip?.split('\n')[0]?.trim() || null;
   614→
   615→    // Run all modules in parallel
   616→    const [
   617→      bannerResult,
   618→      httpVulns,
   619→      sslVulns,
   620→      dnsVulns,
   621→    ] = await Promise.all([
   622→      bannerGrab(target, mainIp),
   623→      scanHTTPVulns(target),
   624→      scanSSLVulns(target),
   625→      scanDNSVulns(target),
   626→    ]);
   627→
   628→    // Build response
   629→    const allFindings = [...httpVulns, ...sslVulns, ...dnsVulns];
   630→    const c = allFindings.filter(f => f.severity === 'critical').length;
   631→    const h = allFindings.filter(f => f.severity === 'high').length;
   632→    const m = allFindings.filter(f => f.severity === 'medium').length;
   633→    const exploitSummary = {
   634→      totalCVEs: bannerResult.cveMatches.length,
   635→      criticalCVEs: bannerResult.cveMatches.filter(v => v.severity === 'critical').length,
   636→      weaponized: bannerResult.cveMatches.filter(v => v.exploitAvailable === 'weaponized').length,
   637→      theoretical: bannerResult.cveMatches.filter(v => v.exploitAvailable === 'theoretical').length,
   638→      avgCVSS: bannerResult.cveMatches.length > 0
   639→        ? Math.round(bannerResult.cveMatches.reduce((a, b) => a + b.cvss, 0) / bannerResult.cveMatches.length * 10) / 10
   640→        : 0,
   641→    };
   642→
   643→    return NextResponse.json({
   644→      success: true,
   645→      scan: {
   646→        target,
   647→        ip: mainIp,
   648→        cveFindings: bannerResult.cveMatches.map(v => ({
   649→          cve: v.id, title: v.title, cvss: v.cvss, severity: v.severity,
   650→          affected: v.affected.join(', '), evidence: `Version pattern match via banner grab`,
   651→          remediation: v.remediation, exploitAvailable: v.exploitAvailable, epss: v.epss,
   652→        })),
   653→        httpVulns: httpVulns.map(f => ({
   654→          ...f, proof: f.evidence,
   655→        })),
   656→        sslVulns: sslVulns.map(f => ({
   657→          ...f, risk: f.severity,
   658→        })),
   659→        dnsVulns,
   660→        banners: bannerResult.banners,
   661→        exploitSummary,
   662→        attackSurface: {
   663→          score: Math.min(100, c * 25 + h * 15 + m * 8 + exploitSummary.weaponized * 20),
   664→          vectors: [
   665→            ...(bannerResult.cveMatches.length > 0 ? ['Known CVE Exploits'] : []),
   666→            ...(httpVulns.some(f => f.severity === 'critical') ? ['Web Application Exploits'] : []),
   667→            ...(sslVulns.some(f => f.severity === 'critical') ? ['Cryptographic Attacks'] : []),
   668→            ...(dnsVulns.some(f => f.severity === 'critical') ? ['DNS Infrastructure Attacks'] : []),
   669→            ...(exploitSummary.weaponized > 0 ? ['Weaponized Exploits Available'] : []),
   670→          ],
   671→        },
   672→      },
   673→    });
   674→  } catch (error) {
   675→    return NextResponse.json({ error: 'Vulnerability scan failed: ' + (error instanceof Error ? error.message : 'unknown') }, { status: 500 });
   676→  }
   677→}
