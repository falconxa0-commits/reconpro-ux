    51→def _render_summary(result, title: str = "RECONPRO") -> None:
    52→    score = result.total_score
    53→    grade = result.grade
    54→    grade_color = GRADE_COLORS.get(grade, "bold bright_red")
    55→    sc = result.severity_counts
    56→    total = len(result.findings)
    57→    bar_width = 50
    58→    filled = int(score / 100 * bar_width)
    59→    bar = "█" * filled + "░" * (bar_width - filled)
    60→    lines = [
    61→        Text(f"\n  {bar} [bold {grade_color}]{score}/100 ({grade})[/{grade_color}]"),
    62→        Text(f"\n  Target: [cyan]{result.target}[/]"),
    63→        Text(f"  Modules: [dim]{', '.join(result.modules_run)}[/]"),
    64→        Text(""),
    65→        Text(
    66→            f"  Findings: [bold]{total}[/]  "
    67→            f"[bright_red]{sc.get('critical', 0)} critical[/], "
    68→            f"[red]{sc.get('high', 0)} high[/], "
    69→            f"[yellow]{sc.get('medium', 0)} medium[/], "
    70→            f"[green]{sc.get('low', 0)} low[/], "
    71→            f"[dim]{sc.get('info', 0)} info[/]",
    72→        ),
    73→    ]
    74→    if result.vibesec_score is not None:
    75→        lines.append(Text(
    76→            f"\n  VibeSec Score: [bold bright_green]{result.vibesec_score}/100 ({result.vibesec_grade})[/]"
    77→        ))
    78→    console.print(Panel(Group(*lines), border_style=grade_color,
    79→                         title=f"[bold]{title}[/bold]", title_align="left", padding=(1, 2)))

    82→def _render_findings_table(result, show_remediation: bool = False) -> None:
    83→    if not result.findings:
    84→        console.print("\n  [bright_green]No findings. Target is clean.[/]")
    85→        return
    86→    table = Table(title=f"Findings — {len(result.findings)} detected",
    87→                  border_style="bright_white", header_style="bold bright_white")
    88→    table.add_column("Severity", style="bold", width=10)
    89→    table.add_column("Module", style="cyan", width=12)
    90→    table.add_column("Category", style="magenta", width=18)
    91→    table.add_column("Finding", style="white")
    92→    table.add_column("Pts", style="yellow", width=5)
    93→    if show_remediation:
    94→        table.add_column("Fix", style="bright_green", width=50)
    95→    for f in result.findings[:80]:
    96→        sev = f.get("severity", "info")
    97→        color = SEV_COLORS.get(sev, "white")
    98→        row = [
    99→            f"[{color}]{sev.upper()}[/{color}]",
   100→            f.get("module", ""), f.get("category", ""),
   101→            f.get("title", "")[:70], str(f.get("points_deducted", 0)),
   102→        ]
   103→        if show_remediation:
   104→            row.append(f.get("remediation", "")[:50])
   105→        table.add_row(*row)
   106→    console.print(table)

   109→def _render_module_breakdown(result) -> None:
   110→    if not result.module_results:
   111→        return
   112→    t = Table(title="Module Breakdown", border_style="dim", header_style="bold dim")
   113→    t.add_column("Module", style="bold", width=16)
   114→    t.add_column("Findings", style="white", width=10)
   115→    t.add_column("Critical", style="bright_red", width=10)
   116→    t.add_column("High", style="red", width=8)
   117→    t.add_column("Medium", style="yellow", width=8)
   118→    for mod_id, mod_data in result.module_results.items():
   119→        findings = mod_data.get("findings", [])
   120→        t.add_row(mod_id.upper(), str(len(findings)),
   121→                   str(sum(1 for f in findings if f.get("severity") == "critical")),
   122→                   str(sum(1 for f in findings if f.get("severity") == "high")),
   123→                   str(sum(1 for f in findings if f.get("severity") == "medium")))
   124→    console.print(t)

   127→def _render_badge(result) -> None:
   128→    if result.badge_markdown:
   129→        console.print(Panel(Group(

   131→            Text(f"  {result.badge_markdown}", style="cyan"),

   135→def _render_remediations(result) -> None:
   136→    actionable = [f for f in result.findings
   137→                  if f.get("remediation") and f.get("severity") not in ("info",)
   138→                  and f.get("points_deducted", 0) > 0]
   139→    if not actionable:
   140→        console.print("\n  [bright_green]No actionable fixes needed.[/]")
   141→        return
   142→    console.print("\n[bold bright_white]  Fix Commands:[/bold bright_white]\n")
   143→    seen = set()
   144→    for f in actionable:
   145→        fix = f.get("remediation", "").strip()
   146→        sev = f.get("severity", "info")
   147→        color = SEV_COLORS.get(sev, "white")
   148→        if fix and fix not in seen:
   149→            seen.add(fix)
   150→            console.print(f"  [{color}]{sev.upper():8}[/{color}]  {fix}")
   151→    console.print()

   154→def _output_result(result, args, title: str = "RECONPRO", show_remediation: bool = False) -> None:
   155→    json_output = getattr(args, "json_output", False)
   156→    output_file = getattr(args, "output_file", None)
   157→    if json_output:
   158→        text = json.dumps(result.to_dict(), indent=2, default=str)
   159→        if output_file:
   160→            with open(output_file, "w") as fp:
   161→                fp.write(text)
   162→            console.print(f"  [green]JSON saved: [cyan]{output_file}[/][/]")
   163→        else:
   164→            console.print(text)
   165→    else:
   166→        _render_summary(result, title=title)
   167→        console.print()
   168→        _render_module_breakdown(result)
   169→        console.print()
   170→        _render_findings_table(result, show_remediation=show_remediation)
   171→        if show_remediation:
   172→            _render_remediations(result)
   173→        _render_badge(result)
   174→        if output_file:
   175→            with open(output_file, "w") as fp:
   176→                fp.write(json.dumps(result.to_dict(), indent=2, default=str))
   177→            console.print(f"\n  [green]JSON saved: [cyan]{output_file}[/][/]")
   178→        # Auto-save to history
   179→        save_scan(result.to_dict())
   180→    console.print()

   183→def _spinner_wrap(msg: str, fn, *a, **kw):
   184→    """Run a function with a spinner."""
   185→    with Progress(SpinnerColumn(), TextColumn("[bold]{task.description}[/]"),
   186→                   TimeElapsedColumn(), console=console) as progress:
   187→        task = progress.add_task(msg, total=None)
   188→        result = fn(*a, **kw)
   189→        progress.update(task, completed=True)
   190→    return result

   460→    # ── REPORT ─────────────────────────────────────────────────────
