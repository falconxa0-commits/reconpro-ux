   246→// Cell renderer for comparison table

   249→function ComparisonCell({ value }: { value: boolean | string | false }) {
   250→  if (value === false) {
   251→    return <Minus className="w-4 h-4 text-white/20 mx-auto" />;

   255→  }
   256→  return <Check className="w-4 h-4 text-[#00ff88] mx-auto" />;
   257→}
