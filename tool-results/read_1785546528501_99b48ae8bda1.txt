    82→    82→// ─── Mini severity donut (SVG) ──────────────────────────────
    83→    83→
    84→    84→function SeverityDonut({ data }: { data: { name: string; value: number; color: string }[] }) {
    85→    85→  const total = data.reduce((s, d) => s + d.value, 0);
    86→    86→  const segments = data.map((d, i) => {
    87→    87→    const pct = total === 0 ? 0 : d.value / total;
    88→    88→    const off = data.slice(0, i).reduce((s2, prev) => s2 + (total === 0 ? 0 : prev.value / total), 0);
    89→    89→    return { ...d, pct, dasharray: `${pct * 283} ${283}`, offset: -off * 283 };
    90→    90→  });
    91→    91→  if (total === 0) return <div className="text-xs text-muted-foreground text-center py-4">No data yet</div>;
    92→    92→  return (
    93→    93→    <svg viewBox="0 0 120 120" className="w-32 h-32 mx-auto">
    94→    94→      {segments.map((d) => (
    95→    95→        <circle key={d.name} cx="60" cy="60" r="45" fill="none"
    96→    96→          stroke={d.color} strokeWidth="18" strokeDasharray={d.dasharray}
    97→    97→          strokeDashoffset={d.offset} strokeLinecap="round" opacity={0.85} />
    98→    98→      ))}
    99→    99→      <text x="60" y="56" textAnchor="middle" fill="#e6edf3" fontSize="22" fontWeight="bold" fontFamily="Geist Sans, sans-serif">{total}</text>
   100→   100→      <text x="60" y="72" textAnchor="middle" fill="#7d8590" fontSize="9" fontFamily="Geist Sans, sans-serif">FINDINGS</text>
   101→   101→    </svg>
   102→   102→  );
   103→   103→}
