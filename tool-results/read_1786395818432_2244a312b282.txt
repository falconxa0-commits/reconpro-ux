   614→   614→        """Return a list of WAF rules for *finding*.

   626→   626→            rules.append(tmpl)

   773→   773→    def generate_code_patch(self, finding: dict) -> str:
   774→   774→        """Generate a unified diff patch for *finding*.
   775→   775→
   776→   776→        Requires ``file`` (path) and ``line`` (1-based) in finding.
   777→   777→        """
   778→   778→        filepath = finding.get("file", "unknown.py")
   779→   779→        line_no = finding.get("line", 1)
   780→   780→        original = finding.get("original_line", "")
   781→   781→
   782→   782→        fix = self._resolve_code_fix(finding)
   783→   783→        fix_lines = fix.split("\n")
   784→   784→
   785→   785→        ctx_before = 3
   786→   786→        start = max(1, line_no - ctx_before)
   787→   787→        end = line_no + ctx_before
   788→   788→        old_count = end - start + 1
   789→   789→        new_count = old_count + len(fix_lines) - 1
   790→   790→
   791→   791→        diff_lines: list[str] = []
   792→   792→        diff_lines.append(f"--- a/{filepath}")
   793→   793→        diff_lines.append(f"+++ b/{filepath}")
   794→   794→        diff_lines.append(f"@@ -{start},{old_count} +{start},{new_count} @@")
   795→   795→
   796→   796→        for i in range(start, end + 1):
   797→   797→            if i == line_no:
   798→   798→                if original:
   799→   799→                    diff_lines.append(f"-{original}")
   800→   800→                else:
   801→   801→                    diff_lines.append(f"-# [REMOVED] vulnerable code at line {line_no}")
   802→   802→                for fl in fix_lines:
   803→   803→                    diff_lines.append(f"+{fl}")
   804→   804→            else:
   805→   805→                diff_lines.append(f" # (context line {i})")
   806→   806→
   807→   807→        return "\n".join(diff_lines) + "\n"

   809→   809→    def _resolve_code_fix(self, finding: dict) -> str:
   810→   810→        """Return the replacement code lines for a finding."""
   811→   811→        category = finding.get("category", "").lower()
   812→   812→        original = finding.get("original_line", "")
   813→   813→
   814→   814→        # Pattern-based detection on the original line
   815→   815→        if "eval(" in original:
   816→   816→            return (
   817→   817→                "import ast\n"
   818→   818→                "result = ast.literal_eval(sanitized_input)  # safe alternative to eval()"
   819→   819→            )
   820→   820→        if "exec(" in original:
   821→   821→            return (
   822→   822→                "# exec() removed \u2014 use a task queue, plugin system, or\n"
   823→   823→                "# sandboxed interpreter for dynamic code execution"
   824→   824→            )
   825→   825→        if "shell=True" in original:
   826→   826→            return (
   827→   827→                "subprocess.run(\n"
   828→   828→                '    ["command", "arg1"],\n'
   829→   829→                "    capture_output=True,\n"
   830→   830→                "    text=True,\n"
   831→   831→                "    shell=False,\n"
   832→   832→                ")"
   833→   833→            )
   834→   834→        if any(kw in original for kw in ("select ", "insert ", "update ", "delete ")) and ("f\"" in original or "f'" in original):
   835→   835→            return (
   836→   836→                'cursor.execute(\n'
   837→   837→                '    "SELECT ... WHERE col = %s",\n'
   838→   838→                '    (user_input,),\n'
   839→   839→                ")  # parameterized query"
   840→   840→            )
   841→   841→        if category == "hardcoded_secret" or any(
   842→   842→            k in original.lower() for k in ("secret", "password", "api_key", "token")
   843→   843→        ):
   844→   844→            var_match = re.match(r'(\w+)\s*=\s*["\']', original)
   845→   845→            var_name = var_match.group(1) if var_match else "CONFIG_VALUE"
   846→   846→            env_key = var_name.upper()
   847→   847→            return (
   848→   848→                "import os\n"
   849→   849→                f'{var_name} = os.environ.get("{env_key}")\n'
   850→   850→                f'if not {var_name}:\n'
   851→   851→                f'    raise RuntimeError("{env_key} environment variable is not set")'
   852→   852→            )
   853→   853→        if "DEBUG = True" in original or "DEBUG=True" in original:
   854→   854→            return "DEBUG = False"
   855→   855→        if "csrf_exempt" in original:
   856→   856→            return "# @csrf_exempt removed \u2014 CSRF protection required for this endpoint"
   857→   857→
   858→   858→        # Fall back to category template
   859→   859→        cat_entry = CATEGORY_PATCH_MAP.get(category, {})
   860→   860→        code_fix = cat_entry.get("code_fix", "")
   861→   861→        if code_fix:
   862→   862→            return code_fix.strip()
   863→   863→
   864→   864→        return "# Apply manual fix: review the vulnerability description and remediation"

   866→   866→    # -- Infrastructure fix generation -------------------------------------

   868→   868→    def generate_infra_fix(self, finding: dict) -> str:
   869→   869→        """Generate an infrastructure fix for *finding*."""
   870→   870→        category = finding.get("category", "").lower()
