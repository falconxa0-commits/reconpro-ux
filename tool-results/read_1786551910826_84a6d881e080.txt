    60→def _render_summary(result, title: str = "RECONPRO") -> None:
    61→    score = result.total_score
    62→    grade = result.grade
    63→    grade_color = GRADE_COLORS.get(grade, "bold bright_red")
    64→    sc = result.severity_counts
    65→    total = len(result.findings)
    66→    bar_width = 50
    67→    filled = int(score / 100 * bar_width)
    68→    bar = "█" * filled + "░" * (bar_width - filled)
    69→    lines = [
    70→        Text(f"\n  {bar} [bold {grade_color}]{score}/100 ({grade})[/{grade_color}]"),
    71→        Text(f"\n  Target: [cyan]{result.target}[/]"),
    72→        Text(f"  Modules: [dim]{', '.join(result.modules_run)}[/]"),
    73→        Text(""),
    74→        Text(
    75→            f"  Findings: [bold]{total}[/]  "
    76→            f"[bright_red]{sc.get('critical', 0)} critical[/], "
    77→            f"[red]{sc.get('high', 0)} high[/], "
    78→            f"[yellow]{sc.get('medium', 0)} medium[/], "
    79→            f"[green]{sc.get('low', 0)} low[/], "
    80→            f"[dim]{sc.get('info', 0)} info[/]",
    81→        ),
    82→    ]
    83→    if result.vibesec_score is not None:
    84→        lines.append(Text(
    85→            f"\n  VibeSec Score: [bold bright_green]{result.vibesec_score}/100 ({result.vibesec_grade})[/]"
    86→        ))
    87→    console.print(Panel(Group(*lines), border_style=grade_color,
    88→                         title=f"[bold]{title}[/bold]", title_align="left", padding=(1, 2)))

    91→def _render_findings_table(result, show_remediation: bool = False) -> None:
    92→    if not result.findings:
    93→        console.print("\n  [bright_green]No findings. Target is clean.[/]")
    94→        return
    95→    table = Table(title=f"Findings — {len(result.findings)} detected",
    96→                  border_style="bright_white", header_style="bold bright_white")
    97→    table.add_column("Severity", style="bold", width=10)
    98→    table.add_column("Module", style="cyan", width=12)
    99→    table.add_column("Category", style="magenta", width=18)
   100→    table.add_column("Finding", style="white")
   101→    table.add_column("Pts", style="yellow", width=5)
   102→    if show_remediation:
   103→        table.add_column("Fix", style="bright_green", width=50)
   104→    displayed = result.findings[:80]
   105→    for f in displayed:
   106→        sev = f.get("severity", "info")
   107→        color = SEV_COLORS.get(sev, "white")
   108→        row = [
   109→            f"[{color}]{sev.upper()}[/{color}]",
   110→            f.get("module", ""), f.get("category", ""),
   111→            f.get("title", "")[:70], str(f.get("points_deducted", 0)),
   112→        ]
   113→        if show_remediation:
   114→            row.append(f.get("remediation", "")[:50])
   115→        table.add_row(*row)
   116→    console.print(table)
   117→    if len(result.findings) > 80:
   118→        console.print(f"  [dim]Showing 80 of {len(result.findings)} findings. Use --json -o report.json for the full list.[/]")

   121→def _render_module_breakdown(result) -> None:
   122→    if not result.module_results:
   123→        return
   124→    t = Table(title="Module Breakdown", border_style="dim", header_style="bold dim")
   125→    t.add_column("Module", style="bold", width=16)
   126→    t.add_column("Findings", style="white", width=10)
   127→    t.add_column("Critical", style="bright_red", width=10)
   128→    t.add_column("High", style="red", width=8)
   129→    t.add_column("Medium", style="yellow", width=8)
   130→    for mod_id, mod_data in result.module_results.items():
   131→        findings = mod_data.get("findings", [])
   132→        t.add_row(mod_id.upper(), str(len(findings)),
   133→                   str(sum(1 for f in findings if f.get("severity") == "critical")),
   134→                   str(sum(1 for f in findings if f.get("severity") == "high")),
   135→                   str(sum(1 for f in findings if f.get("severity") == "medium")))
   136→    console.print(t)

   147→def _render_remediations(result) -> None:
   148→    actionable = [f for f in result.findings
   149→                  if f.get("remediation") and f.get("severity") not in ("info",)
   150→                  and f.get("points_deducted", 0) > 0]
   151→    if not actionable:
   152→        console.print("\n  [bright_green]No actionable fixes needed.[/]")
   153→        return
   154→    console.print("\n[bold bright_white]  Fix Commands:[/bold bright_white]\n")
   155→    seen = set()
   156→    for f in actionable:
   157→        fix = f.get("remediation", "").strip()
   158→        sev = f.get("severity", "info")
   159→        color = SEV_COLORS.get(sev, "white")
   160→        if fix and fix not in seen:
   161→            seen.add(fix)
   162→            console.print(f"  [{color}]{sev.upper():8}[/{color}]  {fix}")
   163→    console.print()

   166→def _output_result(result, args, title: str = "RECONPRO", show_remediation: bool = False) -> None:
   167→    json_output = getattr(args, "json_output", False)
   168→    output_file = getattr(args, "output_file", None)
   169→    if json_output:
   170→        text = json.dumps(result.to_dict(), indent=2, default=str)
   171→        if output_file:
   172→            with open(output_file, "w") as fp:
   173→                fp.write(text)
   174→            console.print(f"  [green]JSON saved: [cyan]{output_file}[/][/]")
   175→        else:
   176→            print(text)
   177→    else:
   178→        _render_summary(result, title=title)
   179→        console.print()
   180→        _render_module_breakdown(result)
   181→        console.print()
   182→        _render_findings_table(result, show_remediation=show_remediation)
   183→        if show_remediation:
   184→            _render_remediations(result)
   185→        _render_badge(result)
   186→        if output_file:
   187→            with open(output_file, "w") as fp:
   188→                fp.write(json.dumps(result.to_dict(), indent=2, default=str))
   189→            console.print(f"\n  [green]JSON saved: [cyan]{output_file}[/][/]")
   190→        # Auto-save to history
   191→        save_scan(result.to_dict())
   192→    console.print()

   205→
   206→def _spinner_wrap(msg: str, fn, *a, _json_mode: bool | None = None, **kw):
   207→    """Run a function with a spinner (suppressed in JSON mode)."""
   208→    if _json_mode is None:
   209→        _json_mode = getattr(_cli_args, 'json_output', False) if _cli_args else False
   210→    if _json_mode:
   211→        return fn(*a, **kw)
   212→    with Progress(SpinnerColumn(), TextColumn("[bold]{task.description}[/]"),
   213→                   TimeElapsedColumn(), console=console) as progress:
   214→        task = progress.add_task(msg, total=None)
   215→        result = fn(*a, **kw)
   216→        progress.update(task, completed=True)
   217→    return result
