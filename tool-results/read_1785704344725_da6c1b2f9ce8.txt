     1→   100→        if data:
     2→   101→            return list(self._nodes.items())
     3→   102→        return list(self._nodes.keys())
     4→   103→
     5→   104→    def edges(self, data: bool = False):
     6→   105→        result = []
     7→   106→        for u, nbrs in self._succ.items():
     8→   107→            for v, attr in nbrs.items():
     9→   108→                if data:
    10→   109→                    result.append((u, v, attr))
    11→   110→                else:
    12→   111→                    result.append((u, v))
    13→   112→        return result
    14→   113→
    15→   114→    def successors(self, node: str):
    16→   115→        return list(self._succ.get(node, {}).keys())
    17→   116→
    18→   117→    def predecessors(self, node: str):
    19→   118→        return list(self._pred.get(node, {}).keys())
    20→   119→
    21→   120→    def out_edges(self, node: str, data: bool = False):
    22→   121→        if data:
    23→   122→            return [(node, v, a) for v, a in self._succ.get(node, {}).items()]
    24→   123→        return [(node, v) for v in self._succ.get(node, {}).keys()]
    25→   124→
    26→   125→    def in_edges(self, node: str, data: bool = False):
    27→   126→        if data:
    28→   127→            return [(u, node, a) for u, a in self._pred.get(node, {}).items()]
    29→   128→        return [(u, node) for u in self._pred.get(node, {}).keys()]
    30→   129→
    31→   130→    def has_node(self, node: str) -> bool:
    32→   131→        return node in self._nodes
    33→   132→
    34→   133→    def get_node_data(self, node: str) -> Dict[str, Any]:
    35→   134→        return dict(self._nodes.get(node, {}))
    36→   135→
    37→   136→    def subgraph(self, nodes: Set[str]):
    38→   137→        g = _FallbackDiGraph()
    39→   138→        for n in nodes:
    40→   139→            if n in self._nodes:
    41→   140→                g.add_node(n, **self._nodes[n])
    42→   141→        for u, nbrs in self._succ.items():
    43→   142→            if u in nodes:
    44→   143→                for v, attr in nbrs.items():
    45→   144→                    if v in nodes:
    46→   145→                        g.add_edge(u, v, **attr)
    47→   146→        return g
    48→   147→
    49→   148→    def to_dict(self) -> Dict[str, Any]:
    50→   149→        return {
    51→   150→            "nodes": dict(self._nodes),
    52→   151→            "edges": [
    53→   152→                {"source": u, "target": v, **a}
    54→   153→                for u, nbrs in self._succ.items()
    55→   154→                for v, a in nbrs.items()
    56→   155→            ],
    57→   156→        }
    58→   157→
    59→   158→    @classmethod
    60→   159→    def from_dict(cls, data: Dict[str, Any]):
    61→   160→        g = cls()
    62→   161→        for node, attr in data.get("nodes", {}).items():
    63→   162→            g.add_node(node, **attr)
    64→   163→        for edge in data.get("edges", []):
    65→   164→            src = edge.pop("source")
    66→   165→            tgt = edge.pop("target")
    67→   166→            g.add_edge(src, tgt, **edge)
    68→   167→        return g
    69→   168→
    70→   169→
    71→   170→def _make_graph():
    72→   171→    """Return a NetworkX DiGraph if available, else the fallback."""
    73→   172→    if HAS_NX:
    74→   173→        return nx.DiGraph()
    75→   174→    return _FallbackDiGraph()

    82→   181→class SecurityKnowledgeGraph:
    83→   182→    """Directed knowledge graph for security scan results.
    84→   183→
    85→   184→    Wraps a NetworkX :class:`DiGraph` when *networkx* is installed;
    86→   185→    otherwise falls back to a lightweight pure-Python implementation.

    88→   187→
    89→   188→    def __init__(self) -> None:
    90→   189→        self._g = _make_graph()
    91→   190→        self._node_type_index: Dict[str, Set[str]] = defaultdict(set)
    92→   191→        self._edge_type_index: Dict[str, List[Tuple[str, str]]] = defaultdict(list)
    93→   192→
    94→   193→    # ------------------------------------------------------------------
    95→   194→    # Internal helpers

    98→   197→    def _node_id(self, ntype: str, label: str) -> str:
    99→   198→        """Deterministic node id: ``<type>:<label>``."""
   100→   199→        return f"{ntype}:{label}"
   101→   200→
   102→   201→    def _add_indexed_node(self, ntype: str, label: str, **attrs: Any) -> str:
   103→   202→        nid = self._node_id(ntype, label)
   104→   203→        self._g.add_node(nid, node_type=ntype, label=label, **attrs)
   105→   204→        self._node_type_index[ntype].add(nid)
   106→   205→        return nid
   107→   206→
   108→   207→    def _add_indexed_edge(self, src: str, dst: str, etype: str, **attrs: Any) -> None:
   109→   208→        self._g.add_edge(src, dst, edge_type=etype, **attrs)
   110→   209→        self._edge_type_index[etype].append((src, dst))

   112→   211→    def _safe_sev(self, raw) -> str:
   113→   212→        """Normalise a severity value to lowercase string."""
   114→   213→        if raw is None:
   115→   214→            return "info"
   116→   215→        s = str(raw).strip().lower()
   117→   216→        if s in SEVERITY_ORDER:
   118→   217→            return s
   119→   218→        return "info"

   125→   224→    def add_scan_result(self, result_dict: Dict[str, Any]) -> None:
   126→   225→        """Ingest a full ``ReconProResult.to_dict()`` payload.
   127→   226→
   128→   227→        Creates nodes/edges for the target, every finding, associated
   129→   228→        technologies, and open ports.
   130→   229→        """
   131→   230→        target = result_dict.get("target") or result_dict.get("host", "unknown")
   132→   231→        self._add_indexed_node("target", target)
   133→   232→
   134→   233→        # Subdomains
   135→   234→        for sub in result_dict.get("subdomains", []):
   136→   235→            self.add_subdomain(sub, target)
   137→   236→
   138→   237→        # Ports & services
   139→   238→        for port_entry in result_dict.get("ports", []):
   140→   239→            if isinstance(port_entry, dict):
   141→   240→                port_num = port_entry.get("port", port_entry.get("number"))
   142→   241→                service = port_entry.get("service", port_entry.get("name", "unknown"))
   143→   242→                state = port_entry.get("state", "open")
   144→   243→            else:
   145→   244→                port_num = port_entry
   146→   245→                service = "unknown"
   147→   246→                state = "open"
   148→   247→
   149→   248→            if port_num is None:
   150→   249→                continue
   151→   250→
   152→   251→            port_id = self._add_indexed_node("port", f"{target}:{port_num}", port=int(port_num), state=state)
   153→   252→            target_id = self._node_id("target", target)
   154→   253→            self._add_indexed_edge(target_id, port_id, "has_port")
   155→   254→
   156→   255→            svc_id = self._add_indexed_node("service", f"{target}:{port_num}:{service}", name=service, port=int(port_num))
   157→   256→            self._add_indexed_edge(port_id, svc_id, "runs_on")
   158→   257→
   159→   258→        # Technologies
   160→   259→        for tech in result_dict.get("technologies", []):
   161→   260→            tech_name = tech if isinstance(tech, str) else tech.get("name", str(tech))
   162→   261→            tech_id = self._add_indexed_node("technology", tech_name)
   163→   262→            target_id = self._node_id("target", target)
   164→   263→            self._add_indexed_edge(target_id, tech_id, "uses_tech")
   165→   264→
   166→   265→        # Findings
   167→   266→        for finding in result_dict.get("findings", []):
   168→   267→            self.add_finding(finding)
   169→   268→
   170→   269→        # Certificates
   171→   270→        for cert in result_dict.get("certificates", []):
   172→   271→            if isinstance(cert, dict):
   173→   272→                cn = cert.get("common_name", cert.get("cn", "unknown"))
   174→   273→                issuer = cert.get("issuer", "")
   175→   274→                expires = cert.get("expires", "")
   176→   275→            else:
   177→   276→                cn = str(cert)
   178→   277→                issuer = ""
   179→   278→                expires = ""
   180→   279→            cert_id = self._add_indexed_node("certificate", f"{target}:{cn}", common_name=cn, issuer=issuer, expires=expires)
   181→   280→            target_id = self._node_id("target", target)
   182→   281→            self._add_indexed_edge(target_id, cert_id, "has_cert")
   183→   282→
   184→   283→        # Endpoints
   185→   284→        for ep in result_dict.get("endpoints", []):
   186→   285→            if isinstance(ep, dict):
   187→   286→                path = ep.get("path", ep.get("url", str(ep)))
   188→   287→                method = ep.get("method", "GET")
   189→   288→            else:
   190→   289→                path = str(ep)
   191→   290→                method = "GET"
   192→   291→            ep_id = self._add_indexed_node("endpoint", f"{target}:{method}:{path}", path=path, method=method)
   193→   292→            target_id = self._node_id("target", target)
   194→   293→            self._add_indexed_edge(target_id, ep_id, "exposes")
   195→   294→
   196→   295→    def add_finding(self, finding_dict: Dict[str, Any]) -> None:
   197→   296→        """Add a finding node and connect it to its parent target."""
   198→   297→        title = finding_dict.get("title", finding_dict.get("name", finding_dict.get("description", "unnamed")))
   199→   298→        severity = self._safe_sev(finding_dict.get("severity"))
   200→   299→        target = finding_dict.get("target", finding_dict.get("host", "unknown"))
   201→   300→        finding_type = finding_dict.get("type", finding_dict.get("category", "generic"))
   202→   301→        description = finding_dict.get("description", "")
   203→   302→        evidence = finding_dict.get("evidence", "")
   204→   303→        cvss_score = finding_dict.get("cvss_score", finding_dict.get("cvss", None))
   205→   304→
   206→   305→        fid = self._add_indexed_node(
   207→   306→            "finding",
   208→   307→            f"{target}:{title}",
   209→   308→            title=title,
   210→   309→            severity=severity,
   211→   310→            finding_type=finding_type,
   212→   311→            description=description,
   213→   312→            evidence=evidence,
   214→   313→            cvss_score=cvss_score,
   215→   314→        )
   216→   315→
   217→   316→        # Ensure target node exists
   218→   317→        self._add_indexed_node("target", target)
   219→   318→        target_id = self._node_id("target", target)
   220→   319→        self._add_indexed_edge(target_id, fid, "has_vuln")
   221→   320→
   222→   321→        # Link to port if specified
   223→   322→        port_num = finding_dict.get("port")
   224→   323→        if port_num is not None:
   225→   324→            port_id = self._node_id("port", f"{target}:{port_num}")
   226→   325→            if self._g.has_node(port_id):
   227→   326→                self._add_indexed_edge(port_id, fid, "has_vuln")
   228→   327→
   229→   328→        # Link related CVEs
   230→   329→        for cve_ref in finding_dict.get("cves", finding_dict.get("references", [])):
   231→   330→            if isinstance(cve_ref, dict):
   232→   331→                cve_id = cve_ref.get("id", cve_ref.get("cve_id", ""))
   233→   332→                cve_sev = cve_ref.get("severity", severity)
   234→   333→                cve_desc = cve_ref.get("description", "")
   235→   334→                affected = cve_ref.get("affected", "")
   236→   335→            else:
   237→   336→                cve_id = str(cve_ref)
   238→   337→                cve_sev = severity
   239→   338→                cve_desc = ""
   240→   339→                affected = ""
   241→   340→            if cve_id:
   242→   341→                self.add_cve(cve_id, cve_sev, cve_desc, affected)
   243→   342→                self._add_indexed_edge(fid, self._node_id("cve", cve_id), "related_cve")
   244→   343→
   245→   344→        # Link related technologies
   246→   345→        for tech in finding_dict.get("technologies", []):
   247→   346→            tech_name = tech if isinstance(tech, str) else tech.get("name", str(tech))
   248→   347→            tech_id = self._add_indexed_node("technology", tech_name)
   249→   348→            self._add_indexed_edge(tech_id, fid, "exposes")
   250→   349→
   251→   350→    def add_cve(
   252→   351→        self,
   253→   352→        cve_id: str,
   254→   353→        severity: str = "info",
   255→   354→        description: str = "",
   256→   355→        affected: str = "",
   257→   356→    ) -> None:
   258→   357→        """Add a CVE node (idempotent) and return its node id."""
   259→   358→        sev = self._safe_sev(severity)
   260→   359→        nid = self._add_indexed_node(
   261→   360→            "cve",
   262→   361→            cve_id,
   263→   362→            severity=sev,
   264→   363→            description=description,
   265→   364→            affected=affected,
   266→   365→        )
   267→   366→        return nid

   269→   368→    def add_subdomain(self, domain: str, parent: str) -> None:
   270→   369→        """Link *domain* as a subdomain of *parent*."""
   271→   370→        sub_id = self._add_indexed_node("subdomain", domain)
   272→   371→        parent_id = self._node_id("target", parent)
   273→   372→        # Parent might be a subdomain itself
   274→   373→        if not self._g.has_node(parent_id):
   275→   374→            parent_id = self._node_id("subdomain", parent)
   276→   375→            if not self._g.has_node(parent_id):
   277→   376→                self._add_indexed_node("target", parent)
   278→   377→                parent_id = self._node_id("target", parent)
   279→   378→        self._add_indexed_edge(parent_id, sub_id, "has_subdomain")

   285→   384→    def get_attack_surface(self, target: str) -> Dict[str, List[Dict[str, Any]]]:
   286→   385→        """Return all reachable endpoints, open ports, and vulns for *target*.
   287→   386→
   288→   387→        Returns a dict with keys ``"endpoints"``, ``"ports"``,
   289→   388→        ``"vulnerabilities"``, and ``"technologies"``.
   290→   389→        """
   291→   390→        target_id = self._node_id("target", target)
   292→   391→        if not self._g.has_node(target_id):
   293→   392→            # Check subdomains
   294→   393→            target_id = self._node_id("subdomain", target)
   295→   394→            if not self._g.has_node(target_id):
   296→   395→                return {"endpoints": [], "ports": [], "vulnerabilities": [], "technologies": []}
   297→   396→
   298→   397→        visited = self._bfs_collect(target_id, max_depth=3)
   299→   398→        result: Dict[str, List[Dict[str, Any]]] = {
   300→   399→            "endpoints": [],
   301→   400→            "ports": [],
   302→   401→            "vulnerabilities": [],
   303→   402→            "technologies": [],
   304→   403→        }
   305→   404→
   306→   405→        for nid in visited:
   307→   406→            if HAS_NX:
   308→   407→                data = self._g.nodes[nid]
   309→   408→            else:
   310→   409→                data = self._g.get_node_data(nid)
   311→   410→            ntype = data.get("node_type", "")
   312→   411→            if ntype == "endpoint":
   313→   412→                result["endpoints"].append(data)
   314→   413→            elif ntype == "port":
   315→   414→                if data.get("state", "open") == "open":
   316→   415→                    result["ports"].append(data)
   317→   416→            elif ntype in ("vulnerability", "finding"):
   318→   417→                result["vulnerabilities"].append(data)
   319→   418→            elif ntype == "technology":
   320→   419→                result["technologies"].append(data)
   321→   420→
   322→   421→        return result

   324→   423→    def get_blast_radius(self, asset: str) -> List[Dict[str, Any]]:
   325→   424→        """BFS from *asset* to find all connected vulnerabilities.
   326→   425→
   327→   426→        *asset* can be a node label, an IP, or a partial node id.
   328→   427→        """
   329→   428→        start = self._resolve_node(asset)
   330→   429→        if start is None:
   331→   430→            return []
   332→   431→
   333→   432→        visited = self._bfs_collect(start, max_depth=10)
   334→   433→        vulns: List[Dict[str, Any]] = []
   335→   434→        for nid in visited:
   336→   435→            if HAS_NX:
   337→   436→                data = self._g.nodes[nid]
   338→   437→            else:
   339→   438→                data = self._g.get_node_data(nid)
   340→   439→            if data.get("node_type") in ("vulnerability", "finding", "cve"):
   341→   440→                vulns.append(data)
   342→   441→        return vulns

   344→   443→    def find_chains(self, target: str) -> List[List[Dict[str, Any]]]:
   345→   444→        """Find multi-hop attack chains originating from *target*.
   346→   445→
   347→   446→        Each chain follows a pattern like::
   348→   447→
   349→   448→            target → port → service → vulnerability → CVE
   350→   449→        """
   351→   450→        target_id = self._resolve_node(target)
   352→   451→        if target_id is None:
   353→   452→            return []
   354→   453→
   355→   454→        chains: List[List[Dict[str, Any]]] = []
   356→   455→        # DFS to find paths that end at CVE or finding nodes
   357→   456→        stack: List[Tuple[str, List[str]]] = [(target_id, [target_id])]
   358→   457→        seen_chains: Set[tuple] = set()
   359→   458→
   360→   459→        while stack:
   361→   460→            current, path = stack.pop()
   362→   461→            succs = self._g.successors(current) if HAS_NX else self._g.successors(current)
   363→   462→            for nxt in succs:
   364→   463→                if nxt in path:
   365→   464→                    continue
   366→   465→                new_path = path + [nxt]
   367→   466→                if HAS_NX:
   368→   467→                    nd = self._g.nodes[nxt]
   369→   468→                else:
   370→   469→                    nd = self._g.get_node_data(nxt)
   371→   470→                ntype = nd.get("node_type", "")
   372→   471→                # A chain is complete if it reaches a CVE or high-sev finding
   373→   472→                if ntype == "cve" or (ntype == "finding" and SEVERITY_ORDER.get(self._safe_sev(nd.get("severity")), 0) >= 2):
   374→   473→                    chain_key = tuple(new_path)
   375→   474→                    if chain_key not in seen_chains:
   376→   475→                        seen_chains.add(chain_key)
   377→   476→                        chain_data = []
   378→   477→                        for nid in new_path:
   379→   478→                            if HAS_NX:
   380→   479→                                chain_data.append(dict(self._g.nodes[nid]))
   381→   480→                            else:
   382→   481→                                chain_data.append(self._g.get_node_data(nid))
   383→   482→                        chains.append(chain_data)
   384→   483→                elif len(new_path) < 8:
   385→   484→                    stack.append((nxt, new_path))
   386→   485→
   387→   486→        # Sort by severity of the terminal node
   388→   487→        def _chain_sort_key(chain):
   389→   488→            last = chain[-1]
   390→   489→            return SEVERITY_ORDER.get(self._safe_sev(last.get("severity")), 0)
   391→   490→
   392→   491→        chains.sort(key=_chain_sort_key, reverse=True)
   393→   492→        return chains

   395→   494→    def get_vulnerable_assets(self) -> List[Dict[str, Any]]:
   396→   495→        """Return all assets (targets/subdomains) with HIGH or CRITICAL findings."""
   397→   496→        vulnerable: List[Dict[str, Any]] = []
   398→   497→        seen: Set[str] = set()
   399→   498→
   400→   499→        for nid in self._node_type_index.get("finding", set()):
   401→   500→            if HAS_NX:
   402→   501→                data = self._g.nodes[nid]
   403→   502→            else:
   404→   503→                data = self._g.get_node_data(nid)
   405→   504→            sev = self._safe_sev(data.get("severity"))
   406→   505→            if SEVERITY_ORDER.get(sev, 0) < 3:
   407→   506→                continue
   408→   507→            # Walk predecessors to find the owning target/subdomain
   409→   508→            for pred in (self._g.predecessors(nid) if HAS_NX else self._g.predecessors(nid)):
   410→   509→                if HAS_NX:
   411→   510→                    pdata = self._g.nodes[pred]
   412→   511→                else:
   413→   512→                    pdata = self._g.get_node_data(pred)
   414→   513→                ptype = pdata.get("node_type", "")
   415→   514→                if ptype in ("target", "subdomain", "asset") and pred not in seen:
   416→   515→                    seen.add(pred)
   417→   516→                    vulnerable.append({"node_id": pred, **pdata, "critical_finding": data})
   418→   517→        return vulnerable

   420→   519→    def correlate(self) -> List[Dict[str, Any]]:
   421→   520→        """Cross-reference findings to suggest related vulnerabilities.
   422→   521→
   423→   522→        Returns a list of correlation suggestions, each containing
   424→   523→        the two related node ids, the relationship type, and a
   425→   524→        confidence score.
   426→   525→        """
   427→   526→        suggestions: List[Dict[str, Any]] = []
   428→   527→        # Group findings by technology
   429→   528→        tech_findings: Dict[str, List[str]] = defaultdict(list)
   430→   529→        for nid in self._node_type_index.get("finding", set()):
   431→   530→            if HAS_NX:
   432→   531→                data = self._g.nodes[nid]
   433→   532→            else:
   434→   533→                data = self._g.get_node_data(nid)
   435→   534→            for pred in (self._g.predecessors(nid) if HAS_NX else self._g.predecessors(nid)):
   436→   535→                if HAS_NX:
   437→   536→                    pdata = self._g.nodes[pred]
   438→   537→                else:
   439→   538→                    pdata = self._g.get_node_data(pred)
   440→   539→                if pdata.get("node_type") == "technology":
   441→   540→                    tech_findings[pred].append(nid)
   442→   541→
   443→   542→        for tech_id, finding_ids in tech_findings.items():
   444→   543→            if len(finding_ids) < 2:
   445→   544→                continue
   446→   545→            for i in range(len(finding_ids)):
   447→   546→                for j in range(i + 1, len(finding_ids)):
   448→   547→                    f1 = finding_ids[i]
   449→   548→                    f2 = finding_ids[j]
   450→   549→                    if HAS_NX:
   451→   550→                        d1 = self._g.nodes[f1]
   452→   551→                        d2 = self._g.nodes[f2]
   453→   552→                    else:
   454→   553→                        d1 = self._g.get_node_data(f1)
   455→   554→                        d2 = self._g.get_node_data(f2)
   456→   555→                    # If both share a technology, suggest they may be related
   457→   556→                    conf = 0.6
   458→   557→                    if d1.get("severity") == d2.get("severity"):
   459→   558→                        conf += 0.2
   460→   559→                    if d1.get("finding_type") == d2.get("finding_type"):
   461→   560→                        conf += 0.2
   462→   561→                    suggestions.append({
   463→   562→                        "source": f1,
   464→   563→                        "target": f2,
   465→   564→                        "relationship": "shared_technology",
   466→   565→                        "technology": tech_id,
   467→   566→                        "confidence": min(conf, 1.0),
   468→   567→                        "source_severity": d1.get("severity"),
   469→   568→                        "target_severity": d2.get("severity"),
   470→   569→                    })
   471→   570→
   472→   571→        # Also correlate by port: findings on the same port may be related
   473→   572→        port_findings: Dict[str, List[str]] = defaultdict(list)
   474→   573→        for nid in self._node_type_index.get("finding", set()):
   475→   574→            for pred in (self._g.predecessors(nid) if HAS_NX else self._g.predecessors(nid)):
   476→   575→                if HAS_NX:
   477→   576→                    pdata = self._g.nodes[pred]
   478→   577→                else:
   479→   578→                    pdata = self._g.get_node_data(pred)
   480→   579→                if pdata.get("node_type") == "port":
   481→   580→                    port_findings[pred].append(nid)
   482→   581→
   483→   582→        for port_id, fids in port_findings.items():
   484→   583→            if len(fids) < 2:
   485→   584→                continue
   486→   585→            for i in range(len(fids)):
   487→   586→                for j in range(i + 1, len(fids)):
   488→   587→                    f1, f2 = fids[i], fids[j]
   489→   588→                    suggestions.append({
   490→   589→                        "source": f1,
   491→   590→                        "target": f2,
   492→   591→                        "relationship": "shared_port",
   493→   592→                        "port": port_id,
   494→   593→                        "confidence": 0.7,
   495→   594→                    })
   496→   595→
   497→   596→        suggestions.sort(key=lambda s: s["confidence"], reverse=True)
   498→   597→        return suggestions

   500→   599→    def to_cypher(self) -> str:
   501→   600→        """Export the graph as Cypher-like CREATE statements."""
   502→   601→        lines: List[str] = []
   503→   602→        # Nodes
   504→   603→        for nid in (self._g.nodes() if HAS_NX else self._g.nodes()):
   505→   604→            if HAS_NX:
   506→   605→                data = self._g.nodes[nid]
   507→   606→            else:
   508→   607→                data = self._g.get_node_data(nid)
   509→   608→            ntype = data.pop("node_type", "Unknown")
   510→   609→            label = data.pop("label", nid)
   511→   610→            props = json.dumps(data, default=str)
   512→   611→            safe_id = nid.replace(":", "__").replace("-", "_")
   513→   612→            lines.append(f'CREATE ({safe_id}:{ntype} {{label: "{label}", {props[1:-1]}}})')
   514→   613→            # Restore for edge generation below
   515→   614→            data["node_type"] = ntype
   516→   615→            data["label"] = label
   517→   616→
   518→   617→        # Edges
   519→   618→        for u, v, attr in (self._g.edges(data=True) if HAS_NX else self._g.edges(data=True)):
   520→   619→            etype = attr.get("edge_type", "RELATED_TO")
   521→   620→            props = {k: v for k, v in attr.items() if k != "edge_type"}
   522→   621→            props_str = json.dumps(props, default=str)
   523→   622→            safe_u = u.replace(":", "__").replace("-", "_")
   524→   623→            safe_v = v.replace(":", "__").replace("-", "_")
   525→   624→            if props_str == "{}":
   526→   625→                lines.append(f'CREATE ({safe_u})-[:{etype}]->({safe_v})')
   527→   626→            else:
   528→   627→                lines.append(f'CREATE ({safe_u})-[:{etype} {{{props_str[1:-1]}}}]->({safe_v})')
   529→   628→
   530→   629→        return "\n".join(lines)

   534→   633→        node_counts: Dict[str, int] = {}

   634→   733→    def _resolve_node(self, label: str) -> Optional[str]:
   635→   734→        """Try to resolve a human-readable label to a full node id."""
   636→   735→        # Exact node id
   637→   736→        if self._g.has_node(label):
   638→   737→            return label
   639→   738→        # Try as target
   640→   739→        candidate = self._node_id("target", label)
   641→   740→        if self._g.has_node(candidate):
   642→   741→            return candidate
   643→   742→        # Try as subdomain
   644→   743→        candidate = self._node_id("subdomain", label)
   645→   744→        if self._g.has_node(candidate):
   646→   745→            return candidate
   647→   746→        # Try as asset
   648→   747→        candidate = self._node_id("asset", label)
   649→   748→        if self._g.has_node(candidate):
   650→   749→            return candidate
   651→   750→        # Fuzzy: search all node labels
   652→   751→        for nid in self._g.nodes():
   653→   752→            if HAS_NX:
   654→   753→                data = self._g.nodes[nid]
   655→   754→            else:
   656→   755→                data = self._g.get_node_data(nid)
   657→   756→            if data.get("label") == label or nid.endswith(f":{label}"):
   658→   757→                return nid
   659→   758→        return None

   661→   760→    def _bfs_collect(self, start: str, max_depth: int = 5) -> Set[str]:
   662→   761→        """BFS from *start* up to *max_depth* hops, return visited nodes."""
   663→   762→        visited: Set[str] = set()
   664→   763→        queue: deque = deque()
   665→   764→        queue.append((start, 0))
   666→   765→        visited.add(start)
   667→   766→        while queue:
   668→   767→            node, depth = queue.popleft()
   669→   768→            if depth >= max_depth:
   670→   769→                continue
   671→   770→            for neighbour in (self._g.successors(node) if HAS_NX else self._g.successors(node)):
   672→   771→                if neighbour not in visited:
   673→   772→                    visited.add(neighbour)
   674→   773→                    queue.append((neighbour, depth + 1))
   675→   774→        return visited
