    95→        border: 'rgba(244,63,94,0.3)',
    96→        icon: XCircle,
    97→        label: 'CRITICAL',
    98→      };
    99→    case 'high':
   100→      return {
   101→        color: '#ff8844',
   102→        bg: 'rgba(251,191,36,0.15)',
   103→        border: 'rgba(251,191,36,0.3)',
   104→        icon: AlertTriangle,
   105→        label: 'HIGH',
   106→      };
   107→    case 'medium':
   108→      return {
   109→        color: '#d29922',
   110→        bg: 'rgba(210,153,34,0.15)',
   111→        border: 'rgba(210,153,34,0.3)',
   112→        icon: AlertCircle,
   113→        label: 'MEDIUM',
   114→      };
   115→    case 'low':
   116→      return {
   117→        color: '#44aaff',
   118→        bg: 'rgba(88,166,255,0.15)',
   119→        border: 'rgba(88,166,255,0.3)',
   120→        icon: FileWarning,
   121→        label: 'LOW',
   122→      };
   123→    default:
   124→      return {
   125→        color: '#444444',
   126→        bg: 'rgba(139,148,158,0.15)',
   127→        border: 'rgba(139,148,158,0.3)',
   128→        icon: AlertCircle,
   129→        label: 'INFO',
   130→      };
   131→  }

   134→function getScheduleColor(schedule: string): string {
   135→  switch (schedule) {
   136→    case 'hourly':
   137→      return 'bg-[rgba(52,211,153,0.15)] text-[#00ff88] border-[rgba(52,211,153,0.3)]';
   138→    case 'daily':
   139→      return 'bg-[rgba(88,166,255,0.15)] text-[#44aaff] border-[rgba(88,166,255,0.3)]';
   140→    case 'weekly':
   141→      return 'bg-[rgba(210,153,34,0.15)] text-[#d29922] border-[rgba(210,153,34,0.3)]';
   142→    case 'monthly':
   143→      return 'bg-[rgba(168,85,247,0.15)] text-[#888888] border-[rgba(168,85,247,0.3)]';
   144→    default:
   145→      return 'bg-[rgba(139,148,158,0.15)] text-[#444444] border-[rgba(139,148,158,0.3)]';
   146→  }
   147→}

   149→function getPolicyStatusConfig(status: string) {
   150→  switch (status) {
   151→    case 'active':
   152→      return { color: '#00ff88', label: 'Active' };
   153→    case 'warning':
   154→      return { color: '#d29922', label: 'Warning' };
   155→    case 'error':
   156→      return { color: '#ff3355', label: 'Error' };
   157→    case 'idle':
   158→      return { color: '#333333', label: 'Paused' };
   159→    default:
   160→      return { color: '#333333', label: 'Unknown' };
   161→  }
   162→}
   163→
   164→function getScanTypeIcon(type: string) {
   165→  switch (type) {
   166→    case 'Full Infrastructure':
   167→    case 'full':
   168→      return Server;
   169→    case 'API Surface':
   170→    case 'api':
   171→      return Globe;
   172→    case 'SSL/TLS':
   173→    case 'ssl':
   174→      return Shield;
   175→    case 'DNS Recon':
   176→    case 'dns':
   177→      return MonitorCheck;
   178→    default:
   179→      return Activity;
   180→  }
   181→}
   182→
   183→function getAlertStatusConfig(status: string) {
   184→  switch (status) {
   185→    case 'new':
   186→      return { color: '#ff3355', bg: 'rgba(244,63,94,0.15)', label: 'NEW' };
   187→    case 'acknowledged':
   188→      return { color: '#d29922', bg: 'rgba(210,153,34,0.15)', label: 'ACK' };
   189→    case 'resolved':
   190→      return { color: '#00ff88', bg: 'rgba(52,211,153,0.15)', label: 'FIXED' };
   191→    default:
   192→      return { color: '#333333', bg: 'rgba(72,79,88,0.15)', label: 'N/A' };

   215→export function MonitoringPanel() {

   217→  const [loading, setLoading] = useState(true);

   228→      setData({
   229→        policies: json.policies || [],
   230→        scheduledRuns: json.scheduledRuns || [],

   239→
   240→  useEffect(() => {
   241→    fetchData();
   242→  }, [fetchData]);
   243→
   244→  const policies = data?.policies || [];
   245→  const scheduledRuns = data?.scheduledRuns || [];
   246→  const alerts = data?.alerts || [];

   248→  const togglePolicy = async (id: string) => {
   249→    const policy = policies.find((p) => p.id === id);

   252→      await fetch('/api/monitoring', {
   253→        method: 'PATCH',
   254→        headers: { 'Content-Type': 'application/json' },
   255→        body: JSON.stringify({ id, enabled: !policy.enabled }),
   256→      });
   257→      await fetchData();
   258→    } catch (err) {
   259→      console.error('Failed to toggle policy:', err);
   260→    }
   261→  };
   262→
   263→  const deletePolicy = async (id: string) => {
   264→    try {
   265→      await fetch(`/api/monitoring?id=${id}`, { method: 'DELETE' });
   266→      await fetchData();
   267→    } catch (err) {
   268→      console.error('Failed to delete policy:', err);
   269→    }
   270→  };
   271→
   272→  const handleNewPolicy = async () => {
   273→    if (!newName || !newDomain) return;
   274→    try {
   275→      await fetch('/api/monitoring', {
   276→        method: 'POST',
   277→        headers: { 'Content-Type': 'application/json' },
   278→        body: JSON.stringify({
   279→          name: newName,
   280→          targetDomain: newDomain,
   281→          schedule: newSchedule,
   282→          scanType: newScanType,
   283→        }),
   284→      });

   286→      setNewName('');
   287→      setNewDomain('');
   288→      setNewSchedule('daily');
   289→      setNewScanType('full');

   292→      console.error('Failed to create policy:', err);

   297→    console.log('Run policy now:', id);
   298→  };
   299→
   300→  const activePolicies = policies.filter((p) => p.enabled).length;
   301→  const totalFindings = policies.reduce((acc, p) => acc + p.findings, 0);
   302→
   303→  if (loading) {

   322→            <Activity className="w-5 h-5 text-[#00ff88]" />
   323→          </div>
   324→          <div>
   325→            <h2 className="text-xl font-bold text-[#f0f0f0]">Continuous Monitoring</h2>
   326→            <p className="text-sm text-[#444444]">Automated attack surface surveillance and policy management</p>
   327→          </div>
