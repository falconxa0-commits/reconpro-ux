   349→   349→// ── Helper: APT Risk Assessment ──────────────────────────────────────────────
   350→   350→
   351→   351→export function assessAPTRisk(industry: Industry, protocols: string[]): {
   352→   352→  activeThreats: APTGroup[];
   353→   353→  riskScore: number;
   354→   354→  targeting: string;
   355→   355→  recommendations: string[];
   356→   356→} {
   357→   357→  const activeThreats = APT_GROUPS.filter(g => g.targets.includes(industry));
   358→   358→
   359→   359→  // Boost groups that target the specific protocols in use
   360→   360→  const protocolRiskBoost = protocols.length;
   361→   361→  const criticalProtocolCount = protocols.filter(p => {
   362→   362→    const proto = SCADA_PROTOCOLS[p];
   363→   363→    return proto && (proto.risk === 'critical' || proto.risk === 'high');
   364→   364→  }).length;
   365→   365→
   366→   366→  const riskScore = Math.min(100,
   367→   367→    activeThreats.filter(t => t.severity === 'critical').length * 18 +
   368→   368→    activeThreats.filter(t => t.severity === 'high').length * 10 +
   369→   369→    criticalProtocolCount * 5 +
   370→   370→    protocolRiskBoost * 2
   371→   371→  );
   372→   372→
   373→   373→  const targeting = activeThreats.length === 0
   374→   374→    ? 'No known APT groups specifically targeting this industry-protocol combination'
   375→   375→    : `${activeThreats.length} APT groups actively target the ${industry} sector, ${activeThreats.filter(t => t.severity === 'critical').length} with critical severity`;
   376→   376→
   377→   377→  const recommendations: string[] = [];
   378→   378→  if (activeThreats.length > 0) {
   379→   379→    recommendations.push('Implement network segmentation with unidirectional gateways between IT and OT zones');
   380→   380→    recommendations.push('Deploy OT-specific intrusion detection systems (IDS) at all zone boundaries');
   381→   381→    recommendations.push('Enable deep packet inspection for all SCADA/ICS protocols in use');
   382→   382→  }
   383→   383→  if (criticalProtocolCount > 0) {

   616→   616→function analyzeProtocols(protocols: string[], networkSegment: NetworkSegment, industry: Industry): ProtocolAnalysis[] {
   617→   617→  return protocols.map(key => {
   618→   618→    const proto = SCADA_PROTOCOLS[key];
   619→   619→    if (!proto) return null;
   620→   620→
   621→   621→    const vulnerabilities: string[] = [...proto.commonVulns];
   622→   622→    const complianceGaps: string[] = [];
   623→   623→    const recommendations: string[] = [];
   624→   624→
   625→   625→    // Segment-specific risk amplification
   626→   626→    if (networkSegment === 'enterprise') {
   627→   627→      complianceGaps.push(`OT protocol ${proto.name} detected in enterprise zone — violates network segmentation policy`);
   628→   628→      vulnerabilities.push('cross-zone protocol exposure');
   629→   629→      recommendations.push('Immediately isolate OT protocols to designated industrial zones');
   630→   630→    }
   631→   631→    if (networkSegment === 'dmz' && proto.risk === 'critical') {
   632→   632→      complianceGaps.push(`Critical-risk protocol ${proto.name} traversing DMZ without encryption`);
   633→   633→      recommendations.push('Deploy protocol-aware gateway with deep packet inspection in DMZ');
   634→   634→    }
   635→   635→
   636→   636→    // Industry-specific vulnerabilities
   637→   637→    if (industry === 'energy' && (key === 'dnp3' || key === 'ipp')) {
   638→   638→      vulnerabilities.push('NERC CIP-regulated protocol without end-to-end security');
   639→   639→      complianceGaps.push('CIP-007: Inadequate port and service security');
   640→   640→    }
   641→   641→    if (industry === 'energy') {
   642→   642→      proto.nercCip.forEach(c => {
   643→   643→        complianceGaps.push(`${c}: Verification required for ${proto.name} deployment`);
   644→   644→      });
   645→   645→    }
   646→   646→
   647→   647→    // Generic hardening recommendations
   648→   648→    if (proto.risk === 'critical') {
   649→   649→      recommendations.push(`Implement protocol-specific IDS signatures for ${proto.name}`);
   650→   650→      recommendations.push(`Deploy unidirectional gateway for ${proto.name} traffic`);
   651→   651→      if (proto.port) recommendations.push(`Restrict ${proto.name} access to port ${proto.port} via ACLs`);
   652→   652→    }
   653→   653→    if (proto.port && proto.risk !== 'medium') {
   654→   654→      recommendations.push(`Monitor all traffic on port ${proto.port} for anomalous patterns`);
   655→   655→    }
   656→   656→    if (!recommendations.some(r => r.includes('encryption')) && proto.commonVulns.some(v => v.includes('no encryption') || v.includes('unauthenticated'))) {
   657→   657→      recommendations.push(`Implement encryption and mutual authentication for ${proto.name} communications`);
   658→   658→    }
   659→   659→
   660→   660→    return {
   661→   661→      protocol: proto.name,
   662→   662→      port: proto.port,
   663→   663→      risk: proto.risk,
   664→   664→      vulnerabilities: Array.from(new Set(vulnerabilities)),
   665→   665→      complianceGaps: Array.from(new Set(complianceGaps)),
   666→   666→      recommendations: Array.from(new Set(recommendations)),
   667→   667→    };
   668→   668→  }).filter(Boolean) as ProtocolAnalysis[];
   669→   669→}

   671→   671→// ── NERC CIP Compliance Assessment ───────────────────────────────────────────
   672→   672→
   673→   673→function assessNERCCompliance(
   674→   674→  industry: Industry,
   675→   675→  protocols: string[],
   676→   676→  networkSegment: NetworkSegment,
   677→   677→  devices?: DeviceInfo[],
   678→   678→  scanFindings?: ScanFinding[]
   679→   679→): NERCCompliance {
   680→   680→  if (industry !== 'energy') {
   681→   681→    return { score: 0, requirements: [] };
   682→   682→  }
   683→   683→
   684→   684→  let totalScore = 0;
   685→   685→  const criticalProtocols = protocols.filter(p => SCADA_PROTOCOLS[p]?.risk === 'critical').length;
   686→   686→  const highProtocols = protocols.filter(p => SCADA_PROTOCOLS[p]?.risk === 'high').length;
   687→   687→
   688→   688→  const requirements = NERC_CIP_REQUIREMENTS.map(req => {
   689→   689→    let status: 'pass' | 'partial' | 'fail' = 'pass';
   690→   690→    const gaps: string[] = [];
   691→   691→    let evidence = 'No issues detected';
   692→   692→
   693→   693→    switch (req.code) {
   694→   694→      case 'CIP-005':
   695→   695→        if (networkSegment === 'enterprise') {
   696→   696→          status = 'fail';
   697→   697→          gaps.push('OT protocols detected in enterprise zone — ESP not properly defined');
   698→   698→          evidence = 'OT protocol traffic found outside ESP';
   699→   699→        } else if (protocols.length > 3) {
   700→   700→          status = 'partial';
   701→   701→          gaps.push('Multiple OT protocols increase ESP attack surface');
   702→   702→          evidence = `${protocols.length} protocols require ESP enforcement`;
   703→   703→        }
   704→   704→        if (criticalProtocols > 0) {
   705→   705→          status = status === 'pass' ? 'partial' : 'fail';
   706→   706→          gaps.push(`${criticalProtocols} critical-risk protocol(s) require enhanced ESP controls`);
   707→   707→        }
   708→   708→        break;
   709→   709→      case 'CIP-007':
   710→   710→        if (devices && devices.some(d => {
   711→   711→          const key = `${d.vendor} ${d.type}`;
   712→   712→          return (FIRMWARE_CVE_DB[key] || FIRMWARE_CVE_DB[`Generic ${d.type}`])?.some(c => c.severity === 'critical');
   713→   713→        })) {
   714→   714→          status = 'fail';
   715→   715→          gaps.push('Devices with critical CVEs detected — patch management insufficient');
   716→   716→          evidence = 'Critical vulnerabilities in deployed firmware';
   717→   717→        }
   718→   718→        if (scanFindings && scanFindings.some(f => f.severity === 'critical')) {
   719→   719→          status = status === 'pass' ? 'partial' : 'fail';
   720→   720→          gaps.push('Critical findings from security scans indicate inadequate security management');
   721→   721→        }
   722→   722→        if (highProtocols > 2) {
   723→   723→          status = status === 'pass' ? 'partial' : status;
   724→   724→          gaps.push('Multiple high-risk ports/services require monitoring');
   725→   725→        }
   726→   726→        break;
   727→   727→      case 'CIP-010':
   728→   728→        if (devices && devices.length > 0) {
   729→   729→          status = 'partial';
   730→   730→          gaps.push('Configuration baselines should be established for all ' + devices.length + ' device(s)');
   731→   731→          evidence = `${devices.length} devices require configuration management`;
   732→   732→        }
   733→   733→        break;
   734→   734→      case 'CIP-002':
   735→   735→        status = 'pass';
   736→   736→        evidence = 'BES Cyber System categorization assumed complete';
   737→   737→        if (criticalProtocols > 0) {
   738→   738→          status = 'partial';
   739→   739→          gaps.push('High Impact BCS systems require re-categorization review');
   740→   740→        }
   741→   741→        break;
   742→   742→      case 'CIP-003':
   743→   743→        status = 'partial';
   744→   744→        gaps.push('Automated verification of policy compliance recommended');
   745→   745→        evidence = 'Policy compliance requires manual audit verification';
   746→   746→        break;
   747→   747→      case 'CIP-004':
   748→   748→        status = 'partial';
   749→   749→        gaps.push('OT-specific security training completion status unknown');
   750→   750→        evidence = 'Training records require verification';
   751→   751→        break;
   752→   752→      case 'CIP-008':
   753→   753→        status = 'pass';
   754→   754→        evidence = 'Incident response plan assumed in place';
   755→   755→        if (criticalProtocols > 0) {
   756→   756→          status = 'partial';
   757→   757→          gaps.push('IR plan should include OT-specific playbooks for critical protocols');
   758→   758→        }
   759→   759→        break;
   760→   760→      case 'CIP-009':
   761→   761→        status = 'partial';
   762→   762→        gaps.push('Recovery plan testing for OT systems not verified');
   763→   763→        evidence = 'Recovery plan requires OT-specific validation';
   764→   764→        break;
   765→   765→      case 'CIP-011':
   766→   766→        if (networkSegment === 'enterprise' || networkSegment === 'dmz') {
   767→   767→          status = 'partial';
   768→   768→          gaps.push('BES Cyber System Information may be accessible outside protected zones');
   769→   769→        }
   770→   770→        break;
   771→   771→      case 'CIP-013':
   772→   772→        status = 'partial';
   773→   773→        gaps.push('Supply chain risk assessment for OT vendors recommended');
   774→   774→        if (devices) {
   775→   775→          const vendors = Array.from(new Set(devices.map(d => d.vendor)));
   776→   776→          gaps.push(`Vendor risk assessment needed for: ${vendors.join(', ')}`);
   777→   777→        }
   778→   778→        break;
   779→   779→      case 'CIP-006':
   780→   780→        status = 'partial';
   781→   781→        gaps.push('Physical security verification for cyber assets not confirmed');
   782→   782→        evidence = 'Physical access controls require on-site audit';
   783→   783→        break;
   784→   784→      case 'CIP-014':
   785→   785→        status = 'partial';
   786→   786→        gaps.push('Transmission station physical security needs verification');
   787→   787→        evidence = 'Physical security assessment pending';
   788→   788→        break;
   789→   789→    }
   790→   790→
   791→   791→    const score = status === 'pass' ? 100 : status === 'partial' ? 60 : 20;
   792→   792→    totalScore += score;
   793→   793→    return { code: req.code, name: req.name, status, gaps, evidence };
   794→   794→  });
   795→   795→
   796→   796→  return {
   797→   797→    score: Math.round(totalScore / NERC_CIP_REQUIREMENTS.length),
   798→   798→    requirements,
   799→   799→  };
   800→   800→}

   802→   802→// ── IEC 62443 Compliance Assessment ──────────────────────────────────────────
   803→   803→
   804→   804→function assessIEC62443(
   805→   805→  protocols: string[],
   806→   806→  networkSegment: NetworkSegment,
   807→   807→  devices?: DeviceInfo[],
   808→   808→  scanFindings?: ScanFinding[]
   809→   809→): IEC62443Compliance {
   810→   810→  const zones = IEC_ZONES.map(zone => {
   811→   811→    let score = 100;
   812→   812→    const gaps: string[] = [];
   813→   813→    const recommendations: string[] = [];
   814→   814→
   815→   815→    // Collect all IEC 62443 requirements referenced by active protocols
   816→   816→    const requiredSRs = new Set<string>();
   817→   817→    protocols.forEach(key => {
   818→   818→      const proto = SCADA_PROTOCOLS[key];
   819→   819→      if (proto) proto.iec62443.forEach(sr => requiredSRs.add(sr));
   820→   820→    });
   821→   821→
   822→   822→    // Zone-specific assessments
   823→   823→    if (zone.zone === 'Zone 0' && networkSegment === 'enterprise') {
   824→   824→      score -= 15;
   825→   825→      gaps.push('Enterprise zone directly adjacent to OT — SR 2.1 (Human User Identification) enforcement needed');
   826→   826→      recommendations.push('Enforce strong authentication at enterprise-OT boundary');
   827→   827→    }
   828→   828→    if (zone.zone === 'Zone 1' || zone.zone === 'Zone 2') {
   829→   829→      score -= 10;
   830→   830→      gaps.push('DMZ zones require enhanced monitoring and logging (SR 6.1, SR 6.2)');
   831→   831→      recommendations.push('Deploy SIEM integration for all DMZ traffic');
   832→   832→    }
   833→   833→    if (zone.zone === 'Zone 3') {
   834→   834→      if (protocols.some(p => SCADA_PROTOCOLS[p]?.risk === 'critical')) {
   835→   835→        score -= 25;
   836→   836→        gaps.push('Critical-risk protocols in SCADA zone without additional hardening');
   837→   837→        recommendations.push('Implement application whitelisting on all SCADA servers');
   838→   838→      }
   839→   839→      if (devices && devices.some(d => d.type.toLowerCase() === 'hmi')) {
   840→   840→        score -= 10;
   841→   841→        gaps.push('HMI devices require hardening — SR 4.1 (Information Integrity)');
   842→   842→        recommendations.push('Apply HMI security baselines and disable unnecessary services');
   843→   843→      }
   844→   844→    }
   845→   845→    if (zone.zone === 'Zone 4') {
   846→   846→      if (protocols.some(p => ['s7comm', 'modbus_tcp', 'modbus_rtu', 'dnp3'].includes(p))) {
   847→   847→        score -= 20;
   848→   848→        gaps.push('Legacy PLC communication protocols lack authentication — SR 1.1, SR 1.2');
   849→   849→        recommendations.push('Deploy PLC-specific IDS/IPS with protocol deep inspection');
   850→   850→      }
   851→   851→      if (devices && devices.some(d => d.type.toLowerCase().includes('plc'))) {
   852→   852→        const plcDevs = devices.filter(d => d.type.toLowerCase().includes('plc'));
   853→   853→        plcDevs.forEach(d => {
   854→   854→          const key = `${d.vendor} ${d.type}`;
   855→   855→          const cves = FIRMWARE_CVE_DB[key] || FIRMWARE_CVE_DB[`Generic ${d.type}`];
   856→   856→          if (cves) {
   857→   857→            score -= cves.filter(c => c.severity === 'critical').length * 10;
   858→   858→            gaps.push(`${d.vendor} ${d.type} at ${d.ip}: ${cves.length} known CVE(s)`);
   859→   859→          }
   860→   860→        });
   861→   861→        recommendations.push('Establish firmware update program for all PLC devices');
   862→   862→      }
   863→   863→    }
   864→   864→    if (zone.zone === 'Zone 3.5') {
   865→   865→      score -= 15;
   866→   866→      gaps.push('Safety system zone requires independent verification — SR 3.3, SR 4.1');
   867→   867→      recommendations.push('Isolate safety systems with hardware-enforced separation');
   868→   868→    }
   869→   869→
   870→   870→    // Scan findings impact
   871→   871→    if (scanFindings && scanFindings.length > 0) {
   872→   872→      const criticalFindings = scanFindings.filter(f => f.severity === 'critical').length;
   873→   873→      score -= criticalFindings * 5;
   874→   874→      if (criticalFindings > 0) {
   875→   875→        gaps.push(`${criticalFindings} critical scan findings impact zone security posture`);
   876→   876→      }
   877→   877→    }
   878→   878→
   879→   879→    score = Math.max(0, Math.min(100, score));
   880→   880→    return { zone: zone.zone, score, gaps, recommendations };
   881→   881→  });
   882→   882→
   883→   883→  const score = Math.round(zones.reduce((s, z) => s + z.score, 0) / zones.length);
   884→   884→  return { score, zones };
   885→   885→}

   887→   887→// ── Network Segmentation Assessment ──────────────────────────────────────────
   888→   888→
   889→   889→function assessNetworkSegmentation(
   890→   890→  networkSegment: NetworkSegment,
   891→   891→  protocols: string[],
   892→   892→  devices?: DeviceInfo[]
   893→   893→): { score: number; issues: string[]; recommendations: string[] } {
   894→   894→  const issues: string[] = [];
   895→   895→  const recommendations: string[] = [];
   896→   896→  let score = 100;
   897→   897→
   898→   898→  // Check for OT protocols in wrong zones
   899→   899→  const otProtocols = protocols.filter(p => SCADA_PROTOCOLS[p]?.risk !== 'medium');
   900→   900→  if (networkSegment === 'enterprise' && otProtocols.length > 0) {
   901→   901→    score -= 35;
   902→   902→    issues.push(`CRITICAL: ${otProtocols.length} OT protocol(s) detected in enterprise zone — flat network architecture detected`);
   903→   903→    recommendations.push('Implement Purdue Model-compliant network architecture with strict zone separation');
   904→   904→    recommendations.push('Deploy industrial DMZ (iDMZ) between enterprise and SCADA zones');
   905→   905→  }
   906→   906→  if (networkSegment === 'dmz') {
   907→   907→    score -= 10;
   908→   908→    issues.push('Protocols traversing DMZ require proxy-based inspection');
   909→   909→    recommendations.push('Deploy protocol-aware proxy/gateway in DMZ for all OT traffic');
   910→   910→  }
   911→   911→
   912→   912→  // Check for unencrypted protocols
   913→   913→  const unencryptedProtos = protocols.filter(p => {
   914→   914→    const proto = SCADA_PROTOCOLS[p];
   915→   915→    return proto && (proto.name.includes('Modbus') || proto.name.includes('DNP3') || proto.name.includes('S7comm') || proto.name.includes('PROFINET') || proto.name.includes('CIP'));
   916→   916→  });
   917→   917→  if (unencryptedProtos.length > 0) {
   918→   918→    score -= unencryptedProtos.length * 5;
   919→   919→    issues.push(`${unencryptedProtos.length} protocol(s) lack native encryption: ${unencryptedProtos.map(p => SCADA_PROTOCOLS[p]?.name).join(', ')}`);
   920→   920→    recommendations.push('Wrap legacy protocols in IPsec tunnels between zones');
   921→   921→  }
   922→   922→
   923→   923→  // Device placement issues
   924→   924→  if (devices) {
   925→   925→    const directInternetDevices = devices.filter(d => d.ip && !d.ip.startsWith('10.') && !d.ip.startsWith('172.') && !d.ip.startsWith('192.168.'));
   926→   926→    if (directInternetDevices.length > 0) {
   927→   927→      score -= 30;
   928→   928→      issues.push(`CRITICAL: ${directInternetDevices.length} device(s) have public IP addresses: ${directInternetDevices.map(d => d.ip).join(', ')}`);
   929→   929→      recommendations.push('Immediately move all OT devices to private address space behind firewalls');
   930→   930→    }
   931→   931→  }
   932→   932→
   933→   933→  // Critical protocol amplification
   934→   934→  if (protocols.includes('s7comm') && protocols.includes('modbus_tcp')) {
   935→   935→    score -= 10;
   936→   936→    issues.push('Multiple PLC vendor protocols coexist — segmentation must enforce vendor isolation');
   937→   937→    recommendations.push('Segment PLC networks by vendor with separate VLANs and firewall policies');
   938→   938→  }
   939→   939→
   940→   940→  score = Math.max(0, Math.min(100, score));
   941→   941→  return { score, issues, recommendations };
   942→   942→}

   944→   944→// ── Firmware Analysis ────────────────────────────────────────────────────────
   945→   945→
   946→   946→function analyzeFirmware(devices?: DeviceInfo[]): FirmwareAnalysis {
   947→   947→  if (!devices || devices.length === 0) {
   948→   948→    return { devices: [] };
   949→   949→  }
   950→   950→
   951→   951→  const analyzedDevices = devices.map(device => {
   952→   952→    const key = `${device.vendor} ${device.type}`;
   953→   953→    const cves = FIRMWARE_CVE_DB[key] || FIRMWARE_CVE_DB[`Generic ${device.type}`] || [];
   954→   954→    const hasCritical = cves.some(c => c.severity === 'critical');
   955→   955→    const hasHigh = cves.some(c => c.severity === 'high');
   956→   956→    const riskLevel = hasCritical ? 'critical' : hasHigh ? 'high' : cves.length > 0 ? 'medium' : 'low';
   957→   957→
   958→   958→    let recommendation = 'Device firmware appears current — maintain regular update cadence';
   959→   959→    if (hasCritical) {
   960→   960→      recommendation = `URGENT: Apply critical security patches. ${cves.filter(c => c.severity === 'critical').length} critical CVE(s) detected. Consider network isolation until patched.`;
   961→   961→    } else if (hasHigh) {
   962→   962→      recommendation = `Schedule firmware update within 30 days. ${cves.filter(c => c.severity === 'high').length} high-severity CVE(s) present.`;
   963→   963→    } else if (cves.length > 0) {
   964→   964→      recommendation = 'Plan firmware update during next maintenance window.';
   965→   965→    }
   966→   966→
   967→   967→    return {
   968→   968→      device: `${device.vendor} ${device.type} (${device.ip})`,
   969→   969→      firmware: device.firmware,
   970→   970→      knownCVEs: cves,
   971→   971→      riskLevel,
   972→   972→      recommendation,
   973→   973→    };
   974→   974→  });
   975→   975→
   976→   976→  return { devices: analyzedDevices };
   977→   977→}

   979→   979→// ── MITRE ATT&CK Mapping ─────────────────────────────────────────────────────
   980→   980→
   981→   981→function mapMITREAttack(
   982→   982→  industry: Industry,
   983→   983→  protocols: string[],
   984→   984→  aptAssessment: { activeThreats: APTGroup[] },
   985→   985→  scanFindings?: ScanFinding[]
   986→   986→): MITREMapping {
   987→   987→  const relevantTechniques = getMITRETechniquesForIndustry(industry);
   988→   988→
   989→   989→  // Collect all techniques from active APT groups
   990→   990→  const aptTechniqueIds = new Set<string>();
   991→   991→  aptAssessment.activeThreats.forEach(apt => {
   992→   992→    apt.techniques.forEach(t => aptTechniqueIds.add(t));
   993→   993→  });
   994→   994→
   995→   995→  const techniques = relevantTechniques.map(technique => {
   996→   996→    const detected = aptTechniqueIds.has(technique.id);
   997→   997→    let confidence = 0;
   998→   998→
   999→   999→    if (detected) {
  1000→  1000→      confidence = 70 + (technique.otSpecific ? 20 : 10);
  1001→  1001→      // Boost confidence if scan findings correlate
  1002→  1002→      if (scanFindings && scanFindings.length > 0) {
  1003→  1003→        const correlatedFindings = scanFindings.filter(f => {
  1004→  1004→          if (technique.tactic === 'Initial Access' && f.type?.includes('phish')) return true;
  1005→  1005→          if (technique.tactic === 'Impact' && f.severity === 'critical') return true;
  1006→  1006→          if (technique.tactic === 'Credential Access' && f.type?.includes('credential')) return true;
  1007→  1007→          return false;
  1008→  1008→        });
  1009→  1009→        confidence = Math.min(99, confidence + correlatedFindings.length * 5);
  1010→  1010→      }
  1011→  1011→    } else if (technique.otSpecific && protocols.length > 0) {
  1012→  1012→      confidence = 30; // OT-specific techniques are relevant even if not directly detected
  1013→  1013→    }
  1014→  1014→
  1015→  1015→    return {
  1016→  1016→      id: technique.id,
  1017→  1017→      name: technique.name,
  1018→  1018→      tactic: technique.tactic,
  1019→  1019→      detected,
  1020→  1020→      confidence,
  1021→  1021→    };
  1022→  1022→  });
  1023→  1023→
  1024→  1024→  const detectedCount = techniques.filter(t => t.detected).length;
  1025→  1025→  const coverageScore = Math.round((detectedCount / Math.max(techniques.length, 1)) * 100);
  1026→  1026→
  1027→  1027→  return { techniques, coverageScore };
  1028→  1028→}

  1032→  1032→export function analyzeCNIThreats(params: CNIAnalysisParams): CNIAnalysisResult {
  1033→  1033→  const { networkSegment, industry, protocols, devices, scanFindings } = params;
  1034→  1034→
  1035→  1035→  // 1. Protocol analysis
  1036→  1036→  const protocolAnalysis = analyzeProtocols(protocols, networkSegment, industry);
  1037→  1037→
  1038→  1038→  // 2. APT risk assessment
  1039→  1039→  const aptAssessment = assessAPTRisk(industry, protocols);
  1040→  1040→
  1041→  1041→  // 3. NERC CIP compliance
  1042→  1042→  const nercCompliance = assessNERCCompliance(industry, protocols, networkSegment, devices, scanFindings);
  1043→  1043→
  1044→  1044→  // 4. IEC 62443 compliance
  1045→  1045→  const iec62443Compliance = assessIEC62443(protocols, networkSegment, devices, scanFindings);
  1046→  1046→
  1047→  1047→  // 5. Network segmentation
  1048→  1048→  const networkSegmentation = assessNetworkSegmentation(networkSegment, protocols, devices);
  1049→  1049→
  1050→  1050→  // 6. Firmware analysis
  1051→  1051→  const firmwareAnalysis = analyzeFirmware(devices);
  1052→  1052→
  1053→  1053→  // 7. MITRE ATT&CK mapping
  1054→  1054→  const mitreAttackMapping = mapMITREAttack(industry, protocols, aptAssessment, scanFindings);
  1055→  1055→
  1056→  1056→  // ── Calculate overall threat level and score ──
  1057→  1057→  let overallScore = 0;
  1058→  1058→
  1059→  1059→  // Protocol risk contribution (0-30)
  1060→  1060→  const criticalProtoCount = protocolAnalysis.filter(p => p.risk === 'critical').length;
  1061→  1061→  const highProtoCount = protocolAnalysis.filter(p => p.risk === 'high').length;
  1062→  1062→  overallScore += Math.min(30, criticalProtoCount * 10 + highProtoCount * 5);
  1063→  1063→
  1064→  1064→  // APT risk contribution (0-25)
  1065→  1065→  overallScore += Math.min(25, Math.round(aptAssessment.riskScore * 0.25));
  1066→  1066→
  1067→  1067→  // Compliance deficit (0-25)
  1068→  1068→  const nercDeficit = industry === 'energy' ? (100 - nercCompliance.score) * 0.15 : 0;
  1069→  1069→  const iecDeficit = (100 - iec62443Compliance.score) * 0.1;
  1070→  1070→  overallScore += Math.min(25, Math.round(nercDeficit + iecDeficit));
  1071→  1071→
  1072→  1072→  // Network segmentation deficit (0-10)
  1073→  1073→  overallScore += Math.min(10, Math.round((100 - networkSegmentation.score) * 0.1));
  1074→  1074→
  1075→  1075→  // Firmware risk contribution (0-10)
  1076→  1076→  if (firmwareAnalysis.devices.length > 0) {
  1077→  1077→    const critFirmware = firmwareAnalysis.devices.filter(d => d.riskLevel === 'critical').length;
  1078→  1078→    const highFirmware = firmwareAnalysis.devices.filter(d => d.riskLevel === 'high').length;
  1079→  1079→    overallScore += Math.min(10, critFirmware * 5 + highFirmware * 2);
  1080→  1080→  }
  1081→  1081→
  1082→  1082→  // Scan findings boost
  1083→  1083→  if (scanFindings) {
  1084→  1084→    const critFindings = scanFindings.filter(f => f.severity === 'critical').length;
  1085→  1085→    overallScore += Math.min(10, critFindings * 5);
  1086→  1086→  }
  1087→  1087→
  1088→  1088→  overallScore = Math.min(100, Math.max(0, overallScore));
  1089→  1089→
  1090→  1090→  const threatLevel: CNIAnalysisResult['threatLevel'] =
  1091→  1091→    overallScore >= 80 ? 'CRITICAL' :
  1092→  1092→    overallScore >= 60 ? 'HIGH' :
  1093→  1093→    overallScore >= 40 ? 'ELEVATED' :
  1094→  1094→    overallScore >= 20 ? 'MODERATE' :
  1095→  1095→    'LOW';
  1096→  1096→
  1097→  1097→  // Build full result
  1098→  1098→  const results: CNIAnalysisResult = {
  1099→  1099→    threatLevel,
  1100→  1100→    overallScore,
  1101→  1101→    protocolAnalysis,
  1102→  1102→    aptAssessment,
  1103→  1103→    nercCompliance,
  1104→  1104→    iec62443Compliance,
  1105→  1105→    networkSegmentation,
  1106→  1106→    firmwareAnalysis,
  1107→  1107→    mitreAttackMapping,
  1108→  1108→    stixReport: '',
  1109→  1109→    iodefReport: '',
  1110→  1110→  };
  1111→  1111→
  1112→  1112→  // Generate reports
  1113→  1113→  results.stixReport = generateSTIXReport(results);
  1114→  1114→  results.iodefReport = generateIODEFReport(results);
  1115→  1115→
  1116→  1116→  return results;
  1117→  1117→}
