   153→# Boot Screen Widget
   154→# ════════════════════════════════════════════════════════════════════════════════
   155→
   156→
   157→class BootScreen(Widget):
   158→    """Full-screen animated boot sequence."""
   159→
   160→    def __init__(self) -> None:
   161→        super().__init__()
   162→        self._step = 0
   163→        self._spinner_idx = 0
   164→        self._module_idx = 0
   165→        self._module_names = list(MODULE_NAMES.values())
   166→        self._done = False
   167→        self._timer: Optional[Timer] = None
   168→
   169→    def compose(self) -> ComposeResult:
   170→        with Vertical(classes="boot-container"):
   171→            yield Static("", id="boot-line-1", classes="boot-title")
   172→            yield Static("", id="boot-line-2", classes="boot-line")
   173→            yield Static("", id="boot-line-3", classes="boot-line")
   174→            yield Static("", id="boot-line-4", classes="boot-line")
   175→
   176→    def on_mount(self) -> None:
   177→        self._timer = self.set_interval(0.04, self._tick)
   178→
   179→    def _tick(self) -> None:
   180→        app = self.app
   181→        if not isinstance(app, NexusApp):
   182→            return
   183→
   184→        line1 = app.query_one("#boot-line-1", Static)
   185→        line2 = app.query_one("#boot-line-2", Static)
   186→        line3 = app.query_one("#boot-line-3", Static)
   187→        line4 = app.query_one("#boot-line-4", Static)
   188→
   189→        if self._step == 0:
   190→            # Letter-by-letter title
   191→            title = "RECONPRO NEXUS"
   192→            chars = title[: self._spinner_idx]
   193→            self._spinner_idx += 1
   194→            line1.update(f"[{CYAN} bold]{chars}[/]")
   195→            if self._spinner_idx > len(title):
   196→                self._spinner_idx = 0
   197→                self._step = 1
   198→                line1.update(f"[{CYAN} bold]{title}[/]  [{DIM_CYAN}]v{VERSION}[/]")
   199→
   200→        elif self._step == 1:
   201→            # Spinner for "Initializing agent..."
   202→            frame = SPINNER_FRAMES[self._spinner_idx % len(SPINNER_FRAMES)]
   203→            self._spinner_idx += 1
   204→            line2.update(f"  [{CYAN}]{frame}[/] [{DIM_CYAN}]Initializing agent...[/]")
   205→            if self._spinner_idx > 30:
   206→                self._spinner_idx = 0
   207→                self._step = 2
   208→                line2.update(f"  [{GREEN}]✓[/] [{DIM_CYAN}]Agent initialized.[/]")
   209→
   210→        elif self._step == 2:
   211→            # Module names appearing one by one
   212→            if self._module_idx < len(self._module_names):
   213→                visible = self._module_names[: self._module_idx + 1]
   214→                self._module_idx += 1
   215→                modules_str = "  ".join(f"[{CYAN}]○[/] [{DIM_CYAN}]{m}[/]" for m in visible)
   216→                remaining = len(self._module_names) - self._module_idx
   217→                if remaining > 0:
   218→                    modules_str += f"  [{DIM_CYAN}]+{remaining} more...[/]"
   219→                frame = SPINNER_FRAMES[self._spinner_idx % len(SPINNER_FRAMES)]
   220→                self._spinner_idx += 1
   221→                line3.update(
   222→                    f"  [{CYAN}]{frame}[/] [{DIM_CYAN}]Loading {len(self._module_names)} modules...[/]\n  {modules_str}"
   223→                )
   224→            else:
   225→                all_mods = "  ".join(
   226→                    f"[{GREEN}]●[/] [{DIM_CYAN}]{m}[/]" for m in self._module_names
   227→                )
   228→                line3.update(
   229→                    f"  [{GREEN}]✓[/] [{DIM_CYAN}]All {len(self._module_names)} modules loaded.[/]\n  {all_mods}"
   230→                )
   231→                self._step = 3
   232→                self._spinner_idx = 0
   233→
   234→        elif self._step == 3:
   235→            # "Ready." with flash
   236→            if self._spinner_idx == 0:
   237→                line4.update(f"[{YELLOW} bold]  ▶  READY.[/]")
   238→            elif self._spinner_idx == 6:
   239→                line4.update(f"[{GREEN} bold]  ▶  READY.[/]")
   240→            elif self._spinner_idx >= 10:
   241→                self._done = True
   242→                if self._timer:
   243→                    self._timer.stop()
   244→                self.app.call_from_thread(self.app._finish_boot)
   245→            self._spinner_idx += 1
   246→

   248→# ════════════════════════════════════════════════════════════════════════════════
   249→# Module Status Cell Widget
   250→# ════════════════════════════════════════════════════════════════════════════════
   251→
   252→
   253→class ModuleCell(Static):
   254→    """Single module status cell in the grid."""
   255→
   256→    def __init__(self, module_id: str, name: str) -> None:
   257→        super().__init__()
   258→        self.module_id = module_id
   259→        self.module_name = name
   260→        self.status = "idle"  # idle | scanning | done | error
   261→        self.finding_count = 0
   262→        self._spinner_idx = 0
   263→        self._timer: Optional[Timer] = None
   264→
   265→    def on_mount(self) -> None:
   266→        self._update_display()
   267→        self._timer = self.set_interval(0.1, self._spin)
   268→
   269→    def _spin(self) -> None:
   270→        if self.status == "scanning":
   271→            self._spinner_idx = (self._spinner_idx + 1) % len(SPINNER_FRAMES)
   272→            self._update_display()
   273→
   274→    def _update_display(self) -> None:
   275→        name_short = self.module_name[:14]
   276→        if self.status == "idle":
   277→            icon = f"[{DIM_CYAN}]○[/]"
   278→            status_text = f"[{DIM_CYAN}]idle[/]"
   279→        elif self.status == "scanning":
   280→            frame = SPINNER_FRAMES[self._spinner_idx]
   281→            icon = f"[{CYAN}]{frame}[/]"
   282→            status_text = f"[{CYAN}]scanning[/]"
   283→        elif self.status == "done":
   284→            icon = f"[{GREEN}]✓[/]"
   285→            status_text = f"[{GREEN}]done ({self.finding_count})[/]"
   286→        elif self.status == "error":
   287→            icon = f"[{RED}]✗[/]"
   288→            status_text = f"[{RED}]error[/]"
   289→        else:
   290→            icon = "○"
   291→            status_text = "idle"
   292→
   293→        self.update(
   294→            f"{icon} [{CYAN} bold]{name_short}[/]\n  {status_text}"
   295→        )
   296→
   297→    def set_status(self, status: str, count: int = 0) -> None:
   298→        self.status = status
   299→        self.finding_count = count
   300→        self._update_display()
   301→
   302→    def on_unmount(self) -> None:
   303→        if self._timer:
   304→            self._timer.stop()
   305→

   985→    def _handle_scan(self, args: List[str]) -> None:
   986→        if not args:
   987→            self._chat(f"[{RED}]Usage: scan <target> [with <mod1> <mod2> ...][/]")
   988→            return
   989→
   990→        # Parse "with" clause
   991→        modules: Optional[List[str]] = None
   992→        target_parts: List[str] = []
   993→        found_with = False
   994→        mod_parts: List[str] = []
   995→        for a in args:
   996→            if a.lower() == "with":
   997→                found_with = True
   998→                continue
   999→            if found_with:
  1000→                mod_parts.append(a)
  1001→            else:
  1002→                target_parts.append(a)
  1003→
  1004→        target = " ".join(target_parts)
  1005→        if not target:
  1006→            self._chat(f"[{RED}]No target specified.[/]")
  1007→            return
  1008→
  1009→        if mod_parts:
  1010→            modules = [m.strip().lower() for m in mod_parts]
  1011→            self._chat(f"[{DIM_CYAN}]Scanning [cyan bold]{target}[/] with modules: {', '.join(modules)}[/]")
  1012→        else:
  1013→            self._chat(f"[{DIM_CYAN}]Scanning [cyan bold]{target}[/]...[/]")
  1014→
  1015→        self.current_target = target
  1016→        self._run_scan_worker(target, modules=modules)

  1097→    @work(exclusive=False, thread=True)
  1098→    def _run_scan_worker(
  1099→        self,
  1100→        target: str,
  1101→        modules: Optional[List[str]] = None,
  1102→        is_local: bool = False,
  1103→    ) -> None:
  1104→        """Run a scan in a background thread."""
  1105→        self.call_from_thread(self.is_scanning.set, True)
  1106→
  1107→        # Set modules to scanning
  1108→        if modules:
  1109→            for m in modules:
  1110→                if m in self._module_cells:
  1111→                    self.call_from_thread(self._module_cells[m].set_status, "scanning")
  1112→        else:
  1113→            all_mods = (
  1114→                list(MODULE_NAMES.keys())
  1115→                if is_local
  1116→                else [k for k in MODULE_NAMES if k not in ("host", "dev", "doctor")]
  1117→            )
  1118→            for m in all_mods:
  1119→                if m in self._module_cells:
  1120→                    self.call_from_thread(self._module_cells[m].set_status, "scanning")
  1121→
  1122→        try:
  1123→            from .scanner import scan, audit_scan
  1124→            from .history import save_scan
  1125→
  1126→            self.call_from_thread(
  1127→                self._chat, f"[{DIM_CYAN}]  ⏳ Scan in progress...[/]"
  1128→            )
  1129→
  1130→            if is_local:
  1131→                result = audit_scan(target=target, modules=modules)
  1132→            else:
  1133→                result = scan(target=target, modules=modules)
  1134→
  1135→            # Process findings
  1136→            data = result.to_dict()
  1137→            self.call_from_thread(self._on_scan_complete, data, modules, is_local)
  1138→
  1139→            # Save to history
  1140→            try:
  1141→                save_scan(data, label="nexus")
  1142→            except Exception:
  1143→                pass
  1144→
  1145→        except Exception as e:
  1146→            self.call_from_thread(self._chat, f"[{RED}]  ✗ Scan error: {e}[/]")
  1147→            if modules:
  1148→                for m in modules:
  1149→                    if m in self._module_cells:
  1150→                        self.call_from_thread(self._module_cells[m].set_status, "error")
  1151→            else:
  1152→                self.call_from_thread(self._set_all_modules_status, "error")
  1153→
  1154→        finally:
  1155→            self.call_from_thread(self.is_scanning.set, False)
  1156→
  1157→    @work(exclusive=False, thread=True)

  1159→        """Run parallel blitz scan."""

  1161→        self.call_from_thread(self._set_all_modules_status, "scanning")

  1170→            result = blitz_scan(targets, save=True)
  1171→
  1172→            self.call_from_thread(
  1173→                self._chat,
  1174→                f"[{GREEN}]  ✓ Blitz complete. {result['successful']}/{result['targets_scanned']} successful, "
  1175→                f"{result['total_findings']} findings, avg score: {result['average_score']} ({result['average_grade']})[/]",

  1178→            for t, r in result.get("results", {}).items():

  1181→        except Exception as e:
