  1054→        if "BucketPolicy" in res_type:
  1055→            issues.extend(_cf_check_bucket_policy(resource, fname))
  1056→        if "DBInstance" in res_type or "RDS" in res_type:
  1057→            issues.extend(_cf_check_rds(resource, fname))
  1058→        if res_type == "AWS::S3::Bucket":
  1059→            issues.extend(_cf_check_s3_bucket(resource, fname))
  1060→        if "IAM" in res_type and ("Role" in res_type or "Policy" in res_type or "User" in res_type):
  1061→            issues.extend(_cf_check_iam(resource, fname))
  1062→
  1063→        # Generic: Lambda without VPC
  1064→        if "Lambda" in res_type and "Function" in res_type:
  1065→            props = resource.get("Properties", {})
  1066→            if isinstance(props, dict) and not props.get("VpcConfig"):
  1067→                issues.append({
  1068→                    "title": f"CloudFormation: Lambda '{res_name}' Not in VPC",
  1069→                    "severity": "low", "category": "network_exposure",
  1070→                    "description": f"Lambda function '{res_name}' has no VpcConfig, running with direct internet access.",
  1071→                    "evidence": f"{fname}: {res_name} missing VpcConfig",
  1072→                    "remediation": "Add VpcConfig with SubnetIds and SecurityGroupIds.",
  1073→                    "dread_score": _dread("low"), "points_deducted": _pts("low"),
  1074→                })
  1075→
  1076→    return issues

  1083→
  1084→# Sensitive file patterns that should never be COPY/ADDed
  1085→_SENSITIVE_FILE_PATTERNS: List[Tuple[re.Pattern, str]] = [
  1086→    (re.compile(r"\.env", re.I), ".env file"),
  1087→    (re.compile(r"\.aws", re.I), ".aws credentials directory"),
  1088→    (re.compile(r"(id_rsa|id_dsa|id_ecdsa|id_ed25519)", re.I), "SSH private key"),
  1089→    (re.compile(r"\.ssh", re.I), ".ssh directory"),
  1090→    (re.compile(r"\.npmrc", re.I), ".npmrc (may contain auth tokens)"),
  1091→    (re.compile(r"\.pem", re.I), ".pem certificate/key file"),
  1092→    (re.compile(r"\.key", re.I), ".key private key file"),
  1093→    (re.compile(r"\.p12|\.pfx", re.I), "PKCS#12 certificate bundle"),
  1094→    (re.compile(r"credentials\.json|service-account", re.I), "GCP service account credentials"),
  1095→]
  1096→
  1097→# Secret environment variable patterns
  1098→_SECRET_ENV_PATTERNS = re.compile(
  1099→    r'(?:PASSWORD|SECRET|TOKEN|API_KEY|PRIVATE_KEY|AUTH_TOKEN|ACCESS_KEY)'
  1100→    r'\s*=\s*["\x27]\S{1,}["\x27]',
  1101→    re.I,
  1102→)
  1103→
  1104→# Dangerous ports to expose
  1105→_DANGEROUS_PORTS: Dict[str, str] = {
  1106→    "22": "SSH",
  1107→    "2375": "Docker daemon (unencrypted)",
  1108→    "2376": "Docker daemon (TLS)",
  1109→    "6379": "Redis",
  1110→    "27017": "MongoDB",
  1111→    "5432": "PostgreSQL",
  1112→    "3306": "MySQL",
  1113→    "9200": "Elasticsearch",
  1114→    "11211": "Memcached",
  1115→}

  1117→
  1118→def parse_dockerfile(path: str) -> List[Dict[str, Any]]:
  1119→    """Parse a Dockerfile and return a list of issue dicts."""
  1120→    issues: List[Dict[str, Any]] = []
  1121→    fname = os.path.basename(path)
  1122→
  1123→    try:
  1124→        text = Path(path).read_text(encoding="utf-8", errors="replace")
  1125→    except OSError:
  1126→        return issues
  1127→
  1128→    lines = text.splitlines()
  1129→    has_user = False
  1130→    has_healthcheck = False
  1131→    has_from = False
  1132→    from_line = 0
  1133→    user_line_num = 0
  1134→
  1135→    for idx, line in enumerate(lines):
  1136→        stripped = line.strip()
  1137→        if not stripped or stripped.startswith("#"):
  1138→            continue
  1139→
  1140→        ln = idx + 1
  1141→
  1142→        # ── FROM checks ───────────────────────────────────────────────
  1143→        if stripped.upper().startswith("FROM "):
  1144→            has_from = True
  1145→            from_line = ln
  1146→            from_val = stripped[5:].strip()
  1147→
  1148→            # Check: FROM scratch
  1149→            if from_val.lower() == "scratch":
  1150→                issues.append({
  1151→                    "title": "Dockerfile Uses FROM scratch",
  1152→                    "severity": "medium", "category": "container_image",
  1153→                    "description": "FROM scratch produces a minimal image but provides no package management or shell for debugging.",
  1154→                    "evidence": _ev(fname, ln),
  1155→                    "remediation": "Use a minimal base like alpine or distroless if possible.",
  1156→                    "dread_score": _dread("medium"), "points_deducted": _pts("medium"),
  1157→                })
  1158→            # Check: FROM without tag (mutable image)
  1159→            elif ":" not in from_val.split("@")[0]:
  1160→                issues.append({
  1161→                    "title": "Dockerfile FROM Without Specific Tag",
  1162→                    "severity": "medium", "category": "container_image",
  1163→                    "description": f"Base image '{from_val}' has no specific tag, using 'latest' which is mutable.",
  1164→                    "evidence": _ev(fname, ln),
  1165→                    "remediation": f"Pin the image: {from_val}:<specific-tag> or use a digest @{from_val}@sha256:...",
  1166→                    "dread_score": _dread("medium"), "points_deducted": _pts("medium"),
  1167→                })
  1168→            # Check: FROM using :latest tag
  1169→            elif from_val.endswith(":latest"):
  1170→                issues.append({
  1171→                    "title": "Dockerfile FROM Uses :latest Tag",
  1172→                    "severity": "medium", "category": "container_image",
  1173→                    "description": f"Base image uses ':latest' tag which is mutable and non-reproducible.",
  1174→                    "evidence": _ev(fname, ln),
  1175→                    "remediation": f"Pin to a specific version tag instead of :latest.",
  1176→                    "dread_score": _dread("medium"), "points_deducted": _pts("medium"),
  1177→                })
  1178→
  1179→        # ── USER checks ───────────────────────────────────────────────
  1180→        if stripped.upper().startswith("USER "):
  1181→            user_val = stripped[5:].strip()
  1182→            has_user = True
  1183→            user_line_num = ln
  1184→
  1185→            if user_val in ("root", "0"):
  1186→                issues.append({
  1187→                    "title": "Dockerfile Explicitly Runs as Root",
  1188→                    "severity": "high", "category": "container_privilege",
  1189→                    "description": "USER root/0 is explicitly set, running the container with full root privileges.",
  1190→                    "evidence": _ev(fname, ln),
  1191→                    "remediation": "Create a non-root user and set USER <username>.",
  1192→                    "dread_score": _dread("high"), "points_deducted": _pts("high"),
  1193→                })
  1194→
  1195→        # ── COPY / ADD checks ─────────────────────────────────────────
  1196→        if stripped.upper().startswith(("COPY ", "ADD ")):
  1197→            # Check for sensitive files
  1198→            for pat, desc in _SENSITIVE_FILE_PATTERNS:
  1199→                if pat.search(stripped):
  1200→                    issues.append({
  1201→                        "title": f"Dockerfile Copies Sensitive File ({desc})",
  1202→                        "severity": "critical", "category": "secrets_leak",
  1203→                        "description": f"COPY or ADD instruction includes {desc} in the image.",
  1204→                        "evidence": _ev(fname, ln),
  1205→                        "remediation": f"Do not copy {desc} into the image. Use runtime secrets or multi-stage builds.",
  1206→                        "dread_score": _dread("critical"), "points_deducted": _pts("critical"),
  1207→                    })
  1208→                    break
  1209→
  1210→            # Check: ADD with remote URL (supply chain risk)
  1211→            if re.search(r'ADD\s+https?://', stripped, re.I):
  1212→                issues.append({
  1213→                    "title": "Dockerfile ADD from Remote URL",
  1214→                    "severity": "high", "category": "supply_chain",
  1215→                    "description": "ADD instruction fetches from a remote URL, creating a supply chain risk.",
  1216→                    "evidence": _ev(fname, ln),
  1217→                    "remediation": "Use RUN curl/wget with checksum verification instead of ADD for remote URLs.",
  1218→                    "dread_score": _dread("high"), "points_deducted": _pts("high"),
  1219→                })
  1220→
  1221→        # ── ENV checks ────────────────────────────────────────────────
  1222→        if stripped.upper().startswith("ENV "):
  1223→            env_val = stripped[4:].strip()
  1224→            if _SECRET_ENV_PATTERNS.search(env_val):
  1225→                issues.append({
  1226→                    "title": "Dockerfile ENV Contains Secret",
  1227→                    "severity": "high", "category": "hardcoded_secret",
  1228→                    "description": "An ENV variable appears to contain a hardcoded secret/password/token.",
  1229→                    "evidence": _ev(fname, ln),
  1230→                    "remediation": "Use Docker secrets, Kubernetes secrets, or runtime environment injection.",
  1231→                    "dread_score": _dread("high"), "points_deducted": _pts("high"),
  1232→                })
  1233→
  1234→        # ── RUN checks ────────────────────────────────────────────────
  1235→        if stripped.upper().startswith("RUN "):
  1236→            run_val = stripped[4:].strip()
  1237→
  1238→            # curl|wget piped to bash/sh
  1239→            if re.search(r'(?:curl|wget)\s+.*\|\s*(?:ba)?sh', run_val):
  1240→                issues.append({
  1241→                    "title": "Dockerfile RUN Pipes Remote Script to Shell",
  1242→                    "severity": "high", "category": "supply_chain",
  1243→                    "description": "RUN instruction pipes curl/wget output directly to shell — supply chain risk.",
  1244→                    "evidence": _ev(fname, ln),
  1245→                    "remediation": "Download the script first, verify its checksum, then execute it.",
  1246→                    "dread_score": _dread("high"), "points_deducted": _pts("high"),
  1247→                })
  1248→
  1249→            # --privileged in RUN (docker run inside docker)
  1250→            if "--privileged" in run_val:
  1251→                issues.append({
  1252→                    "title": "Dockerfile RUN with --privileged Flag",
  1253→                    "severity": "critical", "category": "container_privilege",
  1254→                    "description": "RUN instruction uses --privileged, granting full host access to the build container.",
  1255→                    "evidence": _ev(fname, ln),
  1256→                    "remediation": "Remove --privileged flag. Grant only specific capabilities.",
  1257→                    "dread_score": _dread("critical"), "points_deducted": _pts("critical"),
  1258→                })
  1259→
  1260→            # Running as root in RUN (sudo/su)
  1261→            if re.search(r'\b(?:sudo|su\s)\b', run_val):
  1262→                issues.append({
  1263→                    "title": "Dockerfile RUN Uses sudo/su",
  1264→                    "severity": "medium", "category": "container_privilege",
  1265→                    "description": "RUN uses sudo/su, indicating the container may need root privileges at runtime.",
  1266→                    "evidence": _ev(fname, ln),
  1267→                    "remediation": "Avoid sudo/su. If root is needed for install, switch back to non-root user.",
  1268→                    "dread_score": _dread("medium"), "points_deducted": _pts("medium"),
  1269→                })
  1270→
  1271→            # apt-get without no-install-recommends
  1272→            if re.search(r'apt-get\s+install', run_val) and "--no-install-recommends" not in run_val:
  1273→                issues.append({
  1274→                    "title": "Dockerfile apt-get Without --no-install-recommends",
  1275→                    "severity": "low", "category": "image_size",
  1276→                    "description": "apt-get install without --no-install-recommends increases image size and attack surface.",
  1277→                    "evidence": _ev(fname, ln),
  1278→                    "remediation": "Add --no-install-recommends to apt-get install and combine RUN layers.",
  1279→                    "dread_score": _dread("low"), "points_deducted": _pts("low"),
  1280→                })
  1281→
  1282→            # apt-get update without cache cleanup in same layer
  1283→            if re.search(r'apt-get\s+update', run_val) and not re.search(r'rm\s+-rf\s+/var/lib/apt', run_val):
  1284→                issues.append({
  1285→                    "title": "Dockerfile apt-get Cache Not Cleaned",
  1286→                    "severity": "low", "category": "image_size",
  1287→                    "description": "apt-get update without cache cleanup in the same layer increases image size.",
  1288→                    "evidence": _ev(fname, ln),
  1289→                    "remediation": "Add '&& rm -rf /var/lib/apt/lists/*' in the same RUN instruction.",
  1290→                    "dread_score": _dread("low"), "points_deducted": _pts("low"),
  1291→                })
  1292→
  1293→        # ── EXPOSE checks ─────────────────────────────────────────────
  1294→        if stripped.upper().startswith("EXPOSE "):
  1295→            ports = stripped[7:].strip().split()
  1296→            for port in ports:
  1297→                port_clean = port.split("/")[0]
  1298→                if port_clean in _DANGEROUS_PORTS:
  1299→                    svc = _DANGEROUS_PORTS[port_clean]
  1300→                    issues.append({
  1301→                        "title": f"Dockerfile Exposes {svc} (Port {port_clean})",
  1302→                        "severity": "medium", "category": "network_exposure",
  1303→                        "description": f"Dockerfile exposes port {port_clean} ({svc}) which should not be publicly accessible.",
  1304→                        "evidence": _ev(fname, ln),
  1305→                        "remediation": f"Do not EXPOSE {svc} in production. Use Docker networks for internal communication.",
  1306→                        "dread_score": _dread("medium"), "points_deducted": _pts("medium"),
  1307→                    })
  1308→
  1309→        # ── HEALTHCHECK checks ────────────────────────────────────────
  1310→        if stripped.upper().startswith("HEALTHCHECK"):
  1311→            has_healthcheck = True
  1312→
  1313→    # ── Post-scan checks ──────────────────────────────────────────────
  1314→
  1315→    # Check: No USER directive (runs as root)
  1316→    if has_from and not has_user:
  1317→        issues.append({
  1318→            "title": "Dockerfile Has No USER Directive (Runs as Root)",
  1319→            "severity": "high", "category": "container_privilege",
  1320→            "description": "No USER instruction found — the container process runs as root by default.",
  1321→            "evidence": _ev(fname, from_line or 1),
  1322→            "remediation": "Add 'USER <non-root-user>' before the ENTRYPOINT/CMD.",
  1323→            "dread_score": _dread("high"), "points_deducted": _pts("high"),
  1324→        })
  1325→
  1326→    # Check: No HEALTHCHECK
  1327→    if has_from and not has_healthcheck:
  1328→        issues.append({
  1329→            "title": "Dockerfile Has No HEALTHCHECK Instruction",
  1330→            "severity": "low", "category": "monitoring",
  1331→            "description": "No HEALTHCHECK instruction — the orchestrator cannot determine if the container is healthy.",
  1332→            "evidence": _ev(fname, 1),
  1333→            "remediation": "Add a HEALTHCHECK instruction (e.g., HEALTHCHECK CMD curl -f http://localhost:8080/ || exit 1).",
  1334→            "dread_score": _dread("low"), "points_deducted": _pts("low"),
  1335→        })
  1336→
  1337→    # Check: No .dockerignore (info-level — check if .dockerignore exists alongside)
  1338→    parent = Path(path).parent
  1339→    dockerignore = parent / ".dockerignore"
  1340→    if parent.exists() and not dockerignore.exists():
  1341→        issues.append({
  1342→            "title": "No .dockerignore File Detected",
  1343→            "severity": "info", "category": "best_practice",
  1344→            "description": "No .dockerignore found in the Dockerfile's directory — sensitive files may be included in the build context.",
  1345→            "evidence": _ev(fname, 1),
  1346→            "remediation": "Create a .dockerignore file excluding .git, .env, *.pem, node_modules, etc.",
  1347→            "dread_score": _dread("info"), "points_deducted": _pts("info"),
  1348→        })
  1349→
  1350→    return issues

  1358→def _k8s_get_containers(doc: Dict[str, Any]) -> List[Dict[str, Any]]:
  1359→    """Extract all container definitions from a K8s manifest."""
  1360→    containers: List[Dict[str, Any]] = []
  1361→    spec = doc.get("spec", {})
  1362→    if not isinstance(spec, dict):
  1363→        return containers
  1364→
  1365→    for c in spec.get("containers", []):
  1366→        if isinstance(c, dict):
  1367→            containers.append(c)
  1368→    for c in spec.get("initContainers", []):
  1369→        if isinstance(c, dict):
  1370→            containers.append(c)
  1371→    return containers

  1374→def _k8s_get_pods(doc: Dict[str, Any]) -> List[Dict[str, Any]]:
  1375→    """Extract pod specs from a K8s manifest (handles Deployments, StatefulSets, etc.)."""
  1376→    specs: List[Dict[str, Any]] = []
  1377→    spec = doc.get("spec", {})
  1378→    if not isinstance(spec, dict):
  1379→        return specs
  1380→
  1381→    # Direct Pod spec
  1382→    if spec.get("containers") or spec.get("initContainers"):
  1383→        specs.append(spec)
  1384→    # Template spec (Deployment, StatefulSet, DaemonSet, Job, CronJob)
  1385→    template = spec.get("template", {})
  1386→    if isinstance(template, dict):
  1387→        pod_spec = template.get("spec", {})
  1388→        if isinstance(pod_spec, dict) and (pod_spec.get("containers") or pod_spec.get("initContainers")):
  1389→            specs.append(pod_spec)
  1390→        # CronJob nested template
  1391→        job_spec = template.get("spec", {})
  1392→        if isinstance(job_spec, dict):
  1393→            inner_template = job_spec.get("template", {})
  1394→            if isinstance(inner_template, dict):
  1395→                inner_spec = inner_template.get("spec", {})
  1396→                if isinstance(inner_spec, dict) and (inner_spec.get("containers") or inner_spec.get("initContainers")):
  1397→                    specs.append(inner_spec)
  1398→
  1399→    return specs

  1402→def _k8s_check_security_context(pod_spec: Dict[str, Any], doc_name: str, fname: str) -> List[Dict[str, Any]]:
  1403→    """Check pod and container security contexts."""
  1404→    issues: List[Dict[str, Any]] = []
  1405→
  1406→    # Pod-level security context
  1407→    pod_sc = pod_spec.get("securityContext", {})
  1408→    if not isinstance(pod_sc, dict):
  1409→        pod_sc = {}
  1410→
  1411→    # Check: privileged at pod level
  1412→    if pod_sc.get("privileged") is True:
  1413→        issues.append({
  1414→            "title": f"K8s: '{doc_name}' Pod securityContext.privileged=true",
  1415→            "severity": "critical", "category": "container_privilege",
  1416→            "description": f"Pod '{doc_name}' has privileged=true at pod level — full host access.",
  1417→            "evidence": f"{fname}: {doc_name} securityContext.privileged: true",
  1418→            "remediation": "Remove privileged: true. Grant only specific Linux capabilities.",
  1419→            "dread_score": _dread("critical"), "points_deducted": _pts("critical"),
  1420→        })
  1421→
  1422→    # Check: allowPrivilegeEscalation at pod level
  1423→    if pod_sc.get("allowPrivilegeEscalation") is True:
  1424→        issues.append({
  1425→            "title": f"K8s: '{doc_name}' Pod allowPrivilegeEscalation=true",
  1426→            "severity": "high", "category": "container_privilege",
  1427→            "description": f"Pod '{doc_name}' allows privilege escalation.",
  1428→            "evidence": f"{fname}: {doc_name} allowPrivilegeEscalation: true",
  1429→            "remediation": "Set allowPrivilegeEscalation: false.",
  1430→            "dread_score": _dread("high"), "points_deducted": _pts("high"),
  1431→        })
  1432→
  1433→    # Check: runAsUser: 0 at pod level
  1434→    if pod_sc.get("runAsUser") == 0:
  1435→        issues.append({
  1436→            "title": f"K8s: '{doc_name}' Pod Runs as Root (runAsUser: 0)",
  1437→            "severity": "high", "category": "container_privilege",
  1438→            "description": f"Pod '{doc_name}' explicitly runs as root (runAsUser: 0).",
  1439→            "evidence": f"{fname}: {doc_name} runAsUser: 0",
  1440→            "remediation": "Set runAsUser to a non-zero UID (e.g., 1000).",
  1441→            "dread_score": _dread("high"), "points_deducted": _pts("high"),
  1442→        })
  1443→
  1444→    # Check: readOnlyRootFilesystem not true at pod level
  1445→    if pod_sc.get("readOnlyRootFilesystem") is not True and "readOnlyRootFilesystem" in str(pod_sc):
  1446→        pass  # explicitly false — handled below
  1447→    elif "readOnlyRootFilesystem" not in pod_sc and pod_sc:
  1448→        issues.append({
  1449→            "title": f"K8s: '{doc_name}' Pod Writable Root Filesystem",
  1450→            "severity": "medium", "category": "container_hardening",
  1451→            "description": f"Pod '{doc_name}' does not set readOnlyRootFilesystem: true.",
  1452→            "evidence": f"{fname}: {doc_name} missing readOnlyRootFilesystem: true",
  1453→            "remediation": "Set readOnlyRootFilesystem: true in securityContext.",
  1454→            "dread_score": _dread("medium"), "points_deducted": _pts("medium"),
  1455→        })
  1456→
  1457→    # Container-level checks
  1458→    for container in _k8s_get_containers_from_spec(pod_spec):
  1459→        cname = container.get("name", "<unnamed>")
  1460→        csc = container.get("securityContext", {})
  1461→        if not isinstance(csc, dict):
  1462→            csc = {}
  1463→
  1464→        # Privileged at container level
  1465→        if csc.get("privileged") is True:
  1466→            issues.append({
  1467→                "title": f"K8s: '{doc_name}/{cname}' Container privileged=true",
  1468→                "severity": "critical", "category": "container_privilege",
  1469→                "description": f"Container '{cname}' in '{doc_name}' has privileged=true — full host access.",
  1470→                "evidence": f"{fname}: {doc_name}/{cname} securityContext.privileged: true",
  1471→                "remediation": "Remove privileged: true. Grant only specific Linux capabilities.",
  1472→                "dread_score": _dread("critical"), "points_deducted": _pts("critical"),
  1473→            })
  1474→
  1475→        # allowPrivilegeEscalation at container level
  1476→        if csc.get("allowPrivilegeEscalation") is True:
  1477→            issues.append({
  1478→                "title": f"K8s: '{doc_name}/{cname}' allowPrivilegeEscalation=true",
  1479→                "severity": "high", "category": "container_privilege",
  1480→                "description": f"Container '{cname}' allows privilege escalation.",
  1481→                "evidence": f"{fname}: {doc_name}/{cname} allowPrivilegeEscalation: true",
  1482→                "remediation": "Set allowPrivilegeEscalation: false.",
  1483→                "dread_score": _dread("high"), "points_deducted": _pts("high"),
  1484→            })
  1485→
  1486→        # runAsUser: 0 at container level
  1487→        if csc.get("runAsUser") == 0:
  1488→            issues.append({
  1489→                "title": f"K8s: '{doc_name}/{cname}' Runs as Root (runAsUser: 0)",
  1490→                "severity": "high", "category": "container_privilege",
  1491→                "description": f"Container '{cname}' explicitly runs as root.",
  1492→                "evidence": f"{fname}: {doc_name}/{cname} runAsUser: 0",
  1493→                "remediation": "Set runAsUser to a non-zero UID.",
  1494→                "dread_score": _dread("high"), "points_deducted": _pts("high"),
  1495→            })
  1496→
  1497→        # Writable root filesystem at container level
  1498→        if csc.get("readOnlyRootFilesystem") is not True and pod_sc.get("readOnlyRootFilesystem") is not True:
  1499→            issues.append({
  1500→                "title": f"K8s: '{doc_name}/{cname}' Writable Root Filesystem",
  1501→                "severity": "medium", "category": "container_hardening",
  1502→                "description": f"Container '{cname}' has a writable root filesystem.",
  1503→                "evidence": f"{fname}: {doc_name}/{cname} missing readOnlyRootFilesystem: true",
  1504→                "remediation": "Set readOnlyRootFilesystem: true and use emptyDir for writable paths.",
  1505→                "dread_score": _dread("medium"), "points_deducted": _pts("medium"),
  1506→            })
  1507→
  1508→        # Capabilities check
  1509→        cap = csc.get("capabilities", {})
  1510→        if isinstance(cap, dict):
  1511→            adds = cap.get("add", [])
  1512→            if isinstance(adds, list):
  1513→                dangerous_caps = ["ALL", "NET_RAW", "SYS_ADMIN", "SYS_PTRACE", "SYS_MODULE",
  1514→                                  "DAC_OVERRIDE", "NET_ADMIN", "SYSLOG", "KILL"]
  1515→                for c in adds:
  1516→                    if c in dangerous_caps:
  1517→                        sev = "critical" if c == "ALL" else "high"
  1518→                        issues.append({
  1519→                            "title": f"K8s: '{doc_name}/{cname}' Adds Capability {c}",
  1520→                            "severity": sev, "category": "container_privilege",
  1521→                            "description": f"Container '{cname}' adds Linux capability {c}, expanding its privilege.",
  1522→                            "evidence": f"{fname}: {doc_name}/{cname} capabilities.add: [{c}]",
  1523→                            "remediation": f"Remove {c} from capabilities.add. Drop all capabilities and add only those needed.",
  1524→                            "dread_score": _dread(sev), "points_deducted": _pts(sev),
  1525→                        })
  1526→
  1527→        # No resource limits
  1528→        resources = container.get("resources", {})
  1529→        if isinstance(resources, dict):
  1530→            limits = resources.get("limits", {})
  1531→            if not limits:
  1532→                issues.append({
  1533→                    "title": f"K8s: '{doc_name}/{cname}' No Resource Limits",
  1534→                    "severity": "medium", "category": "resource_limit",
  1535→                    "description": f"Container '{cname}' has no resource limits — can consume all node resources.",
  1536→                    "evidence": f"{fname}: {doc_name}/{cname} missing resources.limits",
  1537→                    "remediation": "Set resources.limits for cpu and memory.",
  1538→                    "dread_score": _dread("medium"), "points_deducted": _pts("medium"),
  1539→                })
  1540→            else:
  1541→                # Check for no memory limit specifically
  1542→                if not limits.get("memory"):
  1543→                    issues.append({
  1544→                        "title": f"K8s: '{doc_name}/{cname}' No Memory Limit",
  1545→                        "severity": "medium", "category": "resource_limit",
  1546→                        "description": f"Container '{cname}' has resource limits but no memory limit.",
  1547→                        "evidence": f"{fname}: {doc_name}/{cname} resources.limits missing memory",
  1548→                        "remediation": "Set resources.limits.memory (e.g., '256Mi').",
  1549→                        "dread_score": _dread("medium"), "points_deducted": _pts("medium"),
  1550→                    })
  1551→                # Check for no CPU limit
  1552→                if not limits.get("cpu"):
  1553→                    issues.append({
  1554→                        "title": f"K8s: '{doc_name}/{cname}' No CPU Limit",
  1555→                        "severity": "low", "category": "resource_limit",
  1556→                        "description": f"Container '{cname}' has no CPU limit.",
  1557→                        "evidence": f"{fname}: {doc_name}/{cname} resources.limits missing cpu",
  1558→                        "remediation": "Set resources.limits.cpu (e.g., '500m').",
  1559→                        "dread_score": _dread("low"), "points_deducted": _pts("low"),
  1560→                    })
  1561→
  1562→        # Secrets in env (plain text) vs envFrom/secretRef
  1563→        env_list = container.get("env", [])
  1564→        if isinstance(env_list, list):
  1565→            secret_env_names = []
  1566→            for env_var in env_list:
  1567→                if isinstance(env_var, dict):
  1568→                    name = env_var.get("name", "")
  1569→                    value = env_var.get("value", "")
  1570→                    if value and re.search(r'(?:PASSWORD|SECRET|TOKEN|KEY|CREDENTIAL)', name, re.I):
  1571→                        secret_env_names.append(name)
  1572→            if secret_env_names:
  1573→                issues.append({
  1574→                    "title": f"K8s: '{doc_name}/{cname}' Secrets in Plain ENV",
  1575→                    "severity": "high", "category": "secrets_management",
  1576→                    "description": f"Container '{cname}' has secret values in plain env: {', '.join(secret_env_names[:3])}",
  1577→                    "evidence": f"{fname}: {doc_name}/{cname} env contains secrets",
  1578→                    "remediation": "Use envFrom with a SecretRef or use valueFrom.secretKeyRef.",
  1579→                    "dread_score": _dread("high"), "points_deducted": _pts("high"),
  1580→                })
  1581→
  1582→        # imagePullPolicy: Always (supply chain risk)
  1583→        pull_policy = container.get("imagePullPolicy", "")
  1584→        if pull_policy == "Always":
  1585→            issues.append({
  1586→                "title": f"K8s: '{doc_name}/{cname}' imagePullPolicy: Always",
  1587→                "severity": "low", "category": "supply_chain",
  1588→                "description": f"Container '{cname}' uses imagePullPolicy: Always, pulling latest on every pod start.",
  1589→                "evidence": f"{fname}: {doc_name}/{cname} imagePullPolicy: Always",

  1591→                "dread_score": _dread("low"), "points_deducted": _pts("low"),
  1592→            })
  1593→
  1594→    return issues

  1672→            # Root filesystem mounts
  1673→            dangerous_roots = ["/", "/etc", "/root", "/var", "/usr", "/bin", "/sbin", "/lib"]
  1674→            if mpath in dangerous_roots:
  1675→                issues.append({

  1705→def _k8s_check_tolerations(pod_spec: Dict[str, Any], doc_name: str, fname: str) -> List[Dict[str, Any]]:
  1706→    """Check for overly permissive tolerations."""

  1709→    tolerations = pod_spec.get("tolerations", [])
  1710→    if not isinstance(tolerations, list):

  1713→    for tol in tolerations:

  1715→            continue

  1719→        if operator == "Exists" and not key:

  1725→                "description": f"Pod '{doc_name}' tolerates all taints{effect_desc} — may schedule on dedicated/master nodes.",
  1726→                "evidence": f"{fname}: {doc_name} toleration operator: Exists, key: (empty)",
  1727→                "remediation": "Specify a specific taint key or restrict tolerations to known taints.",
  1728→                "dread_score": _dread(sev), "points_deducted": _pts(sev),
  1729→            })
  1730→
  1731→    return issues

  1734→def parse_k8s_yaml(path: str) -> List[Dict[str, Any]]:
  1735→    """Parse a Kubernetes YAML manifest and return a list of issue dicts."""
  1736→    issues: List[Dict[str, Any]] = []
  1737→    fname = os.path.basename(path)
  1738→
  1739→    try:
  1740→        text = Path(path).read_text(encoding="utf-8", errors="replace")
  1741→    except OSError:
  1742→        return issues
  1743→
  1744→    docs = _yaml_load_all(text)
  1745→
  1746→    for doc in docs:
  1747→        if not isinstance(doc, dict):
  1748→            continue
  1749→
  1750→        kind = str(doc.get("kind", ""))
  1751→        metadata = doc.get("metadata", {})
  1752→        doc_name = metadata.get("name", "<unnamed>") if isinstance(metadata, dict) else "<unnamed>"
  1753→
  1754→        pod_specs = _k8s_get_pods(doc)
  1755→        if not pod_specs:
  1756→            continue
  1757→
  1758→        for pod_spec in pod_specs:
  1759→            # Security context checks
  1760→            issues.extend(_k8s_check_security_context(pod_spec, doc_name, fname))
  1761→            # Host namespace checks
  1762→            issues.extend(_k8s_check_host_settings(pod_spec, doc_name, fname))
  1763→            # Volume mount checks
  1764→            issues.extend(_k8s_check_volumes(pod_spec, doc_name, fname))
  1765→            # Service account checks
  1766→            issues.extend(_k8s_check_service_account(pod_spec, doc_name, fname))
  1767→            # Toleration checks
  1768→            issues.extend(_k8s_check_tolerations(pod_spec, doc_name, fname))
  1769→
  1770→    return issues

  1785→def _is_likely_cloudformation(path: Path, text: str) -> bool:
  1786→    """Heuristic to determine if a YAML/JSON file is a CloudFormation template."""
  1787→    if re.search(r'AWSTemplateFormatVersion|AWS::', text):
  1788→        return True
  1789→    if re.search(r'Resources:\s*$', text, re.M):
  1790→        # Could be K8s or CF — check for CF-specific keys
  1791→        if re.search(r'Type:\s*AWS::|Description:\s*', text):
  1792→            return True
  1793→    return False

  1796→def _is_likely_k8s(path: Path, text: str) -> bool:
  1797→    """Heuristic to determine if a YAML file is a Kubernetes manifest."""
  1798→    return bool(re.search(r'apiVersion:\s*(?:v1|apps/|extensions/|batch/|networking\.k8s\.io|rbac\.authorization)', text))

  1801→def _is_likely_docker_compose(path: Path, text: str) -> bool:
  1802→    """Heuristic to determine if a YAML file is a docker-compose file."""
  1803→    return bool(re.search(r'version:\s*["\']?3|services:\s*$', text, re.M))

  1806→def run(target: str, base_url: str, **kwargs) -> Tuple[List[Finding], int, str, str]:
  1807→    """Scan a directory for IaC security issues.
  1808→
  1809→    Recursively discovers Terraform, CloudFormation, Dockerfile, and
  1810→    Kubernetes manifest files, runs the appropriate parser, and
  1811→    converts all issues to Finding objects.
  1812→
  1813→    Returns:
  1814→        (findings, max_points, module_name, category)
  1815→    """
  1816→    findings: List[Finding] = []
  1817→    target_path = Path(target)
  1818→
  1819→    max_points = 100
  1820→    module_name = "iac_audit"
  1821→    category = "infrastructure_as_code"
  1822→
  1823→    if not target_path.is_dir():
  1824→        # Single file mode
  1825→        _scan_file(target_path, findings)
  1826→    else:
  1827→        # Recursive directory scan
  1828→        for p in sorted(target_path.rglob("*")):
  1829→            if not p.is_file():
  1830→                continue
  1831→            # Skip hidden dirs and common non-IaC dirs
  1832→            parts = p.parts
  1833→            if any(part.startswith(".") for part in parts if part != "."):
  1834→                continue
  1835→            if any(part in ("node_modules", "vendor", "__pycache__", ".git", "venv") for part in parts):
  1836→                continue
  1837→            _scan_file(p, findings)
  1838→
  1839→    # Deduplicate by title + evidence
  1840→    seen: set = set()
  1841→    deduped: List[Finding] = []
  1842→    for f in findings:
  1843→        key = (f.title, f.evidence)
  1844→        if key not in seen:
  1845→            seen.add(key)
  1846→            deduped.append(f)
  1847→
  1848→    return (deduped, max_points, module_name, category)

  1851→def _scan_file(p: Path, findings: List[Finding]) -> None:
  1852→    """Route a file to the appropriate parser."""
  1853→    name = p.name
  1854→    ext = p.suffix.lower()
  1855→
  1856→    try:
  1857→        text = p.read_text(encoding="utf-8", errors="replace")
  1858→    except OSError:
  1859→        return
  1860→
  1861→    # ── Terraform ─────────────────────────────────────────────────────
  1862→    if ext in _TF_EXTS:
  1863→        for issue in parse_tf_file(str(p)):
  1864→            findings.append(Finding(
  1865→                title=issue["title"],
  1866→                severity=issue["severity"],
  1867→                category=issue["category"],
  1868→                module="iac_audit",
  1869→                description=issue["description"],
  1870→                evidence=issue["evidence"],
  1871→                asset=str(p),
  1872→                points_deducted=issue["points_deducted"],
  1873→                remediation=issue["remediation"],
  1874→                dread_score=issue["dread_score"],
  1875→            ))
  1876→        return
  1877→
  1878→    # ── Dockerfile ─────────────────────────────────────────────────────
  1879→    if name in _DOCKERFILE_NAMES:
  1880→        for issue in parse_dockerfile(str(p)):
  1881→            findings.append(Finding(
  1882→                title=issue["title"],
  1883→                severity=issue["severity"],
  1884→                category=issue["category"],
  1885→                module="iac_audit",
  1886→                description=issue["description"],
  1887→                evidence=issue["evidence"],
  1888→                asset=str(p),
  1889→                points_deducted=issue["points_deducted"],
  1890→                remediation=issue["remediation"],
  1891→                dread_score=issue["dread_score"],
  1892→            ))
  1893→        return
  1894→
  1895→    # ── Docker Compose ────────────────────────────────────────────────
  1896→    if name in _DOCKER_COMPOSE_NAMES:
  1897→        # Analyze docker-compose files for security issues
  1898→        for issue in _parse_docker_compose(str(p), text):
  1899→            findings.append(Finding(
  1900→                title=issue["title"],
  1901→                severity=issue["severity"],
  1902→                category=issue["category"],
  1903→                module="iac_audit",
  1904→                description=issue["description"],
  1905→                evidence=issue["evidence"],
  1906→                asset=str(p),
  1907→                points_deducted=issue["points_deducted"],
  1908→                remediation=issue["remediation"],
  1909→                dread_score=issue["dread_score"],
  1910→            ))
  1911→        return
  1912→
  1913→    # ── YAML / JSON: CloudFormation or Kubernetes ──────────────────────
  1914→    if ext in _CF_EXTS:
  1915→        if _is_likely_cloudformation(p, text):
  1916→            for issue in parse_cf_file(str(p)):
  1917→                findings.append(Finding(
  1918→                    title=issue["title"],
  1919→                    severity=issue["severity"],
  1920→                    category=issue["category"],
  1921→                    module="iac_audit",
  1922→                    description=issue["description"],
  1923→                    evidence=issue["evidence"],
  1924→                    asset=str(p),
  1925→                    points_deducted=issue["points_deducted"],
  1926→                    remediation=issue["remediation"],
  1927→                    dread_score=issue["dread_score"],
  1928→                ))
  1929→        elif _is_likely_k8s(p, text):
  1930→            for issue in parse_k8s_yaml(str(p)):
  1931→                findings.append(Finding(
  1932→                    title=issue["title"],
  1933→                    severity=issue["severity"],
  1934→                    category=issue["category"],
  1935→                    module="iac_audit",
  1936→                    description=issue["description"],
  1937→                    evidence=issue["evidence"],
  1938→                    asset=str(p),

  1940→                    remediation=issue["remediation"],

  1943→        # If neither CF nor K8s, skip silently (not an IaC file)

  1949→def _parse_docker_compose(path: str, text: str) -> List[Dict[str, Any]]:
  1950→    """Analyze docker-compose files for security issues."""
  1951→    issues: List[Dict[str, Any]] = []
  1952→    fname = os.path.basename(path)
  1953→    doc = _simple_yaml_load(text)
  1954→    if not isinstance(doc, dict):
  1955→        return issues
  1956→
  1957→    services = doc.get("services", {})
  1958→    if not isinstance(services, dict):
  1959→        return issues
  1960→
  1961→    for svc_name, svc_config in services.items():
  1962→        if not isinstance(svc_config, dict):
  1963→            continue
  1964→
  1965→        # Check: privileged
  1966→        if svc_config.get("privileged") is True:
  1967→            issues.append({
  1968→                "title": f"Docker Compose: '{svc_name}' Runs Privileged",
  1969→                "severity": "critical", "category": "container_privilege",
  1970→                "description": f"Service '{svc_name}' runs with --privileged flag.",
  1971→                "evidence": f"{fname}: services.{svc_name}.privileged: true",
  1972→                "remediation": "Remove privileged: true from the service.",
  1973→                "dread_score": _dread("critical"), "points_deducted": _pts("critical"),
  1974→            })
  1975→
  1976→        # Check: security_opt with no-new-privileges missing
  1977→        sec_opts = svc_config.get("security_opt", [])
  1978→        if isinstance(sec_opts, list) and "no-new-privileges:true" not in sec_opts and "no-new-privileges" not in sec_opts:
  1979→            issues.append({
  1980→                "title": f"Docker Compose: '{svc_name}' Missing no-new-privileges",
  1981→                "severity": "medium", "category": "container_privilege",
  1982→                "description": f"Service '{svc_name}' does not set security_opt: no-new-privileges:true.",
  1983→                "evidence": f"{fname}: services.{svc_name} missing security_opt",
  1984→                "remediation": "Add 'security_opt: [no-new-privileges:true]' to the service.",
  1985→                "dread_score": _dread("medium"), "points_deducted": _pts("medium"),
  1986→            })
  1987→
  1988→        # Check: sensitive ports exposed to host
  1989→        ports = svc_config.get("ports", [])
  1990→        if isinstance(ports, list):
  1991→            for port_mapping in ports:
  1992→                port_str = str(port_mapping)
  1993→                for port_num, svc_desc in _DANGEROUS_PORTS.items():
  1994→                    if port_num in port_str and (":" in port_str or port_str == port_num):
  1995→                        issues.append({
  1996→                            "title": f"Docker Compose: '{svc_name}' Exposes {svc_desc} (Port {port_num})",
  1997→                            "severity": "medium", "category": "network_exposure",
  1998→                            "description": f"Service '{svc_name}' exposes port {port_num} ({svc_desc}) to the host.",
  1999→                            "evidence": f"{fname}: services.{svc_name}.ports: {port_str}",
  2000→                            "remediation": f"Do not bind port {port_num} to the host. Use Docker internal networking.",
  2001→                            "dread_score": _dread("medium"), "points_deducted": _pts("medium"),
  2002→                        })
  2003→
  2004→        # Check: environment secrets
  2005→        env_vars = svc_config.get("environment", {})
  2006→        if isinstance(env_vars, dict):
  2007→            for k, v in env_vars.items():
  2008→                if v and re.search(r'(?:PASSWORD|SECRET|TOKEN|KEY|CREDENTIAL)', k, re.I) and not str(v).startswith("$"):
  2009→                    issues.append({
  2010→                        "title": f"Docker Compose: '{svc_name}' Hardcoded Secret in Environment",
  2011→                        "severity": "high", "category": "hardcoded_secret",
  2012→                        "description": f"Service '{svc_name}' has a hardcoded secret in env variable '{k}'.",
  2013→                        "evidence": f"{fname}: services.{svc_name}.environment.{k}",
  2014→                        "remediation": "Use env_file or Docker secrets instead of hardcoded values.",
  2015→                        "dread_score": _dread("high"), "points_deducted": _pts("high"),
  2016→                    })
  2017→                    break  # One per service
  2018→
  2019→        # Check: no resource limits
  2020→        deploy = svc_config.get("deploy", {})
  2021→        if isinstance(deploy, dict):
  2022→            resources = deploy.get("resources", {})
  2023→            if isinstance(resources, dict):
  2024→                limits = resources.get("limits", {})
  2025→                if not limits:
  2026→                    issues.append({
  2027→                        "title": f"Docker Compose: '{svc_name}' No Resource Limits",
  2028→                        "severity": "medium", "category": "resource_limit",
  2029→                        "description": f"Service '{svc_name}' has no deploy.resources.limits.",
  2030→                        "evidence": f"{fname}: services.{svc_name}.deploy.resources.limits missing",
  2031→                        "remediation": "Add deploy.resources.limits with cpus and memory.",
  2032→                        "dread_score": _dread("medium"), "points_deducted": _pts("medium"),
  2033→                    })
  2034→        else:
  2035→            # No deploy block at all
  2036→            issues.append({
  2037→                "title": f"Docker Compose: '{svc_name}' No Deploy Resource Limits",
  2038→                "severity": "low", "category": "resource_limit",
  2039→                "description": f"Service '{svc_name}' has no deploy block — no resource limits defined.",
  2040→                "evidence": f"{fname}: services.{svc_name} missing deploy.resources.limits",
  2041→                "remediation": "Add deploy.resources.limits with cpus and memory.",
  2042→                "dread_score": _dread("low"), "points_deducted": _pts("low"),
  2043→            })
  2044→
  2045→        # Check: /var/run/docker.sock mount
  2046→        volumes = svc_config.get("volumes", [])
  2047→        if isinstance(volumes, list):
  2048→            for vol in volumes:
  2049→                vol_str = str(vol)
  2050→                if "/var/run/docker.sock" in vol_str:
  2051→                    issues.append({
  2052→                        "title": f"Docker Compose: '{svc_name}' Mounts Docker Socket",
  2053→                        "severity": "critical", "category": "container_privilege",
  2054→                        "description": f"Service '{svc_name}' mounts docker.sock — full host container escape.",
  2055→                        "evidence": f"{fname}: services.{svc_name}.volumes: {vol_str}",
  2056→                        "remediation": "Never mount docker.sock. Use Docker APIs directly.",
  2057→                        "dread_score": _dread("critical"), "points_deducted": _pts("critical"),
  2058→                    })
  2059→
  2060→        # Check: read_only not set
  2061→        if not svc_config.get("read_only"):
  2062→            issues.append({
  2063→                "title": f"Docker Compose: '{svc_name}' Writable Root Filesystem",
  2064→                "severity": "low", "category": "container_hardening",
  2065→                "description": f"Service '{svc_name}' does not set read_only: true.",
  2066→                "evidence": f"{fname}: services.{svc_name} missing read_only: true",
  2067→                "remediation": "Set read_only: true and use tmpfs/volumes for writable directories.",
  2068→                "dread_score": _dread("low"), "points_deducted": _pts("low"),
  2069→            })
  2070→
  2071→    return issues
