   285→class TestFindingCorrelator(unittest.TestCase):
   286→
   287→    def setUp(self):
   288→        self.correlator = FindingCorrelator()
   289→
   290→    def test_correlate_empty(self):
   291→        groups, deduped = self.correlator.correlate([])
   292→        self.assertEqual(groups, [])
   293→        self.assertEqual(deduped, [])
   294→
   295→    def test_dedup_identical(self):
   296→        findings = [_finding(), _finding()]
   297→        groups, deduped = self.correlator.correlate(findings)
   298→        self.assertEqual(len(deduped), 1)
   299→        # Should have 0 correlation groups (single finding can't correlate)
   300→        self.assertEqual(len(groups), 0)
   301→
   302→    def test_dedup_different(self):
   303→        findings = [_finding(), _finding_xss()]
   304→        groups, deduped = self.correlator.correlate(findings)
   305→        self.assertEqual(len(deduped), 2)
   306→
   307→    def test_group_by_asset(self):
   308→        findings = [
   309→            _finding(asset="example.com/login"),
   310→            _finding_xss().update({"asset": "example.com/login"}) or _finding_xss(),
   311→        ]
   312→        findings[1]["asset"] = "example.com/login"
   313→        groups, deduped = self.correlator.correlate(findings)
   314→        asset_groups = [g for g in groups if g.correlation_type == "shared_asset"]
   315→        self.assertGreater(len(asset_groups), 0)
   316→
   317→    def test_detect_attack_chain(self):
   318→        findings = [_finding_auth_bypass(), _finding_default_creds()]
   319→        groups, deduped = self.correlator.correlate(findings)
   320→        chain_groups = [g for g in groups if g.correlation_type == "attack_chain"]
   321→        self.assertGreater(len(chain_groups), 0)
   322→
   323→    def test_correlation_group_dataclass(self):
   324→        cg = CorrelationGroup(
   325→            findings=[_finding()],
   326→            correlation_type="shared_asset",
   327→            combined_risk="high",
   328→            combined_confidence=0.8,
   329→        )
   330→        self.assertEqual(cg.correlation_type, "shared_asset")
   331→        self.assertEqual(cg.combined_risk, "high")
   332→
   333→    def test_dedup_keeps_more_evidence(self):
   334→        f1 = _finding(evidence="short")
   335→        f2 = _finding(evidence="A" * 200)
   336→        groups, deduped = self.correlator.correlate([f1, f2])
   337→        self.assertEqual(len(deduped), 1)
   338→        self.assertGreater(len(deduped[0].get("evidence", "")), 50)
