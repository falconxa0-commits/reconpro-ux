   120→   120→   120→// ─── Mini severity chart (SVG) ───────────────────────────────
   121→   121→   121→
   122→   122→   122→function SeverityDonut({ data }: { data: { name: string; value: number; color: string }[] }) {
   123→   123→   123→  const total = data.reduce((s, d) => s + d.value, 0);
   124→   124→   124→  const segments = useMemo(() => {
   125→   125→   125→    if (total === 0) return [];
   126→   126→   126→    let off = 0;
   127→   127→   127→    return data.map((d) => {
   128→   128→   128→      const pct = d.value / total;
   129→   129→   129→      const seg = { ...d, pct, dasharray: `${pct * 283} ${283}`, offset: -off };
   130→   130→   130→      off += pct * 283;
   131→   131→   131→      return seg;
   132→   132→   132→    });
   133→   133→   133→  }, [data, total]);
   134→   134→   134→  if (total === 0) return <div className="text-xs text-muted-foreground text-center py-4">No data yet</div>;
   135→   135→   135→  return (
   136→   136→   136→    <svg viewBox="0 0 120 120" className="w-32 h-32 mx-auto">
   137→   137→   137→      {segments.map((d) => (
   138→   138→   138→        <circle
   139→   139→   139→          key={d.name}
   140→   140→   140→          cx="60" cy="60" r="45"
   141→   141→   141→          fill="none"
   142→   142→   142→          stroke={d.color}
   143→   143→   143→          strokeWidth="18"
   144→   144→   144→          strokeDasharray={d.dasharray}
   145→   145→   145→          strokeDashoffset={d.offset}
   146→   146→   146→          strokeLinecap="round"
   147→   147→   147→          opacity={0.85}
   148→   148→   148→        />
   149→   149→   149→      ))}
   150→   150→   150→      <text x="60" y="56" textAnchor="middle" fill="#e6edf3" fontSize="22" fontWeight="bold" fontFamily="Geist Sans, sans-serif">{total}</text>
   151→   151→   151→      <text x="60" y="72" textAnchor="middle" fill="#7d8590" fontSize="9" fontFamily="Geist Sans, sans-serif">FINDINGS</text>
   152→   152→   152→    </svg>
   153→   153→   153→  );
   154→   154→   154→}
