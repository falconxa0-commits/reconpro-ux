    92→    92→function mapFrameworkStatus(

   103→   103→    default:

   175→   175→  const offset = circumference - (score / 100) * circumference;

   178→   178→  return (
   179→   179→    <div className="relative inline-flex items-center justify-center">
   180→   180→      <svg width="180" height="180" viewBox="0 0 180 180" className="transform -rotate-90">
   181→   181→        {/* Background circle */}
   182→   182→        <circle cx="90" cy="90" r={radius} fill="none" stroke="#161b22" strokeWidth="10" />
   183→   183→        {/* Progress arc */}
   184→   184→        <motion.circle
   185→   185→          cx="90"
   186→   186→          cy="90"
   187→   187→          r={radius}
   188→   188→          fill="none"
   189→   189→          stroke={color}
   190→   190→          strokeWidth="10"
   191→   191→          strokeLinecap="round"
   192→   192→          strokeDasharray={circumference}
   193→   193→          initial={{ strokeDashoffset: circumference }}

   218→   218→function TrendSparkline() {
   219→   219→  const points = [62, 65, 68, 72, 70, 75, 78, 82, 85, 84, 87, 89];
   220→   220→  const w = 200;
   221→   221→  const h = 40;
   222→   222→  const step = w / (points.length - 1);
   223→   223→
   224→   224→  const pathData = points
   225→   225→    .map((p, i) => `${i === 0 ? 'M' : 'L'} ${i * step} ${h - (p / 100) * h}`)
   226→   226→    .join(' ');
   227→   227→
   228→   228→  const areaData = `${pathData} L ${w} ${h} L 0 ${h} Z`;
   229→   229→
   230→   230→  return (
   231→   231→    <svg width={w} height={h} viewBox={`0 0 ${w} ${h}`} className="overflow-visible">
   232→   232→      <defs>
   233→   233→        <linearGradient id="sparkGrad" x1="0" y1="0" x2="0" y2="1">
   234→   234→          <stop offset="0%" stopColor="#00ff88" stopOpacity="0.3" />
   235→   235→          <stop offset="100%" stopColor="#00ff88" stopOpacity="0" />
   236→   236→        </linearGradient>
   237→   237→      </defs>
   238→   238→      <motion.path
   239→   239→        d={areaData}
   240→   240→        fill="url(#sparkGrad)"
   241→   241→        initial={{ opacity: 0 }}
   242→   242→        animate={{ opacity: 1 }}
   243→   243→        transition={{ delay: 0.8, duration: 0.6 }}
   244→   244→      />
   245→   245→      <motion.path
   246→   246→        d={pathData}
   247→   247→        fill="none"
   248→   248→        stroke="#00ff88"
   249→   249→        strokeWidth="2"
   250→   250→        strokeLinecap="round"
   251→   251→        strokeLinejoin="round"
   252→   252→        initial={{ pathLength: 0 }}
   253→   253→        animate={{ pathLength: 1 }}
   254→   254→        transition={{ delay: 0.5, duration: 1.2, ease: 'easeOut' as const }}
   255→   255→      />
   256→   256→      <motion.circle
   257→   257→        cx={(points.length - 1) * step}
   258→   258→        cy={h - (points[points.length - 1] / 100) * h}
   259→   259→        r="3"
   260→   260→        fill="#00ff88"
   261→   261→        initial={{ opacity: 0, scale: 0 }}
   262→   262→        animate={{ opacity: 1, scale: 1 }}
   263→   263→        transition={{ delay: 1.5, duration: 0.3 }}
   264→   264→      />
   265→   265→    </svg>
   266→   266→  );
   267→   267→}

   374→   374→export function CompliancePanel({ framework = 'all' }: CompliancePanelProps) {
   375→   375→  const authHeaders = useAuthHeaders();
   376→   376→  const [frameworks, setFrameworks] = useState<ComplianceFramework[]>([]);
   377→   377→  const [overallScore, setOverallScore] = useState(0);
   378→   378→  const [loading, setLoading] = useState(true);
   379→   379→  const [regenerating, setRegenerating] = useState(false);
   380→   380→  const [selectedFramework, setSelectedFramework] = useState(framework === 'all' ? 'soc2' : framework);
   381→   381→  const [expandedControl, setExpandedControl] = useState<string | null>(null);
   382→   382→  const [controlStates, setControlStates] = useState<Record<string, boolean>>({});
   383→   383→  const selectedFrameworkRef = useRef(selectedFramework);
   384→   384→  selectedFrameworkRef.current = selectedFramework;
   385→   385→
   386→   386→  const fetchCompliance = useCallback(async () => {
   387→   387→    try {
   388→   388→      const res = await fetch('/api/compliance', { headers: authHeaders });
   389→   389→      if (!res.ok) throw new Error('Failed to fetch compliance data');
   390→   390→      const data: ApiResponse = await res.json();
   391→   391→      const mapped = data.frameworks.map(mapApiToFramework);
   392→   392→      setFrameworks(mapped);
   393→   393→      setOverallScore(data.overallScore);
   394→   394→      // Auto-select the first framework if the selected one isn't available
   395→   395→      const current = selectedFrameworkRef.current;
   396→   396→      if (mapped.length > 0 && !mapped.find((f) => f.id === current)) {
   397→   397→        setSelectedFramework(mapped[0].id);
   398→   398→      }
   399→   399→    } catch (err) {
   400→   400→      console.error('Failed to load compliance data:', err);
   401→   401→    } finally {
   402→   402→      setLoading(false);
   403→   403→      setRegenerating(false);
   404→   404→    }
   405→   405→  }, [authHeaders]);
   406→   406→
   407→   407→  useEffect(() => {
   408→   408→    fetchCompliance();
   409→   409→  }, [fetchCompliance]);
   410→   410→
   411→   411→  const handleRegenerate = () => {
   412→   412→    setRegenerating(true);
   413→   413→    fetchCompliance();
   414→   414→  };
   415→   415→
   416→   416→  if (loading) {
   417→   417→    return <LoadingSkeleton />;
   418→   418→  }
   419→   419→
   420→   420→  if (frameworks.length === 0) {
   421→   421→    return (
   422→   422→      <div className="w-full flex flex-col items-center justify-center py-20 text-center">
   423→   423→        <Shield className="w-12 h-12 text-[#333333] mb-4" />
   424→   424→        <h3 className="text-lg font-semibold text-[#f0f0f0] mb-2">No Compliance Data</h3>
   425→   425→        <p className="text-sm text-[#444444] mb-4">Run a scan first to generate compliance assessments.</p>
   426→   426→        <Button variant="outline" size="sm" onClick={handleRegenerate} className="border-[#21262d] text-[#444444] hover:bg-[rgba(52,211,153,0.1)] hover:text-[#00ff88] gap-1.5">
   427→   427→          <RefreshCw className={`w-3.5 h-3.5 ${regenerating ? 'animate-spin' : ''}`} />
   428→   428→          Refresh
   429→   429→        </Button>
   430→   430→      </div>
   431→   431→    );
   432→   432→  }
   433→   433→
   434→   434→  const selected = frameworks.find((f) => f.id === selectedFramework) || frameworks[0];
   435→   435→
   436→   436→  const toggleControlStatus = (controlId: string) => {
   437→   437→    setControlStates((prev) => ({ ...prev, [controlId]: !prev[controlId] }));
   438→   438→  };
   439→   439→
   440→   440→  const passedCount = selected.controls.filter((c) => c.status === 'pass').length;
   441→   441→  const failedCount = selected.controls.filter((c) => c.status === 'fail').length;
   442→   442→  const controlPassRate = selected.controls.length > 0
   443→   443→    ? Math.round((passedCount / selected.controls.length) * 100)
   444→   444→    : 0;
   445→   445→
   446→   446→  return (
   447→   447→    <motion.div
   448→   448→      variants={containerVariants}
   449→   449→      initial="hidden"
   450→   450→      animate="visible"
   451→   451→      className="w-full space-y-6"
   452→   452→    >
   453→   453→      {/* ── Header ──────────────────────────────────────────────────────── */}
   454→   454→      <motion.div variants={itemVariants} className="flex flex-col sm:flex-row sm:items-center justify-between gap-4">
   455→   455→        <div className="flex items-center gap-3">
   456→   456→          <div className="p-2 rounded-lg bg-[rgba(52,211,153,0.1)] border border-[rgba(52,211,153,0.2)]">
   457→   457→            <Shield className="w-5 h-5 text-[#00ff88]" />
