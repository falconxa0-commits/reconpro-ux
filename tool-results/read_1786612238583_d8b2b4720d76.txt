   799→function computeThreatScore(findings: any[], injection: any[], multiTurn: any[], cves: any[], secrets: any[], chains: any[]): { score: number; level: string } {
   800→  let score = 0;
   801→  score += findings.filter(f => f.vulnerable).length * 8;
   802→  score += injection.filter(r => r.bypassSuccess).length * 10;
   803→  score += multiTurn.filter(r => r.bypassSuccess).length * 8;
   804→  score += cves.reduce((s, c) => s + Math.min(c.cvss, 10), 0);
   805→  score += secrets.length * 12;
   806→  score += chains.reduce((s, c) => s + ({ critical: 15, high: 10, medium: 5, low: 2 } as Record<string, number>)[c.severity] || 1, 0);
   807→  score = Math.min(100, score);
   808→  const level = score >= 75 ? 'CRITICAL' : score >= 50 ? 'HIGH' : score >= 25 ? 'MEDIUM' : 'LOW';
   809→  return { score, level };
   810→}
