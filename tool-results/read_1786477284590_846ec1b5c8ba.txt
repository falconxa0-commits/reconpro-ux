   213→    def shortest_path(
   214→        self, source: str, target: str
   215→    ) -> Optional[List[Dict[str, Any]]]:
   216→        """BFS shortest path between *source* and *target* nodes.
   217→
   218→        Returns a list of node-data dicts, or ``None`` if no path exists.
   219→        """
   220→        with self._lock:
   221→            src_id = self._graph._resolve_node(source)
   222→            tgt_id = self._graph._resolve_node(target)
   223→            if src_id is None or tgt_id is None:
   224→                return None
   225→            if src_id == tgt_id:
   226→                if src_id in (self._graph._g.nodes() if hasattr(self._graph._g, 'has_node') else []):
   227→                    return []
   228→                return None
   229→
   230→            # BFS
   231→            from collections import deque
   232→            visited: Set[str] = set()
   233→            parent: Dict[str, Optional[str]] = {}
   234→            queue: deque = deque()
   235→            queue.append(src_id)
   236→            visited.add(src_id)
   237→            parent[src_id] = None
   238→            found = False
   239→            while queue and not found:
   240→                current = queue.popleft()
   241→                for neighbour in self._graph._g.successors(current):
   242→                    if neighbour not in visited:
   243→                        visited.add(neighbour)
   244→                        parent[neighbour] = current
   245→                        if neighbour == tgt_id:
   246→                            found = True
   247→                            break
   248→                        queue.append(neighbour)
   249→
   250→            if not found:
   251→                return None
   252→
   253→            # Reconstruct path
   254→            path_ids: List[str] = []
   255→            cur: Optional[str] = tgt_id
   256→            while cur is not None:
   257→                path_ids.append(cur)
   258→                cur = parent.get(cur)
   259→            path_ids.reverse()
   260→
   261→            result: List[Dict[str, Any]] = []
   262→            for nid in path_ids:
   263→                try:
   264→                    data = dict(self._graph._g.nodes[nid])
   265→                except Exception:
   266→                    data = self._graph._g.get_node_data(nid)
   267→                result.append(data)
   268→
   269→            return result

   423→    def trend(self, target: str, days: int = 30) -> List[Dict[str, Any]]:
   424→        """Daily severity breakdown for *target* over the last *days* days.
   425→
   426→        Returns a list of dicts, one per day that has findings::
   427→
   428→            [{"date": "2025-01-15", "critical": 0, "high": 1, "medium": 3, "low": 0, "score": 7.0}, ...]
   429→        """
   430→        with self._lock:
   431→            self._ensure_findings_loaded()
   432→            entries = self._findings.get(target, [])
   433→
   434→            now = datetime.now(timezone.utc)
   435→            cutoff = now - timedelta(days=days)
   436→
   437→            daily: Dict[str, Dict[str, int]] = defaultdict(
   438→                lambda: {"critical": 0, "high": 0, "medium": 0, "low": 0, "info": 0}
   439→            )
   440→
   441→            for entry in entries:
   442→                ts = entry.get("_ts", "")
   443→                if not ts:
   444→                    continue
   445→                try:
   446→                    dt = datetime.fromisoformat(ts)
   447→                except (ValueError, TypeError):
   448→                    continue
   449→                if dt < cutoff:
   450→                    continue
   451→                day_str = dt.strftime("%Y-%m-%d")
   452→                sev = str(entry.get("severity", "info")).strip().lower()
   453→                if sev in daily[day_str]:
   454→                    daily[day_str][sev] += 1
   455→                else:
   456→                    daily[day_str]["info"] += 1
   457→
   458→            # Build result sorted by date
   459→            result: List[Dict[str, Any]] = []
   460→            for day in sorted(daily.keys()):
   461→                counts = daily[day]
   462→                score = (
   463→                    counts.get("critical", 0) * 10.0
   464→                    + counts.get("high", 0) * 5.0
   465→                    + counts.get("medium", 0) * 2.0
   466→                    + counts.get("low", 0) * 0.5
   467→                )
   468→                result.append({
   469→                    "date": day,
   470→                    "critical": counts.get("critical", 0),
   471→                    "high": counts.get("high", 0),
   472→                    "medium": counts.get("medium", 0),
   473→                    "low": counts.get("low", 0),
   474→                    "score": round(score, 1),
   475→                })
   476→
   477→            # Fill in missing days
   478→            if result:
   479→                start_date = datetime.strptime(result[0]["date"], "%Y-%m-%d").date()
   480→                end_date = now.date()
   481→                filled: List[Dict[str, Any]] = []
   482→                idx = 0
   483→                current = start_date
   484→                while current <= end_date:
   485→                    day_str = current.isoformat()
   486→                    if idx < len(result) and result[idx]["date"] == day_str:
   487→                        filled.append(result[idx])
   488→                        idx += 1
   489→                    else:
   490→                        filled.append({
   491→                            "date": day_str,
   492→                            "critical": 0,
   493→                            "high": 0,
   494→                            "medium": 0,
   495→                            "low": 0,
   496→                            "score": 0.0,
   497→                        })
   498→                    current += timedelta(days=1)
   499→                return filled
   500→
   501→            return result

   573→    @staticmethod
   574→    def _machine_key() -> bytes:
   575→        """Derive a deterministic obfuscation key from machine identity."""
   576→        raw = f"reconpro-vault:{platform.node()}:{platform.system()}"
   577→        return hashlib.sha256(raw.encode("utf-8")).digest()

   579→    @staticmethod
   580→    def _obfuscate(plaintext: str, key: bytes) -> str:
   581→        """XOR + base64 obfuscation (NOT cryptographic encryption)."""
   582→        data = plaintext.encode("utf-8")
   583→        xored = bytes(b ^ key[i % len(key)] for i, b in enumerate(data))
   584→        return base64.b64encode(xored).decode("ascii")
   585→
   586→    @staticmethod
   587→    def _deobfuscate(ciphertext: str, key: bytes) -> str:
   588→        """Reverse the XOR + base64 obfuscation."""
   589→        try:
   590→            xored = base64.b64decode(ciphertext.encode("ascii"))
   591→            key_bytes = key
   592→            data = bytes(b ^ key_bytes[i % len(key_bytes)] for i, b in enumerate(xored))
   593→            return data.decode("utf-8")
   594→        except Exception:
   595→            return ""

   668→    def vault_search(self, query: str) -> List[Dict[str, str]]:
   669→        """Search credentials by *query* (matches username, url, notes, source).
   670→
   671→        Case-insensitive substring match.
   672→        """
   673→        with self._lock:
   674→            self._ensure_vault_loaded()
   675→            key = self._machine_key()
   676→            q_lower = query.lower()
   677→            results: List[Dict[str, str]] = []
   678→            for entry in self._vault:
   679→                searchable = " ".join([
   680→                    entry.get("source", ""),
   681→                    entry.get("username", ""),
   682→                    entry.get("url", ""),
   683→                    entry.get("notes", ""),
   684→                ]).lower()
   685→                if q_lower in searchable:
   686→                    decrypted = dict(entry)
   687→                    decrypted["password"] = self._deobfuscate(
   688→                        entry.get("password", ""), key
   689→                    )
   690→                    decrypted.pop("_ts", None)
   691→                    results.append(decrypted)
   692→            return results
   693→
   694→    def vault_purge(self) -> None:
   695→        """Securely clear all credentials from memory and disk."""
   696→        with self._lock:
   697→            # Overwrite in-memory data
   698→            for entry in self._vault:
   699→                for k in list(entry.keys()):
   700→                    entry[k] = "\x00" * len(str(entry[k]))
   701→            self._vault.clear()
   702→            self._vault_loaded = True
   703→
   704→            # Remove or overwrite file
   705→            if VAULT_PATH.is_file():
   706→                try:
   707→                    size = VAULT_PATH.stat().st_size
   708→                    VAULT_PATH.write_bytes(b"\x00" * size)
   709→                    VAULT_PATH.unlink()
   710→                except Exception:
   711→                    pass
   712→

   714→    #  e) Integration methods  (~50 lines)

   717→    def add_scan_result(self, scan_dict: Dict[str, Any]) -> None:
   718→        """Ingest a full scan result dict.
   719→
   720→        Populates both the knowledge graph and the finding store.
   721→
   722→        Expected keys (all optional)::
   723→            target, host, subdomains, ports, technologies,
   724→            findings, certificates, endpoints, module, scan_time
   725→        """
   726→        with self._lock:
   727→            # ── Feed the knowledge graph ──
   728→            self._graph.add_scan_result(scan_dict)
   729→
   730→            # ── Feed the finding store ──
   731→            target = scan_dict.get("target") or scan_dict.get("host", "unknown")
   732→            module = scan_dict.get("module", "")
   733→            scan_time = scan_dict.get("scan_time", "")
   734→
   735→            for finding in scan_dict.get("findings", []):
   736→                entry = dict(finding)
   737→                entry["_target"] = target
   738→                if module:
   739→                    entry["module"] = module
   740→                if scan_time and "_ts" not in entry:
   741→                    entry["_ts"] = scan_time
   742→                self._findings[target].append(entry)
   743→
   744→            # Also record a summary entry for the scan itself
   745→            summary: Dict[str, Any] = {
   746→                "_target": target,
   747→                "title": f"Scan completed: {module or 'general'}",
   748→                "severity": "info",
   749→                "module": module,
   750→                "finding_count": len(scan_dict.get("findings", [])),
   751→                "port_count": len(scan_dict.get("ports", [])),
   752→                "tech_count": len(scan_dict.get("technologies", [])),
   753→            }
   754→            if scan_time:
   755→                summary["_ts"] = scan_time
   756→            self._findings[target].append(summary)
   757→
   758→            # Flush findings for this target
   759→            self._flush_findings_target(target)
   760→
   761→    def export_all(self) -> Dict[str, Any]:
   762→        """Export everything for backup.
   763→
   764→        Returns a dict with keys ``graph``, ``findings``, ``vault``.
   765→        """
   766→        with self._lock:
   767→            # Graph
   768→            try:
   769→                graph_data = self._graph.save()
   770→                with open(graph_data, "r") as fh:
   771→                    graph_json = json.load(fh)
   772→            except Exception:
   773→                graph_json = {}
   774→
   775→            # Findings
   776→            self._ensure_findings_loaded()
   777→            findings_data = {
   778→                target: list(entries)
   779→                for target, entries in self._findings.items()
   780→            }
   781→
   782→            # Vault (export obfuscated — caller can deobfuscate if needed)
   783→            self._ensure_vault_loaded()
   784→            vault_data = list(self._vault)
   785→
   786→            return {
   787→                "graph": graph_json,
   788→                "findings": findings_data,
   789→                "vault": vault_data,
   790→                "exported_at": datetime.now(timezone.utc).isoformat(),
   791→            }
   792→
   793→    def clear_all(self) -> None:
   794→        """Wipe all memory: graph, findings, blackboard, and vault."""
   795→        with self._lock:
   796→            # Graph
   797→            self._graph = KnowledgeGraph()
   798→
   799→            # Findings
   800→            self._findings.clear()
   801→            self._findings_loaded = True
   802→            if FINDINGS_DIR.is_dir():
   803→                for f in FINDINGS_DIR.glob("findings_*.json"):
   804→                    try:
   805→                        f.unlink()
   806→                    except Exception:
   807→                        pass
   808→
   809→            # Blackboard
   810→            self._blackboard.clear()
   811→
   812→            # Vault
   813→            self.vault_purge()
   814→
   815→    def save(self) -> None:
   816→        """Persist graph + findings + vault to disk."""
   817→        with self._lock:
   818→            # Graph
   819→            try:
   820→                self._graph.save()
   821→            except Exception:
   822→                pass
   823→
   824→            # Findings
   825→            self._flush_all_findings()
   826→
   827→            # Vault
   828→            if self._vault:
   829→                self._flush_vault()
   830→
   831→    def load(self) -> None:
   832→        """Restore graph + findings + vault from disk."""
   833→        with self._lock:
   834→            # Graph
   835→            try:
   836→                self._graph.load()
   837→            except Exception:
   838→                pass
   839→
   840→            # Findings
   841→            self._findings_loaded = False
   842→            self._ensure_findings_loaded()
   843→
   844→            # Vault
   845→            self._vault_loaded = False
   846→            self._ensure_vault_loaded()
   847→

   849→    #  Convenience / stats

   851→
   852→    def stats(self) -> Dict[str, Any]:
   853→        """Return a unified stats dict across all sub-systems."""
   854→        with self._lock:
   855→            self._ensure_findings_loaded()
   856→            self._ensure_vault_loaded()
   857→
   858→            total_findings = sum(len(v) for v in self._findings.values())
   859→            target_count = len(self._findings)
   860→
   861→            # Severity breakdown from findings
   862→            sev_counts: Dict[str, int] = defaultdict(int)
   863→            for entries in self._findings.values():
   864→                for e in entries:
   865→                    s = str(e.get("severity", "info")).strip().lower()
   866→                    sev_counts[s] += 1
   867→
   868→            return {
   869→                "graph": self._graph.stats(),
   870→                "findings": {
   871→                    "total_findings": total_findings,
   872→                    "targets_tracked": target_count,
   873→                    "severity_distribution": dict(sev_counts),
   874→                },
   875→                "blackboard": {
   876→                    "active_entries": len(self._blackboard),
   877→                },
   878→                "vault": {
   879→                    "total_credentials": len(self._vault),
   880→                },
   881→            }

   888→_MEMORY_INSTANCE: Optional[UnifiedMemoryStore] = None
   889→_MEMORY_LOCK = threading.Lock()

   891→
   892→def get_memory() -> UnifiedMemoryStore:
   893→    """Return the module-level ``UnifiedMemoryStore`` singleton.
   894→
   895→    Thread-safe: only one instance is ever created per process.
   896→    """
   897→    global _MEMORY_INSTANCE
   898→    if _MEMORY_INSTANCE is None:
   899→        with _MEMORY_LOCK:
   900→            if _MEMORY_INSTANCE is None:
   901→                _MEMORY_INSTANCE = UnifiedMemoryStore()
   902→    return _MEMORY_INSTANCE
