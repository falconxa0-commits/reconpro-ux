    20→# -- ANSI helpers --------------------------------------------------------------
    21→
    22→def _c(code: str, text: str) -> str:
    23→    """Wrap *text* in an ANSI escape sequence."""
    24→    return f"\033[{code}m{text}\033[0m"
    25→
    26→RED = "91"
    27→GREEN = "92"
    28→YELLOW = "93"
    29→BRIGHT_RED = "91;1"
    30→BRIGHT_GREEN = "92;1"
    31→BRIGHT_CYAN = "96"
    32→BRIGHT_WHITE = "97;1"
    33→DIM = "2"
    34→BOLD = "1"
    35→
    36→def red(t: str) -> str:   return _c(RED, t)
    37→def green(t: str) -> str: return _c(GREEN, t)
    38→def yellow(t: str) -> str: return _c(YELLOW, t)
    39→def cyan(t: str) -> str:   return _c(BRIGHT_CYAN, t)
    40→def white(t: str) -> str:  return _c(BRIGHT_WHITE, t)
    41→def dim(t: str) -> str:   return _c(DIM, t)
    42→def bold(t: str) -> str:  return _c(BOLD, t)

   586→
   587→def _match_category(finding_title: str, finding_category: str) -> Optional[str]:
   588→    """Try to map a finding to a remediation rule key."""
   589→    combined = f"{finding_category} {finding_title}".lower()
   590→    best_key: Optional[str] = None
   591→    best_len = 0
   592→    for alias, rule_key in _CATEGORY_ALIAS.items():
   593→        if alias.lower() in combined:
   594→            if len(alias) > best_len:
   595→                best_len = len(alias)
   596→                best_key = rule_key
   597→    return best_key

   688→# -- Adversarial Loop ----------------------------------------------------------
   689→
   690→class AdversarialLoop:
   691→    """Runs HACKER vs CODER vs GUARDIAN in an iterative remediation loop."""
   692→
   693→    def __init__(
   694→        self,
   695→        target: str,
   696→        max_rounds: int = 3,
   697→        modules: Optional[List[str]] = None,
   698→        is_local: bool = False,
   699→    ) -> None:
   700→        self.target = target
   701→        self.max_rounds = max_rounds
   702→        self.modules = modules
   703→        self.is_local = is_local
   704→        self._all_actions: List[RemediationAction] = []
   705→        self._fixed_titles: set = set()
   706→        self._failed_titles: set = set()
   707→        self._feedback_context: Dict[str, List[str]] = {}
   708→        self._converged = False
   709→
   710→    # -- HACKER phase ---------------------------------------------------------
   711→
   712→    def _run_hacker(self) -> Tuple[List[Dict[str, Any]], int, str]:
   713→        """Run a deep scan and return (findings, score, grade)."""
   714→        print(f"  {red('>>  HACKER')}  Scanning {white(self.target)} ...")
   715→        try:
   716→            if self.is_local:
   717→                result = audit_scan(target=self.target, modules=self.modules)
   718→            else:
   719→                result = scan(target=self.target, modules=self.modules, all_modules=False)
   720→            findings = result.findings
   721→            score = result.total_score
   722→            grade = result.grade
   723→        except Exception as exc:
   724→            print(f"    {dim(f'Scan error: {exc}')}")
   725→            return [], 100, "A"
   726→        print(f"    {red(str(len(findings)))} finding(s) discovered")
   727→        return findings, score, grade
   728→
   729→    # -- CODER phase ----------------------------------------------------------
   730→
   731→    def _run_coder(
   732→        self,
   733→        findings: List[Dict[str, Any]],
   734→        round_num: int,
   735→    ) -> List[RemediationAction]:
   736→        """Generate remediation actions for each finding."""
   737→        print(f"  {green('>>  CODER')}  Generating fixes ...")
   738→        actions: List[RemediationAction] = []
   739→        for f in findings:
   740→            title = f.get("title", "")
   741→            category = f.get("category", "")
   742→            severity = f.get("severity", "info")
   743→
   744→            # Skip already-fixed or already-attempted findings
   745→            if title in self._fixed_titles:
   746→                continue
   747→            if title in self._failed_titles:
   748→                continue
   749→
   750→            rule_key = _match_category(title, category)
   751→            if rule_key and rule_key in REMEDIATION_RULES:
   752→                rule = REMEDIATION_RULES[rule_key]
   753→                # Enrich with feedback from previous rounds if available
   754→                fix_cmds = list(rule["fix_commands"])
   755→                if title in self._feedback_context:
   756→                    prior_notes = self._feedback_context[title]
   757→                    fix_cmds.append("")
   758→                    fix_cmds.append("# -- GUARDIAN feedback from prior round(s) --")
   759→                    for note in prior_notes:
   760→                        fix_cmds.append(f"#   > {note}")
   761→                    fix_cmds.append("#   Ensure the above commands are applied before re-verification.")
   762→
   763→                action = RemediationAction(
   764→                    finding_title=title,
   765→                    finding_category=category,
   766→                    severity=severity,
   767→                    rule_key=rule_key,
   768→                    rule_description=rule["description"],
   769→                    fix_commands=fix_cmds,
   770→                    confidence=rule["confidence"],
   771→                    round_applied=round_num,
   772→                )
   773→            else:
   774→                # Generic remediation for unmapped categories
   775→                action = RemediationAction(
   776→                    finding_title=title,
   777→                    finding_category=category,
   778→                    severity=severity,
   779→                    rule_key=None,
   780→                    rule_description=f"Manual remediation needed for category '{category}'",
   781→                    fix_commands=[
   782→                        f"# Manual investigation required:",
   783→                        f"#   Finding : {title}",
   784→                        f"#   Category: {category}",
   785→                        f"#   Severity: {severity}",
   786→                        f"# Review the evidence and apply appropriate fixes.",
   787→                        f"# Re-scan after fixing to verify closure.",
   788→                    ],
   789→                    confidence=0.30,
   790→                    round_applied=round_num,
   791→                )
   792→            actions.append(action)
   793→        print(f"    {green(str(len(actions)))} fix(es) generated")
   794→        return actions
   795→
   796→    # -- GUARDIAN phase --------------------------------------------------------
   797→
   798→    def _run_guardian(
   799→        self,
   800→        actions: List[RemediationAction],
   801→        all_findings: List[Dict[str, Any]],
   802→        round_num: int,
   803→    ) -> Tuple[int, int, List[str]]:
   804→        """Rule-based verification of proposed fixes.
   805→
   806→        Returns (verified_fixed_count, still_vulnerable_count, remaining_titles).
   807→        """
   808→        print(f"  {yellow('>>  GUARDIAN')} Verifying fixes ...")
   809→        verified = 0
   810→        still_open = 0
   811→        remaining: List[str] = []
   812→
   813→        # Build a lookup of current finding titles
   814→        finding_titles = {f.get("title", "") for f in all_findings}
   815→
   816→        for action in actions:
   817→            title = action.finding_title
   818→            rule_key = action.rule_key
   819→
   820→            if rule_key and rule_key in REMEDIATION_RULES:
   821→                rule = REMEDIATION_RULES[rule_key]
   822→                verif_keys = rule.get("verification_keys", [])
   823→                confidence = rule.get("confidence", 0.5)
   824→
   825→                # Check severity and confidence for auto-verification
   826→                if confidence >= 0.90:
   827→                    action.verified = True
   828→                    action.verification_note = (
   829→                        f"High-confidence rule ({confidence:.0%}) -- "
   830→                        f"fix commands should resolve '{title}'."
   831→                    )
   832→                    verified += 1
   833→                    self._fixed_titles.add(title)
   834→                    self._failed_titles.discard(title)
   835→                elif confidence >= 0.80:
   836→                    # Medium-high confidence: verify if the finding is still present
   837→                    if title in finding_titles:
   838→                        # Still present after fix -- mark as still vulnerable
   839→                        action.verified = False
   840→                        action.verification_note = (
   841→                            f"Finding '{title}' still present after remediation attempt. "
   842→                            f"Commands target: {', '.join(verif_keys[:3])}."
   843→                        )
   844→                        still_open += 1
   845→                        remaining.append(title)
   846→                        self._failed_titles.add(title)
   847→                        # Feed back context for next round
   848→                        self._feedback_context.setdefault(title, []).append(
   849→                            f"Round {round_num}: Fix not effective. Try applying all commands."
   850→                        )
   851→                    else:
   852→                        action.verified = True
   853→                        action.verification_note = f"Finding '{title}' no longer detected."
   854→                        verified += 1
   855→                        self._fixed_titles.add(title)
   856→                        self._failed_titles.discard(title)
   857→                else:
   858→                    # Lower confidence -- always flag as needing verification
   859→                    action.verified = False
   860→                    action.verification_note = (
   861→                        f"Rule confidence {confidence:.0%} is below threshold. "
   862→                        f"Manual verification required for '{title}'."
   863→                    )
   864→                    still_open += 1
   865→                    remaining.append(title)
   866→                    self._feedback_context.setdefault(title, []).append(
   867→                        f"Round {round_num}: Low confidence rule. Escalate to manual review."
   868→                    )
   869→            else:
   870→                # No matching rule
   871→                action.verified = False
   872→                action.verification_note = (
   873→                    f"No verification rule for category '{action.finding_category}'. "
   874→                    f"Manual review required."
   875→                )
   876→                still_open += 1
   877→                remaining.append(title)
   878→
   879→        print(f"    {green(str(verified))} verified fixed  |  "
   880→              f"{red(str(still_open))} still vulnerable")
   881→        return verified, still_open, remaining
   882→
   883→    # -- Progress bar ---------------------------------------------------------
   884→
   885→    @staticmethod
   886→    def _progress_bar(fixed: int, total: int, width: int = 30) -> str:
   887→        if total == 0:
   888→            return green("#" * width) + " 100%"
   889→        ratio = fixed / total
   890→        filled = int(width * ratio)
   891→        bar = green("#" * filled) + dim("." * (width - filled))
   892→        return f"{bar} {ratio:.0%}"
   893→
   894→    # -- Round display --------------------------------------------------------
   895→
   896→    @staticmethod
   897→    def _print_round_banner(round_num: int, max_rounds: int) -> None:
   898→        print()
   899→        print(f"{white('=' * 60)}")
   900→        print(f"{bold(f'  ROUND {round_num} / {max_rounds}')}")
   901→        print(f"{white('=' * 60)}")
   902→
   903→    # -- Main loop ------------------------------------------------------------
   904→
   905→    def run(self) -> AdversarialResult:
   906→        """Execute the full adversarial loop and return an AdversarialResult."""
   907→        print()
   908→        print(f"{bold(cyan('  +==========================================+'))}")
   909→        print(f"{bold(cyan('  |   ADVERSARIAL  SELF-PLAY  REMEDIATION   |'))}")
   910→        print(f"{bold(cyan('  |   HACKER -> CODER -> GUARDIAN  (loop)  |'))}")
   911→        print(f"{bold(cyan('  +==========================================+'))}")
   912→        print(f"  Target: {white(self.target)}")
   913→        print(f"  Max rounds: {white(str(self.max_rounds))}")
   914→        print(f"  Mode: {white('LOCAL AUDIT' if self.is_local else 'REMOTE SCAN')}")
   915→
   916→        result = AdversarialResult(
   917→            target=self.target,
   918→            max_rounds=self.max_rounds,
   919→        )
   920→
   921→        initial_findings: List[Dict[str, Any]] = []
   922→        initial_score = 100
   923→        initial_grade = "A"
   924→
   925→        for rnd in range(1, self.max_rounds + 1):
   926→            t0 = time.monotonic()
   927→            self._print_round_banner(rnd, self.max_rounds)
   928→
   929→            # -- HACKER ---------------------------------------------------------
   930→            findings, score_before, grade_before = self._run_hacker()
   931→            if rnd == 1:
   932→                initial_findings = list(findings)
   933→                initial_score = score_before
   934→                initial_grade = grade_before
   935→
   936→            # Track new vs already-known findings
   937→            known_titles = self._fixed_titles | self._failed_titles
   938→            new_findings = [f for f in findings if f.get("title", "") not in known_titles]
   939→            hacker_new = len(new_findings)
   940→            hacker_total = len(findings)
   941→
   942→            # -- CODER ----------------------------------------------------------
   943→            actions = self._run_coder(findings, round_num=rnd)
   944→            self._all_actions.extend(actions)
   945→
   946→            # -- GUARDIAN -------------------------------------------------------
   947→            verified_fixed, still_vulnerable, remaining_titles = self._run_guardian(
   948→                actions, findings, round_num=rnd,
   949→            )
   950→            elapsed = time.monotonic() - t0
   951→
   952→            # Compute projected score after fixes
   953→            projected_score = min(100, 100 - sum(
   954→                f.get("points_deducted", 0)
   955→                for f in findings
   956→                if f.get("title", "") not in self._fixed_titles
   957→            ))
   958→            projected_grade = compute_grade(projected_score)
   959→
   960→            # Open issues: findings not yet fixed
   961→            open_titles = {
   962→                f.get("title", "") for f in findings
   963→            } - self._fixed_titles
   964→            total_open = len(open_titles)
   965→
   966→            progress_str = self._progress_bar(
   967→                len(self._fixed_titles),
   968→                len(initial_findings) if initial_findings else 1,
   969→            )
   970→
   971→            round_summary = RoundSummary(
   972→                round_num=rnd,
   973→                hacker_findings=hacker_total,
   974→                hacker_new=hacker_new,
   975→                coder_fixes=len(actions),
   976→                guardian_verified_fixed=verified_fixed,
   977→                guardian_still_vulnerable=still_vulnerable,
   978→                total_open=total_open,
   979→                score_before=score_before,
   980→                score_after=projected_score,
   981→                grade_before=grade_before,
   982→                grade_after=projected_grade,
   983→                elapsed_seconds=elapsed,
   984→                actions=actions,
   985→                remaining_titles=remaining_titles,
   986→            )
   987→            result.rounds.append(round_summary)
   988→
   989→            # -- Round summary output -------------------------------------------
   990→            print()
   991→            print(f"  {dim('--- Round Summary ------------------------------------')}")
   992→            print(f"  Findings scanned    : {red(str(hacker_total))}  (new: {red(str(hacker_new))})")
   993→            print(f"  Fixes generated     : {green(str(len(actions)))}")
   994→            print(f"  Verified fixed      : {green(str(verified_fixed))}")
   995→            print(f"  Still vulnerable    : {red(str(still_vulnerable))}")
   996→            print(f"  Open issues         : {yellow(str(total_open))}")
   997→            print(f"  Progress            : {progress_str}")
   998→            print(f"  Score               : {grade_before} ({score_before}) -> {projected_grade} ({projected_score})")
   999→            print(f"  Elapsed             : {dim(f'{elapsed:.1f}s')}")
  1000→
  1001→            # -- Show individual fixes ------------------------------------------
  1002→            if actions:
  1003→                print()
  1004→                print(f"  {dim('--- Fixes Applied -----------------------------------')}")
  1005→                for act in actions:
  1006→                    status_icon = green("[OK]") if act.verified else red("[!!]")
  1007→                    sev_color = {
  1008→                        "critical": BRIGHT_RED, "high": RED,
  1009→                        "medium": YELLOW, "low": DIM, "info": DIM,
  1010→                    }.get(act.severity, DIM)
  1011→                    print(f"    {status_icon} {_c(sev_color, act.severity.upper().ljust(8))} "
  1012→                          f"{act.finding_title[:60]}")
  1013→                    if act.rule_key:
  1014→                        print(f"      {dim(f'Rule: {act.rule_key} (confidence {act.confidence:.0%})')}")
  1015→
  1016→            # -- Check convergence ----------------------------------------------
  1017→            if total_open == 0:
  1018→                self._converged = True
  1019→                print()
  1020→                print(f"  {green(bold('  >> ALL FINDINGS REMEDIATED -- loop converged!'))}")
  1021→                break
  1022→
  1023→            if rnd < self.max_rounds and still_vulnerable == 0 and hacker_new == 0:
  1024→                # No new findings and no new fixes -- likely converged
  1025→                self._converged = True
  1026→                print()
  1027→                print(f"  {yellow(bold('  !! No new findings or fixes -- loop stalled. Stopping early.'))}")
  1028→                break
  1029→
  1030→        # -- Build final report ------------------------------------------------
  1031→        initial_title_set = {f.get("title", "") for f in initial_findings}
  1032→        result.total_findings_initial = len(initial_findings)
  1033→        result.total_findings_final = len(initial_title_set - self._fixed_titles)
  1034→        result.findings_fixed = len(self._fixed_titles & initial_title_set)
  1035→        result.findings_unfixed = result.total_findings_initial - result.findings_fixed
  1036→        result.final_score = result.rounds[-1].score_after if result.rounds else 100
  1037→        result.final_grade = compute_grade(result.final_score)
  1038→        result.initial_score = initial_score
  1039→        result.initial_grade = initial_grade
  1040→        result.converged = self._converged
  1041→        result.remediation_report = self._build_remediation_report(result)
  1042→
  1043→        # -- Final scorecard ---------------------------------------------------
  1044→        self._print_scorecard(result)
  1045→
  1046→        return result

  1050→    def _build_remediation_report(self, result: AdversarialResult) -> str:
  1051→        lines: List[str] = []
  1052→        lines.append("RECONPRO ADVERSARIAL REMEDIATION REPORT")
  1053→        lines.append("=" * 50)
  1054→        lines.append(f"Target          : {result.target}")
  1055→        lines.append(f"Initial findings: {result.total_findings_initial}")
  1056→        lines.append(f"Fixed           : {result.findings_fixed}")
  1057→        lines.append(f"Unfixed         : {result.findings_unfixed}")
  1058→        lines.append(f"Score           : {result.initial_grade} ({result.initial_score}) -> "
  1059→                      f"{result.final_grade} ({result.final_score})")
  1060→        lines.append(f"Converged       : {result.converged}")
  1061→        lines.append(f"Rounds played   : {len(result.rounds)}")
  1062→        lines.append("")
  1063→
  1064→        for rnd in result.rounds:
  1065→            lines.append(f"--- Round {rnd.round_num} ---")
  1066→            lines.append(f"  HACKER:   {rnd.hacker_findings} findings ({rnd.hacker_new} new)")
  1067→            lines.append(f"  CODER:    {rnd.coder_fixes} fixes generated")
  1068→            lines.append(f"  GUARDIAN: {rnd.guardian_verified_fixed} fixed, {rnd.guardian_still_vulnerable} still open")
  1069→            lines.append("")
  1070→
  1071→            for act in rnd.actions:
  1072→                status = "FIXED" if act.verified else "OPEN"
  1073→                lines.append(f"  [{status}] {act.finding_title}")
  1074→                lines.append(f"    Category: {act.finding_category}  |  Severity: {act.severity}")
  1075→                if act.rule_key:
  1076→                    lines.append(f"    Rule: {act.rule_key} (confidence {act.confidence:.0%})")
  1077→                lines.append(f"    Fix commands:")
  1078→                for cmd in act.fix_commands:
  1079→                    lines.append(f"      {cmd}")
  1080→                lines.append(f"    Note: {act.verification_note}")
  1081→                lines.append("")
  1082→
  1083→        # Unfixed findings detail
  1084→        unfixed_actions = [a for a in self._all_actions if not a.verified]
  1085→        if unfixed_actions:
  1086→            lines.append("=== REMAINING VULNERABILITIES ===")
  1087→            for act in unfixed_actions:
  1088→                lines.append(f"  - {act.finding_title} ({act.severity})")
  1089→                lines.append(f"    {act.verification_note}")
  1090→            lines.append("")
  1091→
  1092→        return "\n".join(lines)
