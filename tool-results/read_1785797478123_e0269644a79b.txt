   106→   106→class FindingDetailModal(ModalScreen):
   107→   107→    """Modal overlay showing full finding details."""
   108→   108→
   109→   109→    def __init__(self, finding: Dict[str, Any]) -> None:
   110→   110→        super().__init__()
   111→   111→        self.finding = finding
   112→   112→
   113→   113→    def compose(self) -> ComposeResult:
   114→   114→        f = self.finding
   115→   115→        sev = f.get("severity", "info").lower()
   116→   116→        sev_color = SEV_STYLES.get(sev, "white")
   117→   117→        title = f.get("title", "Unknown")
   118→   118→        category = f.get("category", "")
   119→   119→        module = f.get("module", "")
   120→   120→        pts = f.get("points_deducted", 0)
   121→   121→        evidence = f.get("evidence", "")
   122→   122→        remediation = f.get("remediation", "No remediation available.")
   123→   123→        description = f.get("description", "")
   124→   124→        asset = f.get("asset", "")
   125→   125→        related_cves = f.get("related_cves", [])
   126→   126→
   127→   127→        lines: List[str] = []
   128→   128→        lines.append(f"[{sev_color}][{sev.upper()}] {title}[/]")
   129→   129→        lines.append("")
   130→   130→        if asset:
   131→   131→            lines.append(f"[dim]Asset:[/] [cyan]{asset}[/]")
   132→   132→        if category:
   133→   133→            lines.append(f"[dim]Category:[/] {category}")
   134→   134→        if module:
   135→   135→            lines.append(f"[dim]Module:[/] {module}")
   136→   136→        lines.append(f"[dim]Points:[/] -{pts}")
   137→   137→        if related_cves:
   138→   138→            cve_ids = ", ".join(
   139→   139→                c.get("cve_id", "") for c in related_cves[:5] if c.get("cve_id")
   140→   140→            )
   141→   141→            if cve_ids:
   142→   142→                lines.append(f"[dim]CVEs:[/] [red]{cve_ids}[/]")
   143→   143→        lines.append("")
   144→   144→        if description:
   145→   145→            lines.append(f"[bold]Description:[/]")
   146→   146→            lines.append(f"[dim]{description}[/]")
   147→   147→            lines.append("")
   148→   148→        if evidence:
   149→   149→            lines.append(f"[bold]Evidence:[/]")
   150→   150→            lines.append(f"[dim]{evidence}[/]")
   151→   151→            lines.append("")
   152→   152→        lines.append(f"[bold]{GREEN}Remediation:[/]")
   153→   153→        lines.append(f"{GREEN}{remediation}[/]")
   154→   154→
   155→   155→        content = "\n".join(lines)
   156→   156→
   157→   157→        with Vertical(classes="modal-container"):
   158→   158→            yield Label(content, classes="modal-content")
   159→   159→            with Horizontal(classes="modal-actions"):
   160→   160→                yield Button("Close", variant="primary", id="close-modal")
   161→   161→
   162→   162→    def on_button_pressed(self, event: Button.Pressed) -> None:
   163→   163→        if event.button.id == "close-modal":
   164→   164→            self.dismiss()

   168→   168→# Boot Screen Widget
   169→   169→# ════════════════════════════════════════════════════════════════════════════════
   170→   170→
   171→   171→
   172→   172→class BootScreen(Widget):
   173→   173→    """Full-screen animated boot sequence."""
   174→   174→
   175→   175→    def __init__(self) -> None:
   176→   176→        super().__init__()
   177→   177→        self._step = 0
   178→   178→        self._spinner_idx = 0
   179→   179→        self._module_idx = 0
   180→   180→        self._module_names = list(MODULE_NAMES.values())
   181→   181→        self._done = False
   182→   182→        self._timer: Optional[Timer] = None
   183→   183→
   184→   184→    def compose(self) -> ComposeResult:
   185→   185→        with Vertical(classes="boot-container"):
   186→   186→            yield Static("", id="boot-line-1", classes="boot-title")
   187→   187→            yield Static("", id="boot-line-2", classes="boot-line")
   188→   188→            yield Static("", id="boot-line-3", classes="boot-line")
   189→   189→            yield Static("", id="boot-line-4", classes="boot-line")
   190→   190→
   191→   191→    def on_mount(self) -> None:
   192→   192→        self._timer = self.set_interval(0.04, self._tick)
   193→   193→
   194→   194→    def _tick(self) -> None:
   195→   195→        app = self.app
   196→   196→        if not isinstance(app, NexusApp):
   197→   197→            return
   198→   198→
   199→   199→        line1 = app.query_one("#boot-line-1", Static)
   200→   200→        line2 = app.query_one("#boot-line-2", Static)
   201→   201→        line3 = app.query_one("#boot-line-3", Static)
   202→   202→        line4 = app.query_one("#boot-line-4", Static)
   203→   203→
   204→   204→        if self._step == 0:
   205→   205→            # Letter-by-letter title
   206→   206→            title = "RECONPRO NEXUS"
   207→   207→            chars = title[: self._spinner_idx]
   208→   208→            self._spinner_idx += 1
   209→   209→            line1.update(f"[{CYAN} bold]{chars}[/]")
   210→   210→            if self._spinner_idx > len(title):
   211→   211→                self._spinner_idx = 0
   212→   212→                self._step = 1
   213→   213→                line1.update(f"[{CYAN} bold]{title}[/]  [{DIM_CYAN}]v{VERSION}[/]")
   214→   214→
   215→   215→        elif self._step == 1:
   216→   216→            # Spinner for "Initializing agent..."
   217→   217→            frame = SPINNER_FRAMES[self._spinner_idx % len(SPINNER_FRAMES)]
   218→   218→            self._spinner_idx += 1
   219→   219→            line2.update(f"  [{CYAN}]{frame}[/] [{DIM_CYAN}]Initializing agent...[/]")
   220→   220→            if self._spinner_idx > 30:
   221→   221→                self._spinner_idx = 0
   222→   222→                self._step = 2
   223→   223→                line2.update(f"  [{GREEN}]✓[/] [{DIM_CYAN}]Agent initialized.[/]")
   224→   224→
   225→   225→        elif self._step == 2:
   226→   226→            # Module names appearing one by one
   227→   227→            if self._module_idx < len(self._module_names):
   228→   228→                visible = self._module_names[: self._module_idx + 1]
   229→   229→                self._module_idx += 1
   230→   230→                modules_str = "  ".join(f"[{CYAN}]○[/] [{DIM_CYAN}]{m}[/]" for m in visible)
   231→   231→                remaining = len(self._module_names) - self._module_idx
   232→   232→                if remaining > 0:
   233→   233→                    modules_str += f"  [{DIM_CYAN}]+{remaining} more...[/]"
   234→   234→                frame = SPINNER_FRAMES[self._spinner_idx % len(SPINNER_FRAMES)]
   235→   235→                self._spinner_idx += 1
   236→   236→                line3.update(
   237→   237→                    f"  [{CYAN}]{frame}[/] [{DIM_CYAN}]Loading {len(self._module_names)} modules...[/]\n  {modules_str}"
   238→   238→                )
   239→   239→            else:
   240→   240→                all_mods = "  ".join(
   241→   241→                    f"[{GREEN}]●[/] [{DIM_CYAN}]{m}[/]" for m in self._module_names
   242→   242→                )
   243→   243→                line3.update(
   244→   244→                    f"  [{GREEN}]✓[/] [{DIM_CYAN}]All {len(self._module_names)} modules loaded.[/]\n  {all_mods}"
   245→   245→                )
   246→   246→                self._step = 3
   247→   247→                self._spinner_idx = 0
   248→   248→
   249→   249→        elif self._step == 3:
   250→   250→            # "Ready." with flash
   251→   251→            if self._spinner_idx == 0:
   252→   252→                line4.update(f"[{YELLOW} bold]  ▶  READY.[/]")
   253→   253→            elif self._spinner_idx == 6:
   254→   254→                line4.update(f"[{GREEN} bold]  ▶  READY.[/]")
   255→   255→            elif self._spinner_idx >= 10:
   256→   256→                self._done = True
   257→   257→                if self._timer:
   258→   258→                    self._timer.stop()
   259→   259→                self.app.call_from_thread(self.app._finish_boot)
   260→   260→            self._spinner_idx += 1
   261→   261→

   264→   264→# Module Status Cell Widget
   265→   265→# ════════════════════════════════════════════════════════════════════════════════
   266→   266→
   267→   267→
   268→   268→class ModuleCell(Static):
   269→   269→    """Single module status cell in the grid."""
   270→   270→
   271→   271→    def __init__(self, module_id: str, name: str) -> None:
   272→   272→        super().__init__()
   273→   273→        self.module_id = module_id
   274→   274→        self.module_name = name
   275→   275→        self.status = "idle"  # idle | scanning | done | error
   276→   276→        self.finding_count = 0
   277→   277→        self._spinner_idx = 0
   278→   278→        self._timer: Optional[Timer] = None
   279→   279→
   280→   280→    def on_mount(self) -> None:
   281→   281→        self._update_display()
   282→   282→        self._timer = self.set_interval(0.1, self._spin)
   283→   283→
   284→   284→    def _spin(self) -> None:
   285→   285→        if self.status == "scanning":
   286→   286→            self._spinner_idx = (self._spinner_idx + 1) % len(SPINNER_FRAMES)
   287→   287→            self._update_display()
   288→   288→
   289→   289→    def _update_display(self) -> None:
   290→   290→        name_short = self.module_name[:14]
   291→   291→        if self.status == "idle":
   292→   292→            icon = f"[{DIM_CYAN}]○[/]"
   293→   293→            status_text = f"[{DIM_CYAN}]idle[/]"
   294→   294→        elif self.status == "scanning":
   295→   295→            frame = SPINNER_FRAMES[self._spinner_idx]
   296→   296→            icon = f"[{CYAN}]{frame}[/]"
   297→   297→            status_text = f"[{CYAN}]scanning[/]"
   298→   298→        elif self.status == "done":
   299→   299→            icon = f"[{GREEN}]✓[/]"
   300→   300→            status_text = f"[{GREEN}]done ({self.finding_count})[/]"
   301→   301→        elif self.status == "error":
   302→   302→            icon = f"[{RED}]✗[/]"
   303→   303→            status_text = f"[{RED}]error[/]"
   304→   304→        else:
   305→   305→            icon = "○"
   306→   306→            status_text = "idle"
   307→   307→
   308→   308→        self.update(
   309→   309→            f"{icon} [{CYAN} bold]{name_short}[/]\n  {status_text}"
   310→   310→        )
   311→   311→
   312→   312→    def set_status(self, status: str, count: int = 0) -> None:
   313→   313→        self.status = status
   314→   314→        self.finding_count = count
   315→   315→        self._update_display()
   316→   316→
   317→   317→    def on_unmount(self) -> None:
   318→   318→        if self._timer:
   319→   319→            self._timer.stop()
   320→   320→

   323→   323→# Pulsing Status Dot Widget
   324→   324→# ════════════════════════════════════════════════════════════════════════════════
   325→   325→
   326→   326→
   327→   327→class StatusDot(Static):
   328→   328→    """Pulsing dot indicator for scan status."""
   329→   329→
   330→   330→    scanning: reactive[bool] = reactive(False)
   331→   331→
   332→   332→    def __init__(self) -> None:
   333→   333→        super().__init__("")
   334→   334→        self._phase = 0
   335→   335→
   336→   336→    def on_mount(self) -> None:
   337→   337→        self.set_interval(0.3, self._pulse)
   338→   338→
   339→   339→    def watch_scanning(self, old: bool, new: bool) -> None:
   340→   340→        # Update visual state based on scanning status
   341→   341→        if new:
   342→   342→            self.update(f"[{self._get_pulse_color()}]●[/]")
   343→   343→        else:
   344→   344→            self.update(f"[{GREEN}]●[/]")
   345→   345→
   346→   346→    def _get_pulse_color(self) -> str:
   347→   347→        return ["#003322", "#006644", CYAN, "#006644"][self._phase] if self.scanning else GREEN
   348→   348→
   349→   349→    def _pulse(self) -> None:
   350→   350→        self._phase = (self._phase + 1) % 4
   351→   351→        if self.scanning:
   352→   352→            brightness = self._get_pulse_color()
   353→   353→            self.update(f"[{brightness}]●[/]")
   354→   354→        else:
   355→   355→            self.update(f"[{GREEN}]●[/]")

   906→   906→    def on_key(self, event: Key) -> None:
   907→   907→        """Handle key events — boot skip + completer + history + vim-nav."""
   908→   908→        # Skip boot on ANY key press during boot sequence
   909→   909→        if not self._boot_done and not self._boot_skipped:
   910→   910→            self._boot_skipped = True
   911→   911→            try:
   912→   912→                boot = self.query_one("#boot-screen", BootScreen)
   913→   913→                if boot._timer:
   914→   914→                    boot._timer.stop()
   915→   915→            except NoMatches:
   916→   916→                pass
   917→   917→            self._finish_boot()
   918→   918→            return
   919→   919→
   920→   920→        # ── Number keys: quick jump to panels (0=input, 1=chat, 2=findings, 3=modules) ──
   921→   921→        if event.key in ("0", "1", "2", "3") and not self._input_focused():
   922→   922→            jump_map = {"0": "#command-input", "1": "#chat-log", "2": "#findings-feed", "3": "#module-grid"}
   923→   923→            # Skip panel jump to collapsed panels
   924→   924→            target_sel = jump_map[event.key]
   925→   925→            if event.key == "1" and self._left_collapsed:
   926→   926→                target_sel = "#findings-feed"
   927→   927→            if event.key == "3" and self._modules_collapsed:
   928→   928→                target_sel = "#findings-feed"
   929→   929→            self._focus_widget(target_sel)
   930→   930→            event.stop()
   931→   931→            return
   932→   932→
   933→   933→        # ── Phase D: [ and ] to toggle panel collapse ──
   934→   934→        if event.key == "[" and not self._input_focused():
   935→   935→            self._toggle_left_panel()
   936→   936→            event.stop()
   937→   937→            return
   938→   938→        if event.key == "]" and not self._input_focused():
   939→   939→            self._toggle_right_panel()
   940→   940→            event.stop()
   941→   941→            return
   942→   942→        if event.key == "=" and not self._input_focused():
   943→   943→            self._toggle_modules()
   944→   944→            event.stop()
   945→   945→            return
   946→   946→        # ── Phase D: | resets split to auto, \\ resets to 50:50 ──
   947→   947→        if event.key == "|" and not self._input_focused():
   948→   948→            self._user_manually_resized = False
   949→   949→            self._apply_responsive_layout(self.size.width, self.size.height)
   950→   950→            event.stop()
   951→   951→            return
   952→   952→        if event.key == "\\" and not self._input_focused():
   953→   953→            self._user_manually_resized = True
   954→   954→            self._split_ratio = 50
   955→   955→            self._apply_split_ratio()
   956→   956→            event.stop()
   957→   957→            return

  1167→  1167→                parts[-1] = accepted

  1169→  1169→            else:
