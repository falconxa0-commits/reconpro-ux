    53→class ToolResult:
    54→    """Result of a tool execution."""
    55→
    56→    def __init__(self, output: str, success: bool = True, data: Any = None):
    57→        self.output = output
    58→        self.success = success
    59→        self.data = data
    60→
    61→    def to_dict(self) -> Dict:
    62→        return {"output": self.output, "success": self.success, "data": self.data}

    64→
    65→class Tool:
    66→    """A callable tool the agent can use."""
    67→
    68→    def __init__(
    69→        self,
    70→        name: str,
    71→        description: str,
    72→        parameters: Dict[str, Any],
    73→        execute: Callable[..., ToolResult],
    74→        category: str = "scan",
    75→        dangerous: bool = False,
    76→    ):
    77→        self.name = name
    78→        self.description = description
    79→        self.parameters = parameters
    80→        self.execute = execute
    81→        self.category = category
    82→        self.dangerous = dangerous
    83→
    84→    def to_openai_tool(self) -> Dict:
    85→        return {
    86→            "type": "function",
    87→            "function": {
    88→                "name": self.name,
    89→                "description": self.description,
    90→                "parameters": {
    91→                    "type": "object",
    92→                    "properties": self.parameters,
    93→                    "required": list(self.parameters.keys()),
    94→                },
    95→            },
    96→        }
    97→
    98→    def to_anthropic_tool(self) -> Dict:
    99→        return {
   100→            "name": self.name,
   101→            "description": self.description,
   102→            "input_schema": {
   103→                "type": "object",
   104→                "properties": self.parameters,
   105→                "required": list(self.parameters.keys()),
   106→            },
   107→        }

   109→
   110→class ToolRegistry:
   111→    """Registry of all available tools."""
   112→
   113→    def __init__(self):
   114→        self.tools: Dict[str, Tool] = {}
   115→
   116→    def register(self, tool: Tool) -> None:
   117→        self.tools[tool.name] = tool
   118→
   119→    def get(self, name: str) -> Optional[Tool]:
   120→        return self.tools.get(name)
   121→
   122→    def execute(self, name: str, **kwargs) -> ToolResult:
   123→        tool = self.tools.get(name)
   124→        if not tool:
   125→            return ToolResult(f"Tool '{name}' not found.", success=False)
   126→        try:
   127→            return tool.execute(**kwargs)
   128→        except Exception as e:
   129→            return ToolResult(f"Tool error: {e}", success=False)
   130→
   131→    def list_tools(self) -> List[Dict]:
   132→        return [
   133→            {"name": t.name, "description": t.description, "category": t.category}
   134→            for t in self.tools.values()
   135→        ]
   136→
   137→    def to_openai_tools(self) -> List[Dict]:
   138→        return [t.to_openai_tool() for t in self.tools.values()]
   139→
   140→    def to_anthropic_tools(self) -> List[Dict]:
   141→        return [t.to_anthropic_tool() for t in self.tools.values()]

   148→
   149→class AgentMemory:
   150→    """Persistent memory for the Nexus agent across sessions."""
   151→
   152→    def __init__(self):
   153→        MEMORY_DIR.mkdir(parents=True, exist_ok=True)
   154→        self._data = self._load()
   155→
   156→    def _load(self) -> Dict:
   157→        if CONTEXT_FILE.exists():
   158→            try:
   159→                return json.loads(CONTEXT_FILE.read_text())
   160→            except Exception:
   161→                pass
   162→        return {
   163→            "sessions": [],
   164→            "target_contexts": {},
   165→            "learned_patterns": [],
   166→            "created": datetime.now(timezone.utc).isoformat(),
   167→        }
   168→
   169→    def _save(self) -> None:
   170→        # Keep only the last 50 sessions
   171→        if len(self._data.get("sessions", [])) > 50:
   172→            self._data["sessions"] = self._data["sessions"][-50:]
   173→        try:
   174→            CONTEXT_FILE.write_text(json.dumps(self._data, indent=2, default=str))
   175→        except OSError:
   176→            pass
   177→
   178→    def save_session(
   179→        self,
   180→        goal: str,
   181→        target: str,
   182→        plan: List[Dict],
   183→        tool_results: List[Dict],
   184→        summary: str,
   185→    ) -> None:
   186→        session = {
   187→            "goal": goal,
   188→            "target": target,
   189→            "plan": plan,
   190→            "tool_results": tool_results,
   191→            "summary": summary,
   192→            "timestamp": datetime.now(timezone.utc).isoformat(),
   193→        }
   194→        self._data.setdefault("sessions", []).append(session)
   195→        # Update target context
   196→        self._data.setdefault("target_contexts", {})[target] = {
   197→            "last_score": self._extract_score(tool_results),
   198→            "last_grade": self._extract_grade(tool_results),
   199→            "last_run": session["timestamp"],
   200→            "finding_count": self._extract_finding_count(tool_results),
   201→            "known_issues": self._extract_issues(tool_results),
   202→        }
   203→        self._save()
   204→
   205→    def get_target_context(self, target: str) -> Optional[Dict]:
   206→        return self._data.get("target_contexts", {}).get(target)
   207→
   208→    def get_recent_sessions(self, limit: int = 5) -> List[Dict]:
   209→        return self._data.get("sessions", [])[-limit:]
   210→
   211→    @staticmethod
   212→    def _extract_score(results: List[Dict]) -> Optional[int]:
   213→        for r in results:
   214→            d = r.get("data")
   215→            if isinstance(d, dict) and "total_score" in d:
   216→                return d["total_score"]
   217→        return None
   218→
   219→    @staticmethod
   220→    def _extract_grade(results: List[Dict]) -> Optional[str]:
   221→        for r in results:
   222→            d = r.get("data")
   223→            if isinstance(d, dict) and "grade" in d:
   224→                return d["grade"]
   225→        return None
   226→
   227→    @staticmethod
   228→    def _extract_finding_count(results: List[Dict]) -> int:
   229→        total = 0
   230→        for r in results:
   231→            d = r.get("data")
   232→            if isinstance(d, dict) and "findings" in d:
   233→                total += len(d["findings"])
   234→        return total
   235→
   236→    @staticmethod
   237→    def _extract_issues(results: List[Dict]) -> List[str]:
   238→        issues = []
   239→        for r in results:
   240→            d = r.get("data")
   241→            if isinstance(d, dict) and "findings" in d:
   242→                for f in d["findings"]:
   243→                    if isinstance(f, dict) and f.get("severity") in ("critical", "high"):
   244→                        issues.append(f.get("title", ""))
   245→        return issues[:20]

   430→    """Build or query the security knowledge graph."""
   431→    from .knowledge_graph import SecurityKnowledgeGraph
   432→    if not target:
   433→        return ToolResult("No target specified.", success=False)
   434→    kg = SecurityKnowledgeGraph()
   435→    if action == "build":
   436→        from .history import get_latest
   437→        latest = get_latest(target=target)
   438→        if not latest:
   439→            latest = get_latest()
   440→        if latest:
   441→            kg.add_scan_result(latest)
   442→        n_count = len(list(kg._g.nodes(data=False)))
   443→        e_count = len(list(kg._g.edges(data=False)))
   444→        return ToolResult(
   445→            f"Knowledge graph built: {n_count} nodes, {e_count} edges for {target}.",
   446→            success=True,
   447→            data={"node_count": n_count, "edge_count": e_count, "target": target},
   448→        )
   449→    elif action == "attack_chains":
   450→        chains = kg.find_attack_chains(max_depth=4)
   451→        return ToolResult(
   452→            f"Found {len(chains)} attack chain(s).", success=True,
   453→            data={"attack_chains": chains},
   454→        )
   455→    else:
   456→        return ToolResult(f"Unknown action: {action}. Use 'build' or 'attack_chains'.", success=False)

   563→        return ToolResult("No command specified.", success=False)
   564→    dangerous_patterns = ["rm -rf /", "mkfs", "dd if=", ":(){ :|:& };:", "chmod -R 777 /"]
   565→    for dp in dangerous_patterns:
   566→        if dp in command:
   567→            return ToolResult(f"Blocked dangerous command pattern: {dp}", success=False)
   568→    try:
   569→        result = subprocess.run(
   570→            command, shell=True, capture_output=True, text=True, timeout=30,
   571→        )
   572→        output = (result.stdout + result.stderr)[:2000]
   573→        return ToolResult(
   574→            f"Exit code: {result.returncode}\n{output}",
   575→            success=result.returncode == 0,
   576→        )
   577→    except subprocess.TimeoutExpired:
   578→        return ToolResult("Command timed out (30s).", success=False)
   579→    except Exception as e:
   580→        return ToolResult(f"Command error: {e}", success=False)

   902→class CVSSScorer:
   903→    """CVSS-inspired risk scoring for security findings.
   904→
   905→    Computes a base score (0.0–10.0) using the CVSS v3.1 formula
   906→    simplified for rule-based use:
   907→      - Attack Vector (AV): Network=0.85, Adjacent=0.62, Local=0.55, Physical=0.20
   908→      - Attack Complexity (AC): Low=0.77, High=0.44
   909→      - Privileges Required (PR): None=0.85, Low=0.62, High=0.27
   910→      - User Interaction (UI): None=0.85, Required=0.62
   911→      - Impact (C/I/A): None=0.00, Low=0.22, High=0.56
   912→
   913→    BaseScore = roundup(min(Impact + Exploitability, 10))
   914→    Impact = 1 - [(1-C)*(1-I)*(1-A)]
   915→    Exploitability = 8.22 * AV * AC * PR * UI
   916→    """
   917→
   918→    # Attack Vector metrics
   919→    _AV = {"N": 0.85, "A": 0.62, "L": 0.55, "P": 0.20}
   920→    # Attack Complexity metrics
   921→    _AC = {"L": 0.77, "H": 0.44}
   922→    # Privileges Required metrics
   923→    _PR = {"N": 0.85, "L": 0.62, "H": 0.27}
   924→    # User Interaction metrics
   925→    _UI = {"N": 0.85, "R": 0.62}

   930→    # against finding titles/descriptions.

   955→         "av": "N", "ac": "L", "pr": "N", "ui": "N", "c": "H", "i": "L", "a": "N"},

   957→        {"patterns": ["open port", "port ", "service exposed", "database exposed",
   958→                        "redis", "mongodb", "mysql", "ssh exposed"],
   959→         "av": "N", "ac": "L", "pr": "N", "ui": "N", "c": "H", "i": "H", "a": "H"},

   985→        {"patterns": ["cloud", "aws", "azure", "gcp", "s3 bucket", "iam", "oauth",
   986→                        "identity", "federation"],
   987→         "av": "N", "ac": "L", "pr": "N", "ui": "N", "c": "H", "i": "H", "a": "H"},
   988→        # Default fallback
   989→        {"patterns": [],
   990→         "av": "N", "ac": "L", "pr": "N", "ui": "N", "c": "L", "i": "L", "a": "N"},
   991→    ]
   992→
   993→    @classmethod
   994→    def score_finding(cls, finding: Dict) -> Dict[str, Any]:
   995→        """Compute CVSS-like score for a single finding.
   996→
   997→        Returns a dict with: base_score, severity, av, ac, pr, ui, c, i, a, vector_string.
   998→        """
   999→        title = (finding.get("title") or "").lower()
  1000→        desc = (finding.get("description") or "").lower()
  1001→        combined = f"{title} {desc}"
  1002→        original_sev = (finding.get("severity") or "").lower()
  1003→
  1004→        # Find matching profile
  1005→        profile = cls._CATEGORY_PROFILES[-1]  # default
  1006→        for prof in cls._CATEGORY_PROFILES:
  1007→            if prof["patterns"] and any(p in combined for p in prof["patterns"]):
  1008→                profile = prof
  1009→                break
  1010→
  1011→        # Override metrics based on original severity if it's explicitly critical
  1012→        av = cls._AV[profile["av"]]
  1013→        ac = cls._AC[profile["ac"]]
  1014→        pr = cls._PR[profile["pr"]]
  1015→        ui = cls._UI[profile["ui"]]
  1016→        c = cls._CIA[profile["c"]]
  1017→        i = cls._CIA[profile["i"]]
  1018→        a = cls._CIA[profile["a"]]
  1019→
  1020→        # If the original severity is critical, boost impact to High
  1021→        if original_sev == "critical":
  1022→            c = max(c, cls._CIA["H"])
  1023→            i = max(i, cls._CIA["H"])
  1024→            a = max(a, cls._CIA["H"])
  1025→        elif original_sev == "high":
  1026→            c = max(c, cls._CIA["L"])
  1027→            i = max(i, cls._CIA["L"])
  1028→            a = max(a, cls._CIA["L"])
  1029→
  1030→        # CVSS base score calculation
  1031→        iss = 1.0 - ((1.0 - c) * (1.0 - i) * (1.0 - a))
  1032→        impact = 6.42 * iss
  1033→        exploitability = 8.22 * av * ac * pr * ui
  1034→
  1035→        if impact <= 0:
  1036→            base_score = 0.0
  1037→        else:
  1038→            base_score = min(impact + exploitability, 10.0)
  1039→
  1040→        # Round up to nearest 0.1
  1041→        base_score = math.ceil(base_score * 10) / 10
  1042→
  1043→        # Derive severity from score
  1044→        if base_score >= 9.0:
  1045→            derived_sev = "critical"
  1046→        elif base_score >= 7.0:
  1047→            derived_sev = "high"
  1048→        elif base_score >= 4.0:
  1049→            derived_sev = "medium"
  1050→        elif base_score > 0.0:
  1051→            derived_sev = "low"
  1052→        else:
  1053→            derived_sev = "info"
  1054→
  1055→        # Use the more severe of original vs derived
  1056→        severity_order = {"critical": 0, "high": 1, "medium": 2, "low": 3, "info": 4}
  1057→        final_sev = original_sev if severity_order.get(original_sev, 99) <= severity_order.get(derived_sev, 99) else derived_sev
  1058→
  1059→        vector = f"CVSS:3.1/AV:{profile['av']}/AC:{profile['ac']}/PR:{profile['pr']}/UI:{profile['ui']}/C:{cls._cia_label(c)}/I:{cls._cia_label(i)}/A:{cls._cia_label(a)}"
  1060→
  1061→        return {
  1062→            "base_score": base_score,
  1063→            "severity": final_sev,
  1064→            "cvss_vector": vector,
  1065→            "metrics": {
  1066→                "av": profile["av"], "ac": profile["ac"], "pr": profile["pr"],
  1067→                "ui": profile["ui"], "c": cls._cia_label(c), "i": cls._cia_label(i),
  1068→                "a": cls._cia_label(a),
  1069→            },
  1070→        }
  1071→
  1072→    @classmethod
  1073→    def score_findings(cls, findings: List[Dict]) -> List[Dict]:
  1074→        """Score a list of findings, returning enriched list."""
  1075→        enriched = []
  1076→        for f in findings:
  1077→            scored = dict(f)
  1078→            cvss = cls.score_finding(f)
  1079→            scored["cvss_score"] = cvss["base_score"]
  1080→            scored["cvss_severity"] = cvss["severity"]
  1081→            scored["cvss_vector"] = cvss["cvss_vector"]
  1082→            scored["cvss_metrics"] = cvss["metrics"]
  1083→            enriched.append(scored)
  1084→        return enriched
  1085→
  1086→    @staticmethod
  1087→    def _cia_label(value: float) -> str:
  1088→        if value >= 0.56:
  1089→            return "H"
  1090→        elif value >= 0.22:
  1091→            return "L"
  1092→        return "N"

  1403→    @classmethod
  1404→    def correlate(cls, findings: List[Dict]) -> List[Dict]:
  1405→        """Run all correlation rules against findings and return compound findings.
  1406→
  1407→        Each finding is checked as a lowercase string of title + description.
  1408→        A rule fires when ALL trigger groups have at least one pattern matched
  1409→        across the full finding set.
  1410→        """
  1411→        if not findings:
  1412→            return []
  1413→
  1414→        # Build a flat list of all finding text for matching
  1415→        finding_texts: List[str] = []
  1416→        for f in findings:
  1417→            title = (f.get("title") or "").lower()
  1418→            desc = (f.get("description") or "").lower()
  1419→            finding_texts.append(f"{title} {desc}")
  1420→
  1421→        compound_findings: List[Dict] = []
  1422→
  1423→        for rule in cls.CORRELATION_RULES:
  1424→            triggers = rule["triggers"]
  1425→            rule_fired = True
  1426→
  1427→            for trigger_group in triggers:
  1428→                # Each trigger group: at least one pattern must match across all findings
  1429→                group_matched = False
  1430→                for pattern in trigger_group:
  1431→                    for ft in finding_texts:
  1432→                        if pattern in ft:
  1433→                            group_matched = True
  1434→                            break
  1435→                    if group_matched:
  1436→                        break
  1437→                if not group_matched:
  1438→                    rule_fired = False
  1439→                    break
  1440→
  1441→            if rule_fired:
  1442→                compound = {
  1443→                    "title": rule["compound_title"],
  1444→                    "severity": rule["compound_severity"],
  1445→                    "description": rule["compound_description"],
  1446→                    "category": "compound",
  1447→                    "correlation_id": rule["id"],
  1448→                    "is_compound": True,
  1449→                }
  1450→                # Compute CVSS score for the compound finding
  1451→                if "cvss_override" in rule:
  1452→                    override = rule["cvss_override"]
  1453→                    av = CVSSScorer._AV[override["av"]]
  1454→                    ac = CVSSScorer._AC[override["ac"]]
  1455→                    pr = CVSSScorer._PR[override["pr"]]
  1456→                    ui = CVSSScorer._UI[override["ui"]]
  1457→                    c_val = CVSSScorer._CIA[override["c"]]
  1458→                    i_val = CVSSScorer._CIA[override["i"]]
  1459→                    a_val = CVSSScorer._CIA[override["a"]]
  1460→                    iss = 1.0 - ((1.0 - c_val) * (1.0 - i_val) * (1.0 - a_val))
  1461→                    impact = 6.42 * iss
  1462→                    exploitability = 8.22 * av * ac * pr * ui
  1463→                    base_score = math.ceil(min(impact + exploitability, 10.0) * 10) / 10
  1464→                    compound["cvss_score"] = base_score
  1465→                    compound["cvss_vector"] = (
  1466→                        f"CVSS:3.1/AV:{override['av']}/AC:{override['ac']}/"
  1467→                        f"PR:{override['pr']}/UI:{override['ui']}/"
  1468→                        f"C:{override['c']}/I:{override['i']}/A:{override['a']}"
  1469→                    )

  1480→
  1481→class AttackSurfaceMapper:
  1482→    """Computes an attack surface score (0–100, higher = more exposed).
  1483→
  1484→    Factors:
  1485→      1. Open ports (0–25 pts): 0 ports=0, 1–2=5, 3–5=10, 6–10=15, 11+=25
  1486→      2. Exposed paths/endpoints (0–20 pts): 0=0, 1–3=5, 4–7=10, 8–12=15, 13+=20
  1487→      3. Missing security headers (0–20 pts): 0=0, 1=3, 2=6, 3=10, 4=14, 5+=20
  1488→      4. Tech stack complexity (0–15 pts): 1–2 techs=2, 3–4=5, 5–6=8, 7–8=11, 9+=15
  1489→      5. TLS grade (0–20 pts): A=0, B=5, C=10, D=15, F=20
  1490→    """
  1491→
  1492→    @classmethod
  1493→    def compute(cls, findings: List[Dict], scan_data: Optional[Dict] = None) -> Dict[str, Any]:
  1494→        """Compute attack surface score and return breakdown."""
  1495→        findings_lower = []
  1496→        for f in findings:
  1497→            title = (f.get("title") or "").lower()
  1498→            desc = (f.get("description") or "").lower()
  1499→            findings_lower.append(f"{title} {desc}")
  1500→        all_text = " ".join(findings_lower)
  1501→
  1502→        # 1. Open ports
  1503→        port_score = cls._score_ports(findings_lower, scan_data)
  1504→        # 2. Exposed paths
  1505→        path_score = cls._score_exposed_paths(findings_lower)
  1506→        # 3. Missing headers
  1507→        header_score = cls._score_missing_headers(findings_lower)
  1508→        # 4. Tech stack complexity
  1509→        tech_score = cls._score_tech_stack(all_text, scan_data)
  1510→        # 5. TLS grade
  1511→        tls_score = cls._score_tls_grade(findings_lower, scan_data)
  1512→
  1513→        total = min(port_score + path_score + header_score + tech_score + tls_score, 100)
  1514→
  1515→        # Risk rating
  1516→        if total >= 80:
  1517→            rating = "CRITICAL"
  1518→        elif total >= 60:
  1519→            rating = "HIGH"
  1520→        elif total >= 40:
  1521→            rating = "MODERATE"
  1522→        elif total >= 20:
  1523→            rating = "LOW"
  1524→        else:
  1525→            rating = "MINIMAL"
  1526→
  1527→        return {
  1528→            "attack_surface_score": total,
  1529→            "risk_rating": rating,
  1530→            "breakdown": {
  1531→                "open_ports": {"score": port_score, "max": 25},
  1532→                "exposed_paths": {"score": path_score, "max": 20},
  1533→                "missing_headers": {"score": header_score, "max": 20},
  1534→                "tech_complexity": {"score": tech_score, "max": 15},
  1535→                "tls_grade": {"score": tls_score, "max": 20},
  1536→            },
  1537→        }

  1679→
  1680→class RiskMatrixVisualizer:
  1681→    """Generates a text-based risk matrix (Likelihood vs Impact) using
  1682→    Unicode box-drawing characters.
  1683→
  1684→    Matrix layout (5x5):
  1685→
  1686→        Impact →
  1687→        Negligible  Low    Medium   High    Critical

  1692→    l   │         │       │         │        │          │
  1693→    i   │         │       │         │        │          │
  1694→    h   ├─────────┼───────┼─────────┼────────┼──────────┤
  1695→    o   │         │       │    ●    │   ●    │    ●     │
  1696→    o   │         │       │         │        │          │
  1697→    d   ├─────────┼───────┼─────────┼────────┼──────────┤
  1698→        │         │       │         │  ●     │    ●     │
  1699→    ↑   │         │       │         │        │          │
  1700→        ├─────────┼───────┼─────────┼────────┼──────────┤
  1701→        │         │       │         │        │          │
  1702→        │         │       │         │        │          │
  1703→        └─────────┴───────┴─────────┴────────┴──────────┘
  1704→    """
  1705→
  1706→    # Likelihood and Impact bands from CVSS scores
  1707→    _LIKELIHOOD_BANDS = [
  1708→        ("Very High", 9.0, 10.1),
  1709→        ("High", 7.0, 9.0),
  1710→        ("Medium", 4.0, 7.0),
  1711→        ("Low", 1.0, 4.0),
  1712→        ("Very Low", 0.0, 1.0),
  1713→    ]
  1714→
  1715→    _IMPACT_BANDS = [
  1716→        ("Critical", 9.0, 10.1),
  1717→        ("High", 7.0, 9.0),
  1718→        ("Medium", 4.0, 7.0),
  1719→        ("Low", 1.0, 4.0),
  1720→        ("Negligible", 0.0, 1.0),
  1721→    ]
  1722→
  1723→    # Color coding for cell density
  1724→    _DENSITY_SYMBOLS = ["  ", " ·", " •", " ●", " ██"]
  1725→
  1726→    @classmethod
  1727→    def render(cls, findings: List[Dict], width: int = 70) -> str:
  1728→        """Render the risk matrix as a text string with Unicode box-drawing.
  1729→
  1730→        Args:
  1731→            findings: List of finding dicts with 'cvss_score' (or 'base_score') keys.
  1732→            width: Total width of the matrix output.
  1733→
  1734→        Returns:
  1735→            Multi-line string with the matrix.
  1736→        """
  1737→        # Build a 5x5 grid of finding counts
  1738→        # Grid[y][x] where y=0 is top (Very High likelihood), x=0 is left (Negligible impact)
  1739→        grid: List[List[int]] = [[0] * 5 for _ in range(5)]
  1740→        grid_labels: List[List[List[str]]] = [[[] for _ in range(5)] for _ in range(5)]
  1741→
  1742→        for f in findings:
  1743→            score = f.get("cvss_score") or f.get("base_score") or 0.0
  1744→            if not isinstance(score, (int, float)) or score <= 0:
  1745→                continue
  1746→            title = (f.get("title") or "")[:12]
  1747→
  1748→            # Determine likelihood (y-axis) and impact (x-axis) from CVSS score
  1749→            # For simplicity, we use the CVSS score to derive both:
  1750→            # - Impact uses the CIA-derived portion
  1751→            # - Likelihood uses the exploitability portion
  1752→            # Since we may not have the full breakdown, we approximate:
  1753→            impact_idx = cls._score_to_band_index(score)
  1754→            likelihood_idx = cls._score_to_likelihood(score, f)
  1755→
  1756→            if 0 <= likelihood_idx < 5 and 0 <= impact_idx < 5:
  1757→                grid[likelihood_idx][impact_idx] += 1
  1758→                if len(grid_labels[likelihood_idx][impact_idx]) < 3:
  1759→                    grid_labels[likelihood_idx][impact_idx].append(title)
  1760→
  1761→        # Build the matrix string
  1762→        col_width = 11
  1763→        label_width = 12
  1764→        lines: List[str] = []
  1765→
  1766→        # Header row
  1767→        header = f"{'Likelihood / Impact':<{label_width + 2}}"
  1768→        impact_names = [b[0] for b in cls._IMPACT_BANDS]
  1769→        for name in impact_names:
  1770→            header += f"{name[:col_width]:>{col_width}}"
  1771→        lines.append(header)
  1772→
  1773→        # Top border
  1774→        lines.append(f"{'':<{label_width + 2}}┌" + "─" * col_width + ("┬" + "─" * col_width) * 4 + "┐")
  1775→
  1776→        likelihood_names = [b[0] for b in cls._LIKELIHOOD_BANDS]
  1777→        for y in range(5):
  1778→            # Cell content lines (2 lines per row for readability)
  1779→            label = likelihood_names[y]
  1780→            # Line 1: density symbol
  1781→            row1 = f"{label:>{label_width}}  │"
  1782→            row2 = f"{'':>{label_width}}  │"
  1783→            for x in range(5):
  1784→                count = grid[y][x]
  1785→                sym = cls._DENSITY_SYMBOLS[min(count, 4)]
  1786→                row1 += f"{sym:^{col_width}}│"
  1787→                labels = grid_labels[y][x]
  1788→                if labels:
  1789→                    row2 += f"{labels[0][:col_width]:^{col_width}}│"
  1790→                else:
  1791→                    row2 += f"{'':^{col_width}}│"
  1792→
  1793→            lines.append(row1)
  1794→            lines.append(row2)
  1795→
  1796→            # Separator or bottom border
  1797→            if y < 4:
  1798→                lines.append(f"{'':<{label_width + 2}}├" + "─" * col_width + ("┼" + "─" * col_width) * 4 + "┤")
  1799→            else:
  1800→                lines.append(f"{'':<{label_width + 2}}└" + "─" * col_width + ("┴" + "─" * col_width) * 4 + "┘")
  1801→
  1802→        # Summary line
  1803→        total = sum(grid[y][x] for y in range(5) for x in range(5))
  1804→        critical_zone = sum(grid[y][x] for y in range(2) for x in range(3, 5))
  1805→        lines.append("")
  1806→        lines.append(f"  Total findings plotted: {total}")
  1807→        lines.append(f"  Critical zone (High/Very High × High/Critical): {critical_zone}")
  1808→
  1809→        return "\n".join(lines)

  1811→    @classmethod
  1812→    def _score_to_band_index(cls, score: float) -> int:
  1813→        """Map score to impact band index (0=Negligible, 4=Critical)."""
  1814→        for i, (name, lo, hi) in enumerate(cls._IMPACT_BANDS):
  1815→            if lo <= score < hi:
  1816→                return 4 - i  # Reverse so Critical is rightmost
  1817→        return 0
  1818→
  1819→    @classmethod
  1820→    def _score_to_likelihood(cls, score: float, finding: Dict) -> int:
  1821→        """Estimate likelihood from score and finding metadata.
  1822→
  1823→        Higher CVSS exploitability metrics → higher likelihood.
  1824→        We approximate from score and severity.
  1825→        """
  1826→        severity = (finding.get("severity") or "").lower()
  1827→        is_compound = finding.get("is_compound", False)
  1828→        is_remote = True  # Most findings in recon are network-based
  1829→
  1830→        # Base likelihood from score
  1831→        base = score / 10.0
  1832→
  1833→        # Boost for compound findings (more likely to be exploitable)
  1834→        if is_compound:
  1835→            base = min(base + 0.15, 1.0)
  1836→
  1837→        # Adjust for severity
  1838→        sev_boost = {"critical": 0.1, "high": 0.05, "medium": 0, "low": -0.1, "info": -0.2}
  1839→        base += sev_boost.get(severity, 0)
  1840→        base = max(0.0, min(base, 1.0))
  1841→
  1842→        # Map to band index (0=Very Low, 4=Very High)
  1843→        if base >= 0.85:
  1844→            return 0
  1845→        elif base >= 0.65:
  1846→            return 1
  1847→        elif base >= 0.40:
  1848→            return 2
  1849→        elif base >= 0.15:
  1850→            return 3
  1851→        else:
  1852→            return 4
