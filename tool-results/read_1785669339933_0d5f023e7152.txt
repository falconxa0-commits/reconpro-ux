    90→
    91→class FindingDetailModal(ModalScreen):
    92→    """Modal overlay showing full finding details."""
    93→
    94→    def __init__(self, finding: Dict[str, Any]) -> None:
    95→        super().__init__()
    96→        self.finding = finding
    97→
    98→    def compose(self) -> ComposeResult:
    99→        f = self.finding
   100→        sev = f.get("severity", "info").lower()
   101→        sev_color = SEV_STYLES.get(sev, "white")
   102→        title = f.get("title", "Unknown")
   103→        category = f.get("category", "")
   104→        module = f.get("module", "")
   105→        pts = f.get("points_deducted", 0)
   106→        evidence = f.get("evidence", "")
   107→        remediation = f.get("remediation", "No remediation available.")
   108→        description = f.get("description", "")
   109→        asset = f.get("asset", "")
   110→        related_cves = f.get("related_cves", [])
   111→
   112→        lines: List[str] = []
   113→        lines.append(f"[{sev_color}][{sev.upper()}] {title}[/]")
   114→        lines.append("")
   115→        if asset:
   116→            lines.append(f"[dim]Asset:[/] [cyan]{asset}[/]")
   117→        if category:
   118→            lines.append(f"[dim]Category:[/] {category}")
   119→        if module:
   120→            lines.append(f"[dim]Module:[/] {module}")
   121→        lines.append(f"[dim]Points:[/] -{pts}")
   122→        if related_cves:
   123→            cve_ids = ", ".join(
   124→                c.get("cve_id", "") for c in related_cves[:5] if c.get("cve_id")
   125→            )
   126→            if cve_ids:
   127→                lines.append(f"[dim]CVEs:[/] [red]{cve_ids}[/]")
   128→        lines.append("")
   129→        if description:
   130→            lines.append(f"[bold]Description:[/]")
   131→            lines.append(f"[dim]{description}[/]")
   132→            lines.append("")
   133→        if evidence:
   134→            lines.append(f"[bold]Evidence:[/]")
   135→            lines.append(f"[dim]{evidence}[/]")
   136→            lines.append("")
   137→        lines.append(f"[bold]{GREEN}Remediation:[/]")
   138→        lines.append(f"{GREEN}{remediation}[/]")
   139→
   140→        content = "\n".join(lines)
   141→
   142→        with Vertical(classes="modal-container"):
   143→            yield Label(content, classes="modal-content")
   144→            with Horizontal(classes="modal-actions"):
   145→                yield Button("Close", variant="primary", id="close-modal")
   146→
   147→    def on_button_pressed(self, event: Button.Pressed) -> None:
   148→        if event.button.id == "close-modal":
   149→            self.dismiss()

   152→# ════════════════════════════════════════════════════════════════════════════════
   153→# Boot Screen Widget
   154→# ════════════════════════════════════════════════════════════════════════════════
   155→
   156→
   157→class BootScreen(Widget):
   158→    """Full-screen animated boot sequence."""
   159→
   160→    def __init__(self) -> None:
   161→        super().__init__()
   162→        self._step = 0
   163→        self._spinner_idx = 0
   164→        self._module_idx = 0
   165→        self._module_names = list(MODULE_NAMES.values())
   166→        self._done = False
   167→        self._timer: Optional[Timer] = None
   168→
   169→    def compose(self) -> ComposeResult:
   170→        with Vertical(classes="boot-container"):
   171→            yield Static("", id="boot-line-1", classes="boot-title")
   172→            yield Static("", id="boot-line-2", classes="boot-line")
   173→            yield Static("", id="boot-line-3", classes="boot-line")
   174→            yield Static("", id="boot-line-4", classes="boot-line")
   175→
   176→    def on_mount(self) -> None:
   177→        self._timer = self.set_interval(0.04, self._tick)
   178→
   179→    def _tick(self) -> None:
   180→        app = self.app
   181→        if not isinstance(app, NexusApp):
   182→            return
   183→
   184→        line1 = app.query_one("#boot-line-1", Static)
   185→        line2 = app.query_one("#boot-line-2", Static)
   186→        line3 = app.query_one("#boot-line-3", Static)
   187→        line4 = app.query_one("#boot-line-4", Static)
   188→
   189→        if self._step == 0:
   190→            # Letter-by-letter title
   191→            title = "RECONPRO NEXUS"
   192→            chars = title[: self._spinner_idx]
   193→            self._spinner_idx += 1
   194→            line1.update(f"[{CYAN} bold]{chars}[/]")
   195→            if self._spinner_idx > len(title):
   196→                self._spinner_idx = 0
   197→                self._step = 1
   198→                line1.update(f"[{CYAN} bold]{title}[/]  [{DIM_CYAN}]v{VERSION}[/]")
   199→
   200→        elif self._step == 1:
   201→            # Spinner for "Initializing agent..."
   202→            frame = SPINNER_FRAMES[self._spinner_idx % len(SPINNER_FRAMES)]
   203→            self._spinner_idx += 1
   204→            line2.update(f"  [{CYAN}]{frame}[/] [{DIM_CYAN}]Initializing agent...[/]")
   205→            if self._spinner_idx > 30:
   206→                self._spinner_idx = 0
   207→                self._step = 2
   208→                line2.update(f"  [{GREEN}]✓[/] [{DIM_CYAN}]Agent initialized.[/]")
   209→
   210→        elif self._step == 2:
   211→            # Module names appearing one by one
   212→            if self._module_idx < len(self._module_names):
   213→                visible = self._module_names[: self._module_idx + 1]
   214→                self._module_idx += 1
   215→                modules_str = "  ".join(f"[{CYAN}]○[/] [{DIM_CYAN}]{m}[/]" for m in visible)
   216→                remaining = len(self._module_names) - self._module_idx
   217→                if remaining > 0:
   218→                    modules_str += f"  [{DIM_CYAN}]+{remaining} more...[/]"
   219→                frame = SPINNER_FRAMES[self._spinner_idx % len(SPINNER_FRAMES)]
   220→                self._spinner_idx += 1
   221→                line3.update(
   222→                    f"  [{CYAN}]{frame}[/] [{DIM_CYAN}]Loading {len(self._module_names)} modules...[/]\n  {modules_str}"
   223→                )
   224→            else:
   225→                all_mods = "  ".join(
   226→                    f"[{GREEN}]●[/] [{DIM_CYAN}]{m}[/]" for m in self._module_names
   227→                )
   228→                line3.update(
   229→                    f"  [{GREEN}]✓[/] [{DIM_CYAN}]All {len(self._module_names)} modules loaded.[/]\n  {all_mods}"
   230→                )
   231→                self._step = 3
   232→                self._spinner_idx = 0
   233→
   234→        elif self._step == 3:
   235→            # "Ready." with flash
   236→            if self._spinner_idx == 0:
   237→                line4.update(f"[{YELLOW} bold]  ▶  READY.[/]")
   238→            elif self._spinner_idx == 6:
   239→                line4.update(f"[{GREEN} bold]  ▶  READY.[/]")
   240→            elif self._spinner_idx >= 10:
   241→                self._done = True
   242→                if self._timer:
   243→                    self._timer.stop()
   244→                self.app.call_from_thread(self.app._finish_boot)
   245→            self._spinner_idx += 1
   246→

   248→# ════════════════════════════════════════════════════════════════════════════════
   249→# Module Status Cell Widget
   250→# ════════════════════════════════════════════════════════════════════════════════
   251→
   252→
   253→class ModuleCell(Static):
   254→    """Single module status cell in the grid."""
   255→
   256→    def __init__(self, module_id: str, name: str) -> None:
   257→        super().__init__()
   258→        self.module_id = module_id
   259→        self.module_name = name
   260→        self.status = "idle"  # idle | scanning | done | error
   261→        self.finding_count = 0
   262→        self._spinner_idx = 0
   263→        self._timer: Optional[Timer] = None
   264→
   265→    def on_mount(self) -> None:
   266→        self._update_display()
   267→        self._timer = self.set_interval(0.1, self._spin)
   268→
   269→    def _spin(self) -> None:
   270→        if self.status == "scanning":
   271→            self._spinner_idx = (self._spinner_idx + 1) % len(SPINNER_FRAMES)
   272→            self._update_display()
   273→
   274→    def _update_display(self) -> None:
   275→        name_short = self.module_name[:14]
   276→        if self.status == "idle":
   277→            icon = f"[{DIM_CYAN}]○[/]"
   278→            status_text = f"[{DIM_CYAN}]idle[/]"
   279→        elif self.status == "scanning":
   280→            frame = SPINNER_FRAMES[self._spinner_idx]
   281→            icon = f"[{CYAN}]{frame}[/]"
   282→            status_text = f"[{CYAN}]scanning[/]"
   283→        elif self.status == "done":
   284→            icon = f"[{GREEN}]✓[/]"
   285→            status_text = f"[{GREEN}]done ({self.finding_count})[/]"
   286→        elif self.status == "error":
   287→            icon = f"[{RED}]✗[/]"
   288→            status_text = f"[{RED}]error[/]"
   289→        else:
   290→            icon = "○"
   291→            status_text = "idle"
   292→
   293→        self.update(
   294→            f"{icon} [{CYAN} bold]{name_short}[/]\n  {status_text}"
   295→        )
   296→
   297→    def set_status(self, status: str, count: int = 0) -> None:
   298→        self.status = status
   299→        self.finding_count = count
   300→        self._update_display()
   301→
   302→    def on_unmount(self) -> None:
   303→        if self._timer:
   304→            self._timer.stop()
   305→

   309→# ════════════════════════════════════════════════════════════════════════════════

   311→
   312→class StatusDot(Static):
   313→    """Pulsing dot indicator for scan status."""
   314→
   315→    scanning: reactive[bool] = reactive(False)
   316→
   317→    def __init__(self) -> None:
   318→        super().__init__("")
   319→        self._phase = 0
   320→
   321→    def on_mount(self) -> None:
   322→        self.set_interval(0.3, self._pulse)
   323→
   324→    def watch_scanning(self) -> None:
   325→        pass
   326→
   327→    def _pulse(self) -> None:
   328→        self._phase = (self._phase + 1) % 4
   329→        if self.scanning:
   330→            brightness = ["#003322", "#006644", CYAN, "#006644"][self._phase]
   331→            self.update(f"[{brightness}]●[/]")
   332→        else:
   333→            self.update(f"[{GREEN}]●[/]")

   783→        """Handle user command input."""

   799→    def on_key(self, event: Key) -> None:
   800→        """Handle key events for history navigation."""

   802→            try:
   803→                inp = self.query_one("#command-input", Input)
   804→                if inp.has_focus:
   805→                    self._history_index -= 1
   806→                    inp.value = self._command_history[self._history_index]
   807→            except NoMatches:
   808→                pass
   809→        elif event.key == "down":
   810→            try:
   811→                inp = self.query_one("#command-input", Input)
   812→                if inp.has_focus:
   813→                    if self._history_index < len(self._command_history) - 1:
   814→                        self._history_index += 1
   815→                        inp.value = self._command_history[self._history_index]
   816→                    else:
   817→                        self._history_index = len(self._command_history)
   818→                        inp.value = ""
   819→            except NoMatches:
   820→                pass

   987→    def _handle_scan(self, args: List[str]) -> None:
   988→        if not args:
   989→            self._chat(f"[{RED}]Usage: scan <target> [with <mod1> <mod2> ...][/]")
   990→            return
   991→
   992→        # Parse "with" clause
   993→        modules: Optional[List[str]] = None
   994→        target_parts: List[str] = []
   995→        found_with = False
   996→        mod_parts: List[str] = []
   997→        for a in args:
   998→            if a.lower() == "with":
   999→                found_with = True
  1000→                continue
  1001→            if found_with:
  1002→                mod_parts.append(a)
  1003→            else:
  1004→                target_parts.append(a)
  1005→
  1006→        target = " ".join(target_parts)
  1007→        if not target:
  1008→            self._chat(f"[{RED}]No target specified.[/]")
  1009→            return
  1010→
  1011→        if mod_parts:
  1012→            modules = [m.strip().lower() for m in mod_parts]
  1013→            self._chat(f"[{DIM_CYAN}]Scanning [cyan bold]{target}[/] with modules: {', '.join(modules)}[/]")
  1014→        else:
  1015→            self._chat(f"[{DIM_CYAN}]Scanning [cyan bold]{target}[/]...[/]")
  1016→
  1017→        self.current_target = target
  1018→        self._run_scan_worker(target, modules=modules)
  1019→

  1097→    # ════════════════════════════════════════════════════════════════════════
  1098→
  1099→    @work(exclusive=False, thread=True)
  1100→    def _run_scan_worker(
  1101→        self,
  1102→        target: str,
  1103→        modules: Optional[List[str]] = None,
  1104→        is_local: bool = False,
  1105→    ) -> None:
  1106→        """Run a scan in a background thread."""
  1107→        self.call_from_thread(self.is_scanning.set, True)
  1108→
  1109→        # Set modules to scanning
  1110→        if modules:
  1111→            for m in modules:
  1112→                if m in self._module_cells:
  1113→                    self.call_from_thread(self._module_cells[m].set_status, "scanning")
  1114→        else:
  1115→            all_mods = (
  1116→                list(MODULE_NAMES.keys())
  1117→                if is_local
  1118→                else [k for k in MODULE_NAMES if k not in ("host", "dev", "doctor")]
  1119→            )
  1120→            for m in all_mods:
  1121→                if m in self._module_cells:
  1122→                    self.call_from_thread(self._module_cells[m].set_status, "scanning")
  1123→
  1124→        try:
  1125→            from .scanner import scan, audit_scan
  1126→            from .history import save_scan
  1127→
  1128→            self.call_from_thread(
  1129→                self._chat, f"[{DIM_CYAN}]  ⏳ Scan in progress...[/]"
  1130→            )
  1131→
  1132→            if is_local:
  1133→                result = audit_scan(target=target, modules=modules)
  1134→            else:
  1135→                result = scan(target=target, modules=modules)
  1136→
  1137→            # Process findings
  1138→            data = result.to_dict()
  1139→            self.call_from_thread(self._on_scan_complete, data, modules, is_local)
  1140→
  1141→            # Save to history
  1142→            try:
  1143→                save_scan(data, label="nexus")
  1144→            except Exception:
  1145→                pass
