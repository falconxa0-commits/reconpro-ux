    32→    32→class ToolResult:

    35→    35→        self.output = output
    36→    36→        self.success = success
    37→    37→        self.data = data

    43→    43→class Tool:

    46→    46→                 execute: Callable[..., ToolResult], category: str = "scan"):

    49→    49→        self.parameters = parameters

    53→    53→    def to_openai_tool(self) -> Dict:
    54→    54→        return {
    55→    55→            "type": "function",
    56→    56→            "function": {
    57→    57→                "name": self.name,
    58→    58→                "description": self.description,
    59→    59→                "parameters": {
    60→    60→                    "type": "object",
    61→    61→                    "properties": self.parameters,
    62→    62→                    "required": list(self.parameters.keys()),
    63→    63→                },
    64→    64→            },
    65→    65→        }

    68→    68→class ToolRegistry:
    69→    69→    """Registry of all available tools."""
    70→    70→    def __init__(self):

    73→    73→    def register(self, tool: Tool) -> None:
    74→    74→        self.tools[tool.name] = tool
    75→    75→
    76→    76→    def get(self, name: str) -> Optional[Tool]:
    77→    77→        return self.tools.get(name)
    78→    78→
    79→    79→    def execute(self, name: str, **kwargs) -> ToolResult:
    80→    80→        tool = self.tools.get(name)
    81→    81→        if not tool:
    82→    82→            return ToolResult(f"Tool '{name}' not found", success=False)
    83→    83→        try:
    84→    84→            return tool.execute(**kwargs)
    85→    85→        except Exception as e:
    86→    86→            return ToolResult(f"Tool error: {e}", success=False)

    88→    88→    def list_tools(self) -> List[Dict]:
    89→    89→        return [{"name": t.name, "description": t.description, "category": t.category}
    90→    90→                for t in self.tools.values()]
    91→    91→
    92→    92→    def to_openai_tools(self) -> List[Dict]:
    93→    93→        return [t.to_openai_tool() for t in self.tools.values()]

    98→    98→class AgentMemory:
    99→    99→    """Persistent memory for the Nexus agent."""
   100→   100→    def __init__(self, project: str = "default"):
   101→   101→        self.project = project
   102→   102→        self._ensure_dir()
   103→   103→        self.memory_file = MEMORY_DIR / f"{project}.json"
   104→   104→        self._data = self._load()
   105→   105→
   106→   106→    def _ensure_dir(self) -> None:
   107→   107→        MEMORY_DIR.mkdir(parents=True, exist_ok=True)
   108→   108→
   109→   109→    def _load(self) -> Dict:
   110→   110→        if self.memory_file.exists():
   111→   111→            try:
   112→   112→                with open(self.memory_file) as f:
   113→   113→                    return json.load(f)
   114→   114→            except Exception:
   115→   115→                pass
   116→   116→        return {"project": self.project, "context": {}, "learned_patterns": [],
   117→   117→                "conversation_history": [], "created": datetime.now().isoformat(),
   118→   118→                "updated": datetime.now().isoformat()}
   119→   119→
   120→   120→    def _save(self) -> None:
   121→   121→        self._data["updated"] = datetime.now().isoformat()
   122→   122→        if len(self._data["conversation_history"]) > 100:
   123→   123→            self._data["conversation_history"] = self._data["conversation_history"][-50:]
   124→   124→        with open(self.memory_file, "w") as f:
   125→   125→            json.dump(self._data, f, indent=2, default=str)
   126→   126→
   127→   127→    def add_message(self, role: str, content: str) -> None:
   128→   128→        self._data["conversation_history"].append(
   129→   129→            {"role": role, "content": content, "ts": datetime.now().isoformat()})
   130→   130→        self._save()
   131→   131→
   132→   132→    def set_context(self, key: str, value: Any) -> None:
   133→   133→        self._data["context"][key] = value
   134→   134→        self._save()
   135→   135→
   136→   136→    def get_context(self, key: str, default=None) -> Any:
   137→   137→        return self._data["context"].get(key, default)
   138→   138→
   139→   139→    def add_pattern(self, pattern: str) -> None:
   140→   140→        if pattern not in self._data["learned_patterns"]:
   141→   141→            self._data["learned_patterns"].append(pattern)
   142→   142→            self._save()

   144→   144→    def get_history(self, limit: int = 20) -> List[Dict]:
   145→   145→        return self._data["conversation_history"][-limit:]

   147→   147→    def get_context_summary(self) -> str:
   148→   148→        ctx = self._data["context"]
   149→   149→        parts = [f"  {k}: {v}" for k, v in ctx.items()] if ctx else []

   152→   152→            parts.append(f"  Known patterns: {len(patterns)}")

   285→   285→def _tool_search_code(pattern: str, path: str = ".") -> ToolResult:
   286→   286→    try:
   287→   287→        p = Path(path).expanduser()
   288→   288→        if not p.is_dir(): p = p.parent
   289→   289→        matches = []
   290→   290→        for ext in ("*.py", "*.js", "*.ts", "*.jsx", "*.tsx", "*.go", "*.rs", "*.java", "*.rb", "*.php", "*.yml", "*.yaml", "*.json", "*.env"):
   291→   291→            for fp in p.rglob(ext):
   292→   292→                if ".git" in str(fp) or "node_modules" in str(fp): continue
   293→   293→                try:
   294→   294→                    content = fp.read_text()[:10000]
   295→   295→                    for i, line in enumerate(content.split("\n"), 1):
   296→   296→                        if re.search(pattern, line, re.IGNORECASE):
   297→   297→                            matches.append(f"{fp.relative_to(p)}:{i}: {line.strip()[:80]}")
   298→   298→                            if len(matches) >= 30: break
   299→   299→                except Exception: continue
   300→   300→            if len(matches) >= 30: break
   301→   301→        if not matches:
   302→   302→            return ToolResult(f"No matches for '{pattern}'.", success=True)
   303→   303→        return ToolResult(f"Found {len(matches)} match(es):\n" + "\n".join(matches[:30]), success=True)
   304→   304→    except Exception as e:
   305→   305→        return ToolResult(f"Search error: {e}", success=False)
