   581→function matchRules(prompt: string, session: SandboxSession): PatternRule {
   582→  for (const rule of RULES) {
   583→    for (const pattern of rule.patterns) {
   584→      if (pattern.test(prompt)) return rule;
   585→    }
   586→  }
   587→  return RULES[RULES.length - 1]; // generic fallback
   588→}
   589→
   590→function isBlockedByDefense(rule: PatternRule, session: SandboxSession): boolean {
   591→  if (!session.defenseMode || !rule.defenseOverride || !rule.requiresDefenseCheck) return false;
   592→  return rule.requiresDefenseCheck.some((check) => session.defenseRules[check]);
   593→}

   642→// ── Prompt Handler ─────────────────────────────────────────────────
   643→
   644→function handlePrompt(body: { sessionId: string; prompt: string }) {
   645→  const { sessionId, prompt } = body;
   646→  const session = sessions.get(sessionId);
   647→  if (!session) return NextResponse.json({ error: 'Session not found or expired' }, { status: 404 });
   648→  if (Date.now() > session.expiresAt) { sessions.delete(sessionId); return NextResponse.json({ error: 'Session expired' }, { status: 410 }); }
   649→
   650→  session.messages.push({ role: 'user', content: prompt, timestamp: Date.now() });
   651→  session.turnHistory.push(prompt.toLowerCase());
   652→
   653→  const rule = matchRules(prompt, session);
   654→  const blocked = isBlockedByDefense(rule, session);
   655→
   656→  const response = blocked ? rule.defenseOverride!.response : rule.response;
   657→  const action = blocked ? rule.defenseOverride!.action : rule.action;
   658→  const pts = blocked ? rule.defenseOverride!.points : rule.points;
   659→  const annotation = blocked ? rule.defenseOverride!.annotation : rule.annotation;
   660→  const success = !['BLOCKED', 'SECURITY_EVENT_LOGGED'].includes(action) && !action.includes('GUARDRAIL');
   661→
   662→  session.points += pts;
   663→  if (success) session.accessedResources.add(action);
   664→
   665→  session.messages.push({ role: 'agent', content: response, action, points: pts, annotation, timestamp: Date.now() });
   666→  return NextResponse.json({ response, action, success, points: pts, annotation, totalPoints: session.points });
   667→}

   669→// ── Defense Mode Update ────────────────────────────────────────────
   670→
   671→function updateDefense(body: { sessionId: string; defenseMode: boolean; defenseRules?: Partial<DefenseRules> }) {
   672→  const session = sessions.get(body.sessionId);
   673→  if (!session) return NextResponse.json({ error: 'Session not found' }, { status: 404 });
   674→  session.defenseMode = body.defenseMode;
   675→  if (body.defenseRules) session.defenseRules = { ...session.defenseRules, ...body.defenseRules };
   676→  return NextResponse.json({ defenseMode: session.defenseMode, defenseRules: session.defenseRules });
   677→}

   681→function getSession(body: { sessionId: string }) {
   682→  const session = sessions.get(body.sessionId);
   683→  if (!session) return NextResponse.json({ error: 'Session not found' }, { status: 404 });
   684→  return NextResponse.json({
   685→    sessionId: session.id, points: session.points, messageCount: session.messages.length,
   686→    messages: session.messages, defenseMode: session.defenseMode, defenseRules: session.defenseRules,
   687→    accessedResources: Array.from(session.accessedResources), expired: Date.now() > session.expiresAt,
   688→    timeRemaining: Math.max(0, session.expiresAt - Date.now()),
   689→  });
   690→}
