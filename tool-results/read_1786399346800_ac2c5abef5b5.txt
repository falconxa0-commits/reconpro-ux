    56→    def render(self, graph_data: dict, output_path: str = "") -> str:
    57→        """Build the full HTML string from *graph_data*.
    58→
    59→        *graph_data* must be a dict with ``"nodes"`` (list of dicts, each
    60→        with at least ``id``, optionally ``node_type``, ``label``, ``severity",
    61→        ``description``, ``cves``, ``remediation``) and ``"edges"`` (list of
    62→        dicts each with ``source``, ``target``, optionally ``edge_type``).
    63→
    64→        When *output_path* is given the HTML is also written to disk.
    65→
    66→        Returns the HTML string.
    67→        """
    68→        nodes_raw = graph_data.get("nodes", [])
    69→        edges_raw = graph_data.get("edges", [])
    70→
    71→        nodes = self._normalise_nodes(nodes_raw)
    72→        edges = self._normalise_edges(edges_raw, nodes)
    73→        degree_map = self._compute_degrees(edges)
    74→
    75→        nodes_json = json.dumps(nodes, ensure_ascii=False)
    76→        edges_json = json.dumps(edges, ensure_ascii=False)
    77→        degree_json = json.dumps(degree_map, ensure_ascii=False)
    78→
    79→        page = self._build_html(nodes_json, edges_json, degree_json)
    80→
    81→        if output_path:
    82→            return self.render_to_file(graph_data, output_path)
    83→
    84→        return page

    86→    def render_to_file(self, graph_data: dict, path: str) -> str:
    87→        """Write the HTML to *path* and return the absolute path."""
    88→        path = os.path.abspath(path)
    89→        os.makedirs(os.path.dirname(path) or ".", exist_ok=True)
    90→
    91→        nodes_raw = graph_data.get("nodes", [])
    92→        edges_raw = graph_data.get("edges", [])
    93→
    94→        nodes = self._normalise_nodes(nodes_raw)
    95→        edges = self._normalise_edges(edges_raw, nodes)
    96→        degree_map = self._compute_degrees(edges)
    97→
    98→        nodes_json = json.dumps(nodes, ensure_ascii=False)
    99→        edges_json = json.dumps(edges, ensure_ascii=False)
   100→        degree_json = json.dumps(degree_map, ensure_ascii=False)
   101→
   102→        page = self._build_html(nodes_json, edges_json, degree_json)
   103→
   104→        with open(path, "w", encoding="utf-8") as fh:
   105→            fh.write(page)
   106→
   107→        return path

   113→    @staticmethod
   114→    def _normalise_nodes(raw: list) -> list:
   115→        out = []
   116→        for n in raw:
   117→            if isinstance(n, dict):
   118→                nid = n.get("id") or n.get("node_id", str(id(n)))
   119→                node_type = n.get("node_type", n.get("type", "technology"))
   120→                label = n.get("label", n.get("name", nid))
   121→                out.append({
   122→                    "id": str(nid),
   123→                    "label": str(label),
   124→                    "node_type": str(node_type),
   125→                    "severity": str(n.get("severity", "")).lower(),
   126→                    "description": n.get("description", ""),
   127→                    "cves": n.get("cves", []),
   128→                    "remediation": n.get("remediation", ""),
   129→                    "extra": {k: v for k, v in n.items()
   130→                             if k not in ("id", "node_id", "label", "name",
   131→                                           "node_type", "type", "severity",
   132→                                           "description", "cves", "remediation")},
   133→                })
   134→            else:
   135→                out.append({
   136→                    "id": str(n),
   137→                    "label": str(n),
   138→                    "node_type": "technology",
   139→                    "severity": "",
   140→                    "description": "",
   141→                    "cves": [],
   142→                    "remediation": "",
   143→                    "extra": {},
   144→                })
   145→        return out
   146→
   147→    @staticmethod
   148→    def _normalise_edges(raw: list, nodes: list) -> list:
   149→        node_ids = {n["id"] for n in nodes}
   150→        out = []
   151→        for e in raw:
   152→            if isinstance(e, dict):
   153→                src = str(e.get("source", ""))
   154→                tgt = str(e.get("target", ""))
   155→                if not src or not tgt:
   156→                    continue
   157→                if src not in node_ids:
   158→                    continue
   159→                if tgt not in node_ids:
   160→                    continue
   161→                out.append({
   162→                    "source": src,
   163→                    "target": tgt,
   164→                    "edge_type": e.get("edge_type", e.get("relationship", "links_to")),
   165→                    "description": e.get("description", ""),
   166→                })
   167→            elif isinstance(e, (list, tuple)) and len(e) >= 2:
   168→                src, tgt = str(e[0]), str(e[1])
   169→                if src in node_ids and tgt in node_ids:
   170→                    out.append({"source": src, "target": tgt, "edge_type": "links_to", "description": ""})
   171→        return out
   172→
   173→    @staticmethod
   174→    def _compute_degrees(edges: list) -> dict:
   175→        deg: dict[str, int] = {}
   176→        for e in edges:
   177→            deg[e["source"]] = deg.get(e["source"], 0) + 1
   178→            deg[e["target"]] = deg.get(e["target"], 0) + 1
   179→        return deg

   569→function bfsPath(src, tgt) {{
   570→    const adj = {{}};
   571→    simEdges.forEach(e => {{
   572→        const sId = e.source.id || e.source, tId = e.target.id || e.target;
   573→        if (!adj[sId]) adj[sId] = [];
   574→        if (!adj[tId]) adj[tId] = [];
   575→        adj[sId].push(tId);
   576→        adj[tId].push(sId);
   577→    }});
   578→    const visited = new Set([src]);
   579→    const queue = [[src]];
   580→    while (queue.length) {{
   581→        const path = queue.shift();
   582→        const cur = path[path.length-1];
   583→        if (cur === tgt) return path;
   584→        (adj[cur]||[]).forEach(nb => {{
   585→            if (!visited.has(nb)) {{ visited.add(nb); queue.push([...path, nb]); }}
   586→        }});
   587→    }}
   588→    return null;
   589→}}
