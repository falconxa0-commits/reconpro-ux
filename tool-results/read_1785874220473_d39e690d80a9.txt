   171→   171→        # Keep only the last 50 sessions
   172→   172→        if len(self._data.get("sessions", [])) > 50:
   173→   173→            self._data["sessions"] = self._data["sessions"][-50:]
   174→   174→        try:
   175→   175→            CONTEXT_FILE.write_text(json.dumps(self._data, indent=2, default=str))

   180→   180→        self,

   195→   195→        self._data.setdefault("sessions", []).append(session)
   196→   196→        # Update target context

   878→   878→    def classify_intent(self, goal: str) -> Dict[str, float]:
   879→   879→        """Score each intent based on the goal text.
   880→   880→
   881→   881→        Uses weighted keyword matching, proximity scoring, and modifiers.
   882→   882→        Returns a dict mapping intent names to float scores in [0, 1+].
   883→   883→        """
   884→   884→        goal_lower = goal.lower()
   885→   885→        # Pad for whole-word matching
   886→   886→        padded = f' {goal_lower} '
   887→   887→        scores: Dict[str, float] = {}
   888→   888→        words = goal_lower.split()
   889→   889→
   890→   890→        # Calculate global modifier
   891→   891→        modifier = 1.0
   892→   892→        for w in words:
   893→   893→            if w in self._AMPLIFIERS:
   894→   894→                modifier = max(modifier, self._AMPLIFIERS[w])
   895→   895→            if w in self._DAMPENERS:
   896→   896→                modifier = min(modifier, self._DAMPENERS[w])
   897→   897→
   898→   898→        for intent, keywords in self._INTENT_KEYWORDS.items():
   899→   899→            score = 0.0
   900→   900→            for pattern, weight in keywords:
   901→   901→                # Use whole-word matching to avoid false positives
   902→   902→                # (e.g., "repo" matching inside "report")
   903→   903→                if len(pattern.split()) > 1:
   904→   904→                    # Multi-word pattern: check substring (already specific enough)
   905→   905→                    matched = pattern in goal_lower
   906→   906→                else:
   907→   907→                    # Single-word pattern: check whole word with padding
   908→   908→                    matched = f' {pattern} ' in padded
   909→   909→                if matched:
   910→   910→                    # Boost if near a target-like word
   911→   911→                    proximity_boost = 1.0
   912→   912→                    pattern_idx = goal_lower.find(pattern)
   913→   913→                    for tw in (".com", ".io", ".net", ".org", "http", "localhost"):
   914→   914→                        tw_idx = goal_lower.find(tw)
   915→   915→                        if tw_idx >= 0 and abs(pattern_idx - tw_idx) < 40:
   916→   916→                            proximity_boost = 1.3
   917→   917→                            break
   918→   918→                    score += weight * proximity_boost
   919→   919→            scores[intent] = min(score * modifier, 2.0)
   920→   920→
   921→   921→        return scores

   923→   923→    def extract_targets(self, goal: str) -> List[str]:
   924→   924→        """Extract targets (domains, IPs, URLs, paths) from the goal."""
   925→   925→        urls = re.findall(r'https?://[\w\-._~:/?#\[\]@!$&\'()*+,;=%]+', goal)
   926→   926→        domains = re.findall(r'[\w][\w\-]*\.[\w]{2,}(?:\.[\w]{2,})?', goal)
   927→   927→        skip = {
   928→   928→            'the', 'and', 'for', 'with', 'all', 'of', 'to', 'from', 'that', 'this',
   929→   929→            'scan', 'audit', 'check', 'find', 'get', 'show', 'me', 'my',
   930→   930→        }
   931→   931→        targets = list(set(
   932→   932→            urls + [d for d in domains if d.lower() not in skip and len(d) > 4]
   933→   933→        ))
   934→   934→        return targets
   935→   935→
   936→   936→    def detect_target_type(self, goal: str, targets: List[str]) -> str:
   937→   937→        """Classify the target type from context."""
   938→   938→        goal_lower = goal.lower()
   939→   939→        local_indicators = [
   940→   940→            'machine', 'laptop', 'computer', 'local', 'my system', 'this machine',
   941→   941→            'localhost', 'my mac', 'my pc', 'my server',
   942→   942→        ]
   943→   943→        for ind in local_indicators:
   944→   944→            if ind in goal_lower:
   945→   945→                return "local"
   946→   946→        dev_indicators = ['project', 'codebase', 'repo', 'source code', 'code review']
   947→   947→        padded = f' {goal_lower} '
   948→   948→        for ind in dev_indicators:
   949→   949→            if f' {ind} ' in padded and not targets:
   950→   950→                return "codebase"
   951→   951→        if targets:
   952→   952→            for t in targets:
   953→   953→                if t.startswith("http"):
   954→   954→                    return "remote"
   955→   955→                if re.match(r'^\d+\.\d+\.\d+\.\d+$', t):
   956→   956→                    return "remote"
   957→   957→            return "remote"
   958→   958→        return "unknown"
   959→   959→
   960→   960→    def detect_modules(self, goal: str) -> Optional[List[str]]:
   961→   961→        """Detect specific module requests from the goal text."""
   962→   962→        goal_lower = goal.lower()
   963→   963→        for mod_name, keywords in self._MODULE_KEYWORDS.items():
   964→   964→            for kw in keywords:
   965→   965→                if kw in goal_lower:
   966→   966→                    return [mod_name]
   967→   967→        return None

   969→   969→    def generate_plan(
   970→   970→        self, goal: str, scores: Dict[str, float],
   971→   971→        targets: List[str], target_type: str,
   972→   972→    ) -> List[Dict[str, Any]]:
   973→   973→        """Generate an ordered execution plan based on classified intents.
   974→   974→
   975→   975→        Returns a list of step dicts: {"tool", "args", "reason"}
   976→   976→        """
   977→   977→        plan: List[Dict[str, Any]] = []
   978→   978→        domain = ""
   979→   979→        if targets:
   980→   980→            domain = targets[0].replace("https://", "").replace("http://", "").split("/")[0]
   981→   981→
   982→   982→        # Sort intents by score, take significant ones
   983→   983→        sorted_intents = sorted(scores.items(), key=lambda x: x[1], reverse=True)
   984→   984→        significant = [(name, s) for name, s in sorted_intents if s > 0.2]
   985→   985→
   986→   986→        # Check prior context for this target
   987→   987→        prior = None
   988→   988→        if domain:
   989→   989→            prior = self.memory.get_target_context(domain)
   990→   990→
   991→   991→        # ── Subdomain recon phase ──
   992→   992→        if any(n == "subdomain_recon" and s > 0.3 for n, s in significant):
   993→   993→            if domain:
   994→   994→                plan.append({
   995→   995→                    "tool": "subdomain_discover",
   996→   996→                    "args": {"domain": domain},
   997→   997→                    "reason": "User requested subdomain reconnaissance",
   998→   998→                })

  1520→  1520→        """Execute using the intelligent rule-based planner."""
  1521→  1521→        # THINK phase

  1524→  1524→        target_type = self.planner.detect_target_type(goal, targets)

  1529→  1529→

  1532→  1532→        intent_str = " + ".join(f"{n} ({s:.1f})" for n, s in top_intents if s > 0.1)
  1533→  1533→
  1534→  1534→        self._print_header(goal, "rule-based planner")
  1535→  1535→        self._emit(f"  [THINK] Analyzing goal...")

  1562→  1562→

  1569→  1569→            completed.append((step, result))
  1570→  1570→
  1571→  1571→            # Track targets

  1580→  1580→            # VERIFY & ADAPT phase (after each step)

  1583→  1583→            new_steps = len(plan) - old_len
  1584→  1584→            if new_steps > 0:

  1586→  1586→                added = plan[step_idx + 1: step_idx + 1 + new_steps]
  1587→  1587→                self._emit(f"")
