   100→function makeRequest(
   101→  url = 'http://localhost/api/test',
   102→  headers: Record<string, string> = {},
   103→  body?: unknown,
   104→  method = 'GET',
   105→): NextRequest {
   106→  const init: RequestInit = { method };
   107→  const hdrs = new Headers(headers);
   108→  if (body !== undefined) {
   109→    init.body = JSON.stringify(body);
   110→    hdrs.set('content-type', 'application/json');
   111→  }
   112→  init.headers = hdrs;
   113→  return new NextRequest(url, init);
   114→}
