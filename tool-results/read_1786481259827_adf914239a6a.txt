    52→    81→    81→    81→            text = getattr(flowable, 'bookmark_text', '')
    53→    82→    82→    82→            key = getattr(flowable, 'bookmark_key', '')
    54→    83→    83→    83→            self.notify('TOCEntry', (level, text, self.page, key))
    55→    84→    84→    84→
    56→    85→    85→    85→# 2. MUST add TableOfContents to story
    57→    86→    86→    86→toc = TableOfContents()
    58→    87→    87→    87→toc.levelStyles = [toc_level0, toc_level1]  # define your styles
    59→    88→    88→    88→story.append(toc)
    60→    89→    89→    89→story.append(PageBreak())
    61→    90→    90→    90→
    62→    91→    91→    91→# 3. MUST set bookmark attributes on every heading
    63→    92→    92→    92→def add_heading(text, style, level=0):
    64→    93→    93→    93→    key = f'h_{hashlib.md5(text.encode()).hexdigest()[:8]}'
    65→    94→    94→    94→    p = Paragraph(f'<a name="{key}"/>{text}', style)
    66→    95→    95→    95→    p.bookmark_name = key
    67→    96→    96→    96→    p.bookmark_level = level
    68→    97→    97→    97→    p.bookmark_text = text
    69→    98→    98→    98→    p.bookmark_key = key
    70→    99→    99→    99→    return p
    71→   100→   100→   100→
    72→   101→   101→   101→# 4. MUST use multiBuild (NOT build)
    73→   102→   102→   102→doc = TocDocTemplate(output_path, pagesize=A4, ...)
    74→   103→   103→   103→doc.multiBuild(story)
    75→   104→   104→   104→```

   187→   216→   216→   216→## Step 3: EDIT - Content Transformation
   188→   217→   217→   217→
   189→   218→   218→   218→**Before writing any code, restructure raw text into visual roles.** This is the most impactful quality step - skipping it produces "Word document with a PDF extension".
   190→   219→   219→   219→
   191→   220→   220→   220→### 3a. Typographic Role Extraction
   192→   221→   221→   221→
   193→   222→   222→   222→Parse raw text and categorize content into visual roles:
   194→   223→   223→   223→
   195→   224→   224→   224→| Role | ReportLab Mapping | Description |
   196→   225→   225→   225→|------|------------------|-------------|
   197→   226→   226→   226→| **Hero / Display** | Cover title (36-42pt) or section opener | 1-5 words, the emotional hook |
   198→   227→   227→   227→| **Kicker / Eyebrow** | Subtitle or section intro (10-12pt, muted) | Tiny context line above main heading |
   199→   228→   228→   228→| **Data Sculpture** | CalloutBox or bold stat block | Extract impactful numbers (97%, $4.2M, -45ms) |
   200→   229→   229→   229→| **Pull Quote** | BlockQuote (italic + left indent 24pt + accent border) | Most provocative sentence, standalone |
   201→   230→   230→   230→| **Body** | Standard paragraph | Everything else |
   202→   231→   231→   231→
   203→   232→   232→   232→**Rule**: Before writing any body paragraph, scan for numbers with units. Every metric like "revenue grew by 12%" or "latency dropped to 45ms" MUST be extracted into a CalloutBox or metrics table - never buried in paragraph prose. See §Data-to-Ink Ratio Rules below.
   204→   233→   233→   233→
   205→   234→   234→   234→### 3b. Section Pacing
   206→   235→   235→   235→
   207→   236→   236→   236→- If any H1 section has **>400 words** of continuous body text without visual breaks, split into H2 subsections or insert a visual break (table, chart, callout)
   208→   237→   237→   237→- Aim for at least 1 visual element (table/chart/callout) per 2-3 pages of body text
   209→   238→   238→   238→- Dense text walls are the #1 sign of poor report design
   210→   239→   239→   239→
   211→   240→   240→   240→### 3c. Content Sanitization for External Sources (MANDATORY when source is PDF/OCR/web)
   212→   241→   241→   241→
   213→   242→   242→   242→When content is extracted from an external PDF, OCR output, or web scrape, it may contain invisible garbage characters that cause garbled rendering. **You MUST sanitize the text content before writing it into any `Paragraph()` or `canvas.drawString()`.**
   214→   243→   243→   243→
   215→   244→   244→   244→This is different from `code.sanitize` (which cleans Python source code). Content sanitization cleans the **text data itself**.
   216→   245→   245→   245→
   217→   246→   246→   246→```python
   218→   247→   247→   247→import re
   219→   248→   248→   248→
   220→   249→   249→   249→def content_sanitize(text: str) -> str:
   221→   250→   250→   250→    """Strip invisible/control chars from externally-sourced text."""
   222→   251→   251→   251→    # Remove control chars (except \n \t)
   223→   252→   252→   252→    text = re.sub(r'[\x00-\x08\x0b\x0c\x0e-\x1f\x7f]', '', text)
   224→   253→   253→   253→    # Remove zero-width chars
   225→   254→   254→   254→    text = re.sub(r'[\u200b-\u200f\u2028-\u202f\u2060\ufeff]', '', text)
   226→   255→   255→   255→    # Remove replacement char (upstream encoding error)
   227→   256→   256→   256→    text = text.replace('\ufffd', '')
   228→   257→   257→   257→    # Remove variation selectors
   229→   258→   258→   258→    text = re.sub(r'[\ufe00-\ufe0f]', '', text)
   230→   259→   259→   259→    # Remove private use area chars
   231→   260→   260→   260→    text = re.sub(r'[\ue000-\uf8ff]', '', text)
   232→   261→   261→   261→    return text
   233→   262→   262→   262→```
   234→   263→   263→   263→
   235→   264→   264→   264→**When to use:**
   236→   265→   265→   265→| Source | Must sanitize? |
   237→   266→   266→   266→|--------|---------------|
   238→   267→   267→   267→| User typed text in prompt | No (usually clean) |
   239→   268→   268→   268→| Extracted from PDF via `pdf.py extract.text` | **YES** |
   240→   269→   269→   269→| OCR output | **YES** |
   241→   270→   270→   270→| Web scrape / HTML strip | **YES** |
   242→   271→   271→   271→| LLM-generated content | No (usually clean) |
   243→   272→   272→   272→
   244→   273→   273→   273→⚠️ **Do NOT confuse** `code.sanitize` (Step 6, cleans Python code) with `content_sanitize()` (Step 3, cleans text data). They solve different problems.

   273→   302→   302→   302→## Character Safety Rule (MANDATORY)
   274→   303→   303→   303→
   275→   304→   304→   304→Three rules for safe character handling in ReportLab PDFs:
   276→   305→   305→   305→
   277→   306→   306→   306→**a) Superscripts and subscripts**: Use `<super>`/`<sub>` tags, never raw Unicode superscript/subscript characters (e.g., `\u00b2`, `\u2082`).
   278→   307→   307→   307→
   279→   308→   308→   308→**b) Emoji**: ReportLab cannot render emoji. If content contains emoji, use Creative brief (HTML + Playwright).
   280→   309→   309→   309→
   281→   310→   310→   310→**c) Font fallback for mixed CJK/Latin text (MANDATORY)**: After registering fonts, you **MUST** call `install_font_fallback()` once. This automatically wraps missing-glyph characters in `<font>` tags inside every `Paragraph()`. No manual `<font name="...">` wrapping needed for mixed Chinese-English text. **Skipping this call is the #1 cause of Helvetica garbled text in CJK documents.**
   282→   311→   311→   311→
   283→   312→   312→   312→**d) canvas.drawString / drawRightString CJK font rule (MANDATORY)**: `install_font_fallback()` only works on `Paragraph()` objects. It does NOT affect `canvas.drawString()` / `canvas.drawRightString()` / `canvas.drawCentredString()`. If the text passed to any `canvas.draw*String()` call contains CJK characters (Chinese/Japanese/Korean), you **MUST** call `canvas.setFont('NotoSerifSC', size)` (or another registered CJK font) before drawing. Using FreeSerif / Helvetica / Times for CJK canvas text = garbled output.
   284→   313→   313→   313→
   285→   314→   314→   314→```python
   286→   315→   315→   315→# ❌ WRONG — FreeSerif has no CJK glyphs, '科大讯飞' will be garbled
   287→   316→   316→   316→canvas.setFont('FreeSerif-Italic', 8)
   288→   317→   317→   317→canvas.drawRightString(x, y, 'SZSE: 002230  ·  科大讯飞')
   289→   318→   318→   318→
   290→   319→   319→   319→# ✅ CORRECT — use CJK font for mixed text
   291→   320→   320→   320→canvas.setFont('NotoSerifSC', 8)
   292→   321→   321→   321→canvas.drawRightString(x, y, 'SZSE: 002230  ·  科大讯飞')
   293→   322→   322→   322→
   294→   323→   323→   323→# ✅ ALSO CORRECT — pure English text can use FreeSerif
   295→   324→   324→   324→canvas.setFont('FreeSerif-Italic', 8)
   296→   325→   325→   325→canvas.drawString(x, y, 'Prepared by Strategy Advisory Team')
   297→   326→   326→   326→```
   298→   327→   327→   327→
   299→   328→   328→   328→Mathematical/relational operators (×, ÷, ±, ≤, √, ∑, ≅, ∫, π, ∠, Δ, etc.) are safe to use as literal characters in `Paragraph()` - both Noto Sans SC and FreeSerif have these glyphs.
   300→   329→   329→   329→
   301→   330→   330→   330→| Need | Correct Method | Correct Example |
   302→   331→   331→   331→|------|---------------|---------|
   303→   332→   332→   332→| Superscript | `<super>` tag in `Paragraph()` | `Paragraph('10<super>2</super> × 10<super>3</super> = 10<super>5</super>', style)` |
   304→   333→   333→   333→| Subscript | `<sub>` tag in `Paragraph()` | `Paragraph('H<sub>2</sub>O', style)` |
   305→   334→   334→   334→| Bold | `<b>` tag in `Paragraph()` | `Paragraph('<b>Title</b>', style)` |
   306→   335→   335→   335→| Math operators | Literal char in `Paragraph()` | `Paragraph('AB ⊥ AC, ∠A = 90°, and ΔABC ≅ ΔDCF', style)` |
   307→   336→   336→   336→| Scientific notation | Combined tags in `Paragraph()` | `Paragraph('1.2 × 10<super>8</super> kg/m<super>3</super>', style)` |
   308→   337→   337→   337→
   309→   338→   338→   338→```python
   310→   339→   339→   339→from reportlab.platypus import Paragraph
   311→   340→   340→   340→from reportlab.lib.styles import ParagraphStyle
   312→   341→   341→   341→from reportlab.lib.enums import TA_LEFT, TA_CENTER, TA_JUSTIFY
   313→   342→   342→   342→
   314→   343→   343→   343→body_style = ParagraphStyle(
   315→   344→   344→   344→    name="ENBodyStyle",
   316→   345→   345→   345→    fontName="FreeSerif",
   317→   346→   346→   346→    fontSize=10.5,
   318→   347→   347→   347→    leading=18,
   319→   348→   348→   348→    alignment=TA_JUSTIFY,
   320→   349→   349→   349→)
   321→   350→   350→   350→
   322→   351→   351→   351→# CJK Body (Chinese documents)
   323→   352→   352→   352→cn_body_style = ParagraphStyle(
   324→   353→   353→   353→    name="CNBodyStyle",
   325→   354→   354→   354→    fontName="NotoSerifSC", fontSize=10.5, leading=18,
   326→   355→   355→   355→    alignment=TA_LEFT,      # ← NEVER TA_JUSTIFY for CJK
   327→   356→   356→   356→    wordWrap='CJK',         # ← MANDATORY for CJK
   328→   357→   357→   357→    firstLineIndent=2*4.5*mm,
   329→   358→   358→   358→)
   330→   359→   359→   359→
   331→   360→   360→   360→# Superscript: area unit
   332→   361→   361→   361→Paragraph('Total area: 500 m<super>2</super>', body_style)
   333→   362→   362→   362→
   334→   363→   363→   363→# Subscript: chemical formula
   335→   364→   364→   364→Paragraph('The reaction produces CO<sub>2</sub> and H<sub>2</sub>O', body_style)
   336→   365→   365→   365→
   337→   366→   366→   366→# Scientific notation
   338→   367→   367→   367→Paragraph('Speed of light: 3.0 × 10<super>8</super> m/s', body_style)
   339→   368→   368→   368→
   340→   369→   369→   369→# Combined
   341→   370→   370→   370→Paragraph('E<sub>k</sub> = mv<super>2</super>/2', body_style)
   342→   371→   371→   371→
   343→   372→   372→   372→# Bold heading
   344→   373→   373→   373→Paragraph('<b>Chapter 1: Introduction</b>', header_style)

   486→   515→   515→   515→
   487→   516→   516→   516→### Page Breaks
   488→   517→   517→   517→
   489→   518→   518→   518→Follow the document type strategy defined in SOUL.md Rule 1.
   490→   519→   519→   519→
   491→   520→   520→   520→**Structural breaks (always - MANDATORY):**
   492→   521→   521→   521→- Between cover page and TOC - **cover must NEVER share a page with anything else**
   493→   522→   522→   522→- Between TOC and main content
   494→   523→   523→   523→- Between main content and back cover
   495→   524→   524→   524→
   496→   525→   525→   525→**Content breaks (by document type):**
   497→   526→   526→   526→- **Default (all document types)**: ❗ **Absolutely NEVER force page breaks before H1/H2** — content flows naturally, do not insert `PageBreak()` before section headings. This is an iron rule.
   498→   527→   527→   527→- Resume / contract / letter → No content page breaks
   499→   528→   528→   528→- Short article → No content page breaks
   500→   529→   529→   529→- **Exception 1**: Academic papers / textbooks / teaching plans → `PageBreak()` before each H1 is acceptable (only if user explicitly requests academic formatting)
   501→   530→   530→   530→- **Exception 2**: User explicitly requests page breaks between chapters
   502→   531→   531→   531→
   503→   532→   532→   532→**Anti-tear (mandatory — but with height limit):**
   504→   533→   533→   533→```python
   505→   534→   534→   534→from reportlab.platypus import KeepTogether
   506→   535→   535→   535→from reportlab.lib.pagesizes import A4
   507→   536→   536→   536→from reportlab.lib.units import inch
   508→   537→   537→   537→
   509→   538→   538→   538→# Maximum height for KeepTogether blocks: 40% of page height
   510→   539→   539→   539→# Blocks taller than this should NOT be kept together — they cause
   511→   540→   540→   540→# the preceding page to be mostly empty when the block gets pushed down.
   512→   541→   541→   541→MAX_KEEP_HEIGHT = A4[1] * 0.4  # ~336pt ≈ 12cm
   513→   542→   542→   542→
   514→   543→   543→   543→def safe_keep_together(elements):
   515→   544→   544→   544→    """Wrap elements in KeepTogether only if their total height is reasonable.
   516→   545→   545→   545→    If too tall, keep only the FIRST TWO elements together (e.g. title + table/image header)
   517→   546→   546→   546→    to prevent title-content separation, while letting the rest paginate naturally.
   518→   547→   547→   547→    """
   519→   548→   548→   548→    total_h = 0
   520→   549→   549→   549→    for el in elements:
   521→   550→   550→   550→        w, h = el.wrap(A4[0] - 2*inch, A4[1])
   522→   551→   551→   551→        total_h += h
   523→   552→   552→   552→    if total_h <= MAX_KEEP_HEIGHT:
   524→   553→   553→   553→        return [KeepTogether(elements)]
   525→   554→   554→   554→    elif len(elements) >= 2:
   526→   555→   555→   555→        # Keep at least title + first element together to prevent title orphaning
   527→   556→   556→   556→        return [KeepTogether(elements[:2])] + list(elements[2:])
   528→   557→   557→   557→    else:
   529→   558→   558→   558→        return list(elements)
   530→   559→   559→   559→
   531→   560→   560→   560→# Heading stays with first paragraph
   532→   561→   561→   561→story.extend(safe_keep_together([
   533→   562→   562→   562→    heading_paragraph,
   534→   563→   563→   563→    first_body_paragraph,
   535→   564→   564→   564→]))
   536→   565→   565→   565→
   537→   566→   566→   566→# Image + caption: keep together ONLY if image is small
   538→   567→   567→   567→story.extend(safe_keep_together([image, caption_paragraph]))
   539→   568→   568→   568→
   540→   569→   569→   569→# Table title + table: keep together ONLY if table is short
   541→   570→   570→   570→if len(table_data) <= 15:
   542→   571→   571→   571→    story.extend(safe_keep_together([table_title, table]))
   543→   572→   572→   572→else:
   544→   573→   573→   573→    # Long table: just keep title with the table start
   545→   574→   574→   574→    story.extend(safe_keep_together([table_title]))
   546→   575→   575→   575→    story.append(table)
   547→   576→   576→   576→```
   548→   577→   577→   577→
   549→   578→   578→   578→**⚠️ KEY INSIGHT:** `KeepTogether` is the #1 cause of “empty page with chart on next page”.
   550→   579→   579→   579→When a `KeepTogether` block is taller than the remaining space on the current page,
   551→   580→   580→   580→ReportLab pushes the **entire block** to the next page, leaving the current page mostly empty.
   552→   581→   581→   581→**Always use `safe_keep_together()` to cap the maximum block height.**
   553→   582→   582→   582→
   554→   583→   583→   583→**⚠️ Note:** Do NOT use `KeepWithNext` - it is unreliable in ReportLab 4.x.

   566→   595→   595→   595→### Cover Page: HTML/Playwright Unified Cover System
   567→   596→   596→   596→
   568→   597→   597→   597→**⚠️ Report route covers are ALWAYS generated via HTML/Playwright**, using the same 5-template system defined in `typesetting/cover.md`. This ensures visual consistency across all routes (Report, Creative, Academic) and avoids the limitations of ReportLab for complex cover layouts.
   569→   598→   598→   598→
   570→   599→   599→   599→**Pipeline (Report route):**
   571→   600→   600→   600→1. Generate **body PDF** via ReportLab (start with TOC or content - **no cover in story[]**)  
   572→   601→   601→   601→2. Generate **cover HTML** following `typesetting/cover.md` 5-template system  
   573→   602→   602→   602→3. **Run `poster_validate.py check-html` on cover HTML** — fix any ERRORs before rendering (overflow:hidden, font fallback, etc.)  
   574→   603→   603→   603→4. **Run `cover_validate.js` on cover HTML** — detects text-vs-decorative-line overlaps. Non-zero exit = must fix before proceeding.
   575→   604→   604→   604→   ```bash
   576→   605→   605→   605→   node "$PDF_SKILL_DIR/scripts/cover_validate.js" cover.html
   577→   606→   606→   606→   ```
   578→   607→   607→   607→5. Render cover HTML → single-page PDF via Playwright (`html2poster.js`) — **NOT `html2pdf-next.js`** (which converts absolute→static and destroys cover layout). **⚠️ MUST pass `--width 794px` for A4 documents** (A4 = 210mm = 794px @96dpi). Without this, `html2poster.js` defaults to 720px (=540pt), which is narrower than A4 (595pt) and causes page size mismatch after merging with the body PDF.  
   579→   608→   608→   608→6. **Merge: insert cover as page 0** of body PDF using pypdf → output single final PDF
   580→   609→   609→   609→

HTML/CSS handles these natively. One cover system, one visual standard, zero inconsistency.

   591→   620→   620→   620→- Technical documentation, product manuals, user guides
   592→   621→   621→   621→- Annual reports, financial summaries, quarterly reviews

   594→   623→   623→   623→

   604→   633→   633→   633→
   605→   634→   634→   634→#### Step 1: Generate Body PDF (ReportLab, no cover)
   606→   635→   635→   635→
   607→   636→   636→   636→Build the ReportLab document starting directly with TOC or first section. **Do NOT include any cover page in the `story[]` list.**
   608→   637→   637→   637→
   609→   638→   638→   638→```python
   610→   639→   639→   639→# Body PDF - starts with TOC or first section, NO cover
   611→   640→   640→   640→story = []
   612→   641→   641→   641→
   613→   642→   642→   642→# ⚠️ If document has TOC: use TocDocTemplate + multiBuild (see TOC Gate above)
   614→   643→   643→   643→# If no TOC: use SimpleDocTemplate + build
   615→   644→   644→   644→

   659→   688→   688→   688→```python
   660→   689→   689→   689→from pypdf import PdfReader, PdfWriter
   661→   690→   690→   690→
   662→   691→   691→   691→A4_W, A4_H = 595.28, 841.89  # A4 in points
   663→   692→   692→   692→
   664→   693→   693→   693→def normalize_page_to_a4(page):
   665→   694→   694→   694→    """Scale a page to A4 if its dimensions don't match."""
   666→   695→   695→   695→    box = page.mediabox
   667→   696→   696→   696→    w, h = float(box.width), float(box.height)
   668→   697→   697→   697→    if abs(w - A4_W) > 2 or abs(h - A4_H) > 2:
   669→   698→   698→   698→        page.scale_to(A4_W, A4_H)
   670→   699→   699→   699→    return page
   671→   700→   700→   700→
   672→   701→   701→   701→def insert_cover(cover_pdf, body_pdf, output_pdf):
   673→   702→   702→   702→    """Insert cover as first page of body PDF → single output file."""
   674→   703→   703→   703→    writer = PdfWriter()
   675→   704→   704→   704→    # Cover as page 1
   676→   705→   705→   705→    cover_page = PdfReader(cover_pdf).pages[0]
   677→   706→   706→   706→    writer.add_page(normalize_page_to_a4(cover_page))
   678→   707→   707→   707→    # Body pages follow
   679→   708→   708→   708→    for page in PdfReader(body_pdf).pages:
   680→   709→   709→   709→        writer.add_page(normalize_page_to_a4(page))
   681→   710→   710→   710→    writer.add_metadata({'/Title': 'Report Title', '/Author': 'Z.ai', '/Creator': 'Z.ai'})
   682→   711→   711→   711→    with open(output_pdf, 'wb') as f:
   683→   712→   712→   712→        writer.write(f)
   684→   713→   713→   713→```

   875→   904→   904→   904→### Table Width Management (⚠️ Prevent Table Overflow)
   876→   905→   905→   905→
   877→   906→   906→   906→**Most common table bug: columns overflow the right page margin because colWidths are hardcoded without checking available space.**
   878→   907→   907→   907→
   879→   908→   908→   908→**Iron Rule: The sum of all colWidths MUST NOT exceed `available_width`.**
   880→   909→   909→   909→
   881→   910→   910→   910→```python
   882→   911→   911→   911→from reportlab.lib.pagesizes import A4
   883→   912→   912→   912→from reportlab.lib.units import inch
   884→   913→   913→   913→
   885→   914→   914→   914→page_width = A4[0]  # 595.28pt
   886→   915→   915→   915→left_margin = 1.0 * inch   # 72pt (adjust to your doc's actual margins)
   887→   916→   916→   916→right_margin = 1.0 * inch  # 72pt
   888→   917→   917→   917→available_width = page_width - left_margin - right_margin  # ≈ 451pt
   889→   918→   918→   918→
   890→   919→   919→   919→# ─── Method 1: Proportional widths (RECOMMENDED) ───
   891→   920→   920→   920→# Define column ratios, auto-scale to fit available_width
   892→   921→   921→   921→col_ratios = [0.25, 0.40, 0.20, 0.15]  # must sum to 1.0
   893→   922→   922→   922→col_widths = [r * available_width for r in col_ratios]
   894→   923→   923→   923→table = Table(data, colWidths=col_widths)
   895→   924→   924→   924→
   896→   925→   925→   925→# ─── Method 2: Fixed + flex columns ───
   897→   926→   926→   926→# Some columns have fixed width (numbers, dates), rest fills remaining space
   898→   927→   927→   927→fixed_cols = {0: 80, 3: 60}  # col 0 = 80pt, col 3 = 60pt
   899→   928→   928→   928→fixed_total = sum(fixed_cols.values())
   900→   929→   929→   929→flex_count = 4 - len(fixed_cols)  # total columns minus fixed
   901→   930→   930→   930→flex_width = (available_width - fixed_total) / flex_count
   902→   931→   931→   931→col_widths = []
   903→   932→   932→   932→for i in range(4):
   904→   933→   933→   933→    col_widths.append(fixed_cols.get(i, flex_width))
   905→   934→   934→   934→table = Table(data, colWidths=col_widths)
   906→   935→   935→   935→
   907→   936→   936→   936→# ─── Method 3: Let ReportLab auto-calculate (simplest, less control) ───
   908→   937→   937→   937→# Pass colWidths=None - ReportLab auto-sizes based on content
   909→   938→   938→   938→# ⚠️ Risk: may still overflow if content is too wide
   910→   939→   939→   939→table = Table(data)  # no colWidths
   911→   940→   940→   940→```
   912→   941→   941→   941→
   913→   942→   942→   942→**Checklist before adding any table:**
   914→   943→   943→   943→1. Calculate `available_width` from your actual page size and margins
   915→   944→   944→   944→2. Sum all colWidths - must be ≤ `available_width`
   916→   945→   945→   945→3. For CJK text: characters are wider than Latin - budget ~12pt per CJK char at 10pt font
   917→   946→   946→   946→4. If a column has long text (descriptions, policies), use Paragraph() wrapping inside cells (not plain strings) - plain strings don't wrap and will overflow
   918→   947→   947→   947→5. Test with the longest expected content in each column
   919→   948→   948→   948→

   932→   961→   961→   961→

   979→  1008→  1008→  1008→## Auto-Generated Table of Contents
   980→  1009→  1009→  1009→
   981→  1010→  1010→  1010→### ❌ FORBIDDEN: Manual Table of Contents
   982→  1011→  1011→  1011→**NEVER manually create TOC with hardcoded page numbers.**
   983→  1012→  1012→  1012→
   984→  1013→  1013→  1013→### ⚠️ MANDATORY: TOC requires a cover page
   985→  1014→  1014→  1014→**Unless the user explicitly requests no cover, if a document has a Table of Contents, it MUST have a cover page.** Structure: Cover (page 1) → TOC (page 2) → Content (page 3+). Do not generate a TOC without a preceding cover page.
   986→  1015→  1015→  1015→
   987→  1016→  1016→  1016→### ✅ ALWAYS use auto-generated TOC:
   988→  1017→  1017→  1017→
   989→  1018→  1018→  1018→```python
   990→  1019→  1019→  1019→from reportlab.platypus import SimpleDocTemplate, Paragraph, PageBreak, Spacer
   991→  1020→  1020→  1020→from reportlab.platypus.tableofcontents import TableOfContents
   992→  1021→  1021→  1021→from reportlab.lib.utils import simpleSplit
   993→  1022→  1022→  1022→import hashlib
   994→  1023→  1023→  1023→
   995→  1024→  1024→  1024→class TocDocTemplate(SimpleDocTemplate):
   996→  1025→  1025→  1025→    def afterFlowable(self, flowable):
   997→  1026→  1026→  1026→        if hasattr(flowable, 'bookmark_name'):
   998→  1027→  1027→  1027→            level = getattr(flowable, 'bookmark_level', 0)
   999→  1028→  1028→  1028→            text = getattr(flowable, 'bookmark_text', '')
  1000→  1029→  1029→  1029→            key = getattr(flowable, 'bookmark_key', '')
  1001→  1030→  1030→  1030→            # MUST pass key as 4th element - without it, TOC entries won't have clickable links
  1002→  1031→  1031→  1031→            self.notify('TOCEntry', (level, text, self.page, key))
  1003→  1032→  1032→  1032→
  1004→  1033→  1033→  1033→doc = TocDocTemplate("document.pdf", pagesize=letter)
  1005→  1034→  1034→  1034→story = []
  1006→  1035→  1035→  1035→
  1007→  1036→  1036→  1036→toc = TableOfContents()
  1008→  1037→  1037→  1037→toc.levelStyles = [
  1009→  1038→  1038→  1038→    ParagraphStyle(name='TOCHeading1', fontSize=14, leftIndent=20, fontName='FreeSerif'),
  1010→  1039→  1039→  1039→    ParagraphStyle(name='TOCHeading2', fontSize=12, leftIndent=40, fontName='FreeSerif'),
  1011→  1040→  1040→  1040→]
  1012→  1041→  1041→  1041→story.append(Paragraph("<b>Table of Contents</b>", styles['Title']))
  1013→  1042→  1042→  1042→story.append(toc)
  1014→  1043→  1043→  1043→story.append(PageBreak())
  1015→  1044→  1044→  1044→
  1016→  1045→  1045→  1045→def add_heading(text, style, level=0):
  1017→  1046→  1046→  1046→    # Generate a unique bookmark key for this heading
  1018→  1047→  1047→  1047→    key = 'h_%s' % hashlib.md5(text.encode()).hexdigest()[:8]
  1019→  1048→  1048→  1048→    p = Paragraph('<a name="%s"/>%s' % (key, text), style)
  1020→  1049→  1049→  1049→    p.bookmark_name = text
  1021→  1050→  1050→  1050→    p.bookmark_level = level
  1022→  1051→  1051→  1051→    p.bookmark_text = text
  1023→  1052→  1052→  1052→    p.bookmark_key = key  # This key enables clickable TOC links
  1024→  1053→  1053→  1053→    return p
  1025→  1054→  1054→  1054→
  1026→  1055→  1055→  1055→# ⚠️ Orphan Heading Prevention (NOT a page break rule)
  1027→  1056→  1056→  1056→# If an H1 heading would appear in the bottom 15% of the page with no room
  1028→  1057→  1057→  1057→# for at least a few lines of content, push it to the next page.
  1029→  1058→  1058→  1058→# This is NOT "every H1 starts a new page" — it only triggers when the heading
  1030→  1059→  1059→  1059→# would be orphaned at the very bottom.
  1031→  1060→  1060→  1060→available_height = A4[1] - top_margin - bottom_margin
  1032→  1061→  1061→  1061→H1_ORPHAN_THRESHOLD = available_height * 0.15  # ~100pt ≈ 3.5cm — enough for heading + 2 lines
  1033→  1062→  1062→  1062→
  1034→  1063→  1063→  1063→def add_major_section(text, style):
  1035→  1064→  1064→  1064→    """Add an H1-level heading with orphan prevention (NOT forced page break)."""
  1036→  1065→  1065→  1065→    return [
  1037→  1066→  1066→  1066→        CondPageBreak(H1_ORPHAN_THRESHOLD),  # Only break if heading would be orphaned at page bottom
  1038→  1067→  1067→  1067→        add_heading(text, style, level=0),
  1039→  1068→  1068→  1068→    ]
  1040→  1069→  1069→  1069→
  1041→  1070→  1070→  1070→story.extend(add_major_section("Chapter 1: Introduction", styles['Heading1']))
  1042→  1071→  1071→  1071→# ... content ...
  1043→  1072→  1072→  1072→
  1044→  1073→  1073→  1073→doc.multiBuild(story)  # MUST use multiBuild for TOC
  1045→  1074→  1074→  1074→```

  1049→  1078→  1078→  1078→2. `add_heading` MUST set `bookmark_key` AND embed `<a name="key"/>` in the Paragraph text - this creates the link destination

  1053→  1082→  1082→  1082→

Only use this leader-dots approach if you need very specific visual styling that the auto TOC cannot provide - and even then, prefer the auto approach.
  1055→  1084→  1084→  1084→

  1071→  1100→  1100→  1100→

  1073→  1102→  1102→  1102→

  1080→  1109→  1109→  1109→

  1082→  1111→  1111→  1111→toc_style = ParagraphStyle('TOCEntry', parent=styles['Normal'],

  1084→  1113→  1113→  1113→dots_style = ParagraphStyle('LeaderDots', parent=styles['Normal'],

  1086→  1115→  1115→  1115→
  1087→  1116→  1116→  1116→# Build TOC

  1091→  1120→  1120→  1120→]

  1105→  1134→  1134→  1134→    ('ALIGN', (0, 0), (0, -1), 'LEFT'),

  1209→  1238→  1238→  1238→
  1210→  1239→  1239→  1239→**⚠️ CRITICAL: Always constrain to available space - NEVER insert at original size if it exceeds container width/height.**

  1217→  1246→  1246→  1246→from PIL import Image as PILImage
  1218→  1247→  1247→  1247→from reportlab.platypus import Image
  1219→  1248→  1248→  1248→from reportlab.lib.pagesizes import A4
  1220→  1249→  1249→  1249→from reportlab.lib.units import inch
  1221→  1250→  1250→  1250→
  1222→  1251→  1251→  1251→def embed_image(path: str, max_width: float = None, max_height: float = None) -> Image:
  1223→  1252→  1252→  1252→    """Embed image with preserved aspect ratio, constrained to container.
  1224→  1253→  1253→  1253→
  1225→  1254→  1254→  1254→    ALWAYS use this pattern. Never hardcode both width and height.
  1226→  1255→  1255→  1255→    If max_width is None, defaults to available_width (page - margins).
  1227→  1256→  1256→  1256→    """
  1228→  1257→  1257→  1257→    if max_width is None:
  1229→  1258→  1258→  1258→        max_width = A4[0] - 2.0 * inch  # Default: A4 width minus 1" margins each side
  1230→  1259→  1259→  1259→    if max_height is None:
  1231→  1260→  1260→  1260→        max_height = A4[1] * 0.35  # ~294pt ≈ 10cm — leaves room for caption + surrounding text on same page
  1232→  1261→  1261→  1261→
  1233→  1262→  1262→  1262→    pil_img = PILImage.open(path)
  1234→  1263→  1263→  1263→    orig_w, orig_h = pil_img.size
  1235→  1264→  1264→  1264→
  1236→  1265→  1265→  1265→    # Scale to fit within BOTH max_width and max_height
  1237→  1266→  1266→  1266→    ratio_w = max_width / orig_w if orig_w > max_width else 1.0
  1238→  1267→  1267→  1267→    ratio_h = max_height / orig_h if orig_h > max_height else 1.0
  1239→  1268→  1268→  1268→    ratio = min(ratio_w, ratio_h)
  1240→  1269→  1269→  1269→
  1241→  1270→  1270→  1270→    return Image(path, width=orig_w * ratio, height=orig_h * ratio)
  1242→  1271→  1271→  1271→
  1243→  1272→  1272→  1272→# Usage:
  1244→  1273→  1273→  1273→available_width = A4[0] - left_margin - right_margin
  1245→  1274→  1274→  1274→img = embed_image('chart.png', max_width=available_width, max_height=300)
  1246→  1275→  1275→  1275→story.append(img)
  1247→  1276→  1276→  1276→```

  1488→  1517→  1517→  1517→```python
  1489→  1518→  1518→  1518→def format_options(options):
  1490→  1519→  1519→  1519→    """
  1491→  1520→  1520→  1520→    Decide option layout based on content length.
  1492→  1521→  1521→  1521→    Short options (≤4 chars each, ≤4 options) → 2×2 grid table
  1493→  1522→  1522→  1522→    Long options or >4 options → vertical list, one per line
  1494→  1523→  1523→  1523→    """
  1495→  1524→  1524→  1524→    max_len = max(len(opt) for opt in options)
  1496→  1525→  1525→  1525→    if len(options) <= 4 and max_len <= 6:  # CJK: 4 chars; Latin: 6 chars
  1497→  1526→  1526→  1526→        # 2×2 grid using Table
  1498→  1527→  1527→  1527→        grid = Table(
  1499→  1528→  1528→  1528→            [[options[0], options[1]], [options[2], options[3]]],
  1500→  1529→  1529→  1529→            colWidths=[200, 200], hAlign='LEFT'
  1501→  1530→  1530→  1530→        )
  1502→  1531→  1531→  1531→        grid.setStyle(TableStyle([
  1503→  1532→  1532→  1532→            ('LEFTPADDING', (0,0), (-1,-1), 24),  # Match option indent
  1504→  1533→  1533→  1533→            ('VALIGN', (0,0), (-1,-1), 'TOP'),
  1505→  1534→  1534→  1534→        ]))
  1506→  1535→  1535→  1535→        return grid
  1507→  1536→  1536→  1536→    else:
  1508→  1537→  1537→  1537→        # Vertical list
  1509→  1538→  1538→  1538→        return [Paragraph(f'{opt}', option_style) for opt in options]
  1510→  1539→  1539→  1539→```
