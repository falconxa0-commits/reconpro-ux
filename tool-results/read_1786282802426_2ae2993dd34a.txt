   117→   117→        self.tools[tool.name] = tool

   120→   120→        return self.tools.get(name)

   122→   122→    def execute(self, name: str, **kwargs) -> ToolResult:
   123→   123→        tool = self.tools.get(name)
   124→   124→        if not tool:
   125→   125→            return ToolResult(f"Tool '{name}' not found.", success=False)
   126→   126→        try:
   127→   127→            return tool.execute(**kwargs)
   128→   128→        except Exception as e:
   129→   129→            return ToolResult(f"Tool error: {e}", success=False)

   134→   134→            for t in self.tools.values()

   149→   149→class AgentMemory:
   150→   150→    """Persistent memory for the Nexus agent across sessions."""
   151→   151→
   152→   152→    def __init__(self):
   153→   153→        MEMORY_DIR.mkdir(parents=True, exist_ok=True)

   156→   156→    def _load(self) -> Dict:
   157→   157→        if CONTEXT_FILE.exists():
   158→   158→            try:
   159→   159→                return json.loads(CONTEXT_FILE.read_text())
   160→   160→            except Exception:
   161→   161→                pass
   162→   162→        return {
   163→   163→            "sessions": [],
   164→   164→            "target_contexts": {},
   165→   165→            "learned_patterns": [],
   166→   166→            "created": datetime.now(timezone.utc).isoformat(),
   167→   167→        }
   168→   168→
   169→   169→    def _save(self) -> None:
   170→   170→        # Keep only the last 50 sessions
   171→   171→        if len(self._data.get("sessions", [])) > 50:
   172→   172→            self._data["sessions"] = self._data["sessions"][-50:]
   173→   173→        try:
   174→   174→            CONTEXT_FILE.write_text(json.dumps(self._data, indent=2, default=str))
   175→   175→        except OSError:
   176→   176→            pass
   177→   177→
   178→   178→    def save_session(
   179→   179→        self,
   180→   180→        goal: str,
   181→   181→        target: str,
   182→   182→        plan: List[Dict],
   183→   183→        tool_results: List[Dict],
   184→   184→        summary: str,
   185→   185→    ) -> None:
   186→   186→        session = {
   187→   187→            "goal": goal,
   188→   188→            "target": target,
   189→   189→            "plan": plan,
   190→   190→            "tool_results": tool_results,
   191→   191→            "summary": summary,
   192→   192→            "timestamp": datetime.now(timezone.utc).isoformat(),
   193→   193→        }
   194→   194→        self._data.setdefault("sessions", []).append(session)
   195→   195→        # Update target context
   196→   196→        self._data.setdefault("target_contexts", {})[target] = {
   197→   197→            "last_score": self._extract_score(tool_results),
   198→   198→            "last_grade": self._extract_grade(tool_results),
   199→   199→            "last_run": session["timestamp"],
   200→   200→            "finding_count": self._extract_finding_count(tool_results),
   201→   201→            "known_issues": self._extract_issues(tool_results),
   202→   202→        }
   203→   203→        self._save()
   204→   204→
   205→   205→    def get_target_context(self, target: str) -> Optional[Dict]:
   206→   206→        return self._data.get("target_contexts", {}).get(target)
   207→   207→
   208→   208→    def get_recent_sessions(self, limit: int = 5) -> List[Dict]:
   209→   209→        return self._data.get("sessions", [])[-limit:]

   211→   211→    @staticmethod
   212→   212→    def _extract_score(results: List[Dict]) -> Optional[int]:
   213→   213→        for r in results:
   214→   214→            d = r.get("data")
   215→   215→            if isinstance(d, dict) and "total_score" in d:
   216→   216→                return d["total_score"]
   217→   217→        return None
   218→   218→
   219→   219→    @staticmethod
   220→   220→    def _extract_grade(results: List[Dict]) -> Optional[str]:
   221→   221→        for r in results:
   222→   222→            d = r.get("data")
   223→   223→            if isinstance(d, dict) and "grade" in d:
   224→   224→                return d["grade"]
   225→   225→        return None
   226→   226→
   227→   227→    @staticmethod
   228→   228→    def _extract_finding_count(results: List[Dict]) -> int:
   229→   229→        total = 0
   230→   230→        for r in results:
   231→   231→            d = r.get("data")
   232→   232→            if isinstance(d, dict) and "findings" in d:
   233→   233→                total += len(d["findings"])
   234→   234→        return total
   235→   235→
   236→   236→    @staticmethod
   237→   237→    def _extract_issues(results: List[Dict]) -> List[str]:
   238→   238→        issues = []
   239→   239→        for r in results:
   240→   240→            d = r.get("data")
   241→   241→            if isinstance(d, dict) and "findings" in d:
   242→   242→                for f in d["findings"]:
   243→   243→                    if isinstance(f, dict) and f.get("severity") in ("critical", "high"):
   244→   244→                        issues.append(f.get("title", ""))
   245→   245→        return issues[:20]

   570→   570→            command, shell=True, capture_output=True, text=True, timeout=30,

   573→   573→        return ToolResult(

   575→   575→            success=result.returncode == 0,
   576→   576→        )
   577→   577→    except subprocess.TimeoutExpired:
   578→   578→        return ToolResult("Command timed out (30s).", success=False)
   579→   579→    except Exception as e:

   869→   869→         "av": "N", "ac": "L", "pr": "N", "ui": "N", "c": "H", "i": "H", "a": "H"},
   870→   870→        # Default fallback
   871→   871→        {"patterns": [],
   872→   872→         "av": "N", "ac": "L", "pr": "N", "ui": "N", "c": "L", "i": "L", "a": "N"},
   873→   873→    ]
   874→   874→
   875→   875→    @classmethod
   876→   876→    def score_finding(cls, finding: Dict) -> Dict[str, Any]:
   877→   877→        """Compute CVSS-like score for a single finding.
   878→   878→
   879→   879→        Returns a dict with: base_score, severity, av, ac, pr, ui, c, i, a, vector_string.
   880→   880→        """
   881→   881→        title = (finding.get("title") or "").lower()
   882→   882→        desc = (finding.get("description") or "").lower()
   883→   883→        combined = f"{title} {desc}"
   884→   884→        original_sev = (finding.get("severity") or "").lower()
   885→   885→
   886→   886→        # Find matching profile
   887→   887→        profile = cls._CATEGORY_PROFILES[-1]  # default
   888→   888→        for prof in cls._CATEGORY_PROFILES:
   889→   889→            if prof["patterns"] and any(p in combined for p in prof["patterns"]):
   890→   890→                profile = prof
   891→   891→                break
   892→   892→
   893→   893→        # Override metrics based on original severity if it's explicitly critical
   894→   894→        av = cls._AV[profile["av"]]
   895→   895→        ac = cls._AC[profile["ac"]]
   896→   896→        pr = cls._PR[profile["pr"]]
   897→   897→        ui = cls._UI[profile["ui"]]
   898→   898→        c = cls._CIA[profile["c"]]
   899→   899→        i = cls._CIA[profile["i"]]
   900→   900→        a = cls._CIA[profile["a"]]
   901→   901→
   902→   902→        # If the original severity is critical, boost impact to High
   903→   903→        if original_sev == "critical":
   904→   904→            c = max(c, cls._CIA["H"])
   905→   905→            i = max(i, cls._CIA["H"])
   906→   906→            a = max(a, cls._CIA["H"])
   907→   907→        elif original_sev == "high":
   908→   908→            c = max(c, cls._CIA["L"])
   909→   909→            i = max(i, cls._CIA["L"])
   910→   910→            a = max(a, cls._CIA["L"])
   911→   911→
   912→   912→        # CVSS base score calculation
   913→   913→        iss = 1.0 - ((1.0 - c) * (1.0 - i) * (1.0 - a))
   914→   914→        impact = 6.42 * iss
   915→   915→        exploitability = 8.22 * av * ac * pr * ui
   916→   916→
   917→   917→        if impact <= 0:
   918→   918→            base_score = 0.0
   919→   919→        else:
   920→   920→            base_score = min(impact + exploitability, 10.0)
   921→   921→
   922→   922→        # Round up to nearest 0.1
   923→   923→        base_score = math.ceil(base_score * 10) / 10
   924→   924→
   925→   925→        # Derive severity from score
   926→   926→        if base_score >= 9.0:
   927→   927→            derived_sev = "critical"
   928→   928→        elif base_score >= 7.0:
   929→   929→            derived_sev = "high"
   930→   930→        elif base_score >= 4.0:
   931→   931→            derived_sev = "medium"
   932→   932→        elif base_score > 0.0:
   933→   933→            derived_sev = "low"
   934→   934→        else:
   935→   935→            derived_sev = "info"
   936→   936→
   937→   937→        # Use the more severe of original vs derived
   938→   938→        severity_order = {"critical": 4, "high": 3, "medium": 2, "low": 1, "info": 0}
   939→   939→        final_sev = original_sev if severity_order.get(original_sev, -1) >= severity_order.get(derived_sev, -1) else derived_sev
   940→   940→
   941→   941→        vector = f"CVSS:3.1/AV:{profile['av']}/AC:{profile['ac']}/PR:{profile['pr']}/UI:{profile['ui']}/C:{cls._cia_label(c)}/I:{cls._cia_label(i)}/A:{cls._cia_label(a)}"
   942→   942→
   943→   943→        return {
   944→   944→            "base_score": base_score,
   945→   945→            "severity": final_sev,
   946→   946→            "cvss_vector": vector,
   947→   947→            "metrics": {
   948→   948→                "av": profile["av"], "ac": profile["ac"], "pr": profile["pr"],
   949→   949→                "ui": profile["ui"], "c": cls._cia_label(c), "i": cls._cia_label(i),
   950→   950→                "a": cls._cia_label(a),
   951→   951→            },
   952→   952→        }
   953→   953→
   954→   954→    @classmethod
   955→   955→    def score_findings(cls, findings: List[Dict]) -> List[Dict]:
   956→   956→        """Score a list of findings, returning enriched list."""
   957→   957→        enriched = []
   958→   958→        for f in findings:
   959→   959→            scored = dict(f)
   960→   960→            cvss = cls.score_finding(f)
   961→   961→            scored["cvss_score"] = cvss["base_score"]
   962→   962→            scored["cvss_severity"] = cvss["severity"]
   963→   963→            scored["cvss_vector"] = cvss["cvss_vector"]
   964→   964→            scored["cvss_metrics"] = cvss["metrics"]
   965→   965→            enriched.append(scored)
   966→   966→        return enriched
   967→   967→
   968→   968→    @staticmethod
   969→   969→    def _cia_label(value: float) -> str:
   970→   970→        if value >= 0.56:
   971→   971→            return "H"
   972→   972→        elif value >= 0.22:
   973→   973→            return "L"
   974→   974→        return "N"

  1404→  1404→        elif total >= 20:

  1406→  1406→        else:
  1407→  1407→            rating = "MINIMAL"
  1408→  1408→
  1409→  1409→        return {

  1411→  1411→            "risk_rating": rating,
  1412→  1412→            "breakdown": {
