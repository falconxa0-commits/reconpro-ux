   364→def _shannon_entropy(s: str) -> float:
   365→    if not s:
   366→        return 0.0
   367→    freq: Dict[str, int] = {}
   368→    for c in s:
   369→        freq[c] = freq.get(c, 0) + 1
   370→    length = len(s)
   371→    return -sum((count / length) * math.log2(count / length)
   372→                for count in freq.values())

   375→def _extract_host(target: str) -> str:

   379→def _resolve_ips(host: str) -> List[str]:
   380→    try:
   381→        results = socket.getaddrinfo(host, None)
   382→        return list(set(r[4][0] for r in results))
   383→    except Exception:
   384→        return []

   387→def _get_response_time(url: str, timeout: int = 5, verify_tls: bool = True) -> float:
   388→    t0 = time.monotonic()

   488→# Phase 2: Infrastructure Overlap Analysis
   489→# ═══════════════════════════════════════════════════════════════════════════

   491→def _check_infrastructure_overlap(
   492→    target: str, base_url: str, timeout: int, verify_tls: bool,
   493→    ttp_scores: Dict[str, float],
   494→) -> Tuple[Dict[str, float], List[Finding]]:
   495→    findings: List[Finding] = []
   496→    host = _extract_host(target)
   497→    scores = dict(ttp_scores)
   498→    ips = _resolve_ips(host)
   499→
   500→    # Check IP geo patterns
   501→    for ip in ips[:3]:
   502→        # Country inference from reverse DNS
   503→        try:
   504→            rdns = socket.gethostbyaddr(ip)[0]
   505→            for gid, gdata in APT_GROUP_DB.items():
   506→                for pat in gdata.get("c2_patterns", []):
   507→                    if re.search(pat, rdns.lower()):
   508→                        scores[gid] = scores.get(gid, 0) + 25
   509→        except Exception:
   510→            pass
   511→
   512→    # Check TLD
   513→    tld = host.rsplit(".", 1)[-1] if "." in host else ""
   514→    for gid, gdata in APT_GROUP_DB.items():
   515→        for pat in gdata.get("c2_patterns", []):
   516→            if re.search(pat, "." + tld + "$") or re.search(pat, host.lower()):
   517→                scores[gid] = scores.get(gid, 0) + 15
   518→
   519→    # Report top matches
   520→    top_groups = sorted(scores.items(), key=lambda x: x[1], reverse=True)[:5]
   521→    if top_groups:
   522→        details = []
   523→        for gid, score in top_groups:
   524→            g = APT_GROUP_DB[gid]
   525→            details.append(f"{gid} ({g['country']}): {score:.0f}%")
   526→        findings.append(_m(
   527→            f"Infrastructure Attribution: Top candidates identified",
   528→            "high" if top_groups[0][1] > 50 else "medium",
   529→            "infrastructure_attribution",
   530→            f"Based on infrastructure overlap analysis, top APT candidates: "
   531→            f"{'; '.join(details)}. "
   532→            f"Confidence scores are probabilistic — multiple groups share TTPs.",
   533→            "; ".join(details), host, 12,
   534→            "Cross-reference with threat intelligence feeds for confirmation.",
   535→            0.6,
   536→        ))
   537→
   538→    return scores, findings

   542→# Phase 3: Language & Localization Analysis
   543→# ═══════════════════════════════════════════════════════════════════════════
   544→
   545→def _analyze_language_signals(
   546→    target: str, base_url: str, timeout: int, verify_tls: bool,
   547→    overlap_scores: Dict[str, float],
   548→) -> Tuple[Dict[str, float], List[Finding]]:
   549→    findings: List[Finding] = []
   550→    host = _extract_host(target)
   551→    scores = dict(overlap_scores)
   552→
   553→    resp = http_probe(base_url, timeout=timeout, verify_tls=verify_tls, limiter=default_limiter)
   554→    body = resp.get("body", "")
   555→    headers = resp.get("headers", {})
   556→
   557→    # Detect language from headers and content
   558→    lang_signals: Set[str] = set()
   559→    ct = headers.get("content-language", "").lower()
   560→    if ct:
   561→        lang_signals.add(ct)
   562→
   563→    # Check for charset indicators
   564→    charset = headers.get("content-type", "").lower()
   565→    if "gb2312" in charset or "gbk" in charset or "gb18030" in charset:
   566→        lang_signals.add("zh-CN")
   567→    if "utf-8" in charset:
   568→        lang_signals.add("utf8")
   569→
   570→    # Check HTML lang attribute
   571→    lang_match = re.search(r'lang=["\']([^"\']+)["\']', body[:2000])
   572→    if lang_match:
   573→        lang_signals.add(lang_match.group(1).split("-")[0].lower())
   574→
   575→    # Check for CJK characters
   576→    if re.search(r'[\u4e00-\u9fff]', body):
   577→        lang_signals.add("cjk")
   578→
   579→    # Check for Cyrillic
   580→    if re.search(r'[\u0400-\u04ff]', body):
   581→        lang_signals.add("cyrillic")
   582→
   583→    # Check for Arabic
   584→    if re.search(r'[\u0600-\u06ff]', body):
   585→        lang_signals.add("arabic")
   586→
   587→    # Check for Hebrew
   588→    if re.search(r'[\u0590-\u05ff]', body):
   589→        lang_signals.add("hebrew")
   590→
   591→    # Score groups by language overlap
   592→    for gid, gdata in APT_GROUP_DB.items():
   593→        group_langs = set(gdata.get("languages", []))
   594→        for signal in lang_signals:
   595→            for lang in group_langs:
   596→                if signal in lang.lower() or signal in lang.split("-")[0].lower():
   597→                    scores[gid] = scores.get(gid, 0) + 10
   598→
   599→    if lang_signals:
   600→        findings.append(_m(
   601→            f"Language Analysis: Signals detected — {', '.join(sorted(lang_signals))}",
   602→            "medium", "language_analysis",
   603→            f"Language/localization signals: {', '.join(sorted(lang_signals))}. "
   604→            f"Cross-referenced against {len(APT_GROUP_DB)} group language profiles.",
   605→            f"Signals: {sorted(lang_signals)}", host, 5,
   606→            "Language signals alone are not sufficient for attribution.",
   607→            0.3,
   608→        ))
   609→
   610→    return scores, findings

   614→# Phase 4: Temporal Pattern Analysis

   617→def _analyze_temporal_patterns(
   618→    target: str, base_url: str, timeout: int, verify_tls: bool,
   619→    lang_scores: Dict[str, float],
   620→) -> Tuple[Dict[str, float], List[Finding]]:

   622→    host = _extract_host(target)
   623→    scores = dict(lang_scores)
   624→
   625→    # Measure response timing patterns
   626→    timings = []
   627→    for _ in range(5):
   628→        t = _get_response_time(base_url, timeout=min(timeout, 5), verify_tls=verify_tls)
   629→        timings.append(t)
   630→
   631→    if len(timings) >= 3:
   632→        mean_t = sum(timings) / len(timings)
   633→        # Check which groups' active hours match the current server timezone
   634→        current_hour = datetime.now(timezone.utc).hour
   635→
   636→        for gid, gdata in APT_GROUP_DB.items():
   637→            for tz_str in gdata.get("timezones", []):
   638→                try:
   639→                    # Simple timezone offset estimation
   640→                    tz_offsets = {
   641→                        "Asia/Shanghai": 8, "Asia/Hong_Kong": 8, "Asia/Taipei": 8,
   642→                        "Asia/Pyongyang": 9, "Asia/Tokyo": 9,
   643→                        "Europe/Moscow": 3, "Europe/Kiev": 2, "Europe/Berlin": 1,
   644→                        "Europe/Istanbul": 3,
   645→                        "Asia/Tehran": 3.5, "Asia/Dubai": 4,
   646→                        "Asia/Kolkata": 5.5,
   647→                        "Asia/Jerusalem": 2,
   648→                        "America/New_York": -5, "America/Chicago": -6,
   649→                    }
   650→                    offset = tz_offsets.get(tz_str, 0)
   651→                    local_hour = (current_hour + offset) % 24
   652→                    # Business hours = 9-18
   653→                    if 9 <= local_hour <= 18:
   654→                        scores[gid] = scores.get(gid, 0) + 5
   655→                except Exception:
   656→                    pass
   657→
   658→        findings.append(_m(
   659→            "Temporal Analysis: Server response patterns analyzed",
   660→            "info", "temporal_analysis",
   661→            f"Response timing: mean={mean_t:.3f}s, samples={len(timings)}. "
   662→            f"Current UTC hour: {current_hour}. "
   663→            f"Temporal patterns cross-referenced against group operational schedules.",
   664→            f"Mean RTT: {mean_t:.3f}s; UTC hour: {current_hour}",
   665→            host, 2,
   666→            "Temporal analysis provides weak signal — combine with other indicators.",
   667→            0.2,
   668→        ))
   669→
   670→    return scores, findings

   674→# Phase 5: Confidence Scoring & Attribution
   675→# ═══════════════════════════════════════════════════════════════════════════
   676→
   677→def _compute_attribution(final_scores: Dict[str, float]) -> List[Finding]:
   678→    findings: List[Finding] = []
   679→    if not final_scores:
   680→        return findings
   681→
   682→    top = sorted(final_scores.items(), key=lambda x: x[1], reverse=True)[:5]
   683→
   684→    for rank, (gid, score) in enumerate(top):
   685→        g = APT_GROUP_DB[gid]
   686→        confidence = min(score / 100, 1.0)
   687→
   688→        # Determine severity based on confidence
   689→        if confidence > 0.6:
   690→            sev = "critical"
   691→            pts = 20
   692→        elif confidence > 0.4:
   693→            sev = "high"
   694→            pts = 15
   695→        elif confidence > 0.2:
   696→            sev = "medium"
   697→            pts = 10
   698→        else:
   699→            sev = "low"
   700→            pts = 5
   701→
   702→        # Plausible deniability: low confidence = high deniability
   703→        deniability = 1.0 - confidence
   704→
   705→        findings.append(_m(
   706→            f"Attribution #{rank+1}: {gid} ({g['country']}) — {confidence:.0%} confidence",
   707→            sev, "nation_state_attribution",
   708→            f"Group: {gid} — '{g['name']}' ({g.get('aliases', [])[:2]}). "
   709→            f"Country: {g['country']}, Sponsored: {g.get('sponsored', False)}. "
   710→            f"Motivation: {', '.join(g.get('motivation', []))}. "
   711→            f"Sectors: {', '.join(g.get('sectors', [])[:4])}. "
   712→            f"Confidence: {confidence:.0%}, Plausible deniability: {deniability:.0%}. "
   713→            f"Techniques: {', '.join(g.get('techniques', [])[:5])}. "
   714→            f"Tool chains: {', '.join(g.get('tools', [])[:4])}.",
   715→            f"Score: {score:.1f}; Confidence: {confidence:.0%}; "
   716→            f"Aliases: {g.get('aliases', [])[:3]}",
   717→            g['country'], pts,
   718→            "Attribution is probabilistic. Verify with multiple sources before action.",
   719→            confidence * 0.8,
   720→        ))
   721→
   722→    return findings

   726→# Main Entry Point
   727→# ═══════════════════════════════════════════════════════════════════════════
   728→
   729→def run_nation_state_attributor(
   730→    target: str,
   731→    base_url: str,
   732→    timeout: int = 8,
   733→    verify_tls: bool = True,
   734→) -> List[Finding]:
   735→    """Run full nation-state attribution analysis.
   736→
   737→    Pipeline:
   738→    1. TTP correlation against 22 APT groups
   739→    2. Infrastructure overlap analysis
   740→    3. Language/localization signal extraction
   741→    4. Temporal pattern matching
   742→    5. Multi-factor confidence scoring with plausible deniability
   743→    """
   744→    findings: List[Finding] = []
   745→    host = _extract_host(target)
   746→
   747→    # Phase 1: TTP Correlation
   748→    ttp_scores, phase1 = _correlate_ttps(target, base_url, timeout, verify_tls)
   749→    findings.extend(phase1)
   750→
   751→    # Phase 2: Infrastructure Overlap
   752→    overlap_scores, phase2 = _check_infrastructure_overlap(
   753→        target, base_url, timeout, verify_tls, ttp_scores)
   754→    findings.extend(phase2)
   755→
   756→    # Phase 3: Language Analysis
   757→    lang_scores, phase3 = _analyze_language_signals(
   758→        target, base_url, timeout, verify_tls, overlap_scores)
   759→    findings.extend(phase3)
   760→
   761→    # Phase 4: Temporal Analysis
   762→    final_scores, phase4 = _analyze_temporal_patterns(
   763→        target, base_url, timeout, verify_tls, lang_scores)
   764→    findings.extend(phase4)
   765→
   766→    # Phase 5: Final Attribution
   767→    findings.extend(_compute_attribution(final_scores))
   768→
   769→    return findings
