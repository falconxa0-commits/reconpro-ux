   144→def _b64url_encode(data: bytes) -> str:
   145→    """Base64url-encode bytes (no padding)."""
   146→    return base64.urlsafe_b64encode(data).rstrip(b"=").decode("ascii")

   149→def _b64url_decode(s: str) -> bytes:
   150→    """Base64url-decode with padding restoration."""
   151→    padding = 4 - len(s) % 4
   152→    if padding != 4:
   153→        s += "=" * padding
   154→    return base64.urlsafe_b64decode(s)

   157→def _build_jwt(header: Dict[str, Any], payload: Dict[str, Any],
   158→               secret: Optional[bytes] = None) -> str:
   159→    """Build a JWT token. If secret is None, no signature (alg=none)."""
   160→    h = _b64url_encode(json.dumps(header, separators=(",", ":")).encode())
   161→    p = _b64url_encode(json.dumps(payload, separators=(",", ":")).encode())
   162→    signing_input = f"{h}.{p}".encode()
   163→    if secret is None:
   164→        return f"{h}.{p}."
   165→    sig = hmac.new(secret, signing_input, hashlib.sha256).digest()
   166→    return f"{h}.{p}.{_b64url_encode(sig)}"

   315→def _follow_redirects(base_url: str, max_hops: int = 15, timeout: int = 8,
   316→                       verify_tls: bool = True) -> List[Dict[str, Any]]:
   317→    """Follow HTTP redirect chains and record each hop."""
   318→    chain = []
   319→    url = base_url
   320→    visited: Set[str] = set()
   321→
   322→    for _ in range(max_hops):
   323→        if url in visited:
   324→            break
   325→        visited.add(url)
   326→
   327→        resp = http_probe(url, timeout=timeout, verify_tls=verify_tls)
   328→        status = resp.get("status", 0)
   329→        headers = resp.get("headers", {})
   330→        location = headers.get("location", "")
   331→
   332→        chain.append({
   333→            "url": url,
   334→            "status": status,
   335→            "location": location,
   336→            "headers": {k: v for k, v in headers.items()
   337→                       if k.lower().startswith(("x-", "server", "cf-"))},
   338→        })
   339→
   340→        if status not in (301, 302, 303, 307, 308) or not location:
   341→            break
   342→
   343→        # Resolve relative URLs
   344→        if location.startswith("/"):
   345→            from urllib.parse import urljoin
   346→            location = urljoin(url, location)
   347→        elif not location.startswith("http"):
   348→            location = urljoin(url, location)
   349→        url = location
   350→
   351→    return chain
