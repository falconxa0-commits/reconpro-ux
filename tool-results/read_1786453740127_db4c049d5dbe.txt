  1324→def _findings_to_dicts(findings: Any) -> List[Dict[str, Any]]:
  1325→    """Normalize findings (list of dicts or objects) to list of dicts."""
  1326→    result: List[Dict[str, Any]] = []
  1327→    for f in findings:
  1328→        if isinstance(f, dict):
  1329→            result.append(f)
  1330→        elif hasattr(f, "to_dict"):
  1331→            result.append(f.to_dict())
  1332→        elif hasattr(f, "__dict__"):
  1333→            result.append(vars(f))
  1334→        else:
  1335→            result.append({"title": str(f)})
  1336→    return result
  1337→

  1343→_CSV_COLUMNS = [
  1344→    "severity", "module", "category", "title", "description",
  1345→    "evidence", "asset", "points_deducted", "remediation",
  1346→    "cvss_score", "cvss_vector",

  1353→def generate_csv_report(
  1354→    findings: Any,
  1355→    output_path: str,
  1356→    score: Optional[float] = None,
  1357→) -> str:
  1358→    """Export findings to CSV (pure stdlib).
  1359→
  1360→    Parameters
  1361→    ----------
  1362→    findings : list of dict-like
  1363→        Raw findings as produced by scanner modules.
  1364→    output_path : str
  1365→        Destination file path.
  1366→    score : float, optional
  1367→        Overall score (written as comment in header area — CSV has no
  1368→        native comment mechanism, so it is omitted from the CSV itself).
  1369→
  1370→    Returns
  1371→    -------
  1372→    str
  1373→        Absolute path of the written file.
  1374→    """
  1375→    rows = _findings_to_dicts(findings)
  1376→    path = Path(output_path)
  1377→    path.parent.mkdir(parents=True, exist_ok=True)
  1378→
  1379→    with open(str(path), "w", newline="", encoding="utf-8") as fh:
  1380→        writer = csv.DictWriter(fh, fieldnames=_CSV_COLUMNS, extrasaction="ignore")
  1381→        writer.writeheader()
  1382→        for row in rows:
  1383→            writer.writerow({col: row.get(col, "") for col in _CSV_COLUMNS})
  1384→
  1385→    return str(path.resolve())

  1399→def generate_sarif_report(
  1400→    findings: Any,
  1401→    output_path: str,
  1402→    target: str = "",
  1403→) -> str:
  1404→    """Export findings to SARIF v2.1.0 (GitHub Code Scanning compatible).
  1405→
  1406→    Parameters
  1407→    ----------
  1408→    findings : list of dict-like
  1409→    output_path : str
  1410→    target : str
  1411→        Target identifier included in run properties.
  1412→
  1413→    Returns
  1414→    -------
  1415→    str
  1416→        Absolute path of the written file.
  1417→    """
  1418→    rows = _findings_to_dicts(findings)
  1419→    path = Path(output_path)
  1420→    path.parent.mkdir(parents=True, exist_ok=True)
  1421→
  1422→    # Build unique rules from findings
  1423→    rules: List[Dict[str, Any]] = []
  1424→    seen_rules: Dict[str, int] = {}
  1425→    for f in rows:
  1426→        rule_id = f.get("category", f.get("title", "RP-0000"))
  1427→        rule_id = rule_id[:100] if len(rule_id) > 100 else rule_id
  1428→        if rule_id not in seen_rules:
  1429→            idx = len(rules)
  1430→            seen_rules[rule_id] = idx
  1431→            rules.append({
  1432→                "id": rule_id,
  1433→                "shortDescription": {
  1434→                    "text": f.get("title", rule_id),
  1435→                },
  1436→                "helpUri": f.get("remediation", ""),
  1437→                "properties": {
  1438→                    "security-severity": str(f.get("cvss_score", "")),
  1439→                },
  1440→            })
  1441→
  1442→    results: List[Dict[str, Any]] = []
  1443→    for f in rows:
  1444→        severity = _SARIF_SEVERITY_MAP.get(
  1445→            str(f.get("severity", "info")).lower(), "note"
  1446→        )
  1447→        rule_id = f.get("category", f.get("title", "RP-0000"))
  1448→        rule_id = rule_id[:100] if len(rule_id) > 100 else rule_id
  1449→        rule_index = seen_rules.get(rule_id, 0)
  1450→
  1451→        asset = f.get("asset", target)
  1452→        location: Dict[str, Any] = {
  1453→            "physicalLocation": {
  1454→                "artifactLocation": {
  1455→                    "uri": asset,
  1456→                },
  1457→            },
  1458→        }
  1459→
  1460→        results.append({
  1461→            "ruleId": rule_id,
  1462→            "ruleIndex": rule_index,
  1463→            "level": severity,
  1464→            "message": {
  1465→                "text": f.get("description", ""),
  1466→            },
  1467→            "locations": [location],
  1468→            "properties": {
  1469→                "points_deducted": f.get("points_deducted", 0),
  1470→                "module": f.get("module", ""),
  1471→                "evidence": f.get("evidence", "")[:500] if f.get("evidence") else "",
  1472→            },
  1473→        })
  1474→
  1475→    sarif: Dict[str, Any] = {
  1476→        "$schema": "https://raw.githubusercontent.com/oasis-tcs/sarif-spec/master/Schemata/sarif-schema-2.1.0.json",
  1477→        "version": "2.1.0",
  1478→        "runs": [
  1479→            {
  1480→                "tool": {
  1481→                    "driver": {
  1482→                        "name": "ReconPro",
  1483→                        "version": "9.0.0",
  1484→                        "rules": rules,
  1485→                    },
  1486→                },
  1487→                "results": results,
  1488→                "properties": {
  1489→                    "target": target,
  1490→                },
  1491→            }
  1492→        ],
  1493→    }
  1494→
  1495→    with open(str(path), "w", encoding="utf-8") as fh:
  1496→        json.dump(sarif, fh, indent=2, ensure_ascii=False)
  1497→        fh.write("\n")
  1498→
  1499→    return str(path.resolve())

  1656→        for idx, f in enumerate(rows, start=1):
  1657→            sev = str(f.get("severity", "info")).lower()
  1658→            badge = _MD_SEVERITY_BADGE.get(sev, sev.upper())
  1659→            lines.append(f"### {idx}. {badge} — {f.get('title', 'Untitled')}")
  1660→            lines.append("")
  1661→            if f.get("description"):
  1662→                lines.append(f.get("description"))
  1663→                lines.append("")
  1664→            if f.get("evidence"):
  1665→                lines.append(f"**Evidence:**")

  1667→                lines.append(str(f["evidence"])[:1000])
  1668→                lines.append("```")
  1669→                lines.append("")
  1670→            if f.get("remediation"):
  1671→                lines.append(f"**Remediation:** {f.get('remediation')}")
  1672→                lines.append("")

  1674→            lines.append("")
  1675→
  1676→    lines.append("---")
  1677→    lines.append("*Generated by ReconPro v9.0.0*  ")

  1689→def generate_pdf_report(
  1690→    findings: Any,
  1691→    output_path: str,
  1692→    score: Optional[float] = None,
  1693→    target: str = "",
  1694→) -> str:
  1695→    """Generate a printable-HTML report styled for PDF output.
  1696→
  1697→    Creates a self-contained HTML file with print-optimized CSS.
  1698→    The file uses .html extension internally so browsers can render it,
  1699→    but the output_path is respected as given by the caller.
  1700→
  1701→    Parameters
  1702→    ----------
  1703→    findings : list of dict-like
  1704→    output_path : str
  1705→    score : float, optional
  1706→    target : str, optional
  1707→
  1708→    Returns
  1709→    -------
  1710→    str
  1711→        Absolute path of the written file.
  1712→    """
  1713→    rows = _findings_to_dicts(findings)
  1714→    path = Path(output_path)
  1715→    path.parent.mkdir(parents=True, exist_ok=True)
  1716→
  1717→    # If caller passed a .pdf extension, use .html instead
  1718→    actual_path = path
  1719→    if path.suffix.lower() == ".pdf":
  1720→        actual_path = path.with_suffix(".html")
  1721→
  1722→    rows.sort(key=lambda f: _SEVERITY_ORDER.get(str(f.get("severity", "info")).lower(), 0), reverse=True)
  1723→
  1724→    # Severity counts
  1725→    counts: Dict[str, int] = {}
  1726→    for f in rows:
  1727→        s = str(f.get("severity", "info")).lower()
  1728→        counts[s] = counts.get(s, 0) + 1
  1729→
  1730→    # ASCII gauge
  1731→    gauge_width = 40
  1732→    filled = 0
  1733→    if score is not None:
  1734→        filled = int(round(score / 100 * gauge_width))
  1735→    gauge_bar = "█" * filled + "░" * (gauge_width - filled)
  1736→    score_display = f"{score:.1f}" if score is not None else "N/A"
  1737→
  1738→    # Build findings rows
  1739→    finding_rows_html = ""
  1740→    for idx, f in enumerate(rows, start=1):
  1741→        sev = str(f.get("severity", "info")).lower()
  1742→        title = f.get("title", "Untitled")
  1743→        desc = f.get("description", "")
  1744→        rem = f.get("remediation", "")
  1745→        pts = f.get("points_deducted", 0)
  1746→        mod = f.get("module", "")
  1747→        sev_color = {
  1748→            "critical": "#dc2626", "high": "#ea580c",
  1749→            "medium": "#d97706", "low": "#2563eb", "info": "#6b7280",
  1750→        }.get(sev, "#6b7280")
  1751→        finding_rows_html += f"""
  1752→        <tr>
  1753→          <td style="text-align:center">{idx}</td>
  1754→          <td style="color:{sev_color};font-weight:bold;text-transform:uppercase">{sev}</td>
  1755→          <td>{mod}</td>
  1756→          <td>{title}</td>
  1757→          <td style="text-align:center">{pts}</td>
  1758→        </tr>"""
  1759→
  1760→    # Remediation summary
  1761→    rem_html = ""
  1762→    for f in rows:
  1763→        if f.get("remediation"):
  1764→            rem_html += f"<li><strong>{f.get('title', 'Untitled')}</strong>: {f['remediation']}</li>\n"
  1765→
  1766→    html = f"""<!DOCTYPE html>
  1767→<html lang="en">
  1768→<head>
  1769→<meta charset="utf-8">
  1770→<title>ReconPro Report — {target or 'Scan'}</title>
  1771→<style>
  1772→  @page {{ size: A4; margin: 1.5cm; }}
  1773→  * {{ box-sizing: border-box; margin: 0; padding: 0; }}
  1774→  body {{ font-family: 'Segoe UI', Arial, sans-serif; font-size: 11pt; color: #1f2937; line-height: 1.6; padding: 2cm; max-width: 210mm; margin: auto; }}
  1775→  h1 {{ font-size: 20pt; margin-bottom: 4px; }}
  1776→  h2 {{ font-size: 14pt; margin: 20px 0 8px; border-bottom: 2px solid #e5e7eb; padding-bottom: 4px; }}
  1777→  .meta {{ color: #6b7280; font-size: 9pt; margin-bottom: 16px; }}
  1778→  .gauge {{ font-family: 'Courier New', monospace; font-size: 12pt; background: #f9fafb; padding: 12px 16px; border-radius: 6px; border: 1px solid #e5e7eb; margin: 12px 0 20px; }}
  1779→  .gauge-label {{ font-family: 'Segoe UI', Arial, sans-serif; font-size: 9pt; color: #6b7280; margin-bottom: 4px; }}
  1780→  table {{ width: 100%; border-collapse: collapse; margin: 12px 0; font-size: 9pt; }}
  1781→  th {{ background: #f3f4f6; text-align: left; padding: 6px 8px; border-bottom: 2px solid #d1d5db; }}
  1782→  td {{ padding: 5px 8px; border-bottom: 1px solid #e5e7eb; vertical-align: top; }}
  1783→  tr:nth-child(even) {{ background: #f9fafb; }}
  1784→  .summary-grid {{ display: grid; grid-template-columns: repeat(5, 1fr); gap: 8px; margin: 12px 0; }}
  1785→  .summary-box {{ text-align: center; padding: 10px; border-radius: 6px; border: 1px solid #e5e7eb; }}
  1786→  .summary-box .num {{ font-size: 18pt; font-weight: bold; }}
  1787→  .summary-box .label {{ font-size: 8pt; text-transform: uppercase; color: #6b7280; }}
  1788→  .rem-list {{ margin: 8px 0; padding-left: 20px; }}
  1789→  .rem-list li {{ margin-bottom: 4px; font-size: 9pt; }}
  1790→  .footer {{ margin-top: 30px; text-align: center; font-size: 8pt; color: #9ca3af; border-top: 1px solid #e5e7eb; padding-top: 8px; }}
  1791→  @media print {{ body {{ padding: 0; }} }}
  1792→</style>
  1793→</head>
  1794→<body>
  1795→<h1>ReconPro Security Report</h1>
  1796→<div class="meta">Generated: {datetime.utcnow().strftime('%Y-%m-%d %H:%M:%S UTC')} &nbsp;|&nbsp; Target: {target or 'N/A'}</div>
  1797→
  1798→<h2>Score</h2>
  1799→<div class="gauge-label">Security Score (out of 100)</div>
  1800→<div class="gauge">[{gauge_bar}] {score_display}/100</div>
  1801→
  1802→<h2>Severity Breakdown</h2>
  1803→<div class="summary-grid">
  1804→  <div class="summary-box" style="background:#fef2f2"><div class="num" style="color:#dc2626">{counts.get('critical', 0)}</div><div class="label">Critical</div></div>
  1805→  <div class="summary-box" style="background:#fff7ed"><div class="num" style="color:#ea580c">{counts.get('high', 0)}</div><div class="label">High</div></div>
  1806→  <div class="summary-box" style="background:#fffbeb"><div class="num" style="color:#d97706">{counts.get('medium', 0)}</div><div class="label">Medium</div></div>
  1807→  <div class="summary-box" style="background:#eff6ff"><div class="num" style="color:#2563eb">{counts.get('low', 0)}</div><div class="label">Low</div></div>

  1899→def generate_report_auto(
  1900→    findings: Any,
  1901→    output_path: str,
  1902→    fmt: str = "html",
  1903→    score: Optional[float] = None,
  1904→    target: str = "",
  1905→) -> str:
  1906→    """Auto-detect report format from extension or explicit *fmt* parameter.
  1907→
  1908→    Parameters
  1909→    ----------
  1910→    findings : list of dict-like
  1911→    output_path : str
  1912→    fmt : str
  1913→        One of: html, csv, json, sarif, xml, md, markdown, pdf.
  1914→        If ``"auto"``, the format is inferred from *output_path*
  1915→        file extension.
  1916→    score : float, optional
  1917→    target : str, optional
  1918→
  1919→    Returns
  1920→    -------
  1921→    str
  1922→        Absolute path of the written file.
  1923→    """
  1924→    if fmt == "auto":
  1925→        ext = Path(output_path).suffix.lower()
  1926→        fmt = _EXT_MAP.get(ext, "html")
  1927→
  1928→    fmt = fmt.lower().strip()
  1929→    generator = _FORMAT_MAP.get(fmt)
  1930→    if generator is None:
  1931→        raise ValueError(
  1932→            f"Unknown report format '{fmt}'. "
  1933→            f"Supported: {', '.join(sorted(_FORMAT_MAP))}"
  1934→        )
  1935→
  1936→    # Build kwargs based on function signature
  1937→    import inspect
  1938→    sig = inspect.signature(generator)
  1939→    kwargs: Dict[str, Any] = {"findings": findings, "output_path": output_path}
  1940→    if "score" in sig.parameters:
  1941→        kwargs["score"] = score
  1942→    if "target" in sig.parameters:
  1943→        kwargs["target"] = target
  1944→
  1945→    return generator(**kwargs)
