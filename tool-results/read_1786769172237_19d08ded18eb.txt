   140→      }
   141→    }
   142→    return result;
   143→  }, [findings, domain]);
   144→
   145→  useEffect(() => {
   146→    blipsRef.current = blipData.map(b => ({ ...b }));
   147→    discoveredOrderRef.current = [];
   148→  }, [blipData]);
   149→
   150→  // ── Resize ──
   151→  useEffect(() => {
   152→    const el = containerRef.current;
   153→    if (!el) return;
   154→    const ro = new ResizeObserver(entries => {
   155→      for (const entry of entries) {
   156→        setDimensions({ width: entry.contentRect.width, height: entry.contentRect.height });
   157→      }
   158→    });
   159→    ro.observe(el);
   160→    return () => ro.disconnect();
   161→  }, []);
   162→
   163→  // ── Main render loop ──
   164→  useEffect(() => {
   165→    if (dimensions.width === 0 || dimensions.height === 0) return;
   166→    const canvas = canvasRef.current;
   167→    if (!canvas) return;
   168→    const ctx = canvas.getContext('2d');
   169→    if (!ctx) return;
   170→
   171→    const dpr = window.devicePixelRatio || 1;
   172→    canvas.width = dimensions.width * dpr;
   173→    canvas.height = dimensions.height * dpr;
   174→    ctx.scale(dpr, dpr);
   175→
   176→    const W = dimensions.width;
   177→    const H = dimensions.height;
   178→    const cx = W / 2;
   179→    const cy = H / 2;
   180→    const maxR = Math.min(cx, cy) - 36;
   181→    const blips = blipsRef.current;
   182→    const pings = pingsRef.current;
   183→
   184→    let lastTs = 0;
   185→
   186→    function draw(ts: number) {
   187→      const dt = lastTs ? Math.min((ts - lastTs) / 1000, 0.05) : 0.016;
   188→      lastTs = ts;
   189→      timeRef.current += dt;
   190→      const t = timeRef.current;
   191→
   192→      // Sweep: one rotation every 5 seconds
   193→      sweepRef.current = (sweepRef.current + dt * 72) % 360;
   194→      const sweepRad = sweepRef.current * DEG;
   195→
   196→      if (!ctx) return;
   197→      ctx.clearRect(0, 0, W, H);
   198→
   199→      // ──────────────────────────────────
   200→      // 1. RADAR SCREEN BACKGROUND
   201→      // ──────────────────────────────────
   202→
   203→      // Dark CRT-style background with vignette
   204→      const bgGrad = ctx.createRadialGradient(cx, cy, 0, cx, cy, maxR + 40);
   205→      bgGrad.addColorStop(0, '#040a06');
   206→      bgGrad.addColorStop(0.85, '#020804');
   207→      bgGrad.addColorStop(1, '#010302');
   208→      ctx.fillStyle = bgGrad;
   209→      ctx.beginPath();
   210→      ctx.arc(cx, cy, maxR + 6, 0, Math.PI * 2);
   211→      ctx.fill();
   212→
   213→      // Outer ring bezel
   214→      ctx.beginPath();
   215→      ctx.arc(cx, cy, maxR + 6, 0, Math.PI * 2);
   216→      ctx.strokeStyle = 'rgba(0, 255, 136, 0.12)';
   217→      ctx.lineWidth = 2;
   218→      ctx.stroke();
   219→
   220→      ctx.beginPath();
   221→      ctx.arc(cx, cy, maxR + 3, 0, Math.PI * 2);
   222→      ctx.strokeStyle = 'rgba(0, 255, 136, 0.06)';
   223→      ctx.lineWidth = 1;
   224→      ctx.stroke();
   225→
   226→      // ──────────────────────────────────
   227→      // 2. SWEEP TRAIL (phosphor afterglow)
   228→      // ──────────────────────────────────
   229→      // We draw the trail as many thin arc slices going backwards
   230→      const trailSpan = 60 * DEG; // 60 degree trail
   231→      const trailSteps = 40;
   232→      for (let i = 0; i < trailSteps; i++) {
   233→        const frac0 = i / trailSteps;
   234→        const frac1 = (i + 1) / trailSteps;
   235→        const a0 = sweepRad - trailSpan * (1 - frac0);
   236→        const a1 = sweepRad - trailSpan * (1 - frac1);
   237→        const alpha = frac1 * 0.18;
   238→        ctx.beginPath();
   239→        ctx.moveTo(cx, cy);
   240→        ctx.arc(cx, cy, maxR, a0, a1);
   241→        ctx.closePath();
   242→        ctx.fillStyle = `rgba(0, 255, 136, ${alpha})`;
   243→        ctx.fill();
   244→      }
   245→
   246→      // ──────────────────────────────────
   247→      // 3. RANGE RINGS
   248→      // ──────────────────────────────────
   249→      const rings = 5;
   250→      for (let i = 1; i <= rings; i++) {
   251→        const r = (i / rings) * maxR;
   252→        ctx.beginPath();
   253→        ctx.arc(cx, cy, r, 0, Math.PI * 2);
   254→        ctx.strokeStyle = i === rings ? 'rgba(0, 255, 136, 0.2)' : 'rgba(0, 255, 136, 0.07)';
   255→        ctx.lineWidth = i === rings ? 1.5 : 0.7;
   256→        ctx.stroke();
   257→
   258→        // Tick marks every 10 degrees on outer ring
   259→        if (i === rings) {
   260→          for (let deg = 0; deg < 360; deg += 10) {
   261→            const rad = deg * DEG;
   262→            const inner = maxR - (deg % 30 === 0 ? 8 : 4);
   263→            ctx.beginPath();
   264→            ctx.moveTo(cx + inner * Math.cos(rad), cy + inner * Math.sin(rad));
   265→            ctx.lineTo(cx + r * Math.cos(rad), cy + r * Math.sin(rad));
   266→            ctx.strokeStyle = deg % 30 === 0 ? 'rgba(0, 255, 136, 0.2)' : 'rgba(0, 255, 136, 0.08)';
   267→            ctx.lineWidth = deg % 30 === 0 ? 1 : 0.5;
   268→            ctx.stroke();
   269→          }
   270→        }
   271→      }
   272→
   273→      // ──────────────────────────────────
   274→      // 4. RANGE LABELS
   275→      // ──────────────────────────────────
   276→      ctx.fillStyle = 'rgba(0, 255, 136, 0.22)';
   277→      ctx.font = '8px "Geist Mono", monospace';
   278→      ctx.textAlign = 'left';
   279→      ctx.textBaseline = 'bottom';
   280→      for (let i = 1; i <= rings; i++) {
   281→        const r = (i / rings) * maxR;
   282→        ctx.fillText(`${i * 20}%`, cx + r + 5, cy - 3);
   283→      }
   284→
   285→      // ──────────────────────────────────
   286→      // 5. CROSS-HAIR & SECTOR LINES
   287→      // ──────────────────────────────────
   288→      ctx.strokeStyle = 'rgba(0, 255, 136, 0.1)';
   289→      ctx.lineWidth = 0.7;
   290→      // Cardinal lines
   291→      for (let a = 0; a < 360; a += 90) {
   292→        const rad = a * DEG;
   293→        ctx.beginPath();
   294→        ctx.moveTo(cx, cy);
   295→        ctx.lineTo(cx + maxR * Math.cos(rad), cy + maxR * Math.sin(rad));
   296→        ctx.stroke();
   297→      }
   298→      // Diagonal lines (faint)
   299→      ctx.strokeStyle = 'rgba(0, 255, 136, 0.04)';
   300→      for (let a = 45; a < 360; a += 90) {
   301→        const rad = a * DEG;
   302→        ctx.beginPath();
   303→        ctx.moveTo(cx, cy);
   304→        ctx.lineTo(cx + maxR * Math.cos(rad), cy + maxR * Math.sin(rad));
   305→        ctx.stroke();
   306→      }
   307→
   308→      // ──────────────────────────────────
   309→      // 6. BEARING LABELS (compass rose)
   310→      // ──────────────────────────────────
   311→      ctx.font = 'bold 11px "Geist Mono", monospace';
   312→      ctx.textBaseline = 'middle';
   313→      const labelR = maxR + 20;
   314→      const directions = [
   315→        { text: 'N', angle: 0, align: 'center' as CanvasTextAlign, base: 'alphabetic' as CanvasTextBaseline },
   316→        { text: 'E', angle: 90, align: 'left' as CanvasTextAlign, base: 'middle' as CanvasTextBaseline },
   317→        { text: 'S', angle: 180, align: 'center' as CanvasTextAlign, base: 'hanging' as CanvasTextBaseline },
   318→        { text: 'W', angle: 270, align: 'right' as CanvasTextAlign, base: 'middle' as CanvasTextBaseline },
   319→      ];
   320→      for (const d of directions) {
   321→        const rad = d.angle * DEG;
   322→        const lx = cx + labelR * Math.cos(rad);
   323→        const ly = cy + labelR * Math.sin(rad);
   324→        ctx.textAlign = d.align;
   325→        ctx.textBaseline = d.base;
   326→        ctx.fillStyle = d.text === 'N' ? '#00ff88' : 'rgba(0, 255, 136, 0.45)';
   327→        ctx.fillText(d.text, lx, ly);
   328→      }
   329→
   330→      // Degree labels around the outside
   331→      ctx.font = '7px "Geist Mono", monospace';
   332→      ctx.fillStyle = 'rgba(0, 255, 136, 0.18)';
   333→      ctx.textBaseline = 'middle';
   334→      for (let deg = 30; deg < 360; deg += 30) {
   335→        if (deg % 90 === 0) continue;
   336→        const rad = deg * DEG;
   337→        const lx = cx + (maxR + 18) * Math.cos(rad);
   338→        const ly = cy + (maxR + 18) * Math.sin(rad);
   339→        ctx.textAlign = Math.abs(Math.cos(rad)) < 0.01 ? 'center' : Math.cos(rad) > 0 ? 'left' : 'right';
   340→        ctx.fillText(`${String(deg).padStart(3, '0')}`, lx, ly);
   341→      }
   342→
   343→      // ──────────────────────────────────
   344→      // 7. SWEEP BEAM LINE
   345→      // ──────────────────────────────────
   346→      const beamEndX = cx + maxR * Math.cos(sweepRad);
   347→      const beamEndY = cy + maxR * Math.sin(sweepRad);
   348→
   349→      // Beam glow (wide, faint)
   350→      const beamGlow = ctx.createLinearGradient(cx, cy, beamEndX, beamEndY);
   351→      beamGlow.addColorStop(0, 'rgba(0, 255, 136, 0.0)');
   352→      beamGlow.addColorStop(0.3, 'rgba(0, 255, 136, 0.08)');
   353→      beamGlow.addColorStop(1, 'rgba(0, 255, 136, 0.02)');
   354→      ctx.beginPath();
   355→      ctx.moveTo(cx, cy);
   356→      ctx.lineTo(beamEndX, beamEndY);
   357→      ctx.strokeStyle = beamGlow;
   358→      ctx.lineWidth = 12;
   359→      ctx.stroke();
   360→
   361→      // Main beam (sharp)
   362→      const beamSharp = ctx.createLinearGradient(cx, cy, beamEndX, beamEndY);
   363→      beamSharp.addColorStop(0, 'rgba(0, 255, 136, 0.8)');
   364→      beamSharp.addColorStop(0.4, 'rgba(0, 255, 136, 0.45)');
   365→      beamSharp.addColorStop(1, 'rgba(0, 255, 136, 0.1)');
   366→      ctx.beginPath();
   367→      ctx.moveTo(cx, cy);
   368→      ctx.lineTo(beamEndX, beamEndY);
   369→      ctx.strokeStyle = beamSharp;
   370→      ctx.lineWidth = 1.5;
   371→      ctx.stroke();
   372→
   373→      // Beam tip glow
   374→      const tipGlow = ctx.createRadialGradient(beamEndX, beamEndY, 0, beamEndX, beamEndY, 20);
   375→      tipGlow.addColorStop(0, 'rgba(0, 255, 136, 0.25)');
   376→      tipGlow.addColorStop(1, 'transparent');
   377→      ctx.beginPath();
   378→      ctx.arc(beamEndX, beamEndY, 20, 0, Math.PI * 2);
   379→      ctx.fillStyle = tipGlow;
   380→      ctx.fill();
   381→
   382→      // ──────────────────────────────────
   383→      // 8. BLIPS (radar contacts)
   384→      // ──────────────────────────────────
   385→      for (const blip of blips) {
   386→        const blipRad = blip.angle * DEG;
   387→        const bx = cx + blip.distance * maxR * Math.cos(blipRad);
   388→        const by = cy + blip.distance * maxR * Math.sin(blipRad);
   389→
   390→        // Angular diff between sweep and blip
   391→        let angleDiff = sweepRef.current - blip.angle;
   392→        if (angleDiff < 0) angleDiff += 360;
   393→        if (angleDiff > 360) angleDiff -= 360;
   394→
   395→        // Blip illuminates when sweep passes (within 25 degrees behind)
   396→        const hitWindow = 25;
   397→        if (angleDiff < hitWindow && angleDiff >= 0) {
   398→          const wasDiscovered = blip.discovered;
   399→          blip.brightness = 1 - (angleDiff / hitWindow);
   400→          blip.discovered = true;
   401→
   402→          // First discovery: trigger ping + add to order
   403→          if (!wasDiscovered) {
   404→            blip.discoveredAt = t;
   405→            blip.pingPhase = 1;
   406→            discoveredOrderRef.current.push(blip.id);
   407→            const color = SEV_COLORS[blip.severity] || '#00ff88';
   408→            pings.push({ x: bx, y: by, r: 0, alpha: 0.8, color });
   409→          }
   410→        } else {
   411→          // Slow decay
   412→          blip.brightness = Math.max(0, blip.brightness - dt * 0.25);
   413→        }
   414→
   415→        // Ping ripple animation
   416→        if (blip.pingPhase > 0) {
   417→          blip.pingPhase = Math.max(0, blip.pingPhase - dt * 1.2);
   418→          const pingR = (1 - blip.pingPhase) * 30;
   419→          const pingAlpha = blip.pingPhase * 0.5;
   420→          const color = SEV_COLORS[blip.severity] || '#00ff88';
   421→          const pr = parseInt(color.slice(1, 3), 16);
   422→          const pg = parseInt(color.slice(3, 5), 16);
   423→          const pb = parseInt(color.slice(5, 7), 16);
   424→
   425→          ctx.beginPath();
   426→          ctx.arc(bx, by, pingR, 0, Math.PI * 2);
   427→          ctx.strokeStyle = `rgba(${pr},${pg},${pb},${pingAlpha})`;
   428→          ctx.lineWidth = 1.5;
   429→          ctx.stroke();
   430→
   431→          // Second ring
   432→          if (blip.pingPhase < 0.7) {
   433→            const ping2R = (1 - blip.pingPhase * 1.3) * 25;
   434→            if (ping2R > 0) {
   435→              ctx.beginPath();
   436→              ctx.arc(bx, by, ping2R, 0, Math.PI * 2);
   437→              ctx.strokeStyle = `rgba(${pr},${pg},${pb},${pingAlpha * 0.4})`;
   438→              ctx.lineWidth = 1;
   439→              ctx.stroke();
   440→            }
   441→          }
   442→        }
   443→
   444→        // Only draw if discovered
   445→        if (!blip.discovered) continue;
   446→
   447→        const alpha = Math.max(0.12, blip.brightness);
   448→        const color = SEV_COLORS[blip.severity] || '#00ff88';
   449→        const cr = parseInt(color.slice(1, 3), 16);
   450→        const cg = parseInt(color.slice(3, 5), 16);
   451→        const cb = parseInt(color.slice(5, 7), 16);
   452→        const r = blip.size * (0.7 + blip.brightness * 0.5);
   453→
   454→        // Signal noise: slight random position jitter for low-signal blips
   455→        const noiseAmp = (1 - blip.signalStrength) * 1.5;
   456→        const jx = blip.brightness > 0.3 ? 0 : (Math.sin(t * 20 + blip.angle) * noiseAmp);
   457→        const jy = blip.brightness > 0.3 ? 0 : (Math.cos(t * 18 + blip.angle * 2) * noiseAmp);
   458→
   459→        const fx = bx + jx;
   460→        const fy = by + jy;
   461→
   462→        // Outer glow (when bright)
   463→        if (blip.brightness > 0.2) {
   464→          const glowR = r + 12 + blip.brightness * 8;
   465→          const glowGrad = ctx.createRadialGradient(fx, fy, 0, fx, fy, glowR);
   466→          glowGrad.addColorStop(0, `rgba(${cr},${cg},${cb},${blip.brightness * 0.3})`);
   467→          glowGrad.addColorStop(1, 'transparent');
   468→          ctx.beginPath();
   469→          ctx.arc(fx, fy, glowR, 0, Math.PI * 2);
   470→          ctx.fillStyle = glowGrad;
   471→          ctx.fill();
   472→        }
   473→
   474→        // Blip body
   475→        ctx.beginPath();
   476→        ctx.arc(fx, fy, r, 0, Math.PI * 2);
   477→        ctx.fillStyle = `rgba(${cr},${cg},${cb},${alpha})`;
   478→        ctx.fill();
   479→
   480→        // White-hot core (bright blips)
   481→        if (blip.brightness > 0.4) {
   482→          ctx.beginPath();
   483→          ctx.arc(fx, fy, r * 0.35, 0, Math.PI * 2);
   484→          ctx.fillStyle = `rgba(255,255,255,${blip.brightness * 0.7})`;
   485→          ctx.fill();
   486→        }
   487→
   488→        // Critical: pulsing warning ring
   489→        if (blip.severity === 'critical' && blip.brightness > 0.05) {
   490→          const pulseR = r + 8 + Math.sin(t * 5) * 4;
   491→          ctx.beginPath();
   492→          ctx.arc(fx, fy, pulseR, 0, Math.PI * 2);
   493→          ctx.strokeStyle = `rgba(239,68,68,${Math.max(0.08, blip.brightness * 0.5)})`;
   494→          ctx.lineWidth = 1;
   495→          ctx.stroke();
   496→
   497→          // Second pulse ring offset
   498→          const pulse2R = r + 12 + Math.sin(t * 5 + 1) * 3;
   499→          ctx.beginPath();
   500→          ctx.arc(fx, fy, pulse2R, 0, Math.PI * 2);
   501→          ctx.strokeStyle = `rgba(239,68,68,${Math.max(0.04, blip.brightness * 0.25)})`;
   502→          ctx.lineWidth = 0.5;
   503→          ctx.stroke();
   504→        }
   505→
   506→        // High severity: single pulse
   507→        if (blip.severity === 'high' && blip.brightness > 0.1) {
   508→          const hPulseR = r + 6 + Math.sin(t * 3.5) * 3;
   509→          ctx.beginPath();
   510→          ctx.arc(fx, fy, hPulseR, 0, Math.PI * 2);
   511→          ctx.strokeStyle = `rgba(251,191,36,${Math.max(0.05, blip.brightness * 0.3)})`;
   512→          ctx.lineWidth = 0.7;
   513→          ctx.stroke();
   514→        }
   515→
   516→        // Label (visible when bright enough)
   517→        if (blip.brightness > 0.35) {
   518→          const labelAlpha = (blip.brightness - 0.35) * 1.2;
   519→          ctx.fillStyle = `rgba(230,237,243,${Math.min(1, labelAlpha)})`;
   520→          ctx.font = '8px "Geist Mono", monospace';
   521→          ctx.textAlign = 'left';
   522→          ctx.textBaseline = 'middle';
   523→
   524→          // Background for readability
   525→          const labelText = blip.label;
   526→          const metrics = ctx.measureText(labelText);
   527→          ctx.fillStyle = `rgba(4,10,6,${labelAlpha * 0.6})`;
   528→          ctx.fillRect(fx + r + 5, fy - 6, metrics.width + 4, 12);
   529→
   530→          ctx.fillStyle = `rgba(230,237,243,${Math.min(1, labelAlpha * 0.9)})`;
   531→          ctx.fillText(labelText, fx + r + 7, fy);
   532→        }
   533→
   534→        // Connecting line to center for very bright blips
   535→        if (blip.brightness > 0.6) {
   536→          const lineAlpha = (blip.brightness - 0.6) * 0.5;
   537→          ctx.beginPath();
   538→          ctx.moveTo(cx, cy);
   539→          ctx.lineTo(fx, fy);
   540→          ctx.strokeStyle = `rgba(${cr},${cg},${cb},${lineAlpha})`;
   541→          ctx.lineWidth = 0.4;
   542→          ctx.setLineDash([2, 6]);
   543→          ctx.lineDashOffset = -t * 30;
   544→          ctx.stroke();
   545→          ctx.setLineDash([]);
   546→          ctx.lineDashOffset = 0;
   547→        }
   548→      }
   549→
   550→      // ──────────────────────────────────
   551→      // 9. PING RIPPLES (global)
   552→      // ──────────────────────────────────
   553→      for (let i = pings.length - 1; i >= 0; i--) {
   554→        const p = pings[i];
   555→        p.r += dt * 80;
   556→        p.alpha -= dt * 0.6;
   557→        if (p.alpha <= 0) { pings.splice(i, 1); continue; }
   558→        const pr = parseInt(p.color.slice(1, 3), 16);
   559→        const pg = parseInt(p.color.slice(3, 5), 16);
   560→        const pb = parseInt(p.color.slice(5, 7), 16);
   561→        ctx.beginPath();
   562→        ctx.arc(p.x, p.y, p.r, 0, Math.PI * 2);
   563→        ctx.strokeStyle = `rgba(${pr},${pg},${pb},${p.alpha * 0.3})`;
   564→        ctx.lineWidth = 1;
   565→        ctx.stroke();
   566→      }
   567→
   568→      // ──────────────────────────────────
   569→      // 10. CENTER POINT & DOMAIN LABEL
   570→      // ──────────────────────────────────
   571→      const centerGlow = ctx.createRadialGradient(cx, cy, 0, cx, cy, 25);
   572→      centerGlow.addColorStop(0, 'rgba(0, 255, 136, 0.2)');
   573→      centerGlow.addColorStop(1, 'transparent');
   574→      ctx.beginPath();
   575→      ctx.arc(cx, cy, 25, 0, Math.PI * 2);
   576→      ctx.fillStyle = centerGlow;
   577→      ctx.fill();
   578→
   579→      // Center cross
   580→      ctx.strokeStyle = 'rgba(0, 255, 136, 0.6)';
   581→      ctx.lineWidth = 1;
   582→      ctx.beginPath(); ctx.moveTo(cx - 6, cy); ctx.lineTo(cx + 6, cy); ctx.stroke();
   583→      ctx.beginPath(); ctx.moveTo(cx, cy - 6); ctx.lineTo(cx, cy + 6); ctx.stroke();
   584→
   585→      // Center dot
   586→      ctx.beginPath();
   587→      ctx.arc(cx, cy, 2, 0, Math.PI * 2);
   588→      ctx.fillStyle = '#00ff88';
   589→      ctx.fill();
   590→
   591→      // Domain label below center
   592→      ctx.font = 'bold 9px "Geist Mono", monospace';
   593→      ctx.textAlign = 'center';
   594→      ctx.textBaseline = 'top';
   595→      ctx.fillStyle = 'rgba(0, 255, 136, 0.5)';
   596→      ctx.fillText(domain.toUpperCase(), cx, cy + 12);
   597→
   598→      // ──────────────────────────────────
   599→      // 11. SCANNING INDICATOR
   600→      // ──────────────────────────────────
   601→      if (isScanning) {
   602→        const scanPulse = Math.sin(t * 4) * 0.5 + 0.5;
   603→        ctx.beginPath();
   604→        ctx.arc(cx, cy, maxR + 8, 0, Math.PI * 2);
   605→        ctx.strokeStyle = `rgba(0, 255, 136, ${0.08 + scanPulse * 0.08})`;
   606→        ctx.lineWidth = 1.5;
   607→        ctx.setLineDash([6, 6]);
   608→        ctx.lineDashOffset = -t * 40;
   609→        ctx.stroke();
   610→        ctx.setLineDash([]);
   611→        ctx.lineDashOffset = 0;
   612→
   613→        // "SCANNING" text flash
   614→        if (Math.sin(t * 6) > 0) {
   615→          ctx.font = 'bold 10px "Geist Mono", monospace';
   616→          ctx.textAlign = 'center';
   617→          ctx.textBaseline = 'middle';
   618→          ctx.fillStyle = `rgba(0, 255, 136, ${0.4 + scanPulse * 0.3})`;
   619→          ctx.fillText('SCANNING', cx, cy + maxR + 28);
   620→        }
   621→      }
   622→
   623→      // ──────────────────────────────────
   624→      // 12. HUD OVERLAY
   625→      // ──────────────────────────────────
   626→

   640→

   676→      // 13. NOISE / SCAN LINES (CRT effect)

   681→        ctx.fillRect(0, y, W, 1);
   682→      }
   683→

   685→      if (Math.random() < 0.02) {
   686→        const noiseY = Math.random() * H;
   687→        const noiseH = 1 + Math.random() * 2;

   689→        ctx.fillRect(0, noiseY, W, noiseH);
   690→      }
   691→
   692→      animRef.current = requestAnimationFrame(draw);
   693→    }

   695→    function safeDraw(ts: number) {

   697→    }

   701→  }, [dimensions, domain, isScanning]);

   708→    const my = e.clientY - rect.top;

   710→    const cy = dimensions.height / 2;

   712→

   715→      if (!blip.discovered) continue;

   717→      const bx = cx + blip.distance * maxR * Math.cos(rad);
   718→      const by = cy + blip.distance * maxR * Math.sin(rad);
   719→      const dx = mx - bx;
   720→      const dy = my - by;
   721→      if (dx * dx + dy * dy < 120) {
   722→        found = blip;
   723→        setTooltipPos({ x: e.clientX - rect.left, y: e.clientY - rect.top });
   724→        break;
   725→      }
   726→    }
