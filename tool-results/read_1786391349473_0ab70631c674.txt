   153→def dark_table(data, col_widths=None):
   154→    """Create a dark-themed table."""
   155→    if not col_widths:
   156→        col_widths = [None] * len(data[0])
   157→
   158→    t = Table(data, colWidths=col_widths, repeatRows=1)
   159→    style_cmds = [
   160→        ('BACKGROUND', (0, 0), (-1, 0), C_SURFACE),
   161→        ('TEXTCOLOR', (0, 0), (-1, 0), C_WHITE),
   162→        ('FONTNAME', (0, 0), (-1, 0), 'DejaVuSans-Bold'),
   163→        ('FONTSIZE', (0, 0), (-1, 0), 8),
   164→        ('FONTNAME', (0, 1), (-1, -1), 'DejaVuSans'),
   165→        ('FONTSIZE', (0, 1), (-1, -1), 8),
   166→        ('TEXTCOLOR', (0, 1), (-1, -1), C_TEXT),
   167→        ('ALIGN', (0, 0), (-1, -1), 'LEFT'),
   168→        ('VALIGN', (0, 0), (-1, -1), 'MIDDLE'),
   169→        ('GRID', (0, 0), (-1, -1), 0.5, C_BORDER),
   170→        ('TOPPADDING', (0, 0), (-1, -1), 2 * mm),
   171→        ('BOTTOMPADDING', (0, 0), (-1, -1), 2 * mm),
   172→        ('LEFTPADDING', (0, 0), (-1, -1), 3 * mm),
   173→        ('RIGHTPADDING', (0, 0), (-1, -1), 3 * mm),
   174→    ]
   175→    for i in range(1, len(data)):
   176→        if i % 2 == 0:
   177→            style_cmds.append(('BACKGROUND', (0, i), (-1, i), HexColor('#162032')))
   178→    t.setStyle(TableStyle(style_cmds))
   179→    return t
