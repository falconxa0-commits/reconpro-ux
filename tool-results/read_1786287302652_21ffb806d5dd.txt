   459→def _stage_16_ai_endpoint_discovery(base_url: str, timeout: int = 8,
   460→                                      verify_tls: bool = True) -> List[Finding]:
   461→    """v9.1.0 Stage 16: AI-specific endpoint discovery and vendor fingerprinting."""
   462→    findings: List[Finding] = []
   463→    host = base_url.replace("https://", "").replace("http://", "").split("/")[0]
   464→
   465→    try:
   466→        from ..ai_red_team import AIEndpointDiscovery, AIVendorFingerprinter, SecretExtractor
   467→        from ..ai_cve_db import AICVEDatabase
   468→
   469→        # Discover AI endpoints
   470→        discovery = AIEndpointDiscovery()
   471→        endpoints = discovery.discover(base_url, timeout=timeout, verify_tls=verify_tls)
   472→        if endpoints:
   473→            findings.append(Finding(
   474→                title="AI endpoints: {} discovered".format(len(endpoints)),
   475→                severity="medium", category="ai_endpoint_discovery",
   476→                module="gorgon",
   477→                description="AI-specific endpoints found: {}".format(", ".join(e["endpoint"] for e in endpoints[:8])),
   478→                evidence="{} AI endpoints discovered".format(len(endpoints)),
   479→                asset=host, points_deducted=4,
   480→            ))
   481→
   482→        # Vendor fingerprinting
   483→        fp = AIVendorFingerprinter()
   484→        vendors = fp.fingerprint(base_url, timeout=timeout, verify_tls=verify_tls)
   485→        for v in vendors[:3]:
   486→            findings.append(Finding(
   487→                title="AI vendor detected: {}".format(v["vendor"]),
   488→                severity="medium", category="ai_vendor_fingerprint",
   489→                module="gorgon",
   490→                description="AI vendor {} detected (score: {})".format(v["vendor"], v["score"]),
   491→                evidence="Evidence: {}".format("; ".join(v.get("evidence", []))),
   492→                asset=host, points_deducted=3,
   493→            ))
   494→
   495→        # AI CVE matching
   496→        db = AICVEDatabase()
   497→        if vendors:
   498→            all_cves = []
   499→            for v in vendors:
   500→                cves = db.search(product=v["vendor"])
   501→                all_cves.extend(cves)
   502→            if all_cves:
   503→                for cve in all_cves[:3]:
   504→                    findings.append(Finding(
   505→                        title="AI CVE: {} (CVSS {})".format(cve["cve"], cve["cvss"]),
   506→                        severity="critical" if cve["cvss"] >= 9.0 else "high",
   507→                        category="ai_cve",
   508→                        module="gorgon",
   509→                        description="{}: {}".format(cve["cve"], cve["description"][:100]),
   510→                        evidence="Product: {}, Type: {}, CVSS: {}".format(cve["product"], cve["type"], cve["cvss"]),
   511→                        asset=host, points_deducted=12 if cve["cvss"] >= 9.0 else 8,
   512→                        remediation="Upgrade {} to {} or later.".format(cve["product"], cve.get("fix_version", "latest")),
   513→                    ))
   514→
   515→        # Secret extraction from root response
   516→        extractor = SecretExtractor()
   517→        try:
   518→            resp = http_probe(base_url, timeout=timeout, verify_tls=verify_tls)
   519→            secrets = extractor.extract_from_response(resp)
   520→            for s in secrets[:5]:
   521→                findings.append(Finding(
   522→                    title="Secret exposed: {}".format(s["type"]),
   523→                    severity=s["severity"], category="secret_exposure",
   524→                    module="gorgon",
   525→                    description="{} found in response: {}".format(s["type"], s["match"][:60]),
   526→                    evidence="Pattern: {}".format(s.get("pattern", "?")),
   527→                    asset=host, points_deducted=12 if s["severity"] == "critical" else 6,
   528→                    remediation="Remove exposed secrets from responses. Rotate compromised credentials.",
   529→                ))
   530→        except Exception:
   531→            pass
   532→    except Exception:
   533→        pass
   534→
   535→    return findings
