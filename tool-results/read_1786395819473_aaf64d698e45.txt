    93→    93→
    94→    94→class EntropyAnalyzer:
    95→    95→    """Calculate Shannon entropy of byte or string data.
    96→    96→
    97→    97→    High entropy (> 4.0 bits/byte for ASCII) suggests encrypted or compressed
    98→    98→    payloads -- a strong tunneling indicator.
    99→    99→    """
   100→   100→
   101→   101→    @staticmethod
   102→   102→    def shannon_entropy(data: bytes | str) -> float:
   103→   103→        """Calculate Shannon entropy in bits per byte (0-8 range).
   104→   104→
   105→   105→        Args:
   106→   106→            data: Input bytes or string to analyse.
   107→   107→
   108→   108→        Returns:
   109→   109→            Entropy value between 0.0 (uniform/low randomness) and 8.0
   110→   110→            (maximum randomness).
   111→   111→        """
   112→   112→        if isinstance(data, str):
   113→   113→            data = data.encode("utf-8", errors="replace")
   114→   114→        if not data:
   115→   115→            return 0.0
   116→   116→
   117→   117→        length = len(data)
   118→   118→        freq: Dict[int, int] = {}
   119→   119→        for byte in data:
   120→   120→            freq[byte] = freq.get(byte, 0) + 1
   121→   121→
   122→   122→        entropy = 0.0
   123→   123→        for count in freq.values():
   124→   124→            if count > 0:
   125→   125→                p = count / length
   126→   126→                if p > 0:
   127→   127→                    entropy -= p * math.log2(p)
   128→   128→        return round(entropy, 4)
   129→   129→
   130→   130→    @staticmethod
   131→   131→    def is_high_entropy(data: bytes | str, threshold: float = 4.0) -> bool:
   132→   132→        """Return True if *data* has entropy above *threshold*."""
   133→   133→        return EntropyAnalyzer.shannon_entropy(data) >= threshold
   134→   134→
   135→   135→    @staticmethod
   136→   136→    def byte_distribution(data: bytes | str) -> Dict[int, float]:
   137→   137→        """Return a dict mapping byte value (0-255) to its frequency."""
   138→   138→        if isinstance(data, str):
   139→   139→            data = data.encode("utf-8", errors="replace")
   140→   140→        if not data:
   141→   141→            return {}
   142→   142→        length = len(data)
   143→   143→        dist: Dict[int, float] = {}
   144→   144→        for byte in data:
   145→   145→            dist[byte] = dist.get(byte, 0.0) + 1.0 / length
   146→   146→        return dist

   149→   149→# ---------------------------------------------------------------------------
   150→   150→# DNSEntropyChecker
   151→   151→# ---------------------------------------------------------------------------
   152→   152→
   153→   153→class DNSEntropyChecker:
   154→   154→    """Specialised analyser for detecting high-entropy DNS labels.
   155→   155→
   156→   156→    DNS tunneling tools encode data into subdomain labels.  These encoded
   157→   157→    labels have significantly higher entropy than normal domain names.
   158→   158→    """
   159→   159→
   160→   160→    # Normal English domain labels rarely exceed this entropy
   161→   161→    LABEL_ENTROPY_THRESHOLD = 3.5
   162→   162→    # Very long labels are themselves suspicious
   163→   163→    MAX_NORMAL_LABEL_LEN = 30
   164→   164→    # Base64 character set
   165→   165→    BASE64_CHARS = set("ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/=")
   166→   166→    HEX_CHARS = set("0123456789abcdef")
   167→   167→
   168→   168→    def __init__(self, entropy_analyzer: Optional[EntropyAnalyzer] = None) -> None:
   169→   169→        self.entropy = entropy_analyzer or EntropyAnalyzer()
   170→   170→
   171→   171→    def check_label(self, label: str) -> Dict[str, Any]:
   172→   172→        """Analyse a single DNS label for tunneling indicators.
   173→   173→
   174→   174→        Returns a dict with keys: ``entropy``, ``is_high_entropy``,
   175→   175→        ``is_base64_like``, ``is_hex_like``, ``label_length``, ``suspicious``.
   176→   176→        """
   177→   177→        label_lower = label.lower()
   178→   178→        label_len = len(label_lower)
   179→   179→        ent = self.entropy.shannon_entropy(label_lower)
   180→   180→
   181→   181→        is_base64 = (
   182→   182→            label_len > 12
   183→   183→            and all(c in self.BASE64_CHARS for c in label_lower)
   184→   184→        )
   185→   185→        is_hex = (
   186→   186→            label_len > 10
   187→   187→            and all(c in self.HEX_CHARS for c in label_lower)
   188→   188→        )
   189→   189→
   190→   190→        suspicious = (
   191→   191→            ent >= self.LABEL_ENTROPY_THRESHOLD
   192→   192→            or label_len > self.MAX_NORMAL_LABEL_LEN
   193→   193→            or is_base64
   194→   194→            or is_hex
   195→   195→        )
   196→   196→
   197→   197→        return {
   198→   198→            "label": label,
   199→   199→            "entropy": ent,
   200→   200→            "is_high_entropy": ent >= self.LABEL_ENTROPY_THRESHOLD,
   201→   201→            "is_base64_like": is_base64,
   202→   202→            "is_hex_like": is_hex,
   203→   203→            "label_length": label_len,
   204→   204→            "suspicious": suspicious,
   205→   205→        }
   206→   206→
   207→   207→    def check_fqdn(self, fqdn: str) -> Dict[str, Any]:
   208→   208→        """Analyse all labels in a fully-qualified domain name.
   209→   209→
   210→   210→        Returns:
   211→   211→            ``{"fqdn": ..., "labels": [...], "suspicious_labels": [...], "overall_suspicious": bool}``
   212→   212→        """
   213→   213→        labels = fqdn.rstrip(".").split(".")
   214→   214→        results = [self.check_label(lbl) for lbl in labels]
   215→   215→        suspicious = [r for r in results if r["suspicious"]]
   216→   216→        return {
   217→   217→            "fqdn": fqdn,
   218→   218→            "labels": results,
   219→   219→            "suspicious_labels": suspicious,
   220→   220→            "overall_suspicious": len(suspicious) > 0,
   221→   221→        }
   222→   222→

   227→   227→
   228→   228→class TunnelDetector:
   229→   229→    """Main tunnel detection orchestrator.
   230→   230→
   231→   231→    Probes the target host for indicators of various tunnel types and
   232→   232→    returns a list of :class:`TunnelIndicator` results.
   233→   233→    """
   234→   234→
   235→   235→    DEFAULT_TIMEOUT: int = 8
   236→   236→
   237→   237→    def __init__(self) -> None:
   238→   238→        self.entropy_analyzer = EntropyAnalyzer()
   239→   239→        self.dns_checker = DNSEntropyChecker(self.entropy_analyzer)
   240→   240→
   241→   241→    def detect_all(
   242→   242→        self,
   243→   243→        host: str,
   244→   244→        base_url: Optional[str] = None,
   245→   245→        timeout: int = DEFAULT_TIMEOUT,
   246→   246→    ) -> List[TunnelIndicator]:
   247→   247→        """Run all tunnel detectors against *host*.
   248→   248→
   249→   249→        Args:
   250→   250→            host:     Target hostname or IP address.
   251→   251→            base_url: Optional URL for HTTP/HTTPS tunnel checks.
   252→   252→            timeout:  Network timeout in seconds.
   253→   253→
   254→   254→        Returns:
   255→   255→            List of :class:`TunnelIndicator` sorted by confidence descending.
   256→   256→        """
   257→   257→        indicators: List[TunnelIndicator] = []
   258→   258→
   259→   259→        # Resolve target to IP
   260→   260→        dest_ip = self._resolve_host(host)
   261→   261→
   262→   262→        # 1. DNS tunnel detection
   263→   263→        logger.info("[TunnelDetect] Checking DNS tunnel indicators for %s", host)
   264→   264→        indicators.extend(self.detect_dns_tunnel(host, timeout=timeout))
   265→   265→
   266→   266→        # 2. ICMP tunnel detection
   267→   267→        logger.info("[TunnelDetect] Checking ICMP tunnel indicators for %s", host)
   268→   268→        indicators.extend(self.detect_icmp_tunnel(host, timeout=timeout))
   269→   269→
   270→   270→        # 3. IPv6 tunnel detection
   271→   271→        logger.info("[TunnelDetect] Checking IPv6 tunnel indicators for %s", host)
   272→   272→        indicators.extend(self.detect_ipv6_tunnel(host, timeout=timeout))
   273→   273→
   274→   274→        # 4. GRE tunnel detection
   275→   275→        logger.info("[TunnelDetect] Checking GRE tunnel indicators for %s", host)
   276→   276→        indicators.extend(self.detect_gre_tunnel(host, timeout=timeout))
   277→   277→
   278→   278→        # 5. SSH tunnel detection
   279→   279→        logger.info("[TunnelDetect] Checking SSH tunnel indicators for %s", host)
   280→   280→        indicators.extend(self.detect_ssh_tunnel(host, timeout=timeout, dest_ip=dest_ip))
   281→   281→
   282→   282→        # 6. HTTP CONNECT tunnel detection
   283→   283→        if base_url:
   284→   284→            logger.info("[TunnelDetect] Checking HTTP CONNECT tunnel indicators for %s", base_url)
   285→   285→            indicators.extend(self.detect_http_tunnel(base_url, timeout=timeout))
   286→   286→
   287→   287→        # Sort by confidence descending
   288→   288→        indicators.sort(key=lambda i: i.confidence, reverse=True)
   289→   289→        return indicators
   290→   290→
   291→   291→    # -- DNS tunnel --------------------------------------------------------
   292→   292→
   293→   293→    def detect_dns_tunnel(
   294→   294→        self, host: str, timeout: int = 8
   295→   295→    ) -> List[TunnelIndicator]:
   296→   296→        """Analyse DNS response patterns for tunneling indicators.
   297→   297→
   298→   298→        Checks:
   299→   299→        - Unusually long TXT records
   300→   300→        - Entropy of subdomain labels
   301→   301→        - Base64/hex encoded data in DNS responses
   302→   302→        - Frequency of queryable subdomains
   303→   303→        """
   304→   304→        indicators: List[TunnelIndicator] = []
   305→   305→
   306→   306→        # 1. TXT record analysis
   307→   307→        txt_records = self._query_txt_records(host, timeout)
   308→   308→        for txt in txt_records:
   309→   309→            ent = self.entropy_analyzer.shannon_entropy(txt)
   310→   310→            details: Dict[str, Any] = {
   311→   311→                "record_length": len(txt),
   312→   312→                "entropy": ent,
   313→   313→            }
   314→   314→            if len(txt) > 200:
   315→   315→                indicators.append(
   316→   316→                    TunnelIndicator(
   317→   317→                        tunnel_type=TunnelType.DNS_TUNNEL,
   318→   318→                        confidence=min(0.5 + (len(txt) - 200) / 1000, 0.95),
   319→   319→                        evidence=(
   320→   320→                            f"Unusually long TXT record ({len(txt)} bytes, "
   321→   321→                            f"entropy={ent:.2f}): {txt[:80]}..."
   322→   322→                        ),
   323→   323→                        destination_ip=host,
   324→   324→                        protocol_details=details,
   325→   325→                    )
   326→   326→                )
   327→   327→            if ent > 4.0:
   328→   328→                indicators.append(
   329→   329→                    TunnelIndicator(
   330→   330→                        tunnel_type=TunnelType.DNS_TUNNEL,
   331→   331→                        confidence=min(0.4 + (ent - 4.0) * 0.15, 0.9),
   332→   332→                        evidence=(
   333→   333→                            f"High-entropy TXT record (entropy={ent:.2f}): {txt[:80]}..."
   334→   334→                        ),
   335→   335→                        destination_ip=host,
   336→   336→                        protocol_details=details,
   337→   337→                    )
   338→   338→                )
   339→   339→
   340→   340→        # 2. Entropy check on the host itself
   341→   341→        label_analysis = self.dns_checker.check_fqdn(host)
   342→   342→        if label_analysis["overall_suspicious"]:
   343→   343→            sus_labels = label_analysis["suspicious_labels"]
   344→   344→            evidence_parts = [
   345→   345→                f"label '{s['label']}' (len={s['label_length']}, ent={s['entropy']:.2f})"
   346→   346→                for s in sus_labels[:3]
   347→   347→            ]
   348→   348→            indicators.append(
   349→   349→                TunnelIndicator(
   350→   350→                    tunnel_type=TunnelType.DNS_TUNNEL,
   351→   351→                    confidence=0.5,
   352→   352→                    evidence=f"Suspicious DNS labels: {'; '.join(evidence_parts)}",
   353→   353→                    destination_ip=host,
   354→   354→                    protocol_details={"label_analysis": sus_labels},
   355→   355→                )
   356→   356→            )
   357→   357→
   358→   358→        # 3. Check for base64-like subdomain acceptance (tunneling surface)
   359→   359→        test_subdomains = self._probe_dns_subdomains(host, timeout)
   360→   360→        if test_subdomains:
   361→   361→            indicators.append(
   362→   362→                TunnelIndicator(
   363→   363→                    tunnel_type=TunnelType.DNS_TUNNEL,
   364→   364→                    confidence=0.6,
   365→   365→                    evidence=(
   366→   366→                        f"Host accepts base64-like subdomains: {test_subdomains[:3]}"
   367→   367→                    ),
   368→   368→                    destination_ip=host,
   369→   369→                    protocol_details={"resolvable_subdomains": test_subdomains},
   370→   370→                )
   371→   371→            )
   372→   372→
   373→   373→        # 4. Check ANY response size via dig
   374→   374→        any_output = self._dig_any(host, timeout)
   375→   375→        if any_output and len(any_output) > 512:
   376→   376→            ent = self.entropy_analyzer.shannon_entropy(any_output)
   377→   377→            indicators.append(
   378→   378→                TunnelIndicator(
   379→   379→                    tunnel_type=TunnelType.DNS_TUNNEL,
   380→   380→                    confidence=min(0.3 + (len(any_output) - 512) / 2000, 0.85),
   381→   381→                    evidence=(
   382→   382→                        f"Large DNS ANY response ({len(any_output)} bytes, "
   383→   383→                        f"entropy={ent:.2f})"
   384→   384→                    ),
   385→   385→                    destination_ip=host,
   386→   386→                    protocol_details={"response_size": len(any_output), "entropy": ent},
   387→   387→                )
   388→   388→            )
   389→   389→
   390→   390→        return indicators
   391→   391→
   392→   392→    # -- ICMP tunnel -------------------------------------------------------
   393→   393→
   394→   394→    def detect_icmp_tunnel(
   395→   395→        self, host: str, timeout: int = 8
   396→   396→    ) -> List[TunnelIndicator]:
   397→   397→        """Send pings and analyse response patterns for tunneling.
   398→   398→
   399→   399→        Checks:
   400→   400→        - Payload size in responses
   401→   401→        - Response timing consistency (low jitter = stable tunnel)
   402→   402→        - Payload data content (non-standard padding)
   403→   403→        - Whether large payloads are accepted
   404→   404→        """
   405→   405→        indicators: List[TunnelIndicator] = []
   406→   406→
   407→   407→        # 1. Standard 4-ping
   408→   408→        try:
   409→   409→            result = subprocess.run(
   410→   410→                ["ping", "-c", "4", "-W", str(timeout), host],
   411→   411→                capture_output=True, text=True, timeout=timeout + 5,
   412→   412→            )
   413→   413→            output = result.stdout + result.stderr
   414→   414→
   415→   415→            if result.returncode != 0:
   416→   416→                return indicators  # Host doesn't respond to ping
   417→   417→
   418→   418→            # Parse timing
   419→   419→            times = re.findall(r"time=([\d.]+)\s*ms", output)
   420→   420→            if len(times) >= 2:
   421→   421→                float_times = [float(t) for t in times]
   422→   422→                avg = sum(float_times) / len(float_times)
   423→   423→                jitter = max(float_times) - min(float_times)
   424→   424→                std_dev = self._std_dev(float_times)
   425→   425→
   426→   426→                details: Dict[str, Any] = {
   427→   427→                    "avg_ms": round(avg, 2),
   428→   428→                    "jitter_ms": round(jitter, 2),
   429→   429→                    "std_dev_ms": round(std_dev, 2),
   430→   430→                    "samples": len(float_times),
   431→   431→                }
   432→   432→
   433→   433→                # Very low jitter suggests a stable tunnel channel
   434→   434→                if jitter < 2.0 and len(float_times) >= 3:
   435→   435→                    indicators.append(
   436→   436→                        TunnelIndicator(
   437→   437→                            tunnel_type=TunnelType.ICMP_TUNNEL,
   438→   438→                            confidence=0.55,
   439→   439→                            evidence=(
   440→   440→                                f"Very low ICMP jitter ({jitter:.2f}ms, std={std_dev:.2f}ms) "
   441→   441→                                f"across {len(float_times)} pings -- stable timing channel"
   442→   442→                            ),
   443→   443→                            destination_ip=host,
   444→   444→                            protocol_details=details,
   445→   445→                        )
   446→   446→                    )
   447→   447→
   448→   448→            # Parse payload size
   449→   449→            size_match = re.search(r"(\d+)\s*bytes?\s+(?:from|of data)", output)
   450→   450→            if size_match:
   451→   451→                payload = int(size_match.group(1))
   452→   452→                if payload > 64:
   453→   453→                    indicators.append(
   454→   454→                        TunnelIndicator(
   455→   455→                            tunnel_type=TunnelType.ICMP_TUNNEL,
   456→   456→                            confidence=min(0.4 + (payload - 64) / 200, 0.85),
   457→   457→                            evidence=(
   458→   458→                                f"ICMP payload size {payload} bytes exceeds standard 64 bytes"
   459→   459→                            ),
   460→   460→                            destination_ip=host,
   461→   461→                            protocol_details={"payload_bytes": payload},
   462→   462→                        )
   463→   463→                    )
   464→   464→
   465→   465→            # Check TTL consistency
   466→   466→            ttls = re.findall(r"ttl=(\d+)", output)
   467→   467→            if len(ttls) >= 2:
   468→   468→                ttl_values = [int(t) for t in ttls]
   469→   469→                ttl_range = max(ttl_values) - min(ttl_values)
   470→   470→                if ttl_range > 2:
   471→   471→                    indicators.append(
   472→   472→                        TunnelIndicator(
   473→   473→                            tunnel_type=TunnelType.ICMP_TUNNEL,
   474→   474→                            confidence=0.5,
   475→   475→                            evidence=(
   476→   476→                                f"Inconsistent TTL values: {ttl_values} "
   477→   477→                                f"(range={ttl_range}) -- possible tunnel hop"
   478→   478→                            ),
   479→   479→                            destination_ip=host,
   480→   480→                            protocol_details={"ttl_values": ttl_values},
   481→   481→                        )
   482→   482→                    )
   483→   483→

   488→   488→        # 2. Large-payload ping

   498→   498→                        TunnelIndicator(

   506→   506→                            protocol_details={"large_payload_accepted": True},

   508→   508→                    )
   509→   509→        except (FileNotFoundError, subprocess.TimeoutExpired):
   510→   510→            pass
   511→   511→
   512→   512→        return indicators

   546→   546→                                "ipv6_address": ip6,

   549→   549→                        )
   550→   550→                    )
   551→   551→        except (socket.gaierror, OSError) as exc:

   553→   553→
   554→   554→        # 2. Check DNS for tunnel-related subdomains

   579→   579→                    )
   580→   580→            except socket.gaierror:
   581→   581→                continue
   582→   582→
   583→   583→        return indicators

   585→   585→    # -- GRE tunnel --------------------------------------------------------
   586→   586→
   587→   587→    def detect_gre_tunnel(
   588→   588→        self, host: str, timeout: int = 8
   589→   589→    ) -> List[TunnelIndicator]:
   590→   590→        """Check for GRE protocol (IP protocol 47) availability.
   591→   591→
   592→   592→        GRE tunneling cannot be reliably detected without raw sockets or
   593→   593→        captured traffic, so we check indirect indicators:
   594→   594→        - PPTP port 1723 (often used with GRE)
   595→   595→        - DNS records suggesting VPN infrastructure
   596→   596→        - Known GRE-tunnel-friendly patterns
   597→   597→        """
   598→   598→        indicators: List[TunnelIndicator] = []
   599→   599→
   600→   600→        # 1. Check PPTP port (1723) -- commonly paired with GRE
   601→   601→        pptp_open = self._check_port(host, 1723, timeout)
   602→   602→        if pptp_open:
   603→   603→            indicators.append(
   604→   604→                TunnelIndicator(
   605→   605→                    tunnel_type=TunnelType.GRE_TUNNEL,
   606→   606→                    confidence=0.55,
   607→   607→                    evidence=(
   608→   608→                        f"PPTP port 1723 is open on {host} -- "
   609→   609→                        f"PPTP uses GRE (protocol 47) for data"
   610→   610→                    ),
   611→   611→                    destination_ip=host,
   612→   612→                    protocol_details={"port_1723_open": True},
   613→   613→                )
   614→   614→            )
   615→   615→
   616→   616→        # 2. Check for VPN-related DNS TXT records
   617→   617→        vpn_keywords = ["vpn", "tunnel", "gre", "pptp", "ipsec"]
   618→   618→        txt_records = self._query_txt_records(host, timeout)
   619→   619→        for txt in txt_records:
   620→   620→            for kw in vpn_keywords:
   621→   621→                if kw in txt.lower():
   622→   622→                    indicators.append(
   623→   623→                        TunnelIndicator(
   624→   624→                            tunnel_type=TunnelType.GRE_TUNNEL,
   625→   625→                            confidence=0.4,
   626→   626→                            evidence=f"VPN/tunnel-related TXT record: {txt[:100]}",
   627→   627→                            destination_ip=host,
   628→   628→                            protocol_details={"txt_record": txt},
   629→   629→                        )
   630→   630→                    )
   631→   631→                    break
   632→   632→
   633→   633→        # 3. Common GRE/VPN ports
   634→   634→        vpn_ports = {
   635→   635→            47: "GRE",
   636→   636→            1723: "PPTP",
   637→   637→            500: "IKE (IPSec)",
   638→   638→            4500: "NAT-T (IPSec)",
   639→   639→            1194: "OpenVPN",
   640→   640→            1293: "IPSec",
   641→   641→            2134: "GRE over IP",
   642→   642→        }
   643→   643→        open_ports: Dict[int, str] = {}
   644→   644→        for port, name in vpn_ports.items():
   645→   645→            if port == 1723:  # already checked
   646→   646→                continue
   647→   647→            if self._check_port(host, port, timeout):
   648→   648→                open_ports[port] = name
   649→   649→
   650→   650→        if open_ports:
   651→   651→            indicators.append(
   652→   652→                TunnelIndicator(
   653→   653→                    tunnel_type=TunnelType.GRE_TUNNEL,
   654→   654→                    confidence=0.45,
   655→   655→                    evidence=(
   656→   656→                        f"VPN/tunnel ports open: "
   657→   657→                        + ", ".join(f"{p} ({n})" for p, n in open_ports.items())
   658→   658→                    ),
   659→   659→                    destination_ip=host,
   660→   660→                    protocol_details={"open_vpn_ports": open_ports},
   661→   661→                )
   662→   662→            )
   663→   663→
   664→   664→        return indicators

   666→   666→    # -- SSH tunnel --------------------------------------------------------
   667→   667→
   668→   668→    def detect_ssh_tunnel(
   669→   669→        self, host: str, timeout: int = 8,
   670→   670→        dest_ip: Optional[str] = None
   671→   671→    ) -> List[TunnelIndicator]:
   672→   672→        """Check SSH port 22 availability and analyse the banner.
   673→   673→
   674→   674→        An open SSH port with a permissive configuration (e.g. allowing
   675→   675→        port forwarding) is a potential tunneling vector.
   676→   676→        """
   677→   677→        indicators: List[TunnelIndicator] = []
   678→   678→
   679→   679→        ssh_port = 22
   680→   680→        try:
   681→   681→            sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
   682→   682→            sock.settimeout(timeout)
   683→   683→            result = sock.connect_ex((host, ssh_port))
   684→   684→            if result != 0:
   685→   685→                sock.close()
   686→   686→                return indicators
   687→   687→
   688→   688→            # Read banner
   689→   689→            banner = b""
   690→   690→            try:
   691→   691→                banner = sock.recv(1024)
   692→   692→            except OSError:
   693→   693→                pass
   694→   694→            sock.close()
   695→   695→
   696→   696→            banner_text = banner.decode("utf-8", errors="replace").strip()
   697→   697→
   698→   698→            if not banner_text:
   699→   699→                return indicators
   700→   700→
   701→   701→            details: Dict[str, Any] = {
   702→   702→                "port": ssh_port,
   703→   703→                "banner": banner_text,
   704→   704→            }
   705→   705→
   706→   706→            indicators.append(
   707→   707→                TunnelIndicator(
   708→   708→                    tunnel_type=TunnelType.SSH_TUNNEL,
   709→   709→                    confidence=0.5,
   710→   710→                    evidence=f"SSH port {ssh_port} is open: {banner_text}",
   711→   711→                    destination_ip=dest_ip or host,
   712→   712→                    protocol_details=details,
   713→   713→                )
   714→   714→            )
   715→   715→
   716→   716→            # Analyse banner for tunnel-friendly software
   717→   717→            tunnel_friendly = [
   718→   718→                (r"OpenSSH", 0.5, "OpenSSH supports Local/Remote/Dynamic port forwarding"),
   719→   719→                (r"Dropbear", 0.55, "Dropbear is a lightweight SSH often used in embedded tunnel setups"),
   720→   720→                (r"libssh", 0.45, "libssh library-based server -- programmatic tunneling possible"),
   721→   721→            ]
   722→   722→            for pattern, conf, desc in tunnel_friendly:
   723→   723→                if re.search(pattern, banner_text, re.IGNORECASE):
   724→   724→                    indicators.append(
   725→   725→                        TunnelIndicator(
   726→   726→                            tunnel_type=TunnelType.SSH_TUNNEL,
   727→   727→                            confidence=conf,
   728→   728→                            evidence=f"{desc}: {banner_text}",
   729→   729→                            destination_ip=dest_ip or host,
   730→   730→                            protocol_details=details,
   731→   731→                        )
   732→   732→                    )
   733→   733→
   734→   734→            # Check for non-standard SSH port (sometimes used to hide tunnels)
   735→   735→            # Scan common alternate SSH ports
   736→   736→            alt_ssh_ports = [2222, 222, 8022, 443, 8443]
   737→   737→            for port in alt_ssh_ports:
   738→   738→                try:
   739→   739→                    sock2 = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
   740→   740→                    sock2.settimeout(timeout)
   741→   741→                    r = sock2.connect_ex((host, port))
   742→   742→                    if r == 0:
   743→   743→                        alt_banner = b""
   744→   744→                        try:
   745→   745→                            alt_banner = sock2.recv(512)
   746→   746→                        except OSError:
   747→   747→                            pass
   748→   748→                        sock2.close()
   749→   749→                        alt_text = alt_banner.decode("utf-8", errors="replace").strip()
   750→   750→                        if "ssh" in alt_text.lower():
   751→   751→                            indicators.append(
   752→   752→                                TunnelIndicator(
   753→   753→                                    tunnel_type=TunnelType.SSH_TUNNEL,
   754→   754→                                    confidence=0.6,
   755→   755→                                    evidence=(
   756→   756→                                        f"SSH running on non-standard port {port}: {alt_text}"
   757→   757→                                    ),
   758→   758→                                    destination_ip=dest_ip or host,
   759→   759→                                    protocol_details={"port": port, "banner": alt_text},
   760→   760→                                )
   761→   761→                            )
   762→   762→                    else:
   763→   763→                        sock2.close()
   764→   764→                except OSError:
   765→   765→                    pass
   766→   766→
   767→   767→        except OSError as exc:
   768→   768→            logger.debug("SSH tunnel detection failed for %s: %s", host, exc)
   769→   769→
   770→   770→        return indicators

   772→   772→    # -- HTTP CONNECT tunnel ----------------------------------------------
   773→   773→
   774→   774→    def detect_http_tunnel(
   775→   775→        self, base_url: str, timeout: int = 8
   776→   776→    ) -> List[TunnelIndicator]:
   777→   777→        """Check for HTTP CONNECT method support (proxy functionality).
   778→   778→
   779→   779→        An open HTTP proxy allows arbitrary TCP connections, making it a
   780→   780→        powerful tunneling vector.
   781→   781→        """
   782→   782→        indicators: List[TunnelIndicator] = []
   783→   783→
   784→   784→        parsed = urllib.parse.urlparse(base_url)
   785→   785→        proxy_host = parsed.hostname or ""
   786→   786→        proxy_port = parsed.port or (443 if parsed.scheme == "https" else 80)
   787→   787→        scheme = parsed.scheme or "http"
   788→   788→
   789→   789→        # 1. Send CONNECT request
   790→   790→        try:
   791→   791→            connect_target = "example.com:443"
   792→   792→            conn = http.client.HTTPConnection(proxy_host, proxy_port, timeout=timeout)
   793→   793→            conn.request("CONNECT", connect_target, headers={
   794→   794→                "Host": connect_target,
   795→   795→                "User-Agent": "ReconPro/9.2.0",
   796→   796→            })
   797→   797→            resp = conn.getresponse()
   798→   798→            status = resp.status
   799→   799→            resp.read()
   800→   800→            conn.close()
   801→   801→
   802→   802→            if status == 200:
   803→   803→                indicators.append(
   804→   804→                    TunnelIndicator(
   805→   805→                        tunnel_type=TunnelType.HTTP_TUNNEL,
   806→   806→                        confidence=0.85,
   807→   807→                        evidence=(
   808→   808→                            f"HTTP CONNECT to {connect_target} returned 200 -- "
   809→   809→                            f"open proxy / tunnel endpoint"
   810→   810→                        ),
   811→   811→                        destination_ip=proxy_host,
   812→   812→                        protocol_details={
   813→   813→                            "method": "CONNECT",
   814→   814→                            "target": connect_target,
   815→   815→                            "status": status,
   816→   816→                        },
   817→   817→                    )
   818→   818→                )
   819→   819→            elif status in (403, 407):
   820→   820→                indicators.append(
   821→   821→                    TunnelIndicator(
   822→   822→                        tunnel_type=TunnelType.HTTP_TUNNEL,
   823→   823→                        confidence=0.3,
   824→   824→                        evidence=(
   825→   825→                            f"HTTP CONNECT returned {status} -- proxy detected "
   826→   826→                            f"but access restricted (may be exploitable with creds)"
   827→   827→                        ),
   828→   828→                        destination_ip=proxy_host,
   829→   829→                        protocol_details={"status": status},
   830→   830→                    )
   831→   831→                )
   832→   832→
   833→   833→        except (OSError, TimeoutError, Exception) as exc:
   834→   834→            logger.debug("HTTP CONNECT test failed: %s", exc)
   835→   835→
   836→   836→        # 2. Check for proxy-related headers in normal response
   837→   837→        try:
   838→   838→            req = urllib.request.Request(
   839→   839→                base_url, headers={"User-Agent": "ReconPro/9.2.0"},
   840→   840→            )
   841→   841→            with urllib.request.urlopen(req, timeout=timeout) as resp:
   842→   842→                headers = {k.lower(): v for k, v in resp.getheaders()}
   843→   843→
   844→   844→            proxy_headers = [
   845→   845→                ("via", "Proxy chain detected"),
   846→   846→                ("x-forwarded-for", "Forwarding proxy present"),
   847→   847→                ("forwarded", "Forwarded header present"),
   848→   848→                ("proxy-connection", "Proxy-Connection header (proxy behavior)"),
   849→   849→                ("x-proxy-id", "Proxy ID header detected"),
   850→   850→            ]
   851→   851→            for hdr, desc in proxy_headers:
   852→   852→                val = headers.get(hdr)
   853→   853→                if val:
   854→   854→                    indicators.append(
   855→   855→                        TunnelIndicator(
   856→   856→                            tunnel_type=TunnelType.HTTP_TUNNEL,
   857→   857→                            confidence=0.4,
   858→   858→                            evidence=f"{desc}: {hdr}={val}",
   859→   859→                            destination_ip=proxy_host,
   860→   860→                            protocol_details={"header": hdr, "value": val},
   861→   861→                        )
   862→   862→                    )
   863→   863→
   864→   864→        except (urllib.error.URLError, OSError, TimeoutError):
   865→   865→            pass
   866→   866→
   867→   867→        # 3. HTTPS tunnel detection (check for TLS 1.3 + large certificate chain)
   868→   868→        if scheme == "https":
   869→   869→            try:
   870→   870→                # We can't easily inspect TLS from stdlib, but check for HSTS
   871→   871→                # and certificate transparency which suggest HTTPS tunnel surface
   872→   872→                req = urllib.request.Request(
   873→   873→                    base_url, headers={"User-Agent": "ReconPro/9.2.0"},
   874→   874→                )
   875→   875→                with urllib.request.urlopen(req, timeout=timeout) as resp:
   876→   876→                    hdrs = {k.lower(): v for k, v in resp.getheaders()}
   877→   877→                if "strict-transport-security" in hdrs:
   878→   878→                    indicators.append(
   879→   879→                        TunnelIndicator(
   880→   880→                            tunnel_type=TunnelType.HTTPS_TUNNEL,
   881→   881→                            confidence=0.25,
   882→   882→                            evidence=(
   883→   883→                                "HSTS enabled -- encrypted HTTPS tunnel surface exists "
   884→   884→                                "(normal for legitimate sites, but enables covert TLS tunnels)"
   885→   885→                            ),
   886→   886→                            destination_ip=proxy_host,
   887→   887→                            protocol_details={
   888→   888→                                "hsts": hdrs["strict-transport-security"]
   889→   889→                            },
   890→   890→                        )
   891→   891→                    )
   892→   892→            except (urllib.error.URLError, OSError, TimeoutError):
   893→   893→                pass
   894→   894→
   895→   895→        return indicators
