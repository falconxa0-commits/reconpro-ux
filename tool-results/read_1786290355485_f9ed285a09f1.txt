   450→        req_headers["Content-Type"] = "application/octet-stream"
   451→    if "User-Agent" not in req_headers:
   452→        req_headers["User-Agent"] = "ReconPro/9.2.0 (Security Scanner)"
   453→
   454→    req = urllib.request.Request(url, data=data, headers=req_headers, method=method)
   455→    start = time.time()
   456→    try:
   457→        resp = urllib.request.urlopen(req, timeout=timeout)
   458→        elapsed = (time.time() - start) * 1000
   459→        body = resp.read()
   460→        result["status"] = resp.getcode()
   461→        result["headers"] = dict(resp.getheaders())
   462→        result["body_length"] = len(body)
   463→        result["elapsed_ms"] = elapsed
   464→        result["body_preview"] = body[:512].decode("utf-8", errors="replace")
   465→    except urllib.error.HTTPError as e:
   466→        elapsed = (time.time() - start) * 1000
   467→        result["status"] = e.code
   468→        try:
   469→            body = e.read()
   470→            result["body_length"] = len(body)
   471→            result["body_preview"] = body[:512].decode("utf-8", errors="replace")
   472→        except Exception:
   473→            pass
   474→        result["elapsed_ms"] = elapsed
   475→    except (urllib.error.URLError, OSError, socket.timeout) as e:
   476→        result["elapsed_ms"] = (time.time() - start) * 1000
   477→        result["error"] = str(e)
   478→
   479→    return result

   482→def _measure_response_times(url: str, count: int = 8, timeout: float = 8.0) -> List[float]:
   483→    """Collect multiple response time samples for a URL.
   484→
   485→    Args:
   486→        url: URL to measure.
   487→        count: Number of samples to collect.
   488→        timeout: Per-request timeout.
   489→
   490→    Returns:
   491→        List of response times in milliseconds.
   492→    """
   493→    times: List[float] = []
   494→    for _ in range(count):
   495→        result = _http_request(url, method="HEAD", timeout=timeout)
   496→        if result["elapsed_ms"] > 0:
   497→            times.append(result["elapsed_ms"])
   498→        time.sleep(0.1)  # Brief pause between requests
   499→    return times

   502→def _compute_timing_jitter(samples: List[float]) -> float:
   503→    """Compute the coefficient of variation (jitter) of timing samples.
   504→
   505→    A higher coefficient of variation indicates more timing variability,
   506→    which could support timing-based exfiltration channels.
   507→
   508→    Args:
   509→        samples: List of timing measurements in milliseconds.
   510→
   511→    Returns:
   512→        Coefficient of variation (stddev / mean), or 0.0 if insufficient data.
   513→    """
   514→    if len(samples) < 3:
   515→        return 0.0
   516→    mean = sum(samples) / len(samples)
   517→    if mean == 0:
   518→        return 0.0
   519→    variance = sum((s - mean) ** 2 for s in samples) / len(samples)
   520→    stddev = math.sqrt(variance)
   521→    return stddev / mean

   529→def _tls_probe(host: str, port: int = 443, timeout: float = 8.0) -> Dict:
   530→    """Probe TLS configuration of the target host.
   531→
   532→    Tests supported TLS versions, extracts certificate details, and
   533→    evaluates cipher suite strength.
   534→
   535→    Args:
   536→        host: Target hostname.
   537→        port: TLS port.
   538→        timeout: Connection timeout.
   539→
   540→    Returns:
   541→        Dict with TLS version info, cipher details, and certificate metadata.
   542→    """
   543→    result: Dict = {
   544→        "tls_version": None,
   545→        "cipher_name": None,
   546→        "cipher_bits": 0,
   547→        "cert_subject": "",
   548→        "cert_issuer": "",
   549→        "cert_expiry": "",
   550→        "cert_serial": "",
   551→        "hsts_present": False,
   552→        "forward_secrecy": False,
   553→        "ssl_errors": [],
   554→        "supported_protocols": [],
   555→    }
   556→
   557→    # Test each TLS version
   558→    tls_versions = [
   559→        (ssl.TLSVersion.TLSv1_3, "TLSv1.3"),
   560→        (ssl.TLSVersion.TLSv1_2, "TLSv1.2"),
   561→        (ssl.TLSVersion.TLSv1_1, "TLSv1.1"),
   562→        (ssl.TLSVersion.TLSv1, "TLSv1.0"),
   563→    ]
   564→
   565→    context = ssl.SSLContext(ssl.PROTOCOL_TLS_CLIENT)
   566→    context.check_hostname = False
   567→    context.verify_mode = ssl.CERT_NONE
   568→
   569→    for tls_ver, ver_name in tls_versions:
   570→        try:
   571→            test_ctx = ssl.SSLContext(ssl.PROTOCOL_TLS_CLIENT)
   572→            test_ctx.check_hostname = False
   573→            test_ctx.verify_mode = ssl.CERT_NONE
   574→            test_ctx.maximum_version = tls_ver
   575→            test_ctx.minimum_version = tls_ver
   576→
   577→            with socket.create_connection((host, port), timeout=timeout) as sock:
   578→                with test_ctx.wrap_socket(sock, server_hostname=host) as ssock:
   579→                    result["supported_protocols"].append(ver_name)
   580→                    result["tls_version"] = ver_name
   581→                    cipher = ssock.cipher()
   582→                    if cipher:
   583→                        result["cipher_name"] = cipher[0]
   584→                        result["cipher_bits"] = cipher[2] if len(cipher) > 2 else cipher[1]
   585→                    cert = ssock.getpeercert(binary_form=False)
   586→                    if cert:
   587→                        subject_parts = []
   588→                        for rdn in cert.get("subject", ()):
   589→                            for attr_type, attr_value in rdn:
   590→                                subject_parts.append(f"{attr_type}={attr_value}")
   591→                        result["cert_subject"] = ", ".join(subject_parts)
   592→
   593→                        issuer_parts = []
   594→                        for rdn in cert.get("issuer", ()):
   595→                            for attr_type, attr_value in rdn:
   596→                                issuer_parts.append(f"{attr_type}={attr_value}")
   597→                        result["cert_issuer"] = ", ".join(issuer_parts)
   598→
   599→                        result["cert_serial"] = str(cert.get("serialNumber", ""))
   600→
   601→                        not_after = cert.get("notAfter", "")
   602→                        if not_after:
   603→                            result["cert_expiry"] = not_after
   604→        except (ssl.SSLError, OSError, socket.timeout):
   605→            result["ssl_errors"].append(f"{ver_name} not supported or failed")
   606→
   607→    # Use the most secure protocol for full analysis
   608→    try:
   609→        with socket.create_connection((host, port), timeout=timeout) as sock:
   610→            with context.wrap_socket(sock, server_hostname=host) as ssock:
   611→                cipher = ssock.cipher()
   612→                if cipher:
   613→                    result["cipher_name"] = cipher[0]
   614→                    result["cipher_bits"] = cipher[2] if len(cipher) > 2 else cipher[1]
   615→
   616→                # Check for forward secrecy via cipher name heuristics
   617→                fs_ciphers = {
   618→                    "ECDHE", "DHE", "AESGCM", "CHACHA20",
   619→                    "AES_256_GCM", "AES_128_GCM", "TLS_AES_256_GCM_SHA384",
   620→                    "TLS_CHACHA20_POLY1305_SHA256", "TLS_AES_128_GCM_SHA256",
   621→                }
   622→                if cipher and any(fs in cipher[0].upper() for fs in fs_ciphers):
   623→                    result["forward_secrecy"] = True
   624→
   625→                # Check HSTS via a quick HTTP request
   626→                try:
   627→                    ssock.sendall(
   628→                        f"HEAD / HTTP/1.1\r\nHost: {host}\r\n"
   629→                        f"Connection: close\r\n\r\n".encode()
   630→                    )
   631→                    resp_data = b""
   632→                    while True:
   633→                        chunk = ssock.recv(4096)
   634→                        if not chunk:
   635→                            break
   636→                        resp_data += chunk
   637→                        if b"\r\n\r\n" in resp_data:
   638→                            break
   639→                    resp_str = resp_data.decode("utf-8", errors="replace")
   640→                    if "strict-transport-security" in resp_str.lower():
   641→                        result["hsts_present"] = True
   642→                except (OSError, ssl.SSLError):
   643→                    pass
   644→    except (ssl.SSLError, OSError, socket.timeout) as e:
   645→        result["ssl_errors"].append(str(e))
   646→
   647→    return result

   654→
   655→class BandwidthEstimator:
   656→    """Estimates theoretical exfiltration bandwidth for various channels.
   657→
   658→    These are upper-bound theoretical estimates based on protocol
   659→    characteristics, not actual throughput measurements.
   660→    """
   661→
   662→    def __init__(self, timeout: float = 5.0) -> None:
   663→        self.timeout = timeout
   664→
   665→    def estimate_dns_bandwidth(self, host: str) -> float:
   666→        """Estimate DNS tunneling bandwidth in bytes/second.
   667→
   668→        DNS exfiltration works by encoding data in subdomain labels.
   669→        Each label is limited to 63 characters, and the total FQDN
   670→        to 253 characters. TXT records allow ~255 bytes per response.
   671→
   672→        Args:
   673→            host: Target domain to estimate DNS bandwidth for.
   674→
   675→        Returns:
   676→            Estimated bandwidth in bytes/second.
   677→        """
   678→        # Base measurements
   679→        max_label_len = 63
   680→        max_fqdn_len = 253
   681→        base64_efficiency = 0.75  # 3 bytes → 4 base64 chars
   682→        overhead_per_char = 0.2  # DNS protocol overhead
   683→
   684→        # Perform actual DNS timing to calibrate
   685→        result = dns_query_udp(host, qtype=1, timeout=self.timeout)
   686→        if result:
   687→            rtt_ms = result.get("elapsed_ms", 50)
   688→        else:
   689→            rtt_ms = 50.0  # Conservative default
   690→
   691→        if rtt_ms < 1:
   692→            rtt_ms = 1.0
   693→
   694→        # Usable data per query: FQDN overhead minus domain itself
   695→        domain_len = len(host)
   696→        available_label_space = max(0, max_fqdn_len - domain_len - 5)  # dots + padding
   697→        usable_chars_per_query = min(max_label_len, available_label_space)
   698→        bytes_per_query = usable_chars_per_query * base64_efficiency * (1 - overhead_per_char)
   699→
   700→        queries_per_second = 1000.0 / rtt_ms
   701→        bandwidth = bytes_per_query * queries_per_second
   702→
   703→        # Check TXT record capacity (response channel)
   704→        txt_result = dns_query_udp(host, qtype=16, timeout=self.timeout)
   705→        if txt_result and txt_result["answers"]:
   706→            for ans in txt_result["answers"]:
   707→                if ans["type"] == "TXT" and ans["rdata_len"] > 0:
   708→                    # Can exfiltrate via TXT response data
   709→                    txt_bandwidth = ans["rdata_len"] * queries_per_second
   710→                    bandwidth = max(bandwidth, txt_bandwidth)
   711→
   712→        return round(bandwidth, 1)
   713→
   714→    def estimate_http_bandwidth(self, base_url: str) -> float:
   715→        """Estimate HTTP-based exfiltration bandwidth in bytes/second.
   716→
   717→        Measures actual POST throughput by sending data and observing
   718→        accepted payload sizes and response times.
   719→
   720→        Args:
   721→            base_url: Base URL of the target.
   722→
   723→        Returns:
   724→            Estimated bandwidth in bytes/second.
   725→        """
   726→        test_url = base_url.rstrip("/")
   727→        # Try POST to common endpoints
   728→        endpoints = ["/", "/api/upload", "/upload", "/api/data", "/submit"]
   729→        best_bandwidth = 0.0
   730→
   731→        for endpoint in endpoints:
   732→            url = test_url + endpoint
   733→            # Test with increasing payload sizes
   734→            for size in [1024, 4096, 16384, 65536]:
   735→                payload = os.urandom(size)
   736→                result = _http_request(
   737→                    url, method="POST", data=payload, timeout=self.timeout
   738→                )
   739→                if result["status"] > 0 and result["error"] is None:
   740→                    elapsed = result["elapsed_ms"] / 1000.0
   741→                    if elapsed > 0:
   742→                        bw = size / elapsed
   743→                        if bw > best_bandwidth:
   744→                            best_bandwidth = bw
   745→                    break  # Largest accepted size found
   746→                elif result["status"] in (413, 431):
   747→                    break  # Payload too large
   748→
   749→        # If no POST accepted, estimate from GET response times
   750→        if best_bandwidth == 0:
   751→            times = _measure_response_times(test_url + "/", count=5, timeout=self.timeout)
   752→            if times:
   753→                avg_time = sum(times) / len(times) / 1000.0
   754→                if avg_time > 0:
   755→                    # Conservative: assume URL-encoded data in query strings
   756→                    best_bandwidth = 2000.0 / avg_time  # ~2KB in query string
   757→
   758→        return round(best_bandwidth, 1)
   759→
   760→    def estimate_icmp_bandwidth(self, host: str) -> float:
   761→        """Estimate ICMP tunneling bandwidth in bytes/second.
   762→
   763→        ICMP exfiltration encodes data in echo request payload bytes.
   764→        Bandwidth is limited by max payload size and packet rate.
   765→
   766→        Args:
   767→            host: Target host.
   768→
   769→        Returns:
   770→            Estimated bandwidth in bytes/second.
   771→        """
   772→        ping_result = icmp_ping(host, count=2, timeout=self.timeout)
   773→        if not ping_result["reachable"]:
   774→            return 0.0
   775→
   776→        max_payload = _icmp_payload_tolerance(host, self.timeout)
   777→        if max_payload == 0:
   778→            max_payload = ping_result.get("max_payload_accepted", 64)
   779→
   780→        rtt_ms = ping_result["avg_rtt_ms"]
   781→        if rtt_ms < 1:
   782→            rtt_ms = 20.0  # Reasonable default
   783→
   784→        # ICMP tunneling: data encoded in payload, need round-trip for bidirectional
   785→        # But unidirectional exfil only needs outgoing packets
   786→        # Conservative estimate: 10 packets/second to avoid detection
   787→        packets_per_second = min(1000.0 / rtt_ms, 50.0)  # Capped at 50 pps
   788→        bandwidth = max_payload * packets_per_second * 0.7  # 70% efficiency for encoding
   789→
   790→        return round(bandwidth, 1)
   791→
   792→    def estimate_timing_bandwidth(self) -> float:
   793→        """Estimate theoretical bandwidth via timing channels in bits/second.
   794→
   795→        Timing channels encode data in inter-packet delays. With
   796→        millisecond-precision timing, each delay slot can encode
   797→        ~10 bits (1024 distinct values). At ~10 packets/second,
   798→        this yields ~100 bits/second.
   799→
   800→        Returns:
   801→            Estimated bandwidth in bits/second.
   802→        """
   803→        # Assume millisecond timing precision
   804→        precision_ms = 1.0
   805→        # Distinct delay values we can reliably encode
   806→        max_delay_ms = 2000.0
   807→        distinct_values = int(max_delay_ms / precision_ms)
   808→        bits_per_symbol = math.log2(max(distinct_values, 2))
   809→
   810→        # Conservative packet rate (low and slow to avoid detection)
   811→        packets_per_second = 5.0
   812→        bandwidth_bps = bits_per_symbol * packets_per_second
   813→
   814→        return round(bandwidth_bps, 1)
   815→
   816→    def estimate_websocket_bandwidth(self, base_url: str) -> float:
   817→        """Estimate WebSocket exfiltration bandwidth.
   818→
   819→        Args:
   820→            base_url: Base URL to test WebSocket endpoints.
   821→
   822→        Returns:
   823→            Estimated bandwidth in bytes/second.
   824→        """
   825→        # Test common WebSocket endpoints
   826→        ws_endpoints = ["/ws", "/websocket", "/socket", "/live", "/realtime"]
   827→        test_url = base_url.rstrip("/")
   828→
   829→        # Attempt HTTP upgrade to detect WebSocket support
   830→        for endpoint in ws_endpoints:
   831→            url = test_url + endpoint
   832→            headers = {
   833→                "Connection": "Upgrade",
   834→                "Upgrade": "websocket",
   835→                "Sec-WebSocket-Version": "13",
   836→                "Sec-WebSocket-Key": base64.b64encode(os.urandom(16)).decode(),
   837→            }
   838→            result = _http_request(url, method="GET", headers=headers, timeout=self.timeout)
   839→            if result["status"] == 101:
   840→                # WebSocket upgrade successful
   841→                elapsed = result["elapsed_ms"] / 1000.0
   842→                if elapsed > 0:
   843→                    # Assume 64KB frames at connection latency rate
   844→                    return round(65536.0 / elapsed, 1)
   845→            elif result["status"] in (400, 403, 404, 405, 502, 503):
   846→                continue
   847→
   848→        # No WebSocket found; return zero
   849→        return 0.0

   856→
   857→class ExfilRiskScorer:
   858→    """Computes composite risk scores for exfiltration channels.
   859→
   860→    Scoring model:
   861→        overall_risk = w1*viability + w2*stealth + w3*detection_difficulty + w4*bandwidth_factor
   862→    """
   863→
   864→    # Scoring weights
   865→    W_VIABILITY = 0.30
   866→    W_STEALTH = 0.25
   867→    W_DETECTION_DIFFICULTY = 0.25
   868→    W_BANDWIDTH = 0.20
   869→
   870→    # Bandwidth normalization: the bandwidth value that maps to a score of 50
   871→    BANDWIDTH_MIDPOINT = 5000.0  # bytes/sec
   872→
   873→    def score_channel(self, channel: ExfilChannel) -> float:
   874→        """Compute composite risk score for a single channel.
   875→
   876→        The score is a weighted average of viability, stealth,
   877→        detection difficulty, and a log-scaled bandwidth factor,
   878→        all normalized to 0-100.
   879→
   880→        Args:
   881→            channel: The ExfilChannel to score.
   882→
   883→        Returns:
   884→            Composite risk score from 0 to 100.
   885→        """
   886→        # Log-scale bandwidth to 0-100 range
   887→        if channel.bandwidth_estimate > 0:
   888→            bw_factor = 50.0 + 50.0 * math.log10(
   889→                1.0 + channel.bandwidth_estimate / self.BANDWIDTH_MIDPOINT
   890→            ) / math.log10(1.0 + 100.0)  # 100x midpoint = score 100
   891→            bw_factor = max(0.0, min(100.0, bw_factor))
   892→        else:
   893→            bw_factor = 0.0
   894→
   895→        composite = (
   896→            self.W_VIABILITY * channel.viability_score
   897→            + self.W_STEALTH * channel.stealth_score
   898→            + self.W_DETECTION_DIFFICULTY * channel.detection_difficulty
   899→            + self.W_BANDWIDTH * bw_factor
   900→        )
   901→
   902→        # Indicator count bonus: more evidence = slightly higher risk
   903→        indicator_bonus = min(len(channel.indicators) * 2.0, 10.0)
   904→        composite = min(composite + indicator_bonus, 100.0)
   905→
   906→        channel.overall_risk = round(composite, 1)
   907→        return channel.overall_risk
   908→
   909→    def score_overall(self, channels: List[ExfilChannel]) -> Dict:
   910→        """Compute organization-level exfiltration risk from all channels.
   911→
   912→        Args:
   913→            channels: List of analyzed ExfilChannel objects.
   914→
   915→        Returns:
   916→            Dict with 'overall_score', 'risk_level', 'highest_risk_channel',
   917→            'channel_summary', and 'recommendations'.
   918→        """
   919→        if not channels:
   920→            return {
   921→                "overall_score": 0.0,
   922→                "risk_level": "NONE",
   923→                "highest_risk_channel": None,
   924→                "channel_summary": [],
   925→                "recommendations": [
   926→                    "No channels analyzed. Ensure the target is reachable."
   927→                ],
   928→            }
   929→
   930→        # Sort by risk descending
   931→        scored = sorted(channels, key=lambda c: c.overall_risk, reverse=True)
   932→        highest = scored[0]
   933→
   934→        # Weighted overall: top channels matter more
   935→        weights = [0.30, 0.20, 0.15, 0.10, 0.08, 0.07, 0.05, 0.03, 0.02]
   936→        overall = 0.0
   937→        for i, ch in enumerate(scored):
   938→            w = weights[i] if i < len(weights) else 0.01
   939→            overall += w * ch.overall_risk
   940→        overall = min(overall * 2.5, 100.0)  # Scale up to make range meaningful
   941→
   942→        # Risk level classification
   943→        if overall >= 80:
   944→            level = "CRITICAL"
   945→        elif overall >= 60:
   946→            level = "HIGH"
   947→        elif overall >= 40:
   948→            level = "MEDIUM"
   949→        elif overall >= 20:
   950→            level = "LOW"
   951→        else:
   952→            level = "MINIMAL"
   953→
   954→        # Aggregate recommendations
   955→        seen_recs: set = set()
   956→        recommendations: List[str] = []
   957→        for ch in scored:
   958→            if ch.remediation and ch.remediation not in seen_recs:
   959→                seen_recs.add(ch.remediation)
   960→                recommendations.append(ch.remediation)
   961→
   962→        summary = [
   963→            {
   964→                "channel": ch.channel_type.value,
   965→                "risk": ch.overall_risk,
   966→                "viability": ch.viability_score,
   967→                "bandwidth_bps": ch.bandwidth_estimate,
   968→            }
   969→            for ch in scored
   970→        ]
   971→
   972→        return {
   973→            "overall_score": round(overall, 1),
   974→            "risk_level": level,
   975→            "highest_risk_channel": highest.channel_type.value,
   976→            "channel_summary": summary,
   977→            "recommendations": recommendations,
   978→        }
   979→

  1094→

  1096→        channels.append(self._analyze_cloud_storage_exfil(target))
  1097→        channels.append(self._analyze_file_sharing_exfil(target))

  1121→    def _resolve_target(self, target: str) -> str:
  1122→        """Resolve target hostname to IP address.
  1123→
  1124→        Args:
  1125→            target: Hostname or IP string.
  1126→
  1127→        Returns:
  1128→            IP address string, or the original target if resolution fails.
  1129→        """
  1130→        # Already an IP?
  1131→        ip_pattern = re.compile(r"^(\d{1,3}\.){3}\d{1,3}$")
  1132→        if ip_pattern.match(target):
  1133→            return target
