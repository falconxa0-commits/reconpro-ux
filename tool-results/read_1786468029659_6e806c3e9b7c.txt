   189→        _write_plugin(self.tmp, "_hidden_spy", GOOD_PLUGIN)

   191→        _write_plugin(self.tmp, "good_plugin", GOOD_PLUGIN)
   192→
   193→        with patch("reconpro.plugins.PLUGIN_DIR", self.tmp):
   194→            plugins = discover_plugins()

   197→        self.assertNotIn("__private", plugins)
   198→        self.assertIn("good_plugin", plugins)

   208→            plugins = discover_plugins()

   681→    def test_invalid_hook_name_rejected(self):
   682→        """Hook names not in HOOK_NAMES must be rejected."""
   683→        bad_names = [
   684→            "__init__", "__import__", "eval", "exec",
   685→            "system", "popen", "shell", "delete_all",
   686→            "pre_scan_extra", "post_scan_", "",
   687→            "constructor", "__del__", "__class__",

   689→        ]

   692→            self.assertFalse(result, f"Invalid hook '{name}' was accepted!")

   698→            self.assertTrue(result, f"Valid hook '{name}' was rejected!")

   700→    def test_callback_exception_caught(self):
   701→        """Exceptions in callbacks must be caught, not propagated."""
   702→        call_log = []
   703→
   704→        def bad_callback(*args, **kwargs):
   705→            call_log.append("bad")
   706→            raise RuntimeError("Hook callback crashed!")
   707→
   708→        def good_callback(*args, **kwargs):
   709→            call_log.append("good")
   710→            return None  # Explicitly None so fire doesn't short-circuit here
   711→
   712→        # Register bad first (lower priority = runs first), then good
   713→        HookManager.register("error", bad_callback, priority=10)
   714→        HookManager.register("error", good_callback, priority=20)
   715→
   716→        # Must not raise
   717→        result = HookManager.fire("error", error_msg="test")
   718→        # Both callbacks should have been called (bad's exception caught)
   719→        self.assertEqual(call_log, ["bad", "good"])
   720→        # Result is None because both returned None (bad's exception -> no append)
   721→        self.assertIsNone(result)
   722→
   723→    def test_callback_exception_all_bad_no_crash(self):
   724→        """If ALL callbacks raise, fire() must still not crash."""
   725→        HookManager.register("error", lambda *a, **k: (_ for _ in ()).throw(RuntimeError("boom1")), priority=10)
   726→        HookManager.register("error", lambda *a, **k: (_ for _ in ()).throw(RuntimeError("boom2")), priority=20)
   727→
   728→        # Must not raise
   729→        result = HookManager.fire("error", error_msg="test")
   730→        self.assertIsNone(result)
