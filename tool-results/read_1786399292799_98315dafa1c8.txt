   183→   183→# ── Helper: Cloudflare DoH JSON lookup ───────────────────────────────────
   184→   184→
   185→   185→def _doh_lookup(name: str, rtype: str) -> List[str]:
   186→   186→    """Resolve a DNS record via Cloudflare DoH JSON API."""
   187→   187→    url = (
   188→   188→        f"https://cloudflare-dns.com/dns-query?"
   189→   189→        f"name={urllib.parse.quote(name, safe='')}&type={rtype}"
   190→   190→    )
   191→   191→    try:
   192→   192→        req = urllib.request.Request(url, headers={"Accept": "application/dns-json"})
   193→   193→        ctx = ssl.create_default_context()
   194→   194→        with urllib.request.urlopen(req, timeout=5, context=ctx) as resp:
   195→   195→            data = json.loads(resp.read(65536).decode())
   196→   196→        answers = data.get("Answer", [])
   197→   197→        return [a.get("data", "") for a in answers]
   198→   198→    except Exception:
   199→   199→        return []

   201→   201→
   202→   202→def _dns_lookup(host: str) -> Dict[str, Any]:
   203→   203→    result = {"a": [], "aaaa": [], "cname": [], "mx": [], "ns": [], "txt": []}
   204→   204→    # Try Cloudflare DoH for proper record types
   205→   205→    for qtype, attr in [("A", "a"), ("AAAA", "aaaa"), ("CNAME", "cname"),
   206→   206→                         ("MX", "mx"), ("NS", "ns"), ("TXT", "txt")]:
   207→   207→        try:
   208→   208→            records = _doh_lookup(host, qtype)
   209→   209→            if records:
   210→   210→                result[attr] = records
   211→   211→        except Exception:
   212→   212→            pass
   213→   213→
   214→   214→    # Fallback to socket for A/AAAA if DoH returned nothing
   215→   215→    if not result["a"] and not result["aaaa"]:
   216→   216→        try:
   217→   217→            answers = socket.getaddrinfo(host, None, socket.AF_UNSPEC,
   218→   218→                                          socket.SOCK_STREAM)
   219→   219→            seen = set()
   220→   220→            for fam, _, _, _, sockaddr in answers:
   221→   221→                ip = sockaddr[0]
   222→   222→                if ip not in seen:
   223→   223→                    seen.add(ip)
   224→   224→                    if ":" in ip:
   225→   225→                        result["aaaa"].append(ip)
   226→   226→                    else:
   227→   227→                        result["a"].append(ip)
   228→   228→        except Exception:
   229→   229→            pass
   230→   230→    return result

   249→   249→# ── TLS checks ───────────────────────────────────────────────────────────
   250→   250→
   251→   251→def _check_tls(host: str) -> Dict[str, Any]:
   252→   252→    info = {
   253→   253→        "version": None, "issuer": None, "expiry": None,
   254→   254→        "days_left": None, "cipher_name": None, "cipher_bits": None,
   255→   255→        "cipher_protocol": None, "shared_ciphers": [],
   256→   256→    }
   257→   257→    try:
   258→   258→        ctx = ssl.create_default_context()
   259→   259→        with socket.create_connection((host, 443), timeout=5) as sock:
   260→   260→            with ctx.wrap_socket(sock, server_hostname=host) as s:
   261→   261→                cert = s.getpeercert()
   262→   262→                info["version"] = s.version()
   263→   263→                # Cipher details
   264→   264→                cipher = s.cipher()
   265→   265→                if cipher:
   266→   266→                    info["cipher_name"] = cipher[0]
   267→   267→                    info["cipher_protocol"] = cipher[1]
   268→   268→                    info["cipher_bits"] = cipher[2]
   269→   269→                # Shared ciphers
   270→   270→                try:
   271→   271→                    shared = s.shared_ciphers()
   272→   272→                    if shared:
   273→   273→                        info["shared_ciphers"] = [
   274→   274→                            {"name": c[0], "protocol": c[1], "bits": c[2]}
   275→   275→                            for c in shared
   276→   276→                        ]
   277→   277→                except Exception:
   278→   278→                    pass
   279→   279→                if cert:
   280→   280→                    for rdn in cert.get("issuer", ()): 
   281→   281→                        for k, v in rdn:
   282→   282→                            if k == "organizationName":
   283→   283→                                info["issuer"] = v
   284→   284→                    not_after = cert.get("notAfter")
   285→   285→                    if not_after:
   286→   286→                        exp = datetime.strptime(not_after, "%b %d %H:%M:%S %Y %Z")
   287→   287→                        info["expiry"] = exp.isoformat()
   288→   288→                        info["days_left"] = (exp - datetime.utcnow()).days
   289→   289→    except Exception:
   290→   290→        pass
   291→   291→    return info

   294→   294→# ── Technology detection ──────────────────────────────────────────────────
   295→   295→
   296→   296→def _detect_tech(body: str, headers: Dict[str, str]) -> List[str]:
   297→   297→    found = []
   298→   298→    haystack = (body + " " + " ".join(headers.values())).lower()
   299→   299→    for name, sigs in TECH_SIGNATURES:
   300→   300→        if any(s.lower() in haystack for s in sigs):
   301→   301→            found.append(name)
   302→   302→    # Additional header-based detection
   303→   303→    h_lower = {k.lower(): v.lower() for k, v in headers.items()}
   304→   304→    if "x-powered-by" in h_lower:
   305→   305→        powered = h_lower["x-powered-by"]
   306→   306→        # Check if we already detected via signatures
   307→   307→        already = [t.lower() for t in found]
   308→   308→        for tech_name in ["Express", "PHP", "ASP.NET", "FastAPI", "Flask",
   309→   309→                          "Next.js", "Ruby on Rails"]:
   310→   310→            if tech_name.lower() not in already and any(
   311→   311→                kw in powered for kw in TECH_SIGNATURES[[t[0] for t in TECH_SIGNATURES].index(tech_name)][1]
   312→   312→            ):
   313→   313→                found.append(tech_name)
   314→   314→    if "x-generator" in h_lower:
   315→   315→        gen = h_lower["x-generator"]
   316→   316→        for tech_name in ["WordPress", "Drupal", "Joomla", "Ghost", "Wix",
   317→   317→                          "Squarespace", "Webflow"]:
   318→   318→            idx = [t[0] for t in TECH_SIGNATURES].index(tech_name) if tech_name in [t[0] for t in TECH_SIGNATURES] else -1
   319→   319→            if idx >= 0 and any(kw.lower() in gen for kw in TECH_SIGNATURES[idx][1]):
   320→   320→                if tech_name not in found:
   321→   321→                    found.append(tech_name)
   322→   322→    if "server" in h_lower:
   323→   323→        srv = h_lower["server"]
   324→   324→        for tech_name in ["Nginx", "Apache", "Cloudflare"]:
   325→   325→            idx = [t[0] for t in TECH_SIGNATURES].index(tech_name) if tech_name in [t[0] for t in TECH_SIGNATURES] else -1
   326→   326→            if idx >= 0 and any(kw.lower() in srv for kw in TECH_SIGNATURES[idx][1]):
   327→   327→                if tech_name not in found:
   328→   328→                    found.append(tech_name)
   329→   329→    if "via" in h_lower:
   330→   330→        via = h_lower["via"]
   331→   331→        if "fastly" in via and "Fastly" not in found:
   332→   332→            found.append("Fastly")
   333→   333→        if "cloudfront" in via and "Cloudfront" not in found:
   334→   334→            found.append("Cloudfront")
   335→   335→    return found

   350→   350→# ── CORS ──────────────────────────────────────────────────────────────────
   351→   351→
   352→   352→def _check_cors(base_url: str) -> List[Finding]:
   353→   353→    findings = []
   354→   354→    resp = http_probe(base_url)
   355→   355→    h = resp.get("headers", {})
   356→   356→    acao = h.get("Access-Control-Allow-Origin", "")
   357→   357→
   358→   358→    if acao == "*":
   359→   359→        findings.append(Finding(
   360→   360→            title="Permissive CORS (wildcard origin)",
   361→   361→            severity="high", category="cors", module="recon",
   362→   362→            description="Access-Control-Allow-Origin: * — any domain can make cross-origin requests",
   363→   363→            evidence=f"CORS: {acao}", asset="",
   364→   364→    ))
   365→   365→    elif acao:
   366→   366→        probe = http_probe(base_url, headers={"Origin": "https://evil-attacker.com"})
   367→   367→        reflected = probe.get("headers", {}).get("Access-Control-Allow-Origin", "")
   368→   368→        if "evil-attacker" in reflected:
   369→   369→            findings.append(Finding(
   370→   370→                title="CORS origin reflection",
   371→   371→                severity="critical", category="cors", module="recon",
   372→   372→                description="Server reflects any Origin header back",
   373→   373→                evidence=f"Sent Origin: evil-attacker.com, got: {reflected}", asset="",
   374→   374→    ))
   375→   375→    return findings

   378→   378→# ── Cookie security ───────────────────────────────────────────────────────
   379→   379→
   380→   380→def _check_cookies(headers: Dict[str, str]) -> List[Finding]:
   381→   381→    findings = []
   382→   382→    set_cookie = headers.get("set-cookie", "")
   383→   383→    if not set_cookie:
   384→   384→        return findings
   385→   385→    # Cookies may be separated by \n or \r\n in some HTTP libs; also by comma.
   386→   386→    # Split by common delimiters and process each individually.
   387→   387→    cookies = re.split(r'[\n\r]+', set_cookie)
   388→   388→    for raw_cookie in cookies:
   389→   389→        raw_cookie = raw_cookie.strip()
   390→   390→        if not raw_cookie or "=" not in raw_cookie:
   391→   391→            continue
   392→   392→        # Only take the part before any comma that might be part of same line
   393→   393→        cookie = raw_cookie.split(",")[0].strip()
   394→   394→        name = cookie.split("=")[0].strip()
   395→   395→        cookie_lower = cookie.lower()
   396→   396→
   397→   397→        if "httponly" not in cookie_lower and name:
   398→   398→            findings.append(Finding(
   399→   399→                title=f"Missing HttpOnly on cookie: {name[:40]}",
   400→   400→                severity="medium", category="cookies", module="recon",
   401→   401→                description=f"Cookie '{name}' lacks HttpOnly flag — vulnerable to XSS theft",
   402→   402→                evidence=f"Cookie: {cookie[:80]}", asset="",
   403→   403→        ))
   404→   404→        if "secure" not in cookie_lower and name:
   405→   405→            findings.append(Finding(
   406→   406→                title=f"Missing Secure flag on cookie: {name[:40]}",
   407→   407→                severity="medium", category="cookies", module="recon",
   408→   408→                description=f"Cookie '{name}' lacks Secure flag — sent over HTTP",
   409→   409→                evidence=f"Cookie: {cookie[:80]}", asset="",
   410→   410→        ))
   411→   411→        if "samesite" not in cookie_lower and name:
   412→   412→            findings.append(Finding(
   413→   413→                title=f"Missing SameSite on cookie: {name[:40]}",
   414→   414→                severity="low", category="cookies", module="recon",

   864→   864→def _check_meta_tags(body: str, host: str) -> List[Finding]:
   865→   865→    findings = []

   879→   879→

   885→   885→            severity="info", category="meta_tags", module="recon",

   887→   887→            evidence="No <meta name=viewport> found",
   888→   888→            asset=host,
   889→   889→        ))
   890→   890→

   897→   897→            severity="low", category="meta_tags", module="recon",

   902→   902→

   904→   904→    sensitive_metas = {

   916→   916→                evidence=f"<{meta_name}: {val[:60]}",
   917→   917→                asset=host, points_deducted=pts,
   918→   918→            ))
   919→   919→
   920→   920→    return findings
