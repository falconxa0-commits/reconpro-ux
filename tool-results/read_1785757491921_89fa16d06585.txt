   161→   161→
   162→   162→class BootScreen(Widget):
   163→   163→    """Full-screen animated boot sequence."""
   164→   164→
   165→   165→    def __init__(self) -> None:
   166→   166→        super().__init__()
   167→   167→        self._step = 0
   168→   168→        self._spinner_idx = 0
   169→   169→        self._module_idx = 0
   170→   170→        self._module_names = list(MODULE_NAMES.values())
   171→   171→        self._done = False
   172→   172→        self._timer: Optional[Timer] = None
   173→   173→
   174→   174→    def compose(self) -> ComposeResult:
   175→   175→        with Vertical(classes="boot-container"):
   176→   176→            yield Static("", id="boot-line-1", classes="boot-title")
   177→   177→            yield Static("", id="boot-line-2", classes="boot-line")
   178→   178→            yield Static("", id="boot-line-3", classes="boot-line")
   179→   179→            yield Static("", id="boot-line-4", classes="boot-line")
   180→   180→
   181→   181→    def on_mount(self) -> None:
   182→   182→        self._timer = self.set_interval(0.04, self._tick)
   183→   183→
   184→   184→    def _tick(self) -> None:
   185→   185→        app = self.app
   186→   186→        if not isinstance(app, NexusApp):
   187→   187→            return
   188→   188→
   189→   189→        line1 = app.query_one("#boot-line-1", Static)
   190→   190→        line2 = app.query_one("#boot-line-2", Static)
   191→   191→        line3 = app.query_one("#boot-line-3", Static)
   192→   192→        line4 = app.query_one("#boot-line-4", Static)
   193→   193→
   194→   194→        if self._step == 0:
   195→   195→            # Letter-by-letter title
   196→   196→            title = "RECONPRO NEXUS"
   197→   197→            chars = title[: self._spinner_idx]
   198→   198→            self._spinner_idx += 1
   199→   199→            line1.update(f"[{CYAN} bold]{chars}[/]")
   200→   200→            if self._spinner_idx > len(title):
   201→   201→                self._spinner_idx = 0
   202→   202→                self._step = 1
   203→   203→                line1.update(f"[{CYAN} bold]{title}[/]  [{DIM_CYAN}]v{VERSION}[/]")
   204→   204→
   205→   205→        elif self._step == 1:
   206→   206→            # Spinner for "Initializing agent..."
   207→   207→            frame = SPINNER_FRAMES[self._spinner_idx % len(SPINNER_FRAMES)]
   208→   208→            self._spinner_idx += 1
   209→   209→            line2.update(f"  [{CYAN}]{frame}[/] [{DIM_CYAN}]Initializing agent...[/]")
   210→   210→            if self._spinner_idx > 30:
   211→   211→                self._spinner_idx = 0
   212→   212→                self._step = 2
   213→   213→                line2.update(f"  [{GREEN}]✓[/] [{DIM_CYAN}]Agent initialized.[/]")
   214→   214→
   215→   215→        elif self._step == 2:
   216→   216→            # Module names appearing one by one
   217→   217→            if self._module_idx < len(self._module_names):
   218→   218→                visible = self._module_names[: self._module_idx + 1]
   219→   219→                self._module_idx += 1
   220→   220→                modules_str = "  ".join(f"[{CYAN}]○[/] [{DIM_CYAN}]{m}[/]" for m in visible)
   221→   221→                remaining = len(self._module_names) - self._module_idx
   222→   222→                if remaining > 0:
   223→   223→                    modules_str += f"  [{DIM_CYAN}]+{remaining} more...[/]"
   224→   224→                frame = SPINNER_FRAMES[self._spinner_idx % len(SPINNER_FRAMES)]
   225→   225→                self._spinner_idx += 1
   226→   226→                line3.update(
   227→   227→                    f"  [{CYAN}]{frame}[/] [{DIM_CYAN}]Loading {len(self._module_names)} modules...[/]\n  {modules_str}"
   228→   228→                )
   229→   229→            else:
   230→   230→                all_mods = "  ".join(
   231→   231→                    f"[{GREEN}]●[/] [{DIM_CYAN}]{m}[/]" for m in self._module_names
   232→   232→                )
   233→   233→                line3.update(
   234→   234→                    f"  [{GREEN}]✓[/] [{DIM_CYAN}]All {len(self._module_names)} modules loaded.[/]\n  {all_mods}"
   235→   235→                )
   236→   236→                self._step = 3
   237→   237→                self._spinner_idx = 0
   238→   238→
   239→   239→        elif self._step == 3:
   240→   240→            # "Ready." with flash
   241→   241→            if self._spinner_idx == 0:
   242→   242→                line4.update(f"[{YELLOW} bold]  ▶  READY.[/]")
   243→   243→            elif self._spinner_idx == 6:
   244→   244→                line4.update(f"[{GREEN} bold]  ▶  READY.[/]")
   245→   245→            elif self._spinner_idx >= 10:
   246→   246→                self._done = True
   247→   247→                if self._timer:
   248→   248→                    self._timer.stop()
   249→   249→                self.app.call_from_thread(self.app._finish_boot)
   250→   250→            self._spinner_idx += 1
   251→   251→

   254→   254→# Module Status Cell Widget
   255→   255→# ════════════════════════════════════════════════════════════════════════════════

   257→   257→
   258→   258→class ModuleCell(Static):
   259→   259→    """Single module status cell in the grid."""
   260→   260→
   261→   261→    def __init__(self, module_id: str, name: str) -> None:
   262→   262→        super().__init__()
   263→   263→        self.module_id = module_id
   264→   264→        self.module_name = name
   265→   265→        self.status = "idle"  # idle | scanning | done | error
   266→   266→        self.finding_count = 0
   267→   267→        self._spinner_idx = 0
   268→   268→        self._timer: Optional[Timer] = None
   269→   269→
   270→   270→    def on_mount(self) -> None:
   271→   271→        self._update_display()
   272→   272→        self._timer = self.set_interval(0.1, self._spin)
   273→   273→
   274→   274→    def _spin(self) -> None:
   275→   275→        if self.status == "scanning":
   276→   276→            self._spinner_idx = (self._spinner_idx + 1) % len(SPINNER_FRAMES)
   277→   277→            self._update_display()
   278→   278→
   279→   279→    def _update_display(self) -> None:
   280→   280→        name_short = self.module_name[:14]
   281→   281→        if self.status == "idle":
   282→   282→            icon = f"[{DIM_CYAN}]○[/]"
   283→   283→            status_text = f"[{DIM_CYAN}]idle[/]"
   284→   284→        elif self.status == "scanning":
   285→   285→            frame = SPINNER_FRAMES[self._spinner_idx]
   286→   286→            icon = f"[{CYAN}]{frame}[/]"
   287→   287→            status_text = f"[{CYAN}]scanning[/]"
   288→   288→        elif self.status == "done":
   289→   289→            icon = f"[{GREEN}]✓[/]"
   290→   290→            status_text = f"[{GREEN}]done ({self.finding_count})[/]"
   291→   291→        elif self.status == "error":
   292→   292→            icon = f"[{RED}]✗[/]"
   293→   293→            status_text = f"[{RED}]error[/]"
   294→   294→        else:
   295→   295→            icon = "○"
   296→   296→            status_text = "idle"
   297→   297→
   298→   298→        self.update(
   299→   299→            f"{icon} [{CYAN} bold]{name_short}[/]\n  {status_text}"
   300→   300→        )
   301→   301→
   302→   302→    def set_status(self, status: str, count: int = 0) -> None:
   303→   303→        self.status = status
   304→   304→        self.finding_count = count
   305→   305→        self._update_display()
   306→   306→
   307→   307→    def on_unmount(self) -> None:
   308→   308→        if self._timer:
   309→   309→            self._timer.stop()
   310→   310→

   317→   317→class StatusDot(Static):

   322→   322→    def __init__(self) -> None:

   728→   728→                if inp.has_focus:

   730→   730→                    inp.value = self._command_history[self._history_index]

   734→   734→            try:
   735→   735→                inp = self.query_one("#command-input", Input)
   736→   736→                if inp.has_focus:
   737→   737→                    if self._history_index < len(self._command_history) - 1:
   738→   738→                        self._history_index += 1
   739→   739→                        inp.value = self._command_history[self._history_index]
   740→   740→                    else:
   741→   741→                        self._history_index = len(self._command_history)

   969→   969→

   971→   971→        cmd = parts[0].lower()
   972→   972→        args = parts[1:]

   974→   974→        if cmd in ("quit", "q", "exit"):

   977→   977→
   978→   978→        if cmd == "clear":
   979→   979→            self._clear_all_feeds()
   980→   980→            return
   981→   981→
   982→   982→        if cmd == "help":
   983→   983→            self._show_help()
   984→   984→            return
   985→   985→
   986→   986→        if cmd == "scan":
   987→   987→            self._handle_scan(args)
   988→   988→            return
   989→   989→
   990→   990→        if cmd == "audit":
   991→   991→            self._handle_audit(args)
   992→   992→            return
   993→   993→
   994→   994→        if cmd == "dev":
   995→   995→            self._handle_dev(args)
   996→   996→            return
   997→   997→
   998→   998→        if cmd == "doctor":
   999→   999→            self._handle_doctor()
  1000→  1000→            return
  1001→  1001→
  1002→  1002→        if cmd == "blitz":
  1003→  1003→            self._handle_blitz(args)
  1004→  1004→            return
  1005→  1005→
  1006→  1006→        if cmd == "subdomains":
  1007→  1007→            self._handle_subdomains(args)
  1008→  1008→            return

  1219→  1219→    def _run_scan_worker(
  1220→  1220→        self,
  1221→  1221→        target: str,
  1222→  1222→        modules: Optional[List[str]] = None,
  1223→  1223→        is_local: bool = False,
  1224→  1224→    ) -> None:
  1225→  1225→        """Run a scan in a background thread."""
  1226→  1226→        self.call_from_thread(self.is_scanning.set, True)
  1227→  1227→
  1228→  1228→        # Set modules to scanning
  1229→  1229→        if modules:
  1230→  1230→            for m in modules:
  1231→  1231→                if m in self._module_cells:
  1232→  1232→                    self.call_from_thread(self._module_cells[m].set_status, "scanning")
  1233→  1233→        else:
  1234→  1234→            all_mods = (
  1235→  1235→                list(MODULE_NAMES.keys())
  1236→  1236→                if is_local
  1237→  1237→                else [k for k in MODULE_NAMES if k not in ("host", "dev", "doctor")]
  1238→  1238→            )
  1239→  1239→            for m in all_mods:
  1240→  1240→                if m in self._module_cells:
  1241→  1241→                    self.call_from_thread(self._module_cells[m].set_status, "scanning")
  1242→  1242→
  1243→  1243→        try:
  1244→  1244→            from .scanner import scan, audit_scan
  1245→  1245→            from .history import save_scan
  1246→  1246→
  1247→  1247→            self.call_from_thread(
  1248→  1248→                self._chat, f"[{DIM_CYAN}]  ⏳ Scan in progress...[/]"
  1249→  1249→            )
  1250→  1250→
  1251→  1251→            if is_local:
  1252→  1252→                result = audit_scan(target=target, modules=modules)
  1253→  1253→            else:
  1254→  1254→                result = scan(target=target, modules=modules)
  1255→  1255→
  1256→  1256→            # Process findings
  1257→  1257→            data = result.to_dict()
  1258→  1258→            self.call_from_thread(self._on_scan_complete, data, modules, is_local)
  1259→  1259→
  1260→  1260→            # Save to history
  1261→  1261→            try:
  1262→  1262→                save_scan(data, label="nexus")
  1263→  1263→            except Exception:
  1264→  1264→                pass
  1265→  1265→
  1266→  1266→        except Exception as e:
  1267→  1267→            self.call_from_thread(self._chat, f"[{RED}]  ✗ Scan error: {e}[/]")
  1268→  1268→            if modules:
  1269→  1269→                for m in modules:
  1270→  1270→                    if m in self._module_cells:
  1271→  1271→                        self.call_from_thread(self._module_cells[m].set_status, "error")
  1272→  1272→            else:
  1273→  1273→                self.call_from_thread(self._set_all_modules_status, "error")
  1274→  1274→
  1275→  1275→        finally:
  1276→  1276→            self.call_from_thread(self.is_scanning.set, False)

  1629→  1629→    def on_click(self, event: Click) -> None:
  1630→  1630→        """Open finding detail modal for the clicked finding.
  1631→  1631→
  1632→  1632→        Uses the vertical click position to determine which finding index
  1633→  1633→        the user clicked on, mapping it to _findings_list.
  1634→  1634→        """
  1635→  1635→        try:
  1636→  1636→            if (
  1637→  1637→                isinstance(event.widget, RichLog)
  1638→  1638→                and event.widget.id == "findings-feed"
  1639→  1639→                and self._findings_list
  1640→  1640→            ):
  1641→  1641→                # Get the line offset to determine which finding was clicked.
  1642→  1642→                # Each finding entry is 1 line in the RichLog.
  1643→  1643→                # We use the y coordinate relative to the widget to estimate the index.
  1644→  1644→                region = event.widget.region
  1645→  1645→                if region is not None:
  1646→  1646→                    # Calculate which line was clicked relative to the scroll position
  1647→  1647→                    scroll_offset = event.widget.scroll_y
  1648→  1648→                    click_y = event.y - region.y
  1649→  1649→                    # Each finding takes 1 line; estimate index from click position
  1650→  1650→                    # Account for scroll position
  1651→  1651→                    line_height = 1
  1652→  1652→                    if hasattr(event.widget, "_line_height"):
  1653→  1653→                        line_height = event.widget._line_height
  1654→  1654→                    clicked_index = int((click_y + scroll_offset * line_height) / line_height)
  1655→  1655→                    # Clamp to valid range
  1656→  1656→                    clicked_index = max(0, min(clicked_index, len(self._findings_list) - 1))
  1657→  1657→                    finding = self._findings_list[clicked_index]
  1658→  1658→                else:
  1659→  1659→                    finding = self._findings_list[-1]
  1660→  1660→                self.push_screen(FindingDetailModal(finding))
  1661→  1661→        except (NoMatches, Exception):
  1662→  1662→            pass
