   106→   106→def make_table(headers, rows, col_widths=None):
   107→   107→    """Create a styled table."""
   108→   108→    data = [headers] + rows
   109→   109→    if col_widths is None:
   110→   110→        col_widths = [None] * len(headers)
   111→   111→    t = Table(data, colWidths=col_widths, repeatRows=1)
   112→   112→    style_cmds = [
   113→   113→        ('BACKGROUND', (0, 0), (-1, 0), TABLE_HEADER_COLOR),
   114→   114→        ('TEXTCOLOR', (0, 0), (-1, 0), TABLE_HEADER_TEXT),
   115→   115→        ('FONTNAME', (0, 0), (-1, 0), 'DejaVuSans-Bold'),
   116→   116→        ('FONTSIZE', (0, 0), (-1, 0), 9),
   117→   117→        ('FONTNAME', (0, 1), (-1, -1), 'NotoSerifSC'),
   118→   118→        ('FONTSIZE', (0, 1), (-1, -1), 9),
   119→   119→        ('ALIGN', (0, 0), (-1, -1), 'LEFT'),
   120→   120→        ('VALIGN', (0, 0), (-1, -1), 'MIDDLE'),
   121→   121→        ('TOPPADDING', (0, 0), (-1, -1), 4),
   122→   122→        ('BOTTOMPADDING', (0, 0), (-1, -1), 4),
   123→   123→        ('LEFTPADDING', (0, 0), (-1, -1), 6),
   124→   124→        ('RIGHTPADDING', (0, 0), (-1, -1), 6),
   125→   125→        ('GRID', (0, 0), (-1, -1), 0.5, BORDER),
   126→   126→    ]
   127→   127→    for i in range(1, len(data)):
   128→   128→        bg = TABLE_ROW_ODD if i % 2 == 0 else TABLE_ROW_EVEN
   129→   129→        style_cmds.append(('BACKGROUND', (0, i), (-1, i), bg))
   130→   130→    t.setStyle(TableStyle(style_cmds))
   131→   131→    return t
