   626→   785→class RuleBasedPlanner:
   627→   786→    """Smart rule-based planner that classifies intent, generates plans,
   628→   787→    and adapts based on results — no LLM required.
   629→   788→
   630→   789→    The planner uses a weighted intent scoring system that considers:
   631→   790→      - Direct keyword matches with configurable weights
   632→   791→      - Modifier words that boost or reduce intent scores
   633→   792→      - Target type classification (domain, IP, local, codebase)
   634→   793→      - Proximity of intent words to target words
   635→   794→      - Prior context from persistent memory
   636→   795→    """
   637→   796→
   638→   797→    # Intent keyword weights: (word_pattern, base_weight)
   639→   798→    _INTENT_KEYWORDS: Dict[str, List[Tuple[str, float]]] = {
   640→   799→        "full_scan": [
   641→   800→            ("scan", 0.4), ("audit", 0.3), ("assess", 0.3), ("pentest", 0.5),
   642→   801→            ("security", 0.2), ("vuln", 0.4), ("test", 0.2),
   643→   802→        ],
   644→   803→        "subdomain_recon": [
   645→   804→            ("subdomain", 0.8), ("sub-domain", 0.8), ("attack surface", 0.6),
   646→   805→            ("enumerate", 0.5), ("discover", 0.3), ("surface map", 0.7),
   647→   806→        ],
   648→   807→        "local_audit": [
   649→   808→            ("machine", 0.6), ("laptop", 0.6), ("computer", 0.5), ("local", 0.6),
   650→   809→            ("my system", 0.7), ("this machine", 0.6), ("localhost", 0.7),
   651→   810→        ],
   652→   811→        "dev_security": [
   653→   812→            ("project", 0.5), ("codebase", 0.5), ("repo", 0.4), ("source code", 0.5),
   654→   813→            ("dependencies", 0.4), ("package", 0.3), ("code", 0.3),

   719→   878→    def classify_intent(self, goal: str) -> Dict[str, float]:
   720→   879→        """Score each intent based on the goal text.
   721→   880→
   722→   881→        Uses weighted keyword matching, proximity scoring, and modifiers.
   723→   882→        Returns a dict mapping intent names to float scores in [0, 1+].
   724→   883→        """
   725→   884→        goal_lower = goal.lower()
   726→   885→        # Pad for whole-word matching
   727→   886→        padded = f' {goal_lower} '
   728→   887→        scores: Dict[str, float] = {}
   729→   888→        words = goal_lower.split()
   730→   889→
   731→   890→        # Calculate global modifier
   732→   891→        modifier = 1.0
   733→   892→        for w in words:
   734→   893→            if w in self._AMPLIFIERS:
   735→   894→                modifier = max(modifier, self._AMPLIFIERS[w])
   736→   895→            if w in self._DAMPENERS:
   737→   896→                modifier = min(modifier, self._DAMPENERS[w])
   738→   897→
   739→   898→        for intent, keywords in self._INTENT_KEYWORDS.items():
   740→   899→            score = 0.0
   741→   900→            for pattern, weight in keywords:
   742→   901→                # Use whole-word matching to avoid false positives
   743→   902→                # (e.g., "repo" matching inside "report")
   744→   903→                if len(pattern.split()) > 1:
   745→   904→                    # Multi-word pattern: check substring (already specific enough)
   746→   905→                    matched = pattern in goal_lower
   747→   906→                else:
   748→   907→                    # Single-word pattern: check whole word with padding
   749→   908→                    matched = f' {pattern} ' in padded
   750→   909→                if matched:
   751→   910→                    # Boost if near a target-like word
   752→   911→                    proximity_boost = 1.0
   753→   912→                    pattern_idx = goal_lower.find(pattern)
   754→   913→                    for tw in (".com", ".io", ".net", ".org", "http", "localhost"):
   755→   914→                        tw_idx = goal_lower.find(tw)
   756→   915→                        if tw_idx >= 0 and abs(pattern_idx - tw_idx) < 40:
   757→   916→                            proximity_boost = 1.3
   758→   917→                            break
   759→   918→                    score += weight * proximity_boost
   760→   919→            scores[intent] = min(score * modifier, 2.0)
   761→   920→
   762→   921→        return scores

   764→   923→    def extract_targets(self, goal: str) -> List[str]:
   765→   924→        """Extract targets (domains, IPs, URLs, paths) from the goal."""
   766→   925→        urls = re.findall(r'https?://[\w\-._~:/?#\[\]@!$&\'()*+,;=%]+', goal)
   767→   926→        domains = re.findall(r'[\w][\w\-]*\.[\w]{2,}(?:\.[\w]{2,})?', goal)
   768→   927→        skip = {
   769→   928→            'the', 'and', 'for', 'with', 'all', 'of', 'to', 'from', 'that', 'this',
   770→   929→            'scan', 'audit', 'check', 'find', 'get', 'show', 'me', 'my',
   771→   930→        }
   772→   931→        targets = list(set(
   773→   932→            urls + [d for d in domains if d.lower() not in skip and len(d) > 4]
   774→   933→        ))
   775→   934→        return targets

   777→   936→    def detect_target_type(self, goal: str, targets: List[str]) -> str:
   778→   937→        """Classify the target type from context."""
   779→   938→        goal_lower = goal.lower()
   780→   939→        local_indicators = [
   781→   940→            'machine', 'laptop', 'computer', 'local', 'my system', 'this machine',
   782→   941→            'localhost', 'my mac', 'my pc', 'my server',
   783→   942→        ]
   784→   943→        for ind in local_indicators:
   785→   944→            if ind in goal_lower:
   786→   945→                return "local"
   787→   946→        dev_indicators = ['project', 'codebase', 'repo', 'source code', 'code review']
   788→   947→        padded = f' {goal_lower} '
   789→   948→        for ind in dev_indicators:
   790→   949→            if f' {ind} ' in padded and not targets:
   791→   950→                return "codebase"
   792→   951→        if targets:
   793→   952→            for t in targets:
   794→   953→                if t.startswith("http"):
   795→   954→                    return "remote"
   796→   955→                if re.match(r'^\d+\.\d+\.\d+\.\d+$', t):
   797→   956→                    return "remote"
   798→   957→            return "remote"
   799→   958→        return "unknown"

   802→   961→        """Detect specific module requests from the goal text."""

   974→  1133→    def adapt_plan(
   975→  1134→        self,
   976→  1135→        plan: List[Dict[str, Any]],
   977→  1136→        completed_results: List[Tuple[Dict, ToolResult]],
   978→  1137→        current_idx: int,
   979→  1138→    ) -> List[Dict[str, Any]]:
   980→  1139→        """Adapt the remaining plan based on results seen so far.
   981→  1140→
   982→  1141→        This is the 'intelligence' of the rule engine — it inspects intermediate
   983→  1142→        results and inserts new steps or modifies remaining ones.
   984→  1143→        """
   985→  1144→        additions: List[Dict[str, Any]] = []
   986→  1145→        domain = ""
   987→  1146→        last_score = None
   988→  1147→        has_critical = False
   989→  1148→        total_findings = 0
   990→  1149→        all_findings: List[Dict] = []
   991→  1150→        subdomains_found = False
   992→  1151→
   993→  1152→        for step, result in completed_results:
   994→  1153→            d = result.data
   995→  1154→            if isinstance(d, dict):
   996→  1155→                # Extract score from scan results
   997→  1156→                if "total_score" in d:
   998→  1157→                    last_score = d["total_score"]
   999→  1158→                if "vibesec_score" in d:
  1000→  1159→                    last_score = d["vibesec_score"]
  1001→  1160→                # Collect findings
  1002→  1161→                findings = d.get("findings", [])
  1003→  1162→                if isinstance(findings, list):
  1004→  1163→                    all_findings.extend(findings)
  1005→  1164→                    total_findings += len(findings)
  1006→  1165→                    for f in findings:
  1007→  1166→                        if isinstance(f, dict) and f.get("severity") in ("critical", "high"):
  1008→  1167→                            has_critical = True
  1009→  1168→                # Check for subdomains
  1010→  1169→                if "subdomains" in d and len(d["subdomains"]) > 3:
  1011→  1170→                    subdomains_found = True
  1012→  1171→                # Get target from data
  1013→  1172→                if not domain:
  1014→  1173→                    domain = d.get("target", "")
  1015→  1174→
  1016→  1175→        # If any findings found → add CVE enrichment (if not already planned)
  1017→  1176→        has_any_findings = total_findings > 0
  1018→  1177→        if has_any_findings and not any(s["tool"] == "cve_enrich" for s in plan[current_idx:]):
  1019→  1178→            if all_findings:
  1020→  1179→                try:
  1021→  1180→                    findings_json = json.dumps(all_findings[:20])
  1022→  1181→                except (TypeError, ValueError):
  1023→  1182→                    findings_json = "[]"
  1024→  1183→                reason = "Enriching findings with CVE data from NVD"
  1025→  1184→                if has_critical:
  1026→  1185→                    reason = "Critical/high findings detected — enriching with CVE data"
  1027→  1186→                additions.append({
  1028→  1187→                    "tool": "cve_enrich",
  1029→  1188→                    "args": {"findings_json": findings_json},
  1030→  1189→                    "reason": reason,
  1031→  1190→                })
  1032→  1191→
  1033→  1192→        # If many subdomains found and not already doing blitz → suggest blitz
  1034→  1193→        if subdomains_found and not any(s["tool"] == "blitz_scan" for s in plan[current_idx:]):
  1035→  1194→            # Don't add blitz automatically (too aggressive), but add knowledge graph
  1036→  1195→            if domain and not any(s["tool"] == "knowledge_graph" for s in plan[current_idx:]):
  1037→  1196→                additions.append({
  1038→  1197→                    "tool": "knowledge_graph",
  1039→  1198→                    "args": {"action": "build", "target": domain},
  1040→  1199→                    "reason": "Large attack surface detected — building knowledge graph",
  1041→  1200→                })
  1042→  1201→
  1043→  1202→        # If score is very low → add knowledge graph
  1044→  1203→        if last_score is not None and last_score < 40:
  1045→  1204→            if domain and not any(s["tool"] == "knowledge_graph" for s in plan[current_idx:]):
  1046→  1205→                additions.append({
  1047→  1206→                    "tool": "knowledge_graph",
  1048→  1207→                    "args": {"action": "build", "target": domain},
  1049→  1208→                    "reason": f"Critical score ({last_score}/100) — analyzing attack surface via knowledge graph",
  1050→  1209→                })
  1051→  1210→
