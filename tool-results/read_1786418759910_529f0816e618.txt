   353→# ---------------------------------------------------------------------------
   354→# Attack Tree
   355→# ---------------------------------------------------------------------------
   356→
   357→class AttackTree:
   358→    """Generates an attack tree representation from kill-chain findings.
   359→
   360→    Each node in the tree represents a potential action an attacker could take.
   361→    The tree is structured as an OR-tree where any child node represents an
   362→    alternative path toward achieving the parent's goal.
   363→    """
   364→
   365→    def __init__(self) -> None:
   366→        self.root: Dict[str, Any] = {
   367→            "goal": "Compromise Target",
   368→            "type": "or",
   369→            "children": [],
   370→        }
   371→        self._all_techniques: List[str] = []
   372→        self._total_paths: int = 0
   373→
   374→    def build_from_phases(self, phases: Dict[str, KillChainPhase]) -> None:
   375→        """Construct the attack tree from completed phase data.
   376→
   377→        Args:
   378→            phases: Mapping of phase_id -> KillChainPhase results.
   379→        """
   380→        for phase_id in PHASE_NAMES:
   381→            phase = phases.get(phase_id)
   382→            if phase is None or phase.risk_score < 1.0:
   383→                continue
   384→            node: Dict[str, Any] = {
   385→                "goal": phase.label,
   386→                "phase": phase_id,
   387→                "type": "or",
   388→                "risk_score": phase.risk_score,
   389→                "risk_level": phase.risk_level,
   390→                "mitre": phase.mitre_techniques,
   391→                "children": [],
   392→            }
   393→            for finding in phase.findings:
   394→                child: Dict[str, Any] = {
   395→                    "goal": finding.get("title", finding.get("description", "Unknown")),
   396→                    "type": "leaf",
   397→                    "risk_score": finding.get("risk_score", 0.0),
   398→                    "technique": finding.get("technique", ""),
   399→                    "details": finding,
   400→                }
   401→                node["children"].append(child)
   402→                if finding.get("technique"):
   403→                    self._all_techniques.append(finding["technique"])
   404→            if node["children"]:
   405→                self.root["children"].append(node)
   406→                self._total_paths += len(node["children"])
   407→
   408→    def get_max_depth(self) -> int:
   409→        """Return the depth of the deepest branch."""
   410→        def _depth(node: Dict[str, Any]) -> int:
   411→            children = node.get("children", [])
   412→            if not children:
   413→                return 1
   414→            return 1 + max(_depth(c) for c in children)
   415→        return _depth(self.root)
   416→
   417→    def to_dict(self) -> Dict[str, Any]:
   418→        """Serialise the attack tree."""
   419→        return {
   420→            "root": self.root,
   421→            "total_paths": self._total_paths,
   422→            "total_techniques": len(set(self._all_techniques)),
   423→            "unique_techniques": sorted(set(self._all_techniques)),
   424→            "max_depth": self.get_max_depth(),
   425→        }

   428→# ---------------------------------------------------------------------------
   429→# Threat Matrix
   430→# ---------------------------------------------------------------------------
   431→
   432→class ThreatMatrix:
   433→    """Generates a combined threat-matrix view from all kill-chain phases.
   434→
   435→    The matrix cross-references MITRE ATT&CK tactics with detected
   436→    techniques and their associated risk scores, providing a unified
   437→    view of the threat landscape for the assessed target.
   438→    """
   439→
   440→    def __init__(self) -> None:
   441→        self.matrix: Dict[str, Dict[str, Any]] = {}
   442→        self._phase_scores: Dict[str, float] = {}
   443→
   444→    def ingest_phases(self, phases: Dict[str, KillChainPhase]) -> None:
   445→        """Ingest phase results into the threat matrix.
   446→
   447→        Args:
   448→            phases: Mapping of phase_id -> KillChainPhase.
   449→        """
   450→        for phase_id, phase in phases.items():
   451→            self._phase_scores[phase_id] = phase.risk_score
   452→            tactic = PHASE_LABELS.get(phase_id, phase_id)
   453→            if tactic not in self.matrix:
   454→                self.matrix[tactic] = {
   455→                    "tactic_id": phase_id,
   456→                    "techniques": [],
   457→                    "findings_count": 0,
   458→                    "avg_risk": 0.0,
   459→                    "max_risk": 0.0,
   460→                    "mitre_techniques": [],
   461→                }
   462→            bucket = self.matrix[tactic]
   463→            bucket["findings_count"] += len(phase.findings)
   464→            scores = [f.get("risk_score", 0.0) for f in phase.findings if f.get("risk_score")]
   465→            if scores:
   466→                bucket["avg_risk"] = round(sum(scores) / len(scores), 2)
   467→                bucket["max_risk"] = round(max(scores), 2)
   468→            for finding in phase.findings:
   469→                bucket["techniques"].append({
   470→                    "title": finding.get("title", "Unknown"),
   471→                    "risk_score": finding.get("risk_score", 0.0),
   472→                    "description": finding.get("description", ""),
   473→                })
   474→            bucket["mitre_techniques"] = phase.mitre_techniques
   475→
   476→    @property
   477→    def overall_risk(self) -> float:
   478→        """Calculate an overall risk score across all tactics (0–10)."""
   479→        if not self._phase_scores:
   480→            return 0.0
   481→        phase_weights = {
   482→            "reconnaissance": 0.05,
   483→            "weaponization": 0.10,
   484→            "delivery": 0.15,
   485→            "exploitation": 0.25,
   486→            "installation": 0.20,
   487→            "command_and_control": 0.15,
   488→            "actions_on_objectives": 0.10,
   489→        }
   490→        total = 0.0
   491→        for pid, score in self._phase_scores.items():
   492→            w = phase_weights.get(pid, 0.1)
   493→            total += score * w
   494→        return round(total, 2)
   495→
   496→    @property
   497→    def chain_completion_pct(self) -> float:
   498→        """Percentage of kill-chain phases that produced non-zero findings."""
   499→        active = sum(1 for s in self._phase_scores.values() if s > 0)
   500→        return round((active / len(PHASE_NAMES)) * 100, 1) if PHASE_NAMES else 0.0
   501→
   502→    def to_dict(self) -> Dict[str, Any]:
   503→        """Serialise the threat matrix."""
   504→        return {
   505→            "matrix": self.matrix,
   506→            "overall_risk": self.overall_risk,
   507→            "chain_completion_pct": self.chain_completion_pct,
   508→            "phase_scores": self._phase_scores,
   509→            "risk_level": self._classify_risk(self.overall_risk),
   510→        }
   511→
   512→    @staticmethod
   513→    def _classify_risk(score: float) -> str:
   514→        if score >= RISK_CRITICAL:
   515→            return RiskLevel.CRITICAL.value
   516→        if score >= RISK_HIGH:
   517→            return RiskLevel.HIGH.value
   518→        if score >= RISK_MEDIUM:
   519→            return RiskLevel.MEDIUM.value
   520→        if score >= RISK_LOW:
   521→            return RiskLevel.LOW.value
   522→        return RiskLevel.INFO.value
   523→

   529→def _classify_score(score: float) -> str:
   530→    """Map a numeric score (0–10) to a risk-level string."""
   531→    if score >= RISK_CRITICAL:
   532→        return RiskLevel.CRITICAL.value
   533→    if score >= RISK_HIGH:
   534→        return RiskLevel.HIGH.value
   535→    if score >= RISK_MEDIUM:
   536→        return RiskLevel.MEDIUM.value
   537→    if score >= RISK_LOW:
   538→        return RiskLevel.LOW.value
   539→    return RiskLevel.INFO.value

   550→    finding: Dict[str, Any] = {

   552→        "description": description,
   553→        "risk_score": risk_score,
   554→        "risk_level": _classify_score(risk_score),
   555→        "technique": technique,
   556→    }
   557→    if data:
   558→        finding["data"] = data
   559→    return finding

   561→
   562→def _http_fetch(
   563→    url: str,
   564→    timeout: int = 8,
   565→    verify_tls: bool = True,
   566→) -> Tuple[int, Dict[str, str], bytes]:
   567→    """Fetch a URL and return (status_code, headers_dict, body_bytes).
   568→
   569→    Returns (-1, {}, b"") on failure.
   570→    """
   571→    try:
   572→        ctx = None if verify_tls else ssl._create_unverified_context()
   573→        req = urllib.request.Request(url, headers={
   574→            "User-Agent": "Mozilla/5.0 (ReconPro/9.2.0; SecurityScanner)",
   575→            "Accept": "*/*",
   576→        })
   577→        resp = urllib.request.urlopen(req, timeout=timeout, context=ctx)
   578→        hdrs = {k.lower(): v for k, v in resp.getheaders()}
   579→        body = resp.read()
   580→        return resp.getcode(), hdrs, body
   581→    except (urllib.error.URLError, socket.timeout, OSError, ssl.SSLError):
   582→        return -1, {}, b""

   584→
   585→def _dns_resolve(hostname: str, timeout: float = 4.0) -> List[str]:
   586→    """Resolve hostname via socket.getaddrinfo. Returns list of IP strings."""
   587→    results: List[str] = []
   588→    try:
   589→        ai_list = socket.getaddrinfo(hostname, None, socket.AF_UNSPEC, socket.SOCK_STREAM)
   590→        seen: set = set()
   591→        for family, _stype, _proto, _canonname, sockaddr in ai_list:
   592→            ip = sockaddr[0]
   593→            if ip not in seen:
   594→                seen.add(ip)
   595→                results.append(ip)
   596→    except socket.gaierror:
   597→        pass
   598→    return results

   601→def _reverse_dns(ip: str, timeout: float = 4.0) -> str:
   602→    """Perform reverse DNS lookup for an IP address."""
   603→    try:
   604→        hostname, _, _ = socket.gethostbyaddr(ip)
   605→        return hostname
   606→    except (socket.herror, socket.gaierror, OSError):
   607→        return ""

   610→def _run_subprocess(cmd: List[str], timeout: int = 10) -> str:
   611→    """Run a subprocess command and return stdout. Returns '' on error."""
   612→    try:
   613→        result = subprocess.run(
   614→            cmd, capture_output=True, text=True, timeout=timeout,
   615→        )
   616→        return (result.stdout or "") + (result.stderr or "")
   617→    except (subprocess.TimeoutExpired, OSError, FileNotFoundError):
   618→        return ""

   621→def _extract_domain_from_target(target: str) -> str:
   622→    """Normalise a target string to a bare domain name."""
   623→    t = target.strip().lower()
   624→    t = re.sub(r"^https?://", "", t)
   625→    t = re.sub(r"[/:].*$", "", t)
   626→    if not t:
   627→        return "localhost"
   628→    return t

   636→def _calculate_entropy(data: str) -> float:
   637→    """Calculate Shannon entropy of a string (useful for detecting randomness)."""
   638→    if not data:
   639→        return 0.0
   640→    freq: Dict[str, int] = {}
   641→    for ch in data:
   642→        freq[ch] = freq.get(ch, 0) + 1
   643→    length = len(data)
   644→    ent = 0.0
   645→    for count in freq.values():
   646→        p = count / length
   647→        if p > 0:
   648→            ent -= p * (p.bit_length() - 1).bit_length() / (p.bit_length())
   649→            ent -= p * (p.bit_length() - 1)
   650→    import math
   651→    for count in freq.values():
   652→        p = count / length
   653→        if p > 0:
   654→            ent -= p * math.log2(p)
   655→    return round(ent, 3)

   662→class KillChainEngine:
   663→    """Lockheed Martin Cyber Kill Chain automation engine.
   664→
   665→    Executes all seven phases sequentially against a target, collects
   666→    findings, maps them to MITRE ATT&CK techniques, and produces a
   667→    unified threat-matrix view with attack-tree representation.
   668→
   669→    Usage::
   670→
   671→        engine = KillChainEngine()
   672→        result = engine.run_kill_chain("example.com", "https://example.com")
   673→    """
   674→
   675→    def __init__(self, timeout: int = 8, verify_tls: bool = True) -> None:
   676→        self.timeout: int = timeout
   677→        self.verify_tls: bool = verify_tls
   678→        self._target: str = ""
   679→        self._base_url: str = ""
   680→        self._domain: str = ""
   681→        self._ips: List[str] = []
   682→        self._headers: Dict[str, str] = {}
   683→        self._body_text: str = ""
   684→        self._body_bytes: bytes = b""
   685→        self._status_code: int = -1
   686→        self._phases: Dict[str, KillChainPhase] = {}

   688→    # -----------------------------------------------------------------------
   689→    # Public API
   690→    # -----------------------------------------------------------------------
   691→
   692→    def run_kill_chain(
   693→        self,
   694→        target: str,
   695→        base_url: str,
   696→        timeout: int = 8,
   697→        verify_tls: bool = True,
   698→    ) -> Dict[str, Any]:
   699→        """Execute the full 7-phase kill chain against a target.
   700→
   701→        Args:
   702→            target: The target domain or IP (e.g. ``"example.com"``).
   703→            base_url: The base URL to fetch for analysis
   704→                       (e.g. ``"https://example.com"``).
   705→            timeout: Network operation timeout in seconds.
   706→            verify_tls: Whether to verify TLS certificates.
   707→
   708→        Returns:
   709→            A dictionary with keys ``phases``, ``attack_tree``,
   710→            ``threat_matrix``, ``overall_risk``, and ``kill_chain_completion``.
   711→        """
   712→        self.timeout = timeout
   713→        self.verify_tls = verify_tls
   714→        self._target = target
   715→        self._base_url = base_url.rstrip("/")
   716→        self._domain = _extract_domain_from_target(target)
   717→
   718→        # Initial reconnaissance fetch
   719→        self._initial_fetch()
   720→
   721→        # Execute all seven phases in order
   722→        self._phases["reconnaissance"] = self._phase_reconnaissance()
   723→        self._phases["weaponization"] = self._phase_weaponization()
   724→        self._phases["delivery"] = self._phase_delivery()
   725→        self._phases["exploitation"] = self._phase_exploitation()
   726→        self._phases["installation"] = self._phase_installation()
   727→        self._phases["command_and_control"] = self._phase_command_and_control()
   728→        self._phases["actions_on_objectives"] = self._phase_actions_on_objectives()
   729→
   730→        # Build the attack tree
   731→        tree = AttackTree()
   732→        tree.build_from_phases(self._phases)
   733→
   734→        # Build the threat matrix
   735→        matrix = ThreatMatrix()
   736→        matrix.ingest_phases(self._phases)
   737→
   738→        return {
   739→            "target": self._target,
   740→            "domain": self._domain,
   741→            "base_url": self._base_url,
   742→            "ips": self._ips,
   743→            "phases": {pid: p.to_dict() for pid, p in self._phases.items()},
   744→            "attack_tree": tree.to_dict(),
   745→            "threat_matrix": matrix.to_dict(),
   746→            "overall_risk": matrix.overall_risk,
   747→            "kill_chain_completion": matrix.chain_completion_pct,
   748→        }

   754→    def _initial_fetch(self) -> None:
   755→        """Fetch the base URL to capture headers, body, and resolve IPs."""
   756→        self._ips = _dns_resolve(self._domain, timeout=float(self.timeout))

   902→        findings: List[Dict[str, Any]] = []

   905→        all_payloads: Dict[str, List[str]] = {}
   906→        exploitability_scores: Dict[str, float] = {}

   910→        for tech_name, tech_info in TECH_SIGNATURES.items():
   911→            if tech_name in detected:
   912→                tech_count += 1
   913→                base_score = tech_info.get("exploitability_base", 5.0)
   914→                exploit_payloads = tech_info.get("exploits", {})

   920→                    for hdr_key, hdr_val in self._headers.items():
   921→                        if sig_lower in (hdr_key + ": " + hdr_val).lower() or sig_lower in hdr_val.lower():
   922→                            header_match = True

   935→                if header_match:
   936→                    confidence += 0.6
   937→                if body_match:
   938→                    confidence += 0.4
