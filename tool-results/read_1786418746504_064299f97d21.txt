    41→    41→def _b64url_encode(data: bytes) -> str:
    42→    42→    """Base64url-encode bytes (no padding)."""
    43→    43→    return base64.urlsafe_b64encode(data).rstrip(b"=").decode("ascii")

    45→    45→
    46→    46→def _b64url_decode(s: str) -> bytes:
    47→    47→    """Base64url-decode with padding restoration."""
    48→    48→    padding = 4 - len(s) % 4
    49→    49→    if padding != 4:
    50→    50→        s += "=" * padding
    51→    51→    return base64.urlsafe_b64decode(s)

    54→    54→def _build_jwt(header: Dict[str, Any], payload: Dict[str, Any],
    55→    55→               secret: Optional[bytes] = None) -> str:
    56→    56→    """Build a JWT token. If secret is None, no signature (alg=none)."""
    57→    57→    h = _b64url_encode(json.dumps(header, separators=(",", ":")).encode())
    58→    58→    p = _b64url_encode(json.dumps(payload, separators=(",", ":")).encode())
    59→    59→    signing_input = f"{h}.{p}".encode()
    60→    60→    if secret is None:
    61→    61→        return f"{h}.{p}."
    62→    62→    sig = hmac.new(secret, signing_input, hashlib.sha256).digest()
    63→    63→    return f"{h}.{p}.{_b64url_encode(sig)}"

    88→    88→def _extract_bearer_token(resp: Dict[str, Any]) -> Optional[str]:
    89→    89→    """Extract a JWT Bearer token from response headers or body."""
    90→    90→    # Check WWW-Authenticate / Set-Cookie / response body for tokens
    91→    91→    headers = resp.get("headers", {})
    92→    92→    auth_header = headers.get("www-authenticate", "")
    93→    93→    if auth_header.startswith("Bearer "):
    94→    94→        return auth_header[7:].split(",")[0].strip()  # strip params
    95→    95→    # Check body for jwt patterns
    96→    96→    body = resp.get("body", "")
    97→    97→    import re
    98→    98→    patterns = [
    99→    99→        r'"token"\s*:\s*"(eyJ[A-Za-z0-9_-]+\.eyJ[A-Za-z0-9_-]+\.[A-Za-z0-9_-]+)"',
   100→   100→        r'"access_token"\s*:\s*"(eyJ[A-Za-z0-9_-]+\.eyJ[A-Za-z0-9_-]+\.[A-Za-z0-9_-]+)"',
   101→   101→        r'"id_token"\s*:\s*"(eyJ[A-Za-z0-9_-]+\.eyJ[A-Za-z0-9_-]+\.[A-Za-z0-9_-]+)"',
   102→   102→    ]
   103→   103→    for pat in patterns:
   104→   104→        m = re.search(pat, body)
   105→   105→        if m:
   106→   106→            return m.group(1)
   107→   107→    return None
