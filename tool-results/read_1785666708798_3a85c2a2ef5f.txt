    91→    91→class FindingDetailModal(ModalScreen):
    92→    92→    """Modal overlay showing full finding details."""
    93→    93→
    94→    94→    def __init__(self, finding: Dict[str, Any]) -> None:
    95→    95→        super().__init__()
    96→    96→        self.finding = finding
    97→    97→
    98→    98→    def compose(self) -> ComposeResult:
    99→    99→        f = self.finding
   100→   100→        sev = f.get("severity", "info").lower()
   101→   101→        sev_color = SEV_STYLES.get(sev, "white")
   102→   102→        title = f.get("title", "Unknown")
   103→   103→        category = f.get("category", "")
   104→   104→        module = f.get("module", "")
   105→   105→        pts = f.get("points_deducted", 0)
   106→   106→        evidence = f.get("evidence", "")
   107→   107→        remediation = f.get("remediation", "No remediation available.")
   108→   108→        description = f.get("description", "")
   109→   109→        asset = f.get("asset", "")
   110→   110→        related_cves = f.get("related_cves", [])
   111→   111→
   112→   112→        lines: List[str] = []
   113→   113→        lines.append(f"[{sev_color}][{sev.upper()}] {title}[/]")
   114→   114→        lines.append("")
   115→   115→        if asset:
   116→   116→            lines.append(f"[dim]Asset:[/] [cyan]{asset}[/]")
   117→   117→        if category:
   118→   118→            lines.append(f"[dim]Category:[/] {category}")
   119→   119→        if module:
   120→   120→            lines.append(f"[dim]Module:[/] {module}")
   121→   121→        lines.append(f"[dim]Points:[/] -{pts}")
   122→   122→        if related_cves:
   123→   123→            cve_ids = ", ".join(
   124→   124→                c.get("cve_id", "") for c in related_cves[:5] if c.get("cve_id")
   125→   125→            )
   126→   126→            if cve_ids:
   127→   127→                lines.append(f"[dim]CVEs:[/] [red]{cve_ids}[/]")
   128→   128→        lines.append("")
   129→   129→        if description:
   130→   130→            lines.append(f"[bold]Description:[/]")
   131→   131→            lines.append(f"[dim]{description}[/]")
   132→   132→            lines.append("")
   133→   133→        if evidence:
   134→   134→            lines.append(f"[bold]Evidence:[/]")
   135→   135→            lines.append(f"[dim]{evidence}[/]")
   136→   136→            lines.append("")
   137→   137→        lines.append(f"[bold]{GREEN}Remediation:[/]")
   138→   138→            lines.append(f"{GREEN}{remediation}[/]")
   139→   139→
   140→   140→        content = "\n".join(lines)
   141→   141→
   142→   142→        with Vertical(classes="modal-container"):
   143→   143→            yield Label(content, classes="modal-content")
   144→   144→            with Horizontal(classes="modal-actions"):
   145→   145→                yield Button("Close", variant="primary", id="close-modal")
   146→   146→
   147→   147→    def on_button_pressed(self, event: Button.Pressed) -> None:
   148→   148→        if event.button.id == "close-modal":
   149→   149→            self.dismiss()

   153→   153→# Boot Screen Widget
   154→   154→# ════════════════════════════════════════════════════════════════════════════════
   155→   155→
   156→   156→
   157→   157→class BootScreen(Widget):
   158→   158→    """Full-screen animated boot sequence."""
   159→   159→
   160→   160→    def __init__(self) -> None:
   161→   161→        super().__init__()
   162→   162→        self._step = 0
   163→   163→        self._spinner_idx = 0
   164→   164→        self._module_idx = 0
   165→   165→        self._module_names = list(MODULE_NAMES.values())
   166→   166→        self._done = False
   167→   167→        self._timer: Optional[Timer] = None
   168→   168→
   169→   169→    def compose(self) -> ComposeResult:
   170→   170→        with Vertical(classes="boot-container"):
   171→   171→            yield Static("", id="boot-line-1", classes="boot-title")
   172→   172→            yield Static("", id="boot-line-2", classes="boot-line")
   173→   173→            yield Static("", id="boot-line-3", classes="boot-line")
   174→   174→            yield Static("", id="boot-line-4", classes="boot-line")
   175→   175→
   176→   176→    def on_mount(self) -> None:
   177→   177→        self._timer = self.set_interval(0.04, self._tick)
   178→   178→
   179→   179→    def _tick(self) -> None:
   180→   180→        app = self.app
   181→   181→        if not isinstance(app, NexusApp):
   182→   182→            return
   183→   183→
   184→   184→        line1 = app.query_one("#boot-line-1", Static)
   185→   185→        line2 = app.query_one("#boot-line-2", Static)
   186→   186→        line3 = app.query_one("#boot-line-3", Static)
   187→   187→        line4 = app.query_one("#boot-line-4", Static)
   188→   188→
   189→   189→        if self._step == 0:
   190→   190→            # Letter-by-letter title
   191→   191→            title = "RECONPRO NEXUS"
   192→   192→            chars = title[: self._spinner_idx]
   193→   193→            self._spinner_idx += 1
   194→   194→            line1.update(f"[{CYAN} bold]{chars}[/]")
   195→   195→            if self._spinner_idx > len(title):
   196→   196→                self._spinner_idx = 0
   197→   197→                self._step = 1
   198→   198→                line1.update(f"[{CYAN} bold]{title}[/]  [{DIM_CYAN}]v{VERSION}[/]")
   199→   199→
   200→   200→        elif self._step == 1:
   201→   201→            # Spinner for "Initializing agent..."
   202→   202→            frame = SPINNER_FRAMES[self._spinner_idx % len(SPINNER_FRAMES)]
   203→   203→            self._spinner_idx += 1
   204→   204→            line2.update(f"  [{CYAN}]{frame}[/] [{DIM_CYAN}]Initializing agent...[/]")
   205→   205→            if self._spinner_idx > 30:
   206→   206→                self._spinner_idx = 0
   207→   207→                self._step = 2
   208→   208→                line2.update(f"  [{GREEN}]✓[/] [{DIM_CYAN}]Agent initialized.[/]")
   209→   209→
   210→   210→        elif self._step == 2:
   211→   211→            # Module names appearing one by one
   212→   212→            if self._module_idx < len(self._module_names):
   213→   213→                visible = self._module_names[: self._module_idx + 1]
   214→   214→                self._module_idx += 1
   215→   215→                modules_str = "  ".join(f"[{CYAN}]○[/] [{DIM_CYAN}]{m}[/]" for m in visible)
   216→   216→                remaining = len(self._module_names) - self._module_idx
   217→   217→                if remaining > 0:
   218→   218→                    modules_str += f"  [{DIM_CYAN}]+{remaining} more...[/]"
   219→   219→                frame = SPINNER_FRAMES[self._spinner_idx % len(SPINNER_FRAMES)]
   220→   220→                self._spinner_idx += 1
   221→   221→                line3.update(
   222→   222→                    f"  [{CYAN}]{frame}[/] [{DIM_CYAN}]Loading {len(self._module_names)} modules...[/]\n  {modules_str}"
   223→   223→                )
   224→   224→            else:
   225→   225→                all_mods = "  ".join(
   226→   226→                    f"[{GREEN}]●[/] [{DIM_CYAN}]{m}[/]" for m in self._module_names
   227→   227→                )
   228→   228→                line3.update(
   229→   229→                    f"  [{GREEN}]✓[/] [{DIM_CYAN}]All {len(self._module_names)} modules loaded.[/]\n  {all_mods}"
   230→   230→                )
   231→   231→                self._step = 3
   232→   232→                self._spinner_idx = 0
   233→   233→
   234→   234→        elif self._step == 3:
   235→   235→            # "Ready." with flash
   236→   236→            if self._spinner_idx == 0:
   237→   237→                line4.update(f"[{YELLOW} bold]  ▶  READY.[/]")
   238→   238→            elif self._spinner_idx == 6:
   239→   239→                line4.update(f"[{GREEN} bold]  ▶  READY.[/]")
   240→   240→            elif self._spinner_idx >= 10:
   241→   241→                self._done = True
   242→   242→                if self._timer:
   243→   243→                    self._timer.stop()
   244→   244→                self.app.call_from_thread(self.app._finish_boot)
   245→   245→            self._spinner_idx += 1
   246→   246→
   247→   247→

   249→   249→# Module Status Cell Widget
   250→   250→# ════════════════════════════════════════════════════════════════════════════════
   251→   251→
   252→   252→
   253→   253→class ModuleCell(Static):
   254→   254→    """Single module status cell in the grid."""
   255→   255→
   256→   256→    def __init__(self, module_id: str, name: str) -> None:
   257→   257→        super().__init__()
   258→   258→        self.module_id = module_id
   259→   259→        self.module_name = name
   260→   260→        self.status = "idle"  # idle | scanning | done | error
   261→   261→        self.finding_count = 0
   262→   262→        self._spinner_idx = 0
   263→   263→        self._timer: Optional[Timer] = None
   264→   264→
   265→   265→    def on_mount(self) -> None:
   266→   266→        self._update_display()
   267→   267→        self._timer = self.set_interval(0.1, self._spin)
   268→   268→
   269→   269→    def _spin(self) -> None:
   270→   270→        if self.status == "scanning":
   271→   271→            self._spinner_idx = (self._spinner_idx + 1) % len(SPINNER_FRAMES)
   272→   272→            self._update_display()
   273→   273→
   274→   274→    def _update_display(self) -> None:
   275→   275→        name_short = self.module_name[:14]
   276→   276→        if self.status == "idle":
   277→   277→            icon = f"[{DIM_CYAN}]○[/]"
   278→   278→            status_text = f"[{DIM_CYAN}]idle[/]"
   279→   279→        elif self.status == "scanning":
   280→   280→            frame = SPINNER_FRAMES[self._spinner_idx]
   281→   281→            icon = f"[{CYAN}]{frame}[/]"
   282→   282→            status_text = f"[{CYAN}]scanning[/]"
   283→   283→        elif self.status == "done":
   284→   284→            icon = f"[{GREEN}]✓[/]"
   285→   285→            status_text = f"[{GREEN}]done ({self.finding_count})[/]"
   286→   286→        elif self.status == "error":
   287→   287→            icon = f"[{RED}]✗[/]"
   288→   288→            status_text = f"[{RED}]error[/]"
   289→   289→        else:
   290→   290→            icon = "○"
   291→   291→            status_text = "idle"
   292→   292→
   293→   293→        self.update(
   294→   294→            f"{icon} [{CYAN} bold]{name_short}[/]\n  {status_text}"
   295→   295→        )
   296→   296→
   297→   297→    def set_status(self, status: str, count: int = 0) -> None:
   298→   298→        self.status = status
   299→   299→        self.finding_count = count
   300→   300→        self._update_display()
   301→   301→
   302→   302→    def on_unmount(self) -> None:
   303→   303→        if self._timer:
   304→   304→            self._timer.stop()
   305→   305→
