    16→// ── SSRF Protection: Block private/reserved IP ranges ──────────────
    17→function isPrivateIP(ip: string): boolean {
    18→  const parts = ip.split('.').map(Number);
    19→  if (parts.length !== 4) return true; // reject non-IPv4
    20→  const [a, b] = parts;
    21→  // RFC 1918 private ranges
    22→  if (a === 10) return true;
    23→  if (a === 172 && b >= 16 && b <= 31) return true;
    24→  if (a === 192 && b === 168) return true;
    25→  // Loopback
    26→  if (a === 127) return true;
    27→  // Link-local
    28→  if (a === 169 && b === 254) return true; // includes cloud metadata 169.254.169.254
    29→  // Carrier-grade NAT
    30→  if (a === 100 && b >= 64 && b <= 127) return true;
    31→  // Documentation/benchmark ranges
    32→  if (a === 192 && b === 0 && parts[2] === 2) return true;
    33→  if (a === 198 && b === 51 && parts[2] === 100) return true;
    34→  if (a === 203 && b === 0 && parts[2] === 113) return true;
    35→  // Multicast
    36→  if (a >= 224 && a <= 239) return true;
    37→  // Reserved
    38→  if (a >= 240) return true;
    39→  // Default unreachable
    40→  if (a === 0) return true;
    41→  return false;
    42→}

    44→// ── Rate limiter (in-memory, per-domain) ──────────────────────────
    45→const scanRateLimit = new Map<string, { count: number; resetAt: number }>();
    46→function checkRateLimit(domain: string): boolean {
    47→  const now = Date.now();
    48→  const entry = scanRateLimit.get(domain);
    49→  if (!entry || now > entry.resetAt) {
    50→    scanRateLimit.set(domain, { count: 1, resetAt: now + 60_000 }); // 1 per minute
    51→    return true;
    52→  }
    53→  if (entry.count >= 3) return false; // max 3 per minute
    54→  entry.count++;
    55→  return true;
    56→}

    67→// 1. DNS ENUMERATION (all in parallel)
    68→// ═══════════════════════════════════════════════════════════════════════
    69→async function enumerateDNS(domain: string): Promise<{ findings: Finding[]; mainIp: string | null }> {
    70→  const findings: Finding[] = [];
    71→
    72→  // Fire all DNS queries in parallel
    73→  const [aRecords, aaaaRecords, mxRaw, nsRaw, txtRecords, spfDomain] = await Promise.all([
    74→    digShort(domain, 'A'),
    75→    digShort(domain, 'AAAA'),
    76→    digAnswer(domain, 'MX'),
    77→    digAnswer(domain, 'NS'),
    78→    digShort(domain, 'TXT'),
    79→    digShort(`_dmarc.${domain}`, 'TXT'),
    80→  ]);
    81→
    82→  const mainIp = aRecords[0] || null;
    83→
    84→  // A records
    85→  if (aRecords.length > 0) {
    86→    findings.push({
    87→      title: `DNS A Record — ${aRecords.length} IPv4 address(es)`,
    88→      severity: 'info', category: 'dns',
    89→      description: `Domain ${domain} resolves to ${aRecords.length} IPv4 address(es): ${aRecords.join(', ')}. Multiple A records indicate round-robin DNS or CDN usage.`,
    90→      evidence: `A: ${aRecords.join(', ')}`,
    91→      asset: domain,
    92→    });
    93→  }
    94→
    95→  // AAAA
    96→  if (aaaaRecords.length > 0) {
    97→    findings.push({
    98→      title: 'DNS AAAA Record — IPv6 enabled',
    99→      severity: 'info', category: 'dns',
   100→      description: `IPv6 address(es) found: ${aaaaRecords.join(', ')}. IPv6 support is present.`,
   101→      evidence: `AAAA: ${aaaaRecords.join(', ')}`,
   102→      asset: domain,
   103→    });
   104→  }
   105→
   106→  // MX
   107→  const mxLines = mxRaw ? mxRaw.split('\n').filter(l => l.includes('MX')) : [];
   108→  if (mxLines.length > 0) {
   109→    findings.push({
   110→      title: `MX Records — ${mxLines.length} mail server(s)`,
   111→      severity: 'info', category: 'dns',
   112→      description: `${mxLines.length} mail server(s) configured:\n${mxLines.map(m => '  ' + m.trim()).join('\n')}`,
   113→      evidence: mxLines.join('; '),
   114→      asset: domain,
   115→    });
   116→  } else {
   117→    findings.push({
   118→      title: 'No MX Records Found',
   119→      severity: 'low', category: 'dns',
   120→      description: `No MX records for ${domain}. Domain does not receive email directly or uses third-party email.`,
   121→      evidence: 'MX query returned empty',
   122→      asset: domain,
   123→    });
   124→  }
   125→
   126→  // NS
   127→  const nsLines = nsRaw ? nsRaw.split('\n').filter(l => l.includes('NS')) : [];
   128→  if (nsLines.length > 0) {
   129→    const nsNames = nsLines.map(n => n.trim().split(/\s+/).pop());
   130→    findings.push({
   131→      title: `NS Records — ${nsLines.length} nameserver(s)`,
   132→      severity: 'info', category: 'dns',
   133→      description: `Nameservers: ${nsNames.join(', ')}.`,
   134→      evidence: nsLines.join('; '),
   135→      asset: domain,
   136→    });
   137→  }
   138→
   139→  // TXT
   140→  if (txtRecords.length > 0) {
   141→    findings.push({
   142→      title: `TXT Records — ${txtRecords.length} record(s)`,
   143→      severity: 'info', category: 'dns',
   144→      description: `Found ${txtRecords.length} TXT record(s) including verification records, SPF, or other configurations.`,
   145→      evidence: txtRecords.join(' | '),
   146→      asset: domain,
   147→    });
   148→  }
   149→
   150→  // SPF
   151→  const spfRecord = txtRecords.find(t => t.startsWith('"v=spf'));
   152→  if (!spfRecord) {
   153→    findings.push({
   154→      title: 'SPF Record Missing',
   155→      severity: 'high', category: 'dns',
   156→      description: `No SPF record found for ${domain}. Without SPF, attackers can send spoofed emails appearing to come from this domain, enabling phishing attacks and brand damage.`,
   157→      evidence: `TXT query returned ${txtRecords.length} record(s), none with "v=spf1"`,
   158→      asset: domain,
   159→    });
   160→  } else {
   161→    if (spfRecord.includes('+all') || spfRecord.includes('?all')) {
   162→      findings.push({
   163→        title: 'SPF Record Uses +all or ?all (Open)',
   164→        severity: 'critical', category: 'dns',
   165→        description: `SPF record allows ANY server to send email for ${domain}, defeating the purpose of SPF entirely.`,
   166→        evidence: `SPF: ${spfRecord}`,
   167→        asset: domain,
   168→      });
   169→    } else {
   170→      findings.push({
   171→        title: 'SPF Record Configured Properly',
   172→        severity: 'info', category: 'dns',
   173→        description: `SPF record found with proper fail mechanism: ${spfRecord}.`,
   174→        evidence: `SPF: ${spfRecord}`,
   175→        asset: domain,
   176→      });
   177→    }
   178→  }
   179→
   180→  // DMARC
   181→  const dmarc = spfDomain.find(t => t.includes('v=DMARC'));
   182→  if (!dmarc) {
   183→    findings.push({
   184→      title: 'DMARC Record Not Found',
   185→      severity: 'high', category: 'dns',
   186→      description: `No DMARC record for ${domain}. DMARC builds on SPF/DKIM to provide email authentication policy. Without it, the domain cannot specify how receivers handle unauthenticated emails.`,
   187→      evidence: `_dmarc.${domain} TXT query empty`,
   188→      asset: `_dmarc.${domain}`,
   189→    });
   190→  } else {
   191→    const pMatch = dmarc.match(/p=([a-z]+)/);
   192→    const policy = pMatch ? pMatch[1] : 'unknown';
   193→    if (policy === 'none') {
   194→      findings.push({
   195→        title: 'DMARC Policy Set to "none" (Monitor Only)',
   196→        severity: 'medium', category: 'dns',
   197→        description: `DMARC p=none means unauthenticated emails are only monitored, not rejected. Upgrade to p=quarantine or p=reject.`,
   198→        evidence: `DMARC: ${dmarc}`,
   199→        asset: `_dmarc.${domain}`,
   200→      });
   201→    } else {
   202→      findings.push({
   203→        title: `DMARC Policy Active (p=${policy})`,
   204→        severity: 'info', category: 'dns',
   205→        description: `DMARC is enforcing with p=${policy}.`,
   206→        evidence: `DMARC: ${dmarc}`,
   207→        asset: `_dmarc.${domain}`,
   208→      });
   209→    }
   210→  }
   211→
   212→  // DKIM
   213→  const dkimSelectors = ['selector1', 'selector2', 'google', 'k1', 'default', 's1'];
   214→  const dkimChecks = await Promise.all(dkimSelectors.map(async sel => {
   215→    const recs = await digShort(`${sel}._domainkey.${domain}`, 'TXT');
   216→    return recs.find(t => t.includes('v=DKIM')) ? sel : null;
   217→  }));
   218→  const dkimSel = dkimChecks.find(Boolean);
   219→  if (dkimSel) {
   220→    findings.push({
   221→      title: `DKIM Record Found (selector: ${dkimSel})`,
   222→      severity: 'info', category: 'dns',
   223→      description: `DKIM signing configured with selector "${dkimSel}". Outgoing emails can be cryptographically verified.`,
   224→      evidence: `DKIM selector: ${dkimSel}`,
   225→      asset: `${dkimSel}._domainkey.${domain}`,
   226→    });
   227→  } else {
   228→    findings.push({
   229→      title: 'DKIM Record Not Found',
   230→      severity: 'medium', category: 'dns',
   231→      description: `No DKIM record found for common selectors (${dkimSelectors.join(', ')}). Outgoing emails cannot be cryptographically verified.`,
   232→      evidence: `Checked selectors: ${dkimSelectors.join(', ')}`,
   233→      asset: domain,
   234→    });
   235→  }
   236→
   237→  // DNSSEC — native dns.promises doesn't support DNSSEC queries
   238→  // DNSSEC validation requires specialized DNS libraries; skipping this check
   239→  findings.push({
   240→    title: 'DNSSEC Check Skipped',
   241→    severity: 'info', category: 'dns',
   242→    description: `DNSSEC validation requires specialized DNS query support not available in the native resolver. Use a dedicated DNS security tool for full DNSSEC analysis.`,
   243→    evidence: 'Native resolver — DNSSEC check not available',
   244→    asset: domain,
   245→  });
   246→
   247→  return { findings, mainIp };
   248→}

   251→// 2. SUBDOMAIN ENUMERATION (async parallel, batched)

   253→const QUICK_SUBS = [
   254→  'www','api','mail','admin','portal','dashboard','app','dev','staging',
   255→  'cdn','blog','shop','secure','auth','login','sso','vpn','gateway','proxy',
   256→  'ns1','ns2','dns','db','elastic','search','analytics','webhook','graphql',
   257→  'git','gitlab','github','docs','status','monitor','grafana','kibana','logs',
   258→  'jenkins','ci','internal','sandbox','demo','preview','m','mobile','crm',
   259→  'smtp','webmail','origin','edge','panel','cpanel','server','node','us','eu',
   260→];
   261→
   262→const FULL_SUBS = [
   263→  ...QUICK_SUBS,
   264→  'ftp','pop','imap','test','beta','static','assets','images','media','video',
   265→  'news','store','remote','mx1','mx2','redis','track','hooks','api-v1','api-v2',
   266→  'rest','repo','wiki','help','support','health','metrics','prometheus',
   267→  'hr','billing','token','keys','certs','v2','v3','staging1','staging2',
   268→  'phpmyadmin','amp','pwa','erp','payroll','oAuth','oauth2','pki',
   269→  'go','play','run','link','links','short','redirect','webmin','whm','plesk',
   270→  'web1','web2','app1','app2','uk','asia','cdn1','cdn2','static1',
   271→];
   272→
   273→const SENSITIVE_SUBS = ['admin','portal','dashboard','db','database','redis',
   274→  'jenkins','gitlab','git','internal','staging','dev','test','vpn','remote',
   275→  'proxy','elastic','kibana','grafana','prometheus','cpanel','plesk','webmin',
   276→  'phpmyadmin','server','smtp','webmail','crm','sandbox','ci','hooks','api-v1',
   277→  'staging1','staging2','test','dev'];
   278→
   279→async function enumerateSubdomains(domain: string, quick: boolean): Promise<Finding[]> {
   280→  const subs = quick ? QUICK_SUBS : FULL_SUBS;
   281→  const batchSize = 30;
   282→  const discovered: Array<{ fqdn: string; ip: string }> = [];
   283→
   284→  for (let i = 0; i < subs.length; i += batchSize) {
   285→    const batch = subs.slice(i, i + batchSize);
   286→    const results = await Promise.all(
   287→      batch.map(async (sub) => {
   288→        const ip = await nativeResolveIP(`${sub}.${domain}`);
   289→        return ip ? { fqdn: `${sub}.${domain}`, ip } : null;
   290→      })
   291→    );
   292→    for (const r of results) { if (r) discovered.push(r); }
   293→  }
   294→
   295→  return discovered.map(({ fqdn, ip }) => {
   296→    const subName = fqdn.replace(`.${domain}`, '');
   297→    const isSensitive = SENSITIVE_SUBS.some(s => subName.includes(s));
   298→    return {
   299→      title: `Discovered Subdomain: ${fqdn}`,
   300→      severity: isSensitive ? 'high' : 'info',
   301→      category: 'subdomain',
   302→      description: `${fqdn} resolves to ${ip}.${isSensitive ? ` This is a sensitive internal service (${subName}) publicly accessible — significant security concern.` : ' Publicly accessible subdomain adding to attack surface.'}`,
   303→      evidence: `A record: ${ip}`,
   304→      asset: fqdn,
   305→    };
   306→  });
   307→}

   310→// 3. HTTP HEADER ANALYSIS (async)
   311→// ═══════════════════════════════════════════════════════════════════════
   312→async function analyzeHTTPHeaders(domain: string): Promise<{ findings: Finding[]; technologies: string[] }> {
   313→  const findings: Finding[] = [];
   314→  const technologies: string[] = [];
   315→
   316→  const [httpsResult, httpResult] = await Promise.all([
   317→    safeFetch(`https://${domain}`, { timeout: 12000, method: 'HEAD' }),
   318→    safeFetch(`http://${domain}`, { timeout: 8000, method: 'HEAD' }),
   319→  ]);
   320→  const response = httpsResult.ok ? httpsResult : (httpResult.ok ? httpResult : null);
   321→
   322→  if (!response || response.status === 0) {
   323→    findings.push({
   324→      title: 'HTTP Service Not Reachable',
   325→      severity: 'medium', category: 'header',
   326→      description: `Could not connect to ${domain} via HTTP/HTTPS.`,
   327→      evidence: 'Native fetch returned no response',
   328→      asset: domain,
   329→    });
   330→    return { findings, technologies };
   331→  }
   332→
   333→  const hmap: Record<string, string> = response.headers;
   334→
   335→  // HSTS
   336→  if (!hmap['strict-transport-security']) {
   337→    findings.push({
   338→      title: 'Missing Strict-Transport-Security (HSTS)',
   339→      severity: 'high', category: 'header',
   340→      description: `No HSTS header on ${domain}. Browsers may attempt HTTP before HTTPS, enabling SSL stripping MITM attacks. Set HSTS with max-age >= 31536000 and includeSubDomains.`,
   341→      evidence: 'strict-transport-security header not found',
   342→      asset: domain,
   343→    });
   344→  } else {
   345→    const hsts = hmap['strict-transport-security'];
   346→    const ma = hsts.match(/max-age=(\d+)/);
   347→    const age = ma ? parseInt(ma[1]) : 0;
   348→    if (age < 31536000) {
   349→      findings.push({ title: `HSTS max-age Too Short (${Math.round(age/86400)} days)`, severity: 'medium', category: 'header',
   350→        description: `HSTS max-age is ${age}s (${Math.round(age/86400)}d). Best practice: >= 1 year (31536000s).`, evidence: `HSTS: ${hsts}`, asset: domain });
   351→    } else if (!hsts.includes('includeSubDomains')) {
   352→      findings.push({ title: 'HSTS Missing includeSubDomains', severity: 'medium', category: 'header',
   353→        description: `HSTS active but without includeSubDomains — subdomains remain vulnerable.`, evidence: `HSTS: ${hsts}`, asset: domain });
   354→    } else {
   355→      findings.push({ title: 'HSTS Properly Configured', severity: 'info', category: 'header',
   356→        description: `HSTS active with proper configuration.`, evidence: `HSTS: ${hsts}`, asset: domain });
   357→    }
   358→  }
   359→
   360→  // CSP
   361→  if (!hmap['content-security-policy'] && !hmap['content-security-policy-report-only']) {
   362→    findings.push({ title: 'Missing Content-Security-Policy (CSP)', severity: 'high', category: 'header',
   363→      description: `No CSP header. CSP is the primary defense against XSS and data injection. Without it, browsers will execute any inline scripts and load resources from any origin.`, evidence: 'content-security-policy header not found', asset: domain });
   364→  } else {
   365→    const csp = hmap['content-security-policy'] || hmap['content-security-policy-report-only'];
   366→    if (csp.includes("'unsafe-inline'") && csp.includes("'unsafe-eval'")) {
   367→      findings.push({ title: 'CSP Uses unsafe-inline and unsafe-eval', severity: 'medium', category: 'header',
   368→        description: `CSP present but weakened by 'unsafe-inline' and 'unsafe-eval', significantly reducing XSS protection.`, evidence: `CSP: ${csp.substring(0,200)}`, asset: domain });
   369→    } else {
   370→      findings.push({ title: 'Content-Security-Policy Configured', severity: 'info', category: 'header',
   371→        description: `CSP header present, providing XSS protection.`, evidence: `CSP: ${csp.substring(0,150)}`, asset: domain });
   372→    }
   373→  }
   374→
   375→  // X-Frame-Options
   376→  if (!hmap['x-frame-options']) {
   377→    findings.push({ title: 'Missing X-Frame-Options Header', severity: 'medium', category: 'header',
   378→      description: `No X-Frame-Options — site is vulnerable to clickjacking. Set to "DENY" or "SAMEORIGIN".`, evidence: 'x-frame-options not found', asset: domain });
   379→  }
   380→
   381→  // X-Content-Type-Options
   382→  if (hmap['x-content-type-options'] !== 'nosniff') {
   383→    findings.push({ title: 'Missing X-Content-Type-Options: nosniff', severity: 'low', category: 'header',
   384→      description: `Without nosniff, browsers may MIME-sniff content types.`, evidence: `x-content-type-options: ${hmap['x-content-type-options'] || 'not set'}`, asset: domain });
   385→  }
   386→
   387→  // Referrer-Policy
   388→  if (!hmap['referrer-policy']) {
   389→    findings.push({ title: 'Missing Referrer-Policy Header', severity: 'low', category: 'header',
   390→      description: `No Referrer-Policy — full URLs may leak to third parties via Referer header.`, evidence: 'referrer-policy not found', asset: domain });
   391→  }
   392→
   393→  // Permissions-Policy
   394→  if (!hmap['permissions-policy'] && !hmap['feature-policy']) {
   395→    findings.push({ title: 'Missing Permissions-Policy Header', severity: 'low', category: 'header',
   396→      description: `No Permissions-Policy — browser features (camera, mic, etc.) not restricted.`, evidence: 'permissions-policy not found', asset: domain });
   397→  }
   398→
   399→  // CORS
   400→  if (hmap['access-control-allow-origin']) {
   401→    if (hmap['access-control-allow-origin'] === '*') {
   402→      findings.push({ title: 'CORS Allows Any Origin (*)', severity: 'high', category: 'header',
   403→        description: `Access-Control-Allow-Origin: * means any website can read responses. If sensitive data is served, it can be exfiltrated by any malicious site.`, evidence: 'access-control-allow-origin: *', asset: domain });
   404→    }
   405→  }
   406→
   407→  // Server disclosure
   408→  if (hmap['server']) {
   409→    technologies.push(hmap['server']);
   410→    if (/\d+\.\d+/.test(hmap['server'])) {
   411→      findings.push({ title: 'Server Version Disclosure', severity: 'medium', category: 'header',
   412→        description: `Server header reveals version: "${hmap['server']}". Remove version numbers to reduce info leakage.`, evidence: `server: ${hmap['server']}`, asset: domain });
   413→    }
   414→  }
   415→
   416→  // X-Powered-By
   417→  if (hmap['x-powered-by']) {
   418→    technologies.push(hmap['x-powered-by']);
   419→    findings.push({ title: `Technology Disclosure: X-Powered-By: ${hmap['x-powered-by']}`, severity: 'low', category: 'header',
   420→      description: `X-Powered-By reveals backend: "${hmap['x-powered-by']}". Remove in production.`, evidence: `x-powered-by: ${hmap['x-powered-by']}`, asset: domain });
   421→  }
   422→
   423→  // Detect tech from headers
   424→  const allH = JSON.stringify(response.headers).toLowerCase();
   425→  if (hmap['cf-ray'] || hmap['cf-cache-status']) technologies.push('Cloudflare');
   426→  if (allH.includes('X-Amz-Cf-Id')) technologies.push('AWS CloudFront');
   427→  if (hmap['x-vercel-id']) technologies.push('Vercel');
   428→  if (hmap['x-netlify-request-id']) technologies.push('Netlify');
   429→  if (hmap['fly-request-id']) technologies.push('Fly.io');
   430→  if (hmap['alt-svc']?.includes('h3')) technologies.push('HTTP/3 (QUIC)');
   431→
   432→  return { findings, technologies };
   433→}

   436→// 4. SSL/TLS CERTIFICATE ANALYSIS (async)
   437→// ═══════════════════════════════════════════════════════════════════════
   438→async function analyzeSSL(domain: string): Promise<{ findings: Finding[]; technologies: string[] }> {
   439→  const findings: Finding[] = [];
   440→  const technologies: string[] = [];
   441→
   442→  const sslResult = await analyzeSSLNative(domain);
   443→
   444→  if (!sslResult.cert && !sslResult.sslConnect) {
   445→    findings.push({ title: 'SSL/TLS Connection Failed', severity: 'high', category: 'ssl',
   446→      description: `Could not establish SSL/TLS to ${domain}:443. Server may not support HTTPS.`, evidence: sslResult.error || 'TLS connection failed', asset: `${domain}:443` });
   447→    return { findings, technologies };
   448→  }
   449→
   450→  const full = sslResult.certInfo || sslResult.sslConnect;
   451→
   452→  // Subject
   453→  const subjM = full.match(/subject=([^\n]+)/);
   454→  if (subjM) findings.push({ title: `SSL Certificate Subject: ${subjM[1].trim()}`, severity: 'info', category: 'ssl',
   455→    description: `Certificate issued to: ${subjM[1].trim()}.`, evidence: `subject: ${subjM[1].trim()}`, asset: domain });
   456→
   457→  // Issuer + tech detection
   458→  const issM = full.match(/issuer=([^\n]+)/);
   459→  const issuer = issM ? issM[1].trim() : 'unknown';
   460→  if (issuer.includes("Let's Encrypt")) technologies.push("Let's Encrypt");
   461→  else if (issuer.includes('DigiCert')) technologies.push('DigiCert');
   462→  else if (issuer.includes('Sectigo') || issuer.includes('Comodo')) technologies.push('Sectigo');
   463→  else if (issuer.includes('Cloudflare')) technologies.push('Cloudflare SSL');
   464→  else if (issuer.includes('Amazon') || issuer.includes('AWS')) technologies.push('AWS ACM');
   465→  else if (issuer.includes('Google')) technologies.push('Google Trust Services');
   466→
   467→  findings.push({ title: `SSL Certificate Issuer: ${issuer}`, severity: 'info', category: 'ssl',
   468→    description: `Issued by: ${issuer}.`, evidence: `issuer: ${issuer}`, asset: domain });
   469→
   470→  // Dates + expiry
   471→  const nbM = full.match(/notBefore=(.+)/);
   472→  const naM = full.match(/notAfter=(.+)/);
   473→  if (nbM && naM) {
   474→    const nb = new Date(nbM[1].trim());
   475→    const na = new Date(naM[1].trim());
   476→    const days = Math.ceil((na.getTime() - Date.now()) / 86400000);
   477→    const lifespan = Math.ceil((na.getTime() - nb.getTime()) / 86400000);
   478→
   479→    findings.push({ title: `SSL Certificate Validity: ${nb.toISOString().split('T')[0]} — ${na.toISOString().split('T')[0]}`, severity: 'info', category: 'ssl',
   480→      description: `Certificate lifespan: ${lifespan} days. Valid from ${nb.toISOString().split('T')[0]} to ${na.toISOString().split('T')[0]}.`, evidence: `notBefore: ${nbM[1].trim()}, notAfter: ${naM[1].trim()}`, asset: domain });
   481→
   482→    if (days < 0) {
   483→      findings.push({ title: `SSL Certificate EXPIRED ${Math.abs(days)} days ago`, severity: 'critical', category: 'ssl',
   484→        description: `Certificate for ${domain} expired ${Math.abs(days)} days ago. Browsers show security warnings, automated integrations fail with TLS errors. Immediate renewal required.`, evidence: `Expired: ${na.toISOString()}`, asset: domain });
   485→    } else if (days <= 7) {
   486→      findings.push({ title: `SSL Certificate Expiring in ${days} Days — CRITICAL`, severity: 'critical', category: 'ssl',
   487→        description: `Certificate expires in ${days} days. Emergency renewal required.`, evidence: `Expires: ${na.toISOString()} (${days}d)`, asset: domain });
   488→    } else if (days <= 30) {
   489→      findings.push({ title: `SSL Certificate Expiring in ${days} Days`, severity: 'high', category: 'ssl',
   490→        description: `Certificate expires in ${days} days. Prompt renewal needed.`, evidence: `Expires: ${na.toISOString()} (${days}d)`, asset: domain });
   491→    } else if (days <= 90) {
   492→      findings.push({ title: `SSL Certificate Expiring in ${days} Days`, severity: 'medium', category: 'ssl',
   493→        description: `Certificate expires in ${days} days. Plan renewal.`, evidence: `Expires: ${na.toISOString()} (${days}d)`, asset: domain });
   494→    }
   495→
   496→    if (lifespan > 398) {
   497→      findings.push({ title: `Certificate Lifespan Exceeds 398 Days (${lifespan}d)`, severity: 'medium', category: 'ssl',
   498→        description: `Lifespan exceeds Apple/Google 398-day max. New certs must be shorter.`, evidence: `Lifespan: ${lifespan}d (max: 398)`, asset: domain });
   499→    }
   500→  }
   501→
   502→  // SAN
   503→  const sanM = full.match(/Subject Alternative Name[^\n]*\n([^\n]*)/);
   504→  if (sanM) {
   505→    const sans = sanM[1].match(/DNS:[^\s,]+/g);
   506→    if (sans) findings.push({ title: `Certificate Covers ${sans.length} Domain(s) (SAN)`, severity: 'info', category: 'ssl',
   507→      description: `SANs: ${sans.map(s=>s.replace('DNS:','')).join(', ')}.`, evidence: `SAN: ${sans.join(', ')}`, asset: domain });
   508→  }
   509→
   510→  // TLS version
   511→  const proto = sslResult.protocol || '';
   512→  if (proto.includes('TLSv1 ') || proto.includes('TLSv1.0')) {
   513→    findings.push({ title: 'Weak TLS Version: TLS 1.0 Detected', severity: 'high', category: 'ssl',
   514→      description: `TLS 1.0 deprecated (RFC 8996, 2020). Vulnerable to BEAST, POODLE, RC4. PCI-DSS prohibits. Disable immediately.`, evidence: `Protocol: ${proto}`, asset: `${domain}:443` });
   515→  } else if (proto.includes('TLSv1.1')) {
   516→    findings.push({ title: 'Deprecated TLS 1.1 Detected', severity: 'high', category: 'ssl',
   517→      description: `TLS 1.1 deprecated in 2020. Known weaknesses. PCI-DSS requires disabling. Support TLS 1.2+ only.`, evidence: `Protocol: ${proto}`, asset: `${domain}:443` });
   518→  } else if (proto.includes('TLSv1.3')) {
   519→    technologies.push('TLS 1.3');
   520→    findings.push({ title: 'Modern TLS 1.3 Negotiated', severity: 'info', category: 'ssl',
   521→      description: `TLS 1.3 — latest version with 0-RTT, mandatory forward secrecy, no weak ciphers.`, evidence: `Protocol: ${proto}`, asset: `${domain}:443` });
   522→  } else if (proto.includes('TLSv1.2')) {
   523→    technologies.push('TLS 1.2');
   524→    findings.push({ title: 'TLS 1.2 Negotiated', severity: 'info', category: 'ssl',
   525→      description: `TLS 1.2 — secure when configured with strong cipher suites. Consider enabling TLS 1.3.`, evidence: `Protocol: ${proto}`, asset: `${domain}:443` });
   526→  }
   527→
   528→  // Cipher
   529→  const cipher = sslResult.cipher || '';
   530→  const weakC = ['RC4','DES','MD5','NULL','EXPORT','3DES'];
   531→  if (weakC.some(w => cipher.toUpperCase().includes(w))) {
   532→    findings.push({ title: `Weak Cipher Suite: ${cipher}`, severity: 'high', category: 'ssl',
   533→      description: `Weak cipher (${cipher}) — vulnerable to known attacks. Use AEAD ciphers (AES-GCM, ChaCha20-Poly1305) with ECDHE.`, evidence: `Cipher: ${cipher}`, asset: `${domain}:443` });
   534→  } else if (cipher) {
   535→    findings.push({ title: `Cipher Suite: ${cipher}`, severity: 'info', category: 'ssl',
   536→      description: `Negotiated: ${cipher}.`, evidence: `Cipher: ${cipher}`, asset: `${domain}:443` });
   537→  }
   538→
   539→  return { findings, technologies };
   540→}

   567→        const srv = r.headers['server'] ? ` [${r.headers['server']}]` : '';
   568→        return { port, service, risk, desc, status: status + srv };
   569→      }
   570→      return null;
   571→    })
   572→  );

   578→    evidence: status, asset: `${domain}:${port}`,
   579→  }));

   583→// 6. CERTIFICATE TRANSPARENCY LOGS — crt.sh for ALL historical subdomains
   584→// ═══════════════════════════════════════════════════════════════════════
   585→async function enumerateCTLogs(domain: string): Promise<Finding[]> {
   586→  const findings: Finding[] = [];
   587→  try {
   588→    const response = await safeFetch(`https://crt.sh/?q=%25.${domain}&output=json`, { timeout: 20000, skipSSRFCheck: true });
   589→    if (!response.ok || !response.text) return findings;
   590→    const certs = JSON.parse(response.text);
   591→    // Deduplicate subdomains
   592→    const subs = new Set<string>();
   593→    for (const c of certs) {
   594→      if (c.name_value) {
   595→        for (const name of c.name_value.split('\n')) {
   596→          const clean = name.trim().replace(/^\*\./, '');
   597→          if (clean.endsWith(`.${domain}`) && clean !== domain) subs.add(clean);
   598→        }
   599→      }
   600→    }
   601→    const unique = Array.from(subs).sort();
   602→    if (unique.length > 0) {
   603→      findings.push({
   604→        title: `Certificate Transparency: ${unique.length} Subdomain(s) from CT Logs`,
   605→        severity: unique.length > 20 ? 'high' : 'info', category: 'osint',
   606→        description: `crt.sh reveals ${unique.length} unique subdomains from historical SSL certificate logs. These subdomains may not be in current DNS but existed in the past — old infrastructure may still be alive.`,
   607→        evidence: `CT subdomains: ${unique.slice(0, 30).join(', ')}${unique.length > 30 ? ` ... +${unique.length - 30} more` : ''}`,
   608→        asset: domain,
   609→      });
   610→    }
   611→    // Check for expired certs in CT logs
   612→    const now = Date.now();
   613→    const expiredCerts = certs.filter((c: { not_after?: string }) => {
   614→      const exp = c.not_after ? new Date(c.not_after + ' UTC').getTime() : 0;
   615→      return exp > 0 && exp < now;
   616→    });
   617→    if (expiredCerts.length > 3) {
   618→      findings.push({
   619→        title: `${expiredCerts.length} Expired Certificate(s) in CT Logs`,
   620→        severity: 'medium', category: 'ssl',
   621→        description: `${expiredCerts.length} historical certificates found in CT logs have expired. Expired certificates indicate abandoned infrastructure that may still be reachable.`,
   622→        evidence: `Expired certs: ${Math.min(expiredCerts.length, 5)} found in crt.sh`,
   623→        asset: domain,
   624→      });
   625→    }
   626→  } catch { /* crt.sh may fail — non-critical */ }
   627→  return findings;
   628→}

   631→// 7. WAYBACK MACHINE RECON — historical endpoints, hidden paths
   632→// ═══════════════════════════════════════════════════════════════════════
   633→async function enumerateWayback(domain: string): Promise<Finding[]> {
   634→  const findings: Finding[] = [];
   635→  try {
   636→    const response = await safeFetch(`http://web.archive.org/cdx/search/cdx?url=${domain}/*&output=json&collapse=urlkey&fl=original&limit=200`, { timeout: 25000, skipSSRFCheck: true });
   637→    if (!response.ok || !response.text) return findings;
   638→    const lines = response.text.split('\n').filter(Boolean);
   639→    if (lines.length < 2) return findings;
   640→    const urls = lines.slice(1).map(l => {
   641→      try { return JSON.parse(l)[0]; } catch { return null; }
   642→    }).filter(Boolean);
   643→
   644→    // Extract unique paths
   645→    const paths = new Set<string>();

   648→    for (const url of urls) {
   649→      try {
   650→        const u = new URL(url);
   651→        const p = u.pathname;
   652→        paths.add(p);
   653→        if (sensitivePatterns.some(pat => pat.test(p))) sensitivePaths.push(p);
   654→      } catch {}
   655→    }
   656→
   657→    if (paths.size > 0) {
   658→      findings.push({
   659→        title: `Wayback Machine: ${paths.size} Historical URL(s) Archived`,
   660→        severity: 'info', category: 'osint',
   661→        description: `Web Archive reveals ${paths.size} unique paths for ${domain}. Historical pages often expose old API endpoints, admin panels, or backup files still on the server.`,
   662→        evidence: `Sample paths: ${Array.from(paths).slice(0, 15).join(', ')}`,
   663→        asset: domain,
   664→      });
   665→    }
   666→
   667→    if (sensitivePaths.length > 0) {
   668→      const unique = [...new Set(sensitivePaths)].sort();
   669→      findings.push({
   670→        title: `Wayback Machine: ${unique.length} Sensitive Path(s) Discovered`,
   671→        severity: 'high', category: 'vulnerability',
   672→        description: `Historical archives expose ${unique.length} sensitive paths that may still exist on the live server. These include admin panels, API endpoints, debug interfaces, config files, or version control directories.`,
   673→        evidence: `Sensitive: ${unique.slice(0, 20).join(', ')}`,
   674→        asset: domain,
   675→      });
   676→    }
   677→  } catch { /* Wayback may fail — non-critical */ }
   678→  return findings;
   679→}

   725→// 9. DNS ZONE TRANSFER (AXFR) ATTEMPTS — try all NS servers
   726→// ═══════════════════════════════════════════════════════════════════════
   727→async function attemptZoneTransfer(domain: string): Promise<Finding[]> {
   728→  const findings: Finding[] = [];
   729→  const nsOut = await digAnswer(domain, 'NS');
   730→  const nsServers: string[] = [];
   731→  for (const line of nsOut.split('\n')) {
   732→    const m = line.match(/NS\s+(\S+)/);
   733→    if (m) nsServers.push(m[1].replace(/\.$/, ''));
   734→  }
   735→  if (nsServers.length === 0) return findings;
   736→
   737→  // Zone Transfer (AXFR) — native dns.promises does not support AXFR queries
   738→  // AXFR requires a specialized DNS client; skipping this check
   739→  for (const ns of nsServers) {
   740→    findings.push({
   741→      title: `DNS Zone Transfer Check: ${ns}`,
   742→      severity: 'info', category: 'dns',
   743→      description: `AXFR query requires a specialized DNS client not available in the native resolver. Zone transfer checks against ${ns} were skipped.`,
   744→      evidence: `NS: ${ns} — AXFR check skipped (native resolver limitation)`,
   745→      asset: domain,
   746→    });
   747→  }
   748→
   749→  // SOA record analysis
   750→  const soaOut = await digAnswer(domain, 'SOA');
   751→  if (soaOut) {
   752→    findings.push({
   753→      title: 'DNS SOA Record Analyzed',
   754→      severity: 'info', category: 'dns',
   755→      description: `SOA record reveals zone management details for ${domain}.`,
   756→      evidence: soaOut.split('\n').filter(l => l.includes('SOA')).join('; '),
   757→      asset: domain,
   758→    });
   759→  }
   760→  return findings;
   761→}

   764→// 10. ROBOTS.TXT + SITEMAP.XML — hidden paths, API endpoints

   766→async function analyzeRobotsAndSitemap(domain: string): Promise<Finding[]> {
   767→  const findings: Finding[] = [];
   768→  const [robotsResult, sitemapResult] = await Promise.all([
   769→    safeFetch(`https://${domain}/robots.txt`, { timeout: 12000 }).catch(() => safeFetch(`http://${domain}/robots.txt`, { timeout: 12000 })),

   771→  ]);
   772→  const robots = robotsResult?.text || '';
   773→  const sitemap = sitemapResult?.text || '';
   774→
   775→  // Robots.txt
   776→  if (robots && robots.length > 10) {

   778→    const sensitive = disallowed.filter(d => /admin|api|debug|config|backup|\.env|secret|internal|private|tmp|upload|staging/i.test(d));
   779→    if (sensitive.length > 0) {
   780→      findings.push({
   781→        title: `robots.txt: ${sensitive.length} Sensitive Disallowed Path(s)`,
   782→        severity: 'high', category: 'vulnerability',
   783→        description: `robots.txt reveals ${sensitive.length} sensitive paths that the site asks crawlers not to access. These paths likely exist and may be accessible without authentication: ${sensitive.join(', ')}`,
   784→        evidence: `Disallow: ${sensitive.join(', ')}`, asset: `${domain}/robots.txt`,
   785→      });
   786→    } else if (disallowed.length > 0) {
   787→      findings.push({
   788→        title: `robots.txt: ${disallowed.length} Disallowed Path(s) Found`,
   789→        severity: 'info', category: 'osint',
   790→        description: `robots.txt defines ${disallowed.length} restricted paths: ${disallowed.slice(0, 10).join(', ')}`,
   791→        evidence: `Disallow: ${disallowed.slice(0, 10).join(', ')}`, asset: `${domain}/robots.txt`,
   792→      });
   793→    }
   794→    // Check for sitemap reference
   795→    if (robots.includes('Sitemap:')) {
   796→      findings.push({
   797→        title: 'robots.txt References Sitemap', severity: 'info', category: 'osint',
   798→        description: 'Sitemap URL found in robots.txt — reveals site structure.',
   799→        evidence: robots.split('\n').find(l => l.startsWith('Sitemap:')) || '', asset: `${domain}/robots.txt`,
   800→      });
   801→    }
   802→  } else if (!robots || robots.length <= 10) {
   803→    findings.push({
   804→      title: 'robots.txt Not Found or Empty', severity: 'low', category: 'osint',
   805→      description: 'No robots.txt file. This means all pages are crawlable by default, including any admin/debug paths.',
   806→      evidence: 'robots.txt returned empty or 404', asset: `${domain}/robots.txt`,
   807→    });
   808→  }
   809→
   810→  // Sitemap.xml
   811→  if (sitemap && sitemap.includes('<url')) {
   812→    const urls = (sitemap.match(/<loc>([^<]+)<\/loc>/g) || []).map(u => u.replace(/<\/?loc>/g, ''));
   813→    findings.push({
   814→      title: `sitemap.xml: ${urls.length} URL(s) Exposed`,
   815→      severity: urls.length > 100 ? 'medium' : 'info', category: 'osint',
   816→      description: `sitemap.xml exposes ${urls.length} URLs revealing the full site structure. This aids attackers in mapping the application for targeted attacks.`,
   817→      evidence: `URLs: ${urls.slice(0, 10).join(', ')}${urls.length > 10 ? ` ... +${urls.length - 10} more` : ''}`,
   818→      asset: `${domain}/sitemap.xml`,
   819→    });
   820→  }
   821→  return findings;
   822→}

   825→// 11. JAVASCRIPT FILE ANALYSIS — extract API endpoints, secrets, keys
   826→// ═══════════════════════════════════════════════════════════════════════
   827→async function analyzeJSFiles(domain: string): Promise<Finding[]> {
   828→  const findings: Finding[] = [];
   829→  try {
   830→    // Fetch the main page and extract JS file URLs
   831→    const pageResult = await safeFetch(`https://${domain}`, { timeout: 15000 }).catch(() => safeFetch(`http://${domain}`, { timeout: 15000 }));
   832→    const page = pageResult?.text || '';
   833→    if (!page || page.length < 100) return findings;
   834→
   835→    const jsUrls = [...new Set(
   836→      (page.match(/src=["']([^"']+\.js[^"']*)/g) || [])
   837→        .map(m => m.replace(/src=["']/, '').replace(/["']$/, ''))
   838→        .filter(u => !u.startsWith('data:') && !u.includes('chrome-extension'))
   839→    )].slice(0, 8);
   840→
   841→    if (jsUrls.length === 0) return findings;
   842→
   843→    // Fetch and analyze each JS file
   844→    const jsContents = await Promise.all(
   845→      jsUrls.map(async (url) => {
   846→        const fullUrl = url.startsWith('http') ? url : `https://${domain}${url}`;
   847→        const result = await safeFetch(fullUrl, { timeout: 12000 });
   848→        return result?.text || '';
   849→      })
   850→    );
   851→
   852→    const allJs = jsContents.join('\n');
   853→    if (allJs.length < 50) return findings;
   854→
   855→    // Extract API endpoints
   856→    const apiEndpoints = [...new Set(
   857→      (allJs.match(/["'](\/api\/[^"']+)["']/g) || [])
   858→        .map(m => m.replace(/["']/g, ''))
   859→    )];
   860→    if (apiEndpoints.length > 0) {
   861→      findings.push({
   862→        title: `JavaScript: ${apiEndpoints.length} API Endpoint(s) Exposed`,
   863→        severity: 'high', category: 'vulnerability',
   864→        description: `JavaScript files expose ${apiEndpoints.length} API endpoints. Attackers can reverse-engineer the API surface, craft targeted requests, and test for authentication bypass or IDOR vulnerabilities.`,
   865→        evidence: `Endpoints: ${apiEndpoints.slice(0, 20).join(', ')}`,
   866→        asset: domain,
   867→      });
   868→    }
   869→
   870→    // Check for secrets/keys/tokens
   871→    const secretPatterns = [
   872→      { name: 'API Key', regex: /["'](api[_-]?key|apikey)["']\s*[:=]\s*["']([^"']{8,})["']/gi },
   873→      { name: 'Auth Token', regex: /["'](auth[_-]?token|bearer|access[_-]?token)["']\s*[:=]\s*["']([^"']{8,})["']/gi },
   874→      { name: 'AWS Key', regex: /AKIA[0-9A-Z]{16}/g },
   875→      { name: 'Firebase', regex: /firebase[a-zA-Z]*\.appspot\.com/g },
   876→      { name: 'Stripe Key', regex: /pk_(test|live)_[a-zA-Z0-9]{24,}/g },
   877→      { name: 'Generic Secret', regex: /["'](secret|password|token|private[_-]?key)["']\s*[:=]\s*["']([^"']{8,})["']/gi },
   878→    ];
   879→    const secrets: string[] = [];
   880→    for (const { name, regex } of secretPatterns) {
   881→      const matches = allJs.match(regex);
   882→      if (matches) secrets.push(...matches.map(m => `[${name}] ${m.substring(0, 60)}`));
   883→    }
   884→    if (secrets.length > 0) {
   885→      findings.push({
   886→        title: `CRITICAL: ${secrets.length} Secret(s)/Key(s) Found in JavaScript`,
   887→        severity: 'critical', category: 'vulnerability',
   888→        description: `JavaScript files contain ${secrets.length} hardcoded secrets, API keys, or tokens. These can be extracted by anyone viewing the page source and used to access internal services, databases, or third-party APIs.`,
   889→        evidence: `Secrets: ${secrets.slice(0, 5).join(' | ')}`,
   890→        asset: domain,
   891→      });
   892→    }
   893→
   894→    // Extract internal URLs/paths
   895→    const internalPaths = [...new Set(
   896→      (allJs.match(/["'](\/[a-zA-Z0-9_\-\/]+(?:admin|dashboard|console|manage|debug|internal|private|staging)[^"']*)["']/gi) || [])
   897→        .map(m => m.replace(/["']/g, ''))
   898→    )];
   899→    if (internalPaths.length > 0) {
   900→      findings.push({
   901→        title: `JavaScript: ${internalPaths.length} Internal Path(s) Referenced`,
   902→        severity: 'medium', category: 'osint',
   903→        description: `JavaScript files reference ${internalPaths.length} internal/sensitive paths. These may be admin interfaces, debug endpoints, or internal tools.`,
   904→        evidence: `Paths: ${internalPaths.slice(0, 15).join(', ')}`,
   905→        asset: domain,
   906→      });
   907→    }
   908→
   909→    // Count JS files and total size
   910→    findings.push({
   911→      title: `JavaScript Analysis: ${jsUrls.length} File(s), ${Math.round(allJs.length / 1024)}KB Analyzed`,
   912→      severity: 'info', category: 'technology',
   913→      description: `${jsUrls.length} JavaScript files totaling ~${Math.round(allJs.length / 1024)}KB were downloaded and analyzed for API endpoints, secrets, and internal paths.`,
   914→      evidence: `JS files: ${jsUrls.slice(0, 5).join(', ')}`,
   915→      asset: domain,
   916→    });
   917→  } catch {}
   918→  return findings;
   919→}

   922→// 12. WAF/SECURITY CONTROL DETECTION — fingerprint protection layers
   923→// ═══════════════════════════════════════════════════════════════════════
   924→async function detectWAF(domain: string): Promise<Finding[]> {
   925→  const findings: Finding[] = [];
   926→  // Trigger WAF with suspicious paths and analyze response
   927→  const [normalResult, attackResult] = await Promise.all([
   928→    safeFetch(`https://${domain}/`, { timeout: 10000, method: 'HEAD' }),
   929→    safeFetch(`https://${domain}/../../../etc/passwd`, { timeout: 10000, method: 'HEAD' }),
   930→  ]);
   931→  const normalResp = normalResult.headers ? JSON.stringify(normalResult.headers) : '';
   932→  const attackResp = attackResult.headers ? JSON.stringify(attackResult.headers) + ` status:${attackResult.status}` : '';
   933→
   934→  const wafSignatures: Record<string, string[]> = {
   935→    'Cloudflare': ['cf-ray', 'cf-cache-status', '__cf_bm', 'cloudflare'],
   936→    'AWS WAF': ['x-amzn-requestid', 'awselb'],
   937→    'Akamai': ['akamai', 'x-akamai'],
   938→    'Sucuri': ['x-sucuri-id', 'sucuri'],
   939→    'Imperva': ['x-iinfo', 'incap_ses'],
   940→    'Fastly': ['x-fastly-request-id', 'x-served-by'],
   941→    'Varnish': ['x-varnish', 'x-hits'],
   942→    'ModSecurity': ['mod_security', 'modsecurity'],
   943→    'F5 BIG-IP': ['bigip', 'f5'],
   944→  };
   945→
   946→  const detectedWAFs: string[] = [];
   947→  for (const [waf, sigs] of Object.entries(wafSignatures)) {
   948→    const combined = (normalResp + attackResp).toLowerCase();
   949→    if (sigs.some(s => combined.includes(s.toLowerCase()))) detectedWAFs.push(waf);
   950→  }
   951→
   952→  // Check if attack path was blocked differently
   953→  const normalStatus = `HTTP ${normalResult.status}`;
   954→  const attackStatus = `HTTP ${attackResult.status}`;
   955→  const blocked = attackResult.status === 403 || attackResult.status === 429;
   956→
   957→  if (detectedWAFs.length > 0) {
   958→    const techList = detectedWAFs.join(', ');
   959→    findings.push({
   960→      title: `WAF Detected: ${techList}`,
   961→      severity: 'info', category: 'security',
   962→      description: `${techList} detected protecting ${domain}. WAF presence indicates active security posture but also reveals the defense layer for targeted evasion research.`,
   963→      evidence: `Signatures found: ${detectedWAFs.map(w => wafSignatures[w].join(', ')).join('; ')}`,
   964→      asset: domain,
   965→    });
   966→  }
   967→
   968→  if (blocked) {
   969→    findings.push({
   970→      title: 'Path Traversal Attempt Blocked',
   971→      severity: 'info', category: 'security',
   972→      description: `Directory traversal attempt (../../../etc/passwd) was blocked with ${attackStatus}. This indicates active input validation or WAF protection.`,
   973→      evidence: `Attack response: ${attackStatus} | Normal: ${normalStatus}`,
   974→      asset: domain,
   975→    });
   976→  } else if (attackResp.includes('200') || attackResp.includes('301')) {
   977→    findings.push({
   978→      title: 'Path Traversal Attempt NOT Blocked',
   979→      severity: 'high', category: 'vulnerability',
   980→      description: `Directory traversal attempt (../../../etc/passwd) returned ${attackStatus} — the request was NOT blocked. This may indicate missing input validation, no WAF, or a misconfigured security layer.`,
   981→      evidence: `Attack response: ${attackStatus} | Normal: ${normalStatus}`,
   982→      asset: domain,
   983→    });
   984→  }
   985→
   986→  // Rate limit detection
   987→  const burst = await Promise.all([
   988→    safeFetch(`https://${domain}/`, { timeout: 6000, method: 'HEAD' }),
   989→    safeFetch(`https://${domain}/`, { timeout: 6000, method: 'HEAD' }),
   990→    safeFetch(`https://${domain}/`, { timeout: 6000, method: 'HEAD' }),
   991→    safeFetch(`https://${domain}/`, { timeout: 6000, method: 'HEAD' }),
   992→    safeFetch(`https://${domain}/`, { timeout: 6000, method: 'HEAD' }),
   993→  ]);
   994→  const rateLimited = burst.some(r => r.status === 429);
   995→  if (!rateLimited) {
   996→    findings.push({
   997→      title: 'No Rate Limiting Detected',
   998→      severity: 'medium', category: 'vulnerability',
   999→      description: `5 rapid requests returned no rate limiting (429). The server accepts unlimited requests, making it vulnerable to brute force, credential stuffing, and DoS attacks.`,
  1000→      evidence: '5 rapid requests: all returned non-429 status',
  1001→      asset: domain,
  1002→    });
  1003→  } else {
  1004→    findings.push({
  1005→      title: 'Rate Limiting Active', severity: 'info', category: 'security',
  1006→      description: 'Rate limiting detected on rapid requests.', evidence: '429 response received on burst', asset: domain,
  1007→    });
  1008→  }
  1009→  return findings;
  1010→}

  1089→async function harvestEmails(domain: string, pageContent: string): Promise<Finding[]> {
  1090→  const findings: Finding[] = [];
  1091→  const emails = new Set<string>();
  1092→
  1093→  // From page content
  1094→  const escapedDomain = domain.replace(/\./g, '\\.');
  1095→  const emailRegex = new RegExp(`[a-zA-Z0-9._%+-]+@${escapedDomain}`, 'gi');
  1096→  const pageEmails = (pageContent || '').match(emailRegex) || [];
  1097→  pageEmails.forEach(e => emails.add(e.toLowerCase()));
  1098→
  1099→  // From MX records (mail server hostnames often reveal email provider)
  1100→  const mxRecords = await digShort(domain, 'MX');
  1101→  const mxHosts = mxRecords.map(r => {
  1102→    const m = r.match(/\S+/);
  1103→    return m ? m[0] : r;
  1104→  });
  1105→
  1106→  if (emails.size > 0) {
  1107→    findings.push({
  1108→      title: `${emails.size} Email Address(es) Harvested`,
  1109→      severity: 'medium', category: 'osint',
  1110→      description: `${emails.size} email address(es) found on ${domain} via page content analysis. These can be used for social engineering, credential stuffing, or targeted phishing campaigns.`,
  1111→      evidence: `Emails: ${Array.from(emails).slice(0, 10).join(', ')}`,
  1112→      asset: domain,
  1113→    });
  1114→  }
  1115→
  1116→  if (mxHosts.length > 0) {
  1117→    findings.push({
  1118→      title: `Email Infrastructure: ${mxHosts.join(', ')}`,
  1119→      severity: 'info', category: 'osint',
  1120→      description: `Mail routed through: ${mxHosts.join(', ')}. This reveals the email provider (Google Workspace, Microsoft 365, self-hosted, etc.) and can be targeted for email-based attacks.`,
  1121→      evidence: `MX: ${mxHosts.join(', ')}`,
  1122→      asset: domain,
  1123→    });
  1124→  }
  1125→  return findings;
  1126→}

  1128→// ═══════════════════════════════════════════════════════════════════════
  1129→// MAIN SCAN ENDPOINT
  1130→// ═══════════════════════════════════════════════════════════════════════
  1131→export async function POST(request: NextRequest) {
  1132→  try {
  1133→    const body = await request.json();
  1134→    const { domain, scanType } = body;
  1135→
  1136→    if (!domain || typeof domain !== 'string') {
  1137→      return NextResponse.json({ error: 'Domain is required' }, { status: 400 });
  1138→    }
  1139→
  1140→    const cleanDomain = domain.replace(/^(https?:\/\/)?(www\.)?/, '').replace(/\/.*$/, '').replace(/:\d+$/, '').toLowerCase();
  1141→
  1142→    // CRITICAL: Validate domain format — block shell metacharacters, IPs, and internal hosts
  1143→    const DOMAIN_REGEX = /^[a-zA-Z0-9]([a-zA-Z0-9-]*\.)+[a-zA-Z]{2,}$/;
  1144→    if (!DOMAIN_REGEX.test(cleanDomain)) {
  1145→      return NextResponse.json({ error: 'Invalid domain format. Must be a public FQDN (e.g. example.com).' }, { status: 400 });
  1146→    }
  1147→
  1148→    // Rate limit check
  1149→    if (!checkRateLimit(cleanDomain)) {
  1150→      return NextResponse.json({ error: 'Rate limit exceeded. Max 3 scans per minute per domain.' }, { status: 429 });
  1151→    }
  1152→
  1153→    // Block known internal/sensitive domains
  1154→    const BLOCKED_DOMAINS = ['localhost', 'localhost.localdomain', 'internal', 'metadata.google.internal', 'metadata'];
  1155→    if (BLOCKED_DOMAINS.includes(cleanDomain) || cleanDomain.endsWith('.local') || cleanDomain.endsWith('.internal')) {
  1156→      return NextResponse.json({ error: 'Scanning internal domains is not permitted.' }, { status: 403 });
  1157→    }
  1158→
  1159→    // Pre-check: domain must resolve
  1160→    const preCheckIp = await nativeResolveIP(cleanDomain);
  1161→    if (!preCheckIp) {
  1162→      return NextResponse.json({ error: `Domain "${cleanDomain}" does not resolve. Check the name and try again.` }, { status: 400 });
  1163→    }
  1164→
  1165→    // SSRF protection: block scans that resolve to private IPs
  1166→    if (isPrivateIP(preCheckIp)) {
  1167→      return NextResponse.json({ error: 'Domain resolves to a private or reserved IP address. Scanning is not permitted.' }, { status: 403 });
  1168→    }
  1169→
  1170→    // DB: target + scan record
  1171→    let target = await db.scanTarget.findFirst({ where: { domain: cleanDomain } });
  1172→    if (!target) {
  1173→      target = await db.scanTarget.create({ data: { domain: cleanDomain, ip: preCheckIp } });
  1174→    } else {
  1175→      await db.scanTarget.update({ where: { id: target.id }, data: { ip: preCheckIp } });
  1176→    }
  1177→
  1178→    const scan = await db.scan.create({
  1179→      data: { targetId: target.id, status: 'running', scanType: scanType || 'full' },
  1180→    });
  1181→
  1182→    const isQuick = scanType === 'quick';
  1183→    const allFindings: Finding[] = [];
  1184→    const allTech = new Set<string>();
  1185→
  1186→    // ── 1. DNS (always, all queries parallel) ──────────────────────
  1187→    const dnsResult = await enumerateDNS(cleanDomain);
  1188→    allFindings.push(...dnsResult.findings);
  1189→
  1190→    // ── 2. Subdomains (parallel DNS resolution) ────────────────────
  1191→    const subFindings = await enumerateSubdomains(cleanDomain, isQuick);
  1192→    allFindings.push(...subFindings);
  1193→
  1194→    // ── 3. HTTP headers + page content (async curl) ──────────────────
  1195→    const httpResult = await analyzeHTTPHeaders(cleanDomain);
  1196→    allFindings.push(...httpResult.findings);
  1197→    httpResult.technologies.forEach(t => allTech.add(t));
  1198→
  1199→    // Fetch page content once for multiple modules
  1200→    const pageResult = await safeFetch(`https://${cleanDomain}`, { timeout: 15000 }).catch(() => safeFetch(`http://${cleanDomain}`, { timeout: 15000 }));
  1201→    const pageContent = pageResult?.text || '';
  1202→    const rawHeadersResult = await safeFetch(`https://${cleanDomain}`, { timeout: 10000, method: 'HEAD' });
  1203→    const rawHeaders = rawHeadersResult?.text || '';
  1204→
  1205→    // ── 4. SSL/TLS (async openssl) ─────────────────────────────────
  1206→    const sslResult = await analyzeSSL(cleanDomain);
  1207→    allFindings.push(...sslResult.findings);
  1208→    sslResult.technologies.forEach(t => allTech.add(t));
  1209→
  1210→    // ── 5. Port probing (full only, async parallel) ────────────────
  1211→    if (!isQuick) {
  1212→      const portFindings = await probePorts(cleanDomain);
  1213→      allFindings.push(...portFindings);
  1214→    }
  1215→
  1216→    // ═══ NEW DEADLY MODULES (full scan only) ═══════════════════════
  1217→    if (!isQuick) {
  1218→      // ── 6. Certificate Transparency Logs ──────────────────────────
  1219→      const [ctFindings, waybackFindings, rdnsFindings, axfrFindings, robotsFindings, jsFindings, wafFindings, fpFindings, emailFindings] = await Promise.all([
  1220→        enumerateCTLogs(cleanDomain),
  1221→        enumerateWayback(cleanDomain),
  1222→        reconReverseDNS([...dnsResult.mainIp ? [dnsResult.mainIp] : [], ...subFindings.map(f => {
  1223→          const m = f.evidence.match(/\d+\.\d+\.\d+\.\d+/);
  1224→          return m ? m[0] : null;
  1225→        }).filter((v): v is string => Boolean(v))]),
  1226→        attemptZoneTransfer(cleanDomain),
  1227→        analyzeRobotsAndSitemap(cleanDomain),
  1228→        analyzeJSFiles(cleanDomain),
  1229→        detectWAF(cleanDomain),
  1230→        deepFingerprint(cleanDomain, rawHeaders, pageContent),
  1231→        harvestEmails(cleanDomain, pageContent),
  1232→      ]);
  1233→
  1234→      allFindings.push(...ctFindings, ...waybackFindings, ...rdnsFindings, ...axfrFindings, ...robotsFindings, ...jsFindings, ...wafFindings, ...fpFindings, ...emailFindings);
  1235→    } else {
  1236→      // Quick scan: still run CT logs and basic fingerprinting (lightweight)
  1237→      const [ctFindings, fpFindings] = await Promise.all([
  1238→        enumerateCTLogs(cleanDomain),
  1239→        deepFingerprint(cleanDomain, rawHeaders, pageContent),
  1240→      ]);
  1241→      allFindings.push(...ctFindings, ...fpFindings);
  1242→    }
  1243→
  1244→    // ── Technology findings ─────────────────────────────────────────
  1245→    for (const tech of allTech) {
  1246→      allFindings.push({
  1247→        title: `Technology Detected: ${tech}`, severity: 'info', category: 'technology',
  1248→        description: `${tech} identified on ${cleanDomain} via HTTP headers, SSL cert metadata, or response analysis.`,
  1249→        evidence: 'Detected via HTTP/SSL/response fingerprinting', asset: tech,
  1250→      });
  1251→    }
  1252→
  1253→    // ── Wildcard DNS check ─────────────────────────────────────────
  1254→    const wcIp = await nativeResolveIP(`nonexistent-wildcard-test-12345.${cleanDomain}`);
  1255→    if (wcIp) {
  1256→      allFindings.push({
  1257→        title: 'Wildcard DNS Record Detected', severity: 'medium', category: 'dns',
  1258→        description: `Wildcard DNS active — even nonexistent subdomains resolve to ${wcIp}. Subdomain enumeration may have false positives.`,
  1259→        evidence: `nonexistent-wildcard-test-12345.${cleanDomain} resolved to ${wcIp}`, asset: cleanDomain,
  1260→      });
  1261→    }
  1262→
  1263→    // ── Save to DB ─────────────────────────────────────────────────
  1264→    for (const f of allFindings) {
  1265→      // Destructure out any extra fields not in Prisma schema (e.g. technologies)
  1266→      const { title: _t, severity: _s, category: _c, description: _d, evidence: _e, asset: _a, ...rest } = f;
  1267→      const safeData = { scanId: scan.id, title: f.title, severity: f.severity, category: f.category, description: f.description, evidence: f.evidence, asset: f.asset };
  1268→      await db.finding.create({ data: safeData });
  1269→    }
  1270→
  1271→    // ── Risk score ─────────────────────────────────────────────────
  1272→    const c = allFindings.filter(f => f.severity === 'critical').length;
  1273→    const h = allFindings.filter(f => f.severity === 'high').length;
  1274→    const m = allFindings.filter(f => f.severity === 'medium').length;
  1275→    const l = allFindings.filter(f => f.severity === 'low').length;
  1276→    const i = allFindings.filter(f => f.severity === 'info').length;
  1277→    const riskScore = Math.min(100, Math.round(c*25 + h*15 + m*8 + l*3 + i*1));
  1278→
  1279→    await db.scan.update({
  1280→      where: { id: scan.id },
  1281→      data: { status: 'completed', completedAt: new Date(), riskScore, totalVulns: allFindings.length, criticalCount: c, highCount: h, mediumCount: m, lowCount: l, infoCount: i },
  1282→    });
  1283→
  1284→    return NextResponse.json({
  1285→      success: true,
  1286→      scan: {
  1287→        id: scan.id, domain: cleanDomain, status: 'completed', riskScore,
  1288→        totalVulns: allFindings.length, critical: c, high: h, medium: m, low: l, info: i,
  1289→        findings: allFindings.map(f => ({
  1290→          id: f.title.toLowerCase().replace(/\s+/g, '-').substring(0, 20),
  1291→          ...f,
  1292→        })),
  1293→      },
  1294→    });
  1295→  } catch (error) {
  1296→    console.error('Scan error:', error);
  1297→    return NextResponse.json({ error: 'Scan failed: ' + (error instanceof Error ? error.message : 'unknown') }, { status: 500 });
