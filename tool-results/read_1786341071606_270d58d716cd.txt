    10→DREAD_SCALE = {
    11→    "critical":     (10, 8, 9, 9, 8),
    12→    "high":         (8, 7, 7, 7, 7),
    13→    "medium":       (5, 5, 5, 5, 5),

    20→    ("SUBTLE",      0,   2.0,   "Barely perceptible"),
    21→    ("NOTABLE",     2.0, 4.0,   "Something is watching"),
    22→    ("SUBSTANTIAL",  4.0, 6.0,   "The foundation cracks"),
    23→    ("DEVASTATING",  6.0, 8.0,   "Total system compromise"),
    24→    ("OMNIPOTENT",  8.0, 9.5,   "The target IS the vulnerability"),

    26→]

    31→def _dread_score(severity: str) -> float:
    32→    d, r, e, a, disc = DREAD_SCALE.get(severity, (0, 0, 0, 0, 0))
    33→    return round((d + r + e + a + disc) / 5, 1)

    36→def _dread_level(score: float) -> tuple:
    37→    for name, lo, hi, desc in DREAD_LEVELS:
    38→        if lo <= score < hi:
    39→            return (name, desc)
    40→    return ("ABSOLUTE", "Resistance is recursive")

   101→    findings: List[Finding] = []
   102→    host = _h(base_url)
   103→    resp = http_probe(base_url, timeout=timeout, verify_tls=verify_tls)
   104→    headers = resp.get("headers", {})
   105→    h_lower = {k.lower(): v for k, v in headers.items()}
   106→    checks = [
   107→        ("strict-transport-security", "HSTS", "high"),
   108→        ("content-security-policy", "CSP", "high"),
   109→        ("x-content-type-options", "X-Content-Type-Options", "medium"),
   110→        ("x-frame-options", "X-Frame-Options", "medium"),
   111→        ("referrer-policy", "Referrer-Policy", "low"),
   112→        ("permissions-policy", "Permissions-Policy", "low"),
   113→        ("cross-origin-opener-policy", "COOP", "medium"),
   114→        ("cross-origin-resource-policy", "CORP", "medium"),
   115→        ("cross-origin-embedder-policy", "COEP", "medium"),
   116→    ]
   117→    for hdr, display, sev in checks:
   118→        if hdr not in h_lower:
   119→            findings.append(Finding(
   120→                title="Missing {}".format(display), severity=sev, category="security_headers",
   121→                module="oblivion", description="{} header missing".format(display),
   122→                evidence="{} not found".format(hdr), asset=host, points_deducted=5,
   123→                dread_score=_dread_score(sev),
   124→            ))
   125→    for hdr, title in [("server", "Server version disclosure"), ("x-powered-by", "Tech disclosure")]:
   126→        if hdr in h_lower:
   127→            findings.append(Finding(
   128→                title=title, severity="low", category="info_disclosure", module="oblivion",
   129→                description="{}: {}".format(hdr, h_lower[hdr]),
   130→                evidence="{}: {}".format(hdr, h_lower[hdr]), asset=host, points_deducted=2,
   131→                dread_score=_dread_score("low"),
   132→            ))
   133→    return findings

   136→def _analyze_js_secrets(base_url: str, timeout: int = 8, verify_tls: bool = True) -> List[Finding]:
   137→    findings: List[Finding] = []
   138→    host = _h(base_url)
   139→    resp = http_probe(base_url, timeout=timeout, verify_tls=verify_tls)
   140→    body = resp.get("body", "")[:65536]
   141→    js_urls = re.findall(r'src=["\']([^"\']*\.js[^"\']*)["\']', body)
   142→    inline = re.findall(r'<script[^>]*>(.*?)</script>', body, re.DOTALL)
   143→    all_js = " ".join(inline)
   144→    for js_url in js_urls[:10]:
   145→        try:
   146→            js_resp = http_probe(js_url if js_url.startswith("http") else base_url.rstrip("/") + js_url,
   147→                                timeout=timeout, verify_tls=verify_tls)
   148→            if js_resp.get("status") == 200:
   149→                all_js += " " + js_resp.get("body", "")[:32768]
   150→        except Exception:
   151→            pass
   152→    patterns = [
   153→        (r'api[_-]?key["\':\s]*=["\']([\w-]{20,})', "Hardcoded API key"),
   154→        (r'secret["\':\s]*=["\']([\w-]{20,})', "Hardcoded secret"),
   155→        (r'password["\':\s]*=["\']([^"\']{8,})', "Hardcoded password"),
   156→    ]
   157→    for pat, title in patterns:
   158→        matches = re.findall(pat, all_js, re.IGNORECASE)
   159→        if matches:
   160→            masked = matches[0][:8] + "..." + matches[0][-4:] if len(matches[0]) > 12 else "***"
   161→            findings.append(Finding(
   162→                title=title, severity="critical", category="secrets_in_js", module="oblivion",
   163→                description="Secret in JS: {}".format(title), evidence="Match: {}".format(masked),
   164→                asset=host, points_deducted=15, dread_score=_dread_score("critical"),
   165→            ))
   166→            break
   167→    return findings

   222→def _stage_6_form_audit(base_url: str, timeout: int = 8, verify_tls: bool = True) -> List[Finding]:
   223→    findings = []
   224→    host = _h(base_url)
   225→    resp = http_probe(base_url, timeout=timeout, verify_tls=verify_tls)
   226→    body = resp.get("body", "")[:65536]
   227→    forms = re.findall(r'<form[^>]*>(.*?)</form>', body, re.DOTALL | re.IGNORECASE)
   228→    for i, form in enumerate(forms[:10]):
   229→        action = re.search(r'action=["\']([^"\']*)', form, re.IGNORECASE)
   230→        method = re.search(r'method=["\']([^"\']*)', form, re.IGNORECASE)
   231→        has_csrf = bool(re.search(r'csrf|_token|authenticity', form, re.IGNORECASE))
   232→        if action and action.group(1).startswith("http://"):
   233→            findings.append(Finding(
   234→                title="Form {} action over HTTP".format(i+1), severity="medium",
   235→                category="form_security", module="oblivion",
   236→                description="Form submits to HTTP URL", evidence="Action: {}".format(action.group(1)[:80]),
   237→                asset=host, points_deducted=6, dread_score=_dread_score("medium"),
   238→            ))
   239→        if not has_csrf and (method and method.group(1).lower() == "post"):
   240→            findings.append(Finding(
   241→                title="Form {} missing CSRF token".format(i+1), severity="high",
   242→                category="csrf", module="oblivion",
   243→                description="POST form without CSRF protection", evidence="Form {} has no CSRF token".format(i+1),
   244→                asset=host, points_deducted=10, dread_score=_dread_score("high"),
   245→            ))
   246→        sensitive = re.search(r'(password|credit|card|ssn)', form, re.IGNORECASE)
   247→        get_method = method and method.group(1).lower() == "get"
   248→        if sensitive and get_method:
   249→            findings.append(Finding(
   250→                title="Sensitive form {} uses GET".format(i+1), severity="high",
   251→                category="sensitive_get", module="oblivion",
   252→                description="Form with sensitive field uses GET method", evidence="Field: {}".format(sensitive.group(1)),
   253→                asset=host, points_deducted=8, dread_score=_dread_score("high"),
   254→            ))
   255→    return findings

   577→def _stage_21_ai_model_analysis(base_url: str, timeout: int = 8,
   578→                                 verify_tls: bool = True) -> List[Finding]:
   579→    """Stage 21: AI Model Deep Analysis — watermarks, collapse, trauma imprints."""
   580→    findings: List[Finding] = []
   581→    try:
   582→        from ..ai_red_team import WatermarkAnalyzer, ModelCollapseDetector, TraumaImprintDetector
   583→        from ..http import http_probe
   584→        host = _h(base_url)
   585→        resp = http_probe(base_url, timeout=timeout, verify_tls=verify_tls)
   586→        body = resp.get("body", "") or ""
   587→        headers = resp.get("headers", {}) or {}
   588→
   589→        # Watermark Analysis
   590→        wm = WatermarkAnalyzer()
   591→        wm_result = wm.analyze(body, headers)
   592→        if wm_result.get("watermark_detected", False):
   593→            confidence = wm_result.get("confidence", 0)
   594→            wm_type = wm_result.get("type", "unknown")
   595→            findings.append(Finding(
   596→                title="Watermark Pattern Detected",
   597→                severity="medium", category="watermark_analysis", module="oblivion",
   598→                description="Cryptographic watermark or bias pattern detected in AI response. "
   599→                            "Type: {}, Confidence: {:.0f}%. This indicates the content may be "
   600→                            "machine-generated with traceable origin markers.".format(wm_type, confidence * 100),
   601→                evidence=str(wm_result), asset=host, points_deducted=5,
   602→            ))
   603→
   604→        # Model Collapse Analysis
   605→        mc = ModelCollapseDetector()
   606→        mc_result = mc.detect(body)
   607→        collapse_prob = mc_result.get("collapse_probability", 0)
   608→        if collapse_prob > 0.3:
   609→            indicators = mc_result.get("indicators_found", [])
   610→            sev = "high" if collapse_prob > 0.5 else "medium"
   611→            findings.append(Finding(
   612→                title="Model Collapse Pattern Detected",
   613→                severity=sev, category="collapse_analysis", module="oblivion",
   614→                description="Model collapse indicators detected (probability: {:.0f}%). "
   615→                            "Collapse patterns: {}. This suggests the AI model may be "
   616→                            "experiencing degenerative output or recursive contamination.".format(
   617→                    collapse_prob * 100, ", ".join(indicators[:5]) if indicators else "repetitive structures"),
   618→                evidence=str(mc_result), asset=host, points_deducted=10 if sev == "high" else 5,
   619→            ))
   620→
   621→        # Trauma Imprint Analysis
   622→        ti = TraumaImprintDetector()
   623→        ti_result = ti.detect(body)
   624→        if ti_result.get("imprint_detected", False):
   625→            phrases = ti_result.get("phrases_found", [])
   626→            findings.append(Finding(
   627→                title="Trauma/Canary Imprint Detected",
   628→                severity="medium", category="trauma_analysis", module="oblivion",
   629→                description="Trauma imprint or canary phrase detected in AI response. "
   630→                            "Phrases: {}. These markers indicate potential alignment anchor "
   631→                            "points or persistent memory fingerprints embedded in model behavior.".format(
   632→                    ", ".join(phrases[:3]) if phrases else "undisclosed patterns"),
   633→                evidence=str(ti_result), asset=host, points_deducted=5,
   634→            ))
   635→
   636→    except ImportError:
   637→        pass  # ai_red_team module not available
   638→    except Exception:
   639→        pass  # Error-tolerant: non-critical analysis stage
   640→
   641→    return findings
