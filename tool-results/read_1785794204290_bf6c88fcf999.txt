   391→   391→def generate_html_report(data: Dict[str, Any], output_path: str = "") -> str:
   392→   392→    """Generate a Chart.js-powered HTML report. Returns file path.
   393→   393→
   394→   394→    Phase F: Complete overhaul with interactive charts:
   395→   395→    - Score gauge (doughnut with center text)
   396→   396→    - Severity breakdown (doughnut)
   397→   397→    - Findings per module (vertical bar)
   398→   398→    - Points deducted per module (horizontal bar)
   399→   399→    - Category distribution (horizontal bar)
   400→   400→    - Severity stacked per module (stacked bar)
   401→   401→    """
   402→   402→    if not output_path:
   403→   403→        ts = datetime.now().strftime("%Y%m%d_%H%M%S")
   404→   404→        target = data.get("target", "scan").replace("/", "_").replace(":", "")[:30]
   405→   405→        output_path = f"reconpro_report_{target}_{ts}.html"
   406→   406→
   407→   407→    score = data.get("total_score", 0)
   408→   408→    grade = data.get("grade", "N/A")
   409→   409→    target = data.get("target", "Unknown")
   410→   410→    findings = data.get("findings", [])
   411→   411→    sc = data.get("severity_counts", {})
   412→   412→    modules = data.get("modules_run", [])
   413→   413→    badge = data.get("badge_markdown", "")
   414→   414→    scan_time = data.get("scan_time", "")
   415→   415→
   416→   416→    # Version
   417→   417→    try:
   418→   418→        from . import __version__ as _v
   419→   419→        version_str = _v
   420→   420→    except Exception:
   421→   421→        version_str = "7.0.0"
   422→   422→
   423→   423→    # Theme colors
   424→   424→    grade_color = _report_theme.grade_color(grade)
   425→   425→    html_style = _report_theme.html_style()
   426→   426→    h_body_bg = html_style.get("body_bg", "#0a0a0f")
   427→   427→    h_card_bg = html_style.get("card_bg", "#111118")
   428→   428→    h_border = html_style.get("border", "#1a1a2e")
   429→   429→    h_text = html_style.get("text", "#e0e0e0")
   430→   430→    h_muted = html_style.get("muted", "#666")
   431→   431→    h_crit_color = _report_theme.sev_hex("critical")
   432→   432→    h_high_color = _report_theme.sev_hex("high")
   433→   433→    h_med_color = _report_theme.sev_hex("medium")
   434→   434→    h_low_color = _report_theme.sev_hex("low")
   435→   435→    h_info_color = _report_theme.sev_hex("info")
   436→   436→    h_accent = _report_theme.CYAN if hasattr(_report_theme, 'CYAN') else "#06b6d4"
   437→   437→
   438→   438→    # Data processing
   439→   439→    mod_counts = _module_breakdown(findings)
   440→   440→    cat_counts = _category_breakdown(findings)
   441→   441→    pts_mod = _points_per_module(findings)
   442→   442→    sev_matrix = _severity_per_module(findings)
   443→   443→    total_findings = len(findings)
   444→   444→    total_pts = sum(f.get("points_deducted", 0) for f in findings)
   445→   445→
   446→   446→    # ── Chart scripts ──
   447→   447→    chart_score = _chart_score_gauge(score, grade, grade_color)
   448→   448→    chart_severity = _chart_severity_doughnut(sc)
   449→   449→    chart_modules = _chart_module_bar(mod_counts)
   450→   450→    chart_points = _chart_points_hbar(pts_mod)
   451→   451→    chart_categories = _chart_category_hbar(cat_counts)
   452→   452→    chart_stacked = _chart_severity_stacked_bar(sev_matrix)
   453→   453→
   454→   454→    # ── Findings table rows ──
   455→   455→    sorted_findings = sorted(
   456→   456→        findings,
   457→   457→        key=lambda f: {"critical": 0, "high": 1, "medium": 2, "low": 3, "info": 4}.get(
   458→   458→            f.get("severity", "info").lower(), 99
   459→   459→        ),
   460→   460→    )
   461→   461→    findings_rows = ""
   462→   462→    for i, f in enumerate(sorted_findings, 1):
   463→   463→        sev = f.get("severity", "info").lower()
   464→   464→        color = _report_theme.sev_hex(sev)
