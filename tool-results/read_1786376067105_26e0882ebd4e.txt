   730→def render_module_help(module_id: str) -> str:
   731→    """Render formatted help for a specific module."""
   732→    # Try module_id as-is, then with underscores
   733→    mid = module_id.replace("-", "_")
   734→    help_data = MODULE_HELP.get(mid)
   735→    if not help_data:
   736→        help_data = CLI_COMMANDS.get(module_id.replace("-", "-"))
   737→    if not help_data:
   738→        return f"Unknown module: {module_id}\n\nRun 'reconpro list' to see available modules."
   739→
   740→    lines: List[str] = []
   741→    lines.append("")
   742→    lines.append(f"  ╔══════════════════════════════════════════════════╗")
   743→    lines.append(f"  ║  {module_id.upper():^50}║")
   744→    lines.append(f"  ╚══════════════════════════════════════════════════╝")
   745→    lines.append("")
   746→
   747→    # Description
   748→    lines.append(f"  DESCRIPTION")
   749→    lines.append(f"  {'─' * 50}")
   750→    lines.append(f"  {help_data.get('description', 'No description.')}")
   751→    lines.append("")
   752→
   753→    # Details
   754→    if "details" in help_data:
   755→        lines.append(f"  DETAILS")
   756→        lines.append(f"  {'─' * 50}")
   757→        for detail_line in help_data["details"].split(". "):
   758→            lines.append(f"  {detail_line.strip()}.")
   759→            if not detail_line.endswith("."):
   760→                pass
   761→        lines.append("")
   762→
   763→    # Examples
   764→    if "examples" in help_data:
   765→        lines.append(f"  EXAMPLES")
   766→        lines.append(f"  {'─' * 50}")
   767→        for ex in help_data["examples"]:
   768→            lines.append(f"    $ {ex}")
   769→        lines.append("")
   770→
   771→    # Output
   772→    if "output" in help_data:
   773→        lines.append(f"  OUTPUT")
   774→        lines.append(f"  {'─' * 50}")
   775→        lines.append(f"  {help_data['output']}")
   776→        lines.append("")
   777→
   778→    # See also
   779→    if "see_also" in help_data:
   780→        related = help_data["see_also"]
   781→        lines.append(f"  SEE ALSO")
   782→        lines.append(f"  {'─' * 50}")
   783→        lines.append(f"  {', '.join(related)}")
   784→        lines.append("")
   785→
   786→    return "\n".join(lines)
