   178→// --- Circular Progress Ring ---
   179→function CircularProgressRing({ score, size = 140, strokeWidth = 8 }: { score: number; size?: number; strokeWidth?: number }) {
   180→  const radius = (size - strokeWidth * 2) / 2;
   181→  const circumference = 2 * Math.PI * radius;
   182→  const offset = circumference - (score / 100) * circumference;
   183→
   184→  const color = score >= 80 ? '#f85149' : score >= 60 ? '#f97316' : score >= 40 ? '#eab308' : '#00ff88';
   185→  const glowColor = color + '50';
   186→
   187→  return (
   188→    <div className="relative" style={{ width: size, height: size }}>
   189→      <svg viewBox={`0 0 ${size} ${size}`} className="w-full h-full -rotate-90">
   190→        <circle
   191→          cx={size / 2}
   192→          cy={size / 2}
   193→          r={radius}
   194→          fill="none"
   195→          stroke="rgba(255,255,255,0.06)"
   196→          strokeWidth={strokeWidth}
   197→        />
   198→        <motion.circle
   199→          cx={size / 2}
   200→          cy={size / 2}
   201→          r={radius}
   202→          fill="none"
   203→          stroke={color}
   204→          strokeWidth={strokeWidth}
   205→          strokeLinecap="round"
   206→          strokeDasharray={circumference}
   207→          initial={{ strokeDashoffset: circumference }}
   208→          animate={{ strokeDashoffset: offset }}
   209→          transition={{ duration: 2, ease: 'easeOut', delay: 0.3 }}
   210→          style={{ filter: `drop-shadow(0 0 12px ${glowColor})` }}
   211→        />
   212→      </svg>
   213→      <div className="absolute inset-0 flex flex-col items-center justify-center">
   214→        <motion.span
   215→          className="font-bold font-mono text-3xl"
   216→          style={{ color }}
   217→          initial={{ opacity: 0, scale: 0.5 }}
   218→          animate={{ opacity: 1, scale: 1 }}
   219→          transition={{ delay: 0.8, type: 'spring', stiffness: 200 }}
   220→        >
   221→          {score}
   222→        </motion.span>
   223→        <span className="text-[10px] font-semibold uppercase tracking-widest text-[#8b949e]">
   224→          / 100
   225→        </span>
   226→      </div>
   227→    </div>
   228→  );
   229→}

   446→function RiskTrendChart() {
   447→  const data = useMemo(() => generateRiskTrendData(), []);
   448→  const labels = useMemo(() => generateDayLabels(), []);
   449→
   450→  const width = 700;
   451→  const height = 260;
   452→  const padding = { top: 20, right: 20, bottom: 40, left: 50 };
   453→  const chartW = width - padding.left - padding.right;
   454→  const chartH = height - padding.top - padding.bottom;
   455→
   456→  const minVal = Math.min(...data) - 5;
   457→  const maxVal = Math.max(...data) + 5;
   458→
   459→  const points = data.map((v, i) => ({
   460→    x: padding.left + (i / (data.length - 1)) * chartW,
   461→    y: padding.top + (1 - (v - minVal) / (maxVal - minVal)) * chartH,
   462→  }));
   463→
   464→  const linePath = points.map((p, i) => (i === 0 ? `M ${p.x} ${p.y}` : `L ${p.x} ${p.y}`)).join(' ');
   465→  const areaPath = `${linePath} L ${points[points.length - 1].x} ${padding.top + chartH} L ${points[0].x} ${padding.top + chartH} Z`;
   466→
   467→  const gridLines = [0, 0.25, 0.5, 0.75, 1].map((frac) => {
   468→    const y = padding.top + frac * chartH;
   469→    const val = Math.round(maxVal - frac * (maxVal - minVal));
   470→    return { y, val };
   471→  });
   472→
   473→  const xLabels = [0, 7, 14, 21, 29].map((i) => ({
   474→    x: padding.left + (i / (data.length - 1)) * chartW,
   475→    label: labels[i],
   476→  }));
   477→
   478→  return (
   479→    <motion.div
   480→      variants={itemVariants}
   481→      className="cyber-card overflow-hidden"
   482→      style={{
   483→        background: 'linear-gradient(135deg, rgba(0,255,136,0.03) 0%, rgba(10,13,20,0.95) 100%)',
   484→        border: '1px solid rgba(0,255,136,0.1)',
   485→      }}
   486→    >
   487→      <div className="p-6">
   488→        <div className="flex items-center justify-between mb-4">
   489→          <div className="flex items-center gap-2">
   490→            <Activity size={18} style={{ color: '#00ff88' }} />
   491→            <h3 className="text-sm font-bold uppercase tracking-widest text-[#e6edf3]">30-Day Risk Score Trend</h3>
