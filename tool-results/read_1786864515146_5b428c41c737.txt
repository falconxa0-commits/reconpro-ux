   211→function matchClassicalAlgorithm(cryptoName: string): ClassicalAlgorithm | null {
   212→  const name = cryptoName.toLowerCase();
   213→  if (name.includes('rsa-1024') || name === 'rsa_1024') return CLASSICAL_ALGORITHMS.rsa_1024;
   214→  if (name.includes('rsa-2048') || name === 'rsa_2048') return CLASSICAL_ALGORITHMS.rsa_2048;
   215→  if (name.includes('rsa-4096') || name === 'rsa_4096') return CLASSICAL_ALGORITHMS.rsa_4096;
   216→  if (name.includes('ecdsa') && (name.includes('p-256') || name.includes('p256'))) return CLASSICAL_ALGORITHMS.ecdsa_p256;
   217→  if (name.includes('ecdsa') && (name.includes('p-384') || name.includes('p384'))) return CLASSICAL_ALGORITHMS.ecdsa_p384;
   218→  if (name.includes('ecdh') && (name.includes('p-256') || name.includes('p256'))) return CLASSICAL_ALGORITHMS.ecdh_p256;
   219→  if (name.includes('aes-128') || name.includes('aes128')) return CLASSICAL_ALGORITHMS.aes_128;
   220→  if (name.includes('aes-256') || name.includes('aes256')) return CLASSICAL_ALGORITHMS.aes_256;
   221→  if (name.includes('sha-256') || name.includes('sha256')) return CLASSICAL_ALGORITHMS.sha_256;
   222→  if (name.includes('sha-384') || name.includes('sha384')) return CLASSICAL_ALGORITHMS.sha_384;
   223→  if (name.includes('3des') || name.includes('3-des') || name.includes('tripledes')) {
   224→    // 3DES is quantum-breakable via Grover's — treat as weak symmetric
   225→    return { name: '3DES', type: 'symmetric', securityBits: 80, quantumBreakable: true, qubitsRequired: 4096, nistLevel: 0 };
   226→  }
   227→  if (name.includes('sha-1') || name.includes('sha1')) {
   228→    return { name: 'SHA-1', type: 'hash', securityBits: 40, quantumBreakable: true, qubitsRequired: 1024, nistLevel: 0 };
   229→  }
   230→  if (name.includes('chacha20') || name.includes('chacha')) {
   231→    return { name: 'ChaCha20-Poly1305', type: 'symmetric', securityBits: 256, quantumBreakable: false, groverReduction: 128, nistLevel: 2 };
   232→  }
   233→  if (name.includes('tls 1.2')) {
   234→    return { name: 'TLS 1.2', type: 'key_exchange', securityBits: 112, quantumBreakable: true, qubitsRequired: 4096, nistLevel: 0 };
   235→  }
   236→  // Unknown — assume vulnerable for safety
   237→  return null;
   238→}

   242→function generateRecommendations(cryptoName: string, orgType: OrganizationType): PQCRecommendation[] {
   243→  const name = cryptoName.toLowerCase();
   244→  const recs: PQCRecommendation[] = [];
   245→  const isHighAssurance = orgType === 'central_bank' || orgType === 'clearing_house' || orgType === 'government';
   246→
   247→  // RSA key exchange → Kyber KEM
   248→  if (name.includes('rsa') || name.includes('tls 1.2') || name.includes('3des')) {
   249→    const target = isHighAssurance ? 'CRYSTALS-Kyber-1024' : 'CRYSTALS-Kyber-768';
   250→    const nistLevel = isHighAssurance ? 5 : 3;
   251→    recs.push({
   252→      from: cryptoName,
   253→      to: target,
   254→      nistLevel,
   255→      fips: 'FIPS 203',
   256→      priority: name.includes('3des') || name.includes('sha-1') ? 'CRITICAL' : 'HIGH',
   257→      effort: '6-12 months',
   258→    });
   259→  }
   260→
   261→  // ECDSA / ECDH → Kyber KEM + Dilithium signatures
   262→  if (name.includes('ecdsa') || name.includes('ecdh')) {
   263→    const kemTarget = isHighAssurance ? 'CRYSTALS-Kyber-1024' : 'CRYSTALS-Kyber-768';
   264→    const sigTarget = isHighAssurance ? 'CRYSTALS-Dilithium-5' : 'CRYSTALS-Dilithium-3';
   265→    const nistLevel = isHighAssurance ? 5 : 3;
   266→
   267→    if (name.includes('ecdh') || name.includes('key_exchange')) {
   268→      recs.push({
   269→        from: cryptoName,
   270→        to: kemTarget,
   271→        nistLevel,
   272→        fips: 'FIPS 203',
   273→        priority: 'HIGH',
   274→        effort: '4-9 months',
   275→      });
   276→    } else {
   277→      recs.push({
   278→        from: cryptoName,
   279→        to: sigTarget,
   280→        nistLevel,
   281→        fips: 'FIPS 204',
   282→        priority: 'HIGH',
   283→        effort: '6-12 months',
   284→      });
   285→    }
   286→  }
   287→
   288→  // AES-128 → AES-256 (Grover mitigation)
   289→  if (name.includes('aes-128') || name.includes('aes128')) {
   290→    recs.push({
   291→      from: cryptoName,
   292→      to: 'AES-256-GCM',
   293→      nistLevel: 3,
   294→      fips: 'FIPS 140-3',
   295→      priority: 'MODERATE',
   296→      effort: '1-3 months',
   297→    });
   298→  }

   301→  if (name.includes('sha-1') || name.includes('sha1')) {
   302→    recs.push({
   303→      from: cryptoName,
   304→      to: 'SHA-384',
   305→      nistLevel: 3,
   306→      fips: 'FIPS 180-4',
   307→      priority: 'CRITICAL',
   308→      effort: '1-3 months',
   309→    });
   310→  }
   311→
   312→  return recs;
   313→}

   576→// ── Transaction Integrity Assessment ───────────────────────────────────
   577→
   578→function buildTransactionIntegrity(
   579→  protocols: ProtocolAnalysis[],
   580→  orgType: OrganizationType,
   581→): TransactionIntegrity {
   582→  const hasVulnerable = protocols.some(p => p.quantumVulnerable);
   583→  const hasCriticalVulns = protocols.some(p =>
   584→    p.vulnerabilities.some(v => v.quantumBreakable && v.qubitsRequired && v.qubitsRequired < 4096),
   585→  );
   586→
   587→  const isHighAssurance = orgType === 'central_bank' || orgType === 'clearing_house';
   588→
   589→  let currentRisk: string;
   590→  if (hasCriticalVulns && isHighAssurance) {
   591→    currentRisk = 'SOVEREIGN RISK — Quantum adversary can forge or decrypt payment instructions, compromising monetary sovereignty and systemic financial stability.';
   592→  } else if (hasCriticalVulns) {
   593→    currentRisk = 'CRITICAL — Quantum adversary capable of decrypting transaction payloads and forging signatures on payment instructions.';
   594→  } else if (hasVulnerable) {
   595→    currentRisk = 'HIGH — Key exchange vulnerabilities enable "harvest now, decrypt later" attacks on archived transactions.';
   596→  } else {
   597→    currentRisk = 'LOW — All transaction paths protected with quantum-resistant cryptography.';
   598→  }
   599→
   600→  const recommendations: string[] = [];
   601→  if (hasCriticalVulns) {
   602→    recommendations.push('Immediately deploy hybrid classical/PQC key exchange for all payment channels');
   603→    recommendations.push('Implement transaction signing with ML-DSA (FIPS 204) for non-repudiation');
   604→    recommendations.push('Establish quantum-safe key ceremony procedures for HSM migration');
   605→  }
   606→  if (hasVulnerable) {
   607→    recommendations.push('Mandate PQC cipher suites for all inter-institutional communication');
   608→    recommendations.push('Deploy PQC-protected message authentication codes (MAC) for transaction integrity');
   609→  }
   610→  recommendations.push('Implement crypto-agility to enable rapid algorithm rotation as NIST standards evolve');
   611→  if (isHighAssurance) {
   612→    recommendations.push('Establish bilateral PQC key agreement with all correspondent banking partners');
   613→  }
   614→
   615→  return {
   616→    currentRisk,
   617→    pqcProtected: !hasVulnerable,
   618→    recommendations,
   619→  };
   620→}

   640→// ── Main Analysis Function ─────────────────────────────────────────────
   641→
   642→export function analyzePQCVault(params: PQCVaultParams): PQCVaultResult {
   643→  const { organizationType, protocols: protocolIds, tlsData, complianceFrameworks } = params;
   644→
   645→  // ── 1. Analyze each protocol ──
   646→  const protocolResults: ProtocolAnalysis[] = [];
   647→  let totalVulns = 0;
   648→  let criticalVulns = 0;
   649→
   650→  for (const pid of protocolIds) {
   651→    const entry = PROTOCOL_ANALYSIS[pid];
   652→    if (!entry) continue;
   653→
   654→    // Build vulnerabilities from current crypto
   655→    const vulnerabilities: Vulnerability[] = [];
   656→    for (const crypto of entry.currentCrypto) {
   657→      const matched = matchClassicalAlgorithm(crypto);
   658→      if (matched) {
   659→        vulnerabilities.push({
   660→          algorithm: matched.name,
   661→          type: matched.type,
   662→          quantumBreakable: matched.quantumBreakable,
   663→          qubitsRequired: matched.qubitsRequired,
   664→          nistLevel: matched.nistLevel,
   665→        });
   666→        totalVulns++;
   667→        if (matched.quantumBreakable) criticalVulns++;
   668→      } else {
   669→        // Unknown algorithm — assume vulnerable
   670→        vulnerabilities.push({
   671→          algorithm: crypto,
   672→          type: 'unknown',
   673→          quantumBreakable: true,
   674→          nistLevel: 0,
   675→        });
   676→        totalVulns++;
   677→        criticalVulns++;
   678→      }
   679→    }
   680→
   681→    // Build PQC recommendations
   682→    const pqcRecommendations: PQCRecommendation[] = [];
   683→    const seenTargets = new Set<string>();
   684→    for (const crypto of entry.currentCrypto) {
   685→      const recs = generateRecommendations(crypto, organizationType);
   686→      for (const rec of recs) {
   687→        const key = `${rec.from}→${rec.to}`;
   688→        if (!seenTargets.has(key)) {
   689→          seenTargets.add(key);
   690→          pqcRecommendations.push(rec);
   691→        }
   692→      }
   693→    }
   694→
   695→    // Compliance gaps
   696→    const complianceGaps = buildProtocolComplianceGaps(pid, entry);
   697→
   698→    protocolResults.push({
   699→      id: pid,
   700→      name: entry.name,
   701→      quantumVulnerable: entry.quantumVulnerable,
   702→      riskAssessment: entry.riskAssessment,
   703→      currentCrypto: entry.currentCrypto,
   704→      vulnerabilities,
   705→      pqcRecommendations,
   706→      complianceGaps,
   707→    });
   708→  }
   709→
   710→  // ── 2. Compute readiness score ──
   711→  const isHighAssurance = organizationType === 'central_bank' || organizationType === 'clearing_house' || organizationType === 'government';
   712→  const vulnerableProtocols = protocolResults.filter(p => p.quantumVulnerable).length;
   713→  const totalProtocols = protocolResults.length || 1;
   714→  const vulnRatio = vulnerableProtocols / totalProtocols;
   715→
   716→  // Base score deduction for vulnerable protocols
   717→  let score = 100;
   718→  score -= Math.round(vulnRatio * 60);
   719→
   720→  // Extra deduction for critical algorithms (SHA-1, 3DES)
   721→  for (const proto of protocolResults) {
   722→    if (proto.currentCrypto.some(c => c.toLowerCase().includes('sha-1'))) score -= 5;
   723→    if (proto.currentCrypto.some(c => c.toLowerCase().includes('3des'))) score -= 8;
   724→  }
   725→
   726→  // High-assurance organizations need higher bar
   727→  if (isHighAssurance && vulnerableProtocols > 0) {
   728→    score = Math.min(score, 35);
   729→  }
   730→
   731→  score = Math.max(0, Math.min(100, score));
   732→
   733→  let readinessLevel: ReadinessLevel;
   734→  if (score >= 85) readinessLevel = 'EXCELLENT';
   735→  else if (score >= 70) readinessLevel = 'GOOD';
   736→  else if (score >= 45) readinessLevel = 'MODERATE';
   737→  else if (score >= 25) readinessLevel = 'HIGH';
   738→  else readinessLevel = 'CRITICAL';
   739→
   740→  // ── 3. Compliance mapping ──
   741→  const compliance = getComplianceMapping(protocolResults, organizationType, complianceFrameworks);
   742→
   743→  // ── 4. Migration roadmap ──
   744→  const migrationRoadmap = buildMigrationRoadmap(protocolResults, organizationType);
   745→
   746→  // ── 5. Transaction integrity ──
   747→  const transactionIntegrity = buildTransactionIntegrity(protocolResults, organizationType);
   748→
   749→  // ── 6. Cost estimation ──
   750→  const multiplier = getOrgMultiplier(organizationType);
   751→  const baseMigrationCost = criticalVulns * 500000 + (totalVulns - criticalVulns) * 250000;
   752→  const totalCost = Math.round(baseMigrationCost * multiplier) + Math.round(2000000 * multiplier);
   753→  const estimatedMigrationCost = `$${totalCost.toLocaleString()}`;
   754→
   755→  let recommendedTimeline: string;
   756→  if (criticalVulns > 0) recommendedTimeline = '24-36 months (immediate action required)';
   757→  else if (vulnerableProtocols > 0) recommendedTimeline = '30-42 months';
   758→  else recommendedTimeline = '12-18 months (validation phase)';
   759→
   760→  return {
   761→    pqcReadinessScore: score,
   762→    readinessLevel,
   763→    protocols: protocolResults,
   764→    compliance,
   765→    migrationRoadmap,
   766→    transactionIntegrity,
   767→    totalVulnerabilities: totalVulns,
   768→    criticalVulnerabilities: criticalVulns,
   769→    estimatedMigrationCost,
   770→    recommendedTimeline,
   771→  };
   772→}
