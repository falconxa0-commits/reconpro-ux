    60→    60→def _render_summary(result, title: str = "RECONPRO") -> None:
    61→    61→    score = result.total_score
    62→    62→    grade = result.grade
    63→    63→    grade_color = GRADE_COLORS.get(grade, "bold bright_red")
    64→    64→    sc = result.severity_counts
    65→    65→    total = len(result.findings)
    66→    66→    bar_width = 50
    67→    67→    filled = int(score / 100 * bar_width)
    68→    68→    bar = "█" * filled + "░" * (bar_width - filled)
    69→    69→    lines = [
    70→    70→        Text(f"\n  {bar} [bold {grade_color}]{score}/100 ({grade})[/{grade_color}]"),
    71→    71→        Text(f"\n  Target: [cyan]{result.target}[/]"),
    72→    72→        Text(f"  Modules: [dim]{', '.join(result.modules_run)}[/]"),
    73→    73→        Text(""),
    74→    74→        Text(
    75→    75→            f"  Findings: [bold]{total}[/]  "
    76→    76→            f"[bright_red]{sc.get('critical', 0)} critical[/], "
    77→    77→            f"[red]{sc.get('high', 0)} high[/], "
    78→    78→            f"[yellow]{sc.get('medium', 0)} medium[/], "
    79→    79→            f"[green]{sc.get('low', 0)} low[/], "
    80→    80→            f"[dim]{sc.get('info', 0)} info[/]",
    81→    81→        ),
    82→    82→    ]
    83→    83→    if result.vibesec_score is not None:
    84→    84→        lines.append(Text(
    85→    85→            f"\n  VibeSec Score: [bold bright_green]{result.vibesec_score}/100 ({result.vibesec_grade})[/]"
    86→    86→        ))
    87→    87→    console.print(Panel(Group(*lines), border_style=grade_color,
    88→    88→                         title=f"[bold]{title}[/bold]", title_align="left", padding=(1, 2)))

    91→    91→def _render_findings_table(result, show_remediation: bool = False) -> None:
    92→    92→    if not result.findings:
    93→    93→        console.print("\n  [bright_green]No findings. Target is clean.[/]")
    94→    94→        return
    95→    95→    table = Table(title=f"Findings — {len(result.findings)} detected",
    96→    96→                  border_style="bright_white", header_style="bold bright_white")
    97→    97→    table.add_column("Severity", style="bold", width=10)
    98→    98→    table.add_column("Module", style="cyan", width=12)
    99→    99→    table.add_column("Category", style="magenta", width=18)
   100→   100→    table.add_column("Finding", style="white")
   101→   101→    table.add_column("Pts", style="yellow", width=5)
   102→   102→    if show_remediation:
   103→   103→        table.add_column("Fix", style="bright_green", width=50)
   104→   104→    for f in result.findings[:80]:
   105→   105→        sev = f.get("severity", "info")
   106→   106→        color = SEV_COLORS.get(sev, "white")
   107→   107→        row = [
   108→   108→            f"[{color}]{sev.upper()}[/{color}]",
   109→   109→            f.get("module", ""), f.get("category", ""),
   110→   110→            f.get("title", "")[:70], str(f.get("points_deducted", 0)),
   111→   111→        ]
   112→   112→        if show_remediation:
   113→   113→            row.append(f.get("remediation", "")[:50])
   114→   114→        table.add_row(*row)
   115→   115→    console.print(table)

   118→   118→def _render_module_breakdown(result) -> None:
   119→   119→    if not result.module_results:
   120→   120→        return
   121→   121→    t = Table(title="Module Breakdown", border_style="dim", header_style="bold dim")
   122→   122→    t.add_column("Module", style="bold", width=16)
   123→   123→    t.add_column("Findings", style="white", width=10)
   124→   124→    t.add_column("Critical", style="bright_red", width=10)
   125→   125→    t.add_column("High", style="red", width=8)
   126→   126→    t.add_column("Medium", style="yellow", width=8)
   127→   127→    for mod_id, mod_data in result.module_results.items():
   128→   128→        findings = mod_data.get("findings", [])
   129→   129→        t.add_row(mod_id.upper(), str(len(findings)),
   130→   130→                   str(sum(1 for f in findings if f.get("severity") == "critical")),
   131→   131→                   str(sum(1 for f in findings if f.get("severity") == "high")),
   132→   132→                   str(sum(1 for f in findings if f.get("severity") == "medium")))
   133→   133→    console.print(t)

   144→   144→def _render_remediations(result) -> None:
   145→   145→    actionable = [f for f in result.findings
   146→   146→                  if f.get("remediation") and f.get("severity") not in ("info",)
   147→   147→                  and f.get("points_deducted", 0) > 0]
   148→   148→    if not actionable:
   149→   149→        console.print("\n  [bright_green]No actionable fixes needed.[/]")
   150→   150→        return
   151→   151→    console.print("\n[bold bright_white]  Fix Commands:[/bold bright_white]\n")
   152→   152→    seen = set()
   153→   153→    for f in actionable:
   154→   154→        fix = f.get("remediation", "").strip()
   155→   155→        sev = f.get("severity", "info")
   156→   156→        color = SEV_COLORS.get(sev, "white")
   157→   157→        if fix and fix not in seen:
   158→   158→            seen.add(fix)
   159→   159→            console.print(f"  [{color}]{sev.upper():8}[/{color}]  {fix}")
   160→   160→    console.print()

   163→   163→def _output_result(result, args, title: str = "RECONPRO", show_remediation: bool = False) -> None:
   164→   164→    json_output = getattr(args, "json_output", False)
   165→   165→    output_file = getattr(args, "output_file", None)
   166→   166→    if json_output:
   167→   167→        text = json.dumps(result.to_dict(), indent=2, default=str)
   168→   168→        if output_file:
   169→   169→            with open(output_file, "w") as fp:
   170→   170→                fp.write(text)
   171→   171→            console.print(f"  [green]JSON saved: [cyan]{output_file}[/][/]")
   172→   172→        else:
   173→   173→            console.print(text)
   174→   174→    else:
   175→   175→        _render_summary(result, title=title)
   176→   176→        console.print()
   177→   177→        _render_module_breakdown(result)
   178→   178→        console.print()
   179→   179→        _render_findings_table(result, show_remediation=show_remediation)
   180→   180→        if show_remediation:
   181→   181→            _render_remediations(result)
   182→   182→        _render_badge(result)
   183→   183→        if output_file:
   184→   184→            with open(output_file, "w") as fp:
   185→   185→                fp.write(json.dumps(result.to_dict(), indent=2, default=str))
   186→   186→            console.print(f"\n  [green]JSON saved: [cyan]{output_file}[/][/]")
   187→   187→        # Auto-save to history
   188→   188→        save_scan(result.to_dict())
   189→   189→    console.print()

   192→   192→def _spinner_wrap(msg: str, fn, *a, **kw):
   193→   193→    """Run a function with a spinner."""
   194→   194→    with Progress(SpinnerColumn(), TextColumn("[bold]{task.description}[/]"),
   195→   195→                   TimeElapsedColumn(), console=console) as progress:
   196→   196→        task = progress.add_task(msg, total=None)
   197→   197→        result = fn(*a, **kw)
   198→   198→        progress.update(task, completed=True)
   199→   199→    return result
