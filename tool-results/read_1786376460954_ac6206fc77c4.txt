   274→   274→def _extract_domain(target: str) -> str:
   275→   275→    """Extract the root domain from a target."""
   276→   276→    host = target.replace("https://", "").replace("http://", "").split("/")[0]
   277→   277→    host = host.split(":")[0]
   278→   278→    parts = host.rsplit(".", 1)
   279→   279→    if len(parts) >= 2:
   280→   280→        # Check for common TLDs with two-part SLDs
   281→   281→        tld = parts[-1]
   282→   282→        if tld in ("co.uk", "com.au", "org.uk", "ac.uk", "gov.uk", "co.jp"):
   283→   283→            sub_parts = host.rsplit(".", 2)
   284→   284→            return ".".join(sub_parts[-2:]) if len(sub_parts) >= 2 else host
   285→   285→    return host
   286→   286→
   287→   287→
   288→   288→def _extract_email_domains(target: str) -> List[str]:
   289→   289→    """Generate likely email domains for the target."""
   290→   290→    domain = _extract_domain(target)
   291→   291→    domains = [domain]
   292→   292→    # Common email domain patterns
   293→   293→    domains.append(f"mail.{domain}")
   294→   294→    # Strip www
   295→   295→    if domain.startswith("www."):
   296→   296→        domains.append(domain[4:])
   297→   297→    return list(set(domains))

   300→   300→def _safe_probe(url: str, timeout: int = 5, verify_tls: bool = True) -> Dict[str, Any]:
   301→   301→    """Safe HTTP probe with error handling."""
   302→   302→    try:
   303→   303→        return http_probe(
   304→   304→            url, timeout=timeout, verify_tls=verify_tls,
   305→   305→            limiter=default_limiter,
   306→   306→        )
   307→   307→    except Exception:
   308→   308→        return {"ok": False, "status": 0, "reason": "error", "headers": {}, "body": ""}

   323→   323→def _extract_emails_from_domain(domain: str) -> List[str]:
   324→   324→    """Extract likely email addresses from a domain name."""
   325→   325→    prefixes = ["admin", "info", "support", "contact", "root", "test",
   326→   326→                "webmaster", "noreply", "security", "dev", "staging"]
   327→   327→    return [f"{p}@{domain}" for p in prefixes]

   329→   329→
   330→   330→def _decode_jwt(token: str) -> Optional[Dict[str, Any]]:
   331→   331→    """Decode JWT header and payload without verification."""
   332→   332→    parts = token.split(".")
   333→   333→    if len(parts) != 3:
   334→   334→        return None
   335→   335→    result = {}
   336→   336→    for i, label in enumerate(("header", "payload")):
   337→   337→        try:
   338→   338→            padded = parts[i] + "=" * (-len(parts[i]) % 4)
   339→   339→            decoded = base64.urlsafe_b64decode(padded)
   340→   340→            parsed = json.loads(decoded)
   341→   341→            result[label] = parsed
   342→   342→        except Exception:
   343→   343→            result[label] = None
   344→   344→    return result

   348→   348→# Cat 1: Paste Site Monitoring

   351→   351→def _check_paste_sites(
   352→   352→    target: str, base_url: str, timeout: int, verify_tls: bool,
   353→   353→) -> List[Finding]:
   354→   354→    findings: List[Finding] = []
   355→   355→    host = _extract_domain(target)
   356→   356→    queries = [host]
   357→   357→
   358→   358→    for site_id, site in PASTE_SITES.items():
   359→   359→        for query in queries:
   360→   360→            url = site["search_url"].format(query=urllib.parse.quote(query))
   361→   361→            resp = _safe_probe(url, timeout=min(timeout, 5), verify_tls=verify_tls)
   362→   362→
   363→   363→            if resp.get("ok") and resp.get("status") == 200:
   364→   364→                body = resp.get("body", "")
   365→   365→
   366→   366→                # Check for result indicators
   367→   367→                result_indicators = 0
   368→   368→                # Common patterns indicating search results
   369→   369→                if "result" in body.lower():
   370→   370→                    result_indicators += 1
   371→   371→                if "paste" in body.lower():
   372→   372→                    result_indicators += 1
   373→   373→                if host.lower() in body.lower():
   374→   374→                    result_indicators += 3  # Domain mention is strongest signal
   375→   375→                if re.search(rf'{re.escape(host)}[\s/<]', body, re.IGNORECASE):
   376→   376→                    result_indicators += 2
   377→   377→
   378→   378→                # Count occurrences
   379→   379→                domain_count = body.lower().count(host.lower())
   380→   380→
   381→   381→                if result_indicators >= 3:
   382→   382→                    sev = site.get("severity", "high")
   383→   383→                    pts = 18 if sev == "critical" else 12
   384→   384→                    findings.append(_make(
   385→   385→                        f"Paste Site Exposure — {site['name']}: domain found",
   386→   386→                        sev, "paste_exposure",
   387→   387→                        f"Target domain '{host}' found on {site['name']}. "
   388→   388→                        f"Indicators: {result_indicators}, domain mentions: {domain_count}. "
   389→   389→                        f"This may indicate leaked credentials, configuration data, or source code.",
   390→   390→                        f"Source: {site['name']}; Mentions: {domain_count}; URL: {url}",
   391→   391→                        host, pts,
   392→   392→                        "Review the paste content. Request removal if credentials/secrets are exposed.",
   393→   393→                        0.7,
   394→   394→                    ))
   395→   395→
   396→   396→    return findings

   400→   400→# Cat 2: Credential Pattern Detection
   401→   401→# ═══════════════════════════════════════════════════════════════════════════
   402→   402→
   403→   403→def _check_credential_patterns(
   404→   404→    target: str, base_url: str, timeout: int, verify_tls: bool,
   405→   405→) -> List[Finding]:
   406→   406→    """Check the target's own responses for leaked credentials."""
   407→   407→    findings: List[Finding] = []
   408→   408→    host = _extract_domain(target)
   409→   409→
   410→   410→    # Fetch the target's own pages for credential leakage
   411→   411→    paths_to_check = ["/", "/robots.txt", "/sitemap.xml", "/config.js",
   412→   412→                       "/.env", "/package.json", "/webpack.config.js"]
   413→   413→    for path in paths_to_check:
   414→   414→        url = base_url.rstrip("/") + path
   415→   415→        resp = _safe_probe(url, timeout=min(timeout, 5), verify_tls=verify_tls)
   416→   416→        if not resp.get("ok") or resp.get("status") in (404, 403):
   417→   417→            continue
   418→   418→
   419→   419→        body = resp.get("body", "")
   420→   420→        if len(body) < 50:
   421→   421→            continue
   422→   422→
   423→   423→        for cred_id, cred_info in CREDENTIAL_PATTERNS.items():
   424→   424→            matches = cred_info["pattern"].findall(body)
   425→   425→            if matches:
   426→   426→                count = len(matches)
   427→   427→                sev = cred_info.get("severity", "high")
   428→   428→                pts = 20 if sev == "critical" else 12
   429→   429→
   430→   430→                # Decode JWT tokens for extra evidence
   431→   431→                extra = ""
   432→   432→                if cred_id == "jwt_token" and matches:
   433→   433→                    jwt_data = _decode_jwt(matches[0])
   434→   434→                    if jwt_data and jwt_data.get("header"):
   435→   435→                        extra = f" JWT header: {json.dumps(jwt_data['header'])}"
   436→   436→
   437→   437→                evidence_str = (
   438→   438→                    f"Path: {path}; Pattern: {cred_id}; "
   439→   439→                    f"Matches: {count}; Sample: {str(matches[0])[:80]}{extra}"
   440→   440→                )
   441→   441→                findings.append(_make(
   442→   442→                    f"Credential Leak — {cred_info['description']} on {path}",
   443→   443→                    sev, "credential_leak",
   444→   444→                    f"{cred_info['description']} detected {count} time(s) at {path}. "
   445→   445→                    f"Immediate credential rotation recommended.{extra}",
   446→   446→                    evidence_str, host, pts,
