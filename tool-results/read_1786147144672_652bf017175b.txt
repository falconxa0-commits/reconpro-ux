  1678→            # Separator or bottom border

  1693→    @classmethod
  1694→    def _score_to_band_index(cls, score: float) -> int:
  1695→        """Map score to impact band index (0=Negligible, 4=Critical)."""
  1696→        for i, (name, lo, hi) in enumerate(cls._IMPACT_BANDS):
  1697→            if lo <= score < hi:
  1698→                return 4 - i  # Reverse so Critical is rightmost
  1699→        return 0

  1701→    @classmethod
  1702→    def _score_to_likelihood(cls, score: float, finding: Dict) -> int:
  1703→        """Estimate likelihood from score and finding metadata.
  1704→
  1705→        Higher CVSS exploitability metrics → higher likelihood.
  1706→        We approximate from score and severity.
  1707→        """
  1708→        severity = (finding.get("severity") or "").lower()
  1709→        is_compound = finding.get("is_compound", False)
  1710→        is_remote = True  # Most findings in recon are network-based
  1711→
  1712→        # Base likelihood from score
  1713→        base = score / 10.0
  1714→
  1715→        # Boost for compound findings (more likely to be exploitable)
  1716→        if is_compound:
  1717→            base = min(base + 0.15, 1.0)
  1718→
  1719→        # Adjust for severity
  1720→        sev_boost = {"critical": 0.1, "high": 0.05, "medium": 0, "low": -0.1, "info": -0.2}
  1721→        base += sev_boost.get(severity, 0)
  1722→        base = max(0.0, min(base, 1.0))
  1723→
  1724→        # Map to band index (0=Very Low, 4=Very High)
  1725→        if base >= 0.85:
  1726→            return 0
  1727→        elif base >= 0.65:
  1728→            return 1
  1729→        elif base >= 0.40:
  1730→            return 2
  1731→        elif base >= 0.15:
  1732→            return 3
  1733→        else:
  1734→            return 4

  1859→    @classmethod
  1860→    def suggest(
  1861→        cls,
  1862→        findings: List[Dict],
  1863→        domain: str = "",
  1864→        already_planned: Optional[List[str]] = None,
  1865→    ) -> List[Dict[str, Any]]:
  1866→        """Analyze findings and suggest deepening steps.
  1867→
  1868→        Returns a list of suggestion dicts: {tool, args, reason, trigger_findings}.
  1869→        """
  1870→        already_planned = already_planned or []
  1871→        suggestions: List[Dict[str, Any]] = []
  1872→
  1873→        for rule in cls.DEEPENING_RULES:
  1874→            threshold = cls._SEVERITY_ORDER.get(rule["severity_threshold"], 2)
  1875→            tool = rule["suggested_tool"]
  1876→
  1877→            # Skip if already planned
  1878→            if tool in already_planned:
  1879→                continue
  1880→
  1881→            # Check if any finding matches the trigger patterns
  1882→            matching_findings: List[str] = []
  1883→            for f in findings:
  1884→                f_sev = cls._SEVERITY_ORDER.get((f.get("severity") or "").lower(), 0)
  1885→                if f_sev < threshold:
  1886→                    continue
  1887→                f_text = f"{(f.get('title') or '').lower()} {(f.get('description') or '').lower()}"
  1888→                for pattern in rule["trigger_patterns"]:
  1889→                    if pattern in f_text:
  1890→                        matching_findings.append(f.get("title", ""))
  1891→                        break
  1892→

  1894→                args = dict(rule["suggested_args"])
  1895→                # Populate dynamic args
  1896→                if args.get("findings_json") == "__AUTO__":
  1897→                    try:
  1898→                        args["findings_json"] = json.dumps(findings[:20])
  1899→                    except (TypeError, ValueError):
  1900→                        args["findings_json"] = "[]"
  1901→                if args.get("domain") == "__AUTO__" and domain:
  1902→                    args["domain"] = domain
  1903→
  1904→                suggestions.append({
  1905→                    "tool": tool,
  1906→                    "args": args,
  1907→                    "reason": rule["reason"],
  1908→                    "trigger_findings": matching_findings[:5],
  1909→                    "trigger_count": len(matching_findings),
  1910→                })

  2170→    def classify_intent(self, goal: str) -> Dict[str, float]:
  2171→        """Score each intent based on the goal text.
  2172→
  2173→        Uses weighted keyword matching, proximity scoring, and modifiers.
  2174→        Returns a dict mapping intent names to float scores in [0, 1+].
  2175→        Supports 30+ intent categories.
  2176→        """
  2177→        goal_lower = goal.lower()
  2178→        # Pad for whole-word matching
  2179→        padded = f' {goal_lower} '
  2180→        scores: Dict[str, float] = {}
  2181→        words = goal_lower.split()
  2182→
  2183→        # Calculate global modifier
  2184→        modifier = 1.0
  2185→        for w in words:
  2186→            if w in self._AMPLIFIERS:
  2187→                modifier = max(modifier, self._AMPLIFIERS[w])
  2188→            if w in self._DAMPENERS:
  2189→                modifier = min(modifier, self._DAMPENERS[w])
  2190→
  2191→        for intent, keywords in self._INTENT_KEYWORDS.items():
  2192→            score = 0.0
  2193→            for pattern, weight in keywords:
  2194→                # Use whole-word matching to avoid false positives
  2195→                # (e.g., "repo" matching inside "report")
  2196→                if len(pattern.split()) > 1:
  2197→                    # Multi-word pattern: check substring (already specific enough)
  2198→                    matched = pattern in goal_lower
  2199→                else:
  2200→                    # Single-word pattern: check whole word with padding
  2201→                    matched = f' {pattern} ' in padded
  2202→                if matched:
  2203→                    # Boost if near a target-like word
  2204→                    proximity_boost = 1.0
  2205→                    pattern_idx = goal_lower.find(pattern)
  2206→                    for tw in (".com", ".io", ".net", ".org", "http", "localhost"):
  2207→                        tw_idx = goal_lower.find(tw)
  2208→                        if tw_idx >= 0 and abs(pattern_idx - tw_idx) < 40:
  2209→                            proximity_boost = 1.3
  2210→                            break
  2211→                    score += weight * proximity_boost
  2212→            scores[intent] = min(score * modifier, 2.0)
  2213→
  2214→        return scores
