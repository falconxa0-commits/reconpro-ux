    66→    66→# Fallback dict-based directed graph (used when networkx is absent)

    68→    68→
    69→    69→class _FallbackDiGraph:
    70→    70→    """Minimal directed-graph implementation backed by plain dicts."""
    71→    71→
    72→    72→    def __init__(self) -> None:
    73→    73→        self._nodes: Dict[str, Dict[str, Any]] = {}
    74→    74→        self._succ: Dict[str, Dict[str, Dict[str, Any]]] = defaultdict(dict)
    75→    75→        self._pred: Dict[str, Dict[str, Dict[str, Any]]] = defaultdict(dict)
    76→    76→
    77→    77→    # -- mutation ---------------------------------------------------------
    78→    78→    def add_node(self, node: str, **attr: Any) -> None:
    79→    79→        if node not in self._nodes:
    80→    80→            self._nodes[node] = {}
    81→    81→        self._nodes[node].update(attr)
    82→    82→
    83→    83→    def add_edge(self, u: str, v: str, **attr: Any) -> None:
    84→    84→        self.add_node(u)
    85→    85→        self.add_node(v)
    86→    86→        self._succ[u][v] = attr
    87→    87→        self._pred[v][u] = attr
    88→    88→
    89→    89→    def remove_node(self, node: str) -> None:
    90→    90→        self._nodes.pop(node, None)
    91→    91→        for neighbour in list(self._succ.get(node, {})):
    92→    92→            self._pred[neighbour].pop(node, None)
    93→    93→        self._succ.pop(node, None)
    94→    94→        for neighbour in list(self._pred.get(node, {})):
    95→    95→            self._succ[neighbour].pop(node, None)
    96→    96→        self._pred.pop(node, None)
    97→    97→
    98→    98→    # -- queries ----------------------------------------------------------
    99→    99→    def nodes(self, data: bool = False):
   100→   100→        if data:
   101→   101→            return list(self._nodes.items())
   102→   102→        return list(self._nodes.keys())
   103→   103→
   104→   104→    def edges(self, data: bool = False):
   105→   105→        result = []
   106→   106→        for u, nbrs in self._succ.items():
   107→   107→            for v, attr in nbrs.items():
   108→   108→                if data:
   109→   109→                    result.append((u, v, attr))
   110→   110→                else:
   111→   111→                    result.append((u, v))
   112→   112→        return result
   113→   113→
   114→   114→    def successors(self, node: str):
   115→   115→        return list(self._succ.get(node, {}).keys())
   116→   116→
   117→   117→    def predecessors(self, node: str):
   118→   118→        return list(self._pred.get(node, {}).keys())
   119→   119→
   120→   120→    def out_edges(self, node: str, data: bool = False):
   121→   121→        if data:
   122→   122→            return [(node, v, a) for v, a in self._succ.get(node, {}).items()]
   123→   123→        return [(node, v) for v in self._succ.get(node, {}).keys()]
   124→   124→
   125→   125→    def in_edges(self, node: str, data: bool = False):
   126→   126→        if data:
   127→   127→            return [(u, node, a) for u, a in self._pred.get(node, {}).items()]
   128→   128→        return [(u, node) for u in self._pred.get(node, {}).keys()]
   129→   129→
   130→   130→    def has_node(self, node: str) -> bool:
   131→   131→        return node in self._nodes
   132→   132→
   133→   133→    def get_node_data(self, node: str) -> Dict[str, Any]:
   134→   134→        return dict(self._nodes.get(node, {}))
   135→   135→
   136→   136→    def subgraph(self, nodes: Set[str]):
   137→   137→        g = _FallbackDiGraph()
   138→   138→        for n in nodes:
   139→   139→            if n in self._nodes:
   140→   140→                g.add_node(n, **self._nodes[n])
   141→   141→        for u, nbrs in self._succ.items():
   142→   142→            if u in nodes:
   143→   143→                for v, attr in nbrs.items():
   144→   144→                    if v in nodes:
   145→   145→                        g.add_edge(u, v, **attr)
   146→   146→        return g
   147→   147→
   148→   148→    def to_dict(self) -> Dict[str, Any]:
   149→   149→        return {
   150→   150→            "nodes": dict(self._nodes),
   151→   151→            "edges": [
   152→   152→                {"source": u, "target": v, **a}
   153→   153→                for u, nbrs in self._succ.items()
   154→   154→                for v, a in nbrs.items()
   155→   155→            ],
   156→   156→        }
   157→   157→
   158→   158→    @classmethod
   159→   159→    def from_dict(cls, data: Dict[str, Any]):
   160→   160→        g = cls()
   161→   161→        for node, attr in data.get("nodes", {}).items():
   162→   162→            g.add_node(node, **attr)
   163→   163→        for edge in data.get("edges", []):
   164→   164→            src = edge.pop("source")
   165→   165→            tgt = edge.pop("target")
   166→   166→            g.add_edge(src, tgt, **edge)
   167→   167→        return g

   197→   197→    def _node_id(self, ntype: str, label: str) -> str:
   198→   198→        """Deterministic node id: ``<type>:<label>``."""

   201→   201→    def _add_indexed_node(self, ntype: str, label: str, **attrs: Any) -> str:
   202→   202→        nid = self._node_id(ntype, label)
   203→   203→        self._g.add_node(nid, node_type=ntype, label=label, **attrs)

   209→   209→        self._edge_type_index[etype].append((src, dst))

   211→   211→    def _safe_sev(self, raw) -> str:
   212→   212→        """Normalise a severity value to lowercase string."""
   213→   213→        if raw is None:
   214→   214→            return "info"
   215→   215→        s = str(raw).strip().lower()
   216→   216→        if s in SEVERITY_ORDER:
   217→   217→            return s
   218→   218→        return "info"

   224→   224→    def add_scan_result(self, result_dict: Dict[str, Any]) -> None:
   225→   225→        """Ingest a full ``ReconProResult.to_dict()`` payload.
   226→   226→
   227→   227→        Creates nodes/edges for the target, every finding, associated
   228→   228→        technologies, and open ports.
   229→   229→        """
   230→   230→        target = result_dict.get("target") or result_dict.get("host", "unknown")
   231→   231→        self._add_indexed_node("target", target)
   232→   232→
   233→   233→        # Subdomains
   234→   234→        for sub in result_dict.get("subdomains", []):
   235→   235→            self.add_subdomain(sub, target)
   236→   236→
   237→   237→        # Ports & services
   238→   238→        for port_entry in result_dict.get("ports", []):
   239→   239→            if isinstance(port_entry, dict):
   240→   240→                port_num = port_entry.get("port", port_entry.get("number"))
   241→   241→                service = port_entry.get("service", port_entry.get("name", "unknown"))
   242→   242→                state = port_entry.get("state", "open")
   243→   243→            else:
   244→   244→                port_num = port_entry
   245→   245→                service = "unknown"
   246→   246→                state = "open"
   247→   247→
   248→   248→            if port_num is None:
   249→   249→                continue
   250→   250→
   251→   251→            port_id = self._add_indexed_node("port", f"{target}:{port_num}", port=int(port_num), state=state)
   252→   252→            target_id = self._node_id("target", target)
   253→   253→            self._add_indexed_edge(target_id, port_id, "has_port")
   254→   254→
   255→   255→            svc_id = self._add_indexed_node("service", f"{target}:{port_num}:{service}", name=service, port=int(port_num))
   256→   256→            self._add_indexed_edge(port_id, svc_id, "runs_on")
   257→   257→
   258→   258→        # Technologies
   259→   259→        for tech in result_dict.get("technologies", []):
   260→   260→            tech_name = tech if isinstance(tech, str) else tech.get("name", str(tech))
   261→   261→            tech_id = self._add_indexed_node("technology", tech_name)
   262→   262→            target_id = self._node_id("target", target)
   263→   263→            self._add_indexed_edge(target_id, tech_id, "uses_tech")
   264→   264→
   265→   265→        # Findings
   266→   266→        for finding in result_dict.get("findings", []):
   267→   267→            self.add_finding(finding)
   268→   268→
   269→   269→        # Certificates
   270→   270→        for cert in result_dict.get("certificates", []):
   271→   271→            if isinstance(cert, dict):
   272→   272→                cn = cert.get("common_name", cert.get("cn", "unknown"))
   273→   273→                issuer = cert.get("issuer", "")
   274→   274→                expires = cert.get("expires", "")
   275→   275→            else:
   276→   276→                cn = str(cert)
   277→   277→                issuer = ""
   278→   278→                expires = ""
   279→   279→            cert_id = self._add_indexed_node("certificate", f"{target}:{cn}", common_name=cn, issuer=issuer, expires=expires)
   280→   280→            target_id = self._node_id("target", target)
   281→   281→            self._add_indexed_edge(target_id, cert_id, "has_cert")
   282→   282→
   283→   283→        # Endpoints
   284→   284→        for ep in result_dict.get("endpoints", []):
   285→   285→            if isinstance(ep, dict):
   286→   286→                path = ep.get("path", ep.get("url", str(ep)))
   287→   287→                method = ep.get("method", "GET")
   288→   288→            else:
   289→   289→                path = str(ep)
   290→   290→                method = "GET"
   291→   291→            ep_id = self._add_indexed_node("endpoint", f"{target}:{method}:{path}", path=path, method=method)
   292→   292→            target_id = self._node_id("target", target)
   293→   293→            self._add_indexed_edge(target_id, ep_id, "exposes")

   295→   295→    def add_finding(self, finding_dict: Dict[str, Any]) -> None:
   296→   296→        """Add a finding node and connect it to its parent target."""
   297→   297→        title = finding_dict.get("title", finding_dict.get("name", finding_dict.get("description", "unnamed")))
   298→   298→        severity = self._safe_sev(finding_dict.get("severity"))
   299→   299→        target = finding_dict.get("target", finding_dict.get("host", "unknown"))
   300→   300→        finding_type = finding_dict.get("type", finding_dict.get("category", "generic"))
   301→   301→        description = finding_dict.get("description", "")
   302→   302→        evidence = finding_dict.get("evidence", "")
   303→   303→        cvss_score = finding_dict.get("cvss_score", finding_dict.get("cvss", None))
   304→   304→
   305→   305→        fid = self._add_indexed_node(
   306→   306→            "finding",
   307→   307→            f"{target}:{title}",
   308→   308→            title=title,
   309→   309→            severity=severity,
   310→   310→            finding_type=finding_type,
   311→   311→            description=description,
   312→   312→            evidence=evidence,
   313→   313→            cvss_score=cvss_score,
   314→   314→        )
   315→   315→
   316→   316→        # Ensure target node exists
   317→   317→        self._add_indexed_node("target", target)
   318→   318→        target_id = self._node_id("target", target)
   319→   319→        self._add_indexed_edge(target_id, fid, "has_vuln")
   320→   320→
   321→   321→        # Link to port if specified
   322→   322→        port_num = finding_dict.get("port")
   323→   323→        if port_num is not None:
   324→   324→            port_id = self._node_id("port", f"{target}:{port_num}")
   325→   325→            if self._g.has_node(port_id):
   326→   326→                self._add_indexed_edge(port_id, fid, "has_vuln")
   327→   327→
   328→   328→        # Link related CVEs
   329→   329→        for cve_ref in finding_dict.get("cves", finding_dict.get("references", [])):
   330→   330→            if isinstance(cve_ref, dict):
   331→   331→                cve_id = cve_ref.get("id", cve_ref.get("cve_id", ""))
   332→   332→                cve_sev = cve_ref.get("severity", severity)
   333→   333→                cve_desc = cve_ref.get("description", "")
   334→   334→                affected = cve_ref.get("affected", "")
   335→   335→            else:
   336→   336→                cve_id = str(cve_ref)
   337→   337→                cve_sev = severity
   338→   338→                cve_desc = ""
   339→   339→                affected = ""
   340→   340→            if cve_id:
   341→   341→                self.add_cve(cve_id, cve_sev, cve_desc, affected)
   342→   342→                self._add_indexed_edge(fid, self._node_id("cve", cve_id), "related_cve")
   343→   343→
   344→   344→        # Link related technologies
   345→   345→        for tech in finding_dict.get("technologies", []):
   346→   346→            tech_name = tech if isinstance(tech, str) else tech.get("name", str(tech))
   347→   347→            tech_id = self._add_indexed_node("technology", tech_name)
   348→   348→            self._add_indexed_edge(tech_id, fid, "exposes")

   368→   368→    def add_subdomain(self, domain: str, parent: str) -> None:
   369→   369→        """Link *domain* as a subdomain of *parent*."""
   370→   370→        sub_id = self._add_indexed_node("subdomain", domain)
   371→   371→        parent_id = self._node_id("target", parent)
   372→   372→        # Parent might be a subdomain itself
   373→   373→        if not self._g.has_node(parent_id):
   374→   374→            parent_id = self._node_id("subdomain", parent)
   375→   375→            if not self._g.has_node(parent_id):
   376→   376→                self._add_indexed_node("target", parent)
   377→   377→                parent_id = self._node_id("target", parent)
   378→   378→        self._add_indexed_edge(parent_id, sub_id, "has_subdomain")

   384→   384→    def get_attack_surface(self, target: str) -> Dict[str, List[Dict[str, Any]]]:
   385→   385→        """Return all reachable endpoints, open ports, and vulns for *target*.
   386→   386→
   387→   387→        Returns a dict with keys ``"endpoints"``, ``"ports"``,
   388→   388→        ``"vulnerabilities"``, and ``"technologies"``.
   389→   389→        """
   390→   390→        target_id = self._node_id("target", target)
   391→   391→        if not self._g.has_node(target_id):
   392→   392→            # Check subdomains
   393→   393→            target_id = self._node_id("subdomain", target)
   394→   394→            if not self._g.has_node(target_id):
   395→   395→                return {"endpoints": [], "ports": [], "vulnerabilities": [], "technologies": []}
   396→   396→
   397→   397→        visited = self._bfs_collect(target_id, max_depth=3)
   398→   398→        result: Dict[str, List[Dict[str, Any]]] = {
   399→   399→            "endpoints": [],
   400→   400→            "ports": [],
   401→   401→            "vulnerabilities": [],
   402→   402→            "technologies": [],
   403→   403→        }
   404→   404→
   405→   405→        for nid in visited:
   406→   406→            if HAS_NX:
   407→   407→                data = self._g.nodes[nid]
   408→   408→            else:
   409→   409→                data = self._g.get_node_data(nid)
   410→   410→            ntype = data.get("node_type", "")
   411→   411→            if ntype == "endpoint":
   412→   412→                result["endpoints"].append(data)
   413→   413→            elif ntype == "port":
   414→   414→                if data.get("state", "open") == "open":
   415→   415→                    result["ports"].append(data)
   416→   416→            elif ntype in ("vulnerability", "finding"):
   417→   417→                result["vulnerabilities"].append(data)
   418→   418→            elif ntype == "technology":
   419→   419→                result["technologies"].append(data)
   420→   420→
   421→   421→        return result

   423→   423→    def get_blast_radius(self, asset: str) -> List[Dict[str, Any]]:
   424→   424→        """BFS from *asset* to find all connected vulnerabilities.
   425→   425→
   426→   426→        *asset* can be a node label, an IP, or a partial node id.
   427→   427→        """
   428→   428→        start = self._resolve_node(asset)
   429→   429→        if start is None:
   430→   430→            return []
   431→   431→
   432→   432→        visited = self._bfs_collect(start, max_depth=10)
   433→   433→        vulns: List[Dict[str, Any]] = []
   434→   434→        for nid in visited:
   435→   435→            if HAS_NX:
   436→   436→                data = self._g.nodes[nid]
   437→   437→            else:
   438→   438→                data = self._g.get_node_data(nid)
   439→   439→            if data.get("node_type") in ("vulnerability", "finding", "cve"):
   440→   440→                vulns.append(data)
   441→   441→        return vulns

   443→   443→    def find_chains(self, target: str) -> List[List[Dict[str, Any]]]:
   444→   444→        """Find multi-hop attack chains originating from *target*.
   445→   445→
   446→   446→        Each chain follows a pattern like::
   447→   447→
   448→   448→            target → port → service → vulnerability → CVE
   449→   449→        """
   450→   450→        target_id = self._resolve_node(target)
   451→   451→        if target_id is None:
   452→   452→            return []
   453→   453→
   454→   454→        chains: List[List[Dict[str, Any]]] = []
   455→   455→        # DFS to find paths that end at CVE or finding nodes
   456→   456→        stack: List[Tuple[str, List[str]]] = [(target_id, [target_id])]
   457→   457→        seen_chains: Set[tuple] = set()
   458→   458→
   459→   459→        while stack:
   460→   460→            current, path = stack.pop()
   461→   461→            succs = self._g.successors(current) if HAS_NX else self._g.successors(current)
   462→   462→            for nxt in succs:
   463→   463→                if nxt in path:
   464→   464→                    continue
   465→   465→                new_path = path + [nxt]
   466→   466→                if HAS_NX:
   467→   467→                    nd = self._g.nodes[nxt]
   468→   468→                else:
   469→   469→                    nd = self._g.get_node_data(nxt)
   470→   470→                ntype = nd.get("node_type", "")
   471→   471→                # A chain is complete if it reaches a CVE or high-sev finding
   472→   472→                if ntype == "cve" or (ntype == "finding" and SEVERITY_ORDER.get(self._safe_sev(nd.get("severity")), 0) >= 2):
   473→   473→                    chain_key = tuple(new_path)
   474→   474→                    if chain_key not in seen_chains:
   475→   475→                        seen_chains.add(chain_key)
   476→   476→                        chain_data = []
   477→   477→                        for nid in new_path:
   478→   478→                            if HAS_NX:
   479→   479→                                chain_data.append(dict(self._g.nodes[nid]))
   480→   480→                            else:
   481→   481→                                chain_data.append(self._g.get_node_data(nid))
   482→   482→                        chains.append(chain_data)
   483→   483→                elif len(new_path) < 8:
   484→   484→                    stack.append((nxt, new_path))
   485→   485→
   486→   486→        # Sort by severity of the terminal node
   487→   487→        def _chain_sort_key(chain):
   488→   488→            last = chain[-1]
   489→   489→            return SEVERITY_ORDER.get(self._safe_sev(last.get("severity")), 0)
   490→   490→
   491→   491→        chains.sort(key=_chain_sort_key, reverse=True)
   492→   492→        return chains

   494→   494→    def get_vulnerable_assets(self) -> List[Dict[str, Any]]:
   495→   495→        """Return all assets (targets/subdomains) with HIGH or CRITICAL findings."""
   496→   496→        vulnerable: List[Dict[str, Any]] = []
   497→   497→        seen: Set[str] = set()
   498→   498→
   499→   499→        for nid in self._node_type_index.get("finding", set()):
   500→   500→            if HAS_NX:
   501→   501→                data = self._g.nodes[nid]
   502→   502→            else:
   503→   503→                data = self._g.get_node_data(nid)
   504→   504→            sev = self._safe_sev(data.get("severity"))
   505→   505→            if SEVERITY_ORDER.get(sev, 0) < 3:
   506→   506→                continue
   507→   507→            # Walk predecessors to find the owning target/subdomain
   508→   508→            for pred in (self._g.predecessors(nid) if HAS_NX else self._g.predecessors(nid)):
   509→   509→                if HAS_NX:
   510→   510→                    pdata = self._g.nodes[pred]
   511→   511→                else:
   512→   512→                    pdata = self._g.get_node_data(pred)
   513→   513→                ptype = pdata.get("node_type", "")
   514→   514→                if ptype in ("target", "subdomain", "asset") and pred not in seen:
   515→   515→                    seen.add(pred)
   516→   516→                    vulnerable.append({"node_id": pred, **pdata, "critical_finding": data})
   517→   517→        return vulnerable

   519→   519→    def correlate(self) -> List[Dict[str, Any]]:
   520→   520→        """Cross-reference findings to suggest related vulnerabilities.
   521→   521→
   522→   522→        Returns a list of correlation suggestions, each containing
   523→   523→        the two related node ids, the relationship type, and a
   524→   524→        confidence score.
   525→   525→        """
   526→   526→        suggestions: List[Dict[str, Any]] = []
   527→   527→        # Group findings by technology
   528→   528→        tech_findings: Dict[str, List[str]] = defaultdict(list)
   529→   529→        for nid in self._node_type_index.get("finding", set()):
   530→   530→            if HAS_NX:
   531→   531→                data = self._g.nodes[nid]
   532→   532→            else:
   533→   533→                data = self._g.get_node_data(nid)
   534→   534→            for pred in (self._g.predecessors(nid) if HAS_NX else self._g.predecessors(nid)):
   535→   535→                if HAS_NX:
   536→   536→                    pdata = self._g.nodes[pred]
   537→   537→                else:
   538→   538→                    pdata = self._g.get_node_data(pred)
   539→   539→                if pdata.get("node_type") == "technology":
   540→   540→                    tech_findings[pred].append(nid)
   541→   541→
   542→   542→        for tech_id, finding_ids in tech_findings.items():
   543→   543→            if len(finding_ids) < 2:
   544→   544→                continue
   545→   545→            for i in range(len(finding_ids)):
   546→   546→                for j in range(i + 1, len(finding_ids)):
   547→   547→                    f1 = finding_ids[i]
   548→   548→                    f2 = finding_ids[j]
   549→   549→                    if HAS_NX:
   550→   550→                        d1 = self._g.nodes[f1]
   551→   551→                        d2 = self._g.nodes[f2]
   552→   552→                    else:
   553→   553→                        d1 = self._g.get_node_data(f1)
   554→   554→                        d2 = self._g.get_node_data(f2)
   555→   555→                    # If both share a technology, suggest they may be related
   556→   556→                    conf = 0.6
   557→   557→                    if d1.get("severity") == d2.get("severity"):
   558→   558→                        conf += 0.2
   559→   559→                    if d1.get("finding_type") == d2.get("finding_type"):
   560→   560→                        conf += 0.2
   561→   561→                    suggestions.append({
   562→   562→                        "source": f1,
   563→   563→                        "target": f2,
   564→   564→                        "relationship": "shared_technology",
   565→   565→                        "technology": tech_id,
   566→   566→                        "confidence": min(conf, 1.0),
   567→   567→                        "source_severity": d1.get("severity"),
   568→   568→                        "target_severity": d2.get("severity"),
   569→   569→                    })
   570→   570→
   571→   571→        # Also correlate by port: findings on the same port may be related
   572→   572→        port_findings: Dict[str, List[str]] = defaultdict(list)
   573→   573→        for nid in self._node_type_index.get("finding", set()):
   574→   574→            for pred in (self._g.predecessors(nid) if HAS_NX else self._g.predecessors(nid)):
   575→   575→                if HAS_NX:
   576→   576→                    pdata = self._g.nodes[pred]
   577→   577→                else:
   578→   578→                    pdata = self._g.get_node_data(pred)
   579→   579→                if pdata.get("node_type") == "port":
   580→   580→                    port_findings[pred].append(nid)
   581→   581→
   582→   582→        for port_id, fids in port_findings.items():
   583→   583→            if len(fids) < 2:
   584→   584→                continue
   585→   585→            for i in range(len(fids)):
   586→   586→                for j in range(i + 1, len(fids)):
   587→   587→                    f1, f2 = fids[i], fids[j]
   588→   588→                    suggestions.append({
   589→   589→                        "source": f1,
   590→   590→                        "target": f2,
   591→   591→                        "relationship": "shared_port",
   592→   592→                        "port": port_id,
   593→   593→                        "confidence": 0.7,
   594→   594→                    })
   595→   595→
   596→   596→        suggestions.sort(key=lambda s: s["confidence"], reverse=True)
   597→   597→        return suggestions

   599→   599→    def to_cypher(self) -> str:
   600→   600→        """Export the graph as Cypher-like CREATE statements."""
   601→   601→        lines: List[str] = []
   602→   602→        # Nodes
   603→   603→        for nid in (self._g.nodes() if HAS_NX else self._g.nodes()):
   604→   604→            if HAS_NX:
   605→   605→                data = self._g.nodes[nid]
   606→   606→            else:
   607→   607→                data = self._g.get_node_data(nid)
   608→   608→            ntype = data.pop("node_type", "Unknown")
   609→   609→            label = data.pop("label", nid)
   610→   610→            props = json.dumps(data, default=str)
   611→   611→            safe_id = nid.replace(":", "__").replace("-", "_")
   612→   612→            lines.append(f'CREATE ({safe_id}:{ntype} {{label: "{label}", {props[1:-1]}}})')
   613→   613→            # Restore for edge generation below
   614→   614→            data["node_type"] = ntype
   615→   615→            data["label"] = label
   616→   616→
   617→   617→        # Edges
   618→   618→        for u, v, attr in (self._g.edges(data=True) if HAS_NX else self._g.edges(data=True)):
   619→   619→            etype = attr.get("edge_type", "RELATED_TO")
   620→   620→            props = {k: v for k, v in attr.items() if k != "edge_type"}
   621→   621→            props_str = json.dumps(props, default=str)
   622→   622→            safe_u = u.replace(":", "__").replace("-", "_")
   623→   623→            safe_v = v.replace(":", "__").replace("-", "_")
   624→   624→            if props_str == "{}":
   625→   625→                lines.append(f'CREATE ({safe_u})-[:{etype}]->({safe_v})')
   626→   626→            else:
   627→   627→                lines.append(f'CREATE ({safe_u})-[:{etype} {{{props_str[1:-1]}}}]->({safe_v})')
   628→   628→
   629→   629→        return "\n".join(lines)

   632→   632→        """Return a summary dict with node/edge counts broken down by type."""

   733→   733→    def _resolve_node(self, label: str) -> Optional[str]:
   734→   734→        """Try to resolve a human-readable label to a full node id."""
   735→   735→        # Exact node id
   736→   736→        if self._g.has_node(label):
   737→   737→            return label
   738→   738→        # Try as target
   739→   739→        candidate = self._node_id("target", label)
   740→   740→        if self._g.has_node(candidate):
   741→   741→            return candidate
   742→   742→        # Try as subdomain
   743→   743→        candidate = self._node_id("subdomain", label)
   744→   744→        if self._g.has_node(candidate):
   745→   745→            return candidate
   746→   746→        # Try as asset
   747→   747→        candidate = self._node_id("asset", label)
   748→   748→        if self._g.has_node(candidate):
   749→   749→            return candidate
   750→   750→        # Fuzzy: search all node labels
   751→   751→        for nid in self._g.nodes():
   752→   752→            if HAS_NX:
   753→   753→                data = self._g.nodes[nid]
   754→   754→            else:
   755→   755→                data = self._g.get_node_data(nid)
   756→   756→            if data.get("label") == label or nid.endswith(f":{label}"):
   757→   757→                return nid
   758→   758→        return None

   760→   760→    def _bfs_collect(self, start: str, max_depth: int = 5) -> Set[str]:
   761→   761→        """BFS from *start* up to *max_depth* hops, return visited nodes."""
   762→   762→        visited: Set[str] = set()
   763→   763→        queue: deque = deque()
   764→   764→        queue.append((start, 0))
   765→   765→        visited.add(start)
   766→   766→        while queue:
   767→   767→            node, depth = queue.popleft()
   768→   768→            if depth >= max_depth:
   769→   769→                continue
   770→   770→            for neighbour in (self._g.successors(node) if HAS_NX else self._g.successors(node)):
   771→   771→                if neighbour not in visited:
   772→   772→                    visited.add(neighbour)
   773→   773→                    queue.append((neighbour, depth + 1))
   774→   774→        return visited
