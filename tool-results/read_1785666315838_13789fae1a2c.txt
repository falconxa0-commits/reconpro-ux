    32→class ToolResult:
    33→    """Result of a tool execution."""
    34→    def __init__(self, output: str, success: bool = True, data: Any = None):
    35→        self.output = output
    36→        self.success = success
    37→        self.data = data
    38→
    39→    def to_dict(self) -> Dict:
    40→        return {"output": self.output, "success": self.success, "data": self.data}
    41→
    42→
    43→class Tool:
    44→    """A callable tool the agent can use."""
    45→    def __init__(self, name: str, description: str, parameters: Dict[str, Any],
    46→                 execute: Callable[..., ToolResult], category: str = "scan"):
    47→        self.name = name
    48→        self.description = description
    49→        self.parameters = parameters
    50→        self.execute = execute
    51→        self.category = category
    52→
    53→    def to_openai_tool(self) -> Dict:
    54→        return {
    55→            "type": "function",
    56→            "function": {
    57→                "name": self.name,
    58→                "description": self.description,
    59→                "parameters": {
    60→                    "type": "object",
    61→                    "properties": self.parameters,
    62→                    "required": list(self.parameters.keys()),
    63→                },
    64→            },
    65→        }

    68→class ToolRegistry:
    69→    """Registry of all available tools."""
    70→    def __init__(self):
    71→        self.tools: Dict[str, Tool] = {}
    72→
    73→    def register(self, tool: Tool) -> None:
    74→        self.tools[tool.name] = tool
    75→
    76→    def get(self, name: str) -> Optional[Tool]:
    77→        return self.tools.get(name)
    78→
    79→    def execute(self, name: str, **kwargs) -> ToolResult:
    80→        tool = self.tools.get(name)
    81→        if not tool:
    82→            return ToolResult(f"Tool '{name}' not found", success=False)
    83→        try:
    84→            return tool.execute(**kwargs)
    85→        except Exception as e:
    86→            return ToolResult(f"Tool error: {e}", success=False)
    87→
    88→    def list_tools(self) -> List[Dict]:
    89→        return [{"name": t.name, "description": t.description, "category": t.category}
    90→                for t in self.tools.values()]
    91→
    92→    def to_openai_tools(self) -> List[Dict]:
    93→        return [t.to_openai_tool() for t in self.tools.values()]

    98→class AgentMemory:
    99→    """Persistent memory for the Nexus agent."""
   100→    def __init__(self, project: str = "default"):
   101→        self.project = project
   102→        self._ensure_dir()
   103→        self.memory_file = MEMORY_DIR / f"{project}.json"
   104→        self._data = self._load()
   105→
   106→    def _ensure_dir(self) -> None:
   107→        MEMORY_DIR.mkdir(parents=True, exist_ok=True)
   108→
   109→    def _load(self) -> Dict:
   110→        if self.memory_file.exists():
   111→            try:
   112→                with open(self.memory_file) as f:
   113→                    return json.load(f)
   114→            except Exception:
   115→                pass
   116→        return {"project": self.project, "context": {}, "learned_patterns": [],
   117→                "conversation_history": [], "created": datetime.now().isoformat(),
   118→                "updated": datetime.now().isoformat()}
   119→
   120→    def _save(self) -> None:
   121→        self._data["updated"] = datetime.now().isoformat()
   122→        if len(self._data["conversation_history"]) > 100:
   123→            self._data["conversation_history"] = self._data["conversation_history"][-50:]
   124→        with open(self.memory_file, "w") as f:
   125→            json.dump(self._data, f, indent=2, default=str)
   126→
   127→    def add_message(self, role: str, content: str) -> None:
   128→        self._data["conversation_history"].append(
   129→            {"role": role, "content": content, "ts": datetime.now().isoformat()})
   130→        self._save()
   131→
   132→    def set_context(self, key: str, value: Any) -> None:
   133→        self._data["context"][key] = value
   134→        self._save()
   135→
   136→    def get_context(self, key: str, default=None) -> Any:
   137→        return self._data["context"].get(key, default)
   138→
   139→    def add_pattern(self, pattern: str) -> None:
   140→        if pattern not in self._data["learned_patterns"]:
   141→            self._data["learned_patterns"].append(pattern)
   142→            self._save()
   143→
   144→    def get_history(self, limit: int = 20) -> List[Dict]:
   145→        return self._data["conversation_history"][-limit:]
   146→
   147→    def get_context_summary(self) -> str:
   148→        ctx = self._data["context"]
   149→        parts = [f"  {k}: {v}" for k, v in ctx.items()] if ctx else []
   150→        patterns = self._data.get("learned_patterns", [])
   151→        if patterns:
   152→            parts.append(f"  Known patterns: {len(patterns)}")
   153→        return "\n".join(parts) if parts else "No prior context."

   261→def _tool_run_terminal(command: str) -> ToolResult:
   262→    dangerous = ["rm -rf /", "mkfs", "dd if=", ":(){ :|:& };:"]
   263→    for d in dangerous:
   264→        if d in command:
   265→            return ToolResult("Blocked dangerous command.", success=False)

   267→        result = subprocess.run(command, shell=True, capture_output=True, text=True, timeout=30)
   268→        output = (result.stdout + result.stderr)[:2000]
   269→        return ToolResult(f"Exit code: {result.returncode}\n{output}", success=result.returncode == 0)
   270→    except subprocess.TimeoutExpired:
   271→        return ToolResult("Command timed out (30s).", success=False)
   272→    except Exception as e:
   273→        return ToolResult(f"Command error: {e}", success=False)

   285→def _tool_search_code(pattern: str, path: str = ".") -> ToolResult:
   286→    try:
   287→        p = Path(path).expanduser()
   288→        if not p.is_dir(): p = p.parent
   289→        matches = []
   290→        for ext in ("*.py", "*.js", "*.ts", "*.jsx", "*.tsx", "*.go", "*.rs", "*.java", "*.rb", "*.php", "*.yml", "*.yaml", "*.json", "*.env"):
   291→            for fp in p.rglob(ext):
   292→                if ".git" in str(fp) or "node_modules" in str(fp): continue
   293→                try:
   294→                    content = fp.read_text()[:10000]
   295→                    for i, line in enumerate(content.split("\n"), 1):
   296→                        if re.search(pattern, line, re.IGNORECASE):
   297→                            matches.append(f"{fp.relative_to(p)}:{i}: {line.strip()[:80]}")
   298→                            if len(matches) >= 30: break
   299→                except Exception: continue
   300→            if len(matches) >= 30: break
   301→        if not matches:
   302→            return ToolResult(f"No matches for '{pattern}'.", success=True)
   303→        return ToolResult(f"Found {len(matches)} match(es):\n" + "\n".join(matches[:30]), success=True)
   304→    except Exception as e:
   305→        return ToolResult(f"Search error: {e}", success=False)

   390→# \u2500\u2500 Nexus Agent \u2500\u2500\u2500\u2500\u2500\u2500\u2500\u2500\u2500\u2500\u2500\u2500\u2500\u2500\u2500\u2500\u2500\u2500\u2500\u2500\u2500\u2500\u2500\u2500\u2500\u2500\u2500\u2500\u2500\u2500\u2500\u2500\u2500\u2500\u2500\u2500\n
   391→class NexusAgent:
   392→    """The Nexus Agent \u2014 LLM-powered or rule-based agentic security engine."""
   393→    def __init__(self, on_message: Optional[Callable] = None,
   394→                 on_finding: Optional[Callable] = None,
   395→                 on_status: Optional[Callable] = None,
   396→                 project: str = "default"):
   397→        self.registry = build_registry()
   398→        self.memory = AgentMemory(project)
   399→        self.on_message = on_message or (lambda msg, style="msg-nexus": None)
   400→        self.on_finding = on_finding or (lambda f: None)
   401→        self.on_status = on_status or (lambda s: None)
   402→        self._llm_available = bool(os.environ.get("OPENAI_API_KEY") or os.environ.get("ANTHROPIC_API_KEY"))
   403→        self._findings: List[Dict] = []
   404→        self._steps: List[str] = []
   405→        self._targets: List[str] = []
   406→
   407→    def _emit(self, msg: str, style: str = "msg-nexus") -> None:
   408→        self.on_message(msg, style)
   409→
   410→    def _emit_finding(self, finding: Dict) -> None:
   411→        self.on_finding(finding)
   412→        self._findings.append(finding)
   413→
   414→    def _emit_status(self, status: str) -> None:
   415→        self.on_status(status)
   416→
   417→    def execute(self, goal: str) -> Dict[str, Any]:
   418→        self._emit(f"Analyzing goal: [dim]{goal}[/]")
   419→        self.memory.add_message("user", goal)
   420→        if self._llm_available:
   421→            self._emit("LLM detected \u2014 using AI-powered reasoning.")
   422→            result = self._execute_llm(goal)
   423→        else:
   424→            self._emit("No LLM key found \u2014 using intelligent rule-based engine.")
   425→            self._emit("[dim]Set OPENAI_API_KEY or ANTHROPIC_API_KEY for AI mode.[/]")
   426→            result = self._execute_rules(goal)
   427→        self.memory.add_message("agent", f"Completed: {len(self._findings)} findings")
   428→        return result
   429→
   430→    def _execute_rules(self, goal: str) -> Dict[str, Any]:
   431→        goal_lower = goal.lower()
   432→        urls = re.findall(r'https?://[\w\\-.]+(?:/\S*)?', goal)
   433→        domains = re.findall(r'[\w][\w\\-]*\.[\w]{2,}(?:\.[\w]{2,})?', goal)
   434→        skip = {'the', 'and', 'for', 'with', 'all', 'of', 'to', 'from',
   435→                'that', 'this', 'scan', 'audit', 'check', 'find', 'get',

   557→    def _execute_llm(self, goal: str) -> Dict[str, Any]:
   558→        self._emit("Initializing LLM reasoning engine...")
   559→        system_prompt = self._build_system_prompt()
   560→        messages = [{"role": "system", "content": system_prompt}]
   561→        for msg in self.memory.get_history(limit=10):
   562→            if msg["role"] in ("user", "assistant"):
   563→                messages.append({"role": msg["role"], "content": msg["content"]})
   564→        messages.append({"role": "user", "content": goal})
   565→        for i in range(10):
   566→            self._emit_status(f"LLM reasoning (step {i + 1})...")
   567→            response = self._call_llm(messages)
   568→            if not response:
   569→                self._emit("LLM call failed. Falling back to rules.", "msg-warning")
   570→                return self._execute_rules(goal)
   571→            msg = response.get("content", "")
   572→            tool_calls = response.get("tool_calls", [])
   573→            if not tool_calls:
   574→                if msg:
   575→                    self._emit(msg)
   576→                    self.memory.add_message("assistant", msg)
   577→                break
   578→            for tc in tool_calls:
   579→                fn_name = tc.get("function", {}).get("name", "")
   580→                fn_args_str = tc.get("function", {}).get("arguments", "{}")
   581→                try:
   582→                    fn_args = json.loads(fn_args_str) if isinstance(fn_args_str, str) else fn_args_str
   583→                except json.JSONDecodeError:
   584→                    fn_args = {}
   585→                self._emit(f"Calling tool: [bold]{fn_name}[/] with {fn_args}")
   586→                self._emit_status(f"Executing: {fn_name}")
   587→                result = self.registry.execute(fn_name, **fn_args)
   588→                self._emit(result.output, "msg-success" if result.success else "msg-error")
   589→                self._steps.append(fn_name)
   590→                if result.data:
   591→                    if isinstance(result.data, dict) and "findings" in result.data:
   592→                        for f in result.data["findings"]: self._emit_finding(f)
   593→                    elif isinstance(result.data, dict) and "results" in result.data:
   594→                        for t, td in result.data["results"].items():
   595→                            self._targets.append(t)
   596→                            if isinstance(td, dict) and "findings" in td:
   597→                                for f in td["findings"]: self._emit_finding(f)
   598→                messages.append({"role": "assistant", "content": None, "tool_calls": [tc]})
   599→                messages.append({"role": "tool", "tool_call_id": tc.get("id", "call_" + fn_name),
   600→                                "content": result.output[:4000]})
   601→                self.memory.add_message("assistant", f"Called {fn_name}: {result.output[:200]}")
   602→        self._emit(f"Agent loop complete. {len(self._findings)} findings.", "msg-success")
   603→        return {"goal": goal, "steps": self._steps, "targets": self._targets,
   604→                "total_findings": len(self._findings), "findings": self._findings}

   628→    def _call_openai(self, api_key, messages):
   629→        import urllib.request
   630→        import ssl
   631→        url = "https://api.openai.com/v1/chat/completions"
   632→        model = os.environ.get("RECONPRO_MODEL", "gpt-4o-mini")
   633→        payload = {"model": model, "messages": messages,
   634→                   "tools": self.registry.to_openai_tools(),
   635→                   "max_tokens": 4096, "temperature": 0.1}
   636→        try:
   637→            req = urllib.request.Request(url, data=json.dumps(payload).encode(),
   638→                    headers={"Content-Type": "application/json", "Authorization": f"Bearer {api_key}"})
   639→            ctx = ssl.create_default_context()
   640→            with urllib.request.urlopen(req, timeout=60, context=ctx) as resp:
   641→                data = json.loads(resp.read().decode())
   642→            choice = data.get("choices", [{}])[0]
   643→            message = choice.get("message", {})
   644→            return {"content": message.get("content", ""), "tool_calls": message.get("tool_calls", [])}
   645→        except Exception as e:
   646→            self._emit(f"OpenAI error: {e}", "msg-error")
   647→            return None

   649→    def _call_anthropic(self, api_key, messages):
   650→        import urllib.request
   651→        import ssl
   652→        system_msg = ""
   653→        anthropic_messages = []
   654→        for m in messages:
   655→            if m["role"] == "system": system_msg = m["content"]
   656→            else: anthropic_messages.append(m)
   657→        url = "https://api.anthropic.com/v1/messages"
   658→        model = os.environ.get("RECONPRO_MODEL", "claude-sonnet-4-20250514")
   659→        tools = [{"name": t.name, "description": t.description,
   660→                  "input_schema": {"type": "object", "properties": t.parameters,
   661→                                  "required": list(t.parameters.keys())}}
   662→                 for t in self.registry.tools.values()]
   663→        payload = {"model": model, "max_tokens": 4096, "system": system_msg,
   664→                   "messages": anthropic_messages, "tools": tools if tools else None}
   665→        try:
   666→            req = urllib.request.Request(url, data=json.dumps(payload).encode(),
   667→                    headers={"Content-Type": "application/json", "x-api-key": api_key,
   668→                             "anthropic-version": "2023-06-01"})
   669→            ctx = ssl.create_default_context()
   670→            with urllib.request.urlopen(req, timeout=60, context=ctx) as resp:
   671→                data = json.loads(resp.read().decode())
   672→            content = data.get("content", [])
   673→            text_parts, tool_calls = [], []
   674→            for block in content:
   675→                if block["type"] == "text": text_parts.append(block["text"])
   676→                elif block["type"] == "tool_use":
   677→                    tool_calls.append({"id": block["id"], "type": "function",
   678→                                    "function": {"name": block["name"],
   679→                                               "arguments": json.dumps(block.get("input", {}))}})
   680→            return {"content": "\n".join(text_parts), "tool_calls": tool_calls}
   681→        except Exception as e:
   682→            self._emit(f"Anthropic error: {e}", "msg-error")
   683→            return None
