     7→    66→# Fallback dict-based directed graph (used when networkx is absent)
     8→    67→# ---------------------------------------------------------------------------
     9→    68→
    10→    69→class _FallbackDiGraph:
    11→    70→    """Minimal directed-graph implementation backed by plain dicts."""
    12→    71→
    13→    72→    def __init__(self) -> None:
    14→    73→        self._nodes: Dict[str, Dict[str, Any]] = {}
    15→    74→        self._succ: Dict[str, Dict[str, Dict[str, Any]]] = defaultdict(dict)
    16→    75→        self._pred: Dict[str, Dict[str, Dict[str, Any]]] = defaultdict(dict)
    17→    76→
    18→    77→    # -- mutation ---------------------------------------------------------
    19→    78→    def add_node(self, node: str, **attr: Any) -> None:
    20→    79→        if node not in self._nodes:
    21→    80→            self._nodes[node] = {}
    22→    81→        self._nodes[node].update(attr)
    23→    82→
    24→    83→    def add_edge(self, u: str, v: str, **attr: Any) -> None:
    25→    84→        self.add_node(u)
    26→    85→        self.add_node(v)
    27→    86→        self._succ[u][v] = attr
    28→    87→        self._pred[v][u] = attr
    29→    88→
    30→    89→    def remove_node(self, node: str) -> None:
    31→    90→        self._nodes.pop(node, None)
    32→    91→        for neighbour in list(self._succ.get(node, {})):
    33→    92→            self._pred[neighbour].pop(node, None)
    34→    93→        self._succ.pop(node, None)
    35→    94→        for neighbour in list(self._pred.get(node, {})):
    36→    95→            self._succ[neighbour].pop(node, None)
    37→    96→        self._pred.pop(node, None)
    38→    97→
    39→    98→    # -- queries ----------------------------------------------------------
    40→    99→    def nodes(self, data: bool = False):
    41→   100→        if data:
    42→   101→            return list(self._nodes.items())
    43→   102→        return list(self._nodes.keys())
    44→   103→
    45→   104→    def edges(self, data: bool = False):
    46→   105→        result = []
    47→   106→        for u, nbrs in self._succ.items():
    48→   107→            for v, attr in nbrs.items():
    49→   108→                if data:
    50→   109→                    result.append((u, v, attr))
    51→   110→                else:
    52→   111→                    result.append((u, v))
    53→   112→        return result
    54→   113→
    55→   114→    def successors(self, node: str):
    56→   115→        return list(self._succ.get(node, {}).keys())
    57→   116→
    58→   117→    def predecessors(self, node: str):
    59→   118→        return list(self._pred.get(node, {}).keys())
    60→   119→
    61→   120→    def out_edges(self, node: str, data: bool = False):
    62→   121→        if data:
    63→   122→            return [(node, v, a) for v, a in self._succ.get(node, {}).items()]
    64→   123→        return [(node, v) for v in self._succ.get(node, {}).keys()]
    65→   124→
    66→   125→    def in_edges(self, node: str, data: bool = False):
    67→   126→        if data:
    68→   127→            return [(u, node, a) for u, a in self._pred.get(node, {}).items()]
    69→   128→        return [(u, node) for u in self._pred.get(node, {}).keys()]
    70→   129→
    71→   130→    def has_node(self, node: str) -> bool:
    72→   131→        return node in self._nodes
    73→   132→
    74→   133→    def get_node_data(self, node: str) -> Dict[str, Any]:
    75→   134→        return dict(self._nodes.get(node, {}))
    76→   135→
    77→   136→    def subgraph(self, nodes: Set[str]):
    78→   137→        g = _FallbackDiGraph()
    79→   138→        for n in nodes:
    80→   139→            if n in self._nodes:
    81→   140→                g.add_node(n, **self._nodes[n])
    82→   141→        for u, nbrs in self._succ.items():
    83→   142→            if u in nodes:
    84→   143→                for v, attr in nbrs.items():
    85→   144→                    if v in nodes:
    86→   145→                        g.add_edge(u, v, **attr)
    87→   146→        return g
    88→   147→
    89→   148→    def to_dict(self) -> Dict[str, Any]:
    90→   149→        return {
    91→   150→            "nodes": dict(self._nodes),
    92→   151→            "edges": [
    93→   152→                {"source": u, "target": v, **a}
    94→   153→                for u, nbrs in self._succ.items()
    95→   154→                for v, a in nbrs.items()
    96→   155→            ],
    97→   156→        }
    98→   157→
    99→   158→    @classmethod
   100→   159→    def from_dict(cls, data: Dict[str, Any]):
   101→   160→        g = cls()
   102→   161→        for node, attr in data.get("nodes", {}).items():
   103→   162→            g.add_node(node, **attr)
   104→   163→        for edge in data.get("edges", []):
   105→   164→            src = edge.pop("source")
   106→   165→            tgt = edge.pop("target")
   107→   166→            g.add_edge(src, tgt, **edge)
   108→   167→        return g

   113→   172→    if HAS_NX:

   157→   216→        if s in SEVERITY_ORDER:

   364→   423→    def get_blast_radius(self, asset: str) -> List[Dict[str, Any]]:
   365→   424→        """BFS from *asset* to find all connected vulnerabilities.
   366→   425→
   367→   426→        *asset* can be a node label, an IP, or a partial node id.
   368→   427→        """

   372→   431→

   374→   433→        vulns: List[Dict[str, Any]] = []
   375→   434→        for nid in visited:
   376→   435→            if HAS_NX:
   377→   436→                data = self._g.nodes[nid]
   378→   437→            else:
   379→   438→                data = self._g.get_node_data(nid)

   384→   443→    def find_chains(self, target: str) -> List[List[Dict[str, Any]]]:
   385→   444→        """Find multi-hop attack chains originating from *target*.
   386→   445→
   387→   446→        Each chain follows a pattern like::
   388→   447→
   389→   448→            target → port → service → vulnerability → CVE
   390→   449→        """
   391→   450→        target_id = self._resolve_node(target)
   392→   451→        if target_id is None:
   393→   452→            return []
   394→   453→
   395→   454→        chains: List[List[Dict[str, Any]]] = []
   396→   455→        # DFS to find paths that end at CVE or finding nodes
   397→   456→        stack: List[Tuple[str, List[str]]] = [(target_id, [target_id])]
   398→   457→        seen_chains: Set[tuple] = set()
   399→   458→
   400→   459→        while stack:
   401→   460→            current, path = stack.pop()
   402→   461→            succs = self._g.successors(current) if HAS_NX else self._g.successors(current)
   403→   462→            for nxt in succs:
   404→   463→                if nxt in path:
   405→   464→                    continue
   406→   465→                new_path = path + [nxt]
   407→   466→                if HAS_NX:
   408→   467→                    nd = self._g.nodes[nxt]
   409→   468→                else:
   410→   469→                    nd = self._g.get_node_data(nxt)
   411→   470→                ntype = nd.get("node_type", "")
   412→   471→                # A chain is complete if it reaches a CVE or high-sev finding
   413→   472→                if ntype == "cve" or (ntype == "finding" and SEVERITY_ORDER.get(self._safe_sev(nd.get("severity")), 0) >= 2):
   414→   473→                    chain_key = tuple(new_path)
   415→   474→                    if chain_key not in seen_chains:
   416→   475→                        seen_chains.add(chain_key)
   417→   476→                        chain_data = []
   418→   477→                        for nid in new_path:
   419→   478→                            if HAS_NX:
   420→   479→                                chain_data.append(dict(self._g.nodes[nid]))
   421→   480→                            else:
   422→   481→                                chain_data.append(self._g.get_node_data(nid))
   423→   482→                        chains.append(chain_data)
   424→   483→                elif len(new_path) < 8:
   425→   484→                    stack.append((nxt, new_path))
   426→   485→
   427→   486→        # Sort by severity of the terminal node
   428→   487→        def _chain_sort_key(chain):
   429→   488→            last = chain[-1]
   430→   489→            return SEVERITY_ORDER.get(self._safe_sev(last.get("severity")), 0)
   431→   490→
   432→   491→        chains.sort(key=_chain_sort_key, reverse=True)
   433→   492→        return chains

   435→   494→    def get_vulnerable_assets(self) -> List[Dict[str, Any]]:
   436→   495→        """Return all assets (targets/subdomains) with HIGH or CRITICAL findings."""
   437→   496→        vulnerable: List[Dict[str, Any]] = []
   438→   497→        seen: Set[str] = set()
   439→   498→
   440→   499→        for nid in self._node_type_index.get("finding", set()):
   441→   500→            if HAS_NX:
   442→   501→                data = self._g.nodes[nid]
   443→   502→            else:
   444→   503→                data = self._g.get_node_data(nid)
   445→   504→            sev = self._safe_sev(data.get("severity"))
   446→   505→            if SEVERITY_ORDER.get(sev, 0) < 3:
   447→   506→                continue
   448→   507→            # Walk predecessors to find the owning target/subdomain
   449→   508→            for pred in (self._g.predecessors(nid) if HAS_NX else self._g.predecessors(nid)):
   450→   509→                if HAS_NX:
   451→   510→                    pdata = self._g.nodes[pred]
   452→   511→                else:
   453→   512→                    pdata = self._g.get_node_data(pred)
   454→   513→                ptype = pdata.get("node_type", "")
   455→   514→                if ptype in ("target", "subdomain", "asset") and pred not in seen:
   456→   515→                    seen.add(pred)
   457→   516→                    vulnerable.append({"node_id": pred, **pdata, "critical_finding": data})
   458→   517→        return vulnerable

   460→   519→    def correlate(self) -> List[Dict[str, Any]]:
   461→   520→        """Cross-reference findings to suggest related vulnerabilities.
   462→   521→
   463→   522→        Returns a list of correlation suggestions, each containing
   464→   523→        the two related node ids, the relationship type, and a
   465→   524→        confidence score.
   466→   525→        """
   467→   526→        suggestions: List[Dict[str, Any]] = []
   468→   527→        # Group findings by technology
   469→   528→        tech_findings: Dict[str, List[str]] = defaultdict(list)
   470→   529→        for nid in self._node_type_index.get("finding", set()):
   471→   530→            if HAS_NX:
   472→   531→                data = self._g.nodes[nid]
   473→   532→            else:
   474→   533→                data = self._g.get_node_data(nid)
   475→   534→            for pred in (self._g.predecessors(nid) if HAS_NX else self._g.predecessors(nid)):
   476→   535→                if HAS_NX:
   477→   536→                    pdata = self._g.nodes[pred]
   478→   537→                else:
   479→   538→                    pdata = self._g.get_node_data(pred)
   480→   539→                if pdata.get("node_type") == "technology":
   481→   540→                    tech_findings[pred].append(nid)
   482→   541→
   483→   542→        for tech_id, finding_ids in tech_findings.items():
   484→   543→            if len(finding_ids) < 2:
   485→   544→                continue
   486→   545→            for i in range(len(finding_ids)):
   487→   546→                for j in range(i + 1, len(finding_ids)):
   488→   547→                    f1 = finding_ids[i]
   489→   548→                    f2 = finding_ids[j]
   490→   549→                    if HAS_NX:
   491→   550→                        d1 = self._g.nodes[f1]
   492→   551→                        d2 = self._g.nodes[f2]
   493→   552→                    else:
   494→   553→                        d1 = self._g.get_node_data(f1)
   495→   554→                        d2 = self._g.get_node_data(f2)
   496→   555→                    # If both share a technology, suggest they may be related
   497→   556→                    conf = 0.6
   498→   557→                    if d1.get("severity") == d2.get("severity"):
   499→   558→                        conf += 0.2
   500→   559→                    if d1.get("finding_type") == d2.get("finding_type"):
   501→   560→                        conf += 0.2
   502→   561→                    suggestions.append({
   503→   562→                        "source": f1,
   504→   563→                        "target": f2,
   505→   564→                        "relationship": "shared_technology",
   506→   565→                        "technology": tech_id,
   507→   566→                        "confidence": min(conf, 1.0),
   508→   567→                        "source_severity": d1.get("severity"),
   509→   568→                        "target_severity": d2.get("severity"),
   510→   569→                    })
   511→   570→
   512→   571→        # Also correlate by port: findings on the same port may be related
   513→   572→        port_findings: Dict[str, List[str]] = defaultdict(list)
   514→   573→        for nid in self._node_type_index.get("finding", set()):
   515→   574→            for pred in (self._g.predecessors(nid) if HAS_NX else self._g.predecessors(nid)):
   516→   575→                if HAS_NX:
   517→   576→                    pdata = self._g.nodes[pred]
   518→   577→                else:
   519→   578→                    pdata = self._g.get_node_data(pred)
   520→   579→                if pdata.get("node_type") == "port":
   521→   580→                    port_findings[pred].append(nid)
   522→   581→
   523→   582→        for port_id, fids in port_findings.items():
   524→   583→            if len(fids) < 2:
   525→   584→                continue
   526→   585→            for i in range(len(fids)):
   527→   586→                for j in range(i + 1, len(fids)):
   528→   587→                    f1, f2 = fids[i], fids[j]
   529→   588→                    suggestions.append({
   530→   589→                        "source": f1,
   531→   590→                        "target": f2,
   532→   591→                        "relationship": "shared_port",
   533→   592→                        "port": port_id,
   534→   593→                        "confidence": 0.7,
   535→   594→                    })
   536→   595→
   537→   596→        suggestions.sort(key=lambda s: s["confidence"], reverse=True)
   538→   597→        return suggestions

   540→   599→    def to_cypher(self) -> str:
   541→   600→        """Export the graph as Cypher-like CREATE statements."""
   542→   601→        lines: List[str] = []
   543→   602→        # Nodes
   544→   603→        for nid in (self._g.nodes() if HAS_NX else self._g.nodes()):
   545→   604→            if HAS_NX:
   546→   605→                data = self._g.nodes[nid]
   547→   606→            else:
   548→   607→                data = self._g.get_node_data(nid)
   549→   608→            ntype = data.pop("node_type", "Unknown")
   550→   609→            label = data.pop("label", nid)
   551→   610→            props = json.dumps(data, default=str)
   552→   611→            safe_id = nid.replace(":", "__").replace("-", "_")
   553→   612→            lines.append(f'CREATE ({safe_id}:{ntype} {{label: "{label}", {props[1:-1]}}})')
   554→   613→            # Restore for edge generation below
   555→   614→            data["node_type"] = ntype
   556→   615→            data["label"] = label
   557→   616→
   558→   617→        # Edges
   559→   618→        for u, v, attr in (self._g.edges(data=True) if HAS_NX else self._g.edges(data=True)):
   560→   619→            etype = attr.get("edge_type", "RELATED_TO")
   561→   620→            props = {k: v for k, v in attr.items() if k != "edge_type"}
   562→   621→            props_str = json.dumps(props, default=str)
   563→   622→            safe_u = u.replace(":", "__").replace("-", "_")
   564→   623→            safe_v = v.replace(":", "__").replace("-", "_")
   565→   624→            if props_str == "{}":
   566→   625→                lines.append(f'CREATE ({safe_u})-[:{etype}]->({safe_v})')
   567→   626→            else:
   568→   627→                lines.append(f'CREATE ({safe_u})-[:{etype} {{{props_str[1:-1]}}}]->({safe_v})')
   569→   628→
   570→   629→        return "\n".join(lines)

   674→   733→    def _resolve_node(self, label: str) -> Optional[str]:
   675→   734→        """Try to resolve a human-readable label to a full node id."""
   676→   735→        # Exact node id
   677→   736→        if self._g.has_node(label):
   678→   737→            return label
   679→   738→        # Try as target
   680→   739→        candidate = self._node_id("target", label)
   681→   740→        if self._g.has_node(candidate):
   682→   741→            return candidate
   683→   742→        # Try as subdomain
   684→   743→        candidate = self._node_id("subdomain", label)
   685→   744→        if self._g.has_node(candidate):
   686→   745→            return candidate
   687→   746→        # Try as asset
   688→   747→        candidate = self._node_id("asset", label)
   689→   748→        if self._g.has_node(candidate):
   690→   749→            return candidate
   691→   750→        # Fuzzy: search all node labels
   692→   751→        for nid in self._g.nodes():
   693→   752→            if HAS_NX:
   694→   753→                data = self._g.nodes[nid]
   695→   754→            else:
   696→   755→                data = self._g.get_node_data(nid)
   697→   756→            if data.get("label") == label or nid.endswith(f":{label}"):
   698→   757→                return nid
   699→   758→        return None

   701→   760→    def _bfs_collect(self, start: str, max_depth: int = 5) -> Set[str]:
   702→   761→        """BFS from *start* up to *max_depth* hops, return visited nodes."""
   703→   762→        visited: Set[str] = set()
   704→   763→        queue: deque = deque()
   705→   764→        queue.append((start, 0))
   706→   765→        visited.add(start)
   707→   766→        while queue:
   708→   767→            node, depth = queue.popleft()
   709→   768→            if depth >= max_depth:
   710→   769→                continue
   711→   770→            for neighbour in (self._g.successors(node) if HAS_NX else self._g.successors(node)):
   712→   771→                if neighbour not in visited:
   713→   772→                    visited.add(neighbour)
   714→   773→                    queue.append((neighbour, depth + 1))
   715→   774→        return visited
