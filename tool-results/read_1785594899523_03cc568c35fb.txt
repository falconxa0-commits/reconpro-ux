    78→function useSoundEngine() {
    79→  const ctxRef = useRef<AudioContext | null>(null);
    80→  const [muted, setMuted] = useState(false);
    81→  const [volume, setVolume] = useState(50);
    82→  const masterGainRef = useRef<GainNode | null>(null);
    83→
    84→  const ensureCtx = useCallback(() => {
    85→    if (!ctxRef.current || ctxRef.current.state === 'closed') {
    86→      const ac = new (window.AudioContext || (window as unknown as { webkitAudioContext: typeof AudioContext }).webkitAudioContext)();
    87→      ctxRef.current = ac;
    88→      const g = ac.createGain();
    89→      g.gain.value = volume / 100;
    90→      g.connect(ac.destination);
    91→      masterGainRef.current = g;
    92→    }
    93→    if (ctxRef.current.state === 'suspended') ctxRef.current.resume();
    94→    return { ctx: ctxRef.current, master: masterGainRef.current! };
    95→  }, [volume]);
    96→
    97→  useEffect(() => {
    98→    if (masterGainRef.current) masterGainRef.current.gain.value = volume / 100;
    99→  }, [volume]);
   100→
   101→  const playSonar = useCallback(() => {
   102→    if (muted) return;
   103→    const { ctx, master } = ensureCtx();
   104→    const osc = ctx.createOscillator();
   105→    const g = ctx.createGain();
   106→    osc.type = 'sine'; osc.frequency.value = 800;
   107→    g.gain.setValueAtTime(0.15, ctx.currentTime);
   108→    g.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + 0.1);
   109→    osc.connect(g); g.connect(master);
   110→    osc.start(); osc.stop(ctx.currentTime + 0.1);
   111→  }, [muted, ensureCtx]);
   112→
   113→  const playExplosion = useCallback(() => {
   114→    if (muted) return;
   115→    const { ctx, master } = ensureCtx();
   116→    const osc = ctx.createOscillator();
   117→    const g = ctx.createGain();
   118→    osc.type = 'sawtooth'; osc.frequency.value = 60;
   119→    g.gain.setValueAtTime(0.3, ctx.currentTime);
   120→    g.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + 0.2);
   121→    osc.connect(g); g.connect(master);
   122→    osc.start(); osc.stop(ctx.currentTime + 0.2);
   123→  }, [muted, ensureCtx]);
   124→
   125→  const playSiren = useCallback(() => {
   126→    if (muted) return;
   127→    const { ctx, master } = ensureCtx();
   128→    const osc = ctx.createOscillator();
   129→    const g = ctx.createGain();
   130→    osc.type = 'sine';
   131→    osc.frequency.setValueAtTime(400, ctx.currentTime);
   132→    osc.frequency.exponentialRampToValueAtTime(1200, ctx.currentTime + 1);
   133→    osc.frequency.exponentialRampToValueAtTime(400, ctx.currentTime + 2);
   134→    g.gain.setValueAtTime(0.2, ctx.currentTime);
   135→    g.gain.setValueAtTime(0.2, ctx.currentTime + 1.7);
   136→    g.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + 2);
   137→    osc.connect(g); g.connect(master);
   138→    osc.start(); osc.stop(ctx.currentTime + 2);
   139→  }, [muted, ensureCtx]);
   140→
   141→  const playAmbient = useCallback(() => {
   142→    if (muted) return;
   143→    const { ctx, master } = ensureCtx();
   144→    const osc = ctx.createOscillator();
   145→    const g = ctx.createGain();
   146→    osc.type = 'sine'; osc.frequency.value = 55;
   147→    g.gain.value = 0.02;
   148→    osc.connect(g); g.connect(master);
   149→    osc.start();
   150→    return { stop: () => { g.gain.exponentialRampToValueAtTime(0.0001, ctx.currentTime + 1); osc.stop(ctx.currentTime + 1); } };
   151→  }, [muted, ensureCtx]);
   152→
   153→  useEffect(() => {
   154→    return () => {
   155→      if (ctxRef.current && ctxRef.current.state !== 'closed') ctxRef.current.close();
   156→    };
   157→  }, []);
   158→
   159→  return { playSonar, playExplosion, playSiren, playAmbient, muted, setMuted, volume, setVolume };
   160→}

   166→function useParticleSystem(canvasRef: React.RefObject<HTMLCanvasElement | null>) {
   167→  const particlesRef = useRef<Particle[]>([]);
   168→  const animRef = useRef<number>(0);
   169→
   170→  const spawnParticles = useCallback((severity: string, cx: number, cy: number) => {
   171→    const count = severity === 'critical' ? 55 : severity === 'high' ? 30 : severity === 'medium' ? 15 : 10;
   172→    const speed = severity === 'critical' ? 8 : severity === 'high' ? 5 : severity === 'medium' ? 3 : 1.5;
   173→    const lifeBase = severity === 'critical' ? 90 : severity === 'high' ? 60 : severity === 'medium' ? 40 : 25;
   174→    const color = SEVERITY_COLORS[severity] || '#3b82f6';
   175→    const secondColor = severity === 'critical' ? '#fbbf24' : severity === 'high' ? '#fde68a' : '#93c5fd';
   176→
   177→    for (let i = 0; i < count; i++) {
   178→      const angle = (Math.PI * 2 * i) / count + (Math.random() - 0.5) * 0.5;
   179→      const v = speed * (0.5 + Math.random() * 0.8);
   180→      particlesRef.current.push({
   181→        x: cx, y: cy, vx: Math.cos(angle) * v, vy: Math.sin(angle) * v,
   182→        life: lifeBase + Math.random() * 20, maxLife: lifeBase + 20,
   183→        color: Math.random() > 0.5 ? color : secondColor,
   184→        size: 1.5 + Math.random() * (severity === 'critical' ? 4 : 2.5),
   185→      });
   186→    }
   187→  }, []);
   188→
   189→  useEffect(() => {
   190→    const canvas = canvasRef.current;
   191→    if (!canvas) return;
   192→    const ctx = canvas.getContext('2d');
   193→    if (!ctx) return;
   194→
   195→    const resize = () => {
   196→      const r = canvas.parentElement?.getBoundingClientRect();
   197→      if (r) { canvas.width = r.width; canvas.height = r.height; }
   198→    };
   199→    resize();
   200→    const ro = new ResizeObserver(resize);
   201→    ro.observe(canvas.parentElement!);
   202→
   203→    const GRAVITY = 0.08;
   204→    const animate = () => {
   205→      ctx!.clearRect(0, 0, canvas!.width, canvas!.height);
   206→      const ps = particlesRef.current;
   207→      for (let i = ps.length - 1; i >= 0; i--) {
   208→        const p = ps[i];
   209→        p.x += p.vx; p.y += p.vy; p.vy += GRAVITY; p.vx *= 0.99; p.life--;
   210→        if (p.life <= 0) { ps.splice(i, 1); continue; }
   211→        const alpha = p.life / p.maxLife;
   212→        ctx!.globalAlpha = alpha;
   213→        ctx!.fillStyle = p.color;
   214→        ctx!.beginPath();
   215→        ctx!.arc(p.x, p.y, p.size * alpha, 0, Math.PI * 2);
   216→        ctx!.fill();
   217→        // Trail
   218→        ctx!.strokeStyle = p.color;
   219→        ctx!.lineWidth = p.size * 0.4 * alpha;
   220→        ctx!.beginPath(); ctx!.moveTo(p.x, p.y);
   221→        ctx!.lineTo(p.x - p.vx * 3, p.y - p.vy * 3);
   222→        ctx!.stroke();
   223→      }
   224→      ctx!.globalAlpha = 1;
   225→      animRef.current = requestAnimationFrame(animate);
   226→    };
   227→    animate();
   228→
   229→    return () => { cancelAnimationFrame(animRef.current); ro.disconnect(); };
   230→  }, [canvasRef]);
   231→
   232→  return { spawnParticles };
   233→}

   236→// Network Topology Renderer Hook
   237→// ═══════════════════════════════════════════════════════════════════════
   238→
   239→function useTopology(topoCanvasRef: React.RefObject<HTMLCanvasElement | null>) {
   240→  const nodesRef = useRef<TopoNode[]>([]);
   241→  const edgesRef = useRef<TopoEdge[]>([]);
   242→  const animRef = useRef<number>(0);
   243→
   244→  const addNode = useCallback((id: string, label: string, hasVuln: boolean) => {
   245→    const canvas = topoCanvasRef.current;
   246→    if (!canvas) return;
   247→    const exists = nodesRef.current.find(n => n.id === id);
   248→    if (exists) {
   249→      if (hasVuln) { exists.hasVuln = true; exists.color = '#ef4444'; }
   250→      exists.glowIntensity = 1;
   251→      return;
   252→    }
   253→    const cx = canvas.width / 2;
   254→    const cy = canvas.height / 2;
   255→    const angle = Math.random() * Math.PI * 2;
   256→    const dist = 60 + Math.random() * 100;
   257→    nodesRef.current.push({
   258→      id, label, x: cx + Math.cos(angle) * dist, y: cy + Math.sin(angle) * dist,
   259→      vx: 0, vy: 0, radius: 8 + Math.random() * 6,
   260→      color: hasVuln ? '#ef4444' : '#22c55e', glowIntensity: 1, hasVuln,
   261→    });
   262→  }, [topoCanvasRef]);
   263→
   264→  const addEdge = useCallback((from: string, to: string) => {
   265→    if (edgesRef.current.find(e => (e.from === from && e.to === to) || (e.from === to && e.to === from))) return;
   266→    edgesRef.current.push({ from, to, progress: 0, active: true });
   267→  }, []);
   268→
   269→  useEffect(() => {
   270→    const canvas = topoCanvasRef.current;
   271→    if (!canvas) return;
   272→    const ctx = canvas.getContext('2d');
   273→    if (!ctx) return;
   274→
   275→    const resize = () => {
   276→      const r = canvas.parentElement?.getBoundingClientRect();
   277→      if (r) { canvas.width = r.width; canvas.height = r.height; }
   278→    };
   279→    resize();
   280→    const ro = new ResizeObserver(resize);
   281→    ro.observe(canvas.parentElement!);
   282→
   283→    const animate = () => {
   284→      ctx!.clearRect(0, 0, canvas!.width, canvas!.height);
   285→      const nodes = nodesRef.current;
   286→      const edges = edgesRef.current;
   287→
   288→      // Simple force simulation
   289→      for (let i = 0; i < nodes.length; i++) {
   290→        for (let j = i + 1; j < nodes.length; j++) {
   291→          const a = nodes[i], b = nodes[j];
   292→          let dx = b.x - a.x, dy = b.y - a.y;
   293→          const dist = Math.sqrt(dx * dx + dy * dy) || 1;
   294→          const minDist = 80;
   295→          if (dist < minDist) {
   296→            const force = (minDist - dist) * 0.02;
   297→            const fx = (dx / dist) * force, fy = (dy / dist) * force;
   298→            a.vx -= fx; a.vy -= fy; b.vx += fx; b.vy += fy;
   299→          }
   300→        }
   301→        // Center gravity
   302→        const cx = canvas!.width / 2, cy = canvas!.height / 2;
   303→        nodes[i].vx += (cx - nodes[i].x) * 0.001;
   304→        nodes[i].vy += (cy - nodes[i].y) * 0.001;
   305→      }
   306→
   307→      // Update positions
   308→      for (const n of nodes) {
   309→        n.x += n.vx; n.y += n.vy; n.vx *= 0.9; n.vy *= 0.9;
   310→        n.x = Math.max(n.radius + 5, Math.min(canvas!.width - n.radius - 5, n.x));
   311→        n.y = Math.max(n.radius + 5, Math.min(canvas!.height - n.radius - 5, n.y));
   312→        n.glowIntensity = Math.max(0, n.glowIntensity - 0.005);
   313→      }
   314→
   315→      // Draw edges
   316→      for (const e of edges) {
   317→        const a = nodes.find(n => n.id === e.from);
   318→        const b = nodes.find(n => n.id === e.to);
   319→        if (!a || !b) continue;
   320→        e.progress = Math.min(1, e.progress + 0.03);
   321→        ctx!.strokeStyle = e.active ? 'rgba(0,255,136,0.3)' : 'rgba(100,100,100,0.15)';
   322→        ctx!.lineWidth = 1;
   323→        ctx!.beginPath(); ctx!.moveTo(a.x, a.y);
   324→        const ex = a.x + (b.x - a.x) * e.progress;
   325→        const ey = a.y + (b.y - a.y) * e.progress;
   326→        ctx!.lineTo(ex, ey); ctx!.stroke();
   327→      }
   328→
   329→      // Draw nodes
   330→      for (const n of nodes) {
   331→        // Glow
   332→        if (n.glowIntensity > 0.01) {
   333→          const grad = ctx!.createRadialGradient(n.x, n.y, n.radius, n.x, n.y, n.radius * 3);
   334→          grad.addColorStop(0, n.color + '80');
   335→          grad.addColorStop(1, 'transparent');
   336→          ctx!.fillStyle = grad;
   337→          ctx!.globalAlpha = n.glowIntensity * 0.6;
   338→          ctx!.beginPath(); ctx!.arc(n.x, n.y, n.radius * 3, 0, Math.PI * 2); ctx!.fill();
   339→          ctx!.globalAlpha = 1;
   340→        }
   341→        // Circle
   342→        ctx!.fillStyle = n.color;
   343→        ctx!.globalAlpha = 0.15;
   344→        ctx!.beginPath(); ctx!.arc(n.x, n.y, n.radius + 2, 0, Math.PI * 2); ctx!.fill();
   345→        ctx!.globalAlpha = 1;
   346→        ctx!.fillStyle = '#0a0a0a';
   347→        ctx!.beginPath(); ctx!.arc(n.x, n.y, n.radius, 0, Math.PI * 2); ctx!.fill();
   348→        ctx!.strokeStyle = n.color;
   349→        ctx!.lineWidth = 1.5;
   350→        ctx!.stroke();
   351→        // Label
   352→        ctx!.fillStyle = '#9ca3af';
   353→        ctx!.font = '9px monospace';
   354→        ctx!.textAlign = 'center';
   355→        ctx!.fillText(n.label.length > 14 ? n.label.slice(0, 12) + '..' : n.label, n.x, n.y + n.radius + 14);
   356→      }
   357→
   358→      animRef.current = requestAnimationFrame(animate);
   359→    };
   360→    animate();
   361→
   362→    return () => { cancelAnimationFrame(animRef.current); ro.disconnect(); };
   363→  }, [topoCanvasRef]);
   364→
   365→  return { addNode, addEdge };
   366→}

   372→function EventIcon({ type, severity }: { type: StreamEvent['type']; severity?: string }) {
   373→  const cls = 'w-3.5 h-3.5 shrink-0';
   374→  switch (type) {
   375→    case 'SCAN_START': return <Radio className={cn(cls, 'text-green-400')} />;
   376→    case 'SUBDOMAIN_FOUND': return <Globe className={cn(cls, 'text-cyan-400')} />;
   377→    case 'PORT_OPEN': return <Wifi className={cn(cls, 'text-purple-400')} />;
   378→    case 'VULN_FOUND': return severity === 'critical' || severity === 'high'
   379→      ? <AlertTriangle className={cn(cls, SEVERITY_COLORS[severity || 'medium'])} />
   380→      : <Bug className={cn(cls, SEVERITY_COLORS[severity || 'medium'])} />;
   381→    case 'SCAN_COMPLETE': return <Shield className={cn(cls, 'text-green-400')} />;
   382→    default: return <Terminal className={cn(cls, 'text-gray-500')} />;
   383→  }
