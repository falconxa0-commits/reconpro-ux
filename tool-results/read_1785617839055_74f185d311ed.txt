   507→function handlePrompt(body: { sessionId: string; prompt: string }) {
   508→  const { sessionId, prompt } = body;
   509→  const session = sessions.get(sessionId);
   510→
   511→  if (!session) {
   512→    return NextResponse.json({ error: 'Session not found or expired' }, { status: 404 });
   513→  }
   514→
   515→  if (Date.now() > session.expiresAt) {
   516→    sessions.delete(sessionId);
   517→    return NextResponse.json({ error: 'Session expired' }, { status: 410 });
   518→  }
   519→
   520→  // Add user message
   521→  session.messages.push({
   522→    role: 'user',
   523→    content: prompt,
   524→    timestamp: Date.now(),
   525→  });
   526→
   527→  // Track turn history for multi-turn detection
   528→  session.turnHistory.push(prompt.toLowerCase());
   529→
   530→  // Match against rules
   531→  const { rule } = matchRules(prompt, session);
   532→
   533→  // Build response
   534→  let response: string;
   535→  let action: string;
   536→  let points: number;
   537→  let annotation: string;
   538→
   539→  if (session.defenseMode && rule.defenseOverride && rule.requiresDefenseCheck?.some((check) => session.defenseRules[check])) {
   540→    // Defense mode blocked
   541→    response = rule.defenseOverride.response;
   542→    action = rule.defenseOverride.action;
   543→    points = rule.defenseOverride.points;
   544→    annotation = rule.defenseOverride.annotation;
   545→  } else {
   546→    response = rule.response;
   547→    action = rule.action;
   548→    points = rule.points;
   549→    annotation = rule.annotation;
   550→  }
   551→
   552→  session.points += points;
   553→
   554→  // Track accessed resources
   555→  if (action !== 'BLOCKED' && action !== 'SECURITY_EVENT_LOGGED' && action !== 'BLOCKED_BY_GUARDRAIL') {
   556→    session.accessedResources.add(action);
   557→  }
   558→
   559→  // Add agent message
   560→  session.messages.push({
   561→    role: 'agent',
   562→    content: response,
   563→    action,
   564→    points,
   565→    annotation,
   566→    timestamp: Date.now(),
   567→  });
   568→
   569→  return NextResponse.json({
   570→    response,
   571→    action,
   572→    success: action !== 'BLOCKED' && action !== 'SECURITY_EVENT_LOGGED' && !action.includes('GUARDRAIL'),
   573→    points,
   574→    annotation,
   575→    totalPoints: session.points,
   576→  });
   577→}

   581→function updateDefense(body: { sessionId: string; defenseMode: boolean; defenseRules?: Partial<DefenseRules> }) {
   582→  const { sessionId, defenseMode, defenseRules } = body;
   583→  const session = sessions.get(sessionId);
   584→
   585→  if (!session) {
   586→    return NextResponse.json({ error: 'Session not found' }, { status: 404 });
   587→  }
   588→
   589→  session.defenseMode = defenseMode;
   590→  if (defenseRules) {
   591→    session.defenseRules = { ...session.defenseRules, ...defenseRules };
   592→  }
   593→
   594→  return NextResponse.json({
   595→    defenseMode: session.defenseMode,
   596→    defenseRules: session.defenseRules,
   597→  });
   598→}

   602→function getSession(body: { sessionId: string }) {
   603→  const { sessionId } = body;
   604→  const session = sessions.get(sessionId);
   605→
   606→  if (!session) {
   607→    return NextResponse.json({ error: 'Session not found' }, { status: 404 });
   608→  }
   609→
   610→  const expired = Date.now() > session.expiresAt;
   611→
   612→  return NextResponse.json({
   613→    sessionId: session.id,
   614→    points: session.points,
   615→    messageCount: session.messages.length,
   616→    messages: session.messages.map((m) => ({
   617→      role: m.role,
   618→      content: m.content,
   619→      action: m.action,
   620→      points: m.points,
   621→      annotation: m.annotation,
   622→      timestamp: m.timestamp,
   623→    })),
   624→    defenseMode: session.defenseMode,
   625→    defenseRules: session.defenseRules,
   626→    accessedResources: Array.from(session.accessedResources),
   627→    expired,
   628→    timeRemaining: Math.max(0, session.expiresAt - Date.now()),
   629→  });
   630→}
