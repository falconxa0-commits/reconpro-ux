   764→# TECH_DETECTOR — Fingerprinting engine
   765→# ──────────────────────────────────────────────────────────────────────────────
   766→
   767→class TechDetector:
   768→    """Analyzes HTTP responses, headers, HTML, and URLs to identify
   769→    the technology stack of a target application."""
   770→
   771→    def __init__(self) -> None:
   772→        self._reset()
   773→
   774→    def _reset(self) -> None:
   775→        self._frameworks: set[str] = set()
   776→        self._language: Optional[str] = None
   777→        self._server: Optional[str] = None
   778→        self._waf: Optional[str] = None
   779→        self._cms: Optional[str] = None
   780→        self._has_graphql: bool = False
   781→        self._has_rest_api: bool = False
   782→        self._confidence: float = 0.0
   783→        self._hits: int = 0
   784→
   785→    def analyze(
   786→        self,
   787→        target: str,
   788→        response_headers: dict[str, str],
   789→        html_body: str = "",
   790→        js_files: Optional[list[str]] = None,
   791→    ) -> TechProfile:
   792→        """Run full analysis and return a TechProfile."""
   793→        self._reset()
   794→        if js_files is None:
   795→            js_files = []
   796→
   797→        self._analyze_headers(response_headers)
   798→        self._analyze_html(html_body)
   799→        self._analyze_cookies(response_headers)
   800→        self._analyze_url(target)
   801→        self._analyze_js_files(js_files)
   802→        self._analyze_url_paths(target)
   803→
   804→        if self._hits > 0:
   805→            self._confidence = min(self._hits * 0.12, 1.0)
   806→
   807→        return TechProfile(
   808→            frameworks=list(self._frameworks),
   809→            language=self._language,
   810→            server=self._server,
   811→            waf=self._waf,
   812→            cms=self._cms,
   813→            has_graphql=self._has_graphql,
   814→            has_rest_api=self._has_rest_api,
   815→            confidence=self._confidence,
   816→        )
   817→
   818→    def _analyze_headers(self, headers: dict[str, str]) -> None:
   819→        header_signatures = TECH_SIGNATURES["headers"]
   820→        header_map = TECH_SIGNATURES["waf_signatures"]
   821→
   822→        for header_key, mapping in header_signatures.items():
   823→            header_val = headers.get(header_key, "")
   824→            if not isinstance(mapping, dict):
   825→                continue
   826→            for pattern, tech in mapping.items():
   827→                if isinstance(tech, str) and tech and pattern:
   828→                    if pattern.lower() in header_val.lower():
   829→                        if tech == "Drupal" or tech == "WordPress" or tech == "Joomla":
   830→                            self._cms = tech
   831→                            self._hits += 2
   832→                        elif tech in ("nginx", "Apache", "IIS", "Caddy", "LiteSpeed", "OpenResty"):
   833→                            self._server = tech
   834→                            self._hits += 1
   835→                        elif tech == "PHP":
   836→                            self._language = "PHP"
   837→                            self._hits += 1
   838→                        elif tech == "Python":
   839→                            self._language = "Python"
   840→                            self._hits += 1
   841→                        elif tech == "Java":
   842→                            self._language = "Java"
   843→                            self._hits += 1
   844→                        elif tech == "Ruby":
   845→                            self._language = "Ruby"
   846→                            self._hits += 1
   847→                        elif tech == "ASP.NET" or tech == "ASP.NET MVC" or tech == "ASP.NET Core":
   848→                            self._language = "ASP.NET"
   849→                            self._frameworks.add("ASP.NET")
   850→                            self._hits += 1
   851→                        elif tech in ("Express", "Django", "Flask", "Tornado", "Next.js", "React", "Vue", "Nuxt"):
   852→                            self._frameworks.add(tech)
   853→                            self._hits += 1
   854→                        elif tech == "Cloudflare":
   855→                            self._waf = "Cloudflare WAF"
   856→                            self._hits += 1
   857→                        else:
   858→                            self._frameworks.add(tech)
   859→                            self._hits += 1
   860→
   861→                if pattern == "" and mapping.get(header_key, "") == "" and header_val:
   862→                    tech_name = list(mapping.values())[0] if mapping else ""
   863→                    if tech_name and tech_name == "":
   864→                        tech_name = header_key.split("-")[-1].strip()
   865→                        self._hits += 1
   866→
   867→        for sig_header, waf_name in header_map.items():
   868→            if sig_header in headers:
   869→                self._waf = waf_name
   870→                self._hits += 1
   871→                break
   872→
   873→    def _analyze_html(self, html: str) -> None:
   874→        if not html:
   875→            return
   876→        html_lower = html.lower()
   877→
   878→        meta_signatures = TECH_SIGNATURES["html_meta"]
   879→        for meta_type, mapping in meta_signatures.items():
   880→            if meta_type == "generator":
   881→                match = re.search(
   882→                    r'<meta\s+[^>]*?name=["\']generator["\'][^>]*?content=["\']([^"\']+)["\']',
   883→                    html_lower,
   884→                    re.IGNORECASE,
   885→                )
   886→                if match:
   887→                    gen_value = match.group(1).strip().lower()
   888→                    for pattern, tech in mapping.items():
   889→                        if pattern.lower() in gen_value:
   890→                            if tech in ("WordPress", "Joomla", "Drupal", "Ghost", "vBulletin", "phpBB", "MediaWiki"):
   891→                                self._cms = tech
   892→                            else:
   893→                                self._frameworks.add(tech)
   894→                            self._hits += 2
   895→                            break
   896→
   897→            if meta_type == "viewport":
   898→                if 'name="viewport"' in html_lower or 'content="width=device-width' in html_lower:
   899→                    self._hits += 0.5
   900→
   901→        ext_signatures = TECH_SIGNATURES["file_extensions"]
   902→        for ext, lang in ext_signatures.items():
   903→            safe_ext = re.escape(ext)
   904→            pattern = rf'(?:src|href)=["\'][^"\']*{safe_ext}(?:\?[^"\']*)?["\']'
   905→            if re.search(pattern, html_lower):
   906→                if self._language is None:
   907→                    self._language = lang
   908→                self._hits += 1
   909→
   910→    def _analyze_cookies(self, headers: dict[str, str]) -> None:
   911→        cookie_header = headers.get("Set-Cookie", "") or headers.get("Cookie", "")
   912→        cookie_signatures = TECH_SIGNATURES["cookies"]
   913→
   914→        for cookie_name, lang in cookie_signatures.items():
   915→            if cookie_name.lower() in cookie_header.lower():
   916→                if lang == "PHP" and self._language is None:
   917→                    self._language = "PHP"
   918→                elif lang == "Java" and self._language is None:
   919→                    self._language = "Java"
   920→                elif lang == "ASP.NET" and self._language is None:
   921→                    self._language = "ASP.NET"
   922→                elif lang == "Rails" and self._language is None:
   923→                    self._language = "Ruby"
   924→                    self._frameworks.add("Rails")
   925→                elif lang == "Laravel":
   926→                    self._frameworks.add("Laravel")
   927→                elif lang == "CodeIgniter":
   928→                    self._frameworks.add("CodeIgniter")
   929→                elif lang == "Express":
   930→                    self._frameworks.add("Express")
   931→                elif lang == "Django":
   932→                    self._frameworks.add("Django")
   933→                elif lang == "Angular":
   934→                    self._frameworks.add("Angular")
   935→                elif lang == "Next.js":
   936→                    self._frameworks.add("Next.js")
   937→                self._hits += 1
   938→
   939→    def _analyze_url(self, target: str) -> None:
   940→        url_patterns = TECH_SIGNATURES["url_patterns"]
   941→        target_lower = target.lower()
   942→
   943→        for pattern, tech in url_patterns.items():
   944→            if pattern.lower() in target_lower:
   945→                if tech == "GraphQL":
   946→                    self._has_graphql = True
   947→                    self._frameworks.add("GraphQL")
   948→                elif tech == "REST API":
   949→                    self._has_rest_api = True
   950→                elif tech in ("WordPress", "Joomla", "Drupal"):
   951→                    self._cms = tech
   952→                elif tech == "Swagger/OpenAPI":
   953→                    self._has_rest_api = True
   954→                    self._frameworks.add("Swagger")
   955→                elif tech == "Spring Boot":
   956→                    self._frameworks.add("Spring Boot")
   957→                    self._language = "Java"
   958→                elif tech == "Django":
   959→                    self._frameworks.add("Django")
   960→                    self._language = "Python"
   961→                elif tech == "Flask":
   962→                    self._frameworks.add("Flask")
   963→                    self._language = "Python"
   964→                elif tech == "Dotenv":
   965→                    self._hits += 1
   966→                else:
   967→                    self._frameworks.add(tech)
   968→                self._hits += 1
   969→
   970→    def _analyze_js_files(self, js_files: list[str]) -> None:
   971→        js_signatures = TECH_SIGNATURES["js_libraries"]
   972→
   973→        for js_path in js_files:
   974→            js_lower = js_path.lower()
   975→            for pattern, tech in js_signatures.items():
   976→                if pattern.lower() in js_lower:
   977→                    self._frameworks.add(tech)
   978→                    self._hits += 1
   979→                    break

   984→        path = parsed.path.lower()
   985→
   986→        for ext, lang in ext_signatures.items():
   987→            if path.endswith(ext):
   988→                if self._language is None:
   989→                    self._language = lang
   990→                self._hits += 1

  1089→class FuzzSession:
  1090→    """Context-aware fuzzing session that selects payloads based on the
  1091→    target's detected technology stack."""
  1092→
  1093→    def __init__(
  1094→        self,
  1095→        target: str,
  1096→        tech_profile: Optional[TechProfile] = None,
  1097→        timeout: float = 10.0,
  1098→        user_agent: str = "ReconPro/7.5 (Security Scanner)",
  1099→        verify_ssl: bool = False,
  1100→    ) -> None:
  1101→        self.target = target.rstrip("/")
  1102→        self.tech_profile = tech_profile
  1103→        self.timeout = timeout
  1104→        self.user_agent = user_agent
  1105→        self.verify_ssl = verify_ssl
  1106→        self.results: list[FuzzResult] = []
  1107→        self._base_time: float = 0.0
  1108→        self._baseline_status: int = 0
  1109→        self._opener = self._build_opener()
  1110→
  1111→    def _build_opener(self) -> urllib.request.OpenerDirector:
  1112→        """Build a URL opener with optional SSL bypass."""
  1113→        if not self.verify_ssl:
  1114→            import ssl
  1115→            ctx = ssl._create_unverified_context()
  1116→            https_handler = urllib.request.HTTPSHandler(context=ctx)
  1117→            return urllib.request.build_opener(https_handler)
  1118→        return urllib.request.build_opener()
  1119→
  1120→    def _make_request(
  1121→        self,
  1122→        url: str,
  1123→        params: Optional[dict[str, str]] = None,
  1124→        headers: Optional[dict[str, str]] = None,
  1125→        method: str = "GET",
  1126→    ) -> tuple[int, str, float, dict[str, str]]:
  1127→        """Make an HTTP request and return (status_code, body, time, headers)."""
  1128→        if params and method == "GET":
  1129→            query = urllib.parse.urlencode(params)
  1130→            url = f"{url}?{query}"
  1131→
  1132→        req_headers = {
  1133→            "User-Agent": self.user_agent,
  1134→            "Accept": "text/html,application/xhtml+xml,application/xml;q=0.9,*/*;q=0.8",
  1135→            "Accept-Language": "en-US,en;q=0.5",
  1136→        }
  1137→        if headers:
  1138→            req_headers.update(headers)
  1139→
  1140→        body_data: Optional[bytes] = None
  1141→        if params and method == "POST":
  1142→            body_data = urllib.parse.urlencode(params).encode("utf-8")
  1143→            req_headers["Content-Type"] = "application/x-www-form-urlencoded"
  1144→
  1145→        req = urllib.request.Request(url, data=body_data, headers=req_headers, method=method)
  1146→
  1147→        try:
  1148→            start = time.monotonic()
  1149→            resp = self._opener.open(req, timeout=self.timeout)
  1150→            elapsed = time.monotonic() - start
  1151→            resp_body = resp.read().decode("utf-8", errors="replace")
  1152→            resp_headers = dict(resp.headers)
  1153→            return resp.status, resp_body, elapsed, resp_headers
  1154→        except urllib.error.HTTPError as e:
  1155→            elapsed = time.monotonic() - time.monotonic() + 0.001
  1156→            try:
  1157→                resp_body = e.read().decode("utf-8", errors="replace")
  1158→            except Exception:
  1159→                resp_body = ""
  1160→            resp_headers = dict(e.headers) if hasattr(e, "headers") else {}
  1161→            return e.code, resp_body, elapsed, resp_headers
  1162→        except urllib.error.URLError:
  1163→            elapsed = 0.0
  1164→            return 0, "", elapsed, {}
  1165→        except Exception:
  1166→            elapsed = 0.0
  1167→            return 0, "", elapsed, {}
  1168→
  1169→    def calibrate(self) -> None:
  1170→        """Send a baseline request to establish normal response time and status."""
  1171→        status, _, elapsed, _ = self._make_request(self.target)
  1172→        self._baseline_status = status
  1173→        self._base_time = elapsed

  1175→    def get_all_categories(self) -> list[str]:
  1176→        """Return a list of all available payload categories."""
  1177→        return list(PAYLOAD_DATABASE.keys())
  1178→
  1179→    def select_payloads(self, category: str) -> list[str]:
  1180→        """Select the most relevant payloads for the detected tech stack.
  1181→
  1182→        Priority logic:
  1183→          1. If tech_profile is available, prefer framework-specific payloads.
  1184→          2. Language-specific SQL payloads (e.g., Django -> postgres, WordPress -> mysql).
  1185→          3. Framework-specific template payloads (e.g., Django -> jinja2/django SSTI).
  1186→          4. CMS-specific extras for WordPress/Drupal/Joomla.
  1187→          5. Generic payloads always included as fallback.
  1188→        """
  1189→        if category not in PAYLOAD_DATABASE:
  1190→            return []
  1191→
  1192→        cat_payloads = PAYLOAD_DATABASE[category]
  1193→        selected: list[str] = []
  1194→        seen: set[str] = set()
  1195→
  1196→        def _add(payloads: list[str]) -> None:
  1197→            for p in payloads:
  1198→                if p not in seen:
  1199→                    selected.append(p)
  1200→                    seen.add(p)
  1201→
  1202→        # -- If we have a tech profile, prioritize matching payloads --------
  1203→        if self.tech_profile:
  1204→            tp = self.tech_profile
  1205→            fw_set = set(tp.frameworks)
  1206→            lang = tp.language
  1207→
  1208→            # --- SQL injection: route by detected language/backend ----------
  1209→            if category == "sqli":
  1210→                if any(fw in fw_set for fw in ("Django", "Flask", "Python", "FastAPI")):
  1211→                    if "postgres" in cat_payloads:
  1212→                        _add(cat_payloads["postgres"])
  1213→                if lang == "PHP" or tp.cms in ("WordPress", "Drupal", "Joomla"):
  1214→                    if "mysql" in cat_payloads:
  1215→                        _add(cat_payloads["mysql"])
  1216→                if lang == "Java" or any(fw in fw_set for fw in ("Spring Boot", "Tomcat", "Jetty")):
  1217→                    if "oracle" in cat_payloads:
  1218→                        _add(cat_payloads["oracle"])
  1219→                    if "mssql" in cat_payloads:
  1220→                        _add(cat_payloads["mssql"])
  1221→                if lang == "ASP.NET" or "ASP.NET" in fw_set:
  1222→                    if "mssql" in cat_payloads:
  1223→                        _add(cat_payloads["mssql"])
  1224→                if lang == "Ruby" or "Rails" in fw_set:
  1225→                    if "sqlite" in cat_payloads:
  1226→                        _add(cat_payloads["sqlite"])
  1227→                    if "postgres" in cat_payloads:
  1228→                        _add(cat_payloads["postgres"])
  1229→
  1230→            # --- XSS: route by detected frontend framework -----------------
  1231→            elif category == "xss":
  1232→                if "Angular" in fw_set:
  1233→                    _add(cat_payloads.get("angular", []))
  1234→                if "React" in fw_set:
  1235→                    _add(cat_payloads.get("react", []))
  1236→                if "Vue" in fw_set or "Nuxt" in fw_set:
  1237→                    _add(cat_payloads.get("vue", []))
  1238→                _add(cat_payloads.get("html", []))
  1239→
  1240→            # --- SSTI: route by detected backend template engine ------------
  1241→            elif category == "ssti":
  1242→                if any(fw in fw_set for fw in ("Django", "Python", "Flask", "FastAPI")):
  1243→                    _add(cat_payloads.get("jinja2", []))
  1244→                    _add(cat_payloads.get("django", []))
  1245→                if "Ruby" in fw_set or "Rails" in fw_set:
  1246→                    _add(cat_payloads.get("mako", []))
  1247→                if "PHP" in lang or tp.cms == "Drupal":
  1248→                    _add(cat_payloads.get("twig", []))
  1249→
  1250→            # --- SSRF: prefer cloud provider based on headers --------------
  1251→            elif category == "ssrf":
  1252→                if tp.server and "amazon" in tp.server.lower():
  1253→                    _add(cat_payloads.get("aws_metadata", []))
  1254→                elif tp.server and "google" in tp.server.lower():
  1255→                    _add(cat_payloads.get("gcp_metadata", []))
  1256→                elif "azure" in (tp.server or "").lower():
  1257→                    _add(cat_payloads.get("azure_metadata", []))
  1258→                else:
  1259→                    for sub in ("aws_metadata", "gcp_metadata", "azure_metadata"):
  1260→                        _add(cat_payloads.get(sub, []))
  1261→
  1262→            # --- Path traversal: OS-specific based on server ---------------
  1263→            elif category == "path_traversal":
  1264→                if tp.server and "iis" in tp.server.lower():
  1265→                    _add(cat_payloads.get("windows", []))
  1266→                else:
  1267→                    _add(cat_payloads.get("unix", []))
  1268→
  1269→            # --- Command injection: OS-specific based on server -------------
  1270→            elif category == "command_injection":
  1271→                if tp.server and "iis" in tp.server.lower():
  1272→                    _add(cat_payloads.get("windows", []))
  1273→                else:
  1274→                    _add(cat_payloads.get("unix", []))
  1275→
  1276→        # -- Always add generic payloads as fallback -------------------------
  1277→        if "generic" in cat_payloads:
  1278→            _add(cat_payloads["generic"])
  1279→
  1280→        # -- Add any remaining sub-categories not yet covered ---------------
  1281→        for sub_key, sub_payloads in cat_payloads.items():
  1282→            if sub_key != "generic" and sub_key not in ("mysql", "postgres", "mssql", "sqlite", "oracle",
  1283→                                                        "html", "angular", "react", "vue",
  1284→                                                        "jinja2", "django", "mako", "twig",
  1285→                                                        "aws_metadata", "gcp_metadata", "azure_metadata",
  1286→                                                        "unix", "windows"):
  1287→                if not self.tech_profile:
  1288→                    _add(sub_payloads)
  1289→
  1290→        return selected
  1291→
  1292→    def fuzz_parameter(
  1293→        self,
  1294→        url: str,
  1295→        param: str,
  1296→        payloads: Optional[list[str]] = None,
  1297→        category: str = "generic",
  1298→        method: str = "GET",
  1299→    ) -> list[FuzzResult]:
  1300→        """Fuzz a single URL parameter with selected payloads.
  1301→
  1302→        For each payload:
  1303→          1. Substitute the payload into the parameter value.
  1304→          2. Send the request.
  1305→          3. Analyze the response for anomalies.
  1306→          4. Return a list of FuzzResult objects.
  1307→        """
  1308→        if payloads is None:
  1309→            payloads = self.select_payloads(category)
  1310→
  1311→        if not payloads:
  1312→            return []
  1313→
  1314→        # Fetch the original (clean) response for comparison
  1315→        original_status, original_body, _, _ = self._make_request(url)
  1316→
  1317→        results: list[FuzzResult] = []
  1318→
  1319→        for payload in payloads:
  1320→            params = {param: payload}
  1321→
  1322→            status, body, elapsed, resp_headers = self._make_request(url, params=params, method=method)
  1323→
  1324→            result = self._analyze_response(
  1325→                original_body=original_body,
  1326→                fuzzed_body=body,
  1327→                status_code=status,
  1328→                response_time=elapsed,
  1329→            )
  1330→            result.payload = payload
  1331→            result.category = category
  1332→            result.url = url
  1333→            result.param = param
  1334→            result.status_code = status
  1335→
  1336→            results.append(result)
  1337→
  1338→        self.results.extend(results)
  1339→        return results
  1340→

  1342→        self,
  1343→        url: str,
  1344→        payloads: Optional[list[str]] = None,
  1345→        category: str = "sqli",
  1346→        header_name: str = "X-Custom-Header",
  1347→    ) -> list[FuzzResult]:
  1348→        """Fuzz HTTP headers with payloads."""
  1349→        if payloads is None:
  1350→            payloads = self.select_payloads(category)
  1351→
  1352→        if not payloads:
  1353→            return []
  1354→
  1355→        original_status, original_body, _, _ = self._make_request(url)
  1356→        results: list[FuzzResult] = []
  1357→
  1358→        for payload in payloads:
  1359→            headers = {header_name: payload}
  1360→            status, body, elapsed, _ = self._make_request(url, headers=headers)
  1361→
  1362→            result = self._analyze_response(
  1363→                original_body=original_body,
  1364→                fuzzed_body=body,
  1365→                status_code=status,
  1366→                response_time=elapsed,
  1367→            )
  1368→            result.payload = payload
  1369→            result.category = category
  1370→            result.url = url
  1371→            result.param = f"Header:{header_name}"
  1372→            result.status_code = status
  1373→
  1374→            results.append(result)
  1375→
  1376→        self.results.extend(results)
  1377→        return results

  1379→    def fuzz_all_parameters(
  1380→        self,
  1381→        url: str,
  1382→        params: list[str],
  1383→        categories: Optional[list[str]] = None,
  1384→    ) -> list[FuzzResult]:
  1385→        """Fuzz multiple parameters across multiple categories."""
  1386→        if categories is None:
  1387→            categories = ["sqli", "xss", "ssti"]
  1388→
  1389→        all_results: list[FuzzResult] = []
  1390→        for param in params:
  1391→            for category in categories:
  1392→                payloads = self.select_payloads(category)
  1393→                results = self.fuzz_parameter(url, param, payloads, category)
  1394→                all_results.extend(results)
  1395→        return all_results

  1397→    def analyze_response(
  1398→        self,
  1399→        original_body: str,
  1400→        fuzzed_body: str,
  1401→        status_code: int,
  1402→        response_time: float,
  1403→    ) -> FuzzResult:
  1404→        """Analyze a fuzzed response and classify the result.
  1405→
  1406→        Detection methods:
  1407→          - Reflected: the payload itself appears in the response body.
  1408→          - Error-based: SQL errors, stack traces, or framework errors in the body.
  1409→          - Time-based: response took significantly longer than baseline.
  1410→          - Blind: status code change (e.g., 500 instead of 200) without obvious error.
  1411→        """
  1412→        result = FuzzResult()
  1413→        result.status_code = status_code
  1414→        result.response_time = response_time
  1415→        result.matched_pattern = ""
  1416→
  1417→        # --- Check for reflected payload -----------------------------------
  1418→        if fuzzed_body and original_body:
  1419→            # Find the payload in the response (may be HTML-encoded)
  1420→            payload_escaped = re.escape(result.payload)
  1421→            if result.payload and re.search(payload_escaped, fuzzed_body):
  1422→                result.classification = "reflected"
  1423→                result.matched_pattern = "Payload reflected in response body"
  1424→
  1425→            # Also check for partial reflection or HTML entity encoding
  1426→            if result.payload and not result.classification:
  1427→                html_encoded = (
  1428→                    result.payload.replace("<", "&lt;")
  1429→                    .replace(">", "&gt;")
  1430→                    .replace('"', "&quot;")
  1431→                    .replace("'", "&#39;")
  1432→                )
  1433→                if html_encoded in fuzzed_body:
  1434→                    result.classification = "reflected"
  1435→                    result.matched_pattern = "Payload HTML-encoded in response"
  1436→
  1437→        # --- Check for SQL errors in the response --------------------------
  1438→        if fuzzed_body:
  1439→            for pattern in _SQL_ERROR_PATTERNS:
  1440→                match = pattern.search(fuzzed_body)
  1441→                if match:
  1442→                    result.classification = "error_based"
  1443→                    result.matched_pattern = match.group(0)
  1444→                    break
  1445→
  1446→        # --- Check for stack traces ----------------------------------------
  1447→        if fuzzed_body and result.classification != "error_based":
  1448→            for pattern in _STACK_TRACE_PATTERNS:
  1449→                match = pattern.search(fuzzed_body)
  1450→                if match:
  1451→                    result.classification = "error_based"
  1452→                    result.matched_pattern = match.group(0)
  1453→                    break
  1454→
  1455→        # --- Check for command injection output ---------------------------
  1456→        if fuzzed_body and result.classification != "error_based":
  1457→            match = _CMD_INJECTION_PATTERN.search(fuzzed_body)
  1458→            if match:
  1459→                result.classification = "error_based"
  1460→                result.matched_pattern = match.group(0)
  1461→
  1462→        # --- Check for path traversal output --------------------------------
  1463→        if fuzzed_body and result.classification != "error_based":
  1464→            match = _PATH_TRAVERSAL_PATTERN.search(fuzzed_body)
  1465→            if match:
  1466→                result.classification = "error_based"
  1467→                result.matched_pattern = match.group(0)
  1468→
  1469→        # --- Check for XSS indicators -------------------------------------
  1470→        if fuzzed_body and result.classification != "reflected":
  1471→            match = _XSS_PATTERN.search(fuzzed_body)
  1472→            if match:
  1473→                result.classification = "reflected"
  1474→                result.matched_pattern = match.group(0)
  1475→
  1476→        # --- Check for SSTI indicators -------------------------------------
  1477→        if fuzzed_body and result.classification not in ("reflected", "error_based"):
  1478→            match = _SSTI_PATTERN.search(fuzzed_body)
  1479→            if match:
  1480→                result.classification = "reflected"
  1481→                result.matched_pattern = match.group(0)
  1482→
  1483→        # --- Time-based detection ------------------------------------------
  1484→        if self._base_time > 0 and response_time > self._base_time * 3:
  1485→            if response_time > 4.0:
  1486→                if result.classification in ("none", ""):
  1487→                    result.classification = "time_based"
  1488→                    result.matched_pattern = f"Response time {response_time:.2f}s (baseline {self._base_time:.2f}s)"
  1489→                elif result.classification == "reflected":
  1490→                    result.classification = "time_based"
  1491→                    result.matched_pattern = f"Reflected + slow response ({response_time:.2f}s)"
  1492→
  1493→        # --- Status code change detection (blind) ---------------------------
  1494→        if (
  1495→            self._baseline_status > 0
  1496→            and status_code != self._baseline_status
  1497→            and status_code >= 400
  1498→            and result.classification in ("none", "")
  1499→        ):
  1500→            result.classification = "blind"
  1501→            result.matched_pattern = (
  1502→                f"Status changed from {self._baseline_status} to {status_code}"
  1503→            )
  1504→
  1505→        # --- Determine vulnerability ---------------------------------------
  1506→        if result.classification in ("reflected", "error_based", "time_based"):
  1507→            result.is_vulnerable = True
  1508→        elif result.classification == "blind":
  1509→            result.is_vulnerable = True
  1510→
  1511→        return result

  1513→    def get_vulnerable(self) -> list[FuzzResult]:
  1514→        """Return only results classified as vulnerable."""
  1515→        return [r for r in self.results if r.is_vulnerable]

  1542→def quick_scan(
  1543→    target: str,
  1544→    params: list[str],
  1545→    categories: Optional[list[str]] = None,
  1546→    timeout: float = 10.0,
  1547→) -> tuple[TechProfile, list[FuzzResult]]:
  1548→    """Convenience function: detect tech, calibrate, and fuzz all params.
  1549→
  1550→    Returns (TechProfile, list_of_results).
  1551→    """
  1552→    if categories is None:
  1553→        categories = ["sqli", "xss", "ssti", "command_injection", "path_traversal"]
  1554→
  1555→    # Quick tech detection
  1556→    status, body, _, headers = 0, "", 0.0, {}
  1557→    try:
  1558→        parsed = urllib.parse.urlparse(target)
  1559→        base_url = f"{parsed.scheme}://{parsed.netloc}{parsed.path}"
  1560→        ctx = None
  1561→        import ssl
  1562→        try:
  1563→            ctx = ssl._create_unverified_context()
  1564→            handler = urllib.request.HTTPSHandler(context=ctx)
  1565→            opener = urllib.request.build_opener(handler)
  1566→        except Exception:
  1567→            opener = urllib.request.build_opener()
  1568→        req = urllib.request.Request(
  1569→            base_url,
  1570→            headers={"User-Agent": "ReconPro/7.5 (Security Scanner)"},
  1571→        )
  1572→        resp = opener.open(req, timeout=timeout)
  1573→        body = resp.read().decode("utf-8", errors="replace")
  1574→        headers = dict(resp.headers)
  1575→        status = resp.status
  1576→    except Exception:
  1577→        pass
  1578→
  1579→    detector = TechDetector()
  1580→    tech_profile = detector.analyze(target, headers, body)
  1581→
  1582→    session = FuzzSession(target, tech_profile=tech_profile, timeout=timeout)
  1583→    session._baseline_status = status
  1584→    results = session.fuzz_all_parameters(target, params, categories)
  1585→    return tech_profile, results
