    41→def _module_breakdown(findings: List[Dict[str, Any]]) -> Dict[str, int]:
    42→    """Count findings per module."""
    43→    counts: Dict[str, int] = {}
    44→    for f in findings:
    45→        m = f.get("module", "unknown")
    46→        counts[m] = counts.get(m, 0) + 1
    47→    return counts
    48→
    49→
    50→def _category_breakdown(findings: List[Dict[str, Any]]) -> Dict[str, int]:
    51→    """Count findings per category."""
    52→    counts: Dict[str, int] = {}
    53→    for f in findings:
    54→        c = f.get("category", "general")
    55→        counts[c] = counts.get(c, 0) + 1
    56→    return counts
    57→
    58→
    59→def _points_per_module(findings: List[Dict[str, Any]]) -> Dict[str, int]:
    60→    """Sum points deducted per module."""
    61→    pts: Dict[str, int] = {}
    62→    for f in findings:
    63→        m = f.get("module", "unknown")
    64→        pts[m] = pts.get(m, 0) + f.get("points_deducted", 0)
    65→    return pts
    66→
    67→
    68→def _severity_per_module(
    69→    findings: List[Dict[str, Any]],
    70→) -> Dict[str, Dict[str, int]]:
    71→    """Count findings per module per severity."""
    72→    matrix: Dict[str, Dict[str, int]] = {}
    73→    for f in findings:
    74→        m = f.get("module", "unknown")
    75→        s = f.get("severity", "info").lower()
    76→        matrix.setdefault(m, {})
    77→        matrix[m][s] = matrix[m].get(s, 0) + 1
    78→    return matrix

   105→    return [base[i % len(base)] for i in range(n)]

   333→def _chart_severity_stacked_bar(sev_matrix: Dict[str, Dict[str, int]]) -> str:
   334→    """Stacked bar chart: severity breakdown per module."""
   335→    if not sev_matrix:
   336→        return ''
   337→
   338→    # Get all modules and severities
   339→    all_modules = sorted(sev_matrix.keys(), key=lambda m: sum(sev_matrix[m].values()), reverse=True)[:10]
   340→    severities = ["critical", "high", "medium", "low", "info"]
   341→    sev_colors = {s: _report_theme.sev_hex(s) for s in severities}
   342→
   343→    datasets = []
   344→    for sev in severities:
   345→        data = [sev_matrix.get(m, {}).get(sev, 0) for m in all_modules]
   346→        if any(d > 0 for d in data):
   347→            datasets.append({
   348→                "label": sev.capitalize(),
   349→                "data": data,
   350→                "backgroundColor": sev_colors[sev],
   351→                "borderRadius": 2,
   352→                "borderSkipped": False,
   353→            })
   354→
   355→    if not datasets:
   356→        return ''
   357→
   358→    labels = [m.upper() for m in all_modules]
   359→    return f'''
   360→    <script>
   361→      (function() {{
   362→        const ctx = document.getElementById('chart-stacked').getContext('2d');
   363→        new Chart(ctx, {{
   364→          type: 'bar',
   365→          data: {{
   366→            labels: {json.dumps(labels)},
   367→            datasets: {json.dumps(datasets)}
   368→          }},
   369→          options: {{
   370→            responsive: true,
   371→            maintainAspectRatio: true,
   372→            scales: {{
   373→              x: {{ stacked: true, ticks: {{ color: '#666', font: {{ size: 10 }} }}, grid: {{ display: false }} }},
   374→              y: {{ stacked: true, beginAtZero: true, ticks: {{ color: '#666', stepSize: 1 }}, grid: {{ color: '#1a1a2e' }} }}
   375→            }},
   376→            plugins: {{
   377→              legend: {{
   378→                position: 'bottom',

   390→
   391→def generate_html_report(data: Dict[str, Any], output_path: str = "") -> str:
   392→    """Generate a Chart.js-powered HTML report. Returns file path.
   393→
   394→    Phase F: Complete overhaul with interactive charts:
   395→    - Score gauge (doughnut with center text)
   396→    - Severity breakdown (doughnut)
   397→    - Findings per module (vertical bar)
   398→    - Points deducted per module (horizontal bar)
   399→    - Category distribution (horizontal bar)
   400→    - Severity stacked per module (stacked bar)
   401→    """
   402→    if not output_path:
   403→        ts = datetime.now().strftime("%Y%m%d_%H%M%S")
   404→        target = data.get("target", "scan").replace("/", "_").replace(":", "")[:30]
   405→        output_path = f"reconpro_report_{target}_{ts}.html"
   406→
   407→    score = data.get("total_score", 0)
   408→    grade = data.get("grade", "N/A")
   409→    target = data.get("target", "Unknown")
   410→    findings = data.get("findings", [])
   411→    sc = data.get("severity_counts", {})
   412→    modules = data.get("modules_run", [])
   413→    badge = data.get("badge_markdown", "")
   414→    scan_time = data.get("scan_time", "")
   415→
   416→    # Version
   417→    try:
   418→        from . import __version__ as _v
   419→        version_str = _v
   420→    except Exception:
   421→        version_str = "7.0.0"
   422→
   423→    # Theme colors
   424→    grade_color = _report_theme.grade_color(grade)
   425→    html_style = _report_theme.html_style()
   426→    h_body_bg = html_style.get("body_bg", "#0a0a0f")
   427→    h_card_bg = html_style.get("card_bg", "#111118")
   428→    h_border = html_style.get("border", "#1a1a2e")
   429→    h_text = html_style.get("text", "#e0e0e0")
   430→    h_muted = html_style.get("muted", "#666")
   431→    h_crit_color = _report_theme.sev_hex("critical")
   432→    h_high_color = _report_theme.sev_hex("high")
   433→    h_med_color = _report_theme.sev_hex("medium")
   434→    h_low_color = _report_theme.sev_hex("low")
   435→    h_info_color = _report_theme.sev_hex("info")
   436→    h_accent = _report_theme.CYAN if hasattr(_report_theme, 'CYAN') else "#06b6d4"
   437→
   438→    # Data processing
   439→    mod_counts = _module_breakdown(findings)
   440→    cat_counts = _category_breakdown(findings)
   441→    pts_mod = _points_per_module(findings)
   442→    sev_matrix = _severity_per_module(findings)
   443→    total_findings = len(findings)
   444→    total_pts = sum(f.get("points_deducted", 0) for f in findings)
   445→
   446→    # ── Chart scripts ──
   447→    chart_score = _chart_score_gauge(score, grade, grade_color)
   448→    chart_severity = _chart_severity_doughnut(sc)
   449→    chart_modules = _chart_module_bar(mod_counts)
   450→    chart_points = _chart_points_hbar(pts_mod)
   451→    chart_categories = _chart_category_hbar(cat_counts)
   452→    chart_stacked = _chart_severity_stacked_bar(sev_matrix)
   453→
   454→    # ── Findings table rows ──
   455→    sorted_findings = sorted(
   456→        findings,
   457→        key=lambda f: {"critical": 0, "high": 1, "medium": 2, "low": 3, "info": 4}.get(
   458→            f.get("severity", "info").lower(), 99
   459→        ),
   460→    )
   461→    findings_rows = ""
   462→    for i, f in enumerate(sorted_findings, 1):
   463→        sev = f.get("severity", "info").lower()
   464→        color = _report_theme.sev_hex(sev)
   465→        evidence = f.get("evidence", "")
   466→        evidence_short = (evidence[:100] + "..." if len(evidence) > 100 else evidence) if evidence else ""
   467→        evidence_cell = f'<td style="color:{h_muted};font-size:12px;max-width:200px;overflow:hidden;text-overflow:ellipsis;white-space:nowrap">{evidence_short}</td>' if evidence_short else '<td></td>'
   468→        findings_rows += f'''
   469→        <tr>
   470→          <td style="color:#666;width:40px">{i}</td>
   471→          <td style="color:{color};font-weight:600;text-transform:uppercase;font-size:11px;width:70px">{sev}</td>
   472→          <td style="color:{h_muted};font-size:12px;width:90px">{f.get("module", "")}</td>
   473→          <td style="color:{h_text}">{f.get("title", "")}</td>
   474→          <td style="color:{h_muted};font-size:12px;width:120px">{f.get("category", "")}</td>
   475→          <td style="text-align:center;color:{h_med_color};width:40px">-{f.get("points_deducted", 0)}</td>
   476→          {evidence_cell}
   477→        </tr>'''
   478→
   479→    if not findings_rows:
   480→        findings_rows = f'<tr><td colspan="7" style="text-align:center;color:{h_low_color};padding:60px 20px">No findings detected. Target appears clean.</td></tr>'
   481→
   482→    # ── Module tags ──
   483→    module_tags = "".join(f'<span class="module-tag">{m}</span>' for m in modules)
   484→
   485→    # ── Top risk findings (critical + high) ──
   486→    top_risk = [f for f in sorted_findings if f.get("severity", "").lower() in ("critical", "high")]
   487→    risk_items = ""
   488→    for f in top_risk[:5]:
   489→        sev = f.get("severity", "").lower()
   490→        color = _report_theme.sev_hex(sev)
   491→        risk_items += f'''
   492→        <div class="risk-item">
   493→          <span class="risk-sev" style="background:{color}">{sev.upper()[:1]}</span>
   494→          <span class="risk-title">{f.get("title", "")}</span>
   495→          <span class="risk-module">{f.get("module", "")}</span>
   496→        </div>'''
   497→    if not risk_items:
   498→        risk_items = f'<p style="color:{h_low_color};padding:16px;text-align:center">No critical or high findings.</p>'
   499→
   500→    # ── Show/hide chart containers based on data availability ──
   501→    has_modules = bool(mod_counts)
   502→    has_points = bool(pts_mod)
   503→    has_categories = bool(cat_counts) and len(cat_counts) > 1
   504→    has_stacked = bool(sev_matrix)
   505→
   506→    html = f'''<!DOCTYPE html>
   507→<html lang="en">
   508→<head>
   509→<meta charset="UTF-8">
   510→<meta name="viewport" content="width=device-width, initial-scale=1.0">
   511→<title>ReconPro Report \u2014 {target}</title>
   512→{_CHARTJS_CDN}
   513→<style>
   514→  * {{ margin:0; padding:0; box-sizing:border-box; }}
   515→  body {{ background:{h_body_bg}; color:{h_text}; font-family:'SF Mono','Fira Code','Cascadia Code',monospace; padding:24px; line-height:1.5; }}
   516→  .container {{ max-width:1200px; margin:0 auto; }}
   517→
   518→  /* Header */
   519→  .report-header {{ display:flex; align-items:center; justify-content:space-between; padding:28px 32px; background:{h_card_bg}; border:1px solid {h_border}; border-radius:16px; margin-bottom:24px; gap:24px; }}
   520→  .brand {{ flex-shrink:0; }}
   521→  .brand h1 {{ font-size:22px; color:#fff; letter-spacing:2px; }}
   522→  .brand .ver {{ color:{h_muted}; font-size:12px; margin-top:2px; }}
   523→  .header-center {{ flex:1; text-align:center; }}
   524→  .target-name {{ font-size:26px; color:#fff; font-weight:700; margin-bottom:4px; word-break:break-all; }}
   525→  .scan-meta {{ color:{h_muted}; font-size:13px; }}
   526→  .header-right {{ flex-shrink:0; width:180px; height:180px; }}
