    40→def _make_tmp_plugin_dir():
    41→    """Create a temp directory to act as PLUGIN_DIR, return Path."""
    42→    d = Path(tempfile.mkdtemp())
    43→    d.mkdir(parents=True, exist_ok=True)
    44→    return d
    45→
    46→
    47→def _write_plugin(tmp_dir: Path, name: str, content: str) -> Path:
    48→    """Write a plugin file into tmp_dir and return its path."""
    49→    p = tmp_dir / f"{name}.py"
    50→    p.write_text(textwrap.dedent(content))
    51→    return p

   747→    def test_callback_exception_caught(self):
   748→        """Exceptions in callbacks must be caught, not propagated."""
   749→        exceptions_seen = []
   750→
   751→        def bad_callback(*args, **kwargs):
   752→            exceptions_seen.append(True)
   753→            raise RuntimeError("Hook callback crashed!")
   754→
   755→        def good_callback(*args, **kwargs):
   756→            return "safe_result"
   757→
   758→        # Register bad callback first, good callback second (higher priority number)
   759→        HookManager.register("error", bad_callback, priority=10)
   760→        HookManager.register("error", good_callback, priority=20)
   761→
   762→        # Should not raise — bad callback's exception is caught
   763→        result = HookManager.fire("error", error_msg="test")
   764→        self.assertIsNone(result)  # bad_callback returned None (due to exception), good_callback ran after
   765→        # Actually, bad_callback raises, which is caught. Its result is None (not appended
   766→        # because the exception path doesn't append). Then good_callback runs and returns
   767→        # "safe_result", which short-circuits.
   768→        # Wait, let me re-read the code. The for loop iterates in priority order.
   769→        # priority 10 runs first (bad_callback). It raises. The except catches it, logs, continues.
   770→        # result is NOT appended. Then priority 20 runs (good_callback). Returns "safe_result".
   771→        # Since result is not None, it short-circuits and returns "safe_result".
   772→        self.assertEqual(result, "safe_result")
   773→        self.assertEqual(len(exceptions_seen), 1)
   774→
