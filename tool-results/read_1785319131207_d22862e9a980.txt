    85→class _RateLimiter:
    86→    """Token-bucket rate limiter — safe for concurrent use.
    87→
    88→    Default: 10 requests per second. Use throttle.acquire() before any
    89→    network call; it blocks until a token is available.
    90→    """
    91→
    92→    def __init__(self, max_per_second: float = 10.0):
    93→        self._rate = max_per_second
    94→        self._min_interval = 1.0 / max_per_second
    95→        self._lock = Lock()
    96→        self._last = 0.0
    97→
    98→    def acquire(self) -> None:
    99→        """Block until we're allowed to make the next request."""
   100→        with self._lock:
   101→            now = time.monotonic()
   102→            elapsed = now - self._last
   103→            if elapsed < self._min_interval:
   104→                time.sleep(self._min_interval - elapsed)
   105→            self._last = time.monotonic()
   106→

   121→def classify_ipv6(addr: str) -> str:
   122→    """Classify an IPv6 address. Returns: 'global', 'link_local', 'unique_local',
   123→    'loopback', 'mapped_v4', 'documentation', 'multicast', 'unspecified'."""
   124→    if addr == "::":
   125→        return "unspecified"
   126→    if _IPV6_LOOPBACK_RE.match(addr):
   127→        return "loopback"
   128→    if _IPV6_LINK_LOCAL_RE.match(addr):
   129→        return "link_local"
   130→    if _IPV6_UNIQUE_LOCAL_RE.match(addr):
   131→        return "unique_local"
   132→    if _IPV6_MAPPED_V4_RE.match(addr):
   133→        return "mapped_v4"
   134→    if _IPV6_DOC_RE.match(addr):
   135→        return "documentation"
   136→    if _IPV6_MULTICAST_RE.match(addr):
   137→        return "multicast"
   138→    return "global"

   175→def run_argv(argv: list, timeout: int = 8) -> Tuple[str, str]:
   176→    """Execute a pre-parsed argv list without any shell interpretation."""
   177→    try:
   178→        r = subprocess.run(argv, shell=False, capture_output=True, text=True, timeout=timeout)
   179→        return r.stdout.strip(), r.stderr.strip()
   180→    except Exception:
   181→        return "", ""

   198→def http_probe(url: str, method: str = "GET", body: Optional[bytes] = None,
   199→               headers: Optional[Dict[str, str]] = None, timeout: int = 8) -> Dict[str, Any]:
   200→    """Hardened single HTTP layer — all modules delegate through this."""
   201→    RATE_LIMITER.acquire()  # m5: thread-safe rate limiting
   202→    h = {
   203→        "User-Agent": "ReconPro-Unified/1.0 (Six-Blades-One-Target; +https://reconpro.security)",
   204→        "X-ReconPro-Signature": RECONPRO_SIGNATURE,
   205→        "Accept": "application/json,text/plain,*/*",
   206→    }
   207→    if headers:
   208→        h.update(headers)
   209→    req = urllib.request.Request(url, data=body, method=method, headers=h)
   210→    try:
   211→        ctx = ssl.create_default_context()
   212→        if CONFIG.insecure:
   213→            ctx.check_hostname = False
   214→            ctx.verify_mode = ssl.CERT_NONE
   215→        with urllib.request.urlopen(req, timeout=timeout, context=ctx) as resp:
   216→            raw = resp.read(8192)
   217→            return {
   218→                "ok": True, "status": resp.status, "reason": resp.reason,
   219→                "headers": dict(resp.headers.items()),
   220→                "body": raw.decode("utf-8", errors="replace")[:8192],
   221→            }
   222→    except urllib.error.HTTPError as e:
   223→        try:
   224→            raw = e.read(8192)
   225→            body_str = raw.decode("utf-8", errors="replace")[:8192]
   226→        except Exception:
   227→            body_str = ""
   228→        return {
   229→            "ok": False, "status": e.code, "reason": e.reason,
   230→            "headers": dict(e.headers.items()) if e.headers else {},
   231→            "body": body_str,
   232→        }
   233→    except Exception as e:
   234→        return {"ok": False, "status": 0, "reason": str(e), "headers": {}, "body": ""}

   939→_FILENAME_UNSAFE_RE = re.compile(r'[\x00-\x1f\x7f\x80-\x9f/\\:\n\r\t]')

   942→def _safe_filename(raw: str) -> str:
   943→    """Sanitize a user-supplied string for use as a filename component.
   944→
   945→    Strips: null bytes, control chars, path separators, backslashes.
   946→    Truncates to _FILENAME_MAX_LEN. Rejects empty results (returns 'unnamed').
   947→    """
   948→    cleaned = _FILENAME_UNSAFE_RE.sub('_', raw)
   949→    # Collapse consecutive underscores
   950→    cleaned = re.sub(r'_{2,}', '_', cleaned)
   951→    cleaned = cleaned.strip('._-')
   952→    cleaned = cleaned[:_FILENAME_MAX_LEN]
   953→    if not cleaned or cleaned in ('.', '..'):
   954→        return 'unnamed'
   955→    return cleaned

   957→def _find_cache(prefix: str, host: str) -> Optional[str]:
   958→    """Find a cached JSON for this host with the given prefix."""
   959→    if not os.path.isdir(CACHE_DIR):
   960→        return None
   961→    safe = _safe_filename(host)
   962→    candidates = [
   963→        f"{CACHE_DIR}/{prefix}_{safe}.json",
   964→        f"{CACHE_DIR}/{prefix}_{safe}_v2.json",
   965→    ]
   966→    # also any *{host}*.json
   967→    for f in sorted(os.listdir(CACHE_DIR), reverse=True):
   968→        if f.startswith(prefix) and safe in f and f.endswith(".json"):
   969→            candidates.append(f"{CACHE_DIR}/{f}")
   970→    for c in candidates:
   971→        if os.path.exists(c) and os.path.getsize(c) > 100:
   972→            return c
   973→    return None

  1060→def module_vibesec(host: str) -> Dict[str, Any]:
  1061→    """Rapid AI/vibe-coding vulnerability benchmark.
  1062→
  1063→    Checks four categories:
  1064→      1. Exposed env/config endpoints
  1065→      2. Unauthenticated API/webhook routes
  1066→      3. Permissive CORS policies
  1067→      4. Exposed anon/public backend keys (Supabase, Firebase, S3, R2)
  1068→
  1069→    Returns a 100-point VibeSec score with A+ to F grade mapping.
  1070→    Every finding is verified via native HTTP response validation (zero fabrication).
  1071→    """
  1072→    findings: List[Dict[str, Any]] = []
  1073→    deductions = 0  # each vulnerability deducts from 100
  1074→
  1075→    base_url = host if host.startswith("http") else f"https://{host}"
  1076→
  1077→    def add(title, severity, category, description, evidence, points_deducted):
  1078→        nonlocal deductions
  1079→        deductions += points_deducted
  1080→        findings.append({
  1081→            "title": title, "severity": severity, "category": category,
  1082→            "description": description, "evidence": evidence, "asset": host,
  1083→            "points_deducted": points_deducted,
  1084→        })
  1085→
  1086→    # ── Category 1: Exposed Environment/Config Files ──────────────────────
  1087→    audit_log("vibesec.config.start", detail=host)
  1088→    for path in VIBESEC_SENSITIVE_PATHS:
  1089→        url = base_url.rstrip("/") + path
  1090→        resp = http_probe(url, timeout=5)
  1091→        status = resp.get("status", 0)
  1092→        body = resp.get("body", "")[:2048]
  1093→        if status == 200 and len(body) > 10:
  1094→            # Verify: real content, not just a redirect page
  1095→            has_secrets = any(kw in body.lower() for kw in [
  1096→                "api_key", "secret", "password", "token", "database_url",
  1097→                "private_key", "supabase", "firebase", "aws_access",
  1098→            ])
  1099→            if has_secrets:
  1100→                add(f"Exposed config — {path}", "critical", "exposed_config",
  1101→                    f"Sensitive configuration endpoint accessible: {path} (200 OK, {len(body)} bytes, contains secrets)",
  1102→                    f"GET {path} → {status} ({len(body)}B, secrets detected)", 15)
  1103→            else:
  1104→                add(f"Exposed config — {path}", "high", "exposed_config",
  1105→                    f"Configuration endpoint accessible: {path} (200 OK, {len(body)} bytes)",
  1106→                    f"GET {path} → {status} ({len(body)}B)", 10)
  1107→        elif status in (200, 201, 301, 302, 307, 308) and status != 404:
  1108→            add(f"Config path accessible — {path}", "medium", "exposed_config",
  1109→                f"Endpoint returned {status} (may redirect or serve partial content)",
  1110→                f"GET {path} → {status}", 5)
  1111→    audit_log("vibesec.config.done", detail=f"{len([f for f in findings if f['category'] == 'exposed_config'])} findings")
  1112→
  1113→    # ── Category 2: Unauthenticated API/Webhook Routes ────────────────────
  1114→    audit_log("vibesec.api.start", detail=host)
  1115→    for path in VIBESEC_API_PATHS:
  1116→        url = base_url.rstrip("/") + path
  1117→        resp = http_probe(url, timeout=5)
  1118→        status = resp.get("status", 0)
  1119→        body = resp.get("body", "")[:2048]
  1120→        if status == 200:
  1121→            # Check for auth-required indicators
  1122→            body_lower = body.lower()
  1123→            is_protected = any(kw in body_lower for kw in [
  1124→                "unauthorized", "401", "forbidden", "authentication required",
  1125→                "\"error\"", "login required",
  1126→            ])
  1127→            if not is_protected and len(body) > 20:
  1128→                add(f"Unauthenticated API — {path}", "high", "unauth_api",
  1129→                    f"API route accessible without authentication: {path} (200 OK, {len(body)} bytes)",
  1130→                    f"GET {path} → {status} (no auth required)", 10)
  1131→            elif status == 200:
  1132→                add(f"API route reachable — {path}", "medium", "unauth_api",
  1133→                    f"API route returned 200 but may have auth checks in POST/DELETE",
  1134→                    f"GET {path} → {status}", 5)
  1135→        elif status in (403, 401):
  1136→            pass  # properly protected — no finding
  1137→        elif status == 404:
  1138→            pass  # not found — no finding
  1139→        elif status == 405:
  1140→            add(f"API route exists — {path}", "low", "unauth_api",
  1141→                f"API route exists (405 Method Not Allowed) — may be exploitable with correct method",
  1142→                f"GET {path} → 405", 3)
  1143→    audit_log("vibesec.api.done", detail=f"{len([f for f in findings if f['category'] == 'unauth_api'])} findings")
  1144→
  1145→    # ── Category 3: CORS Policy Analysis ────────────────────────────────────
  1146→    audit_log("vibesec.cors.start", detail=host)
  1147→    # Test with an origin probe
  1148→    cors_resp = http_probe(base_url, timeout=5)
  1149→    cors_headers = cors_resp.get("headers", {})
  1150→    acao = cors_headers.get("Access-Control-Allow-Origin", "")
  1151→    if acao == "*":
  1152→        add("Permissive CORS — wildcard origin", "high", "cors",
  1153→            "Access-Control-Allow-Origin: * — any domain can make cross-origin requests",
  1154→            f"CORS header: {acao}", 12)
  1155→    elif acao and acao != "null":
  1156→        # Specific origin — check if it reflects the Origin header
  1157→        reflect_resp = http_probe(base_url, headers={"Origin": "https://evil-attacker.com"}, timeout=5)
  1158→        reflected = reflect_resp.get("headers", {}).get("Access-Control-Allow-Origin", "")
  1159→        if "evil-attacker" in reflected:
  1160→            add("CORS origin reflection vulnerability", "critical", "cors",
  1161→                "Server reflects any Origin header back — allows cross-origin attacks from any domain",
  1162→                f"Sent Origin: https://evil-attacker.com, got back: {reflected}", 15)
  1163→    allow_cred = cors_headers.get("Access-Control-Allow-Credentials", "")
  1164→    if acao != "" and allow_cred.lower() == "true":
  1165→        add("CORS credentials exposed", "high", "cors",
  1166→            "Access-Control-Allow-Credentials: true with non-empty Allow-Origin",
  1167→            f"Allow-Credentials: true, Allow-Origin: {acao or '(reflected)'}", 10)
  1168→    if not acao:
  1169→        # No CORS header at all — check if it's a JSON API that should have CORS
  1170→        ct = cors_headers.get("content-type", "")
  1171→        if "json" in ct.lower():
  1172→            add("JSON API without CORS headers", "low", "cors",
  1173→                "API returns JSON but sets no CORS headers — may be intentional or oversight",
  1174→            f"Content-Type: {ct}, no CORS headers", 3)
  1175→    audit_log("vibesec.cors.done", detail=f"{len([f for f in findings if f['category'] == 'cors'])} findings")
  1176→
  1177→    # ── Category 4: Exposed Anon/Public Backend Keys ────────────────────────
  1178→    audit_log("vibesec.anon_keys.start", detail=host)
  1179→    # Check the main page body for exposed service URLs
  1180→    main_body = http_probe(base_url, timeout=5).get("body", "")[:16384]
  1181→    for name, pattern in VIBESEC_ANON_KEY_PATTERNS:
  1182→        matches = pattern.findall(main_body)
  1183→        if matches:
  1184→            unique = list(set(matches))[:5]
  1185→            # Verify each match resolves (zero fabrication)
  1186→            verified = []
  1187→            for m in unique:
  1188→                test_url = f"https://{m}" if not m.startswith("http") else m
  1189→                probe = http_probe(test_url, timeout=3)
  1190→                if probe.get("status", 0) > 0:
  1191→                    verified.append(m)
  1192→            if verified:
  1193→                add(f"Exposed {name} backend — {len(verified)} instance(s)", "critical", "anon_keys",
  1194→                    f"Public {name} URLs found in page source and confirmed reachable",
  1195→                    f"{name}: {', '.join(verified[:3])}", 15)
  1196→            else:
  1197→                add(f"Detected {name} references — not verified", "medium", "anon_keys",
  1198→                    f"{name} URLs found in page source but could not confirm reachability",
  1199→                    f"{name}: {', '.join(unique[:3])}", 5)
  1200→
  1201→    # Also check robots.txt and sitemap for exposed service URLs
  1202→    for check_path in ["/robots.txt", "/sitemap.xml"]:
  1203→        check_resp = http_probe(base_url.rstrip("/") + check_path, timeout=5)
  1204→        if check_resp.get("status") == 200:
  1205→            check_body = check_resp.get("body", "")[:8192]
  1206→            for name, pattern in VIBESEC_ANON_KEY_PATTERNS:
  1207→                matches = pattern.findall(check_body)
  1208→                if matches and not any(f["title"].startswith(f"Exposed {name}") for f in findings):
  1209→                    unique = list(set(matches))[:3]
  1210→                    add(f"{name} URLs in {check_path}", "medium", "anon_keys",
  1211→                        f"Public {name} URLs exposed in {check_path}",
  1212→                        f"{check_path}: {', '.join(unique)}", 5)
  1213→    audit_log("vibesec.anon_keys.done", detail=f"{len([f for f in findings if f['category'] == 'anon_keys'])} findings")
  1214→
  1215→    # ── Score Calculation ──────────────────────────────────────────────────
  1216→    raw_score = max(0, 100 - deductions)
  1217→    # Normalize: if no findings at all, perfect score
  1218→    if not findings:
  1219→        raw_score = 100
  1220→    # Clamp
  1221→    raw_score = min(100, max(0, raw_score))
  1222→    grade, grade_color = _vibesec_compute_grade(raw_score)
  1223→    badge_md = _vibesec_render_badge(host, grade, raw_score)
  1224→
  1225→    severity_counts: Dict[str, int] = {}
  1226→    for f in findings:
  1227→        sev = f["severity"]
  1228→        severity_counts[sev] = severity_counts.get(sev, 0) + 1
  1229→
  1230→    result = {
  1231→        "module": "VIBESEC",
  1232→        "vibesec_score": raw_score,
  1233→        "grade": grade,
  1234→        "grade_color": grade_color,
  1235→        "badge_markdown": badge_md,
  1236→        "total_findings": len(findings),
  1237→        "severity_counts": severity_counts,
  1238→        "findings": findings,
  1239→        "categories_checked": ["exposed_config", "unauth_api", "cors", "anon_keys"],
  1240→        "max_possible_score": 100,
  1241→    }
  1242→    audit_log("vibesec.complete", detail=f"score={raw_score} grade={grade} findings={len(findings)}")
  1243→    return result

  1316→def compute_unified_verdict(report: Dict[str, Any]) -> Dict[str, Any]:
  1317→    """Combine all module scores into a unified verdict."""
  1318→    scores = {}
  1319→    # RECON: weighted by severity counts
  1320→    recon = report.get("recon", {})
  1321→    sc = recon.get("severity_counts", {})
  1322→    scores["recon"] = min(100, sc.get("critical", 0) * 25 + sc.get("high", 0) * 15 +
  1323→                          sc.get("medium", 0) * 8 + sc.get("low", 0) * 3 + sc.get("info", 0))
  1324→    # AUTH: bypasses / 75 * 100
  1325→    auth = report.get("auth_bypass", {})
  1326→    scores["auth"] = min(100, int((auth.get("bypasses_successful", 0) / max(1, auth.get("total_attempts", 1))) * 400))
  1327→    # CHAIN: SSRF + open redirects
  1328→    chain = report.get("chain_hunter", {})
  1329→    scores["chain"] = min(100, chain.get("ssrf_detected", 0) * 25 + chain.get("open_redirects", 0) * 15)
  1330→    # BOT: bot hits
  1331→    bot = report.get("bot_hunter", {})
  1332→    scores["bot"] = min(100, bot.get("bot_hits", 0) * 20)
  1333→    # GORGON
  1334→    gorgon = report.get("gorgon", {})
  1335→    scores["gorgon"] = gorgon.get("threatScore", gorgon.get("verdict", {}).get("threatScore", 0)) if isinstance(gorgon, dict) else 0
  1336→    # OBLIVION
  1337→    oblivion = report.get("oblivion", {})
  1338→    scores["oblivion"] = oblivion.get("verdict", {}).get("threatScore", 0) if isinstance(oblivion, dict) else 0
  1339→    # VIBESEC: inverted score (low vibesec = more dangerous for unified)
  1340→    vibesec = report.get("vibesec", {})
  1341→    vibesec_score = vibesec.get("vibesec_score", 100) if isinstance(vibesec, dict) else 100
  1342→    scores["vibesec"] = 100 - vibesec_score  # invert: more vibe-vulns = higher unified threat
  1343→
  1344→    # Weighted unified score
  1345→    weights = {"recon": 0.08, "auth": 0.13, "chain": 0.13, "bot": 0.08, "gorgon": 0.18, "oblivion": 0.28, "vibesec": 0.12}
  1346→    unified = int(sum(scores.get(k, 0) * w for k, w in weights.items()))
  1347→
  1348→    if unified >= 90: verdict = "OMNIPOTENT VERDICT — The target has been comprehensively dissolved."
  1349→    elif unified >= 75: verdict = "DEVASTATING VERDICT — The target's defenses are thoroughly compromised."
  1350→    elif unified >= 50: verdict = "SUBSTANTIAL VERDICT — Multiple critical exposures confirmed."
  1351→    elif unified >= 25: verdict = "NOTABLE VERDICT — Some exposures detected."
  1352→    else: verdict = "MUNDANE VERDICT — The target resisted most probes."
  1353→
  1354→    return {
  1355→        "unified_score": unified,
  1356→        "module_scores": scores,
  1357→        "verdict_text": verdict,
  1358→        "verdict_level": "OMNIPOTENT" if unified >= 90 else "DEVASTATING" if unified >= 75 else
  1359→                          "SUBSTANTIAL" if unified >= 50 else "NOTABLE" if unified >= 25 else "MUNDANE",
  1360→    }
