   582→   871→  1220→        self,
   583→   872→  1221→        target: str,

   585→   874→  1223→        is_local: bool = False,

   599→   888→  1237→                else [k for k in MODULE_NAMES if k not in ("host", "dev", "doctor")]
   600→   889→  1238→            )

   612→   901→  1250→
   613→   902→  1251→            if is_local:
   614→   903→  1252→                result = audit_scan(target=target, modules=modules)

   634→   923→  1272→            else:
   635→   924→  1273→                self.call_from_thread(self._set_all_modules_status, "error")
   636→   925→  1274→

   992→  1281→  1630→        """Open finding detail modal for the clicked finding.
   993→  1282→  1631→
   994→  1283→  1632→        Uses the vertical click position to determine which finding index
