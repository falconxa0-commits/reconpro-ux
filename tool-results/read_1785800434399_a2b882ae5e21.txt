    11→results = []  # (name, status, detail, duration)
    12→
    13→def test(name, fn, timeout=30):
    14→    """Run a single test and record result."""
    15→    start = time.time()
    16→    try:
    17→        result = fn()
    18→        dur = time.time() - start
    19→        status = "PASS" if result else "FAIL"
    20→        results.append((name, status, "OK", dur))
    21→        symbol = "\033[92mPASS\033[0m" if result else "\033[91mFAIL\033[0m"
    22→        print(f"  [{symbol}] {name} ({dur:.1f}s)")
    23→        return result
    24→    except Exception as e:
    25→        dur = time.time() - start
    26→        results.append((name, "ERROR", str(e), dur))
    27→        print(f"  [\033[91mERROR\033[0m] {name} ({dur:.1f}s): {e}")
    28→        return False
    29→
    30→
    31→def report():
    32→    passed = sum(1 for _, s, _, _ in results if s == "PASS")
    33→    failed = sum(1 for _, s, _, _ in results if s in ("FAIL", "ERROR"))
    34→    total = len(results)
    35→    print(f"\n{'='*70}")
    36→    print(f"RESULTS: {passed}/{total} passed, {failed} failed")
    37→    if failed:
    38→        print(f"\n\033[91mFAILURES:\033[0m")
    39→        for name, status, detail, dur in results:
    40→            if status in ("FAIL", "ERROR"):
    41→                print(f"  x {name}")
    42→                print(f"    {detail[:300]}")
    43→                traceback_detail = detail if '\n' in detail else ''
    44→                if traceback_detail:
    45→                    for line in traceback_detail.split('\n')[:5]:
    46→                        print(f"    {line}")
    47→    else:
    48→        print(f"\n\033[92mALL TESTS PASSED\033[0m")
    49→    print(f"{'='*70}")
    50→    return failed == 0
