    48→    48→def _run(cmd: str, timeout: int = 10) -> Tuple[int, str]:
    49→    49→    """Run a shell command, return (exit_code, stdout+stderr)."""
    50→    50→    try:
    51→    51→        r = subprocess.run(
    52→    52→            cmd, shell=True, capture_output=True, text=True, timeout=timeout

    80→    80→def _check_open_ports() -> List[Finding]:
    81→    81→    """Find open listening ports via ss/netstat."""
    82→    82→    findings: List[Finding] = []
    83→    83→    hostname = _hostname()
    84→    84→
    85→    85→    # Platform-specific port enumeration
    86→    86→    if _is_windows():
    87→    87→        code, out = _run('netstat -an 2>NUL | findstr LISTENING')
    88→    88→        if code != 0 or not out:
    89→    89→            findings.append(Finding(
    90→    90→                title="Could not enumerate open ports",
    91→    91→                severity="low", category="ports",
    92→    92→                module="host",
    93→    93→                description="Unable to enumerate open ports.",
    94→    94→                evidence="netstat returned no output",
    95→    95→                asset=hostname, points_deducted=1,
    96→    96→                remediation="Run as Administrator for full port visibility.",
    97→    97→            ))
    98→    98→            return findings
    99→    99→    else:
   100→   100→        code, out = _run("ss -tlnp 2>/dev/null || netstat -tlnp 2>/dev/null")
   101→   101→        if code != 0 or not out:
   102→   102→            findings.append(Finding(
   103→   103→                title="Could not enumerate open ports",
   104→   104→                severity="low", category="ports",
   105→   105→                module="host",
   106→   106→                description="Unable to run ss or netstat. Try running with sudo.",
   107→   107→                evidence="ss/netstat returned no output",
   108→   108→                asset=hostname, points_deducted=1,
   109→   109→                remediation="Run with sudo for full port visibility.",
   110→   110→            ))
   111→   111→            return findings
   112→   112→
   113→   113→    ports = []
   114→   114→    for line in out.splitlines():
   115→   115→        # Parse ss output: 0.0.0.0:3000 or [::]:8080
   116→   116→        match = re.search(r':(\d+)\s', line)
   117→   117→        if match:
   118→   118→            port = int(match.group(1))
   119→   119→            ports.append(port)
   120→   120→
   121→   121→    # Categorize risky ports
   122→   122→    risky_ports = {
   123→   123→        23: ("high", "Telnet (unencrypted)"),
   124→   124→        21: ("high", "FTP (unencrypted)"),
   125→   125→        445: ("high", "SMB/CIFS"),
   126→   126→        3389: ("high", "RDP"),
   127→   127→        5900: ("high", "VNC"),
   128→   128→        6379: ("high", "Redis (no auth by default)"),
   129→   129→        27017: ("high", "MongoDB (no auth by default)"),
   130→   130→        11211: ("high", "Memcached"),
   131→   131→        9200: ("medium", "Elasticsearch"),
   132→   132→        5432: ("medium", "PostgreSQL"),
   133→   133→        3306: ("medium", "MySQL"),
   134→   134→        8080: ("low", "HTTP alt"),
   135→   135→        8443: ("low", "HTTPS alt"),
   136→   136→        3000: ("low", "Node.js dev server"),
   137→   137→        5173: ("low", "Vite dev server"),
   138→   138→        8000: ("low", "Python dev server"),
   139→   139→    }
   140→   140→
   141→   141→    for port in sorted(set(ports)):
   142→   142→        if port in risky_ports:
   143→   143→            sev, desc = risky_ports[port]
   144→   144→            findings.append(Finding(
   145→   145→                title=f"Risky open port: {port} ({desc})",
   146→   146→                severity=sev, category="ports",
   147→   147→                module="host",
   148→   148→                description=f"Port {port} ({desc}) is open and listening. This could expose services to network attackers.",
   149→   149→                evidence=f"Port {port} found in ss/netstat output",
   150→   150→                asset=hostname, points_deducted=8 if sev == "high" else 4,
   151→   151→                remediation=f"Close port {port} if not needed, or bind to 127.0.0.1 only.",
   152→   152→            ))
   153→   153→
   154→   154→    if not findings and ports:
   155→   155→        findings.append(Finding(
   156→   156→            title=f"{len(set(ports))} open ports detected",
   157→   157→            severity="info", category="ports",
   158→   158→            module="host",
   159→   159→            description=f"Found {len(set(ports))} open ports. None are known high-risk.",
   160→   160→            evidence=", ".join(str(p) for p in sorted(set(ports))[:20]),
   161→   161→            asset=hostname, points_deducted=0,
   162→   162→            remediation="Review all open ports periodically.",
   163→   163→        ))
   164→   164→
   165→   165→    return findings

   215→   215→                module="host",

   285→   285→def _check_users() -> List[Finding]:
   286→   286→    """Check user accounts and sudo access."""
   287→   287→    findings: List[Finding] = []
   288→   288→    hostname = _hostname()
   289→   289→
   290→   290→    if _is_windows():
   291→   291→        # Check administrator accounts on Windows
   292→   292→        code, out = _run('net localgroup Administrators 2>NUL')
   293→   293→        if code == 0 and out:
   294→   294→            lines = [l.strip() for l in out.splitlines() if l.strip()]
   295→   295→            # First two lines are header, last line is "The command completed..."
   296→   296→            admins = [l for l in lines[2:] if l and "command completed" not in l.lower()]
   297→   297→            if len(admins) > 2:
   298→   298→                findings.append(Finding(
   299→   299→                    title=f"{len(admins)} administrator accounts found",
   300→   300→                    severity="medium", category="users",
   301→   301→                    module="host",
   302→   302→                    description=f"{len(admins)} users have administrator privileges: {', '.join(admins[:5])}. Minimize admin access to reduce risk.",
   303→   303→                    evidence=f"Administrators: {', '.join(admins[:5])}",
   304→   304→                    asset=hostname, points_deducted=5,
   305→   305→                    remediation="Remove unnecessary users from the Administrators group via Computer Management.",
   306→   306→                ))
   307→   307→        return findings
   308→   308→
   309→   309→    # Non-Windows: check sudo users
   310→   310→    code, out = _run("getent group sudo 2>/dev/null")
   311→   311→    if code == 0 and out:
   312→   312→        users = [u for u in out.split(":")[1].split(",") if u.strip()]
   313→   313→        if len(users) > 2:
   314→   314→            findings.append(Finding(
   315→   315→                title=f"{len(users)} users have sudo privileges",
   316→   316→                severity="medium", category="users",
   317→   317→                module="host",
   318→   318→                description=f"{len(users)} users can run sudo: {', '.join(users[:5])}. Minimize sudo access to reduce risk.",
   319→   319→                evidence=f"sudo group: {out[:200]}",
   320→   320→                asset=hostname, points_deducted=5,
   321→   321→                remediation="Remove unnecessary users from the sudo group. Use specific capabilities instead.",
   322→   322→            ))
   323→   323→
   324→   324→    # Check for passwordless sudo (non-Windows only)
   325→   325→    code, out = _run("sudo -n true 2>&1")
   326→   326→    if code == 0:
   327→   327→        findings.append(Finding(
   328→   328→            title="Passwordless sudo is enabled",
   329→   329→            severity="high", category="users",
   330→   330→            module="host",
   331→   331→            description="Current user can run sudo without a password. Any code running as your user has root access.",
   332→   332→            evidence="sudo -n returned exit code 0",
   333→   333→            asset=hostname, points_deducted=10,
   334→   334→            remediation="Require password for sudo: remove NOPASSWD from /etc/sudoers",
   335→   335→        ))
   336→   336→
   337→   337→    # Check for empty password users (Linux only)
   338→   338→    code, out = _run("sudo awk -F: '($2 == \"\") {print $1}' /etc/shadow 2>/dev/null")
   339→   339→    if code == 0 and out.strip():
   340→   340→        users = out.strip().splitlines()
   341→   341→        findings.append(Finding(
   342→   342→            title=f"{len(users)} user(s) with empty password",
   343→   343→            severity="critical", category="users",
   344→   344→            module="host",
   345→   345→            description=f"Users with no password: {', '.join(users[:5])}. This is a critical security risk.",
   346→   346→            evidence=out[:200],
   347→   347→            asset=hostname, points_deducted=15,
   348→   348→            remediation="Set passwords for all users immediately: sudo passwd <username>",
   349→   349→        ))
   350→   350→
   351→   351→    return findings

   354→   354→def _check_ssh() -> List[Finding]:
   355→   355→    """Check SSH configuration hardening."""
   356→   356→    findings: List[Finding] = []
   357→   357→    hostname = _hostname()
   358→   358→    sshd_config = "/etc/ssh/sshd_config"
   359→   359→
   360→   360→    if not _file_exists(sshd_config):
   361→   361→        return findings
   362→   362→
   363→   363→    try:
   364→   364→        with open(sshd_config) as f:
   365→   365→            config = f.read()
   366→   366→    except Exception:
   367→   367→        return findings
   368→   368→
   369→   369→    ssh_checks = [
   370→   370→        (r'^#?PermitRootLogin\s+yes', "high", "SSH root login is permitted", "Set PermitRootLogin no"),
   371→   371→        (r'^#?PasswordAuthentication\s+yes', "medium", "SSH password authentication enabled", "Use SSH keys only: PasswordAuthentication no"),
   372→   372→        (r'^#?PermitEmptyPasswords\s+yes', "critical", "SSH allows empty passwords", "Set PermitEmptyPasswords no"),
   373→   373→        (r'^#?X11Forwarding\s+yes', "low", "SSH X11 forwarding enabled", "Set X11Forwarding no if not needed"),
   374→   374→        (r'^#?MaxAuthTries\s+(\d+)', None, "", ""),  # just read value
   375→   375→        (r'^#?Protocol\s+1', "high", "SSH Protocol 1 in use (insecure)", "Use Protocol 2"),
   376→   376→        (r'^#?AllowAgentForwarding\s+yes', "low", "SSH agent forwarding enabled", "Disable if not needed: AllowAgentForwarding no"),
   377→   377→    ]
   378→   378→
   379→   379→    for pattern, sev, title, remediation in ssh_checks:
   380→   380→        if re.search(pattern, config, re.MULTILINE | re.IGNORECASE):
   381→   381→            if sev is None:
   382→   382→                # MaxAuthTries check
   383→   383→                match = re.search(r'^#?MaxAuthTries\s+(\d+)', config, re.MULTILINE)
   384→   384→                if match and int(match.group(1)) > 6:
   385→   385→                    findings.append(Finding(
   386→   386→                        title=f"SSH MaxAuthTries is {match.group(1)} (too high)",
   387→   387→                        severity="low", category="ssh",
   388→   388→                        module="host",
   389→   389→                        description=f"MaxAuthTries={match.group(1)} allows more brute-force attempts.",
   390→   390→                        evidence=f"MaxAuthTries {match.group(1)}",
   391→   391→                        asset=hostname, points_deducted=2,
   392→   392→                        remediation="Set MaxAuthTries 4",
   393→   393→                    ))
   394→   394→            else:
   395→   395→                findings.append(Finding(
   396→   396→                    title=title,
   397→   397→                    severity=sev, category="ssh",
   398→   398→                    module="host",
   399→   399→                    description=f"SSH config issue: {title}",
   400→   400→                    evidence=f"Found in {sshd_config}",
   401→   401→                    asset=hostname, points_deducted=8 if sev == "high" else (15 if sev == "critical" else 3),
   402→   402→                    remediation=remediation,
   403→   403→                ))
   404→   404→
   405→   405→    return findings

   408→   408→def _check_docker() -> List[Finding]:
   409→   409→    """Check Docker security."""
   410→   410→    findings: List[Finding] = []
   411→   411→    hostname = _hostname()
   412→   412→
   413→   413→    # Check if Docker is running
   414→   414→    code, out = _run("docker ps --format '{{.Names}} {{.Ports}}' 2>/dev/null")
   415→   415→    if code != 0 or not out:
   416→   416→        return findings
   417→   417→
   418→   418→    containers = [l for l in out.splitlines() if l.strip()]
   419→   419→    if containers:
   420→   420→        findings.append(Finding(
   421→   421→            title=f"{len(containers)} Docker container(s) running",
   422→   422→            severity="info", category="docker",
   423→   423→            module="host",
   424→   424→            description=f"Found {len(containers)} running Docker containers.",
   425→   425→            evidence=out[:300],
   426→   426→            asset=hostname, points_deducted=0,
   427→   427→            remediation="",
   428→   428→        ))
   429→   429→
   430→   430→    # Check for containers with host networking
   431→   431→    code, out = _run("docker ps --format '{{.Names}}' -q 2>/dev/null | xargs -I{} docker inspect {} --format '{{.Name}} {{.HostConfig.NetworkMode}}' 2>/dev/null")
   432→   432→    if code == 0 and out:
   433→   433→        for line in out.splitlines():
   434→   434→            if "host" in line.lower():
   435→   435→                findings.append(Finding(
   436→   436→                    title=f"Docker container using host networking: {line.split()[0]}",
   437→   437→                    severity="high", category="docker",
   438→   438→                    module="host",
   439→   439→                    description="Container with host network mode shares the host's network stack, bypassing network isolation.",
   440→   440→                    evidence=line[:200],
   441→   441→                    asset=hostname, points_deducted=8,
   442→   442→                    remediation="Use bridge networking instead of host mode.",
   443→   443→                ))
   444→   444→
   445→   445→    # Check for privileged containers
   446→   446→    code, out = _run("docker ps -q 2>/dev/null | xargs -I{} docker inspect {} --format '{{.Name}} {{.HostConfig.Privileged}}' 2>/dev/null")
   447→   447→    if code == 0 and out:
   448→   448→        for line in out.splitlines():
   449→   449→            if "true" in line.lower():
   450→   450→                findings.append(Finding(
   451→   451→                    title=f"Privileged Docker container: {line.split()[0]}",
   452→   452→                    severity="critical", category="docker",
   453→   453→                    module="host",
   454→   454→                    description="Privileged containers have full access to the host system. This is extremely dangerous.",
   455→   455→                    evidence=line[:200],
   456→   456→                    asset=hostname, points_deducted=15,
   457→   457→                    remediation="Remove --privileged flag. Use specific capabilities instead.",
   458→   458→                ))

   815→   815→def _check_suid_sgid() -> List[Finding]:
   816→   816→    """Find world-writable SUID/SGID binaries (Linux) or auto-elevatable binaries (Windows)."""
   817→   817→    findings: List[Finding] = []

   832→   832→        for d in path_dirs[:30]:
   833→   833→            if not os.path.isdir(d):
   834→   834→                continue

   843→   843→                title=f"{len(found_elevatable)} auto-elevatable binaries found in PATH",

   852→   852→

   873→   873→                    remediation="Remove world-write: sudo chmod o-w <binary> for each listed file.",

  1041→  1041→def _parse_kernel_version(ver: str) -> Optional[Tuple[int, ...]]:
  1042→  1042→    """Parse kernel version string into tuple of ints for comparison."""
  1043→  1043→    # Extract version part, stripping distro suffixes
  1044→  1044→    match = re.match(r'(\d+\.\d+(?:\.\d+)?)', ver)
  1045→  1045→    if not match:
  1046→  1046→        return None
  1047→  1047→    parts = []
  1048→  1048→    for p in match.group(1).split("."):
  1049→  1049→        try:
  1050→  1050→            parts.append(int(p))
  1051→  1051→        except ValueError:
  1052→  1052→            break
  1053→  1053→    return tuple(parts) if parts else None

  1056→  1056→def _version_in_range(ver: Tuple[int, ...], min_v: str, max_v: str) -> bool:
  1057→  1057→    """Check if ver falls within [min_v, max_v] inclusive."""
  1058→  1058→    v_min = _parse_kernel_version(min_v)

  1060→  1060→    if v_min is None or v_max is None:
  1061→  1061→        return False
  1062→  1062→    return v_min <= ver <= v_max
