    20→   119→   368→                and snapshot_a.page_hash != snapshot_b.page_hash):
    21→   120→   369→            events.append(DriftEvent(
    22→   121→   370→                timestamp=ts,
    23→   122→   371→                field_changed="CONTENT_CHANGE",
    24→   123→   372→                old_value=snapshot_a.page_hash,
    25→   124→   373→                new_value=snapshot_b.page_hash,
    26→   125→   374→                severity="low",
    27→   126→   375→                risk_impact="Page content changed",
    28→   127→   376→            ))
    29→   128→   377→
    30→   129→   378→        # Status code change
    31→   130→   379→        if snapshot_a.status_code != snapshot_b.status_code:
    32→   131→   380→            sev = "high" if snapshot_b.status_code >= 500 else "medium"
    33→   132→   381→            events.append(DriftEvent(
    34→   133→   382→                timestamp=ts,
    35→   134→   383→                field_changed="STATUS_CHANGE",
    36→   135→   384→                old_value=str(snapshot_a.status_code),
    37→   136→   385→                new_value=str(snapshot_b.status_code),
    38→   137→   386→                severity=sev,

    40→   139→   388→                    f"HTTP status changed from {snapshot_a.status_code} "
    41→   140→   389→                    f"to {snapshot_b.status_code}"
    42→   141→   390→                ),
    43→   142→   391→            ))
    44→   143→   392→
    45→   144→   393→        # Body length change (significant)
    46→   145→   394→        if snapshot_a.body_length > 0 and snapshot_b.body_length > 0:
    47→   146→   395→            ratio = snapshot_b.body_length / max(snapshot_a.body_length, 1)
    48→   147→   396→            if ratio > 1.5 or ratio < 0.5:
    49→   148→   397→                events.append(DriftEvent(
    50→   149→   398→                    timestamp=ts,
    51→   150→   399→                    field_changed="CONTENT_CHANGE",
    52→   151→   400→                    old_value=str(snapshot_a.body_length),
    53→   152→   401→                    new_value=str(snapshot_b.body_length),
    54→   153→   402→                    severity="medium",
    55→   154→   403→                    risk_impact=(
    56→   155→   404→                        f"Body length changed by {abs(ratio - 1) * 100:.0f}%"
    57→   156→   405→                    ),
    58→   157→   406→                ))
    59→   158→   407→
    60→   159→   408→        return events

    62→   161→   410→    def detect_drift(
    63→   162→   411→        self, target: str, base_url: str, timeout: int = 8
    64→   163→   412→    ) -> Dict[str, Any]:
    65→   164→   413→        """Take a snapshot and compare against the last saved one."""
    66→   165→   414→        current = self.take_snapshot(target, base_url, timeout=timeout)
    67→   166→   415→        self.save_snapshot(current)

   118→   217→   466→    def analyze_drift_trend(self, target: str) -> Dict[str, Any]:
   119→   218→   467→        """Analyze drift patterns over time.
   120→   219→   468→
   121→   220→   469→        Returns trend analysis: increasing, decreasing, cyclical,
   122→   221→   470→        or stable, along with category breakdowns.
   123→   222→   471→        """
   124→   223→   472→        snapshots = self.load_snapshots(target)
   125→   224→   473→        if len(snapshots) < 2:
   126→   225→   474→            return {
   127→   226→   475→                "target": target,
   128→   227→   476→                "trend": "insufficient_data",
   129→   228→   477→                "snapshot_count": len(snapshots),
   130→   229→   478→                "category_breakdown": {},
   131→   230→   479→                "details": [],
   132→   231→   480→            }
   133→   232→   481→
   134→   233→   482→        all_events: List[DriftEvent] = []
   135→   234→   483→        for i in range(len(snapshots) - 1):
   136→   235→   484→            events = self.compare_snapshots(snapshots[i + 1], snapshots[i])
   137→   236→   485→            all_events.extend(events)
   138→   237→   486→
   139→   238→   487→        # Category breakdown
   140→   239→   488→        category_counts: Dict[str, int] = {}
   141→   240→   489→        for ev in all_events:
   142→   241→   490→            cat = ev.field_changed
   143→   242→   491→            category_counts[cat] = category_counts.get(cat, 0) + 1
   144→   243→   492→
   145→   244→   493→        # Group events by snapshot pair to detect trends
   146→   245→   494→        pairs: List[int] = []
   147→   246→   495→        for i in range(len(snapshots) - 1):
   148→   247→   496→            pair_events = self.compare_snapshots(
   149→   248→   497→                snapshots[i + 1], snapshots[i]
   150→   249→   498→            )
   151→   250→   499→            pairs.append(len(pair_events))
   152→   251→   500→
   153→   252→   501→        # Determine trend
   154→   253→   502→        trend = self._determine_trend(pairs)
   155→   254→   503→
   156→   255→   504→        # Time between changes
   157→   256→   505→        change_intervals: List[str] = []
   158→   257→   506→        for i in range(len(snapshots) - 1):
   159→   258→   507→            events = self.compare_snapshots(snapshots[i + 1], snapshots[i])
   160→   259→   508→            if events:
   161→   260→   509→                try:
   162→   261→   510→                    ts_a = datetime.fromisoformat(snapshots[i + 1].timestamp)
   163→   262→   511→                    ts_b = datetime.fromisoformat(snapshots[i].timestamp)
   164→   263→   512→                    delta = abs((ts_b - ts_a).total_seconds())
   165→   264→   513→                    if delta < 3600:
   166→   265→   514→                        interval = f"{int(delta // 60)} minutes"
   167→   266→   515→                    elif delta < 86400:
   168→   267→   516→                        interval = f"{delta / 3600:.1f} hours"
   169→   268→   517→                    else:
   170→   269→   518→                        interval = f"{delta / 86400:.1f} days"
   171→   270→   519→                    change_intervals.append(interval)
   172→   271→   520→                except (ValueError, TypeError):
   173→   272→   521→                    pass
   174→   273→   522→
   175→   274→   523→        return {
   176→   275→   524→            "target": target,
   177→   276→   525→            "trend": trend,
   178→   277→   526→            "snapshot_count": len(snapshots),
   179→   278→   527→            "total_drift_events": len(all_events),
   180→   279→   528→            "category_breakdown": category_counts,
   181→   280→   529→            "events_per_snapshot_pair": pairs,
   182→   281→   530→            "change_intervals": change_intervals,
   183→   282→   531→            "risk_trend": self._risk_trend(snapshots),
   184→   283→   532→            "details": [e.to_dict() for e in all_events],
   185→   284→   533→        }

   307→   406→   655→                            v for item in cert.get("subjectAltName", [])

   463→   562→   811→    @staticmethod
   464→   563→   812→    def _determine_trend(pairs: List[int]) -> str:
   465→   564→   813→        """Classify the drift trend from a list of event counts."""
   466→   565→   814→        if len(pairs) < 2:
   467→   566→   815→            return "stable"
   468→   567→   816→
   469→   568→   817→        first_half = pairs[: len(pairs) // 2]
   470→   569→   818→        second_half = pairs[len(pairs) // 2 :]
   471→   570→   819→        avg_first = sum(first_half) / len(first_half)
   472→   571→   820→        avg_second = sum(second_half) / len(second_half)
   473→   572→   821→
   474→   573→   822→        if avg_first == 0 and avg_second == 0:
   475→   574→   823→            return "stable"
   476→   575→   824→        if avg_first == 0:
   477→   576→   825→            return "increasing"
   478→   577→   826→
   479→   578→   827→        ratio = avg_second / avg_first
   480→   579→   828→        if ratio > 1.5:
   481→   580→   829→            return "increasing"
   482→   581→   830→        elif ratio < 0.5:
   483→   582→   831→            return "decreasing"
   484→   583→   832→        else:
   485→   584→   833→            # Check for cyclical pattern (alternating high/low)
   486→   585→   834→            if len(pairs) >= 4:
   487→   586→   835→                alternating = 0
   488→   587→   836→                for i in range(1, len(pairs)):
   489→   588→   837→                    if pairs[i] == pairs[i - 1]:
   490→   589→   838→                        continue
   491→   590→   839→                    if i == 1:
   492→   591→   840→                        alternating += 1
   493→   592→   841→                        continue
   494→   593→   842→                    curr_up = pairs[i] > pairs[i - 1]
   495→   594→   843→                    prev_up = pairs[i - 1] > pairs[i - 2]
   496→   595→   844→                    if prev_up != curr_up:
   497→   596→   845→                        alternating += 1
   498→   597→   846→                if alternating > len(pairs) * 0.6:
   499→   598→   847→                    return "cyclical"
   500→   599→   848→            return "stable"

   502→   601→   850→    def _risk_trend(self, snapshots: List[DriftSnapshot]) -> str:
   503→   602→   851→        """Determine if overall risk is trending up or down."""
   504→   603→   852→        if len(snapshots) < 3:
   505→   604→   853→            return "insufficient_data"
   506→   605→   854→
   507→   606→   855→        scores: List[float] = []
   508→   607→   856→        for i in range(len(snapshots) - 1):
   509→   608→   857→            events = self.compare_snapshots(snapshots[i + 1], snapshots[i])
   510→   609→   858→            scores.append(self._scorer.score_drift(events))
   511→   610→   859→
   512→   611→   860→        recent = scores[: max(1, len(scores) // 2)]
   513→   612→   861→        older = scores[max(1, len(scores) // 2) :]
   514→   613→   862→        avg_recent = sum(recent) / len(recent)
   515→   614→   863→        avg_older = sum(older) / len(older) if older else 0.0
   516→   615→   864→
   517→   616→   865→        if avg_recent > avg_older * 1.3:
   518→   617→   866→            return "increasing_risk"
   519→   618→   867→        elif avg_recent < avg_older * 0.7:
   520→   619→   868→            return "decreasing_risk"
   521→   620→   869→        return "stable_risk"
