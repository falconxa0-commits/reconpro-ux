  1388→@dataclass
  1389→class SecretFinding:
  1390→    """A single secret detection finding.
  1391→
  1392→    Attributes:
  1393→        secret_type: Category of the secret (e.g., "aws_key", "private_key_rsa").
  1394→        match: The matched text (redacted for high-severity findings).
  1395→        line: Line number where found (1-based), or 0 if not line-based.
  1396→        offset: Character offset within the line.
  1397→        severity: "critical", "high", "medium", or "low".
  1398→        description: Human-readable description of what was found.
  1399→        source: Where the content came from (e.g., filename, "string_input").
  1400→    """
  1401→    secret_type: str
  1402→    match: str
  1403→    line: int
  1404→    offset: int
  1405→    severity: str
  1406→    description: str
  1407→    source: str = "unknown"
  1408→
  1409→
  1410→class SecretsManager:
  1411→    """Enhanced secret detection and management.
  1412→
  1413→    Builds on top of security.detect_secrets_in_text() with:
  1414→    - Additional patterns (Slack, Google, Stripe, Twilio, SSH, PGP, etc.)
  1415→    - File scanning capability
  1416→    - Structured severity reporting
  1417→    - Summary statistics
  1418→
  1419→    Usage::
  1420→
  1421→        sm = SecretsManager()
  1422→        findings = sm.scan_content("some text with AKIAIOSFODNN7EXAMPLE key")
  1423→        report = sm.get_report()
  1424→    """
  1425→
  1426→    SEVERITY_ORDER: Dict[str, int] = {
  1427→        "critical": 0,
  1428→        "high": 1,
  1429→        "medium": 2,
  1430→        "low": 3,
  1431→    }
  1432→
  1433→    def __init__(self, audit_logger: Optional[SecurityAuditLogger] = None) -> None:
  1434→        self._findings: List[SecretFinding] = []
  1435→        self._audit = audit_logger
  1436→        self._scan_count = 0
  1437→        self._files_scanned = 0
  1438→        self._lock = threading.Lock()
  1439→
  1440→    def scan_for_secrets(
  1441→        self,
  1442→        content: str,
  1443→        source: str = "string_input",
  1444→    ) -> List[Dict[str, Any]]:
  1445→        """Scan content for secrets using both built-in and extended patterns.
  1446→
  1447→        This method first uses the existing detect_secrets_in_text() from
  1448→        security.py, then applies additional patterns from this module.
  1449→
  1450→        Args:
  1451→            content: Text content to scan.
  1452→            source: Identifier for where the content came from.
  1453→
  1454→        Returns:
  1455→            List of dicts with keys:
  1456→            - secret_type: str
  1457→            - match: str (redacted for critical/high)
  1458→            - line: int
  1459→            - offset: int
  1460→            - severity: str
  1461→            - description: str
  1462→            - source: str
  1463→            - detector: str ("core" or "extended")
  1464→        """
  1465→        if not content:
  1466→            return []
  1467→
  1468→        self._scan_count += 1
  1469→        results: List[Dict[str, Any]] = []
  1470→        lines = content.split("\n")
  1471→
  1472→        # Phase 1: Use existing core detection from security.py
  1473→        core_findings = detect_secrets_in_text(content)
  1474→        for finding in core_findings:
  1475→            redacted = self._redact_match(finding["match"], finding["confidence"])
  1476→            results.append({
  1477→                "secret_type": finding["type"],
  1478→                "match": redacted,
  1479→                "line": finding["line"],
  1480→                "offset": finding["offset"],
  1481→                "severity": finding["confidence"],
  1482→                "description": self._describe_finding(finding["type"]),
  1483→                "source": source,
  1484→                "detector": "core",
  1485→            })
  1486→
  1487→        # Phase 2: Extended patterns
  1488→        for secret_type, pattern, severity, description in _EXTENDED_SECRET_PATTERNS:
  1489→            for line_num, line in enumerate(lines, start=1):
  1490→                for match in pattern.finditer(line):
  1491→                    matched_text = match.group(0)
  1492→                    redacted = self._redact_match(matched_text, severity)
  1493→                    results.append({
  1494→                        "secret_type": secret_type,
  1495→                        "match": redacted,
  1496→                        "line": line_num,
  1497→                        "offset": match.start(),
  1498→                        "severity": severity,
  1499→                        "description": description,
  1500→                        "source": source,
  1501→                        "detector": "extended",
  1502→                    })
  1503→
  1504→        # Store findings
  1505→        with self._lock:
  1506→            for r in results:
  1507→                self._findings.append(SecretFinding(
  1508→                    secret_type=r["secret_type"],
  1509→                    match=r["match"],
  1510→                    line=r["line"],
  1511→                    offset=r["offset"],
  1512→                    severity=r["severity"],
  1513→                    description=r["description"],
  1514→                    source=source,
  1515→                ))
  1516→
  1517→        # Audit log
  1518→        if results and self._audit:
  1519→            severity_counts: Dict[str, int] = {}
  1520→            for r in results:
  1521→                s = r["severity"]
  1522→                severity_counts[s] = severity_counts.get(s, 0) + 1
  1523→            self._audit.warn(
  1524→                "SECRETS_DETECTED",
  1525→                target=source,
  1526→                details={
  1527→                    "total": len(results),
  1528→                    "severity_counts": severity_counts,
  1529→                    "types": list(set(r["secret_type"] for r in results)),
  1530→                },
  1531→            )
  1532→
  1533→        return results
  1534→
  1535→    def scan_file(self, filepath: str) -> List[Dict[str, Any]]:
  1536→        """Scan a file for secrets.
  1537→
  1538→        Args:
  1539→            filepath: Path to the file to scan.
  1540→
  1541→        Returns:
  1542→            Same format as scan_for_secrets().
  1543→        """
  1544→        if not os.path.isfile(filepath):
  1545→            return []
  1546→
  1547→        try:
  1548→            with open(filepath, "r", encoding="utf-8", errors="ignore") as f:
  1549→                content = f.read()
  1550→        except OSError as exc:
  1551→            logger.warning("Cannot read file for secret scanning: %s: %s", filepath, exc)
  1552→            return []
  1553→
  1554→        self._files_scanned += 1
  1555→        return self.scan_for_secrets(content, source=filepath)
  1556→
  1557→    def scan_directory(
  1558→        self,
  1559→        dirpath: str,
  1560→        extensions: Optional[Set[str]] = None,
  1561→        max_files: int = 1000,
  1562→    ) -> List[Dict[str, Any]]:
  1563→        """Recursively scan a directory for secrets.
  1564→
  1565→        Args:
  1566→            dirpath: Root directory to scan.
  1567→            extensions: Set of file extensions to scan (e.g., {".py", ".js"}).
  1568→                      If None, scans all files.
  1569→            max_files: Maximum number of files to scan.
  1570→
  1571→        Returns:
  1572→            Combined list of all findings.
  1573→        """
  1574→        all_results: List[Dict[str, Any]] = []
  1575→        files_scanned = 0
  1576→
  1577→        if not os.path.isdir(dirpath):
  1578→            return []
  1579→
  1580→        for root, _dirs, files in os.walk(dirpath):
  1581→            if files_scanned >= max_files:
  1582→                break
  1583→            for fname in sorted(files):
  1584→                if files_scanned >= max_files:
  1585→                    break
  1586→                # Skip hidden files and common non-text files
  1587→                if fname.startswith("."):
  1588→                    continue
  1589→                if fname.endswith((".pyc", ".pyo", ".so", ".dll", ".exe",
  1590→                                  ".png", ".jpg", ".jpeg", ".gif", ".ico",
  1591→                                  ".zip", ".tar", ".gz", ".bz2", ".xz")):
  1592→                    continue
  1593→
  1594→                filepath = os.path.join(root, fname)
  1595→                if extensions is not None:
  1596→                    _, ext = os.path.splitext(fname)
  1597→                    if ext.lower() not in extensions:
  1598→                        continue
  1599→
  1600→                results = self.scan_file(filepath)
  1601→                all_results.extend(results)
  1602→                files_scanned += 1
  1603→
  1604→        return all_results

  1606→    def get_report(self) -> Dict[str, Any]:
  1607→        """Generate a comprehensive secrets scanning report.
  1608→
  1609→        Returns:
  1610→            Dict with keys:
  1611→            - total_findings: int
  1612→            - by_severity: dict mapping severity -> count
  1613→            - by_type: dict mapping secret_type -> count
  1614→            - by_source: dict mapping source -> count
  1615→            - by_detector: dict mapping detector -> count
  1616→            - critical_findings: list of critical/high severity findings
  1617→            - scan_stats: scanning statistics
  1618→        """
  1619→        with self._lock:
  1620→            findings = list(self._findings)
  1621→
  1622→        by_severity: Dict[str, int] = {}
  1623→        by_type: Dict[str, int] = {}
  1624→        by_source: Dict[str, int] = {}
  1625→        by_detector: Dict[str, int] = {}
  1626→        critical_findings: List[Dict[str, Any]] = []
  1627→
  1628→        for f in findings:
  1629→            by_severity[f.severity] = by_severity.get(f.severity, 0) + 1
  1630→            by_type[f.secret_type] = by_type.get(f.secret_type, 0) + 1
  1631→            by_source[f.source] = by_source.get(f.source, 0) + 1
  1632→            _CORE_TYPES = {
  1633→                "aws_key", "github_token", "github_token_fine_grained",
  1634→                "generic_api_key", "private_key_rsa", "private_key_ec",
  1635→                "private_key_generic", "password_in_url", "db_connection_string",
  1636→                "jwt_token",
  1637→            }
  1638→            detector = "core" if f.secret_type in _CORE_TYPES else "extended"
  1639→            by_detector[detector] = by_detector.get(detector, 0) + 1
  1640→
  1641→            if f.severity in ("critical", "high"):
  1642→                critical_findings.append({
  1643→                    "type": f.secret_type,
  1644→                    "line": f.line,
  1645→                    "severity": f.severity,
  1646→                    "description": f.description,
  1647→                    "source": f.source,
  1648→                })
  1649→
  1650→        return {
  1651→            "total_findings": len(findings),
  1652→            "by_severity": by_severity,
  1653→            "by_type": by_type,
  1654→            "by_source": by_source,
  1655→            "by_detector": by_detector,
  1656→            "critical_findings": critical_findings,
  1657→            "scan_stats": {
  1658→                "content_scans": self._scan_count,
  1659→                "files_scanned": self._files_scanned,
  1660→            },
  1661→            "generated_at": datetime.now(timezone.utc).isoformat(),
  1662→        }
  1663→

  1703→# 4. TAMPER-EVIDENT SECURITY AUDIT LOGGER
  1704→# ══════════════════════════════════════════════════════════════════════════════

  1706→
  1707→class TamperEvidenceLogger:
  1708→    """Tamper-evident security event logger.
  1709→
  1710→    Extends the concept of SecurityAuditLogger with hash-chained entries
  1711→    that make it detectable if log entries have been added, removed, or
  1712→    modified.
  1713→
  1714→    Each log entry includes:
  1715→    - sequence number (monotonically increasing)
  1716→    - timestamp (ISO 8601 UTC)
  1717→    - level (INFO, WARN, ALERT)
  1718→    - event_type
  1719→    - module
  1720→    - target
  1721→    - details (dict)
  1722→    - previous_hash: SHA-256 of the previous entry's canonical form
  1723→    - entry_hash: SHA-256 of this entry (including previous_hash)
  1724→
  1725→    The chain can be verified at any time to detect tampering.
  1726→    """
  1727→
  1728→    def __init__(
  1729→        self,
  1730→        log_dir: Optional[str] = None,
  1731→        module_name: str = "security_hardening",
  1732→        chain_file: Optional[str] = None,
  1733→    ) -> None:
  1734→        self.module_name = module_name
  1735→        if log_dir is None:
  1736→            log_dir = os.path.join(os.path.expanduser("~"), ".reconpro")
  1737→        self.log_dir = log_dir
  1738→        os.makedirs(log_dir, exist_ok=True)
  1739→
  1740→        if chain_file is None:
  1741→            chain_file = os.path.join(log_dir, "security_chain.log")
  1742→        self.chain_file = chain_file
  1743→
  1744→        self._entries: List[Dict[str, Any]] = []
  1745→        self._last_hash: str = "GENESIS"  # Seed for the chain
  1746→        self._sequence: int = 0
  1747→        self._lock = threading.Lock()
  1748→        self._event_counts: Dict[str, int] = {}
  1749→        self._level_counts: Dict[str, int] = {"INFO": 0, "WARN": 0, "ALERT": 0}
  1750→
  1751→        # Try to load existing chain
  1752→        self._load_chain()
  1753→
  1754→    def _load_chain(self) -> None:
  1755→        """Load existing chain from disk to continue the hash sequence."""
  1756→        if not os.path.isfile(self.chain_file):
  1757→            return
  1758→        try:
  1759→            with open(self.chain_file, "r", encoding="utf-8") as f:
  1760→                for line in f:
  1761→                    line = line.strip()
  1762→                    if not line:
  1763→                        continue
  1764→                    try:
  1765→                        entry = json.loads(line)
  1766→                        self._entries.append(entry)
  1767→                        self._sequence = entry.get("sequence", 0)
  1768→                        self._last_hash = entry.get("entry_hash", "GENESIS")
  1769→                        level = entry.get("level", "INFO")
  1770→                        if level in self._level_counts:
  1771→                            self._level_counts[level] += 1
  1772→                        event_type = entry.get("event_type", "")
  1773→                        self._event_counts[event_type] = self._event_counts.get(event_type, 0) + 1
  1774→                    except (json.JSONDecodeError, KeyError):
  1775→                        continue
  1776→        except OSError:
  1777→            pass
  1778→
  1779→    def _compute_hash(self, canonical: str) -> str:
  1780→        """Compute SHA-256 hash of a canonical string."""
  1781→        return hashlib.sha256(canonical.encode("utf-8")).hexdigest()
  1782→
  1783→    def _make_canonical(self, entry: Dict[str, Any]) -> str:
  1784→        """Create a canonical string representation for hashing.
  1785→
  1786→        The canonical form includes all fields in a deterministic order.
  1787→        """
  1788→        # Build a deterministic representation
  1789→        parts = [
  1790→            str(entry.get("sequence", "")),
  1791→            str(entry.get("timestamp", "")),
  1792→            str(entry.get("level", "")),
  1793→            str(entry.get("event_type", "")),
  1794→            str(entry.get("module", "")),
  1795→            sanitize_log(str(entry.get("target", ""))),
  1796→            str(entry.get("previous_hash", "")),
  1797→            json.dumps(entry.get("details", {}), sort_keys=True, default=str),
  1798→        ]
  1799→        return "|".join(parts)
  1800→
  1801→    def log(
  1802→        self,
  1803→        level: str,
  1804→        event_type: str,
  1805→        target: str = "",
  1806→        details: Optional[Dict[str, Any]] = None,
  1807→    ) -> Dict[str, Any]:
  1808→        """Append a tamper-evident log entry.
  1809→
  1810→        Args:
  1811→            level: "INFO", "WARN", or "ALERT".
  1812→            event_type: Category of security event.
  1813→            target: Target associated with the event.
  1814→            details: Optional dict of additional details.
  1815→
  1816→        Returns:
  1817→            The complete log entry dict including hashes.
  1818→        """
  1819→        if level not in ("INFO", "WARN", "ALERT"):
  1820→            level = "INFO"
  1821→
  1822→        with self._lock:
  1823→            self._sequence += 1
  1824→            entry: Dict[str, Any] = {
  1825→                "sequence": self._sequence,
  1826→                "timestamp": datetime.now(timezone.utc).isoformat(),
  1827→                "level": level,
  1828→                "event_type": event_type,
  1829→                "module": self.module_name,
  1830→                "target": sanitize_log(target),
  1831→                "details": details or {},
  1832→                "previous_hash": self._last_hash,
  1833→            }
  1834→
  1835→            # Compute hash of this entry
  1836→            canonical = self._make_canonical(entry)
  1837→            entry["entry_hash"] = self._compute_hash(canonical)
  1838→
  1839→            self._last_hash = entry["entry_hash"]
  1840→            self._entries.append(entry)
  1841→
  1842→            # Update statistics
  1843→            self._level_counts[level] = self._level_counts.get(level, 0) + 1
  1844→            self._event_counts[event_type] = self._event_counts.get(event_type, 0) + 1
  1845→
  1846→            # Persist to disk
  1847→            self._persist_entry(entry)
  1848→
  1849→            return dict(entry)
  1850→
  1851→    def _persist_entry(self, entry: Dict[str, Any]) -> None:
  1852→        """Append a single entry to the chain file on disk."""
  1853→        try:
  1854→            with open(self.chain_file, "a", encoding="utf-8") as f:
  1855→                f.write(json.dumps(entry, default=str) + "\n")
  1856→        except OSError as exc:
  1857→            logger.error("Failed to persist security log entry: %s", exc)

  1859→    def verify_chain(self) -> Dict[str, Any]:
  1860→        """Verify the integrity of the entire log chain.
  1861→
  1862→        Checks that:
  1863→        1. Sequence numbers are monotonically increasing.
  1864→        2. Each entry's hash matches its content.
  1865→        3. Each entry's previous_hash matches the prior entry's hash.
  1866→
  1867→        Returns:
  1868→            Dict with keys:
  1869→            - valid: bool (True if chain is intact)
  1870→            - entries_checked: int
  1871→            - first_violation: dict describing the first issue found (if any)
  1872→            - violations: list of all violation descriptions
  1873→        """
  1874→        violations: List[str] = []
  1875→        first_violation: Optional[Dict[str, Any]] = None
  1876→        prev_hash = "GENESIS"
  1877→        expected_sequence = 1
  1878→
  1879→        for i, entry in enumerate(self._entries):
  1880→            entry_issues: List[str] = []
  1881→
  1882→            # Check sequence
  1883→            if entry.get("sequence") != expected_sequence:
  1884→                issue = (
  1885→                    f"Entry {i}: expected sequence {expected_sequence}, "
  1886→                    f"got {entry.get('sequence')}"
  1887→                )
  1888→                entry_issues.append(issue)
  1889→
  1890→            # Check previous_hash
  1891→            if entry.get("previous_hash") != prev_hash:
  1892→                issue = (
  1893→                    f"Entry {i}: previous_hash mismatch. "
  1894→                    f"Expected {prev_hash[:16]}..., got {str(entry.get('previous_hash', ''))[:16]}..."
  1895→                )
  1896→                entry_issues.append(issue)
  1897→
  1898→            # Check entry_hash
  1899→            entry_copy = {k: v for k, v in entry.items() if k != "entry_hash"}
  1900→            canonical = self._make_canonical(entry_copy)
  1901→            computed_hash = self._compute_hash(canonical)
  1902→            if entry.get("entry_hash") != computed_hash:
  1903→                issue = (
  1904→                    f"Entry {i}: hash mismatch. "
  1905→                    f"Stored {str(entry.get('entry_hash', ''))[:16]}..., "
  1906→                    f"computed {computed_hash[:16]}..."
  1907→                )
  1908→                entry_issues.append(issue)
  1909→
  1910→            if entry_issues:
  1911→                if first_violation is None:
  1912→                    first_violation = {
  1913→                        "entry_index": i,
  1914→                        "sequence": entry.get("sequence"),
  1915→                        "issues": entry_issues,
  1916→                    }
  1917→                violations.extend(entry_issues)
  1918→
  1919→            prev_hash = entry.get("entry_hash", "")
  1920→            expected_sequence += 1
  1921→
  1922→        return {
  1923→            "valid": len(violations) == 0,
  1924→            "entries_checked": len(self._entries),
  1925→            "first_violation": first_violation,
  1926→            "violations": violations,
  1927→            "verified_at": datetime.now(timezone.utc).isoformat(),
  1928→        }
  1929→
  1930→    def get_statistics(self) -> Dict[str, Any]:
  1931→        """Return statistics about logged security events.
  1932→
  1933→        Returns:
  1934→            Dict with keys:
  1935→            - total_entries: int
  1936→            - by_level: dict mapping level -> count
  1937→            - by_event_type: dict mapping event_type -> count
  1938→            - chain_valid: bool
  1939→            - first_entry: ISO timestamp of first entry
  1940→            - last_entry: ISO timestamp of last entry
  1941→        """
  1942→        first_ts = (
  1943→            self._entries[0]["timestamp"]
  1944→            if self._entries
  1945→            else None
  1946→        )
  1947→        last_ts = (
  1948→            self._entries[-1]["timestamp"]
  1949→            if self._entries
  1950→            else None
  1951→        )
  1952→
  1953→        return {
  1954→            "total_entries": len(self._entries),
  1955→            "by_level": dict(self._level_counts),
  1956→            "by_event_type": dict(self._event_counts),
  1957→            "chain_valid": self.verify_chain()["valid"],
  1958→            "first_entry": first_ts,
  1959→            "last_entry": last_ts,
  1960→        }

  1962→    def get_entries(
  1963→        self,
  1964→        level: Optional[str] = None,
  1965→        event_type: Optional[str] = None,
  1966→        limit: int = 100,
  1967→        offset: int = 0,
  1968→    ) -> List[Dict[str, Any]]:
  1969→        """Query log entries with optional filters.
  1970→
  1971→        Args:
  1972→            level: Filter by level (INFO, WARN, ALERT).
  1973→            event_type: Filter by event type.
  1974→            limit: Maximum entries to return.
  1975→            offset: Skip this many entries from the start.
  1976→
  1977→        Returns:
  1978→            List of matching log entry dicts.
  1979→        """
  1980→        filtered: List[Dict[str, Any]] = []
  1981→        for entry in self._entries:
  1982→            if level and entry.get("level") != level:
  1983→                continue
  1984→            if event_type and entry.get("event_type") != event_type:
  1985→                continue
  1986→            filtered.append(dict(entry))
  1987→
  1988→        return filtered[offset:offset + limit]
