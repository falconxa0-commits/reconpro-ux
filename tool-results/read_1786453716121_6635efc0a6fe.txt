    97→# Data Model
    98→# ═══════════════════════════════════════════════════════════════════════════

   100→
   101→@dataclass
   102→class Recommendation:
   103→    """A single engineering recommendation.
   104→
   105→    Attributes:
   106→        id:              Unique SHA-256 based identifier.
   107→        title:           Short human-readable title.
   108→        description:     Detailed explanation of the recommendation.
   109→        severity:        One of critical/high/medium/low/info.
   110→        category:        Engineering category (diagnostics, testing, etc.).
   111→        effort_estimate: How much work to address (trivial/small/medium/large/epic).
   112→        affected_files:  List of file paths or module names affected.
   113→        suggested_fix:   Concrete action to take.
   114→        evidence:        Why this recommendation was generated.
   115→        confidence:      0.0 – 1.0 confidence in the recommendation.
   116→        created_at:      ISO-8601 UTC timestamp of creation.
   117→        dismissed:       Whether the user has dismissed this recommendation.
   118→        dismissed_reason: Optional reason for dismissal.
   119→        priority_score:  Computed score (severity * 10 + confidence * 10 - effort * 2).
   120→        source:          Which input generated this (diagnostics, drift, etc.).
   121→    """
   122→
   123→    id: str = ""
   124→    title: str = ""
   125→    description: str = ""
   126→    severity: str = "medium"
   127→    category: str = "engineering"
   128→    effort_estimate: str = "small"
   129→    affected_files: List[str] = field(default_factory=list)
   130→    suggested_fix: str = ""
   131→    evidence: str = ""
   132→    confidence: float = 0.7
   133→    created_at: str = ""
   134→    dismissed: bool = False
   135→    dismissed_reason: str = ""
   136→    priority_score: float = 0.0
   137→    source: str = ""
   138→
   139→    def __post_init__(self) -> None:
   140→        """Derive the ID if not provided and normalise fields."""
   141→        if not self.created_at:
   142→            self.created_at = datetime.now(timezone.utc).isoformat()
   143→        if not self.id:
   144→            self.id = self._generate_id()
   145→        # Normalise severity
   146→        self.severity = self.severity.lower()
   147→        if self.severity not in VALID_SEVERITIES:
   148→            self.severity = "medium"
   149→        # Normalise category
   150→        self.category = self.category.lower()
   151→        # Clamp confidence
   152→        self.confidence = max(0.0, min(1.0, self.confidence))
   153→        # Clamp effort
   154→        self.effort_estimate = self.effort_estimate.lower()
   155→        if self.effort_estimate not in VALID_EFFORTS:
   156→            self.effort_estimate = "small"
   157→        # Compute priority score
   158→        self.priority_score = self._compute_priority()
   159→
   160→    def _generate_id(self) -> str:
   161→        """Generate a deterministic SHA-256 ID from title + category + source."""
   162→        payload = f"{self.title}|{self.category}|{self.source}|{self.affected_files}"
   163→        return hashlib.sha256(payload.encode("utf-8")).hexdigest()[:16]
   164→
   165→    def _compute_priority(self) -> float:
   166→        """Compute priority score: higher = more urgent.
   167→
   168→        Formula: severity_weight * 10 + confidence * 10 - effort_weight * 2
   169→        Range approximately: -14 to +108
   170→        """
   171→        sev = _SEVERITY_SCORE.get(self.severity, 5.0)
   172→        eff = _EFFORT_SCORE.get(self.effort_estimate, 3.0)
   173→        return round(sev * 10.0 + self.confidence * 10.0 - eff * 2.0, 2)
   174→
   175→    def to_dict(self) -> Dict[str, Any]:
   176→        """Serialise to a plain dict suitable for JSON persistence."""
   177→        return {
   178→            "id": self.id,
   179→            "title": self.title,
   180→            "description": self.description,
   181→            "severity": self.severity,
   182→            "category": self.category,
   183→            "effort_estimate": self.effort_estimate,
   184→            "affected_files": list(self.affected_files),
   185→            "suggested_fix": self.suggested_fix,
   186→            "evidence": self.evidence,
   187→            "confidence": self.confidence,
   188→            "created_at": self.created_at,
   189→            "dismissed": self.dismissed,
   190→            "dismissed_reason": self.dismissed_reason,
   191→            "priority_score": self.priority_score,
   192→            "source": self.source,
   193→        }
   194→
   195→    @classmethod
   196→    def from_dict(cls, data: Dict[str, Any]) -> Recommendation:
   197→        """Deserialise from a dict (typically loaded from JSON)."""
   198→        return cls(
   199→            id=data.get("id", ""),
   200→            title=data.get("title", ""),
   201→            description=data.get("description", ""),
   202→            severity=data.get("severity", "medium"),
   203→            category=data.get("category", "engineering"),
   204→            effort_estimate=data.get("effort_estimate", "small"),
   205→            affected_files=data.get("affected_files", []),
   206→            suggested_fix=data.get("suggested_fix", ""),
   207→            evidence=data.get("evidence", ""),
   208→            confidence=data.get("confidence", 0.7),
   209→            created_at=data.get("created_at", ""),
   210→            dismissed=data.get("dismissed", False),
   211→            dismissed_reason=data.get("dismissed_reason", ""),
   212→            priority_score=data.get("priority_score", 0.0),
   213→            source=data.get("source", ""),
   214→        )

   217→# ═══════════════════════════════════════════════════════════════════════════
   218→# RecommendationStore — Persistent JSON Storage
   219→# ═══════════════════════════════════════════════════════════════════════════
   220→
   221→
   222→class RecommendationStore:
   223→    """Persistent JSON-backed storage for recommendations.
   224→
   225→    Stores all recommendations (active and dismissed) in
   226→    ``RECONPRO_HOME/memory/recommendations.json``.
   227→    """
   228→
   229→    def __init__(self, path: Optional[Path] = None) -> None:
   230→        self._path: Path = path or RECOMMENDATIONS_FILE
   231→        self._recommendations: Dict[str, Recommendation] = {}
   232→        self._load()
   233→
   234→    # ── Persistence ─────────────────────────────────────────────────────
   235→
   236→    def _load(self) -> None:
   237→        """Load recommendations from the JSON file."""
   238→        if not self._path.exists():
   239→            self._recommendations = {}
   240→            return
   241→        try:
   242→            with open(self._path, "r", encoding="utf-8") as fh:
   243→                data = json.load(fh)
   244→            if not isinstance(data, dict):
   245→                logger.warning("Recommendation store: expected dict, got %s", type(data).__name__)
   246→                self._recommendations = {}
   247→                return
   248→            for rec_id, rec_data in data.items():
   249→                try:
   250→                    self._recommendations[rec_id] = Recommendation.from_dict(rec_data)
   251→                except Exception as exc:
   252→                    logger.warning("Failed to load recommendation %s: %s", rec_id, exc)
   253→        except (json.JSONDecodeError, OSError) as exc:
   254→            logger.warning("Failed to load recommendation store: %s", exc)
   255→            self._recommendations = {}
   256→
   257→    def _save(self) -> None:
   258→        """Persist recommendations to the JSON file."""
   259→        try:
   260→            self._path.parent.mkdir(parents=True, exist_ok=True)
   261→            data = {rid: rec.to_dict() for rid, rec in self._recommendations.items()}
   262→            with open(self._path, "w", encoding="utf-8") as fh:
   263→                json.dump(data, fh, indent=2, default=str)
   264→        except OSError as exc:
   265→            logger.error("Failed to save recommendation store: %s", exc)
   266→
   267→    # ── CRUD ────────────────────────────────────────────────────────────
   268→
   269→    def add(self, recommendation: Recommendation) -> bool:
   270→        """Add a recommendation. Returns True if it was newly added."""
   271→        if not recommendation.id:
   272→            return False
   273→        # Don't overwrite existing non-dismissed recommendations.
   274→        existing = self._recommendations.get(recommendation.id)
   275→        if existing is not None and not existing.dismissed:
   276→            return False
   277→        self._recommendations[recommendation.id] = recommendation
   278→        self._prune_if_needed()
   279→        self._save()
   280→        return True
   281→
   282→    def get(self, rec_id: str) -> Optional[Recommendation]:
   283→        """Retrieve a recommendation by ID."""
   284→        return self._recommendations.get(rec_id)
   285→
   286→    def get_all(self) -> List[Recommendation]:
   287→        """Return all stored recommendations."""
   288→        return list(self._recommendations.values())
   289→
   290→    def get_active(self) -> List[Recommendation]:
   291→        """Return only non-dismissed recommendations, sorted by priority desc."""
   292→        active = [r for r in self._recommendations.values() if not r.dismissed]
   293→        active.sort(key=lambda r: r.priority_score, reverse=True)
   294→        return active
   295→
   296→    def get_dismissed(self) -> List[Recommendation]:
   297→        """Return only dismissed recommendations."""
   298→        return [r for r in self._recommendations.values() if r.dismissed]
   299→
   300→    def dismiss(self, rec_id: str, reason: str = "") -> bool:
   301→        """Mark a recommendation as dismissed."""
   302→        rec = self._recommendations.get(rec_id)
   303→        if rec is None:
   304→            return False
   305→        rec.dismissed = True
   306→        rec.dismissed_reason = reason
   307→        self._save()
   308→        return True
   309→
   310→    def restore(self, rec_id: str) -> bool:
   311→        """Un-dismiss a recommendation."""
   312→        rec = self._recommendations.get(rec_id)
   313→        if rec is None:
   314→            return False
   315→        rec.dismissed = False
   316→        rec.dismissed_reason = ""
   317→        self._save()
   318→        return True
   319→
   320→    def remove(self, rec_id: str) -> bool:
   321→        """Delete a recommendation entirely from the store."""
   322→        if rec_id in self._recommendations:
   323→            del self._recommendations[rec_id]
   324→            self._save()
   325→            return True
   326→        return False
   327→
   328→    def clear_dismissed(self) -> int:
   329→        """Remove all dismissed recommendations. Returns count removed."""
   330→        to_remove = [rid for rid, r in self._recommendations.items() if r.dismissed]
   331→        for rid in to_remove:
   332→            del self._recommendations[rid]
   333→        if to_remove:
   334→            self._save()
   335→        return len(to_remove)
   336→
   337→    def count(self) -> int:
   338→        """Total number of stored recommendations."""
   339→        return len(self._recommendations)
   340→
   341→    def active_count(self) -> int:
   342→        """Number of non-dismissed recommendations."""
   343→        return sum(1 for r in self._recommendations.values() if not r.dismissed)
   344→
   345→    # ── Filtering ───────────────────────────────────────────────────────
   346→
   347→    def filter_by_severity(self, severity: str) -> List[Recommendation]:
   348→        """Return active recommendations matching a severity level."""
   349→        sev = severity.lower()
   350→        return [r for r in self.get_active() if r.severity == sev]
   351→
   352→    def filter_by_category(self, category: str) -> List[Recommendation]:
   353→        """Return active recommendations matching a category."""
   354→        cat = category.lower()
   355→        return [r for r in self.get_active() if r.category == cat]
   356→
   357→    def filter_by_source(self, source: str) -> List[Recommendation]:
   358→        """Return active recommendations from a specific source."""
   359→        return [r for r in self.get_active() if r.source == source]
   360→
   361→    def search(self, query: str) -> List[Recommendation]:
   362→        """Full-text search across title, description, and evidence."""
   363→        q = query.lower()
   364→        results: List[Recommendation] = []
   365→        for r in self.get_active():
   366→            if (q in r.title.lower()
   367→                    or q in r.description.lower()
   368→                    or q in r.evidence.lower()
   369→                    or q in r.suggested_fix.lower()):
   370→                results.append(r)
   371→        return results
   372→
   373→    # ── Pruning ─────────────────────────────────────────────────────────
   374→
   375→    def _prune_if_needed(self) -> None:
   376→        """Remove oldest dismissed entries when store exceeds _MAX_STORED."""
   377→        if len(self._recommendations) <= _MAX_STORED:
   378→            return
   379→        # Sort dismissed by created_at ascending, remove oldest first.
   380→        dismissed = [
   381→            (rid, r) for rid, r in self._recommendations.items() if r.dismissed
   382→        ]
   383→        dismissed.sort(key=lambda pair: pair[1].created_at)
   384→        to_remove = len(self._recommendations) - _MAX_STORED
   385→        for i in range(min(to_remove, len(dismissed))):
   386→            del self._recommendations[dismissed[i][0]]
   387→
   388→
   389→# ═══════════════════════════════════════════════════════════════════════════
   390→# RecommendationEngine — Core Rule-Based Generation
   391→# ═══════════════════════════════════════════════════════════════════════════
   392→
   393→
   394→class RecommendationEngine:
   395→    """Core rule-based recommendation generation engine.
   396→
   397→    Responsibilities:
   398→    - Rule-based recommendation generation (if X then recommend Y)
   399→    - Priority scoring based on severity + effort + impact
   400→    - Deduplication of similar recommendations via Jaccard similarity
   401→    - Aging: old recommendations get escalated severity
   402→    """
   403→
   404→    def __init__(self, store: Optional[RecommendationStore] = None) -> None:
   405→        self.store: RecommendationStore = store or RecommendationStore()
   406→
   407→    # ── Jaccard Similarity for Deduplication ────────────────────────────
   408→
   409→    @staticmethod
   410→    def _tokenise(text: str) -> Set[str]:
   411→        """Split text into a set of lowercase word tokens."""
   412→        return set(re.findall(r"\b\w+\b", text.lower()))
   413→
   414→    def _jaccard(self, a: str, b: str) -> float:
   415→        """Compute Jaccard similarity between two strings."""
   416→        tokens_a = self._tokenise(a)
   417→        tokens_b = self._tokenise(b)
   418→        if not tokens_a or not tokens_b:
   419→            return 0.0
   420→        intersection = tokens_a & tokens_b
   421→        union = tokens_a | tokens_b
   422→        return len(intersection) / len(union)
   423→
   424→    def _is_duplicate(
   425→        self, candidate: Recommendation, existing: List[Recommendation]
   426→    ) -> bool:
   427→        """Check if *candidate* duplicates any *existing* recommendation."""
   428→        for ex in existing:
   429→            # Same category is required for a duplicate match.
   430→            if ex.category != candidate.category:
   431→                continue
   432→            if ex.dismissed:
   433→                continue
   434→            # Compare by title similarity and affected files overlap.
   435→            title_sim = self._jaccard(candidate.title, ex.title)
   436→            if title_sim >= _DEDUP_THRESHOLD:
   437→                return True
   438→            # Also check if they affect the same files with similar titles.
   439→            if (candidate.affected_files and ex.affected_files
   440→                    and set(candidate.affected_files) & set(ex.affected_files)
   441→                    and title_sim >= 0.4):
   442→                return True
   443→        return False
   444→
   445→    # ── Aging / Escalation ──────────────────────────────────────────────
   446→
   447→    @staticmethod
   448→    def _age_recommendation(rec: Recommendation) -> Recommendation:
   449→        """Escalate severity of old recommendations.
   450→
   451→        - After _STALE_DAYS: boost severity by one level.
   452→        - After _VERY_STALE_DAYS: boost severity by two levels.
   453→        Returns a *new* Recommendation with updated fields.
   454→        """
   455→        now = datetime.now(timezone.utc)
   456→        try:
   457→            created = datetime.fromisoformat(rec.created_at)
   458→            if created.tzinfo is None:
   459→                created = created.replace(tzinfo=timezone.utc)
   460→        except (ValueError, TypeError):
   461→            return rec
   462→
   463→        age_days = (now - created).total_seconds() / 86400.0
   464→
   465→        if age_days >= _VERY_STALE_DAYS:
   466→            new_sev = RecommendationEngine._escalate_severity(rec.severity, steps=2)
   467→        elif age_days >= _STALE_DAYS:
   468→            new_sev = RecommendationEngine._escalate_severity(rec.severity, steps=1)
   469→        else:
   470→            return rec
   471→
   472→        if new_sev == rec.severity:
   473→            return rec
   474→
   475→        # Build a new Recommendation with the escalated severity.
   476→        updated = Recommendation(
   477→            id=rec.id,
   478→            title=rec.title,
   479→            description=rec.description,
   480→            severity=new_sev,
   481→            category=rec.category,
   482→            effort_estimate=rec.effort_estimate,
   483→            affected_files=rec.affected_files,
   484→            suggested_fix=rec.suggested_fix,
   485→            evidence=rec.evidence,
   486→            confidence=rec.confidence,
   487→            created_at=rec.created_at,
   488→            dismissed=rec.dismissed,
   489→            dismissed_reason=rec.dismissed_reason,
   490→            source=rec.source,
   491→        )
   492→        return updated
   493→
   494→    @staticmethod
   495→    def _escalate_severity(current: str, steps: int = 1) -> str:
   496→        """Move severity *steps* levels higher (more severe).
   497→
   498→        Order: info -> low -> medium -> high -> critical
   499→        """
   500→        order = ["info", "low", "medium", "high", "critical"]
   501→        current = current.lower()
   502→        try:
   503→            idx = order.index(current)
   504→        except ValueError:
   505→            return current
   506→        new_idx = min(idx + steps, len(order) - 1)
   507→        return order[new_idx]
   508→
   509→    # ── Add with Dedup + Persist ────────────────────────────────────────
   510→
   511→    def _add_recommendations(self, recommendations: List[Recommendation]) -> int:
   512→        """Add recommendations to the store, skipping duplicates.
   513→
   514→        Returns the number of newly added recommendations.
   515→        """
   516→        active = self.store.get_active()
   517→        # Also age existing recommendations while we're here.
   518→        for i, rec in enumerate(active):
   519→            aged = self._age_recommendation(rec)
   520→            if aged.id != rec.id or aged.severity != rec.severity:
   521→                self.store._recommendations[aged.id] = aged
   522→
   523→        added = 0
   524→        for rec in recommendations:
   525→            if not self._is_duplicate(rec, active):
   526→                if self.store.add(rec):
   527→                    added += 1
   528→                    active.append(rec)  # Update active list for intra-batch dedup
   529→        return added

   531→    # ═══════════════════════════════════════════════════════════════════
   532→    # Recommendation Generation Rules
   533→    # ═══════════════════════════════════════════════════════════════════
   534→
   535→    def generate_from_diagnostics(self, diag_results: Dict[str, Any]) -> List[Recommendation]:
   536→        """Generate engineering recommendations from diagnostic check results.
   537→
   538→        Expected input shape (from diagnostics.run_diagnostics):
   539→            {
   540→                "timestamp": str,
   541→                "total_checks": int,
   542→                "passed": int,
   543→                "failed": int,
   544→                "checks": [
   545→                    {"label": str, "status": "ok"|"error", "value": Any, "error": str}
   546→                ]
   547→            }
   548→        """
   549→        recommendations: List[Recommendation] = []
   550→        if not diag_results or "checks" not in diag_results:
   551→            return recommendations
   552→
   553→        checks = diag_results.get("checks", [])
   554→        failed_checks = [c for c in checks if c.get("status") == "error"]
   555→
   556→        if not failed_checks:
   557→            return recommendations
   558→
   559→        # ── Broken module registry ──────────────────────────────────────
   560→        registry_check = next(
   561→            (c for c in failed_checks if c.get("label") == "module_registry"), None
   562→        )
   563→        if registry_check:
   564→            value = registry_check.get("value", {})
   565→            broken = value.get("broken", []) if isinstance(value, dict) else []
   566→            recommendations.append(Recommendation(

  1021→    def generate_from_test_results(self, test_results: Dict[str, Any]) -> List[Recommendation]:
  1022→        """Generate engineering recommendations from test suite results.
  1023→
  1024→        Expected input shape:

  1028→                "failed": int,

  1059→        if pass_rate < 0.8 and total >= 5:

  1079→            ))
  1080→

  1082→        if errors > 0:

  1092→                effort_estimate="medium",
  1093→                affected_files=[

  1095→                    if f.get("file") and "error" in f.get("error_message", "").lower()
  1096→                ],
  1097→                suggested_fix=(

  1104→            ))

  1106→        # ── High skip rate ──────────────────────────────────────────────

  1185→    def generate_from_performance(self, perf_data: Dict[str, Any]) -> List[Recommendation]:

  1192→                    "module_name": {"duration_s": float, "findings_count": int}

  1194→                "memory_peak_mb": float,
  1195→                "request_count": int,
  1196→                "error_count": int,
  1197→                "timeout_count": int,

  1201→        if not perf_data:
  1202→            return recommendations
  1203→

  1205→        module_timings = perf_data.get("module_timings", {})

  1213→            error_rate = error_count / request_count

  1224→                affected_files=[],

  1226→                    "Check if the target is rate-limiting requests. Consider increasing "
  1227→                    "the timeout, reducing concurrency, or adding retry logic with "
  1228→                    "exponential backoff."
  1229→                ),

  1237→            timeout_rate = timeout_count / request_count

  1239→                title=f"High request timeout rate: {timeout_rate:.1%}",
  1240→                description=(
  1241→                    f"{timeout_count} out of {request_count} requests timed out "
  1242→                    f"({timeout_rate:.1%}). This suggests the target or network is "
  1243→                    f"slow to respond, or the timeout is set too aggressively."
  1244→                ),
  1245→                severity="medium",
  1246→                category="performance",
  1247→                effort_estimate="small",
  1248→                affected_files=[],
  1249→                suggested_fix=(
  1250→                    "Increase the request timeout from the current value. For slow "
  1251→                    f"targets, consider a timeout of 15-30s. Also check if the "
  1252→                    f"network path has high latency."
  1253→                ),
  1254→                evidence=f"timeouts={timeout_count}, requests={request_count}",
  1255→                confidence=0.85,
  1256→                source="performance",
  1257→            ))
  1258→
  1259→        # ── Slow modules ────────────────────────────────────────────────
  1260→        if module_timings:
  1261→            slow_modules: List[Tuple[str, float]] = []
  1262→            for mod_name, mod_data in module_timings.items():
  1263→                dur = mod_data.get("duration_s", 0.0) if isinstance(mod_data, dict) else 0.0
  1264→                if dur > 10.0:
  1265→                    slow_modules.append((mod_name, dur))
  1266→
  1267→            if slow_modules:
  1268→                slow_modules.sort(key=lambda x: x[1], reverse=True)
  1269→                for mod_name, dur in slow_modules[:3]:  # Top 3 slowest
  1270→                    findings = 0
  1271→                    if isinstance(module_timings.get(mod_name), dict):
  1272→                        findings = module_timings[mod_name].get("findings_count", 0)
  1273→                    efficiency = "low" if findings < 5 and dur > 30.0 else "moderate"
  1274→                    recommendations.append(Recommendation(

  1276→                        description=(
  1277→                            f"Module '{mod_name}' took {dur:.1f}s and produced "

  1279→                            f"This module dominates scan time."
  1280→                        ),
  1281→                        severity="medium" if efficiency == "low" else "low",
  1282→                        category="performance",
  1283→                        effort_estimate="medium",
  1284→                        affected_files=[f"module:{mod_name}"],
  1285→                        suggested_fix=(
  1286→                            f"Profile '{mod_name}' to identify bottlenecks. Consider: "
  1287→                            f"(1) reducing the scope of checks, (2) parallelising "
  1288→                            f"independent operations, (3) caching repeated lookups."
  1289→                        ),
  1290→                        evidence=f"module={mod_name}, duration={dur:.1f}s, findings={findings}",
  1291→                        confidence=0.8,
  1292→                        source="performance",
  1293→                    ))
  1294→
  1295→        # ── High memory usage ───────────────────────────────────────────
  1296→        if memory_peak > 500.0:
  1297→            recommendations.append(Recommendation(
  1298→                title=f"High memory usage: {memory_peak:.0f}MB peak",
  1299→                description=(
  1300→                    f"Peak memory usage reached {memory_peak:.0f}MB during the scan. "
  1301→                    f"High memory usage can cause OOM kills on constrained systems "

  1305→                category="performance",
  1306→                effort_estimate="large",
  1307→                affected_files=[],
  1308→                suggested_fix=(
  1309→                    "Profile memory usage to identify the largest consumers. Common "
  1310→                    "fixes: (1) stream large responses instead of buffering, "
  1311→                    "(2) limit the number of concurrent requests, (3) release "

  1313→                ),
  1314→                evidence=f"memory_peak_mb={memory_peak:.0f}",
  1315→                confidence=0.8,
  1316→                source="performance",
  1317→            ))
  1318→
  1319→        # ── Overall scan duration ───────────────────────────────────────
  1320→        if total_duration > 300.0:
  1321→            recommendations.append(Recommendation(
  1322→                title=f"Long scan duration: {total_duration:.0f}s ({total_duration/60:.1f}min)",
  1323→                description=(
  1324→                    f"The total scan took {total_duration:.0f}s. Long scan times "

  1327→                severity="medium",
  1328→                category="performance",
  1329→                effort_estimate="medium",
  1330→                affected_files=[],
  1331→                suggested_fix=(
  1332→                    "Consider: (1) running only essential modules, (2) increasing "

  1336→                evidence=f"scan_duration_s={total_duration:.0f}",
  1337→                confidence=0.75,
  1338→                source="performance",
  1339→            ))
  1340→
  1341→        return recommendations

  1343→    def generate_from_security(self, sec_findings: Dict[str, Any]) -> List[Recommendation]:

  1400→                    f"Low validation rates suggest detection logic may be producing "

  1402→                ),
  1403→                severity="medium",
  1404→                category="code_quality",
  1405→                effort_estimate="medium",
  1406→                affected_files=[],
  1407→                suggested_fix=(

  1412→                evidence=f"cross_validation_rate={cv_rate:.1%}",

  1414→                source="security",
  1415→            ))
  1416→
  1417→        # ── Limited module coverage ─────────────────────────────────────
  1418→        if len(modules_run) < 5:

  1435→                confidence=0.9,

  1439→        # ── No critical/high findings but low coverage ──────────────────

  1458→                ),

  1463→
  1464→        # ── Many info/low findings with no actionable ones ──────────────
  1465→        info_count = by_severity.get("info", 0)
  1466→        low_count = by_severity.get("low", 0)

  1479→                suggested_fix=(

  1484→                evidence=(
