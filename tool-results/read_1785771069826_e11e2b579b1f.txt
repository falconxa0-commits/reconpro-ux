    99→
   100→class FindingDetailModal(ModalScreen):
   101→    """Modal overlay showing full finding details."""
   102→
   103→    def __init__(self, finding: Dict[str, Any]) -> None:
   104→        super().__init__()
   105→        self.finding = finding
   106→
   107→    def compose(self) -> ComposeResult:
   108→        f = self.finding
   109→        sev = f.get("severity", "info").lower()
   110→        sev_color = SEV_STYLES.get(sev, "white")
   111→        title = f.get("title", "Unknown")
   112→        category = f.get("category", "")
   113→        module = f.get("module", "")
   114→        pts = f.get("points_deducted", 0)
   115→        evidence = f.get("evidence", "")
   116→        remediation = f.get("remediation", "No remediation available.")
   117→        description = f.get("description", "")
   118→        asset = f.get("asset", "")
   119→        related_cves = f.get("related_cves", [])
   120→
   121→        lines: List[str] = []
   122→        lines.append(f"[{sev_color}][{sev.upper()}] {title}[/]")
   123→        lines.append("")
   124→        if asset:
   125→            lines.append(f"[dim]Asset:[/] [cyan]{asset}[/]")
   126→        if category:
   127→            lines.append(f"[dim]Category:[/] {category}")
   128→        if module:
   129→            lines.append(f"[dim]Module:[/] {module}")
   130→        lines.append(f"[dim]Points:[/] -{pts}")
   131→        if related_cves:
   132→            cve_ids = ", ".join(
   133→                c.get("cve_id", "") for c in related_cves[:5] if c.get("cve_id")
   134→            )
   135→            if cve_ids:
   136→                lines.append(f"[dim]CVEs:[/] [red]{cve_ids}[/]")
   137→        lines.append("")
   138→        if description:
   139→            lines.append(f"[bold]Description:[/]")
   140→            lines.append(f"[dim]{description}[/]")
   141→            lines.append("")
   142→        if evidence:
   143→            lines.append(f"[bold]Evidence:[/]")
   144→            lines.append(f"[dim]{evidence}[/]")
   145→            lines.append("")
   146→        lines.append(f"[bold]{GREEN}Remediation:[/]")
   147→        lines.append(f"{GREEN}{remediation}[/]")
   148→
   149→        content = "\n".join(lines)
   150→
   151→        with Vertical(classes="modal-container"):
   152→            yield Label(content, classes="modal-content")
   153→            with Horizontal(classes="modal-actions"):
   154→                yield Button("Close", variant="primary", id="close-modal")
   155→
   156→    def on_button_pressed(self, event: Button.Pressed) -> None:
   157→        if event.button.id == "close-modal":
   158→            self.dismiss()
   159→

   161→# ════════════════════════════════════════════════════════════════════════════════
   162→# Boot Screen Widget
   163→# ════════════════════════════════════════════════════════════════════════════════
   164→
   165→
   166→class BootScreen(Widget):
   167→    """Full-screen animated boot sequence."""
   168→
   169→    def __init__(self) -> None:
   170→        super().__init__()
   171→        self._step = 0
   172→        self._spinner_idx = 0
   173→        self._module_idx = 0
   174→        self._module_names = list(MODULE_NAMES.values())
   175→        self._done = False
   176→        self._timer: Optional[Timer] = None
   177→
   178→    def compose(self) -> ComposeResult:
   179→        with Vertical(classes="boot-container"):
   180→            yield Static("", id="boot-line-1", classes="boot-title")
   181→            yield Static("", id="boot-line-2", classes="boot-line")
   182→            yield Static("", id="boot-line-3", classes="boot-line")
   183→            yield Static("", id="boot-line-4", classes="boot-line")
   184→
   185→    def on_mount(self) -> None:
   186→        self._timer = self.set_interval(0.04, self._tick)
   187→
   188→    def _tick(self) -> None:
   189→        app = self.app
   190→        if not isinstance(app, NexusApp):
   191→            return
   192→
   193→        line1 = app.query_one("#boot-line-1", Static)
   194→        line2 = app.query_one("#boot-line-2", Static)
   195→        line3 = app.query_one("#boot-line-3", Static)
   196→        line4 = app.query_one("#boot-line-4", Static)
   197→
   198→        if self._step == 0:
   199→            # Letter-by-letter title
   200→            title = "RECONPRO NEXUS"
   201→            chars = title[: self._spinner_idx]
   202→            self._spinner_idx += 1
   203→            line1.update(f"[{CYAN} bold]{chars}[/]")
   204→            if self._spinner_idx > len(title):
   205→                self._spinner_idx = 0
   206→                self._step = 1
   207→                line1.update(f"[{CYAN} bold]{title}[/]  [{DIM_CYAN}]v{VERSION}[/]")
   208→
   209→        elif self._step == 1:
   210→            # Spinner for "Initializing agent..."
   211→            frame = SPINNER_FRAMES[self._spinner_idx % len(SPINNER_FRAMES)]
   212→            self._spinner_idx += 1
   213→            line2.update(f"  [{CYAN}]{frame}[/] [{DIM_CYAN}]Initializing agent...[/]")
   214→            if self._spinner_idx > 30:
   215→                self._spinner_idx = 0
   216→                self._step = 2
   217→                line2.update(f"  [{GREEN}]✓[/] [{DIM_CYAN}]Agent initialized.[/]")
   218→
   219→        elif self._step == 2:
   220→            # Module names appearing one by one
   221→            if self._module_idx < len(self._module_names):
   222→                visible = self._module_names[: self._module_idx + 1]
   223→                self._module_idx += 1
   224→                modules_str = "  ".join(f"[{CYAN}]○[/] [{DIM_CYAN}]{m}[/]" for m in visible)
   225→                remaining = len(self._module_names) - self._module_idx
   226→                if remaining > 0:
   227→                    modules_str += f"  [{DIM_CYAN}]+{remaining} more...[/]"
   228→                frame = SPINNER_FRAMES[self._spinner_idx % len(SPINNER_FRAMES)]
   229→                self._spinner_idx += 1
   230→                line3.update(
   231→                    f"  [{CYAN}]{frame}[/] [{DIM_CYAN}]Loading {len(self._module_names)} modules...[/]\n  {modules_str}"
   232→                )
   233→            else:
   234→                all_mods = "  ".join(
   235→                    f"[{GREEN}]●[/] [{DIM_CYAN}]{m}[/]" for m in self._module_names
   236→                )
   237→                line3.update(
   238→                    f"  [{GREEN}]✓[/] [{DIM_CYAN}]All {len(self._module_names)} modules loaded.[/]\n  {all_mods}"
   239→                )
   240→                self._step = 3
   241→                self._spinner_idx = 0
   242→
   243→        elif self._step == 3:
   244→            # "Ready." with flash
   245→            if self._spinner_idx == 0:
   246→                line4.update(f"[{YELLOW} bold]  ▶  READY.[/]")
   247→            elif self._spinner_idx == 6:
   248→                line4.update(f"[{GREEN} bold]  ▶  READY.[/]")
   249→            elif self._spinner_idx >= 10:
   250→                self._done = True
   251→                if self._timer:
   252→                    self._timer.stop()
   253→                self.app.call_from_thread(self.app._finish_boot)
   254→            self._spinner_idx += 1
   255→

   257→# ════════════════════════════════════════════════════════════════════════════════
   258→# Module Status Cell Widget
   259→# ════════════════════════════════════════════════════════════════════════════════
   260→
   261→
   262→class ModuleCell(Static):
   263→    """Single module status cell in the grid."""
   264→
   265→    def __init__(self, module_id: str, name: str) -> None:
   266→        super().__init__()
   267→        self.module_id = module_id
   268→        self.module_name = name
   269→        self.status = "idle"  # idle | scanning | done | error
   270→        self.finding_count = 0
   271→        self._spinner_idx = 0
   272→        self._timer: Optional[Timer] = None
   273→
   274→    def on_mount(self) -> None:
   275→        self._update_display()
   276→        self._timer = self.set_interval(0.1, self._spin)
   277→
   278→    def _spin(self) -> None:
   279→        if self.status == "scanning":
   280→            self._spinner_idx = (self._spinner_idx + 1) % len(SPINNER_FRAMES)
   281→            self._update_display()
   282→
   283→    def _update_display(self) -> None:
   284→        name_short = self.module_name[:14]
   285→        if self.status == "idle":
   286→            icon = f"[{DIM_CYAN}]○[/]"
   287→            status_text = f"[{DIM_CYAN}]idle[/]"
   288→        elif self.status == "scanning":
   289→            frame = SPINNER_FRAMES[self._spinner_idx]
   290→            icon = f"[{CYAN}]{frame}[/]"
   291→            status_text = f"[{CYAN}]scanning[/]"
   292→        elif self.status == "done":
   293→            icon = f"[{GREEN}]✓[/]"
   294→            status_text = f"[{GREEN}]done ({self.finding_count})[/]"
   295→        elif self.status == "error":
   296→            icon = f"[{RED}]✗[/]"
   297→            status_text = f"[{RED}]error[/]"
   298→        else:
   299→            icon = "○"
   300→            status_text = "idle"
   301→
   302→        self.update(
   303→            f"{icon} [{CYAN} bold]{name_short}[/]\n  {status_text}"
   304→        )
   305→
   306→    def set_status(self, status: str, count: int = 0) -> None:
   307→        self.status = status
   308→        self.finding_count = count
   309→        self._update_display()
   310→
   311→    def on_unmount(self) -> None:
   312→        if self._timer:
   313→            self._timer.stop()
   314→

   316→# ════════════════════════════════════════════════════════════════════════════════
   317→# Pulsing Status Dot Widget
   318→# ════════════════════════════════════════════════════════════════════════════════
   319→
   320→
   321→class StatusDot(Static):
   322→    """Pulsing dot indicator for scan status."""
   323→
   324→    scanning: reactive[bool] = reactive(False)
   325→
   326→    def __init__(self) -> None:
   327→        super().__init__("")
   328→        self._phase = 0
   329→
   330→    def on_mount(self) -> None:
   331→        self.set_interval(0.3, self._pulse)
   332→
   333→    def watch_scanning(self, old: bool, new: bool) -> None:
   334→        # Update visual state based on scanning status
   335→        if new:
   336→            self.update(f"[{self._get_pulse_color()}]●[/]")
   337→        else:
   338→            self.update(f"[{GREEN}]●[/]")
   339→
   340→    def _get_pulse_color(self) -> str:
   341→        return ["#003322", "#006644", CYAN, "#006644"][self._phase] if self.scanning else GREEN
   342→
   343→    def _pulse(self) -> None:
   344→        self._phase = (self._phase + 1) % 4
   345→        if self.scanning:
   346→            brightness = self._get_pulse_color()
   347→            self.update(f"[{brightness}]●[/]")
   348→        else:
   349→            self.update(f"[{GREEN}]●[/]")

   841→    def on_key(self, event: Key) -> None:
   842→        """Handle key events — boot skip + completer + history + vim-nav."""
   843→        # Skip boot on ANY key press during boot sequence
   844→        if not self._boot_done and not self._boot_skipped:
   845→            self._boot_skipped = True
   846→            try:
   847→                boot = self.query_one("#boot-screen", BootScreen)
   848→                if boot._timer:
   849→                    boot._timer.stop()
   850→            except NoMatches:
   851→                pass
   852→            self._finish_boot()
   853→            return
   854→
   855→        # ── Number keys: quick jump to panels (0=input, 1=chat, 2=findings, 3=modules) ──
   856→        if event.key in ("0", "1", "2", "3") and not self._input_focused():
   857→            jump_map = {"0": "#command-input", "1": "#chat-log", "2": "#findings-feed", "3": "#module-grid"}
   858→            # Skip panel jump to collapsed panels
   859→            target_sel = jump_map[event.key]
   860→            if event.key == "1" and self._left_collapsed:
   861→                target_sel = "#findings-feed"
   862→            if event.key == "3" and self._modules_collapsed:
   863→                target_sel = "#findings-feed"
   864→            self._focus_widget(target_sel)
   865→            event.stop()
   866→            return
   867→
   868→        # ── Phase D: [ and ] to toggle panel collapse ──
   869→        if event.key == "[" and not self._input_focused():
   870→            self._toggle_left_panel()
   871→            event.stop()
   872→            return
   873→        if event.key == "]" and not self._input_focused():
   874→            self._toggle_right_panel()
   875→            event.stop()
   876→            return
   877→        if event.key == "=" and not self._input_focused():
   878→            self._toggle_modules()
   879→            event.stop()
   880→            return
   881→
   882→        # ── Tab: completer accept/cycle / quick-pick on empty / cycle focus ──
   883→        if event.key == "tab":
   884→            try:
   885→                inp = self.query_one("#command-input", Input)
   886→                comp = self.query_one("#cmd-completer", CommandCompleter)
   887→                if inp.has_focus:
   888→                    # Completer visible → accept top suggestion
   889→                    if comp.visible:
   890→                        accepted = comp.accept_top()
   891→                        if accepted:
   892→                            self._apply_completion(inp, comp, accepted)
   893→                        event.stop()
   894→                        return
   895→                    # Empty input → quick-pick menu
   896→                    if not inp.value.strip():
   897→                        comp.show_quick_pick()
   898→                        event.stop()
   899→                        return
   900→            except NoMatches:
   901→                pass
   902→            # Fall through to default Tab behavior (cycle focus)
   903→            return
   904→
   905→        # ── Escape: dismiss completer, close modal, or jump to input ──
   906→        if event.key == "escape":
   907→            try:
   908→                comp = self.query_one("#cmd-completer", CommandCompleter)
   909→                if comp.visible:
   910→                    comp.hide()
   911→                    event.stop()
   912→                    return
   913→            except NoMatches:
   914→                pass
   915→            # If any modal screen is open, Escape will dismiss it via Textual
   916→            # Otherwise jump back to input
   917→            if not self._input_focused():
   918→                self._focus_input()
   919→                event.stop()
   920→                return
   921→
   922→        # ── Up/Down in completer: cycle suggestions ──
   923→        if event.key in ("up", "down"):
   924→            try:
   925→                comp = self.query_one("#cmd-completer", CommandCompleter)
   926→                inp = self.query_one("#command-input", Input)
   927→                if comp.visible and inp.has_focus:
   928→                    direction = -1 if event.key == "up" else 1
   929→                    comp.cycle_selection(direction)
   930→                    event.stop()
   931→                    return
   932→            except NoMatches:
   933→                pass
   934→
   935→        # ── j/k vim-style navigation: findings feed + chat log ──
   936→        if event.key in ("j", "k"):
   937→            # Findings feed navigation
   938→            try:
   939→                feed = self.query_one("#findings-feed", RichLog)
   940→                if feed.has_focus and self._findings_list:
   941→                    self._vim_navigate_findings(event.key)
   942→                    event.stop()
   943→                    return
   944→            except NoMatches:
   945→                pass
   946→            # Chat log scroll (j=down, k=up)
   947→            try:
   948→                chat = self.query_one("#chat-log", RichLog)
   949→                if chat.has_focus:
   950→                    if event.key == "j":
   951→                        chat.scroll_relative(3)
   952→                    else:
   953→                        chat.scroll_relative(-3)
   954→                    event.stop()
   955→                    return
   956→            except NoMatches:
   957→                pass
   958→
   959→        # ── Arrow keys on module grid: navigate cells ──
   960→        if event.key in ("up", "down", "left", "right"):
   961→            try:
   962→                grid = self.query_one("#module-grid", Container)
   963→                if grid.has_focus:
   964→                    self._arrow_navigate_modules(event.key)
   965→                    event.stop()
   966→                    return
   967→            except NoMatches:
   968→                pass
   969→
   970→        # ── Enter on findings: open detail modal ──
   971→        if event.key == "enter":
   972→            try:
   973→                feed = self.query_one("#findings-feed", RichLog)
   974→                if feed.has_focus and self._findings_list:
   975→                    idx = self._findings_cursor if 0 <= self._findings_cursor < len(self._findings_list) else len(self._findings_list) - 1
   976→                    self.push_screen(FindingDetailModal(self._findings_list[idx]))
   977→                    event.stop()
   978→                    return
   979→            except NoMatches:
   980→                pass
   981→
   982→        # Command history navigation
   983→        if event.key == "up" and self._history_index > 0:
   984→            try:
   985→                inp = self.query_one("#command-input", Input)
   986→                if inp.has_focus:
   987→                    self._history_index -= 1
   988→                    inp.value = self._command_history[self._history_index]
   989→            except NoMatches:
   990→                pass
   991→        elif event.key == "down":
   992→            try:
   993→                inp = self.query_one("#command-input", Input)
   994→                if inp.has_focus:
   995→                    if self._history_index < len(self._command_history) - 1:
   996→                        self._history_index += 1
   997→                        inp.value = self._command_history[self._history_index]
   998→                    else:
   999→                        self._history_index = len(self._command_history)
  1000→                        inp.value = ""
  1001→            except NoMatches:
  1002→                pass

  1065→    def _apply_completion(self, inp: Input, comp: CommandCompleter, accepted: str) -> None:
  1066→        """Apply a completion suggestion to the input field.
  1067→
  1068→        Phase C: handles both command-level and arg-level completions.
  1069→        For arg mode, appends/replaces the last word instead of the first.
  1070→        """
  1071→        parts = inp.value.split()
  1072→
  1073→        if comp._mode == "command":
  1074→            # Replace the first word with the accepted command
  1075→            rest = parts[1:] if len(parts) > 1 else []
  1076→            inp.value = accepted + (" " + " ".join(rest) if rest else "")
  1077→        else:
  1078→            # Arg-level: replace the last word
  1079→            if len(parts) >= 2:
  1080→                parts[-1] = accepted
  1081→                inp.value = " ".join(parts)
  1082→            else:
  1083→                inp.value = accepted
  1084→
  1085→        inp.cursor_position = len(inp.value)

  1087→    def _vim_navigate_findings(self, direction: str) -> None:

  1097→        else:  # k

  1112→            finding = self._findings_list[self._findings_cursor]
  1113→            sev = finding.get("severity", "info").lower()
  1114→            title = finding.get("title", "?")[:40]

  1124→    def _arrow_navigate_modules(self, direction: str) -> None:
  1125→        """Navigate module cells with arrow keys when module-grid has focus."""
  1126→        module_keys = list(self._module_cells.keys())
  1127→        if not module_keys:
  1128→            return
  1129→
  1130→        # Current focused module index
  1131→        current_idx = 0
  1132→        for i, key in enumerate(module_keys):
  1133→            try:
  1134→                if self._module_cells[key].has_focus:
  1135→                    current_idx = i
  1136→                    break
  1137→            except Exception:
  1138→                pass
  1139→
  1140→        new_idx = current_idx
  1141→        if direction == "right":
  1142→            new_idx = min(current_idx + 1, len(module_keys) - 1)
  1143→        elif direction == "left":
  1144→            new_idx = max(current_idx - 1, 0)
  1145→        elif direction == "down":
  1146→            new_idx = min(current_idx + self._GRID_COLS, len(module_keys) - 1)
  1147→        elif direction == "up":
  1148→            new_idx = max(current_idx - self._GRID_COLS, 0)
  1149→
  1150→        if new_idx != current_idx:
  1151→            key = module_keys[new_idx]
  1152→            self._module_cells[key].focus()
  1153→            # Show module hint
  1154→            mod_name = MODULE_NAMES.get(key, key)
  1155→            try:
  1156→                el = self.query_one("#bottom-info", Label)
  1157→                el.update(f"[{CYAN}]{mod_name}[/{CYAN}] [{MUTED}]{key}[/{MUTED}]")
  1158→            except NoMatches:
  1159→                pass
