    86→class _RateLimiter:
    87→    """Token-bucket rate limiter — safe for concurrent use.
    88→
    89→    Default: 10 requests per second. Use throttle.acquire() before any
    90→    network call; it blocks until a token is available.
    91→    """
    92→
    93→    def __init__(self, max_per_second: float = 10.0):
    94→        self._rate = max_per_second
    95→        self._min_interval = 1.0 / max_per_second
    96→        self._lock = Lock()
    97→        self._last = 0.0
    98→
    99→    def acquire(self) -> None:
   100→        """Block until we're allowed to make the next request."""
   101→        with self._lock:
   102→            now = time.monotonic()
   103→            elapsed = now - self._last
   104→            if elapsed < self._min_interval:
   105→                time.sleep(self._min_interval - elapsed)
   106→            self._last = time.monotonic()
   107→

   122→def classify_ipv6(addr: str) -> str:
   123→    """Classify an IPv6 address. Returns: 'global', 'link_local', 'unique_local',
   124→    'loopback', 'mapped_v4', 'documentation', 'multicast', 'unspecified'."""
   125→    if addr == "::":
   126→        return "unspecified"
   127→    if _IPV6_LOOPBACK_RE.match(addr):
   128→        return "loopback"
   129→    if _IPV6_LINK_LOCAL_RE.match(addr):
   130→        return "link_local"
   131→    if _IPV6_UNIQUE_LOCAL_RE.match(addr):
   132→        return "unique_local"
   133→    if _IPV6_MAPPED_V4_RE.match(addr):
   134→        return "mapped_v4"
   135→    if _IPV6_DOC_RE.match(addr):
   136→        return "documentation"
   137→    if _IPV6_MULTICAST_RE.match(addr):
   138→        return "multicast"
   139→    return "global"

   175→def run_argv(argv: list, timeout: int = 8) -> Tuple[str, str]:
   176→    """Execute a pre-parsed argv list without any shell interpretation."""
   177→    try:
   178→        r = subprocess.run(argv, shell=False, capture_output=True, text=True, timeout=timeout)
   179→        return r.stdout.strip(), r.stderr.strip()
   180→    except Exception:
   181→        return "", ""

   183→
   184→def run(cmd: str, timeout: int = 8) -> Tuple[str, str]:
   185→    """Execute a command string safely via shlex.split — no /bin/sh spawned."""
   186→    if CONFIG.dry_run:
   187→        audit_log("subprocess.skip", detail=f"dry-run: {cmd[:120]}")
   188→        return "", ""
   189→    try:
   190→        argv = shlex.split(cmd)
   191→    except ValueError as e:
   192→        audit_log("subprocess.shlex.error", status=type(e).__name__, detail=str(e))
   193→        return "", ""
   194→    if not argv:
   195→        return "", ""
   196→    return run_argv(argv, timeout=timeout)

   198→def http_probe(url: str, method: str = "GET", body: Optional[bytes] = None,
   199→               headers: Optional[Dict[str, str]] = None, timeout: int = 8) -> Dict[str, Any]:
   200→    """Hardened single HTTP layer — all modules delegate through this."""
   201→    RATE_LIMITER.acquire()  # m5: thread-safe rate limiting
   202→    h = {
   203→        "User-Agent": "ReconPro-Unified/1.0 (Six-Blades-One-Target; +https://reconpro.security)",
   204→        "X-ReconPro-Signature": RECONPRO_SIGNATURE,
   205→        "Accept": "application/json,text/plain,*/*",
   206→    }
   207→    if headers:
   208→        h.update(headers)
   209→    req = urllib.request.Request(url, data=body, method=method, headers=h)
   210→    try:
   211→        ctx = ssl.create_default_context()
   212→        if CONFIG.insecure:
   213→            ctx.check_hostname = False
   214→            ctx.verify_mode = ssl.CERT_NONE
   215→        with urllib.request.urlopen(req, timeout=timeout, context=ctx) as resp:
   216→            raw = resp.read(8192)
   217→            return {
   218→                "ok": True, "status": resp.status, "reason": resp.reason,
   219→                "headers": dict(resp.headers.items()),
   220→                "body": raw.decode("utf-8", errors="replace")[:8192],
   221→            }
   222→    except urllib.error.HTTPError as e:
   223→        try:
   224→            raw = e.read(8192)
   225→            body_str = raw.decode("utf-8", errors="replace")[:8192]
   226→        except Exception:
   227→            body_str = ""
   228→        return {
   229→            "ok": False, "status": e.code, "reason": e.reason,
   230→            "headers": dict(e.headers.items()) if e.headers else {},
   231→            "body": body_str,
   232→        }
   233→    except Exception as e:
   234→        return {"ok": False, "status": 0, "reason": str(e), "headers": {}, "body": ""}

   238→def audit_log(event: str, status: str = "ok", detail: str = "") -> None:
   239→    """Append a JSON Lines record to the audit log. Injection-immune via json.dumps."""
   240→    global _AUDIT_FAIL_WARNED
   241→    record = {
   242→        "ts": datetime.utcnow().isoformat() + "Z",
   243→        "event": event,
   244→        "status": status,
   245→        "detail": detail,
   246→    }
   247→    try:
   248→        os.makedirs(os.path.dirname(AUDIT_LOG_PATH) or ".", exist_ok=True)
   249→        with open(AUDIT_LOG_PATH, "a") as f:
   250→            f.write(json.dumps(record, ensure_ascii=False, separators=(",", ":")) + "\n")
   251→    except Exception:
   252→        if not _AUDIT_FAIL_WARNED:
   253→            _AUDIT_FAIL_WARNED = True
   254→            sys.stderr.write(f"[reconpro] WARNING: audit log write failed for {AUDIT_LOG_PATH}\n")

   412→def _save_credentials(data: Dict[str, str], primary: bool = True) -> bool:
   413→    """Save credentials to ~/.reconpro/config.json (primary) or credentials.json (fallback).
   414→    When primary=True, also writes to credentials.json for backward compat."""
   415→    target = CONFIG_FILE if primary else CREDENTIALS_FILE
   416→    try:
   417→        os.makedirs(CREDENTIALS_DIR, mode=0o700, exist_ok=True)
   418→        fd = os.open(target, os.O_WRONLY | os.O_CREAT | os.O_TRUNC, 0o600)
   419→        with os.fdopen(fd, "w") as f:
   420→            json.dump(data, f, indent=2)
   421→        audit_log("credentials.saved", detail=f"api_key stored at {target}")
   422→        # Write to fallback too if primary
   423→        if primary and target != CREDENTIALS_FILE:
   424→            try:
   425→                fd2 = os.open(CREDENTIALS_FILE, os.O_WRONLY | os.O_CREAT | os.O_TRUNC, 0o600)
   426→                with os.fdopen(fd2, "w") as f2:
   427→                    json.dump(data, f2, indent=2)
   428→            except Exception:
   429→                pass  # best-effort fallback
   430→        return True
   431→    except Exception as e:
   432→        audit_log("credentials.save.error", status=type(e).__name__, detail=str(e)[:120])
   433→        return False

   517→def _sign_report(report: Dict[str, Any]) -> str:
   518→    """HMAC-SHA256 sign a report for tamper-proof telemetry."""
   519→    creds = _load_credentials()
   520→    api_key = creds.get("api_key", "")
   521→    canonical = json.dumps(report, sort_keys=True, separators=(",", ":"), ensure_ascii=False)
   522→    signature = hmac.new(api_key.encode(), canonical.encode(), hashlib.sha256).hexdigest()
   523→    return signature

   526→def _upload_telemetry(report: Dict[str, Any]) -> bool:
   527→    """POST signed report to telemetry endpoint. Returns True on success."""
   528→    creds = _load_credentials()
   529→    if not creds.get("api_key"):
   530→        audit_log("telemetry.skip", detail="no api_key configured")
   531→        return False
   532→    signature = _sign_report(report)
   533→    payload = json.dumps({
   534→        "report": report,
   535→        "signature": signature,
   536→        "api_key_prefix": creds["api_key"][:8],
   537→    }).encode()
   538→    try:
   539→        RATE_LIMITER.acquire()
   540→        req = urllib.request.Request(
   541→            TELEMETRY_ENDPOINT,
   542→            data=payload,
   543→            method="POST",
   544→            headers={
   545→                "Content-Type": "application/json",
   546→                "User-Agent": f"ReconPro-Unified/{RECONPRO_VERSION}",
   547→                "X-ReconPro-Signature": RECONPRO_SIGNATURE,
   548→                "Authorization": f"Bearer {creds['api_key']}",
   549→            },
   550→        )
   551→        ctx = ssl.create_default_context()
   552→        if CONFIG.insecure:
   553→            ctx.check_hostname = False
   554→            ctx.verify_mode = ssl.CERT_NONE
   555→        with urllib.request.urlopen(req, timeout=15, context=ctx) as resp:
   556→            if resp.status < 300:
   557→                audit_log("telemetry.upload.ok", detail=f"status={resp.status}")
   558→                return True
   559→            else:
   560→                audit_log("telemetry.upload.error", detail=f"status={resp.status}")
   561→                return False
   562→    except Exception as e:
   563→        audit_log("telemetry.upload.error", status=type(e).__name__, detail=str(e)[:120])
   564→        return False

  1056→_FILENAME_UNSAFE_RE = re.compile(r'[\x00-\x1f\x7f\x80-\x9f/\\:\n\r\t]')
  1057→_FILENAME_MAX_LEN = 120
  1058→
  1059→def _safe_filename(raw: str) -> str:
  1060→    """Sanitize a user-supplied string for use as a filename component.
  1061→
  1062→    Strips: null bytes, control chars, path separators, backslashes.
  1063→    Truncates to _FILENAME_MAX_LEN. Rejects empty results (returns 'unnamed').
  1064→    """
  1065→    cleaned = _FILENAME_UNSAFE_RE.sub('_', raw)
  1066→    # Collapse consecutive underscores
  1067→    cleaned = re.sub(r'_{2,}', '_', cleaned)
  1068→    cleaned = cleaned.strip('._-')
  1069→    cleaned = cleaned[:_FILENAME_MAX_LEN]
  1070→    if not cleaned or cleaned in ('.', '..'):
  1071→        return 'unnamed'
  1072→    return cleaned

  1074→def _find_cache(prefix: str, host: str) -> Optional[str]:
  1075→    """Find a cached JSON for this host with the given prefix."""
  1076→    if not os.path.isdir(CACHE_DIR):
  1077→        return None
  1078→    safe = _safe_filename(host)
  1079→    candidates = [
  1080→        f"{CACHE_DIR}/{prefix}_{safe}.json",
  1081→        f"{CACHE_DIR}/{prefix}_{safe}_v2.json",
  1082→    ]
  1083→    # also any *{host}*.json
  1084→    for f in sorted(os.listdir(CACHE_DIR), reverse=True):
  1085→        if f.startswith(prefix) and safe in f and f.endswith(".json"):
  1086→            candidates.append(f"{CACHE_DIR}/{f}")
  1087→    for c in candidates:
  1088→        if os.path.exists(c) and os.path.getsize(c) > 100:
  1089→            return c
  1090→    return None

  1504→def _nhi_verify_match(value: str, body: str) -> bool:
  1505→    """Verify an identity match is real, not documentation/placeholder."""
  1506→    if not value:
  1507→        return False
  1508→    placeholders = {"example", "your-", "xxx", "placeholder", "replace", "changeme", "todo", "insert"}
  1509→    val_lower = value.lower()
  1510→    for ph in placeholders:
  1511→        if ph in val_lower:
  1512→            return False
  1513→    # Reject if surrounded by documentation markers
  1514→    try:
  1515→        if re.search(r'<!--.*?-->.*' + re.escape(value), body[:2000]):
  1516→            return False
  1517→        if re.search(r'["\'][\s]*' + re.escape(value) + r'[\s]*["\'].*#.*example', body[:2000], re.I):
  1518→            return False
  1519→    except Exception:
  1520→        pass
  1521→    # Reject if too short
  1522→    if len(value) < 8:
  1523→        return False
  1524→    # Require at least 2 different character classes
  1525→    classes = 0
  1526→    if re.search(r'[a-z]', value):
  1527→        classes += 1
  1528→    if re.search(r'[A-Z]', value):
  1529→        classes += 1
  1530→    if re.search(r'[0-9]', value):
  1531→        classes += 1
  1532→    if re.search(r'[_\-.@/]', value):
  1533→        classes += 1
  1534→    return classes >= 2

  1537→def _nhi_op_severity(op_name: str) -> str:
  1538→    """Map over-permission pattern name to severity."""
  1539→    severity_map = {
  1540→        "wildcard_action": "critical",
  1541→        "wildcard_resource": "critical",
  1542→        "admin_wildcard": "critical",
  1543→        "s3_full_bucket": "high",
  1544→        "iam_passrole": "high",

  1549→def _nhi_compute_risk(identities: list, overpermissions: list, blast_radius: dict) -> int:
  1550→    """Compute 0-100 risk score for NHI findings."""
  1551→    score = 0
  1552→    verified = [i for i in identities if i.get("verified")]
  1553→    score += min(30, len(verified) * 8)
  1554→    critical_ops = [o for o in overpermissions if o.get("severity") == "critical"]
  1555→    high_ops = [o for o in overpermissions if o.get("severity") == "high"]
  1556→    score += min(40, len(critical_ops) * 15 + len(high_ops) * 8)
  1557→    score += min(30, blast_radius.get("max_depth", 0) * 5 + blast_radius.get("reachable_services", 0) * 5)
  1558→    return min(100, score)

  1664→def module_nhi_graph(host: str) -> Dict[str, Any]:
  1665→    """Non-Human Identity & Blast-Radius Mapping.
  1666→
  1667→    Scans target for:
  1668→      1. Cloud identity signals in page source, JS, metadata
  1669→      2. Over-permissioned IAM/role patterns
  1670→      3. Exposed service account credentials
  1671→      4. Graph construction: nodes (Identity, Role, Endpoint, Database)
  1672→         edges (HAS_ACCESS_TO, CAN_ASSUME, EXPOSES_TOKEN)
  1673→      5. Blast-radius: traverse from any vulnerability to connected resources
  1674→      6. Generate Terraform remediation patch for detected wildcards
  1675→    """
  1676→    audit_log("nhi_graph.start", detail=f"target={host}")
  1677→
  1678→    # Paths to probe for identity signals
  1679→    probe_paths = [
  1680→        "/",
  1681→        "/.well-known/",
  1682→        "/robots.txt",
  1683→        "/sitemap.xml",
  1684→        "/api/",
  1685→        "/config.json",
  1686→        "/.env",
  1687→        "/package.json",
  1688→        "/firebase.json",
  1689→        "/.aws/config",
  1690→        "/.github/workflows/",
  1691→    ]
  1692→
  1693→    base = host if host.startswith("http") else f"https://{host}"
  1694→    if not base.endswith("/"):
  1695→        base += "/"
  1696→
  1697→    identities_found = []
  1698→    all_bodies = []  # (path, body_text, resp_dict) pairs for context analysis
  1699→
  1700→    # Phase 1: Fetch and scan for identity patterns
  1701→    for path in probe_paths:
  1702→        url = base + path.lstrip("/")
  1703→        try:
  1704→            resp = http_probe(url)
  1705→        except Exception as e:
  1706→            audit_log("nhi_graph.probe.error", status=type(e).__name__, detail=f"url={url} err={str(e)[:120]}")
  1707→            continue
  1708→        body = resp.get("body", "")
  1709→        if body:
  1710→            all_bodies.append((path, body, resp))
  1711→
  1712→        for pat_name, pat_re in NHI_IDENTITY_PATTERNS:
  1713→            matches = pat_re.findall(body)
  1714→            seen = set()
  1715→            for match in matches:
  1716→                val = match if isinstance(match, str) else (match[0] if match and isinstance(match, tuple) else "")
  1717→                if not val or val in seen:
  1718→                    continue
  1719→                seen.add(val)
  1720→
  1721→                verified = _nhi_verify_match(val, body)
  1722→
  1723→                identities_found.append({
  1724→                    "type": pat_name,
  1725→                    "value": val,
  1726→                    "source": path,
  1727→                    "verified": verified,
  1728→                })
  1729→
  1730→    audit_log("nhi_graph.scan", detail=f"identities_raw={len(identities_found)}")
  1731→
  1732→    # Phase 2: Build identity graph
  1733→    nodes = []
  1734→    edges = []
  1735→    node_id_map = {}  # label -> node_id
  1736→    node_counter = [0]
  1737→
  1738→    def _add_node(label: str, ntype: str) -> str:
  1739→        if label in node_id_map:
  1740→            return node_id_map[label]
  1741→        nid = f"nhi_{node_counter[0]}"
  1742→        node_counter[0] += 1
  1743→        node_id_map[label] = nid
  1744→        nodes.append({"id": nid, "type": ntype, "label": label})
  1745→        return nid
  1746→
  1747→    def _add_edge(src_id: str, tgt_id: str, relation: str):
  1748→        if src_id and tgt_id and src_id != tgt_id:
  1749→            edges.append({"source": src_id, "target": tgt_id, "relation": relation})
  1750→
  1751→    # Target endpoint node
  1752→    target_node = _add_node(host, "Endpoint")
  1753→
  1754→    # Identity nodes
  1755→    for ident in identities_found:
  1756→        id_node = _add_node(ident["value"], "Identity")
  1757→        _add_edge(target_node, id_node, "EXPOSES_TOKEN")
  1758→
  1759→    # Phase 3: Check for over-permission patterns in response bodies
  1760→    overpermissions = []
  1761→    for path, body, resp in all_bodies:
  1762→        for op_name, op_re in NHI_OVERPERMISSION_PATTERNS:
  1763→            op_matches = op_re.findall(body)
  1764→            seen_op = set()
  1765→            for op_val in op_matches:
  1766→                val = op_val if isinstance(op_val, str) else (op_val[0] if op_val and isinstance(op_val, tuple) else "")
  1767→                if not val or val in seen_op:
  1768→                    continue
  1769→                seen_op.add(val)
  1770→
  1771→                linked = [i for i in identities_found if i["source"] == path]
  1772→                linked_strs = [f'{i["type"]}:{i["value"][:30]}' for i in linked[:3]]
  1773→
  1774→                severity = _nhi_op_severity(op_name)
  1775→                overpermissions.append({
  1776→                    "identity": linked_strs[0] if linked_strs else "unknown",
  1777→                    "pattern": op_name,
  1778→                    "severity": severity,
  1779→                    "description": f"Found '{val}' in {path}",
  1780→                    "value": val,
  1781→                })
  1782→
  1783→                op_node = _add_node(f"OVERPERM:{op_name}", "Role")
  1784→                _add_edge(target_node, op_node, "HAS_ACCESS_TO")
  1785→
  1786→    audit_log("nhi_graph.overpermissions", detail=f"count={len(overpermissions)}")
  1787→
  1788→    # Phase 4: Compute blast radius via BFS
  1789→    adj = {}  # adjacency list (undirected)
  1790→    for node in nodes:
  1791→        adj[node["id"]] = set()
  1792→    for edge in edges:
  1793→        if edge["source"] in adj:
  1794→            adj[edge["source"]].add(edge["target"])
  1795→        if edge["target"] in adj:
  1796→            adj[edge["target"]].add(edge["source"])
  1797→
  1798→    max_depth = 0
  1799→    reachable_types = {"Database": set(), "Secret": set(), "Service": set()}
  1800→
  1801→    for ident in identities_found:
  1802→        if ident["value"] not in node_id_map:
  1803→            continue
  1804→        start = node_id_map[ident["value"]]
  1805→        visited = {start}
  1806→        queue = [(start, 0)]
  1807→        while queue:
  1808→            current, depth = queue.pop(0)
  1809→            if depth > max_depth:
  1810→                max_depth = depth
  1811→            for neighbor in adj.get(current, set()):
  1812→                if neighbor not in visited:
  1813→                    visited.add(neighbor)
  1814→                    queue.append((neighbor, depth + 1))
  1815→            for node in nodes:
  1816→                if node["id"] == current:
  1817→                    ntype = node["type"]
  1818→                    if ntype == "Database":
  1819→                        reachable_types["Database"].add(current)
  1820→                    elif ntype == "Identity":
  1821→                        reachable_types["Secret"].add(current)
  1822→                    elif ntype in ("Role", "Service"):
  1823→                        reachable_types["Service"].add(current)
  1824→
  1825→    blast_radius = {
  1826→        "max_depth": max_depth,
  1827→        "reachable_databases": len(reachable_types["Database"]),
  1828→        "reachable_secrets": len(reachable_types["Secret"]),
  1829→        "reachable_services": len(reachable_types["Service"]),
  1830→    }
  1831→
  1832→    # Phase 5: Generate Terraform remediation
  1833→    remediation_tf = _nhi_generate_remediation(overpermissions, identities_found, host)
  1834→
  1835→    # Phase 6: Compute risk score
  1836→    risk_score = _nhi_compute_risk(identities_found, overpermissions, blast_radius)
  1837→
  1838→    result = {
  1839→        "module": "NHI GRAPH",
  1840→        "identities_found": identities_found,
  1841→        "nodes": nodes,
  1842→        "edges": edges,
  1843→        "overpermissions": overpermissions,
  1844→        "blast_radius": blast_radius,
  1845→        "remediation_tf": remediation_tf,
  1846→        "total_identities": len(identities_found),
  1847→        "risk_score": risk_score,
  1848→    }
  1849→
  1850→    audit_log("nhi_graph.complete", detail=f"risk={risk_score} identities={len(identities_found)} ops={len(overpermissions)}")
  1851→    return result
