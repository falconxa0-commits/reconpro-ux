    41→def _b64url_encode(data: bytes) -> str:
    42→    """Base64url-encode bytes (no padding)."""
    43→    return base64.urlsafe_b64encode(data).rstrip(b"=").decode("ascii")
    44→
    45→
    46→def _b64url_decode(s: str) -> bytes:
    47→    """Base64url-decode with padding restoration."""
    48→    padding = 4 - len(s) % 4
    49→    if padding != 4:
    50→        s += "=" * padding
    51→    return base64.urlsafe_b64decode(s)

    54→def _build_jwt(header: Dict[str, Any], payload: Dict[str, Any],
    55→               secret: Optional[bytes] = None) -> str:
    56→    """Build a JWT token. If secret is None, no signature (alg=none)."""
    57→    h = _b64url_encode(json.dumps(header, separators=(",", ":")).encode())
    58→    p = _b64url_encode(json.dumps(payload, separators=(",", ":")).encode())
    59→    signing_input = f"{h}.{p}".encode()
    60→    if secret is None:
    61→        return f"{h}.{p}."
    62→    sig = hmac.new(secret, signing_input, hashlib.sha256).digest()
    63→    return f"{h}.{p}.{_b64url_encode(sig)}"

    66→def _try_decode_jwt_payload(token: str) -> Optional[Dict[str, Any]]:
    67→    """Try to decode JWT payload without verification."""
    68→    try:
    69→        parts = token.split(".")
    70→        if len(parts) != 3:
    71→            return None
    72→        return json.loads(_b64url_decode(parts[1]))
    73→    except Exception:
    74→        return None

    77→def _try_decode_jwt_header(token: str) -> Optional[Dict[str, Any]]:
    78→    """Try to decode JWT header without verification."""
    79→    try:
    80→        parts = token.split(".")
    81→        if len(parts) < 2:
    82→            return None
    83→        return json.loads(_b64url_decode(parts[0]))
    84→    except Exception:
    85→        return None

    88→def _extract_bearer_token(resp: Dict[str, Any]) -> Optional[str]:
    89→    """Extract a JWT Bearer token from response headers or body."""
    90→    # Check WWW-Authenticate / Set-Cookie / response body for tokens
    91→    headers = resp.get("headers", {})
    92→    auth_header = headers.get("www-authenticate", "")
    93→    if auth_header.startswith("Bearer "):
    94→        return auth_header[7:].split(",")[0].strip()  # strip params
    95→    # Check body for jwt patterns
    96→    body = resp.get("body", "")
    97→    import re
    98→    patterns = [
    99→        r'"token"\s*:\s*"(eyJ[A-Za-z0-9_-]+\.eyJ[A-Za-z0-9_-]+\.[A-Za-z0-9_-]+)"',
   100→        r'"access_token"\s*:\s*"(eyJ[A-Za-z0-9_-]+\.eyJ[A-Za-z0-9_-]+\.[A-Za-z0-9_-]+)"',
   101→        r'"id_token"\s*:\s*"(eyJ[A-Za-z0-9_-]+\.eyJ[A-Za-z0-9_-]+\.[A-Za-z0-9_-]+)"',
   102→    ]
   103→    for pat in patterns:
   104→        m = re.search(pat, body)
   105→        if m:
   106→            return m.group(1)
   107→    return None
