   555→        f"Fixed: {len(d['fixed'])} | New: {len(d['new'])} | Persistent: {len(d['persistent'])}",

   561→def _tool_shell_command(command: str = "") -> ToolResult:
   562→    """Run an arbitrary shell command. DANGEROUS — use with caution."""
   563→    if not command:
   564→        return ToolResult("No command specified.", success=False)
   565→    dangerous_patterns = ["rm -rf /", "mkfs", "dd if=", ":(){ :|:& };:", "chmod -R 777 /"]
   566→    for dp in dangerous_patterns:
   567→        if dp in command:
   568→            return ToolResult(f"Blocked dangerous command pattern: {dp}", success=False)
   569→    try:
   570→        result = subprocess.run(
   571→            command, shell=True, capture_output=True, text=True, timeout=30,
   572→        )
   573→        output = (result.stdout + result.stderr)[:2000]
   574→        return ToolResult(
   575→            f"Exit code: {result.returncode}\n{output}",
   576→            success=result.returncode == 0,
   577→        )
   578→    except subprocess.TimeoutExpired:
   579→        return ToolResult("Command timed out (30s).", success=False)
   580→    except Exception as e:
   581→        return ToolResult(f"Command error: {e}", success=False)

   905→
   906→    Computes a base score (0.0–10.0) using the CVSS v3.1 formula
   907→    simplified for rule-based use:

   909→      - Attack Complexity (AC): Low=0.77, High=0.44

   912→      - Impact (C/I/A): None=0.00, Low=0.22, High=0.56
   913→
   914→    BaseScore = roundup(min(Impact + Exploitability, 10))
   915→    Impact = 1 - [(1-C)*(1-I)*(1-A)]
   916→    Exploitability = 8.22 * AV * AC * PR * UI

   993→
   994→    @classmethod
   995→    def score_finding(cls, finding: Dict) -> Dict[str, Any]:
   996→        """Compute CVSS-like score for a single finding.
   997→
   998→        Returns a dict with: base_score, severity, av, ac, pr, ui, c, i, a, vector_string.
   999→        """
  1000→        title = (finding.get("title") or "").lower()
  1001→        desc = (finding.get("description") or "").lower()
  1002→        combined = f"{title} {desc}"
  1003→        original_sev = (finding.get("severity") or "").lower()
  1004→
  1005→        # Find matching profile
  1006→        profile = cls._CATEGORY_PROFILES[-1]  # default
  1007→        for prof in cls._CATEGORY_PROFILES:
  1008→            if prof["patterns"] and any(p in combined for p in prof["patterns"]):
  1009→                profile = prof
  1010→                break
  1011→
  1012→        # Override metrics based on original severity if it's explicitly critical
  1013→        av = cls._AV[profile["av"]]
  1014→        ac = cls._AC[profile["ac"]]
  1015→        pr = cls._PR[profile["pr"]]
  1016→        ui = cls._UI[profile["ui"]]
  1017→        c = cls._CIA[profile["c"]]
  1018→        i = cls._CIA[profile["i"]]
  1019→        a = cls._CIA[profile["a"]]
  1020→
  1021→        # If the original severity is critical, boost impact to High
  1022→        if original_sev == "critical":
  1023→            c = max(c, cls._CIA["H"])
  1024→            i = max(i, cls._CIA["H"])
  1025→            a = max(a, cls._CIA["H"])
  1026→        elif original_sev == "high":
  1027→            c = max(c, cls._CIA["L"])
  1028→            i = max(i, cls._CIA["L"])
  1029→            a = max(a, cls._CIA["L"])
  1030→
  1031→        # CVSS base score calculation
  1032→        iss = 1.0 - ((1.0 - c) * (1.0 - i) * (1.0 - a))
  1033→        impact = 6.42 * iss
  1034→        exploitability = 8.22 * av * ac * pr * ui
  1035→
  1036→        if impact <= 0:
  1037→            base_score = 0.0
  1038→        else:
  1039→            base_score = min(impact + exploitability, 10.0)
  1040→
  1041→        # Round up to nearest 0.1
  1042→        base_score = math.ceil(base_score * 10) / 10
  1043→
  1044→        # Derive severity from score
  1045→        if base_score >= 9.0:
  1046→            derived_sev = "critical"
  1047→        elif base_score >= 7.0:
  1048→            derived_sev = "high"
  1049→        elif base_score >= 4.0:
  1050→            derived_sev = "medium"
  1051→        elif base_score > 0.0:
  1052→            derived_sev = "low"
  1053→        else:
  1054→            derived_sev = "info"
  1055→
  1056→        # Use the more severe of original vs derived
  1057→        severity_order = {"critical": 0, "high": 1, "medium": 2, "low": 3, "info": 4}
  1058→        final_sev = original_sev if severity_order.get(original_sev, 99) <= severity_order.get(derived_sev, 99) else derived_sev
  1059→
  1060→        vector = f"CVSS:3.1/AV:{profile['av']}/AC:{profile['ac']}/PR:{profile['pr']}/UI:{profile['ui']}/C:{cls._cia_label(c)}/I:{cls._cia_label(i)}/A:{cls._cia_label(a)}"
  1061→
  1062→        return {
  1063→            "base_score": base_score,
  1064→            "severity": final_sev,
  1065→            "cvss_vector": vector,
  1066→            "metrics": {
  1067→                "av": profile["av"], "ac": profile["ac"], "pr": profile["pr"],
  1068→                "ui": profile["ui"], "c": cls._cia_label(c), "i": cls._cia_label(i),
  1069→                "a": cls._cia_label(a),
  1070→            },
  1071→        }
  1072→
  1073→    @classmethod
  1074→    def score_findings(cls, findings: List[Dict]) -> List[Dict]:
  1075→        """Score a list of findings, returning enriched list."""
  1076→        enriched = []
  1077→        for f in findings:
  1078→            scored = dict(f)
  1079→            cvss = cls.score_finding(f)
  1080→            scored["cvss_score"] = cvss["base_score"]
  1081→            scored["cvss_severity"] = cvss["severity"]
  1082→            scored["cvss_vector"] = cvss["cvss_vector"]
  1083→            scored["cvss_metrics"] = cvss["metrics"]
  1084→            enriched.append(scored)
  1085→        return enriched
  1086→
  1087→    @staticmethod
  1088→    def _cia_label(value: float) -> str:
  1089→        if value >= 0.56:
  1090→            return "H"
  1091→        elif value >= 0.22:
  1092→            return "L"
  1093→        return "N"

  1404→    @classmethod
  1405→    def correlate(cls, findings: List[Dict]) -> List[Dict]:
  1406→        """Run all correlation rules against findings and return compound findings.
  1407→
  1408→        Each finding is checked as a lowercase string of title + description.
  1409→        A rule fires when ALL trigger groups have at least one pattern matched
  1410→        across the full finding set.
  1411→        """
  1412→        if not findings:
  1413→            return []
  1414→
  1415→        # Build a flat list of all finding text for matching
  1416→        finding_texts: List[str] = []
  1417→        for f in findings:
  1418→            title = (f.get("title") or "").lower()
  1419→            desc = (f.get("description") or "").lower()
  1420→            finding_texts.append(f"{title} {desc}")
  1421→
  1422→        compound_findings: List[Dict] = []
  1423→
  1424→        for rule in cls.CORRELATION_RULES:
  1425→            triggers = rule["triggers"]
  1426→            rule_fired = True
  1427→
  1428→            for trigger_group in triggers:
  1429→                # Each trigger group: at least one pattern must match across all findings
  1430→                group_matched = False
  1431→                for pattern in trigger_group:
  1432→                    for ft in finding_texts:
  1433→                        if pattern in ft:
  1434→                            group_matched = True
  1435→                            break
  1436→                    if group_matched:
  1437→                        break
  1438→                if not group_matched:
  1439→                    rule_fired = False
  1440→                    break
  1441→
  1442→            if rule_fired:
  1443→                compound = {
  1444→                    "title": rule["compound_title"],
  1445→                    "severity": rule["compound_severity"],
  1446→                    "description": rule["compound_description"],
  1447→                    "category": "compound",
  1448→                    "correlation_id": rule["id"],
  1449→                    "is_compound": True,
  1450→                }
  1451→                # Compute CVSS score for the compound finding
  1452→                if "cvss_override" in rule:
  1453→                    override = rule["cvss_override"]
  1454→                    av = CVSSScorer._AV[override["av"]]
  1455→                    ac = CVSSScorer._AC[override["ac"]]
  1456→                    pr = CVSSScorer._PR[override["pr"]]
  1457→                    ui = CVSSScorer._UI[override["ui"]]
  1458→                    c_val = CVSSScorer._CIA[override["c"]]
  1459→                    i_val = CVSSScorer._CIA[override["i"]]
  1460→                    a_val = CVSSScorer._CIA[override["a"]]
  1461→                    iss = 1.0 - ((1.0 - c_val) * (1.0 - i_val) * (1.0 - a_val))
  1462→                    impact = 6.42 * iss
  1463→                    exploitability = 8.22 * av * ac * pr * ui
  1464→                    base_score = math.ceil(min(impact + exploitability, 10.0) * 10) / 10
  1465→                    compound["cvss_score"] = base_score
  1466→                    compound["cvss_vector"] = (
  1467→                        f"CVSS:3.1/AV:{override['av']}/AC:{override['ac']}/"
  1468→                        f"PR:{override['pr']}/UI:{override['ui']}/"
  1469→                        f"C:{override['c']}/I:{override['i']}/A:{override['a']}"
  1470→                    )
  1471→
  1472→                compound_findings.append(compound)
  1473→
  1474→        return compound_findings

  1481→
  1482→class AttackSurfaceMapper:
  1483→    """Computes an attack surface score (0–100, higher = more exposed).
  1484→
  1485→    Factors:
  1486→      1. Open ports (0–25 pts): 0 ports=0, 1–2=5, 3–5=10, 6–10=15, 11+=25
  1487→      2. Exposed paths/endpoints (0–20 pts): 0=0, 1–3=5, 4–7=10, 8–12=15, 13+=20
  1488→      3. Missing security headers (0–20 pts): 0=0, 1=3, 2=6, 3=10, 4=14, 5+=20
  1489→      4. Tech stack complexity (0–15 pts): 1–2 techs=2, 3–4=5, 5–6=8, 7–8=11, 9+=15
  1490→      5. TLS grade (0–20 pts): A=0, B=5, C=10, D=15, F=20
  1491→    """
  1492→
  1493→    @classmethod
  1494→    def compute(cls, findings: List[Dict], scan_data: Optional[Dict] = None) -> Dict[str, Any]:
  1495→        """Compute attack surface score and return breakdown."""
  1496→        findings_lower = []
  1497→        for f in findings:
  1498→            title = (f.get("title") or "").lower()
  1499→            desc = (f.get("description") or "").lower()
  1500→            findings_lower.append(f"{title} {desc}")
  1501→        all_text = " ".join(findings_lower)
  1502→
  1503→        # 1. Open ports
  1504→        port_score = cls._score_ports(findings_lower, scan_data)
  1505→        # 2. Exposed paths
  1506→        path_score = cls._score_exposed_paths(findings_lower)
  1507→        # 3. Missing headers
  1508→        header_score = cls._score_missing_headers(findings_lower)
  1509→        # 4. Tech stack complexity
  1510→        tech_score = cls._score_tech_stack(all_text, scan_data)
  1511→        # 5. TLS grade
  1512→        tls_score = cls._score_tls_grade(findings_lower, scan_data)
  1513→
  1514→        total = min(port_score + path_score + header_score + tech_score + tls_score, 100)
  1515→
  1516→        # Risk rating
  1517→        if total >= 80:
  1518→            rating = "CRITICAL"
  1519→        elif total >= 60:
  1520→            rating = "HIGH"
  1521→        elif total >= 40:
  1522→            rating = "MODERATE"
  1523→        elif total >= 20:
  1524→            rating = "LOW"
  1525→        else:
  1526→            rating = "MINIMAL"
  1527→
  1528→        return {
  1529→            "attack_surface_score": total,
  1530→            "risk_rating": rating,
  1531→            "breakdown": {
  1532→                "open_ports": {"score": port_score, "max": 25},
  1533→                "exposed_paths": {"score": path_score, "max": 20},
  1534→                "missing_headers": {"score": header_score, "max": 20},
  1535→                "tech_complexity": {"score": tech_score, "max": 15},
  1536→                "tls_grade": {"score": tls_score, "max": 20},
  1537→            },
  1538→        }

  1540→    @classmethod

  1549→            for f in scan_data.get("findings", []):
  1550→                title = (f.get("title") or "").lower()

  1554→            return 0

  1556→            return 5
  1557→        elif port_count <= 5:
  1558→            return 10
  1559→        elif port_count <= 10:
  1560→            return 15
  1561→        else:
  1562→            return 25

  1681→class RiskMatrixVisualizer:
  1682→    """Generates a text-based risk matrix (Likelihood vs Impact) using
  1683→    Unicode box-drawing characters.
  1684→
  1685→    Matrix layout (5x5):

  1694→    i   │         │       │         │        │          │

  1697→    o   │         │       │         │        │          │
  1698→    d   ├─────────┼───────┼─────────┼────────┼──────────┤
  1699→        │         │       │         │  ●     │    ●     │
  1700→    ↑   │         │       │         │        │          │
  1701→        ├─────────┼───────┼─────────┼────────┼──────────┤
  1702→        │         │       │         │        │          │
  1703→        │         │       │         │        │          │
  1704→        └─────────┴───────┴─────────┴────────┴──────────┘
  1705→    """
  1706→
  1707→    # Likelihood and Impact bands from CVSS scores
  1708→    _LIKELIHOOD_BANDS = [
  1709→        ("Very High", 9.0, 10.1),
  1710→        ("High", 7.0, 9.0),
  1711→        ("Medium", 4.0, 7.0),
  1712→        ("Low", 1.0, 4.0),
  1713→        ("Very Low", 0.0, 1.0),
  1714→    ]
  1715→
  1716→    _IMPACT_BANDS = [
  1717→        ("Critical", 9.0, 10.1),
  1718→        ("High", 7.0, 9.0),
  1719→        ("Medium", 4.0, 7.0),
  1720→        ("Low", 1.0, 4.0),
  1721→        ("Negligible", 0.0, 1.0),
  1722→    ]
  1723→
  1724→    # Color coding for cell density
  1725→    _DENSITY_SYMBOLS = ["  ", " ·", " •", " ●", " ██"]
  1726→
  1727→    @classmethod
  1728→    def render(cls, findings: List[Dict], width: int = 70) -> str:
  1729→        """Render the risk matrix as a text string with Unicode box-drawing.
  1730→
  1731→        Args:
  1732→            findings: List of finding dicts with 'cvss_score' (or 'base_score') keys.
  1733→            width: Total width of the matrix output.
  1734→
  1735→        Returns:
  1736→            Multi-line string with the matrix.
  1737→        """
  1738→        # Build a 5x5 grid of finding counts
  1739→        # Grid[y][x] where y=0 is top (Very High likelihood), x=0 is left (Negligible impact)
  1740→        grid: List[List[int]] = [[0] * 5 for _ in range(5)]
  1741→        grid_labels: List[List[List[str]]] = [[[] for _ in range(5)] for _ in range(5)]
  1742→

  1744→            score = f.get("cvss_score") or f.get("base_score") or 0.0
  1745→            if not isinstance(score, (int, float)) or score <= 0:
  1746→                continue
  1747→            title = (f.get("title") or "")[:12]
  1748→
  1749→            # Determine likelihood (y-axis) and impact (x-axis) from CVSS score

  1751→            # - Impact uses the CIA-derived portion
  1752→            # - Likelihood uses the exploitability portion

  1754→            impact_idx = cls._score_to_band_index(score)
  1755→            likelihood_idx = cls._score_to_likelihood(score, f)
  1756→

  1758→                grid[likelihood_idx][impact_idx] += 1
  1759→                if len(grid_labels[likelihood_idx][impact_idx]) < 3:
  1760→                    grid_labels[likelihood_idx][impact_idx].append(title)
  1761→

  1763→        col_width = 11
  1764→        label_width = 12
  1765→        lines: List[str] = []
  1766→
  1767→        # Header row
  1768→        header = f"{'Likelihood / Impact':<{label_width + 2}}"
  1769→        impact_names = [b[0] for b in cls._IMPACT_BANDS]
  1770→        for name in impact_names:
  1771→            header += f"{name[:col_width]:>{col_width}}"
  1772→        lines.append(header)

  1776→
  1777→        likelihood_names = [b[0] for b in cls._LIKELIHOOD_BANDS]
  1778→        for y in range(5):

  1780→            label = likelihood_names[y]
  1781→            # Line 1: density symbol
  1782→            row1 = f"{label:>{label_width}}  │"
  1783→            row2 = f"{'':>{label_width}}  │"

  1785→                count = grid[y][x]
  1786→                sym = cls._DENSITY_SYMBOLS[min(count, 4)]
  1787→                row1 += f"{sym:^{col_width}}│"
  1788→                labels = grid_labels[y][x]
  1789→                if labels:
  1790→                    row2 += f"{labels[0][:col_width]:^{col_width}}│"
  1791→                else:
  1792→                    row2 += f"{'':^{col_width}}│"
  1793→
  1794→            lines.append(row1)
  1795→            lines.append(row2)
  1796→
  1797→            # Separator or bottom border
  1798→            if y < 4:
  1799→                lines.append(f"{'':<{label_width + 2}}├" + "─" * col_width + ("┼" + "─" * col_width) * 4 + "┤")
  1800→            else:
  1801→                lines.append(f"{'':<{label_width + 2}}└" + "─" * col_width + ("┴" + "─" * col_width) * 4 + "┘")
  1802→

  1804→        total = sum(grid[y][x] for y in range(5) for x in range(5))

  1806→        lines.append("")
  1807→        lines.append(f"  Total findings plotted: {total}")
  1808→        lines.append(f"  Critical zone (High/Very High × High/Critical): {critical_zone}")
  1809→
  1810→        return "\n".join(lines)

  1812→    @classmethod
  1813→    def _score_to_band_index(cls, score: float) -> int:
  1814→        """Map score to impact band index (0=Negligible, 4=Critical)."""
  1815→        for i, (name, lo, hi) in enumerate(cls._IMPACT_BANDS):
  1816→            if lo <= score < hi:
  1817→                return 4 - i  # Reverse so Critical is rightmost
  1818→        return 0

  1820→    @classmethod
  1821→    def _score_to_likelihood(cls, score: float, finding: Dict) -> int:
  1822→        """Estimate likelihood from score and finding metadata.
  1823→
  1824→        Higher CVSS exploitability metrics → higher likelihood.
  1825→        We approximate from score and severity.
  1826→        """
  1827→        severity = (finding.get("severity") or "").lower()
  1828→        is_compound = finding.get("is_compound", False)
  1829→        is_remote = True  # Most findings in recon are network-based
  1830→
  1831→        # Base likelihood from score
  1832→        base = score / 10.0
  1833→
  1834→        # Boost for compound findings (more likely to be exploitable)
  1835→        if is_compound:
  1836→            base = min(base + 0.15, 1.0)
  1837→
  1838→        # Adjust for severity
  1839→        sev_boost = {"critical": 0.1, "high": 0.05, "medium": 0, "low": -0.1, "info": -0.2}
  1840→        base += sev_boost.get(severity, 0)
  1841→        base = max(0.0, min(base, 1.0))
  1842→
  1843→        # Map to band index (0=Very Low, 4=Very High)
  1844→        if base >= 0.85:
  1845→            return 0
  1846→        elif base >= 0.65:
  1847→            return 1
  1848→        elif base >= 0.40:
  1849→            return 2
  1850→        elif base >= 0.15:
  1851→            return 3
  1852→        else:
  1853→            return 4

  2289→    def classify_intent(self, goal: str) -> Dict[str, float]:
  2290→        """Score each intent based on the goal text.
  2291→
  2292→        Uses weighted keyword matching, proximity scoring, and modifiers.
  2293→        Returns a dict mapping intent names to float scores in [0, 1+].
  2294→        Supports 30+ intent categories.
  2295→        """
  2296→        goal_lower = goal.lower()
  2297→        # Pad for whole-word matching
  2298→        padded = f' {goal_lower} '
  2299→        scores: Dict[str, float] = {}
  2300→        words = goal_lower.split()
  2301→
  2302→        # Calculate global modifier
  2303→        modifier = 1.0
  2304→        for w in words:
  2305→            if w in self._AMPLIFIERS:
  2306→                modifier = max(modifier, self._AMPLIFIERS[w])
  2307→            if w in self._DAMPENERS:
  2308→                modifier = min(modifier, self._DAMPENERS[w])
  2309→
  2310→        for intent, keywords in self._INTENT_KEYWORDS.items():
  2311→            score = 0.0
  2312→            for pattern, weight in keywords:
  2313→                # Use whole-word matching to avoid false positives
  2314→                # (e.g., "repo" matching inside "report")
  2315→                if len(pattern.split()) > 1:
  2316→                    # Multi-word pattern: check substring (already specific enough)
  2317→                    matched = pattern in goal_lower
  2318→                else:
  2319→                    # Single-word pattern: check whole word with padding
  2320→                    matched = f' {pattern} ' in padded
  2321→                if matched:
  2322→                    # Boost if near a target-like word
  2323→                    proximity_boost = 1.0
  2324→                    pattern_idx = goal_lower.find(pattern)
  2325→                    for tw in (".com", ".io", ".net", ".org", "http", "localhost"):
  2326→                        tw_idx = goal_lower.find(tw)
  2327→                        if tw_idx >= 0 and abs(pattern_idx - tw_idx) < 40:
  2328→                            proximity_boost = 1.3
  2329→                            break
  2330→                    score += weight * proximity_boost
  2331→            scores[intent] = min(score * modifier, 2.0)
  2332→
  2333→        return scores
