   141→   141→class HeaderRotator:
   142→   142→    """Rotates through 50+ real browser User-Agent strings and produces a
   143→   143→    complete, consistent set of HTTP headers for each rotation.
   144→   144→
   145→   145→    Maintains referer state across calls so successive requests look like
   146→   146→    a real browsing session.
   147→   147→    """
   148→   148→
   149→   149→    def __init__(self, seed: Optional[int] = None) -> None:
   150→   150→        self._rng = random.Random(seed)
   151→   151→        self._ua_pool = list(_USER_AGENTS)
   152→   152→        self._rng.shuffle(self._ua_pool)
   153→   153→        self._index = 0
   154→   154→        self._referer: Optional[str] = None
   155→   155→        self._lang_codes = list(_ACCEPT_LANGS.keys())
   156→   156→
   157→   157→    # -- public API ----------------------------------------------------------
   158→   158→
   159→   159→    def set_referer(self, url: str) -> None:
   160→   160→        """Store a referer URL that will be included on the *next* rotate()."""
   161→   161→        self._referer = url
   162→   162→
   163→   163→    def clear_referer(self) -> None:
   164→   164→        """Clear stored referer."""
   165→   165→        self._referer = None
   166→   166→
   167→   167→    def rotate(self, custom_headers: Optional[Dict[str, str]] = None) -> Dict[str, str]:
   168→   168→        """Return a full set of HTTP headers with a randomised User-Agent.
   169→   169→
   170→   170→        Parameters
   171→   171→        ----------
   172→   172→        custom_headers:
   173→   173→            Optional dict of headers that override / extend the generated set.
   174→   174→
   175→   175→        Returns
   176→   176→        -------
   177→   177→        dict  –  mapping of header name → value
   178→   178→        """
   179→   179→        ua = self._pick_ua()
   180→   180→        headers: Dict[str, str] = {
   181→   181→            "User-Agent": ua,
   182→   182→            "Accept": self._accept_for(ua),
   183→   183→            "Accept-Language": _ACCEPT_LANGS[self._rng.choice(self._lang_codes)],
   184→   184→            "Accept-Encoding": "gzip, deflate, br, zstd",
   185→   185→            "Connection": "keep-alive",
   186→   186→            "Cache-Control": "max-age=0",
   187→   187→            "Sec-Fetch-Dest": "document",
   188→   188→            "Sec-Fetch-Mode": "navigate",
   189→   189→            "Sec-Fetch-Site": "none",
   190→   190→        }
   191→   191→
   192→   192→        # Add Sec-CH-UA only for Chrome / Edge
   193→   193→        sec_ch = self._sec_ch_ua(ua)
   194→   194→        if sec_ch is not None:
   195→   195→            headers["Sec-CH-UA"] = sec_ch[0]
   196→   196→            headers["Sec-CH-UA-Mobile"] = sec_ch[1]
   197→   197→            headers["Sec-CH-UA-Platform"] = sec_ch[2]
   198→   198→
   199→   199→        if self._referer is not None:
   200→   200→            headers["Referer"] = self._referer
   201→   201→            self._referer = None
   202→   202→
   203→   203→        if custom_headers:
   204→   204→            headers.update(custom_headers)
   205→   205→
   206→   206→        return headers
   207→   207→
   208→   208→    # -- internals -----------------------------------------------------------
   209→   209→
   210→   210→    def _pick_ua(self) -> str:
   211→   211→        """Pick next UA in shuffled order, re-shuffle when exhausted."""
   212→   212→        if self._index >= len(self._ua_pool):
   213→   213→            self._rng.shuffle(self._ua_pool)
   214→   214→            self._index = 0
   215→   215→        ua = self._ua_pool[self._index]
   216→   216→        self._index += 1
   217→   217→        return ua
   218→   218→
   219→   219→    @staticmethod
   220→   220→    def _accept_for(ua: str) -> str:
   221→   221→        """Return an Accept header appropriate for the browser family."""
   222→   222→        if "Edg/" in ua:
   223→   223→            return (
   224→   224→                "text/html,application/xhtml+xml,application/xml;q=0.9,"
   225→   225→                "image/avif,image/webp,image/apng,*/*;q=0.8,"
   226→   226→                "application/signed-exchange;v=b3;q=0.7"
   227→   227→            )
   228→   228→        if "Firefox/" in ua:
   229→   229→            return (
   230→   230→                "text/html,application/xhtml+xml,application/xml;q=0.9,"
   231→   231→                "image/avif,image/webp,*/*;q=0.8"
   232→   232→            )
   233→   233→        if "Safari/" in ua and "Chrome/" not in ua:
   234→   234→            return (
   235→   235→                "text/html,application/xhtml+xml,application/xml;q=0.9,"
   236→   236→                "*/*;q=0.8"
   237→   237→            )
   238→   238→        # Chrome / generic Chromium
   239→   239→        return (
   240→   240→            "text/html,application/xhtml+xml,application/xml;q=0.9,"
   241→   241→            "image/avif,image/webp,image/apng,*/*;q=0.8,"
   242→   242→            "application/signed-exchange;v=b3;q=0.7"
   243→   243→        )
   244→   244→
   245→   245→    @staticmethod
   246→   246→    def _sec_ch_ua(ua: str) -> Optional[tuple]:
   247→   247→        """Return (Sec-CH-UA, Mobile, Platform) or None for non-Chrome."""
   248→   248→        if "Edg/" in ua:
   249→   249→            ver = _extract_chrome_ver(ua)
   250→   250→            ch = _SEC_CH_UA_EDGE.get(ver)
   251→   251→            platform = _extract_platform(ua)
   252→   252→            if ch:
   253→   253→                return (ch, "?0", f'"{platform}"')
   254→   254→            return None
   255→   255→        if "Chrome/" in ua and "Safari/" in ua and "Edg/" not in ua:
   256→   256→            ver = _extract_chrome_ver(ua)
   257→   257→            is_mobile = "Mobile" in ua
   258→   258→            pool = _SEC_CH_UA_MOBILE if is_mobile else _SEC_CH_UA
   259→   259→            ch = pool.get(ver)
   260→   260→            platform = _extract_platform(ua)
   261→   261→            if ch:
   262→   262→                return (ch, "?1" if is_mobile else "?0", f'"{platform}"')
   263→   263→            return None
   264→   264→        return None
   265→   265→
   266→   266→
   267→   267→# helpers for HeaderRotator
   268→   268→
   269→   269→
   270→   270→def _extract_chrome_ver(ua: str) -> str:
   271→   271→    """Extract the major Chrome version number from a UA string."""
   272→   272→    import re
   273→   273→    m = re.search(r"Chrome/(\d+)\.", ua)
   274→   274→    return m.group(1) if m else "124"
   275→   275→
   276→   276→
   277→   277→def _extract_platform(ua: str) -> str:
   278→   278→    """Guess the platform string for Sec-CH-UA-Platform."""
   279→   279→    if "Windows" in ua:
   280→   280→        return "Windows"
   281→   281→    if "Macintosh" in ua or "Mac OS X" in ua:
   282→   282→        return "macOS"
   283→   283→    if "Android" in ua:
   284→   284→        return "Android"
   285→   285→    if "iPhone" in ua or "iPad" in ua:
   286→   286→        return "iOS"
   287→   287→    if "Linux" in ua:
   288→   288→        return "Linux"
   289→   289→    return "Unknown"

   420→   420→        if profile is None:

   423→   423→                f"Choose from: {', '.join(self._PRESETS)}, {self.RANDOM}"

   437→   437→        ciphers = list(base["cipher_suites"])

   439→   439→        # Keep AES_128_GCM near the top 60 % of the time (most browsers prefer it)
   440→   440→        if self._rng.random() < 0.6:
   441→   441→            aes128 = "TLS_AES_128_GCM_SHA256"

   443→   443→                ciphers.remove(aes128)
   444→   444→                ciphers.insert(self._rng.randint(0, 2), aes128)

   448→   448→        if "server_name" in extensions:
   449→   449→            extensions.remove("server_name")
   450→   450→            extensions.insert(0, "server_name")

   452→   452→            "cipher_suites": ciphers,
   453→   453→            "tls_version": base["tls_version"],
   454→   454→            "extensions": extensions,
   455→   455→            "grease": base["grease"],

   468→   468→class TimingJitter:
   469→   469→    """Add human-like timing jitter between requests.
   470→   470→
   471→   471→    Uses a Gaussian distribution with configurable stddev.  Supports
   472→   472→    burst patterns that simulate real browsing (rapid clicks then pause).
   473→   473→    """
   474→   474→
   475→   475→    MIN_DELAY: float = 0.05
   476→   476→    MAX_DELAY: float = 5.0
   477→   477→    DEFAULT_STDDEV: float = 0.3
   478→   478→
   479→   479→    def __init__(
   480→   480→        self,
   481→   481→        stddev: float = DEFAULT_STDDEV,
   482→   482→        min_delay: float = MIN_DELAY,
   483→   483→        max_delay: float = MAX_DELAY,
   484→   484→        burst_enabled: bool = True,
   485→   485→        seed: Optional[int] = None,
   486→   486→    ) -> None:
   487→   487→        self._rng = random.Random(seed)
   488→   488→        self._stddev = stddev
   489→   489→        self._min = min_delay
   490→   490→        self._max = max_delay
   491→   491→        self._burst_enabled = burst_enabled
   492→   492→        self._burst_remaining = 0
   493→   493→        self._in_burst = False
   494→   494→
   495→   495→    def add_jitter(self) -> float:
   496→   496→        """Return a delay in seconds drawn from a Gaussian distribution,
   497→   497→        clamped to [min_delay, max_delay].
   498→   498→
   499→   499→        If burst mode is enabled, some calls will return near-zero delay to
   500→   500→        simulate rapid human clicks, followed by a longer pause.
   501→   501→        """
   502→   502→        if self._burst_enabled:
   503→   503→            delay = self._burst_jitter()
   504→   504→        else:
   505→   505→            delay = self._gaussian_jitter()
   506→   506→        return max(self._min, min(self._max, delay))
   507→   507→
   508→   508→    def _gaussian_jitter(self) -> float:
   509→   509→        """Pure Gaussian jitter: mean=0, clamped so result is >= min."""
   510→   510→        sample = self._rng.gauss(0, self._stddev)
   511→   511→        # Shift so the result is a positive delay
   512→   512→        return max(self._min, abs(sample) + self._min)
   513→   513→
   514→   514→    def _burst_jitter(self) -> float:
   515→   515→        """Simulate a burst of rapid requests then a pause.
   516→   516→
   517→   517→        Each burst consists of 2-5 requests with minimal delay (~0.05-0.15s),
   518→   518→        then a pause of 1.5-4.0s before the next burst.
   519→   519→        """
   520→   520→        if not self._in_burst:
   521→   521→            # Decide whether to start a burst (40 % chance)
   522→   522→            if self._rng.random() < 0.4:
   523→   523→                self._in_burst = True
   524→   524→                self._burst_remaining = self._rng.randint(2, 5)
   525→   525→            else:
   526→   526→                # Normal delay
   527→   527→                return self._gaussian_jitter()
   528→   528→
   529→   529→        if self._in_burst:
   530→   530→            if self._burst_remaining > 1:
   531→   531→                self._burst_remaining -= 1
   532→   532→                return self._rng.uniform(0.05, 0.15)
   533→   533→            else:
   534→   534→                # End of burst — return a longer pause
   535→   535→                self._in_burst = False
   536→   536→                self._burst_remaining = 0
   537→   537→                return self._rng.uniform(1.5, 4.0)
   538→   538→
   539→   539→        return self._gaussian_jitter()

   546→   546→
   547→   547→class PathCanonicalizer:
   548→   548→    """Generate variations of a URL path to test WAF / CDN normalisation.
   549→   549→
   550→   550→    Produces 8-12 variants using techniques like dot-segments, URL-encoding,
   551→   551→    mixed case, trailing slashes, and overlong UTF-8 sequences.
   552→   552→    """
   553→   553→
   554→   554→    def generate_variants(self, path: str) -> List[str]:
   555→   555→        """Return a list of 8-12 path variations for WAF testing.
   556→   556→
   557→   557→        Parameters
   558→   558→        ----------
   559→   559→        path : str
   560→   560→            The original URL path (e.g. ``/api/v1/users``).
   561→   561→
   562→   562→        Returns
   563→   563→        -------
   564→   564→        list[str]
   565→   565→        """
   566→   566→        if not path.startswith("/"):
   567→   567→            path = "/" + path
   568→   568→
   569→   569→        variants: List[str] = []
   570→   570→        # 1. Original
   571→   571→        variants.append(path)
   572→   572→        # 2. Dot-segment prefix
   573→   573→        variants.append("/./" + path.lstrip("/"))
   574→   574→        # 3. Double-encoding of first char after slash
   575→   575→        variants.append(self._double_encode_char(path))
   576→   576→        # 4. URL-encoded full first segment
   577→   577→        variants.append(self._encode_segment(path))
   578→   578→        # 5. Mixed case — title-case each segment
   579→   579→        variants.append(self._mixed_case(path))
   580→   580→        # 6. UPPERCASE
   581→   581→        variants.append(path.upper())
   582→   582→        # 7. Trailing slash
   583→   583→        variants.append(path.rstrip("/") + "/")
   584→   584→        # 8. Double trailing slash
   585→   585→        variants.append(path.rstrip("/") + "//")
   586→   586→        # 9. Trailing dot-segment
   587→   587→        variants.append(path.rstrip("/") + "/.")
   588→   588→        # 10. Overlong UTF-8 for first slash-separated segment
   589→   589→        variants.append(self._overlong_utf8(path))
   590→   590→        # 11. Null-byte injection (common WAF bypass)
   591→   591→        variants.append(path.rstrip("/") + "%00")
   592→   592→        # 12. Semicolon / path-param injection
   593→   593→        variants.append(path.rstrip("/") + ";jsessionid=" + _rand_hex(8))
   594→   594→
   595→   595→        # De-duplicate while preserving order
   596→   596→        seen: set = set()
   597→   597→        unique: List[str] = []
   598→   598→        for v in variants:
   599→   599→            if v not in seen:
   600→   600→                seen.add(v)
   601→   601→                unique.append(v)
   602→   602→
   603→   603→        return unique
   604→   604→
   605→   605→    # -- variant generators --------------------------------------------------
   606→   606→
   607→   607→    @staticmethod
   608→   608→    def _double_encode_char(path: str) -> str:
   609→   609→        """Double-encode the first character of each path segment.
   610→   610→
   611→   611→        E.g. '/a/b' → '/%2561/%2562'
   612→   612→        """
   613→   613→        parts = path.strip("/").split("/")
   614→   614→        encoded_parts: List[str] = []
   615→   615→        for part in parts:
   616→   616→            if not part:
   617→   617→                continue
   618→   618→            first = part[0]
   619→   619→            double = "%25" + ("%02X" % ord(first))
   620→   620→            encoded_parts.append(double + part[1:])
   621→   621→        return "/" + "/".join(encoded_parts)
   622→   622→
   623→   623→    @staticmethod
   624→   624→    def _encode_segment(path: str) -> str:
   625→   625→        """Fully URL-encode the first path segment."""
   626→   626→        parts = path.strip("/").split("/")
   627→   627→        if not parts or not parts[0]:
   628→   628→            return path
   629→   629→        encoded_first = urllib.parse.quote(parts[0], safe="")
   630→   630→        return "/" + encoded_first + ("/" + "/".join(parts[1:]) if len(parts) > 1 else "")
   631→   631→
   632→   632→    @staticmethod
   633→   633→    def _mixed_case(path: str) -> str:
   634→   634→        """Title-case each segment: /api/v1/users → /Api/V1/Users."""
   635→   635→        parts = path.strip("/").split("/")
   636→   636→        titled = [p[:1].upper() + p[1:].lower() if p else "" for p in parts]
   637→   637→        return "/" + "/".join(titled)
   638→   638→
   639→   639→    @staticmethod
   640→   640→    def _overlong_utf8(path: str) -> str:
   641→   641→        """Replace the first character of the first segment with an overlong
   642→   642→        UTF-8 2-byte sequence (e.g. 'a' → %C0%AF for '/').
   643→   643→
   644→   644→        Specifically this targets the forward-slash character to produce
   645→   645→        /path%C0%AFrest style bypasses.
   646→   646→        """
   647→   647→        parts = path.strip("/").split("/")
   648→   648→        if not parts:
   649→   649→            return path
   650→   650→        # Use a valid overlong encoding of 'a' (U+0061) as 2-byte: 0xC1 0xA1
   651→   651→        # Note: many modern systems reject overlong UTF-8, but some WAFs still
   652→   652→        # mishandle it. We encode the first char of the first segment.
   653→   653→        first = parts[0][0] if parts[0] else "a"
   654→   654→        cp = ord(first)
   655→   655→        if cp < 0x80:
   656→   656→            # Overlong 2-byte: 110xxxxx 10xxxxxx → 0xC0|cp>>6, 0x80|cp&0x3F
   657→   657→            b1 = 0xC0 | (cp >> 6)
   658→   658→            b2 = 0x80 | (cp & 0x3F)
   659→   659→            overlong = f"%{b1:02X}%{b2:02X}"
   660→   660→            parts[0] = overlong + parts[0][1:]
   661→   661→        return "/" + "/".join(parts)
   662→   662→
   663→   663→
   664→   664→def _rand_hex(n: int) -> str:
   665→   665→    """Return a random hex string of length *n*."""
   666→   666→    return "".join(random.choice("0123456789abcdef") for _ in range(n))

  1093→  1093→class RateLimiter:
  1094→  1094→    """Token bucket rate limiter for scan requests."""
  1095→  1095→
  1096→  1096→    def __init__(self, max_requests: float = 10.0, period: float = 1.0):
  1097→  1097→        self._max = max_requests
  1098→  1098→        self._period = period
  1099→  1099→        self._tokens = max_requests
  1100→  1100→        self._last_refill = time.monotonic()
  1101→  1101→
  1102→  1102→    def acquire(self) -> bool:
  1103→  1103→        """Try to acquire a token. Returns True if request allowed."""
  1104→  1104→        now = time.monotonic()
  1105→  1105→        elapsed = now - self._last_refill
  1106→  1106→        self._tokens = min(self._max, self._tokens + elapsed * (self._max / self._period))
  1107→  1107→        self._last_refill = now

  1109→  1109→            self._tokens -= 1.0
  1110→  1110→            return True
  1111→  1111→        return False

  1113→  1113→    def wait(self) -> float:
  1114→  1114→        """How long to wait for the next token."""
  1115→  1115→        if self._tokens >= 1.0:
  1116→  1116→            return 0.0
  1117→  1117→        return (1.0 - self._tokens) * (self._period / self._max)
