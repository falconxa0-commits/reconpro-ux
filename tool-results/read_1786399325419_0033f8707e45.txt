    28→def _hostname() -> str:
    29→    """Cross-platform hostname (works on Linux, macOS, Windows)."""
    30→    try:
    31→        return os.uname().nodename
    32→    except AttributeError:
    33→        return os.environ.get("COMPUTERNAME", os.environ.get("HOSTNAME", "localhost"))

    36→def _run(cmd: str, timeout: int = 8) -> Tuple[int, str]:
    37→    """Run a shell command, return (exit_code, output)."""
    38→    try:
    39→        r = subprocess.run(
    40→            cmd, shell=True, capture_output=True, text=True, timeout=timeout
    41→        )
    42→        return r.returncode, (r.stdout + r.stderr).strip()
    43→    except Exception:
    44→        return -1, ""
    45→

    47→def _is_windows() -> bool:
    48→    """Check if running on Windows."""
    49→    return os.name == "nt"

    52→def _is_macos() -> bool:
    53→    """Check if running on macOS."""
    54→    return os.name == "posix" and hasattr(os, "uname") and os.uname().sysname == "Darwin"

    57→def _check_password_policy() -> List[Finding]:
    58→    """Check password and authentication policies."""
    59→    findings: List[Finding] = []
    60→    hostname = _hostname()
    61→
    62→    if _is_windows():
    63→        # Windows: check password policy via net accounts
    64→        code, out = _run('net accounts 2>NUL')
    65→        if code == 0 and out:
    66→            max_age = None
    67→            min_len = None
    68→            for line in out.splitlines():
    69→                line_lower = line.lower().strip()
    70→                if "maximum password age" in line_lower:
    71→                    parts = line.split(":")
    72→                    if len(parts) >= 2:
    73→                        try:
    74→                            val = parts[-1].strip()
    75→                            max_age = int(val) if val.lower() != "unlimited" else 99999
    76→                        except (ValueError, IndexError):
    77→                            pass
    78→                elif "minimum password length" in line_lower:
    79→                    parts = line.split(":")
    80→                    if len(parts) >= 2:
    81→                        try:
    82→                            min_len = int(parts[-1].strip())
    83→                        except (ValueError, IndexError):
    84→                            pass
    85→            if max_age is not None and max_age > 90:
    86→                findings.append(Finding(
    87→                    title=f"Password max age is {max_age} days (too long)",
    88→                    severity="medium", category="auth_policy",
    89→                    module="doctor",
    90→                    description=f"Password expiration is set to {max_age} days. NIST recommends 90 days or less.",
    91→                    evidence=f"Maximum password age: {max_age}",
    92→                    asset=hostname, points_deducted=4,
    93→                    remediation="Set password expiration: net accounts /maxpwage:90",
    94→                ))
    95→            if min_len is not None and min_len < 12:
    96→                findings.append(Finding(
    97→                    title=f"Minimum password length is {min_len} (too short)",
    98→                    severity="medium", category="auth_policy",
    99→                    module="doctor",
   100→                    description="Minimum password length is less than 12 characters. Longer passwords are more secure.",
   101→                    evidence=f"Minimum password length: {min_len}",
   102→                    asset=hostname, points_deducted=5,
   103→                    remediation="Set minimum password length: net accounts /minpwlen:12",
   104→                ))
   105→        return findings
   106→
   107→    # Non-Windows: check Linux password policies
   108→    # Check if password aging is configured
   109→    code, out = _run("cat /etc/login.defs 2>/dev/null")
   110→    if code == 0 and out:
   111→        max_days = 99999
   112→        for line in out.splitlines():
   113→            if line.strip().startswith("PASS_MAX_DAYS"):
   114→                try:
   115→                    max_days = int(line.split()[1])
   116→                except (ValueError, IndexError):
   117→                    pass
   118→        if max_days > 90:
   119→            findings.append(Finding(
   120→                title=f"Password max age is {max_days} days (too long)",
   121→                severity="medium", category="auth_policy",
   122→                module="doctor",
   123→                description=f"Password expiration is set to {max_days} days. NIST recommends 90 days or less.",
   124→                evidence=f"PASS_MAX_DAYS={max_days}",
   125→                asset=hostname, points_deducted=4,
   126→                remediation="Set PASS_MAX_DAYS=90 in /etc/login.defs",
   127→            ))
   128→
   129→    # Check minimum password length
   130→    code, out = _run("cat /etc/security/pwquality.conf 2>/dev/null || cat /etc/pam.d/common-password 2>/dev/null")
   131→    if code == 0 and out:
   132→        if "minlen" not in out.lower() and "pam_pwquality" not in out and "pam_cracklib" not in out:
   133→            findings.append(Finding(
   134→                title="No password complexity requirements configured",
   135→                severity="medium", category="auth_policy",
   136→                module="doctor",
   137→                description="No minimum password length or complexity requirements found.",
   138→                evidence="pwquality.conf not configured",
   139→                asset=hostname, points_deducted=5,
   140→                remediation="Configure /etc/security/pwquality.conf: minlen=12 dcredit=-1 ucredit=-1 ocredit=-1 lcredit=-1",
   141→            ))
   142→
   143→    return findings

   146→def _check_disk_encryption() -> List[Finding]:
   147→    """Check if disk encryption is active."""
   148→    findings: List[Finding] = []
   149→    hostname = _hostname()
   150→
   151→    # Windows: check BitLocker
   152→    if _is_windows():
   153→        code, out = _run('manage-bde -status C: 2>NUL')
   154→        if code == 0 and out and "Percentage Encrypted" in out:
   155→            pct_match = re.search(r'Percentage Encrypted[:\s]+(\d+)%', out)
   156→            if pct_match and int(pct_match.group(1)) > 0:
   157→                return findings  # BitLocker is active
   158→        findings.append(Finding(
   159→            title="No disk encryption detected",
   160→            severity="medium", category="disk_security",
   161→            module="doctor",
   162→            description="No BitLocker encryption found on the system drive. Laptop data is at risk if stolen.",
   163→            evidence="BitLocker not active or not detected",
   164→            asset=hostname, points_deducted=5,
   165→            remediation="Enable BitLocker: Settings > System > BitLocker or manage-bde -on C:",
   166→        ))
   167→        return findings
   168→
   169→    # macOS: check FileVault
   170→    if _is_macos():
   171→        code, out = _run("fdesetup status 2>/dev/null")
   172→        if code == 0 and "On" in (out or ""):
   173→            return findings  # FileVault is active
   174→        findings.append(Finding(
   175→            title="No disk encryption detected",
   176→            severity="medium", category="disk_security",
   177→            module="doctor",
   178→            description="No FileVault encryption found. Laptop data is at risk if stolen.",
   179→            evidence="FileVault not active",
   180→            asset=hostname, points_deducted=5,
   181→            remediation="Enable FileVault: System Preferences > Security & Privacy > FileVault",
   182→        ))
   183→        return findings
   184→
   185→    # Linux: check LUKS / ecryptfs
   186→    code, out = _run("lsblk -o NAME,FSTYPE,TYPE -n 2>/dev/null | grep -i crypt")
   187→    if code != 0 or not out:
   188→        # Check for ecryptfs
   189→        code2, out2 = _run("mount | grep -i ecryptfs")
   190→        if code2 != 0 or not out2:
   191→            findings.append(Finding(
   192→                title="No disk encryption detected",
   193→                severity="medium", category="disk_security",
   194→                module="doctor",
   195→                description="No LUKS or ecryptfs encryption found. Laptop data is at risk if stolen.",
   196→                evidence="No encrypted volumes found",
   197→                asset=hostname, points_deducted=5,
   198→                remediation="Enable full disk encryption: cryptsetup luksFormat /dev/sdX",
   199→            ))
   200→    return findings

   203→def _check_auto_lock() -> List[Finding]:
   204→    """Check screen lock / auto-lock settings."""
   205→    findings: List[Finding] = []
   206→    hostname = _hostname()
   207→
   208→    # Windows: check screen saver lock via registry
   209→    if _is_windows():
   210→        code, out = _run('reg query "HKCU\\Control Panel\\Desktop" /v ScreenSaveActive 2>NUL')
   211→        if code == 0 and out:
   212→            if "REG_SZ" in out and "1" in out.split("REG_SZ")[-1]:
   213→                return findings  # Screen saver with lock is active
   214→        findings.append(Finding(
   215→            title="Screen lock may not be enabled",
   216→            severity="medium", category="physical_security",
   217→            module="doctor",
   218→            description="Windows screen lock does not appear to be active. Your machine can be accessed when unattended.",
   219→            evidence="ScreenSaveActive not set to 1",
   220→            asset=hostname, points_deducted=5,
   221→            remediation="Enable screen lock: Settings > Personalization > Lock screen > Screen timeout",
   222→        ))
   223→        return findings
   224→
   225→    # macOS: check screen saver idle time
   226→    if _is_macos():
   227→        code, out = _run("defaults read com.apple.screensaver idleTime 2>/dev/null")
   228→        if code == 0 and out.strip():
   229→            try:
   230→                seconds = int(out.strip())
   231→                if seconds > 0 and seconds <= 300:  # 5 minutes or less
   232→                    return findings  # Auto-lock is configured
   233→            except ValueError:
   234→                pass
   235→        findings.append(Finding(
   236→            title="Screen lock may not be configured",
   237→            severity="medium", category="physical_security",
   238→            module="doctor",
   239→            description="macOS screen lock idle time is not set or too long. Your machine can be accessed when unattended.",
   240→            evidence=f"idleTime: {out.strip() if code == 0 else 'not set'}",
   241→            asset=hostname, points_deducted=5,
   242→            remediation="Enable screen lock: System Preferences > Desktop & Screen Saver > Start After",
   243→        ))
   244→        return findings
   245→
   246→    # Linux: check gnome screensaver or xautolock
   247→    code, out = _run("gsettings get org.gnome.desktop.screensaver lock-enabled 2>/dev/null")
   248→    if code == 0:
   249→        if "false" in out.lower():
   250→            findings.append(Finding(
   251→                title="Screen lock is disabled",
   252→                severity="medium", category="physical_security",
   253→                module="doctor",
   254→                description="GNOME screen lock is disabled. Your machine can be accessed when unattended.",
   255→                evidence="lock-enabled: false",
   256→                asset=hostname, points_deducted=5,
   257→                remediation="Enable screen lock: gsettings set org.gnome.desktop.screensaver lock-enabled true",
   258→            ))
   259→    else:
   260→        # Check xautolock or xscreensaver
   261→        code2, out2 = _run("pgrep -x xautolock 2>/dev/null || pgrep -x xscreensaver 2>/dev/null")
   262→        if code2 != 0 or not out2:
   263→            findings.append(Finding(
   264→                title="No auto-lock mechanism detected",
   265→                severity="low", category="physical_security",
   266→                module="doctor",
   267→                description="No screen locker daemon (xautolock/xscreensaver) is running.",
   268→                evidence="No lock process found",
   269→                asset=hostname, points_deducted=3,
   270→                remediation="Install xautolock: sudo apt install xautolock && xautolock -time 5 -locker lock",
   271→            ))
   272→
   273→    return findings

   539→def _check_ntp_sync() -> List[Finding]:
   540→    """Check if system clock is synchronized via NTP."""
   541→    findings: List[Finding] = []
   542→    hostname = _hostname()
   543→
   544→    if _is_windows():
   545→        # Windows: use w32tm
   546→        code, out = _run('w32tm /query /status 2>NUL')
   547→        if code == 0 and out:
   548→            # Look for "Source" and "Last Successful Sync" lines
   549→            source = ""

  1102→def _check_ssh_key_strength() -> List[Finding]:
  1103→    """Check SSH key types and lengths in ~/.ssh/."""
  1104→    findings: List[Finding] = []
  1105→    hostname = _hostname()
  1106→
  1107→    ssh_dir = os.path.expanduser("~/.ssh")
  1108→    if not os.path.isdir(ssh_dir):
  1109→        return findings
  1110→
  1111→    # Known key type patterns and minimum bit lengths
  1112→    # Format: (file_glob_pattern, key_type_regex, recommended_min_bits, severity_if_short, description)
  1113→    key_files = []
  1114→    for entry in os.listdir(ssh_dir):
  1115→        full = os.path.join(ssh_dir, entry)
  1116→        if not os.path.isfile(full):
  1117→            continue
  1118→        # Check for private keys (no .pub extension, and known key file names)
  1119→        if entry.endswith(".pub"):
  1120→            continue  # We'll read the .pub to check key type
  1121→        if entry.startswith("id_") and not entry.endswith(".pub"):
  1122→            key_files.append((full, entry, os.path.join(ssh_dir, entry + ".pub")))
  1123→
  1124→    for priv_path, priv_name, pub_path in key_files:
  1125→        # Read the public key for key type and bit length
  1126→        if not os.path.isfile(pub_path):
  1127→            continue
  1128→        try:
  1129→            with open(pub_path, errors="replace") as f:
  1130→                content = f.read().strip()
  1131→        except Exception:
  1132→            continue
  1133→
  1134→        # Parse: ssh-rsa AAAA... comment  OR  ssh-ed25519 AAAA... comment
  1135→        parts = content.split()
  1136→        if len(parts) < 2:
  1137→            continue
  1138→
  1139→        key_type = parts[0]
  1140→        key_data_b64 = parts[1]
  1141→
  1142→        # Decode base64 key data to get bit length
  1143→        import base64
  1144→        try:
  1145→            key_bytes = base64.b64decode(key_data_b64)
  1146→            bit_length = len(key_bytes) * 8
  1147→        except Exception:
  1148→            bit_length = 0
  1149→
  1150→        if key_type == "ssh-rsa":
  1151→            # RSA keys: minimum 3072 bits recommended, 2048 is bare minimum
  1152→            if bit_length < 2048:
  1153→                findings.append(Finding(
  1154→                    title=f"Weak RSA SSH key: {priv_name} (~{bit_length} bits)",
  1155→                    severity="high", category="ssh_key_strength",
  1156→                    module="doctor",
  1157→                    description=f"RSA key {priv_name} is approximately {bit_length} bits. RSA keys should be at least 3072 bits.",
  1158→                    evidence=f"Key type: {key_type}, ~{bit_length} bits",
  1159→                    asset=hostname, points_deducted=8,
  1160→                    remediation=f"Generate a new key: ssh-keygen -t ed25519 -C 'your@email.com'. Or use RSA 4096: ssh-keygen -t rsa -b 4096",
  1161→                ))
  1162→            elif bit_length < 3072:
  1163→                findings.append(Finding(
  1164→                    title=f"RSA SSH key could be stronger: {priv_name} (~{bit_length} bits)",
  1165→                    severity="medium", category="ssh_key_strength",
  1166→                    module="doctor",
  1167→                    description=f"RSA key {priv_name} is approximately {bit_length} bits. NIST recommends 3072+ bits for RSA.",
  1168→                    evidence=f"Key type: {key_type}, ~{bit_length} bits",
  1169→                    asset=hostname, points_deducted=4,
  1170→                    remediation=f"Consider regenerating: ssh-keygen -t ed25519 -C 'your@email.com'",
  1171→                ))
  1172→            else:
  1173→                findings.append(Finding(
  1174→                    title=f"SSH key {priv_name}: RSA {bit_length} bits (good)",
  1175→                    severity="info", category="ssh_key_strength",
  1176→                    module="doctor",
  1177→                    description=f"RSA key {priv_name} has sufficient bit length.",
  1178→                    evidence=f"{key_type} ~{bit_length} bits",
  1179→                    asset=hostname, points_deducted=0,
  1180→                    remediation="",
  1181→                ))
  1182→        elif key_type == "ssh-dss":
  1183→            # DSA keys are deprecated
  1184→            findings.append(Finding(
  1185→                title=f"Deprecated DSA SSH key: {priv_name}",
  1186→                severity="high", category="ssh_key_strength",
  1187→                module="doctor",
  1188→                description="DSA (ssh-dss) keys are deprecated and disabled by default in modern OpenSSH. They provide weak security.",
  1189→                evidence=f"Key type: {key_type}",
  1190→                asset=hostname, points_deducted=8,
  1191→                remediation=f"Generate a new key: ssh-keygen -t ed25519 -C 'your@email.com'",
  1192→            ))
  1193→        elif key_type == "ecdsa-sha2-nistp256":
  1194→            findings.append(Finding(
  1195→                title=f"SSH key {priv_name}: ECDSA P-256 (acceptable)",
  1196→                severity="info", category="ssh_key_strength",
  1197→                module="doctor",
  1198→                description="ECDSA P-256 provides ~128-bit security. Consider Ed25519 for better performance and security.",
  1199→                evidence=f"Key type: {key_type}",
  1200→                asset=hostname, points_deducted=0,
  1201→                remediation="Consider migrating to ed25519: ssh-keygen -t ed25519 -C 'your@email.com'",

  1203→        elif key_type in ("ssh-ed25519", "sk-ssh-ed25519@openssh.com"):

  1209→                evidence=f"Key type: {key_type}",
  1210→                asset=hostname, points_deducted=0,
  1211→                remediation="",
  1212→            ))
  1213→        else:
  1214→            findings.append(Finding(
  1215→                title=f"Unrecognized SSH key type: {key_type} in {priv_name}",
  1216→                severity="low", category="ssh_key_strength",

  1219→                evidence=f"Key type: {key_type}",
  1220→                asset=hostname, points_deducted=3,
