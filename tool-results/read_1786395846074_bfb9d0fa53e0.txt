   454→    def detect_bot_activity(
   455→        self, base_url: str, timeout: int = 8
   456→    ) -> BotProfile:
   457→        """Detect signs of bot armies and automated content generation.
   458→
   459→        Analyzes page content for bot signatures and behavioral patterns.
   460→        """
   461→        body = self._fetch_content(base_url, timeout=timeout)
   462→        indicators: List[InfluenceIndicator] = []
   463→        total_score = 0.0
   464→        max_possible = 0.0
   465→
   466→        for sig in BOT_SIGNATURES:
   467→            if not sig["patterns"]:
   468→                continue
   469→            matches: List[str] = []
   470→            for pattern in sig["patterns"]:
   471→                found = re.findall(pattern, body)
   472→                matches.extend(found)
   473→
   474→            if matches:
   475→                required = sig.get("requires_count", 1)
   476→                if len(matches) >= required:
   477→                    evidence = f"Pattern '{sig['name']}' matched {len(matches)} times"
   478→                    confidence = sig["confidence"]
   479→                    severity = "high" if confidence > 0.8 else "medium"
   480→                    indicators.append(InfluenceIndicator(
   481→                        indicator_type=f"bot_signature:{sig['name']}",
   482→                        confidence=confidence,
   483→                        evidence=evidence,
   484→                        severity=severity,
   485→                    ))
   486→                    total_score += confidence
   487→            max_possible += sig["confidence"]
   488→
   489→        # Check for URL patterns common in bot operations
   490→        bot_url_patterns = [
   491→            r"(?:bit\.ly|tinyurl\.com|t\.co)/[a-zA-Z0-9]+",
   492→            r"(?:telegram\.me|t\.me)/joinchat/[a-zA-Z0-9_-]+",
   493→        ]
   494→        for pattern in bot_url_patterns:
   495→            matches = re.findall(pattern, body)
   496→            if matches:
   497→                indicators.append(InfluenceIndicator(
   498→                    indicator_type="bot_url_pattern",
   499→                    confidence=0.5,
   500→                    evidence=f"Bot-typical URL shortener/link pattern: {matches[0]}",
   501→                    severity="low",
   502→                ))
   503→                total_score += 0.5
   504→                max_possible += 1.0
   505→
   506→        # Normalize to 0-1
   507→        bot_score = total_score / max_possible if max_possible > 0 else 0.0
   508→        bot_score = min(bot_score, 1.0)
   509→
   510→        # Determine behavior pattern
   511→        if bot_score > 0.7:
   512→            pattern = "coordinated_bot_army"
   513→        elif bot_score > 0.4:
   514→            pattern = "mixed_automated_content"
   515→        elif bot_score > 0.1:
   516→            pattern = "some_automated_elements"
   517→        else:
   518→            pattern = "primarily_human"
   519→
   520→        return BotProfile(
   521→            bot_score=bot_score,
   522→            indicators=indicators,
   523→            platform="web",
   524→            behavior_pattern=pattern,
   525→        )

   527→    def analyze_sentiment_manipulation(self, body: str) -> Dict[str, Any]:
   528→        """Detect sentiment manipulation techniques in page content.
   529→
   530→        Scans for emotional trigger words, fear/anger language,
   531→        and known manipulation patterns from SENTIMENT_PATTERNS.
   532→        """
   533→        results: List[Dict[str, Any]] = []
   534→        total_weight = 0.0
   535→        max_weight = 0.0
   536→
   537→        for pattern_def in SENTIMENT_PATTERNS:
   538→            matches: List[str] = []
   539→            for regex in pattern_def["indicators"]:
   540→                found = re.findall(regex, body)
   541→                matches.extend(found)
   542→
   543→            if matches:
   544→                severity_weight = {"high": 3.0, "medium": 2.0, "low": 1.0}
   545→                weight = severity_weight.get(pattern_def["severity"], 1.0)
   546→                total_weight += weight * len(matches)
   547→                max_weight += weight * 3  # normalize against expected max
   548→
   549→                results.append({
   550→                    "technique": pattern_def["name"],
   551→                    "category": pattern_def["category"],
   552→                    "severity": pattern_def["severity"],
   553→                    "match_count": len(matches),
   554→                    "cognitive_bias": pattern_def.get("cognitive_bias", ""),
   555→                    "sample_evidence": matches[:3],
   556→                })
   557→
   558→        # Calculate manipulation score 0-100
   559→        manipulation_score = (total_weight / max_weight * 100) if max_weight > 0 else 0.0
   560→        manipulation_score = min(manipulation_score, 100.0)
   561→
   562→        # Categorize dominant manipulation type
   563→        categories: Dict[str, int] = {}
   564→        for r in results:
   565→            cat = r["category"]
   566→            categories[cat] = categories.get(cat, 0) + r["match_count"]
   567→
   568→        dominant = max(categories, key=categories.get) if categories else "none"  # type: ignore[arg-type]
   569→
   570→        return {
   571→            "manipulation_score": round(manipulation_score, 1),
   572→            "manipulation_risk": self._risk_level(manipulation_score),
   573→            "techniques_detected": results,
   574→            "technique_count": len(results),
   575→            "category_breakdown": categories,
   576→            "dominant_category": dominant,
   577→            "recommendation": self._sentiment_recommendation(manipulation_score, dominant),
   578→        }

   580→    def detect_astroturfing(self, body: str) -> Dict[str, Any]:
   581→        """Detect fake grassroots (astroturfing) patterns.
   582→
   583→        Looks for coordinated messaging, artificial consensus signals,
   584→        and manufactured support indicators.
   585→        """
   586→        indicators: List[Dict[str, Any]] = []
   587→        score = 0.0
   588→
   589→        # Coordinated messaging: repeated phrases
   590→        sentences = re.split(r'[.!?]+', body)
   591→        phrase_freq: Dict[str, int] = {}
   592→        for sent in sentences:
   593→            sent = sent.strip()
   594→            if len(sent) < 20 or len(sent) > 200:
   595→                continue
   596→            phrase_freq[sent.lower()] = phrase_freq.get(sent.lower(), 0) + 1
   597→
   598→        repeated = {p: c for p, c in phrase_freq.items() if c >= 2}
   599→        if repeated:
   600→            top_phrase = max(repeated, key=repeated.get)  # type: ignore[arg-type]
   601→            indicators.append({
   602→                "type": "repeated_phrases",
   603→                "evidence": f"{len(repeated)} phrase(s) repeated, top: '{top_phrase[:80]}...' ({repeated[top_phrase]}x)",
   604→                "severity": "medium",
   605→            })
   606→            score += min(len(repeated) * 10, 30)
   607→
   608→        # Artificial consensus indicators
   609→        consensus_patterns = [
   610→            (r"(?i)\b(?:\d{1,3}(?:,\d{3})*\s*(?:people|citizens|voters|experts))\b.{0,30}(?:agree|support|sign|demand)", "statistical_claim"),
   611→            (r"(?i)\b(?:petition|movement|campaign)\b.{0,30}(?:has (?:already )?(?:\d|gained|reached|collected))", "petition_claim"),
   612→            (r"(?i)\b(?:overwhelming|vast|strong|massive)\b.{0,15}(?:majority|support|consensus|agreement)", "consensus_claim"),
   613→            (r"(?i)\b(?:polls?|survey|research)\b.{0,30}(?:shows?|reveals?|finds?|confirms?)\b.{0,20}(?:\d+\s*%)", "poll_claim"),
   614→        ]
   615→        for pattern, label in consensus_patterns:
   616→            matches = re.findall(pattern, body)
   617→            if matches:
   618→                indicators.append({
   619→                    "type": f"artificial_consensus:{label}",
   620→                    "evidence": f"Matched {len(matches)} time(s)",
   621→                    "severity": "medium",
   622→                })
   623→                score += min(len(matches) * 15, 30)
   624→
   625→        # Check for "grassroots" self-identification
   626→        grassroots = re.findall(
   627→            r"(?i)(?:grassroots|ground.up|people.{0,5}powered|community.{0,5}led|citizen.{0,5}driven)",
   628→            body,
   629→        )
   630→        if grassroots:
   631→            indicators.append({
   632→                "type": "grassroots_self_identification",
   633→                "evidence": f"'{grassroots[0]}' self-identification found",
   634→                "severity": "low",
   635→            })
   636→            score += 5
   637→
   638→        # Check for call-to-action coordination patterns
   639→        cta_patterns = [
   640→            r"(?i)(?:share|repost|retweet|forward|send)\s+(?:this|it)\s+(?:now|widely|everywhere|with everyone)",
   641→            r"(?i)(?:copy and paste|copy.{0,5}paste|forward this message)",
   642→            r"(?i)(?:tag (?:your|\d+|everyone)|share with (?:your|\d+) friends)",
   643→        ]
   644→        for pattern in cta_patterns:
   645→            if re.search(pattern, body):
   646→                indicators.append({
   647→                    "type": "coordinated_cta",
   648→                    "evidence": "Coordinated call-to-action detected",
   649→                    "severity": "high",
   650→                })
   651→                score += 25
   652→
   653→        score = min(score, 100.0)
   654→
   655→        return {
   656→            "astroturf_score": round(score, 1),
   657→            "astroturf_risk": self._risk_level(score),
   658→            "indicators": indicators,
   659→            "indicator_count": len(indicators),
   660→            "assessment": "likely_astroturf" if score > 60 else "possible_astroturf" if score > 30 else "unlikely_astroturf",
   661→        }

   663→    def analyze_information_laundering(
   664→        self,
   665→        base_url: str,
   666→        headers: Optional[Dict[str, str]] = None,
   667→        timeout: int = 8,
   668→    ) -> Dict[str, Any]:
   669→        """Detect information laundering: obscured sources, circular reporting,
   670→        and multi-hop redirection chains.
   671→        """
   672→        indicators: List[Dict[str, Any]] = []
   673→        score = 0.0
   674→        parsed = urllib.parse.urlparse(base_url)
   675→        host = parsed.hostname or ""
   676→
   677→        # Check redirect chain
   678→        redirect_chain = self._trace_redirects(base_url, timeout=timeout)
   679→        if len(redirect_chain) > 3:
   680→            indicators.append({
   681→                "type": "long_redirect_chain",
   682→                "evidence": f"{len(redirect_chain)}-hop redirect chain",
   683→                "severity": "high",
   684→            })
   685→            score += 40
   686→        elif len(redirect_chain) > 1:
   687→            indicators.append({
   688→                "type": "redirect_chain",
   689→                "evidence": f"{len(redirect_chain)}-hop redirect: {' -> '.join(redirect_chain[:4])}",
   690→                "severity": "medium",
   691→            })
   692→            score += 15
   693→
   694→        # Check for referrer masking
   695→        if headers:
   696→            if "x-frame-options" not in {k.lower() for k in headers}:
   697→                indicators.append({
   698→                    "type": "framing_allowed",
   699→                    "evidence": "No X-Frame-Options header — page can be embedded to obscure source",
   700→                    "severity": "low",
   701→                })
   702→                score += 5
   703→
   704→        # Check for anonymous hosting / WhoisGuard
   705→        whois_privacy_patterns = [
   706→            r"(?i)(?:whoisguard|domains by proxy|privacy protect|anonymous|redacted|redacted for privacy)",
   707→        ]
   708→        body = self._fetch_content(base_url, timeout=timeout)
   709→        for pattern in whois_privacy_patterns:
   710→            if re.search(pattern, body):
   711→                indicators.append({
   712→                    "type": "anonymous_hosting_signal",
   713→                    "evidence": "Domain privacy/WHOIS protection indicators found",
   714→                    "severity": "medium",
   715→                })
   716→                score += 15
   717→
   718→        # Check for circular reporting: site references itself or sister sites
   719→        # as authoritative source
   720→        circular = re.findall(
   721→            r"(?i)(?:according to|reported by|sources? (?:say|confirm|report))\b.{0,50}" + re.escape(host or "__none__"),
   722→            body,
   723→        )
   724→        if circular:
   725→            indicators.append({
   726→                "type": "self_referential",
   727→                "evidence": f"Site references itself as authority source ({len(circular)} time(s))",
   728→                "severity": "high",
   729→            })
   730→            score += 30
   731→
   732→        # Check for content scraping indicators
   733→        scrape_patterns = [
   734→            r"(?i)(?:source|originally published|reposted|syndicated from|via)",
   735→            r"(?i)\b(?:read more|continue reading|full article)\b.{0,30}(?:at|on|here)",
   736→        ]
   737→        for pattern in scrape_patterns:
   738→            if re.search(pattern, body):
   739→                indicators.append({
   740→                    "type": "content_aggregation_signal",
   741→                    "evidence": "Content appears to be aggregated/syndicated",
   742→                    "severity": "low",
   743→                })
   744→                score += 5
   745→
   746→        score = min(score, 100.0)
   747→
   748→        return {
   749→            "laundering_score": round(score, 1),
   750→            "laundering_risk": self._risk_level(score),
   751→            "redirect_chain": redirect_chain,
   752→            "indicators": indicators,
   753→            "indicator_count": len(indicators),
   754→            "assessment": "likely_laundering" if score > 60 else "possible_laundering" if score > 30 else "unlikely_laundering",
   755→        }

   845→    def assess_cognitive_impact(
   846→        self,
   847→        target: str,
   848→        sentiment: Dict[str, Any],
   849→        bot_profile: BotProfile,
   850→        astroturf: Dict[str, Any],
   851→    ) -> Dict[str, Any]:
   852→        """Assess potential cognitive impact on visitors.
   853→
   854→        Combines manipulation, bot, and astroturf scores to estimate
   855→        the likely cognitive effect on a typical visitor.
   856→        """
   857→        manipulation_score = sentiment.get("manipulation_score", 0)
   858→        bot_score = bot_profile.bot_score * 100
   859→        astroturf_score = astroturf.get("astroturf_score", 0)
   860→
   861→        # Weighted cognitive impact assessment
   862→        impact_score = (
   863→            manipulation_score * 0.4
   864→            + bot_score * 0.3
   865→            + astroturf_score * 0.3
   866→        )
   867→        impact_score = min(impact_score, 100.0)
   868→
   869→        # Determine vulnerability level
   870→        if impact_score > 75:
   871→            vuln_level = "extremely_high"
   872→            description = (
   873→                "This content employs multiple sophisticated manipulation techniques. "
   874→                "Visitors are at extreme risk of being misinformed or radicalized."
   875→            )
   876→        elif impact_score > 50:
   877→            vuln_level = "high"
   878→            description = (
   879→                "This content uses multiple influence techniques. "
   880→                "Visitors with lower media literacy are likely to be affected."
   881→            )
   882→        elif impact_score > 25:
   883→            vuln_level = "moderate"
   884→            description = (
   885→                "Some manipulation indicators present. "
   886→                "Cautious evaluation is recommended."
   887→            )
   888→        elif impact_score > 10:
   889→            vuln_level = "low"
   890→            description = (
   891→                "Minimal manipulation indicators detected. "
   892→                "Standard critical thinking applies."
   893→            )
   894→        else:
   895→            vuln_level = "negligible"
   896→            description = (
   897→                "No significant cognitive security threats detected."
   898→            )
   899→

   966→    def _trace_redirects(
   967→        self, base_url: str, timeout: int = 8
   968→    ) -> List[str]:
   969→        """Follow redirects and return the full chain of URLs."""
   970→        chain: List[str] = []
   971→        url = base_url
   972→        for _ in range(10):  # max 10 hops
   973→            chain.append(url)
   974→            try:
   975→                req = urllib.request.Request(
   976→                    url, headers={"User-Agent": self._ua}
   977→                )
   978→                opener = urllib.request.build_opener(
   979→                    urllib.request.HTTPRedirectHandler
   980→                )
   981→                # Manually handle redirect to capture chain
   982→                resp = urllib.request.urlopen(req, timeout=timeout)
   983→                final_url = resp.url
   984→                if final_url != url and final_url not in chain:
   985→                    chain.append(final_url)
   986→                break
   987→            except urllib.error.HTTPError as e:
   988→                if e.code in (301, 302, 303, 307, 308):
   989→                    location = e.headers.get("Location", "")
   990→                    if location:
   991→                        url = urllib.parse.urljoin(url, location)
   992→                        continue
   993→                break
   994→            except (urllib.error.URLError, socket.timeout, OSError):
   995→                break
   996→        return chain
