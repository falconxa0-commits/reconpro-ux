    57→  ok: boolean;
    58→  verified: boolean;
    59→  expired: boolean;
    60→  broadcast: Broadcast | null;
    61→  error?: string;
    62→}

    97→// ─── Component ────────────────────────────────────────────────────
    98→
    99→export function BroadcastCenterPanel() {
   100→  // State
   101→  const [broadcasts, setBroadcasts] = useState<Broadcast[]>([]);
   102→  const [activeBroadcasts, setActiveBroadcasts] = useState<Broadcast[]>([]);
   103→  const [loading, setLoading] = useState(true);
   104→  const [issuing, setIssuing] = useState(false);
   105→  const [confirmIssue, setConfirmIssue] = useState(false);
   106→  const [expandedId, setExpandedId] = useState<string | null>(null);
   107→  const [filterPriority, setFilterPriority] = useState<Priority | ''>('');
   108→  const [filterChannel, setFilterChannel] = useState<Channel | ''>('');
   109→  const [verifyId, setVerifyId] = useState('');
   110→  const [verifyResult, setVerifyResult] = useState<VerifyResult | null>(null);
   111→  const [verifying, setVerifying] = useState(false);
   112→  const [bannerIndex, setBannerIndex] = useState(0);
   113→  const [copiedId, setCopiedId] = useState<string | null>(null);
   114→  const [tab, setTab] = useState<'issue' | 'history' | 'channels' | 'verify' | 'stats'>('issue');
   115→
   116→  // Form state
   117→  const [formTitle, setFormTitle] = useState('');
   118→  const [formBody, setFormBody] = useState('');
   119→  const [formPriority, setFormPriority] = useState<Priority>('INFO');
   120→  const [formChannel, setFormChannel] = useState<Channel>('web');
   121→  const [formScope, setFormScope] = useState<Scope>('all');
   122→
   123→  const bannerRef = useRef<ReturnType<typeof setInterval> | undefined>(undefined);
   124→
   125→  // ── Data fetching ────────────────────────────────────────────────
   126→
   127→  const fetchBroadcasts = useCallback(async () => {
   128→    try {
   129→      const params = new URLSearchParams();
   130→      if (filterPriority) params.set('priority', filterPriority);
   131→      if (filterChannel) params.set('channel', filterChannel);
   132→      const res = await fetch(`/api/broadcast?${params}`);
   133→      const data = await res.json();
   134→      if (data.ok) setBroadcasts(data.broadcasts);
   135→    } catch { /* silent */ }
   136→  }, [filterPriority, filterChannel]);
   137→
   138→  const fetchActive = useCallback(async () => {
   139→    try {
   140→      const res = await fetch('/api/broadcast/active');
   141→      const data = await res.json();
   142→      if (data.ok) setActiveBroadcasts(data.broadcasts);
   143→    } catch { /* silent */ }
   144→  }, []);
   145→
   146→  useEffect(() => {
   147→    Promise.all([fetchBroadcasts(), fetchActive()]).then(() => setLoading(false));
   148→  }, [fetchBroadcasts, fetchActive]);
   149→
   150→  // ── Banner auto-scroll ───────────────────────────────────────────
   151→
   152→  useEffect(() => {
   153→    const criticalActive = activeBroadcasts.filter(
   154→      (b) => b.priority === 'CRITICAL' || b.priority === 'SOVEREIGN'
   155→    );
   156→    if (criticalActive.length === 0) return;
   157→    bannerRef.current = setInterval(() => {
   158→      setBannerIndex((i) => (i + 1) % criticalActive.length);
   159→    }, 4000);
   160→    return () => clearInterval(bannerRef.current);
   161→  }, [activeBroadcasts]);
   162→
   163→  // ── Issue broadcast ──────────────────────────────────────────────
   164→
   165→  const handleIssue = async () => {
   166→    if (!confirmIssue) {
   167→      setConfirmIssue(true);
   168→      setTimeout(() => setConfirmIssue(false), 5000);
   169→      return;
   170→    }
   171→    setIssuing(true);
   172→    setConfirmIssue(false);
   173→    try {
   174→      const res = await fetch('/api/broadcast', {
   175→        method: 'POST',
   176→        headers: { 'Content-Type': 'application/json' },
   177→        body: JSON.stringify({
   178→          priority: formPriority,
   179→          title: formTitle,
   180→          body: formBody,
   181→          channel: formChannel,
   182→          targetScope: formScope,
   183→        }),
   184→      });
   185→      const data = await res.json();
   186→      if (data.ok) {
   187→        setFormTitle('');
   188→        setFormBody('');
   189→        setFormPriority('INFO');
   190→        setFormChannel('web');
   191→        setFormScope('all');
   192→        fetchBroadcasts();
   193→        fetchActive();
   194→      }
   195→    } catch { /* silent */ }
   196→    setIssuing(false);
   197→  };
   198→
   199→  // ── Verify broadcast ─────────────────────────────────────────────
   200→
   201→  const handleVerify = async () => {
   202→    if (!verifyId.trim()) return;
   203→    setVerifying(true);
   204→    try {
   205→      const res = await fetch(`/api/broadcast/verify/${verifyId.trim()}`);
   206→      const data = await res.json();
   207→      setVerifyResult(data);
   208→    } catch {
   209→      setVerifyResult({ ok: false, verified: false, expired: false, broadcast: null, error: 'Network error' });
   210→    }
   211→    setVerifying(false);
   212→  };
   213→
   214→  // ── Copy to clipboard ────────────────────────────────────────────
   215→
   216→  const copyId = (id: string) => {
   217→    navigator.clipboard.writeText(id);
   218→    setCopiedId(id);
   219→    setTimeout(() => setCopiedId(null), 2000);
   220→  };
   221→
   222→  // ── Quick fill from template ─────────────────────────────────────
   223→
   224→  const fillTemplate = (tmpl: { title: string; body: string; priority: Priority; channel: Channel; targetScope: Scope }) => {
   225→    setFormTitle(tmpl.title);
   226→    setFormBody(tmpl.body);
   227→    setFormPriority(tmpl.priority);
   228→    setFormChannel(tmpl.channel);
   229→    setFormScope(tmpl.targetScope);
   230→    setTab('issue');
   231→  };
   232→
   233→  // ── Stats computation ────────────────────────────────────────────
   234→
   235→  const stats = {
   236→    total: broadcasts.length,
   237→    byPriority: {
   238→      INFO: broadcasts.filter((b) => b.priority === 'INFO').length,
   239→      WARNING: broadcasts.filter((b) => b.priority === 'WARNING').length,
   240→      CRITICAL: broadcasts.filter((b) => b.priority === 'CRITICAL').length,
   241→      SOVEREIGN: broadcasts.filter((b) => b.priority === 'SOVEREIGN').length,
   242→    },
   243→    byChannel: {
   244→      cli: broadcasts.filter((b) => b.channel === 'cli').length,
   245→      web: broadcasts.filter((b) => b.channel === 'web').length,
   246→      email: broadcasts.filter((b) => b.channel === 'email').length,
   247→      slack: broadcasts.filter((b) => b.channel === 'slack').length,
   248→      pagerduty: broadcasts.filter((b) => b.channel === 'pagerduty').length,
   249→      webhook: broadcasts.filter((b) => b.channel === 'webhook').length,
   250→    },
   251→    active: activeBroadcasts.length,
   252→  };
   253→
   254→  const criticalActive = activeBroadcasts.filter(
   255→    (b) => b.priority === 'CRITICAL' || b.priority === 'SOVEREIGN'
   256→  );
   257→  const bannerBroadcast = criticalActive[bannerIndex] || activeBroadcasts[0];
   258→
   259→  // ── Channel status (simulated) ───────────────────────────────────
   260→
   261→  const channelStatus = (ch: Channel) => {
   262→    const count = broadcasts.filter((b) => b.channel === ch).length;
   263→    const lastBc = broadcasts.find((b) => b.channel === ch);
   264→    return { count, lastAt: lastBc?.issuedAt || null, active: count > 0 };
   265→  };
   266→
   267→  // ═══════════════════════════════════════════════════════════════
   268→  // RENDER
   269→  // ═══════════════════════════════════════════════════════════════
   270→
   271→  if (loading) {
   272→    return (
   273→      <div className="flex items-center justify-center h-96">
   274→        <motion.div animate={{ rotate: 360 }} transition={{ duration: 2, repeat: Infinity, ease: 'linear' as const }}>
   275→          <Radio className="w-8 h-8 text-amber-400" />
   276→        </motion.div>
   277→        <span className="ml-3 text-zinc-400 text-sm">Initializing Echo-Sign Protocol...</span>
   278→      </div>
   279→    );
   280→  }
   281→
   282→  return (
   283→    <div className="space-y-4">
   284→      {/* ═══ A. ACTIVE BROADCASTS BANNER ═══ */}
   285→      <AnimatePresence>
   286→        {bannerBroadcast && (
   287→          <motion.div
   288→            initial={{ opacity: 0, y: -20 }}
   289→            animate={{ opacity: 1, y: 0 }}
   290→            exit={{ opacity: 0, y: -20 }}
   291→            className={cn(
   292→              'relative overflow-hidden rounded-lg border p-4',
   293→              bannerBroadcast.priority === 'SOVEREIGN'
   294→                ? 'bg-gradient-to-r from-yellow-900/40 via-amber-900/30 to-yellow-900/40 border-yellow-500/50'
   295→                : 'bg-gradient-to-r from-red-900/40 via-red-800/30 to-red-900/40 border-red-500/50'
   296→            )}
   297→          >
   298→            {criticalActive.length > 1 && (
   299→              <div className="absolute top-2 right-3 text-xs text-zinc-500">
   300→                {bannerIndex + 1}/{criticalActive.length}
   301→              </div>
   302→            )}
   303→            <div className="flex items-center gap-3">
   304→              <motion.div
   305→                animate={{ opacity: [1, 0.3, 1] }}
   306→                transition={{ duration: 1.5, repeat: Infinity }}
   307→              >
   308→                <Zap className="w-5 h-5 text-amber-400" />
   309→              </motion.div>
   310→              <span className={cn(
   311→                'text-xs font-bold px-2 py-0.5 rounded',
   312→                bannerBroadcast.priority === 'SOVEREIGN' ? 'bg-yellow-500/20 text-yellow-300' : 'bg-red-500/20 text-red-400'
   313→              )}>
   314→                {bannerBroadcast.priority}
   315→              </span>
   316→              <span className="text-white font-semibold text-sm truncate flex-1">
   317→                {bannerBroadcast.title}
   318→              </span>
   319→              <span className="text-zinc-400 text-xs hidden sm:block">
   320→                {bannerBroadcast.issuedBy.split('@')[0]}
   321→              </span>
   322→              <span className="text-zinc-500 text-xs">
   323→                {new Date(bannerBroadcast.issuedAt).toLocaleTimeString()}
   324→              </span>
   325→              {bannerBroadcast.verified ? (
   326→                <ShieldCheck className="w-4 h-4 text-emerald-400" />
   327→              ) : (
   328→                <XCircle className="w-4 h-4 text-red-400" />
   329→              )}
   330→            </div>
   331→          </motion.div>
   332→        )}
   333→      </AnimatePresence>
   334→
   335→      {/* ═══ HEADER ═══ */}
   336→      <div className="flex items-center gap-3">
   337→        <div className="p-2 rounded-lg bg-amber-500/10 border border-amber-500/20">
   338→          <Radio className="w-5 h-5 text-amber-400" />
   339→        </div>
   340→        <div>
   341→          <h2 className="text-lg font-bold text-white">Echo-Sign Broadcast Protocol</h2>
   342→          <p className="text-xs text-zinc-500">Global Sovereign Broadcast · Ed25519 Signed Security Bulletins</p>
   343→        </div>
   344→        <div className="ml-auto flex items-center gap-2">
   345→          <div className="flex items-center gap-1.5 px-2.5 py-1 rounded-full bg-emerald-500/10 border border-emerald-500/20">
   346→            <div className="w-1.5 h-1.5 rounded-full bg-emerald-400 animate-pulse" />
   347→            <span className="text-[10px] text-emerald-400 font-medium">PROTOCOL ACTIVE</span>
   348→          </div>
   349→          {activeBroadcasts.length > 0 && (
   350→            <div className="flex items-center gap-1.5 px-2.5 py-1 rounded-full bg-amber-500/10 border border-amber-500/20">
   351→              <Megaphone className="w-3 h-3 text-amber-400" />
   352→              <span className="text-[10px] text-amber-400 font-medium">{activeBroadcasts.length} LIVE</span>
   353→            </div>
   354→          )}
   355→        </div>
   356→      </div>
   357→
   358→      {/* ═══ TAB NAV ═══ */}
   359→      <div className="flex gap-1 bg-zinc-900/50 rounded-lg p-1 border border-zinc-800/50">
   360→        {([
   361→          ['issue', Megaphone, 'Issue Broadcast'],
   362→          ['history', FileText, 'History'],
   363→          ['channels', Radio, 'Channels'],
   364→          ['verify', Shield, 'Verify'],
   365→          ['stats', Zap, 'Stats'],
   366→        ] as const).map(([key, Icon, label]) => (
   367→          <button
   368→            key={key}
   369→            onClick={() => setTab(key)}
   370→            className={cn(
   371→              'flex-1 flex items-center justify-center gap-1.5 py-2 px-3 rounded-md text-xs font-medium transition-all',
   372→              tab === key
   373→                ? 'bg-amber-500/10 text-amber-400 border border-amber-500/20'
   374→                : 'text-zinc-500 hover:text-zinc-300 hover:bg-zinc-800/50'
   375→            )}
   376→          >
   377→            <Icon className="w-3.5 h-3.5" />
   378→            <span className="hidden sm:inline">{label}</span>
   379→          </button>
   380→        ))}
   381→      </div>
   382→
   383→      {/* ═══ TAB CONTENT ═══ */}
   384→      <AnimatePresence mode="wait">
   385→        <motion.div key={tab} initial={{ opacity: 0, y: 8 }} animate={{ opacity: 1, y: 0 }} exit={{ opacity: 0, y: -8 }} transition={{ duration: 0.15 }}>
   386→
   387→          {/* ─── B. ISSUE BROADCAST ─── */}
   388→          {tab === 'issue' && (
   389→            <div className="space-y-4">
   390→              {/* Template quick-fill */}
   391→              <div className="rounded-lg border border-zinc-800/50 bg-zinc-900/30 p-3">
   392→                <div className="flex items-center gap-2 mb-2.5">
   393→                  <Zap className="w-3.5 h-3.5 text-amber-400" />
   394→                  <span className="text-xs font-semibold text-zinc-300">Quick Templates</span>
   395→                </div>
   396→                <div className="grid grid-cols-2 lg:grid-cols-4 gap-2">
   397→                  {[
   398→                    { label: 'Zero-Day Disclosure', t: { title: '[ZERO-DAY] Critical Vulnerability Disclosure', body: 'A critical zero-day vulnerability has been identified. All tenants should review attack surface and apply emergency mitigations.', priority: 'SOVEREIGN' as Priority, channel: 'web' as Channel, targetScope: 'all' as Scope } },
   399→                    { label: 'Platform Maintenance', t: { title: 'Scheduled Platform Maintenance', body: 'ReconPro will undergo scheduled maintenance. Scans may be queued during the window.', priority: 'INFO' as Priority, channel: 'web' as Channel, targetScope: 'all' as Scope } },
   400→                    { label: 'Emergency Patch', t: { title: '[EMERGENCY PATCH] Immediate Action Required', body: 'Emergency security patch released. All enterprise tenants must update within 24 hours.', priority: 'CRITICAL' as Priority, channel: 'email' as Channel, targetScope: 'enterprise' as Scope } },
   401→                    { label: 'Threat Advisory', t: { title: 'Threat Advisory: Active Campaign', body: 'Active adversarial campaign detected targeting ReconPro tenants. Rotate API keys and enable MFA.', priority: 'WARNING' as Priority, channel: 'slack' as Channel, targetScope: 'all' as Scope } },
   402→                  ].map((tmpl) => (
   403→                    <button
   404→                      key={tmpl.label}
   405→                      onClick={() => fillTemplate(tmpl.t)}
   406→                      className="text-left p-2.5 rounded-md bg-zinc-800/40 border border-zinc-700/30 hover:border-amber-500/30 hover:bg-amber-500/5 transition-all group"
   407→                    >
   408→                      <div className="text-[11px] font-semibold text-zinc-300 group-hover:text-amber-300">{tmpl.label}</div>
   409→                      <div className="flex items-center gap-1 mt-1">
   410→                        <span className={cn('text-[9px] px-1.5 py-0.5 rounded font-bold', PRIORITY_CONFIG[tmpl.t.priority].bg, PRIORITY_CONFIG[tmpl.t.priority].color)}>{tmpl.t.priority}</span>
   411→                        <span className="text-[9px] text-zinc-600">{CHANNEL_LABELS[tmpl.t.channel]}</span>
   412→                      </div>
   413→                    </button>
   414→                  ))}
   415→                </div>
   416→              </div>
   417→
   418→              {/* Form */}
   419→              <div className="rounded-lg border border-zinc-800/50 bg-zinc-900/30 p-4 space-y-4">
   420→                {/* Title */}
   421→                <div>
   422→                  <label className="text-xs font-medium text-zinc-400 mb-1.5 block">Title</label>
   423→                  <input
   424→                    value={formTitle}
   425→                    onChange={(e) => setFormTitle(e.target.value)}
   426→                    placeholder="Broadcast title..."
   427→                    className="w-full bg-zinc-800/50 border border-zinc-700/50 rounded-md px-3 py-2 text-sm text-white placeholder-zinc-600 focus:outline-none focus:border-amber-500/50 focus:ring-1 focus:ring-amber-500/20"
   428→                  />
   429→                </div>
   430→                {/* Body */}
   431→                <div>
   432→                  <label className="text-xs font-medium text-zinc-400 mb-1.5 block">Body</label>
   433→                  <textarea
   434→                    value={formBody}
   435→                    onChange={(e) => setFormBody(e.target.value)}
   436→                    rows={4}
   437→                    placeholder="Broadcast message body..."
   438→                    className="w-full bg-zinc-800/50 border border-zinc-700/50 rounded-md px-3 py-2 text-sm text-white placeholder-zinc-600 focus:outline-none focus:border-amber-500/50 focus:ring-1 focus:ring-amber-500/20 resize-none font-mono"
   439→                  />
   440→                </div>
   441→                {/* Priority + Channel + Scope row */}
   442→                <div className="grid grid-cols-1 sm:grid-cols-3 gap-3">
   443→                  {/* Priority */}
   444→                  <div>
   445→                    <label className="text-xs font-medium text-zinc-400 mb-1.5 block">Priority</label>
   446→                    <div className="grid grid-cols-2 gap-1.5">
   447→                      {(Object.keys(PRIORITY_CONFIG) as Priority[]).map((p) => {
   448→                        const cfg = PRIORITY_CONFIG[p];
   449→                        const PIcon = cfg.icon;
   450→                        return (
   451→                          <button
   452→                            key={p}
   453→                            onClick={() => setFormPriority(p)}
   454→                            className={cn(
   455→                              'flex items-center gap-1.5 p-2 rounded-md text-[11px] font-semibold transition-all border',
   456→                              formPriority === p
   457→                                ? `${cfg.bg} ${cfg.border} ${cfg.color}`
   458→                                : 'bg-zinc-800/30 border-zinc-700/30 text-zinc-500 hover:text-zinc-300'
   459→                            )}
   460→                          >
   461→                            <PIcon className="w-3 h-3" />
   462→                            {p === 'SOVEREIGN' && <Crown className="w-3 h-3" />}

   467→                    </div>
   468→                  </div>
   469→                  {/* Channel */}
   470→                  <div>
   471→                    <label className="text-xs font-medium text-zinc-400 mb-1.5 block">Channel</label>
   472→                    <div className="grid grid-cols-2 gap-1.5">
   473→                      {(Object.keys(CHANNEL_LABELS) as Channel[]).map((ch) => {
   474→                        const ChIcon = CHANNEL_ICONS[ch];
   475→                        return (
   476→                          <button
   477→                            key={ch}
   478→                            onClick={() => setFormChannel(ch)}
   479→                            className={cn(
   480→                              'flex items-center gap-1.5 p-2 rounded-md text-[11px] font-semibold transition-all border',
   481→                              formChannel === ch
   482→                                ? 'bg-amber-500/10 border-amber-500/30 text-amber-400'
   483→                                : 'bg-zinc-800/30 border-zinc-700/30 text-zinc-500 hover:text-zinc-300'
   484→                            )}
   485→                          >
   486→                            <ChIcon className="w-3 h-3" />
   487→                            {CHANNEL_LABELS[ch]}
   488→                          </button>
   489→                        );
   490→                      })}
   491→                    </div>
