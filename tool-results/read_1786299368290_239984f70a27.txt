   571→def _shannon_entropy(data: str) -> float:
   572→    """Compute Shannon entropy of a string in bits."""
   573→    if not data:
   574→        return 0.0
   575→    freq: Counter = Counter(data)
   576→    length = len(data)
   577→    return -sum(
   578→        (count / length) * math.log2(count / length)
   579→        for count in freq.values()
   580→    )
   581→
   582→
   583→def _coefficient_of_variation(values: List[float]) -> Optional[float]:
   584→    """CV = stddev / mean.  Returns None if mean is zero or insufficient data."""
   585→    if len(values) < 2:
   586→        return None
   587→    m = statistics.mean(values)
   588→    if m == 0:
   589→        return None
   590→    return statistics.stdev(values) / m
   591→
   592→
   593→def _detect_periodicity(values: List[float]) -> Optional[Dict[str, Any]]:
   594→    """Simple autocorrelation-based periodicity detection.
   595→
   596→    Returns dict with 'period', 'confidence', 'mean_interval' or None.
   597→    """
   598→    n = len(values)
   599→    if n < 4:
   600→        return None
   601→
   602→    mean_val = statistics.mean(values)
   603→    if mean_val == 0:
   604→        return None
   605→
   606→    centered = [v - mean_val for v in values]
   607→    variance = sum(c * c for c in centered) / n
   608→    if variance == 0:
   609→        return None
   610→
   611→    best_lag = 0
   612→    best_corr = 0.0
   613→
   614→    for lag in range(1, n // 2):
   615→        corr_sum = sum(centered[i] * centered[i + lag] for i in range(n - lag))
   616→        corr = corr_sum / ((n - lag) * variance)
   617→        if corr > best_corr:
   618→            best_corr = corr
   619→            best_lag = lag
   620→
   621→    if best_corr < 0.4 or best_lag == 0:
   622→        return None
   623→
   624→    period_values = [values[i] for i in range(0, n, best_lag) if i < n]
   625→    if len(period_values) < 2:
   626→        return None
   627→
   628→    return {
   629→        "period": best_lag,
   630→        "confidence": round(best_corr, 3),
   631→        "mean_interval": round(statistics.mean(values), 4),
   632→        "period_mean": round(statistics.mean(period_values), 4),
   633→    }
   634→
   635→
   636→def _rounds_to_seconds_estimate(rounds: int) -> float:
   637→    """Estimate the observation window in seconds (rough timing)."""
   638→    # Each round has _PROBE_ROUNDS probes with rate limiter delays.
   639→    return rounds * _PROBE_ROUNDS * 0.15

   769→# ═══════════════════════════════════════════════════════════════════════════
   770→# 2. BEACONING DETECTION
   771→# ═══════════════════════════════════════════════════════════════════════════
   772→
   773→def _detect_beaconing(
   774→    samples: List[ProbeSample], target: str, base_url: str
   775→) -> List[Finding]:
   776→    """Detect C2 beaconing patterns: regular intervals, small payloads,
   777→    consistent timing with jitter.
   778→    """
   779→    findings: List[Finding] = []
   780→    ok_samples = [s for s in samples if s.get("ok")]
   781→
   782→    if len(ok_samples) < BEACON_PATTERNS["regular_interval"]["detection_criteria"]["min_samples"]:
   783→        return findings
   784→
   785→    intervals: List[float] = []
   786→    for i in range(1, len(ok_samples)):
   787→        dt = ok_samples[i]["timestamp"] - ok_samples[i - 1]["timestamp"]
   788→        if dt > 0:
   789→            intervals.append(dt)
   790→
   791→    if not intervals:
   792→        return findings
   793→
   794→    cv = _coefficient_of_variation(intervals)
   795→    mean_interval = statistics.mean(intervals)
   796→    resp_sizes: List[int] = [s["response_size"] for s in ok_samples]
   797→    req_sizes: List[int] = [s["request_size"] for s in ok_samples]
   798→
   799→    matched_beacons: List[str] = []
   800→    evidence_parts: List[str] = [
   801→        f"Successful samples: {len(ok_samples)}",
   802→        f"Mean interval: {mean_interval:.3f}s",
   803→    ]
   804→    if cv is not None:
   805→        evidence_parts.append(f"Interval CV: {cv:.4f}")
   806→    evidence_parts.append(
   807→        f"Response sizes: min={min(resp_sizes) if resp_sizes else 0}, "
   808→        f"max={max(resp_sizes) if resp_sizes else 0}, "
   809→        f"mean={statistics.mean(resp_sizes) if resp_sizes else 0:.0f}"
   810→    )
   811→
   812→    # Check against beacon patterns
   813→    # Regular interval beacon
   814→    criteria = BEACON_PATTERNS["regular_interval"]["detection_criteria"]
   815→    if cv is not None and cv < criteria["cv_threshold"]:
   816→        # Check if interval matches known beacon intervals
   817→        nearest_known = min(
   818→            criteria["typical_intervals"],
   819→            key=lambda t: abs(t - mean_interval),
   820→        )
   821→        if abs(nearest_known - mean_interval) / max(nearest_known, 1) < 0.30:
   822→            matched_beacons.append("regular_interval")
   823→            evidence_parts.append(
   824→                f"MATCH: Regular interval ~{nearest_known}s beacon "
   825→                f"(actual: {mean_interval:.2f}s, deviation: "
   826→                f"{abs(nearest_known - mean_interval):.2f}s)"
   827→            )
   828→
   829→    # Jittered beacon
   830→    criteria_j = BEACON_PATTERNS["jittered_beacon"]["detection_criteria"]
   831→    if (cv is not None
   832→            and criteria_j["cv_min"] <= cv <= criteria_j["cv_max"]):
   833→        matched_beacons.append("jittered_beacon")
   834→        evidence_parts.append(
   835→            f"MATCH: Jittered beacon (CV={cv:.4f} within "
   836→            f"[{criteria_j['cv_min']}, {criteria_j['cv_max']}])"
   837→        )
   838→
   839→    # High-jitter beacon
   840→    criteria_hj = BEACON_PATTERNS["high_jitter_beacon"]["detection_criteria"]
   841→    if (cv is not None
   842→            and criteria_hj["cv_min"] <= cv <= criteria_hj["cv_max"]):
   843→        matched_beacons.append("high_jitter_beacon")
   844→        evidence_parts.append(
   845→            f"MATCH: High-jitter beacon (CV={cv:.4f} within "
   846→            f"[{criteria_hj['cv_min']}, {criteria_hj['cv_max']}])"
   847→        )
   848→
   849→    # Small payload beacon
   850→    criteria_sp = BEACON_PATTERNS["small_payload_beacon"]["detection_criteria"]
   851→    if (len(resp_sizes) >= criteria_sp["min_samples"]
   852→            and max(resp_sizes) <= criteria_sp["max_response_bytes"]):
   853→        matched_beacons.append("small_payload_beacon")
   854→        evidence_parts.append(
   855→            f"MATCH: Small payload beacon (max response: {max(resp_sizes)}B "
   856→            f"<= {criteria_sp['max_response_bytes']}B)"
   857→        )
   858→
   859→    # Consistent response size beacon
   860→    criteria_cr = BEACON_PATTERNS["consistent_response_size"]["detection_criteria"]
   861→    if len(resp_sizes) >= criteria_cr["min_samples"]:
   862→        resp_cv = _coefficient_of_variation([float(s) for s in resp_sizes])
   863→        if resp_cv is not None and resp_cv < criteria_cr["response_size_cv_threshold"]:
   864→            matched_beacons.append("consistent_response_size")
   865→            evidence_parts.append(
   866→                f"MATCH: Consistent response size (CV={resp_cv:.4f})"
   867→            )
   868→
   869→    # Generate findings for matched beacons
   870→    for beacon_key in matched_beacons:
   871→        pattern = BEACON_PATTERNS[beacon_key]
   872→        findings.append(Finding(
   873→            title=f"Beacon Pattern Detected: {pattern['name']}",
   874→            severity=pattern["severity"],
   875→            category="beacon_detection",
   876→            module="signal_intelligence",
   877→            description=(
   878→                f"{pattern['description']} Observed on {target} with "
   879→                f"mean interval {mean_interval:.3f}s, CV={cv:.4f if cv else 'N/A'}. "
   880→                f"This pattern is consistent with automated C2 communication."
   881→            ),
   882→            evidence="\n".join(evidence_parts),
   883→            asset=base_url,
   884→            points_deducted=8 if pattern["severity"] in ("critical", "high") else 3,
   885→            dread_score=pattern["dread_score"],
   886→            remediation=(
   887→                "Isolate and analyze the communication endpoint. Correlate with "
   888→                "endpoint process information. Implement beacon detection rules in "
   889→                "SIEM. Block or rate-limit the endpoint if confirmed malicious."
   890→            ),
   891→        ))
   892→
   893→    return findings

   897→# 3. C2 PATTERN MATCHING
   898→# ═══════════════════════════════════════════════════════════════════════════
   899→
   900→def _match_c2_patterns(
   901→    samples: List[ProbeSample], target: str, base_url: str
   902→) -> List[Finding]:
   903→    """Match observed communication patterns against known C2 frameworks.
   904→    """
   905→    findings: List[Finding] = []
   906→    ok_samples = [s for s in samples if s.get("ok")]
   907→
   908→    if not ok_samples:
   909→        return findings
   910→
   911→    # Collect all observed response headers and body content
   912→    observed_headers: Dict[str, List[str]] = {}
   913→    observed_bodies: List[str] = []
   914→    observed_paths: List[str] = []
   915→    observed_status_codes: List[int] = []
   916→
   917→    for s in ok_samples:
   918→        for hdr, val in s.get("response_headers", {}).items():
   919→            observed_headers.setdefault(hdr.lower(), []).append(val)
   920→        if s.get("response_body"):
   921→            observed_bodies.append(s["response_body"])
   922→        if s.get("path"):
   923→            observed_paths.append(s["path"])
   924→        if s.get("status"):
   925→            observed_status_codes.append(s["status"])
   926→
   927→    # Also collect request headers from any observed
   928→    request_uas: List[str] = []
   929→    for s in ok_samples:
   930→        ua = s.get("request_headers", {}).get("user-agent", "")
   931→        if ua:
   932→            request_uas.append(ua)
   933→
   934→    for c2_key, c2_sig in C2_SIGNATURES.items():
   935→        match_score = 0
   936→        match_evidence: List[str] = []
   937→        max_possible = 0
   938→
   939→        # Check header indicators
   940→        for hdr_name, hdr_pattern in c2_sig["header_indicators"].items():
   941→            max_possible += 2
   942→            hdr_values = observed_headers.get(hdr_name.lower(), [])
   943→            for val in hdr_values:
   944→                if re.search(hdr_pattern, val, re.IGNORECASE):
   945→                    match_score += 2
   946→                    match_evidence.append(
   947→                        f"  Header [{hdr_name}] matches: \"{val[:80]}\""
   948→                    )
   949→                    break
   950→
   951→        # Check URI patterns against response body / redirect locations
   952→        for uri_pat in c2_sig["uri_patterns"]:
   953→            max_possible += 1
   954→            for path in observed_paths:
   955→                if re.search(uri_pat, path, re.IGNORECASE):
   956→                    match_score += 1
   957→                    match_evidence.append(f"  URI pattern matches: {path}")
   958→                    break
   959→            # Also check body content for references to these patterns
   960→            for body in observed_bodies[:3]:  # limit body scans
   961→                if re.search(uri_pat, body, re.IGNORECASE):
   962→                    match_score += 1
   963→                    match_evidence.append(
   964→                        f"  Body contains URI pattern: {uri_pat[:50]}"
   965→                    )
   966→                    break
   967→
   968→        # Check response size range
   969→        resp_sizes = [s["response_size"] for s in ok_samples]
   970→        if resp_sizes:
   971→            max_possible += 1
   972→            rmin, rmax = c2_sig["response_size_range"]
   973→            in_range = sum(1 for sz in resp_sizes if rmin <= sz <= rmax)
   974→            if in_range / len(resp_sizes) > 0.7:
   975→                match_score += 1
   976→                match_evidence.append(
   977→                    f"  {in_range}/{len(resp_sizes)} responses in size range "
   978→                    f"[{rmin}, {rmax}]"
   979→                )
   980→
   981→        # Only report if we have meaningful matches
   982→        if match_score > 0 and max_possible > 0:
   983→            confidence = match_score / max_possible
   984→            if confidence >= 0.2:  # 20% minimum match threshold
   985→                mitre_str = ", ".join(c2_sig.get("mitre_attack", []))
   986→                findings.append(Finding(
   987→                    title=f"C2 Framework Pattern Match: {c2_sig['name']}",
   988→                    severity=c2_sig["severity"],
   989→                    category="c2_pattern_match",
   990→                    module="signal_intelligence",
   991→                    description=(
   992→                        f"Observed HTTP traffic from {target} exhibits patterns "
   993→                        f"consistent with {c2_sig['name']} ({c2_sig['developer']}). "
   994→                        f"{c2_sig['description']} Match confidence: "
   995→                        f"{confidence:.1%}. MITRE ATT&CK: {mitre_str}."
   996→                    ),
   997→                    evidence=(
   998→                        f"C2 Framework: {c2_sig['name']}\n"
   999→                        f"Developer: {c2_sig['developer']}\n"
  1000→                        f"Match Score: {match_score}/{max_possible} "
  1001→                        f"({confidence:.1%})\n"
  1002→                        f"Matched Indicators:\n"
  1003→                        + ("\n".join(match_evidence) if match_evidence else "  (size range only)")
  1004→                        + f"\nMITRE ATT&CK: {mitre_str}"
  1005→                    ),
  1006→                    asset=base_url,
  1007→                    points_deducted=10 if c2_sig["severity"] == "critical" else 6,
  1008→                    dread_score=c2_sig["dread_score"] * confidence,
  1009→                    remediation=(
  1010→                        f"Investigate endpoint for {c2_sig['name']} artifacts: "
  1011→                        f"dropped files, persistence mechanisms, lateral movement "
  1012→                        f"indicators. Deploy YARA/Sigma rules for {c2_sig['name']}. "
  1013→                        f"Review MITRE techniques: {mitre_str}."
  1014→                    ),
  1015→                ))
  1016→
  1017→    return findings

  1020→# ═══════════════════════════════════════════════════════════════════════════
  1021→# 4. COMMUNICATION SCHEDULE EXTRACTION
  1022→# ═══════════════════════════════════════════════════════════════════════════
  1023→
  1024→def _extract_communication_schedule(
  1025→    samples: List[ProbeSample], target: str, base_url: str
  1026→) -> List[Finding]:
  1027→    """Extract the 'when' of communications — maintenance windows,
  1028→    active hours, backup schedules.
  1029→    """
  1030→    findings: List[Finding] = []
  1031→    ok_samples = [s for s in samples if s.get("ok")]
  1032→
  1033→    if len(ok_samples) < 3:
  1034→        return findings
  1035→
  1036→    # Analyze response time distribution for patterns
  1037→    response_times = [s["response_time"] for s in ok_samples]
  1038→    resp_sizes = [s["response_size"] for s in ok_samples]
  1039→    status_codes = [s["status"] for s in ok_samples]
  1040→
  1041→    # Bucket by response time (fast / normal / slow)
  1042→    fast = [s for s in ok_samples if s["response_time"] < 0.3]
  1043→    normal = [s for s in ok_samples if 0.3 <= s["response_time"] < 1.5]
  1044→    slow = [s for s in ok_samples if s["response_time"] >= 1.5]
  1045→
  1046→    # Look for error rate patterns
  1047→    error_count = sum(1 for sc in status_codes if sc >= 400)
  1048→    error_rate = error_count / len(status_codes) if status_codes else 0
  1049→
  1050→    evidence_parts = [
  1051→        f"Total probes: {len(ok_samples)}",
  1052→        f"Fast (<0.3s): {len(fast)}",
  1053→        f"Normal (0.3-1.5s): {len(normal)}",
  1054→        f"Slow (>=1.5s): {len(slow)}",
  1055→        f"Error rate: {error_rate:.1%}",
  1056→        f"Status codes: {dict(Counter(str(sc) for sc in status_codes))}",
  1057→    ]
  1058→
  1059→    if resp_sizes:
  1060→        evidence_parts.append(
  1061→            f"Response size range: {min(resp_sizes)} - {max(resp_sizes)} bytes"
  1062→        )
  1063→
  1064→    # Check for high error rate (possible maintenance window / rate limiting)
  1065→    if error_rate > 0.4:
  1066→        findings.append(Finding(
  1067→            title="High Error Rate Pattern — Possible Rate Limiting or Maintenance",
  1068→            severity="medium",
  1069→            category="comm_schedule",
  1070→            module="signal_intelligence",
  1071→            description=(

  1191→def _analyze_payload_sizes(
  1192→    samples: List[ProbeSample], target: str, base_url: str

  1194→    """Analyze request/response sizes for encrypted tunnel indicators.
  1195→    """

  1198→
  1199→    if len(ok_samples) < 3:
  1200→        return findings
  1201→
  1202→    resp_sizes = [s["response_size"] for s in ok_samples]
  1203→    bodies = [s.get("response_body", "") for s in ok_samples]
  1204→    body_entropies = [_shannon_entropy(b) for b in bodies if b]
  1205→
  1206→    mean_size = statistics.mean(resp_sizes)
  1207→    size_cv = _coefficient_of_variation([float(s) for s in resp_sizes])
  1208→    mean_entropy = statistics.mean(body_entropies) if body_entropies else 0.0

  1211→        f"Samples analyzed: {len(ok_samples)}",

  1218→        evidence_parts.append(
  1219→            f"Body entropy: min={min(body_entropies):.2f}, "
  1220→            f"max={max(body_entropies):.2f}, mean={mean_entropy:.2f} bits"
  1221→        )
  1222→
  1223→    # Check for consistent small responses (tunnel indicator)
  1224→    if (len(resp_sizes) >= 4
  1225→            and max(resp_sizes) <= 256
  1226→            and (size_cv is not None and size_cv < 0.20)):

  1232→            description=(

  1234→                f"with low size variation (CV={size_cv:.4f}). This pattern is "
  1235→                f"consistent with encrypted C2 tunnel endpoints where the server "
  1236→                f"returns minimal acknowledgment payloads. Legitimate small "
  1237→                f"responses include health checks, API heartbeats, and redirects."
  1238→            ),
  1239→            evidence="\n".join(evidence_parts),
  1240→            asset=base_url,
  1241→            points_deducted=4,
  1242→            dread_score=5.5,
  1243→            remediation=(
  1244→                "Verify the endpoint's intended purpose. Check if small responses "
  1245→                "contain meaningful data or are stub acknowledgments. Correlate "
  1246→                "with endpoint process and network flow data."
  1247→            ),
  1248→        ))
  1249→
  1250→    # Check for high-entropy responses (encrypted/compressed data)
  1251→    if mean_entropy > 5.5 and body_entropies:
  1252→        high_entropy_count = sum(1 for e in body_entropies if e > 5.0)
  1253→        findings.append(Finding(

  1267→            points_deducted=1,
  1268→            dread_score=3.0,
  1269→            remediation=(

  1271→                "Compare against known-good baselines. Investigate if "
  1272→                "entropy is anomalous for this endpoint type."
  1273→            ),
  1274→        ))
  1275→
  1276→    # Check for suspiciously uniform response sizes (fixed-block encryption)
  1277→    unique_sizes = len(set(resp_sizes))
  1278→    if len(resp_sizes) >= 6 and unique_sizes <= 2:
  1279→        findings.append(Finding(
  1280→            title="Uniform Response Sizes — Possible Fixed-Block Encryption",
  1281→            severity="medium",
  1282→            category="payload_size",
  1283→            module="signal_intelligence",
  1284→            description=(
  1285→                f"Only {unique_sizes} distinct response size(s) across "
  1286→                f"{len(resp_sizes)} responses from {target}. Uniform sizes are "
  1287→                f"characteristic of fixed-block encryption (AES-CBC, etc.) or "
  1288→                f"templated responses. Distinct sizes: {sorted(set(resp_sizes))}."

  1290→            evidence="\n".join(evidence_parts),

  1299→        ))
  1300→
  1301→    # Size profile summary
  1302→    size_histogram: Dict[str, int] = {}
  1303→    for sz in resp_sizes:
  1304→        bucket = (
  1305→            "0-128B" if sz <= 128 else
  1306→            "129-512B" if sz <= 512 else
  1307→            "513-2KB" if sz <= 2048 else
  1308→            "2KB-8KB" if sz <= 8192 else
  1309→            "8KB-32KB" if sz <= 32768 else
  1310→            ">32KB"

  1312→        size_histogram[bucket] = size_histogram.get(bucket, 0) + 1
  1313→
  1314→    evidence_parts.append(
  1315→        "Size distribution: "
  1316→        + ", ".join(f"{k}:{v}" for k, v in sorted(size_histogram.items()))

  1318→
  1319→    findings.append(Finding(

  1340→    ))
  1341→
  1342→    return findings

  1351→) -> List[Finding]:
  1352→    """Analyze UA strings for automation patterns, headless browsers, C2 agents.
  1353→    """
  1354→    findings: List[Finding] = []
  1355→
  1356→    # Collect all observed User-Agent strings
  1357→    observed_uas: List[str] = []
  1358→    for s in samples:
  1359→        ua = s.get("response_headers", {}).get("server", "")
  1360→        if ua:
  1361→            observed_uas.append(ua)

  1363→    for s in samples:

  1370→    powered_by: List[str] = []

  1376→        if pb:
  1377→            powered_by.append(pb)

  1395→    # (Server header sometimes leaks C2 agent info)
  1396→    all_header_values: List[str] = []
  1397→    for s in samples:

  1399→            all_header_values.append(val)
  1400→
  1401→    matched_ua_sigs: List[str] = []

  1405→                if re.search(pattern, val, re.IGNORECASE | re.MULTILINE):
  1406→                    if sig_key not in matched_ua_sigs:
  1407→                        matched_ua_sigs.append(sig_key)
  1408→                        evidence_parts.append(
  1409→                            f"MATCH [{sig_data['name']}]: pattern \"{pattern}\" "
  1410→                            f"found in header value \"{val[:80]}\""
  1411→                        )
  1412→                    break
  1413→
  1414→    for sig_key in matched_ua_sigs:

  1431→                f"Verify if {sig['name']} presence is expected for this service. "

  1442→    header_counter = Counter(h.lower() for h in all_headers_flat)
  1443→    unique_headers = len(header_counter)
  1444→
  1445→    # Check for suspiciously minimal headers (C2 often has minimal header sets)
  1446→    if unique_headers <= 3 and len(samples) >= 4:
  1447→        findings.append(Finding(
  1448→            title="Minimal HTTP Header Set — Possible C2 or Minimal Server",
  1449→            severity="medium",

  1473→
  1474→    # General UA profile

  1501→# ═══════════════════════════════════════════════════════════════════════════
  1502→# 7. DNS COMMUNICATION PATTERN ANALYSIS (via HTTP observation)
  1503→# ═══════════════════════════════════════════════════════════════════════════
  1504→
  1505→def _analyze_dns_comm_patterns(
  1506→    samples: List[ProbeSample], target: str, base_url: str
  1507→) -> List[Finding]:
  1508→    """Detect C2-over-DNS patterns via HTTP observation.
  1509→
  1510→    Since we can only observe HTTP traffic, we look for indirect indicators:
  1511→    - Redirect chains to suspicious domains
  1512→    - References to DNS-like encoded data in responses
  1513→    - Long subdomain-like strings in response bodies or headers
  1514→    - High-entropy encoded strings suggesting DNS tunnel data
  1515→    """
  1516→    findings: List[Finding] = []
  1517→    ok_samples = [s for s in samples if s.get("ok")]
  1518→
  1519→    dns_indicators: List[str] = []
  1520→
  1521→    # Patterns suggesting DNS tunneling activity observable via HTTP
  1522→    dns_tunnel_patterns = [
  1523→        (r"[a-z0-9-]{24,}\.[a-z]{2,}\.[a-z]{2,}", "Long subdomain label"),
  1524→        (r"[a-f0-9]{32,}\.", "Hex-encoded subdomain"),
  1525→        (r"[A-Za-z0-9+/=]{20,}\.", "Base64-like subdomain"),
  1526→        (r"dnscat2|iodine|dns2tcp", "Known DNS tunnel tool reference"),
  1527→        (r"TXT|CNAME|NULL record", "DNS record type in HTTP content"),
  1528→    ]
  1529→
  1530→    # Check response bodies and headers for DNS tunnel indicators
  1531→    for s in ok_samples:
  1532→        body = s.get("response_body", "")
  1533→        headers_str = " ".join(s.get("response_headers", {}).values())
  1534→        combined = f"{body} {headers_str}"
  1535→
  1536→        for pattern, description in dns_tunnel_patterns:
  1537→            matches = re.findall(pattern, combined, re.IGNORECASE)
  1538→            if matches:
  1539→                for m in matches[:3]:  # limit evidence
  1540→                    truncated = m[:80] if len(m) > 80 else m
  1541→                    dns_indicators.append(
  1542→                        f"[{description}] Found: \"{truncated}\" "
  1543→                        f"in {s.get('path', 'unknown')}"
  1544→                    )
  1545→
  1546→    # Check for .well-known path exploitation (DNS rebinding, etc.)
  1547→    wellknown_samples = [
  1548→        s for s in ok_samples
  1549→        if ".well-known" in s.get("path", "")
  1550→    ]
  1551→    if wellknown_samples:
  1552→        for s in wellknown_samples:
  1553→            body = s.get("response_body", "")
  1554→            if body:
  1555→                entropy = _shannon_entropy(body)
  1556→                if entropy > 4.5:
  1557→                    dns_indicators.append(
  1558→                        f"[.well-known] High-entropy content at {s['path']} "
  1559→                        f"(entropy={entropy:.2f})"
  1560→                    )
  1561→
  1562→    if dns_indicators:
  1563→        severity = "high" if len(dns_indicators) >= 3 else "medium"
