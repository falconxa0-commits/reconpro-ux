    10→results = []

    12→def test(name, fn):
    13→    start = time.time()
    14→    try:
    15→        result = fn()
    16→        dur = time.time() - start
    17→        status = "PASS" if result else "FAIL"
    18→        results.append((name, status, "OK", dur))
    19→        symbol = "\033[92mPASS\033[0m" if result else "\033[91mFAIL\033[0m"
    20→        print(f"  [{symbol}] {name} ({dur:.1f}s)")
    21→        return result
    22→    except Exception as e:
    23→        dur = time.time() - start
    24→        tb = traceback.format_exc()
    25→        results.append((name, "ERROR", tb, dur))
    26→        print(f"  [\033[91mERROR\033[0m] {name} ({dur:.1f}s): {type(e).__name__}: {e}")
    27→        return False
    28→
    29→
    30→def report():
    31→    passed = sum(1 for _, s, _, _ in results if s == "PASS")
    32→    failed = sum(1 for _, s, _, _ in results if s in ("FAIL", "ERROR"))
    33→    total = len(results)
    34→    print(f"\n{'='*70}")
    35→    print(f"RESULTS: {passed}/{total} passed, {failed} failed")
    36→    if failed:
    37→        print(f"\n\033[91mFAILURES:\033[0m")
    38→        for name, status, detail, dur in results:
    39→            if status in ("FAIL", "ERROR"):
    40→                print(f"  x {name} ({dur:.1f}s)")
    41→                # Print just the exception line, not full traceback
    42→                for line in detail.split('\n'):
    43→                    if line.strip() and not line.startswith('Traceback') and not line.startswith('  File') and not line.strip() == '^':
    44→                        print(f"    {line.strip()[:200]}")
    45→                        break
    46→    else:
    47→        print(f"\n\033[92mALL TESTS PASSED\033[0m")
    48→    print(f"{'='*70}")
    49→    return failed == 0
