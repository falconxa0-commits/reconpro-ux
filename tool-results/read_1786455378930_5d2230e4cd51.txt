    42→def _render_summary(result, title: str = "RECONPRO") -> None:
    43→    score = result.total_score
    44→    grade = result.grade
    45→    grade_color = GRADE_COLORS.get(grade, "bold bright_red")
    46→    sc = result.severity_counts
    47→    total = len(result.findings)
    48→    bar_width = 50
    49→    filled = int(score / 100 * bar_width)
    50→    bar = "█" * filled + "░" * (bar_width - filled)
    51→    lines = [
    52→        Text(f"\n  {bar} [bold {grade_color}]{score}/100 ({grade})[/{grade_color}]"),
    53→        Text(f"\n  Target: [cyan]{result.target}[/]"),
    54→        Text(f"  Modules: [dim]{', '.join(result.modules_run)}[/]"),
    55→        Text(""),
    56→        Text(
    57→            f"  Findings: [bold]{total}[/]  "
    58→            f"[bright_red]{sc.get('critical', 0)} critical[/], "
    59→            f"[red]{sc.get('high', 0)} high[/], "
    60→            f"[yellow]{sc.get('medium', 0)} medium[/], "
    61→            f"[green]{sc.get('low', 0)} low[/], "
    62→            f"[dim]{sc.get('info', 0)} info[/]",
    63→        ),
    64→    ]
    65→    if result.vibesec_score is not None:
    66→        lines.append(Text(
    67→            f"\n  VibeSec Score: [bold bright_green]{result.vibesec_score}/100 ({result.vibesec_grade})[/]"
    68→        ))
    69→    console.print(Panel(Group(*lines), border_style=grade_color,
    70→                         title=f"[bold]{title}[/bold]", title_align="left", padding=(1, 2)))

    73→def _render_findings_table(result, show_remediation: bool = False) -> None:
    74→    if not result.findings:
    75→        console.print("\n  [bright_green]No findings. Target is clean.[/]")
    76→        return
    77→    table = Table(title=f"Findings — {len(result.findings)} detected",
    78→                  border_style="bright_white", header_style="bold bright_white")
    79→    table.add_column("Severity", style="bold", width=10)
    80→    table.add_column("Module", style="cyan", width=12)
    81→    table.add_column("Category", style="magenta", width=18)
    82→    table.add_column("Finding", style="white")
    83→    table.add_column("Pts", style="yellow", width=5)
    84→    if show_remediation:
    85→        table.add_column("Fix", style="bright_green", width=50)
    86→    for f in result.findings[:80]:
    87→        sev = f.get("severity", "info")
    88→        color = SEV_COLORS.get(sev, "white")
    89→        row = [
    90→            f"[{color}]{sev.upper()}[/{color}]",
    91→            f.get("module", ""), f.get("category", ""),
    92→            f.get("title", "")[:70], str(f.get("points_deducted", 0)),
    93→        ]
    94→        if show_remediation:
    95→            row.append(f.get("remediation", "")[:50])
    96→        table.add_row(*row)
    97→    console.print(table)

   100→def _render_module_breakdown(result) -> None:
   101→    if not result.module_results:
   102→        return
   103→    t = Table(title="Module Breakdown", border_style="dim", header_style="bold dim")
   104→    t.add_column("Module", style="bold", width=16)
   105→    t.add_column("Findings", style="white", width=10)
   106→    t.add_column("Critical", style="bright_red", width=10)
   107→    t.add_column("High", style="red", width=8)
   108→    t.add_column("Medium", style="yellow", width=8)
   109→    for mod_id, mod_data in result.module_results.items():
   110→        findings = mod_data.get("findings", [])
   111→        t.add_row(mod_id.upper(), str(len(findings)),
   112→                   str(sum(1 for f in findings if f.get("severity") == "critical")),
   113→                   str(sum(1 for f in findings if f.get("severity") == "high")),
   114→                   str(sum(1 for f in findings if f.get("severity") == "medium")))
   115→    console.print(t)

   126→def _render_remediations(result) -> None:
   127→    actionable = [f for f in result.findings
   128→                  if f.get("remediation") and f.get("severity") not in ("info",)
   129→                  and f.get("points_deducted", 0) > 0]
   130→    if not actionable:
   131→        console.print("\n  [bright_green]No actionable fixes needed.[/]")
   132→        return
   133→    console.print("\n[bold bright_white]  Fix Commands:[/bold bright_white]\n")
   134→    seen = set()
   135→    for f in actionable:
   136→        fix = f.get("remediation", "").strip()
   137→        sev = f.get("severity", "info")
   138→        color = SEV_COLORS.get(sev, "white")
   139→        if fix and fix not in seen:
   140→            seen.add(fix)
   141→            console.print(f"  [{color}]{sev.upper():8}[/{color}]  {fix}")
   142→    console.print()

   145→def _output_result(result, args, title: str = "RECONPRO", show_remediation: bool = False) -> None:
   146→    json_output = getattr(args, "json_output", False)
   147→    output_file = getattr(args, "output_file", None)
   148→    if json_output:
   149→        text = json.dumps(result.to_dict(), indent=2, default=str)
   150→        if output_file:
   151→            with open(output_file, "w") as fp:
   152→                fp.write(text)
   153→            console.print(f"  [green]JSON saved: [cyan]{output_file}[/][/]")
   154→        else:
   155→            print(text)
   156→    else:
   157→        _render_summary(result, title=title)
   158→        console.print()
   159→        _render_module_breakdown(result)
   160→        console.print()
   161→        _render_findings_table(result, show_remediation=show_remediation)
   162→        if show_remediation:
   163→            _render_remediations(result)
   164→        _render_badge(result)
   165→        if output_file:
   166→            with open(output_file, "w") as fp:
   167→                fp.write(json.dumps(result.to_dict(), indent=2, default=str))
   168→            console.print(f"\n  [green]JSON saved: [cyan]{output_file}[/][/]")
   169→        # Auto-save to history
   170→        save_scan(result.to_dict())
   171→    console.print()

   185→def _spinner_wrap(msg: str, fn, *a, _json_mode: bool | None = None, **kw):
   186→    """Run a function with a spinner (suppressed in JSON mode)."""
   187→    if _json_mode is None:
   188→        _json_mode = getattr(_cli_args, 'json_output', False) if _cli_args else False
   189→    if _json_mode:
   190→        return fn(*a, **kw)
   191→    with Progress(SpinnerColumn(), TextColumn("[bold]{task.description}[/]"),
   192→                   TimeElapsedColumn(), console=console) as progress:
   193→        task = progress.add_task(msg, total=None)
   194→        result = fn(*a, **kw)
   195→        progress.update(task, completed=True)
   196→    return result

   199→def _print_findings(findings, args, title: str = "SCAN") -> None:
   200→    """Print findings from a module run, respecting JSON mode."""
   201→    if getattr(args, 'json_output', False):
   202→        out = []
   203→        for f in findings:
   204→            out.append({
   205→                "title": f.title,
   206→                "severity": f.severity,
   207→                "category": f.category,
   208→                "module": f.module,
   209→                "description": f.description,
   210→                "evidence": f.evidence,
   211→                "asset": f.asset,
   212→                "points_deducted": f.points_deducted,
   213→                "remediation": f.remediation,
   214→            })
   215→        console.print_json(json.dumps(out, indent=2, default=str))
   216→    else:
   217→        if not findings:
   218→            console.print("  [dim]No findings.[/]")
   219→        else:
   220→            for f in findings:
   221→                c = SEV_COLORS.get(f.severity, "white")
   222→                console.print(f"  [{c}]{f.severity.upper():8}[/{c}]  {f.title}")
   223→                if f.evidence:
   224→                    console.print(f"       [dim]{f.evidence[:200]}[/]")
   225→            console.print(f"  [bold]{len(findings)} finding(s).[/]")
   226→    console.print()
