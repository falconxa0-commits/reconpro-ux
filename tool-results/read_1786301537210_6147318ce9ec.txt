  1718→            "tracking-id", "correlation-id", "x-request-id",
  1719→            "x-correlation-id", "visitor-id", "session-id",
  1720→            "user-id", "recipient-id", "doc-id", "report-id",
  1721→        ]
  1722→        for tname in tracking_names:
  1723→            if re.search(rf"(?:name|http-equiv)\s*=\s*["']?{re.escape(tname)}["']?",
  1724→                        meta_attrs, re.IGNORECASE):
  1725→                content_match = re.search(
  1726→                    r'''content\s*=\s*["']([^"']+)["']''',
  1727→                    meta_attrs, re.IGNORECASE,
  1728→                )
  1729→                if content_match:
  1730→                    tracking_meta_fields.append(f"{tname}={content_match.group(1)}")
  1731→
  1732→    if tracking_meta_fields:
  1733→        findings.append(Finding(
  1734→            title="Document Metadata — Tracking IDs in HTML Meta Tags",
  1735→            severity="high",
  1736→            category="metadata-tracking",
  1737→            module=MODULE_NAME,
  1738→            description=(
  1739→                f"Document contains {len(tracking_meta_fields)} tracking-related meta fields. "
  1740→                f"These unique identifiers can link document access to specific recipients."
  1741→            ),
  1742→            evidence=_truncate_evidence("; ".join(tracking_meta_fields[:5])),
  1743→            asset=target,
  1744→            points_deducted=15,
  1745→            remediation="Remove tracking identifiers from meta tags before distribution.",
  1746→            dread_score=0.7,
  1747→        ))
  1748→
  1749→    # ── HTTP response header tracking ──
  1750→    tracking_headers: List[str] = []
  1751→    suspicious_header_names = [
  1752→        "x-request-id", "x-correlation-id", "x-trace-id",
  1753→        "x-session-id", "x-visitor-id", "x-tracking-id",
  1754→        "x-b3-traceid", "x-b3-spanid", "x-datadog-trace-id",
  1755→        "x-amz-request-id", "x-aws-request-id",
  1756→    ]
  1757→    for hname, hval in headers.items():
  1758→        if hname.lower() in suspicious_header_names:
  1759→            tracking_headers.append(f"{hname}: {hval}")
  1760→
  1761→    if tracking_headers:
  1762→        findings.append(Finding(
  1763→            title="Document Metadata — Tracking IDs in HTTP Response Headers",
  1764→            severity="medium",
  1765→            category="metadata-tracking",

  1771→            evidence="; ".join(tracking_headers[:5]),
  1772→            asset=target,
  1773→            points_deducted=8,

  1775→            dread_score=0.5,

  1780→        r'<meta\s+[^>]*name\s*=\s*["\']generator[""][^>]*content\s*=\s*["\']([^"']+)["\']',

  1782→    )

  1784→        gen_val = gen_meta.group(1)
  1785→        # Check for tracking-capable generators
  1786→        tracking_generators = [
  1787→            "mailchimp", "campaign monitor", "sendgrid", "hubspot",
  1788→            "marketo", "pardot", "activecampaign", "convertkit",
  1789→            "mailerlite", "brevo", "constant contact",
  1790→        ]
  1791→        for tg in tracking_generators:
  1792→            if tg in gen_val.lower():
  1793→                findings.append(Finding(
  1794→                    title="Document Metadata — Email Marketing Platform Generator Detected",
  1795→                    severity="medium",
  1796→                    category="metadata-tracking",
  1797→                    module=MODULE_NAME,
  1798→                    description=(
  1799→                        f"Document generator is '{gen_val}' ({tg}). Email marketing platforms "
  1800→                        f"automatically embed tracking pixels and click tracking in all generated content."
  1801→                    ),
  1802→                    evidence=f"Generator: {gen_val}",
  1803→                    asset=target,
  1804→                    points_deducted=8,
  1805→                    remediation=(

  1807→                        f"strip all tracking elements before redistribution."
  1808→                    ),
  1809→                    dread_score=0.5,
  1810→                ))
  1811→                break
  1812→
  1813→    # ── Comment-embedded tracking data ──
  1814→    html_comments = re.findall(r"<!--(.*?)-->", body, re.DOTALL)
  1815→    tracking_comments: List[str] = []

  1819→            continue

  1822→                     comment_stripped, re.IGNORECASE):
  1823→            tracking_comments.append(comment_stripped[:100])
  1824→        # Look for base64-encoded data in comments
  1825→        b64_in_comment = re.findall(r"[A-Za-z0-9+/]{40,}={0,2}", comment_stripped)
  1826→        if b64_in_comment:

  1828→
  1829→    if tracking_comments:

  1833→            category="metadata-tracking",

  1844→            dread_score=0.5,

  1849→    for hname, hval in headers.items():
  1850→        hlower = hname.lower()
  1851→        if hlower.startswith("x-") and hlower not in {
  1852→            "x-powered-by", "x-frame-options", "x-content-type-options",
  1853→            "x-xss-protection", "x-robots-tag", "x-requested-with",
  1854→        }:
  1855→            # Check if the value looks like a tracking ID (UUID, hex, long base64)
  1856→            if (re.match(r"^[a-f0-9]{8}-[a-f0-9]{4}-", hval, re.IGNORECASE) or  # UUID
  1857→                re.match(r"^[a-f0-9]{32,}$", hval, re.IGNORECASE) or  # Hex ID
  1858→                len(hval) > 50):  # Long opaque value
  1859→                custom_tracking_headers.append(f"{hname}: {hval[:60]}")
  1860→
  1861→    if len(custom_tracking_headers) >= 2:
  1862→        findings.append(Finding(
  1863→            title="Document Metadata — Multiple Suspicious Custom Headers",
  1864→            severity="low",
  1865→            category="metadata-tracking",
  1866→            module=MODULE_NAME,
  1867→            description=(
  1868→                f"Found {len(custom_tracking_headers)} custom X- headers with ID-like values. "
  1869→                f"May be used for request correlation and user tracking."
  1870→            ),
  1871→            evidence="; ".join(custom_tracking_headers[:5]),
  1872→            asset=target,
  1873→            points_deducted=3,
  1874→            remediation="Audit custom headers and remove any that serve only tracking purposes.",
  1875→            dread_score=0.3,
  1876→        ))
  1877→
  1878→    return findings

  1881→def _check_js_trackers(
  1882→    body: str, base_url: str, target: str
  1883→) -> List[Finding]:
  1884→    """Cat 6: Detect analytics, fingerprinting, and tracking scripts."""
  1885→    findings: List[Finding] = []
  1886→    if not body:
  1887→        return findings
  1888→
  1889→    # Extract all script content
  1890→    script_blocks = _RE_SCRIPT.findall(body)
  1891→    inline_js = "\n".join(script_blocks)
  1892→    all_js = body  # Also scan the full HTML for script src references
  1893→
  1894→    # ── Check against TRACKER_SIGNATURES database ──
  1895→    detected_trackers: List[Dict[str, Any]] = []
  1896→    for tracker in _compiled_trackers:
  1897→        matches = _count_matching_patterns(all_js, tracker["compiled"])
  1898→        if matches:
  1899→            detected_trackers.append({
  1900→                **tracker,
  1901→                "matched_evidence": matches[:3],

  1905→    seen_categories: set = set()

  1907→        cat = dt["category"]

  1910→            continue

  1912→        if cat == "analytics" and len([d for d in detected_trackers if d["category"] == "analytics"]) > 3:
  1913→            if dt != [d for d in detected_trackers if d["category"] == "analytics"][0]:
  1914→                continue
  1915→

  1917→

  1919→        category_count = len([d for d in detected_trackers if d["category"] == cat])
  1920→        other_names = [d["name"] for d in detected_trackers if d["category"] == cat and d["name"] != dt["name"]]
  1921→
  1922→        desc = dt["description"]
  1923→        if category_count > 1:
  1924→            desc += f" Also detected in same category: {', '.join(other_names[:3])}."
  1925→
  1926→        evidence_str = "; ".join(dt["matched_evidence"])
  1927→        if category_count > 1:
  1928→            evidence_str += f" (plus {category_count - 1} more in {cat} category)"
  1929→

  1942→                f"anonymized collection)."
  1943→            ),
  1944→            dread_score=_severity_dread(dt["severity"]),
  1945→        ))
  1946→
  1947→    # ── Check for beacon-like outbound requests in JS ──
  1948→    if inline_js:
  1949→        beacon_requests = _RE_JS_FETCH_XHR.findall(inline_js)
  1950→        # Check URLs in fetch/XHR calls
  1951→        fetch_urls = re.findall(
  1952→            r"(?:fetch|open|send)\s*\(\s*["']([^"']+)["']",
  1953→            inline_js, re.IGNORECASE,
  1954→        )
  1955→        tracking_fetch_urls: List[str] = []
  1956→        for furl in fetch_urls:
  1957→            full_url = _normalize_url(base_url, furl)
  1958→            if _RE_SUSPICIOUS_HOST.search(full_url) and not _is_same_domain(full_url, base_url):
  1959→                tracking_fetch_urls.append(furl)
  1960→
  1961→        if tracking_fetch_urls and not any(f["category"] == "js-beacon" for f in findings):
  1962→            findings.append(Finding(

  1965→                category="js-beacon",
  1966→                module=MODULE_NAME,
  1967→                description=(
  1968→                    f"JavaScript makes {len(tracking_fetch_urls)} fetch/XHR/sendBeacon requests to "

  1971→                evidence=_truncate_evidence("; ".join(tracking_fetch_urls[:5])),
  1972→                asset=target,
  1973→                points_deducted=15,
  1974→                remediation="Remove outbound tracking requests from document JavaScript.",
  1975→                dread_score=0.7,
  1976→            ))
  1977→
  1978→        # ── Check for sendBeacon specifically (harder to block) ──
  1979→        if "sendBeacon" in inline_js:
  1980→            beacon_urls = re.findall(
  1981→                r"sendBeacon\s*\(\s*["']([^"']+)["']",
  1982→                inline_js, re.IGNORECASE,
  1983→            )
  1984→            if beacon_urls:
  1985→                findings.append(Finding(
  1986→                    title="JavaScript Tracker — navigator.sendBeacon Detected",
  1987→                    severity="high",
  1988→                    category="js-beacon",
  1989→                    module=MODULE_NAME,
  1990→                    description=(
  1991→                        f"JavaScript uses navigator.sendBeacon() to {len(beacon_urls)} endpoint(s). "
  1992→                        f"sendBeacon is designed to be reliable even during page unload, making it "
  1993→                        f"difficult to block. It's commonly used for analytics tracking that persists "
  1994→                        f"even if the user navigates away quickly."
  1995→                    ),
  1996→                    evidence="; ".join(beacon_urls[:5]),
  1997→                    asset=target,
  1998→                    points_deducted=15,
  1999→                    remediation=(
  2000→                        "Remove sendBeacon calls. This API is specifically designed to guarantee delivery "

  2002→                    ),
  2003→                    dread_score=0.7,
  2004→                ))
  2005→
  2006→        # ── Check for Image.src beacons (no-JS fallback tracking) ──
  2007→        img_src_beacons = re.findall(
  2008→            r"(?:new\s+Image|document\.createElement\(['\"]img['\"])"  # noqa: handle string concat in regex

  2012→        if img_src_beacons:

  2015→                severity="medium",
  2016→                category="js-beacon",
  2017→                module=MODULE_NAME,
  2018→                description=(
  2019→                    f"JavaScript creates Image objects and sets .src to trigger HTTP requests. "
  2020→                    f"This is a classic tracking technique that works even without XHR/fetch. "
  2021→                    f"Found {len(img_src_beacons)} beacon URL(s)."
  2022→                ),
  2023→                evidence="; ".join(img_src_beacons[:5]),
  2024→                asset=target,
  2025→                points_deducted=8,
  2026→            remediation="Remove Image.src beacon patterns from JavaScript.",
  2027→            dread_score=0.5,

  2041→                f"{', '.join(d['name'] for d in detected_trackers[:10])}."

  2058→def _check_css_tracking(
  2059→    body: str, base_url: str, target: str
  2060→) -> List[Finding]:
  2061→    """Cat 7: Detect CSS-based tracking techniques."""

  2063→    if not body:
  2064→        return findings
  2065→

  2097→            desc = (
  2098→                f"CSS contains :visited pseudo-class styling ({len(visited_matches)} occurrences). "
  2099→                f"While often legitimate, :visited styles can be abused for history sniffing."
  2100→            )
  2101→
  2102→        findings.append(Finding(
  2103→            title="CSS Tracking — :visited Link Detection (History Sniffing)",

  2105→            category="css-visited-tracking",
  2106→            module=MODULE_NAME,
  2107→            description=desc,
  2108→            evidence=f":visited occurrences: {len(visited_matches)}",
  2109→            asset=target,
  2110→            points_deducted=_severity_points(severity),
  2111→            remediation=(
  2112→                "Remove :visited styling rules from documents. Modern browsers restrict :visited "

  2123→            r"document\.fonts\.(load|check|ready)",
  2124→            r"FontFace\(",
  2125→            r"font-family\s*:\s*['\"]?[A-Z][a-zA-Z]+",
  2126→        ]

  2128→            re.search(p, body, re.IGNORECASE) for p in font_enum_patterns
  2129→        )
  2130→

  2132→        unique_fonts = set(re.findall(
  2133→            r"font-family\s*:\s*['\"]?([^'";}{]+)", all_css, re.IGNORECASE
  2134→        ))
  2135→        # Filter out generic font families
  2136→        generic_families = {
  2137→            "serif", "sans-serif", "monospace", "cursive", "fantasy",
  2138→            "system-ui", "ui-serif", "ui-sans-serif", "ui-monospace",
  2139→            "ui-rounded", "emoji", "math", "fangsong",
  2140→        }
  2141→        custom_fonts = {f.strip() for f in unique_fonts if f.strip().lower() not in generic_families}
  2142→

  2144→            findings.append(Finding(

  2149→                description=(
  2150→                    f"Document declares {len(custom_fonts)} custom @font-face rules and uses JavaScript "

  2152→                    f"(via fallback rendering), a unique fingerprint can be created. Custom fonts: "
  2153→                    f"{', '.join(list(custom_fonts)[:5])}."
  2154→                ),
  2155→                evidence=f"@font-face count: {len(font_face_matches)}; custom fonts: {', '.join(list(custom_fonts)[:5])}",
  2156→                asset=target,
  2157→                points_deducted=15,
  2158→                remediation=(
  2159→                    "Remove unnecessary @font-face declarations. Avoid JavaScript font enumeration. "
  2160→                    "Use a minimal set of web-safe or self-hosted fonts."
  2161→                ),
  2162→                dread_score=0.7,
  2163→            ))
  2164→        elif len(custom_fonts) > 5:
  2165→            findings.append(Finding(
  2166→                title="CSS Tracking — Excessive Custom Font Declarations",
  2167→                severity="medium",
  2168→                category="css-font-fingerprinting",
  2169→                module=MODULE_NAME,
  2170→                description=(
  2171→                    f"Document declares {len(custom_fonts)} custom @font-face rules. Large font sets "
  2172→                    f"can be used for system fingerprinting by measuring rendering differences."
  2173→                ),
  2174→                evidence=f"Custom fonts ({len(custom_fonts)}): {', '.join(list(custom_fonts)[:5])}",
  2175→                asset=target,
  2176→                points_deducted=8,
  2177→                remediation="Reduce custom font usage to the minimum required for display.",
  2178→                dread_score=0.5,
  2179→            ))
  2180→
  2181→    # ── CSS url() beacons ──
  2182→    css_urls = _RE_CSS_URL.findall(all_css)
  2183→    tracking_css_urls: List[str] = []
  2184→    for css_url in css_urls:
  2185→        if css_url.startswith("data:"):
  2186→            continue
  2187→        full_url = _normalize_url(base_url, css_url)
  2188→        is_external = not _is_same_domain(full_url, base_url)
  2189→        has_tracking_path = bool(_RE_SUSPICIOUS_HOST.search(full_url))
  2190→        if is_external or has_tracking_path:
  2191→            tracking_css_urls.append(css_url)
  2192→
  2193→    if tracking_css_urls:
  2194→        findings.append(Finding(
  2195→            title="CSS Tracking — External URL() Requests in Stylesheets",
  2196→            severity="medium",
  2197→            category="css-beacon",
  2198→            module=MODULE_NAME,
  2199→            description=(
  2200→                f"CSS contains {len(tracking_css_urls)} url() references to external/tracking domains. "
  2201→                f"CSS url() requests fire automatically when the stylesheet is parsed, even without JavaScript. "
