    38→    38→def sanitize_target(target: str) -> str:
    39→    39→    """Strip dangerous characters and normalize a scan target.
    40→    40→
    41→    41→    Removes null bytes, control characters (except space), shell metacharacters,
    42→    42→    and excessive whitespace. Strips leading/trailing whitespace.
    43→    43→
    44→    44→    Args:
    45→    45→        target: Raw user-supplied target string.
    46→    46→
    47→    47→    Returns:
    48→    48→        Sanitized target string.
    49→    49→    """
    50→    50→    if not target:
    51→    51→        return ""
    52→    52→    # Remove null bytes
    53→    53→    result = target.replace("\x00", "")
    54→    54→    # Remove control characters (except space = 0x20, tab = 0x09)
    55→    55→    result = "".join(
    56→    56→        ch for ch in result
    57→    57→        if ord(ch) >= 0x20 or ch in ("\t",)
    58→    58→    )
    59→    59→    # Strip whitespace
    60→    60→    result = result.strip()
    61→    61→    # Remove shell metacharacters that could cause injection
    62→    62→    result = re.sub(r'[;|`$&><]', '', result)
    63→    63→    # Collapse multiple whitespace into single space
    64→    64→    result = re.sub(r'\s+', ' ', result)
    65→    65→    return result

    68→    68→def sanitize_path(path: str) -> str:
    69→    69→    """Prevent path traversal attacks.
    70→    70→
    71→    71→    Strips null bytes, resolves path traversal sequences, and limits
    72→    72→    the resulting path to a safe representation.
    73→    73→
    74→    74→    Args:
    75→    75→        path: Raw file path string.
    76→    76→
    77→    77→    Returns:
    78→    78→        Sanitized path with traversal sequences removed.
    79→    79→    """
    80→    80→    if not path:
    81→    81→        return ""
    82→    82→    # Remove null bytes
    83→    83→    result = path.replace("\x00", "")
    84→    84→    # Remove control characters
    85→    85→    result = "".join(ch for ch in result if ord(ch) >= 0x20 or ch == "\t")
    86→    86→    # Strip whitespace
    87→    87→    result = result.strip()
    88→    88→    # Decode percent-encoded dots and slashes
    89→    89→    try:
    90→    90→        result = urllib.parse.unquote(result)
    91→    91→    except Exception:
    92→    92→        pass
    93→    93→    # Collapse path separators and remove traversal
    94→    94→    # Replace backslashes with forward slashes for uniformity
    95→    95→    result = result.replace("\\", "/")
    96→    96→    # Split on /, filter out empty and traversal components
    97→    97→    parts = result.split("/")
    98→    98→    safe_parts: List[str] = []
    99→    99→    for part in parts:
   100→   100→        if part == "" or part == ".":
   101→   101→            continue
   102→   102→        if part == "..":
   103→   103→            # Don't go above root — pop last safe part if any
   104→   104→            if safe_parts:
   105→   105→                safe_parts.pop()
   106→   106→            continue
   107→   107→        safe_parts.append(part)
   108→   108→    # Rejoin
   109→   109→    result = "/".join(safe_parts)
   110→   110→    # Limit total length
   111→   111→    if len(result) > 4096:
   112→   112→        result = result[:4096]
   113→   113→    return result

   116→   116→def sanitize_filename(name: str) -> str:
   117→   117→    """Remove directory components and dangerous characters from a filename.
   118→   118→
   119→   119→    Args:
   120→   120→        name: Raw filename string.
   121→   121→
   122→   122→    Returns:
   123→   123→        Safe filename with no path components or dangerous characters.
   124→   124→    """
   125→   125→    if not name:
   126→   126→        return "unnamed"
   127→   127→    # Remove null bytes
   128→   128→    result = name.replace("\x00", "")
   129→   129→    # Take only the basename (last component after / or \\)
   130→   130→    result = result.replace("\\", "/")
   131→   131→    result = result.split("/")[-1]
   132→   132→    # Remove characters not safe for filenames
   133→   133→    # Allow: letters, digits, dots, dashes, underscores
   134→   134→    result = re.sub(r'[^a-zA-Z0-9._\-]', '_', result)
   135→   135→    # Don't allow hidden files to start with dot (security risk)
   136→   136→    if result.startswith("."):
   137→   137→        result = "_" + result[1:]
   138→   138→    # Limit length
   139→   139→    if len(result) > 255:
   140→   140→        result = result[:255]
   141→   141→    # Don't return empty string — use a fallback
   142→   142→    if not result:
   143→   143→        return "unnamed"
   144→   144→    return result

   168→   168→def sanitize_shell(text: str) -> str:
   169→   169→    """Escape shell metacharacters to prevent command injection.
   170→   170→
   171→   171→    Wraps dangerous shell characters in single quotes and handles
   172→   172→    single quotes within the text by ending the quoted section,
   173→   173→    adding an escaped single quote, and restarting the quote.
   174→   174→
   175→   175→    Args:
   176→   176→        text: Raw text that may be used in a shell context.
   177→   177→
   178→   178→    Returns:
   179→   179→        Shell-escaped string.
   180→   180→    """
   181→   181→    if not text:
   182→   182→        return ""
   183→   183→    # Remove null bytes
   184→   184→    result = text.replace("\x00", "")
   185→   185→    # Escape shell metacharacters by wrapping in single quotes
   186→   186→    # and handling embedded single quotes
   187→   187→    # Split on single quotes, wrap each part, rejoin with escaped single quotes
   188→   188→    parts = result.split("'")
   189→   189→    quoted = "'\\''".join(parts)
   190→   190→    return f"'{quoted}'"

   276→   276→def detect_secrets_in_text(text: str) -> List[Dict[str, Any]]:
   277→   277→    """Detect secrets and credentials in text using regex patterns.
   278→   278→
   279→   279→    Scans the provided text for known secret patterns including AWS keys,
   280→   280→    GitHub tokens, API keys, private keys, passwords in URLs, database
   281→   281→    connection strings, and JWT tokens.
   282→   282→
   283→   283→    Args:
   284→   284→        text: The text content to scan for secrets.
   285→   285→
   286→   286→    Returns:
   287→   287→        List of dicts with keys: type, match, line, offset, confidence.
   288→   288→        Each dict represents a detected secret.
   289→   289→    """
   290→   290→    if not text:
   291→   291→        return []
   292→   292→
   293→   293→    findings: List[Dict[str, Any]] = []
   294→   294→    lines = text.split("\n")
   295→   295→
   296→   296→    for secret_type, pattern, confidence in _SECRET_PATTERNS:
   297→   297→        for line_num, line in enumerate(lines, start=1):
   298→   298→            for match in pattern.finditer(line):
   299→   299→                matched_text = match.group(0)
   300→   300→                findings.append({
   301→   301→                    "type": secret_type,
   302→   302→                    "match": matched_text,
   303→   303→                    "line": line_num,
   304→   304→                    "offset": match.start(),
   305→   305→                    "confidence": confidence,
   306→   306→                })
   307→   307→
   308→   308→    return findings

   321→   321→def safe_json_parse(text: str) -> Tuple[Optional[Dict[str, Any]], Optional[str]]:
   322→   322→    """Safely parse JSON with size, depth, and key count limits.
   323→   323→
   324→   324→    Args:
   325→   325→        text: JSON string to parse.
   326→   326→
   327→   327→    Returns:
   328→   328→        Tuple of (parsed_dict_or_None, error_message_or_None).
   329→   329→        On success: (dict, None)
   330→   330→        On failure: (None, error_description)
   331→   331→    """
   332→   332→    if not text:
   333→   333→        return None, "Empty input"
   334→   334→
   335→   335→    # Size check
   336→   336→    if len(text) > _MAX_JSON_SIZE:
   337→   337→        return None, f"JSON input exceeds maximum size of {_MAX_JSON_SIZE} bytes"
   338→   338→
   339→   339→    # Try parsing
   340→   340→    try:
   341→   341→        data = json.loads(text)
   342→   342→    except json.JSONDecodeError as e:
   343→   343→        return None, f"Invalid JSON: {e}"
   344→   344→    except Exception as e:
   345→   345→        return None, f"JSON parse error: {e}"
   346→   346→
   347→   347→    # Must be a dict at top level
   348→   348→    if not isinstance(data, dict):
   349→   349→        return None, "JSON root must be an object (dict)"
   350→   350→
   351→   351→    # Check key count
   352→   352→    key_count = _count_keys(data)
   353→   353→    if key_count > _MAX_JSON_KEYS:
   354→   354→        return None, f"JSON has {key_count} keys, exceeding limit of {_MAX_JSON_KEYS}"
   355→   355→
   356→   356→    # Check depth
   357→   357→    depth = _measure_depth(data)
   358→   358→    if depth > _MAX_JSON_DEPTH:
   359→   359→        return None, f"JSON has depth {depth}, exceeding limit of {_MAX_JSON_DEPTH}"
   360→   360→
   361→   361→    return data, None
   362→   362→
   363→   363→
   364→   364→def _count_keys(obj: Any) -> int:
   365→   365→    """Recursively count all keys in a JSON-like structure."""
   366→   366→    if isinstance(obj, dict):
   367→   367→        count = len(obj)
   368→   368→        for v in obj.values():
   369→   369→            count += _count_keys(v)
   370→   370→        return count
   371→   371→    elif isinstance(obj, list):
   372→   372→        count = 0
   373→   373→        for item in obj:
   374→   374→            count += _count_keys(item)
   375→   375→        return count
   376→   376→    return 0
   377→   377→
   378→   378→
   379→   379→def _measure_depth(obj: Any, current: int = 1) -> int:
   380→   380→    """Measure the maximum nesting depth of a JSON-like structure."""
   381→   381→    if isinstance(obj, dict):
   382→   382→        if not obj:
   383→   383→            return current
   384→   384→        return max(_measure_depth(v, current + 1) for v in obj.values())
   385→   385→    elif isinstance(obj, list):
   386→   386→        if not obj:
   387→   387→            return current
   388→   388→        return max(_measure_depth(item, current + 1) for item in obj)
   389→   389→    return current

   392→   392→def safe_url_parse(url: str) -> Tuple[Optional[Dict[str, Any]], Optional[str]]:
   393→   393→    """Safely parse a URL with scheme whitelist and length limits.
   394→   394→
   395→   395→    Only allows http and https schemes. Enforces a maximum URL length.
   396→   396→
   397→   397→    Args:
   398→   398→        url: URL string to parse.
   399→   399→
   400→   400→    Returns:
   401→   401→        Tuple of (parsed_components_or_None, error_message_or_None).
   402→   402→        On success: (dict with scheme, netloc, path, params, query, fragment, None)
   403→   403→        On failure: (None, error_description)
   404→   404→    """
   405→   405→    if not url:
   406→   406→        return None, "Empty URL"
   407→   407→
   408→   408→    # Length check
   409→   409→    if len(url) > _MAX_URL_LENGTH:
   410→   410→        return None, f"URL exceeds maximum length of {_MAX_URL_LENGTH} characters"
   411→   411→
   412→   412→    # Parse
   413→   413→    try:
   414→   414→        parsed = urllib.parse.urlparse(url.strip())
   415→   415→    except Exception as e:
   416→   416→        return None, f"URL parse error: {e}"
   417→   417→
   418→   418→    # Scheme whitelist
   419→   419→    scheme = parsed.scheme.lower()
   420→   420→    if not scheme:
   421→   421→        return None, "URL must include a scheme (http or https)"
   422→   422→    if scheme not in _ALLOWED_SCHEMES:
   423→   423→        return None, f"URL scheme '{scheme}' is not allowed. Allowed: {', '.join(sorted(_ALLOWED_SCHEMES))}"
   424→   424→
   425→   425→    # Must have a netloc (hostname)
   426→   426→    if not parsed.netloc:
   427→   427→        return None, "URL must include a hostname"
   428→   428→
   429→   429→    return {
   430→   430→        "scheme": scheme,
   431→   431→        "netloc": parsed.netloc,
   432→   432→        "path": parsed.path,
   433→   433→        "params": parsed.params,
   434→   434→        "query": parsed.query,
   435→   435→        "fragment": parsed.fragment,
   436→   436→    }, None

   440→   440→    """Safely parse XML with entity expansion prevention.
   441→   441→
   442→   442→    Uses Python's xml.etree.ElementTree with security limits to prevent

   448→   448→    Returns:

   459→   459→
   460→   460→    # Define a custom parser that disables entity expansion

   462→   462→        """XML parser that blocks entity declarations."""

   464→   464→

   472→   472→    except ET.ParseError as e:
   473→   473→        return None, f"XML parse error: {e}"
   474→   474→    except ValueError as e:
   475→   475→        return None, f"XML security violation: {e}"
   476→   476→    except Exception as e:
   477→   477→        return None, f"XML error: {e}"

   498→   498→    _BACKUP_COUNT = 5

   500→   500→    _LOG_FILENAME = "audit.log"
   501→   501→
   502→   502→    def __init__(

   509→   509→        Args:

   511→   511→            module_name: Name of the module using this logger (for the 'module' field).
   512→   512→        """
   513→   513→        self.module_name = module_name
   514→   514→        if log_dir is None:
   515→   515→            log_dir = os.path.join(os.path.expanduser("~"), self._LOG_DIR_NAME)
   516→   516→        self.log_dir = log_dir
   517→   517→        self._log_path = os.path.join(log_dir, self._LOG_FILENAME)
   518→   518→
   519→   519→        # Ensure directory exists
   520→   520→        os.makedirs(log_dir, exist_ok=True)
   521→   521→
   522→   522→        # Set up rotating file handler
   523→   523→        self._handler = logging.handlers.RotatingFileHandler(
   524→   524→            self._log_path,
   525→   525→            maxBytes=self._MAX_BYTES,
   526→   526→            backupCount=self._BACKUP_COUNT,
   527→   527→            encoding="utf-8",
   528→   528→        )
   529→   529→        self._handler.setFormatter(logging.Formatter("%(message)s"))
   530→   530→
   531→   531→        self._logger = logging.getLogger(f"reconpro.audit.{module_name}")
   532→   532→        self._logger.setLevel(logging.DEBUG)
   533→   533→        # Avoid duplicate handlers if logger already configured
   534→   534→        if not self._logger.handlers:
   535→   535→            self._logger.addHandler(self._handler)
   536→   536→
   537→   537→    def _emit(self, level: str, event_type: str, target: str = "", details: Optional[Dict[str, Any]] = None) -> None:
   538→   538→        """Emit a structured audit log entry."""
   539→   539→        entry = {
   540→   540→            "timestamp": datetime.now(timezone.utc).isoformat(),
   541→   541→            "level": level,
   542→   542→            "event_type": event_type,
   543→   543→            "module": self.module_name,
   544→   544→            "target": sanitize_log(target),
   545→   545→            "details": details or {},
   546→   546→        }
   547→   547→        msg = json.dumps(entry, default=str)
   548→   548→        if level == "ALERT":
   549→   549→            self._logger.critical(msg)
   550→   550→        elif level == "WARN":
   551→   551→            self._logger.warning(msg)
   552→   552→        else:
   553→   553→            self._logger.info(msg)
   554→   554→
   555→   555→    def info(self, event_type: str, target: str = "", details: Optional[Dict[str, Any]] = None) -> None:
   556→   556→        """Log an INFO-level audit event."""
   557→   557→        self._emit("INFO", event_type, target, details)
   558→   558→
   559→   559→    def warn(self, event_type: str, target: str = "", details: Optional[Dict[str, Any]] = None) -> None:
   560→   560→        """Log a WARN-level audit event."""
   561→   561→        self._emit("WARN", event_type, target, details)
   562→   562→
   563→   563→    def alert(self, event_type: str, target: str = "", details: Optional[Dict[str, Any]] = None) -> None:
   564→   564→        """Log an ALERT-level audit event."""
   565→   565→        self._emit("ALERT", event_type, target, details)
   566→   566→
   567→   567→    @property
   568→   568→    def log_path(self) -> str:
   569→   569→        """Return the path to the audit log file."""
   570→   570→        return self._log_path

   572→   572→    def close(self) -> None:
   573→   573→        """Close the logger and release resources."""
   574→   574→        self._handler.close()
   575→   575→        self._logger.removeHandler(self._handler)

   580→   580→def compute_file_hash(filepath: str, algorithm: str = "sha256") -> str:
   581→   581→    """Compute the hash of a file.
   582→   582→
   583→   583→    Args:
   584→   584→        filepath: Path to the file to hash.
   585→   585→        algorithm: Hash algorithm (sha256, sha512, md5, sha1).
   586→   586→
   587→   587→    Returns:
   588→   588→        Hex-encoded hash string, or empty string on error.
   589→   589→    """
   590→   590→    if not os.path.isfile(filepath):
   591→   591→        return ""
   592→   592→    try:
   593→   593→        h = hashlib.new(algorithm)
   594→   594→        with open(filepath, "rb") as f:
   595→   595→            while True:
   596→   596→                chunk = f.read(65536)  # 64 KB chunks
   597→   597→                if not chunk:
   598→   598→                    break
   599→   599→                h.update(chunk)
   600→   600→        return h.hexdigest()
   601→   601→    except (ValueError, OSError, IOError):
   602→   602→        return ""

   626→   626→def check_dependency_integrity() -> Dict[str, Any]:
   627→   627→    """Check that reconpro has no unexpected dependencies.
   628→   628→
   629→   629→    Scans the reconpro package for import statements and reports
   630→   630→    which external (non-stdlib) packages are imported.
   631→   631→
   632→   632→    Returns:
   633→   633→        Dict with keys:
   634→   634→        - "status": "ok" or "warning"
   635→   635→        - "stdlib_imports": set of stdlib module names found
   636→   636→        - "external_imports": set of non-stdlib module names found
   637→   637→        - "details": human-readable summary
   638→   638→    """
   639→   639→    stdlib_modules = {
   640→   640→        "os", "sys", "re", "json", "hashlib", "html", "logging",
   641→   641→        "urllib", "xml", "datetime", "pathlib", "typing", "math",
   642→   642→        "collections", "itertools", "functools", "time", "socket",
   643→   643→        "ssl", "http", "subprocess", "threading", "asyncio",
   644→   644→        "io", "struct", "unittest", "argparse", "copy", "csv",
   645→   645→        "base64", "hmac", "secrets", "shutil", "tempfile",
   646→   646→        "importlib", "pkgutil", "traceback", "warnings", "contextlib",
   647→   647→        "dataclasses", "enum", "abc", "operator", "statistics",
   648→   648→        "textwrap", "difflib", "fileinput", "fnmatch", "glob",
   649→   649→        "linecache", "posixpath", "ntpath", "genericpath",
   650→   650→        "codecs", "binascii", "mimetypes", "email", "html",
   651→   651→        "xmlrpc", "multiprocessing", "concurrent", "queue",
   652→   652→        "smtplib", "ftplib", "poplib", "imaplib", "nntplib",
   653→   653→        "telnetlib", "uuid", "string", "random", "numbers",
   654→   654→        "decimal", "fractions", "array", "weakref",
   655→   655→        "types", "pdb", "profile", "cProfile", "timeit",
   656→   656→        "pprint", "reprlib", "enum", "numbers",
   657→   657→    }
   658→   658→
   659→   659→    # Read all Python files in the reconpro package
   660→   660→    reconpro_dir = os.path.dirname(os.path.abspath(__file__))
   661→   661→    external_imports: Set[str] = set()
   662→   662→    stdlib_imports: Set[str] = set()
   663→   663→
   664→   664→    import_pattern = re.compile(
   665→   665→        r'^(?:from|import)\s+([a-zA-Z_][a-zA-Z0-9_]*)',
   666→   666→        re.MULTILINE,
   667→   667→    )
   668→   668→
   669→   669→    for root, _dirs, files in os.walk(reconpro_dir):
   670→   670→        # Skip __pycache__
   671→   671→        if "__pycache__" in root:
   672→   672→            continue
   673→   673→        for fname in files:
   674→   674→            if not fname.endswith(".py"):
   675→   675→                continue
   676→   676→            filepath = os.path.join(root, fname)
   677→   677→            try:
   678→   678→                with open(filepath, "r", encoding="utf-8", errors="ignore") as f:
   679→   679→                    content = f.read()
   680→   680→            except OSError:
   681→   681→                continue
   682→   682→
   683→   683→            for m in import_pattern.finditer(content):
   684→   684→                mod = m.group(1)
   685→   685→                # Skip relative imports and reconpro internal imports
   686→   686→                if mod.startswith(".") or mod == "reconpro":
   687→   687→                    continue
   688→   688→                if mod in stdlib_modules:
   689→   689→                    stdlib_imports.add(mod)
   690→   690→                elif mod not in ("__future__", "__main__"):
   691→   691→                    external_imports.add(mod)
   692→   692→
   693→   693→    status = "ok" if not external_imports else "warning"
   694→   694→    details = (
   695→   695→        f"Found {len(stdlib_imports)} stdlib imports, "
   696→   696→        f"{len(external_imports)} external imports."
   697→   697→    )
   698→   698→
   699→   699→    return {
   700→   700→        "status": status,
   701→   701→        "stdlib_imports": stdlib_imports,
   702→   702→        "external_imports": external_imports,
   703→   703→        "details": details,
   704→   704→    }

   746→   746→def classify_threat(finding: Dict[str, Any]) -> str:
   747→   747→    """Map a finding dict to a threat category.
   748→   748→
   749→   749→    Looks at the finding's 'type', 'category', 'severity', and 'description'
   750→   750→    fields to determine the most appropriate threat category.
   751→   751→
   752→   752→    Args:
   753→   753→        finding: A dict with at least 'type' or 'category' key.
   754→   754→
   755→   755→    Returns:
   756→   756→        Threat category string (e.g., "injection", "crypto", "network", "data", "auth")
   757→   757→        or "unknown" if no match is found.
   758→   758→    """
   759→   759→    if not finding:
   760→   760→        return "unknown"
   761→   761→
   762→   762→    # Collect all text to search through
   763→   763→    search_fields: List[str] = []
   764→   764→    for key in ("type", "category", "subtype", "name"):
   765→   765→        if key in finding:
   766→   766→            search_fields.append(str(finding[key]).lower())
   767→   767→    if "description" in finding:
   768→   768→        search_fields.append(str(finding["description"]).lower())
   769→   769→    if "severity" in finding:
   770→   770→        search_fields.append(str(finding["severity"]).lower())
   771→   771→
   772→   772→    combined = " ".join(search_fields)
   773→   773→
   774→   774→    # Check each category for keyword matches
   775→   775→    for category, keywords in THREAT_CATEGORIES.items():
   776→   776→        for keyword in keywords:
   777→   777→            if keyword in combined:
   778→   778→                return category
   779→   779→
   780→   780→    # Additional heuristic mappings
   781→   781→    # SQL-related patterns
   782→   782→    sql_indicators = {"sql", "query", "select", "inject", "sqli"}
   783→   783→    if any(ind in combined for ind in sql_indicators):
   784→   784→        return "injection"
   785→   785→
   786→   786→    # XSS-related patterns
   787→   787→    xss_indicators = {"xss", "cross-site", "script", "javascript", "dom"}
   788→   788→    if any(ind in combined for ind in xss_indicators):
   789→   789→        return "injection"
   790→   790→
   791→   791→    # Crypto-related patterns
   792→   792→    crypto_indicators = {"tls", "ssl", "certificate", "cipher", "encrypt", "hash"}
   793→   793→    if any(ind in combined for ind in crypto_indicators):
   794→   794→        return "crypto"
   795→   795→
   796→   796→    # Network patterns
   797→   797→    network_indicators = {"port", "service", "open", "exposed", "firewall"}
   798→   798→    if any(ind in combined for ind in network_indicators):
   799→   799→        return "network"
   800→   800→
   801→   801→    # Data patterns
   802→   802→    data_indicators = {"data", "leak", "exposure", "pii", "personal", "gdpr"}
   803→   803→    if any(ind in combined for ind in data_indicators):
   804→   804→        return "data"
   805→   805→
   806→   806→    # Auth patterns
   807→   807→    auth_indicators = {"auth", "login", "password", "session", "token", "mfa", "credential"}
   808→   808→    if any(ind in combined for ind in auth_indicators):
   809→   809→        return "auth"
   810→   810→
   811→   811→    return "unknown"
