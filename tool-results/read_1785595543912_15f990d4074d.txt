    87→// ─── Mini severity donut (SVG) ──────────────────────────────
    88→
    89→function SeverityDonut({ data }: { data: { name: string; value: number; color: string }[] }) {
    90→  const total = data.reduce((s, d) => s + d.value, 0);
    91→  const segments = data.map((d, i) => {
    92→    const pct = total === 0 ? 0 : d.value / total;
    93→    const off = data.slice(0, i).reduce((s2, prev) => s2 + (total === 0 ? 0 : prev.value / total), 0);
    94→    return { ...d, pct, dasharray: `${pct * 283} ${283}`, offset: -off * 283 };
    95→  });
    96→  if (total === 0) return <div className="text-xs text-muted-foreground text-center py-4">No data yet</div>;
    97→  return (
    98→    <svg viewBox="0 0 120 120" className="w-32 h-32 mx-auto">
    99→      {segments.map((d) => (
   100→        <circle key={d.name} cx="60" cy="60" r="45" fill="none"
   101→          stroke={d.color} strokeWidth="18" strokeDasharray={d.dasharray}
   102→          strokeDashoffset={d.offset} strokeLinecap="round" opacity={0.85} />
   103→      ))}
   104→      <text x="60" y="56" textAnchor="middle" fill="#e6edf3" fontSize="22" fontWeight="bold" fontFamily="Geist Sans, sans-serif">{total}</text>
   105→      <text x="60" y="72" textAnchor="middle" fill="#7d8590" fontSize="9" fontFamily="Geist Sans, sans-serif">FINDINGS</text>
   106→    </svg>
   107→  );
   108→}
