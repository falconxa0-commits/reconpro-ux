  2288→    def classify_intent(self, goal: str) -> Dict[str, float]:
  2289→        """Score each intent based on the goal text.
  2290→
  2291→        Uses weighted keyword matching, proximity scoring, and modifiers.
  2292→        Returns a dict mapping intent names to float scores in [0, 1+].
  2293→        Supports 30+ intent categories.
  2294→        """
  2295→        goal_lower = goal.lower()
  2296→        # Pad for whole-word matching
  2297→        padded = f' {goal_lower} '
  2298→        scores: Dict[str, float] = {}
  2299→        words = goal_lower.split()
  2300→
  2301→        # Calculate global modifier
  2302→        modifier = 1.0
  2303→        for w in words:
  2304→            if w in self._AMPLIFIERS:
  2305→                modifier = max(modifier, self._AMPLIFIERS[w])
  2306→            if w in self._DAMPENERS:
  2307→                modifier = min(modifier, self._DAMPENERS[w])
  2308→
  2309→        for intent, keywords in self._INTENT_KEYWORDS.items():
  2310→            score = 0.0
  2311→            for pattern, weight in keywords:
  2312→                # Use whole-word matching to avoid false positives
  2313→                # (e.g., "repo" matching inside "report")
  2314→                if len(pattern.split()) > 1:
  2315→                    # Multi-word pattern: check substring (already specific enough)
  2316→                    matched = pattern in goal_lower
  2317→                else:
  2318→                    # Single-word pattern: check whole word with padding
  2319→                    matched = f' {pattern} ' in padded
  2320→                if matched:
  2321→                    # Boost if near a target-like word
  2322→                    proximity_boost = 1.0
  2323→                    pattern_idx = goal_lower.find(pattern)
  2324→                    for tw in (".com", ".io", ".net", ".org", "http", "localhost"):
  2325→                        tw_idx = goal_lower.find(tw)
  2326→                        if tw_idx >= 0 and abs(pattern_idx - tw_idx) < 40:
  2327→                            proximity_boost = 1.3
  2328→                            break
  2329→                    score += weight * proximity_boost
  2330→            scores[intent] = min(score * modifier, 2.0)
  2331→
  2332→        return scores

  2651→    def adapt_plan(
  2652→        self,
  2653→        plan: List[Dict[str, Any]],
  2654→        completed_results: List[Tuple[Dict, ToolResult]],
  2655→        current_idx: int,
  2656→    ) -> List[Dict[str, Any]]:
  2657→        """Adapt the remaining plan based on results seen so far.
  2658→
  2659→        This is the 'intelligence' of the rule engine — it inspects intermediate
  2660→        results and inserts new steps or modifies remaining ones.
  2661→        Enhanced with: adaptive deepening, correlation-aware suggestions,
  2662→        and smarter conditional logic.
  2663→        """
  2664→        additions: List[Dict[str, Any]] = []
  2665→        domain = ""
  2666→        last_score = None
  2667→        has_critical = False
  2668→        total_findings = 0
  2669→        all_findings: List[Dict] = []
  2670→        subdomains_found = False
  2671→        scan_data: Optional[Dict] = None
  2672→
  2673→        for step, result in completed_results:
  2674→            d = result.data
  2675→            if isinstance(d, dict):
  2676→                # Extract score from scan results
  2677→                if "total_score" in d:
  2678→                    last_score = d["total_score"]
  2679→                if "vibesec_score" in d:
  2680→                    last_score = d["vibesec_score"]
  2681→                # Collect findings
  2682→                findings = d.get("findings", [])
  2683→                if isinstance(findings, list):
  2684→                    all_findings.extend(findings)
  2685→                    total_findings += len(findings)
  2686→                    for f in findings:
  2687→                        if isinstance(f, dict) and f.get("severity") in ("critical", "high"):
  2688→                            has_critical = True
  2689→                # Check for subdomains
  2690→                if "subdomains" in d and len(d["subdomains"]) > 3:
  2691→                    subdomains_found = True
  2692→                # Get target from data
  2693→                if not domain:
  2694→                    domain = d.get("target", "")
  2695→                # Keep scan data reference for deepening
  2696→                if step.get("tool") in ("remote_scan", "vibesec_benchmark", "local_audit"):
  2697→                    scan_data = d
  2698→
  2699→        # Extract domain from the current step if not yet known
  2700→        if not domain and current_idx < len(plan):
  2701→            step = plan[current_idx]
  2702→            for k, v in step.get("args", {}).items():
  2703→                if k in ("target", "domain") and v:
  2704→                    domain = v.replace("https://", "").replace("http://", "").split("/")[0]
  2705→                    break
  2706→
  2707→        already_planned_tools = [s["tool"] for s in plan]
  2708→
  2709→        # If any findings found → add CVE enrichment (if not already planned)
  2710→        has_any_findings = total_findings > 0
  2711→        if has_any_findings and not any(s["tool"] == "cve_enrich" for s in plan[current_idx:]):
  2712→            if all_findings:
  2713→                try:
  2714→                    findings_json = json.dumps(all_findings[:20])
  2715→                except (TypeError, ValueError):
  2716→                    findings_json = "[]"
  2717→                reason = "Enriching findings with CVE data from NVD"
  2718→                if has_critical:
  2719→                    reason = "Critical/high findings detected — enriching with CVE data"
  2720→                additions.append({
  2721→                    "tool": "cve_enrich",
  2722→                    "args": {"findings_json": findings_json},
  2723→                    "reason": reason,
  2724→                })
  2725→
  2726→        # If many subdomains found and not already doing blitz → suggest knowledge graph
  2727→        if subdomains_found and not any(s["tool"] == "blitz_scan" for s in plan[current_idx:]):
  2728→            if domain and not any(s["tool"] == "knowledge_graph" for s in plan[current_idx:]):
  2729→                additions.append({
  2730→                    "tool": "knowledge_graph",
  2731→                    "args": {"action": "build", "target": domain},
  2732→                    "reason": "Large attack surface detected — building knowledge graph",
  2733→                })
  2734→
  2735→        # If score is very low → add knowledge graph
  2736→        if last_score is not None and last_score < 40:
  2737→            if domain and not any(s["tool"] == "knowledge_graph" for s in plan[current_idx:]):
  2738→                additions.append({
  2739→                    "tool": "knowledge_graph",
  2740→                    "args": {"action": "build", "target": domain},
  2741→                    "reason": f"Critical score ({last_score}/100) — analyzing attack surface via knowledge graph",
  2742→                })

  2745→        if last_score is not None and last_score < 50:
  2746→            if domain and not any(s["tool"] in ("adversarial_loop", "swarm_run") for s in plan + additions):
  2747→                additions.append({
  2748→                    "tool": "adversarial_loop",
  2749→                    "args": {"target": domain, "rounds": 2},

  2763→                        "reason": sug["reason"],

  2766→        # Insert additions after current step
