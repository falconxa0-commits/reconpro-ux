   400→  const findings: Finding[] = [];
   401→  const technologies: string[] = [];
   402→
   403→  const sslResult = await analyzeSSLNative(domain);
   404→
   405→  if (!sslResult.cert && !sslResult.sslConnect) {
   406→    findings.push({ title: 'SSL/TLS Connection Failed', severity: 'high', category: 'ssl',
   407→      description: `Could not establish SSL/TLS to ${domain}:443. Server may not support HTTPS.`, evidence: sslResult.error || 'TLS connection failed', asset: `${domain}:443` });
   408→    return { findings, technologies };
   409→  }
   410→
   411→  const full = sslResult.certInfo || sslResult.sslConnect;
   412→
   413→  // Subject
   414→  const subjM = full.match(/subject=([^\n]+)/);
   415→  if (subjM) findings.push({ title: `SSL Certificate Subject: ${subjM[1].trim()}`, severity: 'info', category: 'ssl',
   416→    description: `Certificate issued to: ${subjM[1].trim()}.`, evidence: `subject: ${subjM[1].trim()}`, asset: domain });
   417→
   418→  // Issuer + tech detection
   419→  const issM = full.match(/issuer=([^\n]+)/);
   420→  const issuer = issM ? issM[1].trim() : 'unknown';
   421→  if (issuer.includes("Let's Encrypt")) technologies.push("Let's Encrypt");
   422→  else if (issuer.includes('DigiCert')) technologies.push('DigiCert');
   423→  else if (issuer.includes('Sectigo') || issuer.includes('Comodo')) technologies.push('Sectigo');
   424→  else if (issuer.includes('Cloudflare')) technologies.push('Cloudflare SSL');
   425→  else if (issuer.includes('Amazon') || issuer.includes('AWS')) technologies.push('AWS ACM');
   426→  else if (issuer.includes('Google')) technologies.push('Google Trust Services');
   427→
   428→  findings.push({ title: `SSL Certificate Issuer: ${issuer}`, severity: 'info', category: 'ssl',
   429→    description: `Issued by: ${issuer}.`, evidence: `issuer: ${issuer}`, asset: domain });
   430→
   431→  // Dates + expiry
   432→  const nbM = full.match(/notBefore=(.+)/);
   433→  const naM = full.match(/notAfter=(.+)/);
   434→  if (nbM && naM) {
   435→    const nb = new Date(nbM[1].trim());
   436→    const na = new Date(naM[1].trim());
   437→    const days = Math.ceil((na.getTime() - Date.now()) / 86400000);
   438→    const lifespan = Math.ceil((na.getTime() - nb.getTime()) / 86400000);
   439→
   440→    findings.push({ title: `SSL Certificate Validity: ${nb.toISOString().split('T')[0]} — ${na.toISOString().split('T')[0]}`, severity: 'info', category: 'ssl',
   441→      description: `Certificate lifespan: ${lifespan} days. Valid from ${nb.toISOString().split('T')[0]} to ${na.toISOString().split('T')[0]}.`, evidence: `notBefore: ${nbM[1].trim()}, notAfter: ${naM[1].trim()}`, asset: domain });
   442→
   443→    if (days < 0) {
   444→      findings.push({ title: `SSL Certificate EXPIRED ${Math.abs(days)} days ago`, severity: 'critical', category: 'ssl',
   445→        description: `Certificate for ${domain} expired ${Math.abs(days)} days ago. Browsers show security warnings, automated integrations fail with TLS errors. Immediate renewal required.`, evidence: `Expired: ${na.toISOString()}`, asset: domain });
   446→    } else if (days <= 7) {
   447→      findings.push({ title: `SSL Certificate Expiring in ${days} Days — CRITICAL`, severity: 'critical', category: 'ssl',
   448→        description: `Certificate expires in ${days} days. Emergency renewal required.`, evidence: `Expires: ${na.toISOString()} (${days}d)`, asset: domain });
   449→    } else if (days <= 30) {
   450→      findings.push({ title: `SSL Certificate Expiring in ${days} Days`, severity: 'high', category: 'ssl',
   451→        description: `Certificate expires in ${days} days. Prompt renewal needed.`, evidence: `Expires: ${na.toISOString()} (${days}d)`, asset: domain });
   452→    } else if (days <= 90) {
   453→      findings.push({ title: `SSL Certificate Expiring in ${days} Days`, severity: 'medium', category: 'ssl',
   454→        description: `Certificate expires in ${days} days. Plan renewal.`, evidence: `Expires: ${na.toISOString()} (${days}d)`, asset: domain });
   455→    }
   456→
   457→    if (lifespan > 398) {
   458→      findings.push({ title: `Certificate Lifespan Exceeds 398 Days (${lifespan}d)`, severity: 'medium', category: 'ssl',
   459→        description: `Lifespan exceeds Apple/Google 398-day max. New certs must be shorter.`, evidence: `Lifespan: ${lifespan}d (max: 398)`, asset: domain });
   460→    }
   461→  }
   462→
   463→  // SAN
   464→  const sanM = full.match(/Subject Alternative Name[^\n]*\n([^\n]*)/);
   465→  if (sanM) {
   466→    const sans = sanM[1].match(/DNS:[^\s,]+/g);
   467→    if (sans) findings.push({ title: `Certificate Covers ${sans.length} Domain(s) (SAN)`, severity: 'info', category: 'ssl',
   468→      description: `SANs: ${sans.map(s=>s.replace('DNS:','')).join(', ')}.`, evidence: `SAN: ${sans.join(', ')}`, asset: domain });
   469→  }
   470→
   471→  // TLS version
   472→  const proto = sslResult.protocol || '';
   473→  if (proto.includes('TLSv1 ') || proto.includes('TLSv1.0')) {
   474→    findings.push({ title: 'Weak TLS Version: TLS 1.0 Detected', severity: 'high', category: 'ssl',
   475→      description: `TLS 1.0 deprecated (RFC 8996, 2020). Vulnerable to BEAST, POODLE, RC4. PCI-DSS prohibits. Disable immediately.`, evidence: `Protocol: ${proto}`, asset: `${domain}:443` });
   476→  } else if (proto.includes('TLSv1.1')) {
   477→    findings.push({ title: 'Deprecated TLS 1.1 Detected', severity: 'high', category: 'ssl',
   478→      description: `TLS 1.1 deprecated in 2020. Known weaknesses. PCI-DSS requires disabling. Support TLS 1.2+ only.`, evidence: `Protocol: ${proto}`, asset: `${domain}:443` });
   479→  } else if (proto.includes('TLSv1.3')) {
   480→    technologies.push('TLS 1.3');
   481→    findings.push({ title: 'Modern TLS 1.3 Negotiated', severity: 'info', category: 'ssl',
   482→      description: `TLS 1.3 — latest version with 0-RTT, mandatory forward secrecy, no weak ciphers.`, evidence: `Protocol: ${proto}`, asset: `${domain}:443` });
   483→  } else if (proto.includes('TLSv1.2')) {
   484→    technologies.push('TLS 1.2');
   485→    findings.push({ title: 'TLS 1.2 Negotiated', severity: 'info', category: 'ssl',
   486→      description: `TLS 1.2 — secure when configured with strong cipher suites. Consider enabling TLS 1.3.`, evidence: `Protocol: ${proto}`, asset: `${domain}:443` });
   487→  }
   488→
   489→  // Cipher
   490→  const cipher = sslResult.cipher || '';
   491→  const weakC = ['RC4','DES','MD5','NULL','EXPORT','3DES'];
   492→  if (weakC.some(w => cipher.toUpperCase().includes(w))) {
   493→    findings.push({ title: `Weak Cipher Suite: ${cipher}`, severity: 'high', category: 'ssl',
   494→      description: `Weak cipher (${cipher}) — vulnerable to known attacks. Use AEAD ciphers (AES-GCM, ChaCha20-Poly1305) with ECDHE.`, evidence: `Cipher: ${cipher}`, asset: `${domain}:443` });
   495→  } else if (cipher) {
   496→    findings.push({ title: `Cipher Suite: ${cipher}`, severity: 'info', category: 'ssl',
   497→      description: `Negotiated: ${cipher}.`, evidence: `Cipher: ${cipher}`, asset: `${domain}:443` });
   498→  }
   499→
   500→  return { findings, technologies };
   501→}

   504→// 5. PORT PROBING (async parallel)
   505→// ═══════════════════════════════════════════════════════════════════════
   506→const WEB_PORTS = [
   507→  { port: 80, service: 'HTTP', risk: 'low', desc: 'Standard HTTP — should redirect to HTTPS' },
   508→  { port: 443, service: 'HTTPS', risk: 'info', desc: 'Standard HTTPS' },
   509→  { port: 8080, service: 'HTTP-Alt', risk: 'high', desc: 'Dev servers, proxies, Tomcat' },
   510→  { port: 8443, service: 'HTTPS-Alt', risk: 'medium', desc: 'Admin panels, Java apps' },
   511→  { port: 3000, service: 'Dev Server', risk: 'high', desc: 'Node.js/Express dev default' },
   512→  { port: 5000, service: 'Dev Server', risk: 'high', desc: 'Flask/React dev default' },
   513→  { port: 8000, service: 'HTTP-Dev', risk: 'high', desc: 'Django, Go dev' },
   514→  { port: 8888, service: 'Jupyter/Proxy', risk: 'critical', desc: 'Jupyter Notebook or proxy' },
   515→  { port: 9090, service: 'Prometheus', risk: 'high', desc: 'Prometheus monitoring' },
   516→  { port: 9200, service: 'Elasticsearch', risk: 'critical', desc: 'Elasticsearch REST API' },
   517→  { port: 5601, service: 'Kibana', risk: 'high', desc: 'Kibana dashboard' },
   518→  { port: 27017, service: 'MongoDB', risk: 'critical', desc: 'MongoDB default port' },
   519→];
   520→
   521→async function probePorts(domain: string): Promise<Finding[]> {
   522→  const results = await Promise.all(
   523→    WEB_PORTS.map(async ({ port, service, risk, desc }) => {
   524→      const r = await safeFetch(`https://${domain}:${port}`, { timeout: 5000, method: 'HEAD' })
   525→        .catch(() => safeFetch(`http://${domain}:${port}`, { timeout: 5000, method: 'HEAD' }));
   526→      if (r && r.status > 0) {
   527→        const status = `HTTP/${r.status}`;
   528→        const srv = r.headers['server'] ? ` [${r.headers['server']}]` : '';
   529→        return { port, service, risk, desc, status: status + srv };
   530→      }
   531→      return null;
   532→    })
   533→  );
   534→
   535→  return results.filter((r): r is NonNullable<typeof r> => Boolean(r)).map(({ port, service, risk, desc, status }) => ({
   536→    title: `Port ${port}/tcp OPEN — ${service}`,
   537→    severity: risk === 'info' ? 'info' : risk, category: 'port',
   538→    description: `${service} accessible on port ${port}. ${desc}.${risk === 'critical' || risk === 'high' ? ' This should not be publicly accessible.' : ''}`,
   539→    evidence: status, asset: `${domain}:${port}`,
   540→  }));
   541→}

   544→// 6. CERTIFICATE TRANSPARENCY LOGS — crt.sh for ALL historical subdomains
   545→// ═══════════════════════════════════════════════════════════════════════
   546→async function enumerateCTLogs(domain: string): Promise<Finding[]> {
   547→  const findings: Finding[] = [];
   548→  try {
   549→    const response = await safeFetch(`https://crt.sh/?q=%25.${domain}&output=json`, { timeout: 20000, skipSSRFCheck: true });
   550→    if (!response.ok || !response.text) return findings;
   551→    const certs = JSON.parse(response.text);
   552→    // Deduplicate subdomains
   553→    const subs = new Set<string>();
   554→    for (const c of certs) {
   555→      if (c.name_value) {
   556→        for (const name of c.name_value.split('\n')) {
   557→          const clean = name.trim().replace(/^\*\./, '');
   558→          if (clean.endsWith(`.${domain}`) && clean !== domain) subs.add(clean);
   559→        }
   560→      }
   561→    }
   562→    const unique = Array.from(subs).sort();
   563→    if (unique.length > 0) {
   564→      findings.push({
   565→        title: `Certificate Transparency: ${unique.length} Subdomain(s) from CT Logs`,
   566→        severity: unique.length > 20 ? 'high' : 'info', category: 'osint',
   567→        description: `crt.sh reveals ${unique.length} unique subdomains from historical SSL certificate logs. These subdomains may not be in current DNS but existed in the past — old infrastructure may still be alive.`,
   568→        evidence: `CT subdomains: ${unique.slice(0, 30).join(', ')}${unique.length > 30 ? ` ... +${unique.length - 30} more` : ''}`,
   569→        asset: domain,
   570→      });
   571→    }
   572→    // Check for expired certs in CT logs
   573→    const now = Date.now();
   574→    const expiredCerts = certs.filter((c: { not_after?: string }) => {
   575→      const exp = c.not_after ? new Date(c.not_after + ' UTC').getTime() : 0;
   576→      return exp > 0 && exp < now;
   577→    });
   578→    if (expiredCerts.length > 3) {
   579→      findings.push({
   580→        title: `${expiredCerts.length} Expired Certificate(s) in CT Logs`,
   581→        severity: 'medium', category: 'ssl',
   582→        description: `${expiredCerts.length} historical certificates found in CT logs have expired. Expired certificates indicate abandoned infrastructure that may still be reachable.`,
   583→        evidence: `Expired certs: ${Math.min(expiredCerts.length, 5)} found in crt.sh`,
   584→        asset: domain,
   585→      });
   586→    }
   587→  } catch { /* crt.sh may fail — non-critical */ }
   588→  return findings;
   589→}

   592→// 7. WAYBACK MACHINE RECON — historical endpoints, hidden paths
   593→// ═══════════════════════════════════════════════════════════════════════
   594→async function enumerateWayback(domain: string): Promise<Finding[]> {
   595→  const findings: Finding[] = [];
   596→  try {
   597→    const response = await safeFetch(`http://web.archive.org/cdx/search/cdx?url=${domain}/*&output=json&collapse=urlkey&fl=original&limit=200`, { timeout: 25000, skipSSRFCheck: true });
   598→    if (!response.ok || !response.text) return findings;
   599→    const lines = response.text.split('\n').filter(Boolean);
   600→    if (lines.length < 2) return findings;
   601→    const urls = lines.slice(1).map(l => {
   602→      try { return JSON.parse(l)[0]; } catch { return null; }
   603→    }).filter(Boolean);
   604→
   605→    // Extract unique paths
   606→    const paths = new Set<string>();
   607→    const sensitivePaths: string[] = [];
   608→    const sensitivePatterns = [/\/admin/i, /\/api\//i, /\/debug/i, /\/test/i, /\/staging/i, /\/backup/i, /\/config/i, /\/env/i, /\/\.git/i, /\/\.svn/i, /\/\.env/i, /\/wp-/i, /\/phpmyadmin/i, /\/server-status/i, /\/actuator/i, /\/console/i, /\/graphql/i, /\/swagger/i, /\/\.well-known\//i];
   609→    for (const url of urls) {
   610→      try {
   611→        const u = new URL(url);
   612→        const p = u.pathname;
   613→        paths.add(p);
   614→        if (sensitivePatterns.some(pat => pat.test(p))) sensitivePaths.push(p);
   615→      } catch {}
   616→    }
   617→
   618→    if (paths.size > 0) {
   619→      findings.push({
   620→        title: `Wayback Machine: ${paths.size} Historical URL(s) Archived`,
   621→        severity: 'info', category: 'osint',
   622→        description: `Web Archive reveals ${paths.size} unique paths for ${domain}. Historical pages often expose old API endpoints, admin panels, or backup files still on the server.`,
   623→        evidence: `Sample paths: ${Array.from(paths).slice(0, 15).join(', ')}`,
   624→        asset: domain,
   625→      });
   626→    }
   627→
   628→    if (sensitivePaths.length > 0) {
   629→      const unique = [...new Set(sensitivePaths)].sort();
   630→      findings.push({
   631→        title: `Wayback Machine: ${unique.length} Sensitive Path(s) Discovered`,
   632→        severity: 'high', category: 'vulnerability',
   633→        description: `Historical archives expose ${unique.length} sensitive paths that may still exist on the live server. These include admin panels, API endpoints, debug interfaces, config files, or version control directories.`,
   634→        evidence: `Sensitive: ${unique.slice(0, 20).join(', ')}`,
   635→        asset: domain,
   636→      });
   637→    }
   638→  } catch { /* Wayback may fail — non-critical */ }
   639→  return findings;
   640→}

   643→// 8. REVERSE DNS + ASN INTELLIGENCE — map IPs to networks
   644→// ═══════════════════════════════════════════════════════════════════════
   645→async function reconReverseDNS(ips: string[]): Promise<Finding[]> {
   646→  const findings: Finding[] = [];
   647→  const uniqueIPs = [...new Set(ips)].slice(0, 10);
   648→  const rdnsResults = await Promise.all(
   649→    uniqueIPs.map(async (ip) => {
   650→      const ptr = await reverseDNS(ip);
   651→      return { ip, ptr: ptr || '' };
   652→    })
   653→  );
   654→  for (const { ip, ptr } of rdnsResults) {
   655→    if (ptr) {
   656→      findings.push({
   657→        title: `Reverse DNS: ${ip} → ${ptr}`,
   658→        severity: 'info', category: 'osint',
   659→        description: `IP ${ip} resolves to hostname ${ptr} via reverse DNS. This reveals the hosting provider, CDN edge, or internal network naming convention.`,
   660→        evidence: `PTR: ${ptr}`, asset: ip,
   661→      });
   662→    }
   663→  }
   664→  // ASN lookup via ipinfo.io (free tier, no auth needed)
   665→  if (uniqueIPs[0]) {
   666→    try {
   667→      const asnResponse = await safeFetch(`https://ipinfo.io/${uniqueIPs[0]}/json`, { timeout: 10000, skipSSRFCheck: true });
   668→      if (asnResponse.ok && asnResponse.text) {
   669→        const info = JSON.parse(asnResponse.text);
   670→        if (info.org || info.asn) {
   671→          findings.push({
   672→            title: `ASN Intelligence: ${info.org || info.asn}`,
   673→            severity: 'info', category: 'osint',
   674→            description: `Primary IP ${uniqueIPs[0]} belongs to ${info.org || info.asn}. Location: ${info.city || '?'}, ${info.region || '?'}, ${info.country || '?'}. ${info.hostname ? `Hostname: ${info.hostname}.` : ''}`,
   675→            evidence: `ASN: ${info.asn || '?'}, Org: ${info.org || '?'}, IP: ${uniqueIPs[0]}, Loc: ${info.city || '?'}/${info.country || '?'}`,
   676→            asset: uniqueIPs[0],
   677→          });
   678→        }
   679→      }
   680→    } catch {}
   681→  }
   682→  return findings;
   683→}

   686→// 9. DNS ZONE TRANSFER (AXFR) ATTEMPTS — try all NS servers
   687→// ═══════════════════════════════════════════════════════════════════════
   688→async function attemptZoneTransfer(domain: string): Promise<Finding[]> {
   689→  const findings: Finding[] = [];
   690→  const nsOut = await digAnswer(domain, 'NS');
   691→  const nsServers: string[] = [];
   692→  for (const line of nsOut.split('\n')) {
   693→    const m = line.match(/NS\s+(\S+)/);
   694→    if (m) nsServers.push(m[1].replace(/\.$/, ''));
   695→  }
   696→  if (nsServers.length === 0) return findings;
   697→
   698→  // Zone Transfer (AXFR) — native dns.promises does not support AXFR queries
   699→  // AXFR requires a specialized DNS client; skipping this check
   700→  for (const ns of nsServers) {
   701→    findings.push({
   702→      title: `DNS Zone Transfer Check: ${ns}`,
   703→      severity: 'info', category: 'dns',
   704→      description: `AXFR query requires a specialized DNS client not available in the native resolver. Zone transfer checks against ${ns} were skipped.`,
   705→      evidence: `NS: ${ns} — AXFR check skipped (native resolver limitation)`,
   706→      asset: domain,
   707→    });
   708→  }
   709→
   710→  // SOA record analysis
   711→  const soaOut = await digAnswer(domain, 'SOA');
   712→  if (soaOut) {
   713→    findings.push({
   714→      title: 'DNS SOA Record Analyzed',
   715→      severity: 'info', category: 'dns',
   716→      description: `SOA record reveals zone management details for ${domain}.`,
   717→      evidence: soaOut.split('\n').filter(l => l.includes('SOA')).join('; '),
   718→      asset: domain,
   719→    });
   720→  }
   721→  return findings;
   722→}

   725→// 10. ROBOTS.TXT + SITEMAP.XML — hidden paths, API endpoints
   726→// ═══════════════════════════════════════════════════════════════════════
   727→async function analyzeRobotsAndSitemap(domain: string): Promise<Finding[]> {
   728→  const findings: Finding[] = [];
   729→  const [robotsResult, sitemapResult] = await Promise.all([
   730→    safeFetch(`https://${domain}/robots.txt`, { timeout: 12000 }).catch(() => safeFetch(`http://${domain}/robots.txt`, { timeout: 12000 })),
   731→    safeFetch(`https://${domain}/sitemap.xml`, { timeout: 12000 }).catch(() => safeFetch(`http://${domain}/sitemap.xml`, { timeout: 12000 })),
   732→  ]);
   733→  const robots = robotsResult?.text || '';
   734→  const sitemap = sitemapResult?.text || '';
   735→
   736→  // Robots.txt
   737→  if (robots && robots.length > 10) {
   738→    const disallowed = robots.split('\n').filter(l => l.startsWith('Disallow:')).map(l => l.replace('Disallow:', '').trim()).filter(Boolean);
   739→    const sensitive = disallowed.filter(d => /admin|api|debug|config|backup|\.env|secret|internal|private|tmp|upload|staging/i.test(d));
   740→    if (sensitive.length > 0) {
   741→      findings.push({
   742→        title: `robots.txt: ${sensitive.length} Sensitive Disallowed Path(s)`,
   743→        severity: 'high', category: 'vulnerability',
   744→        description: `robots.txt reveals ${sensitive.length} sensitive paths that the site asks crawlers not to access. These paths likely exist and may be accessible without authentication: ${sensitive.join(', ')}`,
   745→        evidence: `Disallow: ${sensitive.join(', ')}`, asset: `${domain}/robots.txt`,
   746→      });
   747→    } else if (disallowed.length > 0) {
   748→      findings.push({
   749→        title: `robots.txt: ${disallowed.length} Disallowed Path(s) Found`,
   750→        severity: 'info', category: 'osint',
   751→        description: `robots.txt defines ${disallowed.length} restricted paths: ${disallowed.slice(0, 10).join(', ')}`,
   752→        evidence: `Disallow: ${disallowed.slice(0, 10).join(', ')}`, asset: `${domain}/robots.txt`,
   753→      });
   754→    }
   755→    // Check for sitemap reference
   756→    if (robots.includes('Sitemap:')) {
   757→      findings.push({
   758→        title: 'robots.txt References Sitemap', severity: 'info', category: 'osint',
   759→        description: 'Sitemap URL found in robots.txt — reveals site structure.',
   760→        evidence: robots.split('\n').find(l => l.startsWith('Sitemap:')) || '', asset: `${domain}/robots.txt`,
   761→      });
   762→    }
   763→  } else if (!robots || robots.length <= 10) {
   764→    findings.push({
   765→      title: 'robots.txt Not Found or Empty', severity: 'low', category: 'osint',
   766→      description: 'No robots.txt file. This means all pages are crawlable by default, including any admin/debug paths.',
   767→      evidence: 'robots.txt returned empty or 404', asset: `${domain}/robots.txt`,
   768→    });
   769→  }
   770→
   771→  // Sitemap.xml
   772→  if (sitemap && sitemap.includes('<url')) {
   773→    const urls = (sitemap.match(/<loc>([^<]+)<\/loc>/g) || []).map(u => u.replace(/<\/?loc>/g, ''));
   774→    findings.push({
   775→      title: `sitemap.xml: ${urls.length} URL(s) Exposed`,
   776→      severity: urls.length > 100 ? 'medium' : 'info', category: 'osint',
   777→      description: `sitemap.xml exposes ${urls.length} URLs revealing the full site structure. This aids attackers in mapping the application for targeted attacks.`,
   778→      evidence: `URLs: ${urls.slice(0, 10).join(', ')}${urls.length > 10 ? ` ... +${urls.length - 10} more` : ''}`,
   779→      asset: `${domain}/sitemap.xml`,
   780→    });
   781→  }
   782→  return findings;
   783→}

   786→// 11. JAVASCRIPT FILE ANALYSIS — extract API endpoints, secrets, keys
   787→// ═══════════════════════════════════════════════════════════════════════
   788→async function analyzeJSFiles(domain: string): Promise<Finding[]> {
   789→  const findings: Finding[] = [];
   790→  try {
   791→    // Fetch the main page and extract JS file URLs
   792→    const pageResult = await safeFetch(`https://${domain}`, { timeout: 15000 }).catch(() => safeFetch(`http://${domain}`, { timeout: 15000 }));
   793→    const page = pageResult?.text || '';
   794→    if (!page || page.length < 100) return findings;
   795→
   796→    const jsUrls = [...new Set(
   797→      (page.match(/src=["']([^"']+\.js[^"']*)/g) || [])
   798→        .map(m => m.replace(/src=["']/, '').replace(/["']$/, ''))
   799→        .filter(u => !u.startsWith('data:') && !u.includes('chrome-extension'))
   800→    )].slice(0, 8);
   801→
   802→    if (jsUrls.length === 0) return findings;
   803→
   804→    // Fetch and analyze each JS file
   805→    const jsContents = await Promise.all(
   806→      jsUrls.map(async (url) => {
   807→        const fullUrl = url.startsWith('http') ? url : `https://${domain}${url}`;
   808→        const result = await safeFetch(fullUrl, { timeout: 12000 });
   809→        return result?.text || '';
   810→      })
   811→    );
   812→
   813→    const allJs = jsContents.join('\n');
   814→    if (allJs.length < 50) return findings;
   815→
   816→    // Extract API endpoints
   817→    const apiEndpoints = [...new Set(
   818→      (allJs.match(/["'](\/api\/[^"']+)["']/g) || [])
   819→        .map(m => m.replace(/["']/g, ''))
   820→    )];
   821→    if (apiEndpoints.length > 0) {
   822→      findings.push({
   823→        title: `JavaScript: ${apiEndpoints.length} API Endpoint(s) Exposed`,
   824→        severity: 'high', category: 'vulnerability',
   825→        description: `JavaScript files expose ${apiEndpoints.length} API endpoints. Attackers can reverse-engineer the API surface, craft targeted requests, and test for authentication bypass or IDOR vulnerabilities.`,
   826→        evidence: `Endpoints: ${apiEndpoints.slice(0, 20).join(', ')}`,
   827→        asset: domain,
   828→      });
   829→    }
   830→
   831→    // Check for secrets/keys/tokens
   832→    const secretPatterns = [
   833→      { name: 'API Key', regex: /["'](api[_-]?key|apikey)["']\s*[:=]\s*["']([^"']{8,})["']/gi },
   834→      { name: 'Auth Token', regex: /["'](auth[_-]?token|bearer|access[_-]?token)["']\s*[:=]\s*["']([^"']{8,})["']/gi },
   835→      { name: 'AWS Key', regex: /AKIA[0-9A-Z]{16}/g },
   836→      { name: 'Firebase', regex: /firebase[a-zA-Z]*\.appspot\.com/g },
   837→      { name: 'Stripe Key', regex: /pk_(test|live)_[a-zA-Z0-9]{24,}/g },
   838→      { name: 'Generic Secret', regex: /["'](secret|password|token|private[_-]?key)["']\s*[:=]\s*["']([^"']{8,})["']/gi },
   839→    ];
   840→    const secrets: string[] = [];
   841→    for (const { name, regex } of secretPatterns) {
   842→      const matches = allJs.match(regex);
   843→      if (matches) secrets.push(...matches.map(m => `[${name}] ${m.substring(0, 60)}`));
   844→    }
   845→    if (secrets.length > 0) {
   846→      findings.push({
   847→        title: `CRITICAL: ${secrets.length} Secret(s)/Key(s) Found in JavaScript`,
   848→        severity: 'critical', category: 'vulnerability',
   849→        description: `JavaScript files contain ${secrets.length} hardcoded secrets, API keys, or tokens. These can be extracted by anyone viewing the page source and used to access internal services, databases, or third-party APIs.`,
   850→        evidence: `Secrets: ${secrets.slice(0, 5).join(' | ')}`,
   851→        asset: domain,
   852→      });
   853→    }
   854→
   855→    // Extract internal URLs/paths
   856→    const internalPaths = [...new Set(
   857→      (allJs.match(/["'](\/[a-zA-Z0-9_\-\/]+(?:admin|dashboard|console|manage|debug|internal|private|staging)[^"']*)["']/gi) || [])
   858→        .map(m => m.replace(/["']/g, ''))
   859→    )];
   860→    if (internalPaths.length > 0) {
   861→      findings.push({
   862→        title: `JavaScript: ${internalPaths.length} Internal Path(s) Referenced`,
   863→        severity: 'medium', category: 'osint',
   864→        description: `JavaScript files reference ${internalPaths.length} internal/sensitive paths. These may be admin interfaces, debug endpoints, or internal tools.`,
   865→        evidence: `Paths: ${internalPaths.slice(0, 15).join(', ')}`,
   866→        asset: domain,
   867→      });
   868→    }
   869→
   870→    // Count JS files and total size
   871→    findings.push({
   872→      title: `JavaScript Analysis: ${jsUrls.length} File(s), ${Math.round(allJs.length / 1024)}KB Analyzed`,
   873→      severity: 'info', category: 'technology',
   874→      description: `${jsUrls.length} JavaScript files totaling ~${Math.round(allJs.length / 1024)}KB were downloaded and analyzed for API endpoints, secrets, and internal paths.`,
   875→      evidence: `JS files: ${jsUrls.slice(0, 5).join(', ')}`,
   876→      asset: domain,
   877→    });
   878→  } catch {}
   879→  return findings;
   880→}

   883→// 12. WAF/SECURITY CONTROL DETECTION — fingerprint protection layers
   884→// ═══════════════════════════════════════════════════════════════════════
   885→async function detectWAF(domain: string): Promise<Finding[]> {
   886→  const findings: Finding[] = [];
   887→  // Trigger WAF with suspicious paths and analyze response
   888→  const [normalResult, attackResult] = await Promise.all([
   889→    safeFetch(`https://${domain}/`, { timeout: 10000, method: 'HEAD' }),
   890→    safeFetch(`https://${domain}/../../../etc/passwd`, { timeout: 10000, method: 'HEAD' }),
   891→  ]);
   892→  const normalResp = normalResult.headers ? JSON.stringify(normalResult.headers) : '';
   893→  const attackResp = attackResult.headers ? JSON.stringify(attackResult.headers) + ` status:${attackResult.status}` : '';
   894→
   895→  const wafSignatures: Record<string, string[]> = {
   896→    'Cloudflare': ['cf-ray', 'cf-cache-status', '__cf_bm', 'cloudflare'],
   897→    'AWS WAF': ['x-amzn-requestid', 'awselb'],
   898→    'Akamai': ['akamai', 'x-akamai'],
   899→    'Sucuri': ['x-sucuri-id', 'sucuri'],
   900→    'Imperva': ['x-iinfo', 'incap_ses'],
   901→    'Fastly': ['x-fastly-request-id', 'x-served-by'],
   902→    'Varnish': ['x-varnish', 'x-hits'],
   903→    'ModSecurity': ['mod_security', 'modsecurity'],
   904→    'F5 BIG-IP': ['bigip', 'f5'],
   905→  };
   906→
   907→  const detectedWAFs: string[] = [];
   908→  for (const [waf, sigs] of Object.entries(wafSignatures)) {
   909→    const combined = (normalResp + attackResp).toLowerCase();
   910→    if (sigs.some(s => combined.includes(s.toLowerCase()))) detectedWAFs.push(waf);
   911→  }
   912→
   913→  // Check if attack path was blocked differently
   914→  const normalStatus = `HTTP ${normalResult.status}`;
   915→  const attackStatus = `HTTP ${attackResult.status}`;
   916→  const blocked = attackResult.status === 403 || attackResult.status === 429;
   917→
   918→  if (detectedWAFs.length > 0) {
   919→    const techList = detectedWAFs.join(', ');
   920→    findings.push({
   921→      title: `WAF Detected: ${techList}`,
   922→      severity: 'info', category: 'security',
   923→      description: `${techList} detected protecting ${domain}. WAF presence indicates active security posture but also reveals the defense layer for targeted evasion research.`,
   924→      evidence: `Signatures found: ${detectedWAFs.map(w => wafSignatures[w].join(', ')).join('; ')}`,
   925→      asset: domain,
   926→    });
   927→  }
   928→
   929→  if (blocked) {
   930→    findings.push({
   931→      title: 'Path Traversal Attempt Blocked',
   932→      severity: 'info', category: 'security',
   933→      description: `Directory traversal attempt (../../../etc/passwd) was blocked with ${attackStatus}. This indicates active input validation or WAF protection.`,
   934→      evidence: `Attack response: ${attackStatus} | Normal: ${normalStatus}`,
   935→      asset: domain,
   936→    });
   937→  } else if (attackResp.includes('200') || attackResp.includes('301')) {
   938→    findings.push({
   939→      title: 'Path Traversal Attempt NOT Blocked',
   940→      severity: 'high', category: 'vulnerability',
   941→      description: `Directory traversal attempt (../../../etc/passwd) returned ${attackStatus} — the request was NOT blocked. This may indicate missing input validation, no WAF, or a misconfigured security layer.`,
   942→      evidence: `Attack response: ${attackStatus} | Normal: ${normalStatus}`,
   943→      asset: domain,
   944→    });
   945→  }
   946→
   947→  // Rate limit detection
   948→  const burst = await Promise.all([
   949→    safeFetch(`https://${domain}/`, { timeout: 6000, method: 'HEAD' }),
   950→    safeFetch(`https://${domain}/`, { timeout: 6000, method: 'HEAD' }),
   951→    safeFetch(`https://${domain}/`, { timeout: 6000, method: 'HEAD' }),
   952→    safeFetch(`https://${domain}/`, { timeout: 6000, method: 'HEAD' }),
   953→    safeFetch(`https://${domain}/`, { timeout: 6000, method: 'HEAD' }),
   954→  ]);
   955→  const rateLimited = burst.some(r => r.status === 429);
   956→  if (!rateLimited) {
   957→    findings.push({
   958→      title: 'No Rate Limiting Detected',
   959→      severity: 'medium', category: 'vulnerability',
   960→      description: `5 rapid requests returned no rate limiting (429). The server accepts unlimited requests, making it vulnerable to brute force, credential stuffing, and DoS attacks.`,
   961→      evidence: '5 rapid requests: all returned non-429 status',
   962→      asset: domain,
   963→    });
   964→  } else {
   965→    findings.push({
   966→      title: 'Rate Limiting Active', severity: 'info', category: 'security',
   967→      description: 'Rate limiting detected on rapid requests.', evidence: '429 response received on burst', asset: domain,
   968→    });
   969→  }
   970→  return findings;
   971→}

  1048→// 14. EMAIL HARVESTING — WHOIS, website, JS, DNS

  1050→async function harvestEmails(domain: string, pageContent: string): Promise<Finding[]> {
  1051→  const findings: Finding[] = [];
  1052→  const emails = new Set<string>();
  1053→
  1054→  // From page content
  1055→  const escapedDomain = domain.replace(/\./g, '\\.');
  1056→  const emailRegex = new RegExp(`[a-zA-Z0-9._%+-]+@${escapedDomain}`, 'gi');
  1057→  const pageEmails = (pageContent || '').match(emailRegex) || [];
  1058→  pageEmails.forEach(e => emails.add(e.toLowerCase()));
  1059→
  1060→  // From MX records (mail server hostnames often reveal email provider)
  1061→  const mxRecords = await digShort(domain, 'MX');
  1062→  const mxHosts = mxRecords.map(r => {
  1063→    const m = r.match(/\S+/);
  1064→    return m ? m[0] : r;
  1065→  });
  1066→
  1067→  if (emails.size > 0) {
  1068→    findings.push({
  1069→      title: `${emails.size} Email Address(es) Harvested`,
  1070→      severity: 'medium', category: 'osint',
  1071→      description: `${emails.size} email address(es) found on ${domain} via page content analysis. These can be used for social engineering, credential stuffing, or targeted phishing campaigns.`,
  1072→      evidence: `Emails: ${Array.from(emails).slice(0, 10).join(', ')}`,
  1073→      asset: domain,
  1074→    });
  1075→  }
  1076→
  1077→  if (mxHosts.length > 0) {
  1078→    findings.push({
  1079→      title: `Email Infrastructure: ${mxHosts.join(', ')}`,
  1080→      severity: 'info', category: 'osint',
  1081→      description: `Mail routed through: ${mxHosts.join(', ')}. This reveals the email provider (Google Workspace, Microsoft 365, self-hosted, etc.) and can be targeted for email-based attacks.`,
  1082→      evidence: `MX: ${mxHosts.join(', ')}`,
  1083→      asset: domain,
  1084→    });
  1085→  }
  1086→  return findings;
  1087→}

  1090→// MAIN SCAN ENDPOINT
  1091→// ═══════════════════════════════════════════════════════════════════════
  1092→export async function POST(request: NextRequest) {
  1093→  const { error: protErr, domain: protDomain } = await withProtection(request.clone() as unknown as NextRequest, { requireAuth: true, validateDomainFromBody: true, rateLimit: { maxRequests: 3, windowMs: 60000 } });
  1094→  if (protErr) return protErr;
  1095→
  1096→  try {
  1097→    const body = await request.json();
  1098→    const { scanType } = body;
  1099→
  1100→    if (!protDomain) {
  1101→      return NextResponse.json({ error: 'Domain is required' }, { status: 400 });
  1102→    }
  1103→
  1104→    const cleanDomain = protDomain;
  1105→
  1106→    // Pre-check: domain must resolve
  1107→    const preCheckIp = await nativeResolveIP(cleanDomain);
  1108→    if (!preCheckIp) {
  1109→      return NextResponse.json({ error: `Domain "${cleanDomain}" does not resolve. Check the name and try again.` }, { status: 400 });
  1110→    }
  1111→
  1112→    // SSRF protection: block scans that resolve to private IPs
  1113→    if (isPrivateIP(preCheckIp)) {
  1114→      return NextResponse.json({ error: 'Domain resolves to a private or reserved IP address. Scanning is not permitted.' }, { status: 403 });
  1115→    }
  1116→
  1117→    // DB: target + scan record
  1118→    let target = await db.scanTarget.findFirst({ where: { domain: cleanDomain } });
  1119→    if (!target) {
  1120→      target = await db.scanTarget.create({ data: { domain: cleanDomain, ip: preCheckIp } });
  1121→    } else {
  1122→      await db.scanTarget.update({ where: { id: target.id }, data: { ip: preCheckIp } });
  1123→    }
  1124→
  1125→    const scan = await db.scan.create({
  1126→      data: { targetId: target.id, status: 'running', scanType: scanType || 'full' },
  1127→    });
  1128→
  1129→    const isQuick = scanType === 'quick';
  1130→    const allFindings: Finding[] = [];
  1131→    const allTech = new Set<string>();
  1132→
  1133→    // ── 1. DNS (always, all queries parallel) ──────────────────────
  1134→    const dnsResult = await enumerateDNS(cleanDomain);
  1135→    allFindings.push(...dnsResult.findings);
  1136→
  1137→    // ── 2. Subdomains (parallel DNS resolution) ────────────────────
  1138→    const subFindings = await enumerateSubdomains(cleanDomain, isQuick);
  1139→    allFindings.push(...subFindings);
  1140→
  1141→    // ── 3. HTTP headers + page content (async curl) ──────────────────
  1142→    const httpResult = await analyzeHTTPHeaders(cleanDomain);
  1143→    allFindings.push(...httpResult.findings);
  1144→    httpResult.technologies.forEach(t => allTech.add(t));
  1145→
  1146→    // Fetch page content once for multiple modules
  1147→    const pageResult = await safeFetch(`https://${cleanDomain}`, { timeout: 15000 }).catch(() => safeFetch(`http://${cleanDomain}`, { timeout: 15000 }));
  1148→    const pageContent = pageResult?.text || '';
  1149→    const rawHeadersResult = await safeFetch(`https://${cleanDomain}`, { timeout: 10000, method: 'HEAD' });
  1150→    const rawHeaders = rawHeadersResult?.text || '';
  1151→
  1152→    // ── 4. SSL/TLS (async openssl) ─────────────────────────────────
  1153→    const sslResult = await analyzeSSL(cleanDomain);
  1154→    allFindings.push(...sslResult.findings);
  1155→    sslResult.technologies.forEach(t => allTech.add(t));
  1156→
  1157→    // ── 5. Port probing (full only, async parallel) ────────────────
  1158→    if (!isQuick) {
  1159→      const portFindings = await probePorts(cleanDomain);
  1160→      allFindings.push(...portFindings);
  1161→    }
  1162→
  1163→    // ═══ NEW DEADLY MODULES (full scan only) ═══════════════════════
  1164→    if (!isQuick) {
  1165→      // ── 6. Certificate Transparency Logs ──────────────────────────
  1166→      const [ctFindings, waybackFindings, rdnsFindings, axfrFindings, robotsFindings, jsFindings, wafFindings, fpFindings, emailFindings] = await Promise.all([
  1167→        enumerateCTLogs(cleanDomain),
  1168→        enumerateWayback(cleanDomain),
  1169→        reconReverseDNS([...dnsResult.mainIp ? [dnsResult.mainIp] : [], ...subFindings.map(f => {
  1170→          const m = f.evidence.match(/\d+\.\d+\.\d+\.\d+/);
  1171→          return m ? m[0] : null;
  1172→        }).filter((v): v is string => Boolean(v))]),
  1173→        attemptZoneTransfer(cleanDomain),
  1174→        analyzeRobotsAndSitemap(cleanDomain),
  1175→        analyzeJSFiles(cleanDomain),
  1176→        detectWAF(cleanDomain),
  1177→        deepFingerprint(cleanDomain, rawHeaders, pageContent),
  1178→        harvestEmails(cleanDomain, pageContent),
  1179→      ]);
  1180→
  1181→      allFindings.push(...ctFindings, ...waybackFindings, ...rdnsFindings, ...axfrFindings, ...robotsFindings, ...jsFindings, ...wafFindings, ...fpFindings, ...emailFindings);
  1182→    } else {
  1183→      // Quick scan: still run CT logs and basic fingerprinting (lightweight)
  1184→      const [ctFindings, fpFindings] = await Promise.all([
  1185→        enumerateCTLogs(cleanDomain),
  1186→        deepFingerprint(cleanDomain, rawHeaders, pageContent),
  1187→      ]);
  1188→      allFindings.push(...ctFindings, ...fpFindings);
  1189→    }
  1190→
  1191→    // ── Technology findings ─────────────────────────────────────────
  1192→    for (const tech of allTech) {
  1193→      allFindings.push({
  1194→        title: `Technology Detected: ${tech}`, severity: 'info', category: 'technology',
  1195→        description: `${tech} identified on ${cleanDomain} via HTTP headers, SSL cert metadata, or response analysis.`,
  1196→        evidence: 'Detected via HTTP/SSL/response fingerprinting', asset: tech,
  1197→      });
  1198→    }
  1199→
  1200→    // ── Wildcard DNS check ─────────────────────────────────────────
  1201→    const wcIp = await nativeResolveIP(`nonexistent-wildcard-test-12345.${cleanDomain}`);
  1202→    if (wcIp) {
  1203→      allFindings.push({
  1204→        title: 'Wildcard DNS Record Detected', severity: 'medium', category: 'dns',
  1205→        description: `Wildcard DNS active — even nonexistent subdomains resolve to ${wcIp}. Subdomain enumeration may have false positives.`,
  1206→        evidence: `nonexistent-wildcard-test-12345.${cleanDomain} resolved to ${wcIp}`, asset: cleanDomain,
  1207→      });
  1208→    }
  1209→
