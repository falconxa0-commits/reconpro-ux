   268→class TestDiagnosticsToRecommendationsFlow(unittest.TestCase):
   269→    """Verify auto_engineering diagnostics results feed into engineering_recommendations."""
   270→
   271→    def test_health_check_diagnostics_feed_recommendations(self):
   272→        """Health check diagnostic results should produce engineering recommendations.
   273→
   274→        The EngineeringRecommender.recommend_from_diagnostics() method
   275→        accepts the diagnostics dict shape and returns Recommendation objects.
   276→        """
   277→        with tempfile.TemporaryDirectory() as td:
   278→            store_path = Path(td) / "recs.json"
   279→            recommender = EngineeringRecommender(store_path=store_path)
   280→
   281→            diag_results = _make_diagnostics_result()
   282→            recs = recommender.recommend_from_diagnostics(diag_results)
   283→
   284→            # Should generate at least one recommendation from failed checks
   285→            self.assertIsInstance(recs, list)
   286→            self.assertGreater(len(recs), 0)
   287→
   288→            # All recommendations should be Recommendation instances
   289→            for rec in recs:
   290→                self.assertIsInstance(rec, Recommendation)
   291→                self.assertIn(rec.source, ("diagnostics", "testing", "performance"))
   292→                self.assertTrue(rec.id)
   293→                self.assertTrue(rec.title)
   294→                self.assertIn(rec.severity, ("critical", "high", "medium", "low", "info"))
   295→
   296→            # Broken module registry should generate a recommendation
   297→            module_recs = [r for r in recs if "broken" in r.title.lower() or "module" in r.title.lower()]
   298→            self.assertGreater(len(module_recs), 0)
