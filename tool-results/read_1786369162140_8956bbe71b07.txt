    95→   219→
    96→   220→class MetricsCollector:
    97→   221→    """Thread-safe metrics collection.
    98→   222→
    99→   223→    Metrics:
   100→   224→    - counters: monotonically increasing values (requests, findings, errors)
   101→   225→    - gauges: point-in-time values (memory, connections, queue_depth)
   102→   226→    - histograms: distribution tracking (latency, response_size)
   103→   227→    - timers: named timers with start/stop
   104→   228→
   105→   229→    All methods are O(1) and thread-safe.
   106→   230→    Minimal overhead: disabled metrics cost ~50ns per call.
   107→   231→    """
   108→   232→
   109→   233→    def __init__(self, enabled: bool = True) -> None:
   110→   234→        self._enabled = enabled
   111→   235→        self._lock = threading.Lock()
   112→   236→        self._counters: Dict[str, float] = {}
   113→   237→        self._gauges: Dict[str, float] = {}
   114→   238→        self._histograms: Dict[str, List[float]] = {}
   115→   239→        self._active_timers: Dict[str, float] = {}
   116→   240→
   117→   241→    @property
   118→   242→    def enabled(self) -> bool:
   119→   243→        return self._enabled
   120→   244→
   121→   245→    @enabled.setter
   122→   246→    def enabled(self, value: bool) -> None:
   123→   247→        self._enabled = value
   124→   248→
   125→   249→    def counter_increment(self, name: str, amount: float = 1) -> None:
   126→   250→        """Increment a counter by amount (default 1)."""
   127→   251→        if not self._enabled:
   128→   252→            return
   129→   253→        with self._lock:
   130→   254→            self._counters[name] = self._counters.get(name, 0) + amount
   131→   255→
   132→   256→    def gauge_set(self, name: str, value: float) -> None:
   133→   257→        """Set a gauge to a specific value."""
   134→   258→        if not self._enabled:
   135→   259→            return
   136→   260→        with self._lock:
   137→   261→            self._gauges[name] = value
   138→   262→
   139→   263→    def gauge_increment(self, name: str, amount: float = 1) -> None:
   140→   264→        """Increment (or decrement) a gauge."""
   141→   265→        if not self._enabled:
   142→   266→            return
   143→   267→        with self._lock:
   144→   268→            self._gauges[name] = self._gauges.get(name, 0) + amount
   145→   269→
   146→   270→    def histogram_observe(self, name: str, value: float) -> None:
   147→   271→        """Record a value in a histogram."""
   148→   272→        if not self._enabled:
   149→   273→            return
   150→   274→        with self._lock:
   151→   275→            if name not in self._histograms:
   152→   276→                self._histograms[name] = []
   153→   277→            self._histograms[name].append(value)
   154→   278→
   155→   279→    def timer_start(self, name: str) -> str:
   156→   280→        """Start a named timer. Returns the timer name."""
   157→   281→        if not self._enabled:
   158→   282→            return name
   159→   283→        with self._lock:
   160→   284→            self._active_timers[name] = time.perf_counter()
   161→   285→        return name
   162→   286→
   163→   287→    def timer_stop(self, name: str) -> Optional[float]:
   164→   288→        """Stop a named timer and record elapsed ms in histogram. Returns elapsed ms or None."""
   165→   289→        if not self._enabled:
   166→   290→            return None
   167→   291→        with self._lock:
   168→   292→            start = self._active_timers.pop(name, None)
   169→   293→        if start is None:
   170→   294→            return None
   171→   295→        elapsed_ms = (time.perf_counter() - start) * 1000.0
   172→   296→        self.histogram_observe(f"{name}_duration_ms", elapsed_ms)
   173→   297→        return elapsed_ms
   174→   298→
   175→   299→    def timer_context(self, name: str) -> TimerContext:
   176→   300→        """Return a TimerContext for use as a context manager."""
   177→   301→        return TimerContext(self, name)
   178→   302→
   179→   303→    def get_snapshot(self) -> Dict[str, Any]:
   180→   304→        """Return all current metrics as a dictionary."""
   181→   305→        with self._lock:
   182→   306→            snapshot: Dict[str, Any] = {
   183→   307→                "counters": dict(self._counters),
   184→   308→                "gauges": dict(self._gauges),
   185→   309→            }
   186→   310→            hist_summary: Dict[str, Dict[str, Any]] = {}
   187→   311→            for name, values in self._histograms.items():
   188→   312→                if values:
   189→   313→                    sorted_vals = sorted(values)
   190→   314→                    n = len(sorted_vals)
   191→   315→                    hist_summary[name] = {
   192→   316→                        "count": n,
   193→   317→                        "min": sorted_vals[0],
   194→   318→                        "max": sorted_vals[-1],
   195→   319→                        "avg": sum(sorted_vals) / n,
   196→   320→                        "p50": sorted_vals[n // 2],
   197→   321→                        "p95": sorted_vals[int(n * 0.95)] if n >= 20 else sorted_vals[-1],
   198→   322→                        "p99": sorted_vals[int(n * 0.99)] if n >= 100 else sorted_vals[-1],
   199→   323→                    }
   200→   324→                else:
   201→   325→                    hist_summary[name] = {"count": 0}
   202→   326→            snapshot["histograms"] = hist_summary
   203→   327→            snapshot["active_timers"] = len(self._active_timers)
   204→   328→        return snapshot
   205→   329→
   206→   330→    def reset(self) -> None:
   207→   331→        """Clear all metrics."""
   208→   332→        with self._lock:
   209→   333→            self._counters.clear()
   210→   334→            self._gauges.clear()
   211→   335→            self._histograms.clear()
   212→   336→            self._active_timers.clear()
   213→   337→
   214→   338→    def get_counter(self, name: str) -> float:
   215→   339→        """Get current counter value."""
   216→   340→        with self._lock:
   217→   341→            return self._counters.get(name, 0)
   218→   342→
   219→   343→    def get_gauge(self, name: str) -> float:
   220→   344→        """Get current gauge value."""
   221→   345→        with self._lock:
   222→   346→            return self._gauges.get(name, 0)

   251→   375→
   252→   376→class ScanTracer:
   253→   377→    """Trace a scan execution with detailed timing.
   254→   378→
   255→   379→    Produces an execution receipt:
   256→   380→    {
   257→   381→        "trace_id": "...",
   258→   382→        "target": "example.com",
   259→   383→        "started_at": "...",
   260→   384→        "completed_at": "...",
   261→   385→        "total_duration_ms": 12345,
   262→   386→        "modules": [
   263→   387→            {"id": "recon", "duration_ms": 1234, "findings": 5, "status": "ok"},
   264→   388→            {"id": "auth", "duration_ms": 567, "findings": 2, "status": "ok"},
   265→   389→        ],
   266→   390→        "errors": [],
   267→   391→        "score": 85,
   268→   392→        "grade": "B"
   269→   393→    }
   270→   394→    """
   271→   395→
   272→   396→    def __init__(self, target: str, enabled: bool = True) -> None:
   273→   397→        self._enabled = enabled
   274→   398→        self.trace_id: str = _generate_trace_id()
   275→   399→        self.target: str = target
   276→   400→        self.started_at: Optional[str] = None
   277→   401→        self.completed_at: Optional[str] = None
   278→   402→        self._modules: OrderedDict[str, ModuleTrace] = OrderedDict()
   279→   403→        self._errors: List[str] = []
   280→   404→        self._score: Optional[int] = None
   281→   405→        self._lock = threading.Lock()
   282→   406→        self._start_perf: float = 0.0
   283→   407→
   284→   408→    @property
   285→   409→    def enabled(self) -> bool:
   286→   410→        return self._enabled
   287→   411→
   288→   412→    @enabled.setter
   289→   413→    def enabled(self, value: bool) -> None:
   290→   414→        self._enabled = value
   291→   415→
   292→   416→    def start_scan(self) -> None:
   293→   417→        """Mark the scan as started."""
   294→   418→        self.started_at = _utc_iso()
   295→   419→        self._start_perf = time.perf_counter()
   296→   420→
   297→   421→    def end_scan(self, score: Optional[int] = None) -> Dict[str, Any]:
   298→   422→        """Mark the scan as completed and return the execution receipt."""
   299→   423→        self.completed_at = _utc_iso()
   300→   424→        total_ms = (time.perf_counter() - self._start_perf) * 1000.0 if self._start_perf else 0.0
   301→   425→        if score is not None:
   302→   426→            self._score = max(MIN_SCORE, min(MAX_SCORE, score))
   303→   427→
   304→   428→        receipt: Dict[str, Any] = {
   305→   429→            "trace_id": self.trace_id,
   306→   430→            "target": self.target,
   307→   431→            "started_at": self.started_at,
   308→   432→            "completed_at": self.completed_at,
   309→   433→            "total_duration_ms": round(total_ms, 2),
   310→   434→            "modules": [m.to_dict() for m in self._modules.values()],
   311→   435→            "errors": list(self._errors),
   312→   436→        }
   313→   437→        if self._score is not None:
   314→   438→            receipt["score"] = self._score
   315→   439→            receipt["grade"] = _score_to_grade(self._score)
   316→   440→        return receipt
   317→   441→
   318→   442→    def start_module(self, module_id: str) -> None:
   319→   443→        """Begin timing a module."""
   320→   444→        if not self._enabled:
   321→   445→            return
   322→   446→        with self._lock:
   323→   447→            mt = ModuleTrace(
   324→   448→                id=module_id,
   325→   449→                status="running",
   326→   450→                started_at=_utc_iso(),
   327→   451→            )
   328→   452→            self._modules[module_id] = mt
   329→   453→
   330→   454→    def end_module(self, module_id: str, findings: int = 0, status: str = "ok", error: Optional[str] = None) -> None:
   331→   455→        """End timing a module and record results."""
   332→   456→        if not self._enabled:
   333→   457→            return
   334→   458→        with self._lock:
   335→   459→            mt = self._modules.get(module_id)
   336→   460→            if mt is None:
   337→   461→                return
   338→   462→            mt.completed_at = _utc_iso()
   339→   463→            if mt.started_at:
   340→   464→                try:
   341→   465→                    started = datetime.fromisoformat(mt.started_at.replace("Z", "+00:00"))
   342→   466→                    ended = datetime.fromisoformat(mt.completed_at.replace("Z", "+00:00"))
   343→   467→                    mt.duration_ms = (ended - started).total_seconds() * 1000.0
   344→   468→                except (ValueError, OSError):
   345→   469→                    mt.duration_ms = 0.0
   346→   470→            mt.findings = findings
   347→   471→            mt.status = status
   348→   472→            mt.error = error
   349→   473→            if error:
   350→   474→                self._errors.append(f"{module_id}: {error}")
   351→   475→
   352→   476→    def add_error(self, message: str) -> None:
   353→   477→        """Add an error to the scan trace."""
   354→   478→        self._errors.append(message)
   355→   479→
   356→   480→    def set_module_detail(self, module_id: str, key: str, value: Any) -> None:
   357→   481→        """Set an arbitrary detail on a module trace."""
   358→   482→        if not self._enabled:
   359→   483→            return
   360→   484→        with self._lock:
   361→   485→            mt = self._modules.get(module_id)
   362→   486→            if mt:
   363→   487→                mt.details[key] = value
   364→   488→
   365→   489→    def get_receipt(self) -> Dict[str, Any]:
   366→   490→        """Return the current execution receipt without ending the scan."""
   367→   491→        total_ms = 0.0
   368→   492→        if self._start_perf and self.started_at:
   369→   493→            total_ms = (time.perf_counter() - self._start_perf) * 1000.0
   370→   494→
   371→   495→        receipt: Dict[str, Any] = {
   372→   496→            "trace_id": self.trace_id,
   373→   497→            "target": self.target,
   374→   498→            "started_at": self.started_at,
   375→   499→            "total_duration_ms": round(total_ms, 2),
   376→   500→            "modules": [m.to_dict() for m in self._modules.values()],
   377→   501→            "errors": list(self._errors),
   378→   502→        }
   379→   503→        if self._score is not None:
   380→   504→            receipt["score"] = self._score
   381→   505→            receipt["grade"] = _score_to_grade(self._score)
   382→   506→        return receipt
   383→   507→
   384→   508→    def reset(self) -> None:
   385→   509→        """Reset tracer for reuse."""
   386→   510→        self.trace_id = _generate_trace_id()
   387→   511→        self.started_at = None
   388→   512→        self.completed_at = None
   389→   513→        self._modules.clear()
   390→   514→        self._errors.clear()
   391→   515→        self._score = None
   392→   516→        self._start_perf = 0.0

   420→   544→
   421→   545→class PerformanceProfiler:
   422→   546→    """Per-module performance profiling.
   423→   547→
   424→   548→    Track per-module:
   425→   549→    - Wall clock time
   426→   550→    - Number of HTTP requests made
   427→   551→    - Bytes sent/received
   428→   552→    - Findings produced
   429→   553→
   430→   554→    Generates performance report comparing modules.
   431→   555→    """
   432→   556→
   433→   557→    def __init__(self, enabled: bool = True) -> None:
   434→   558→        self._enabled = enabled
   435→   559→        self._profiles: OrderedDict[str, ModuleProfile] = OrderedDict()
   436→   560→        self._start_times: Dict[str, float] = {}
   437→   561→        self._lock = threading.Lock()
   438→   562→
   439→   563→    @property
   440→   564→    def enabled(self) -> bool:
   441→   565→        return self._enabled
   442→   566→
   443→   567→    @enabled.setter
   444→   568→    def enabled(self, value: bool) -> None:
   445→   569→        self._enabled = value
   446→   570→
   447→   571→    def start_module(self, module_id: str) -> None:
   448→   572→        """Begin profiling a module."""
   449→   573→        if not self._enabled:
   450→   574→            return
   451→   575→        with self._lock:
   452→   576→            self._start_times[module_id] = time.perf_counter()
   453→   577→            self._profiles[module_id] = ModuleProfile(module_id=module_id)
   454→   578→
   455→   579→    def end_module(self, module_id: str, status: str = "ok") -> None:
   456→   580→        """End profiling a module."""
   457→   581→        if not self._enabled:
   458→   582→            return
   459→   583→        with self._lock:
   460→   584→            start = self._start_times.pop(module_id, None)
   461→   585→            profile = self._profiles.get(module_id)
   462→   586→            if profile and start is not None:
   463→   587→                profile.wall_time_ms = (time.perf_counter() - start) * 1000.0
   464→   588→                profile.status = status
   465→   589→
   466→   590→    def record_http_request(self, module_id: str, bytes_sent: int = 0, bytes_received: int = 0) -> None:
   467→   591→        """Record an HTTP request for a module."""
   468→   592→        if not self._enabled:
   469→   593→            return
   470→   594→        with self._lock:
   471→   595→            profile = self._profiles.get(module_id)
   472→   596→            if profile:
   473→   597→                profile.http_requests += 1
   474→   598→                profile.bytes_sent += bytes_sent
   475→   599→                profile.bytes_received += bytes_received
   476→   600→
   477→   601→    def record_findings(self, module_id: str, count: int = 1) -> None:
   478→   602→        """Record findings produced by a module."""
   479→   603→        if not self._enabled:
   480→   604→            return
   481→   605→        with self._lock:
   482→   606→            profile = self._profiles.get(module_id)
   483→   607→            if profile:
   484→   608→                profile.findings += count
   485→   609→
   486→   610→    def get_report(self) -> Dict[str, Any]:
   487→   611→        """Generate a performance report comparing all profiled modules."""
   488→   612→        with self._lock:
   489→   613→            profiles = [p.to_dict() for p in self._profiles.values()]
   490→   614→
   491→   615→        total_wall_ms = sum(p["wall_time_ms"] for p in profiles)
   492→   616→        total_requests = sum(p["http_requests"] for p in profiles)
   493→   617→        total_bytes_sent = sum(p["bytes_sent"] for p in profiles)
   494→   618→        total_bytes_recv = sum(p["bytes_received"] for p in profiles)
   495→   619→        total_findings = sum(p["findings"] for p in profiles)
   496→   620→
   497→   621→        # Sort by wall time descending for ranking
   498→   622→        ranked = sorted(profiles, key=lambda p: p["wall_time_ms"], reverse=True)
   499→   623→
   500→   624→        report: Dict[str, Any] = {
   501→   625→            "total_wall_time_ms": round(total_wall_ms, 2),
   502→   626→            "total_http_requests": total_requests,
   503→   627→            "total_bytes_sent": total_bytes_sent,
   504→   628→            "total_bytes_received": total_bytes_recv,
   505→   629→            "total_findings": total_findings,
   506→   630→            "modules": ranked,
   507→   631→            "slowest_module": ranked[0]["module_id"] if ranked else None,
   508→   632→            "fastest_module": ranked[-1]["module_id"] if ranked else None,
   509→   633→        }
   510→   634→        return report
   511→   635→
   512→   636→    def get_module_profile(self, module_id: str) -> Optional[Dict[str, Any]]:
   513→   637→        """Get the profile for a specific module."""
   514→   638→        with self._lock:
   515→   639→            profile = self._profiles.get(module_id)
   516→   640→            return profile.to_dict() if profile else None
   517→   641→
   518→   642→    def reset(self) -> None:
   519→   643→        """Clear all profiles."""
   520→   644→        with self._lock:
   521→   645→            self._profiles.clear()
   522→   646→            self._start_times.clear()

   551→   675→    def _get_memory_usage(self) -> Dict[str, Any]:
   552→   676→        """Get memory usage information."""
   553→   677→        result: Dict[str, Any] = {"status": "unknown", "rss_mb": 0, "method": "none"}
   554→   678→        try:
   555→   679→            # Try /proc/self/status first (Linux)
   556→   680→            if os.path.exists("/proc/self/status"):
   557→   681→                with open("/proc/self/status", "r") as f:
   558→   682→                    for line in f:
   559→   683→                        if line.startswith("VmRSS:"):
   560→   684→                            kb = int(line.split()[1])
   561→   685→                            result["rss_mb"] = round(kb / 1024, 1)
   562→   686→                            result["method"] = "proc"
   563→   687→                            break
   564→   688→                # Also get VmSize for context
   565→   689→                with open("/proc/self/status", "r") as f:
   566→   690→                    for line in f:
   567→   691→                        if line.startswith("VmSize:"):
   568→   692→                            kb = int(line.split()[1])
   569→   693→                            result["vm_size_mb"] = round(kb / 1024, 1)
   570→   694→                            break
   571→   695→            else:
   572→   696→                # Fallback: try resource module
   573→   697→                import resource
   574→   698→                rss = resource.getrusage(resource.RUSAGE_SELF).ru_maxrss
   575→   699→                # On Linux, ru_maxrss is in KB; on macOS, in bytes
   576→   700→                if os.uname().sysname == "Darwin":
   577→   701→                    result["rss_mb"] = round(rss / (1024 * 1024), 1)
   578→   702→                else:
   579→   703→                    result["rss_mb"] = round(rss / 1024, 1)
   580→   704→                result["method"] = "resource"
   581→   705→        except Exception:
   582→   706→            pass
   583→   707→        return result

   616→   740→        """Check basic network connectivity via DNS resolution."""

   621→   745→            elapsed = (time.perf_counter() - start) * 1000.0
   622→   746→            result["dns_resolve_ms"] = round(elapsed, 1)

   624→   748→        except socket.gaierror:

   626→   750→        except Exception:

   630→   754→    def _get_ssl_check(self) -> Dict[str, Any]:
   631→   755→        """Check SSL certificate for a known site."""
   632→   756→        result: Dict[str, Any] = {
   633→   757→            "host": "example.com",
   634→   758→            "status": "unknown",
   635→   759→            "days_until_expiry": None,
   636→   760→        }
   637→   761→        try:
   638→   762→            context = ssl.create_default_context()
   639→   763→            conn = context.wrap_socket(socket.socket(socket.AF_INET), server_hostname="example.com")
   640→   764→            conn.settimeout(3)
   641→   765→            conn.connect(("example.com", 443))
   642→   766→            cert = conn.getpeercert()
   643→   767→            conn.close()
   644→   768→
   645→   769→            # Parse notAfter
   646→   770→            import email.utils
   647→   771→            for field in cert.get("notAfter", ""):
   648→   772→                pass  # no-op
   649→   773→            not_after_str = cert.get("notAfter", "")
   650→   774→            if not_after_str:
   651→   775→                not_after = email.utils.parsedate_to_datetime(not_after_str)
   652→   776→                now = datetime.now(timezone.utc)
   653→   777→                days = (not_after - now).days
   654→   778→                result["days_until_expiry"] = days
   655→   779→                if days <= 0:
   656→   780→                    result["status"] = "critical"
   657→   781→                elif days <= 30:
   658→   782→                    result["status"] = "degraded"
   659→   783→                else:
   660→   784→                    result["status"] = "healthy"
   661→   785→        except Exception:
   662→   786→            pass
   663→   787→        return result

   665→   789→    def check_health(self) -> Dict[str, Any]:
   666→   790→        """Run all health checks and return a comprehensive report."""
   667→   791→        if not self._enabled:
   668→   792→            return {"status": "disabled", "checks": {}}
   669→   793→
   670→   794→        memory = self._get_memory_usage()
   671→   795→        disk = self._get_disk_usage()
   672→   796→        network = self._get_network_connectivity()
   673→   797→        ssl = self._get_ssl_check()
   674→   798→
   675→   799→        checks = {
   676→   800→            "memory": memory,
   677→   801→            "disk": disk,
   678→   802→            "network": network,
   679→   803→            "ssl": ssl,
   680→   804→        }
   681→   805→
   682→   806→        # Determine overall status
   683→   807→        statuses = []
   684→   808→        for check in checks.values():
   685→   809→            s = check.get("status", "unknown")
   686→   810→            if s == "critical":
   687→   811→                statuses.append(0)
   688→   812→            elif s == "degraded":
   689→   813→                statuses.append(1)
   690→   814→            elif s == "healthy":
   691→   815→                statuses.append(2)
   692→   816→            else:
   693→   817→                statuses.append(1)  # unknown -> degraded
   694→   818→
   695→   819→        worst = min(statuses)
   696→   820→        overall = ["critical", "degraded", "healthy"][worst]
   697→   821→
   698→   822→        return {
   699→   823→            "status": overall,
   700→   824→            "timestamp": _utc_iso(),
   701→   825→            "checks": checks,
   702→   826→        }
