    71→    71→def _check_open_ports() -> List[Finding]:

    73→    73→    findings: List[Finding] = []
    74→    74→    hostname = _hostname()
    75→    75→
    76→    76→    # Platform-specific port enumeration
    77→    77→    if _is_windows():
    78→    78→        code, out = _run('netstat -an 2>NUL | findstr LISTENING')
    79→    79→        if code != 0 or not out:
    80→    80→            findings.append(Finding(
    81→    81→                title="Could not enumerate open ports",
    82→    82→                severity="low", category="ports",
    83→    83→                module="host",
    84→    84→                description="Unable to enumerate open ports.",
    85→    85→                evidence="netstat returned no output",
    86→    86→                asset=hostname, points_deducted=1,
    87→    87→                remediation="Run as Administrator for full port visibility.",
    88→    88→            ))
    89→    89→            return findings
    90→    90→    else:
    91→    91→        code, out = _run("ss -tlnp 2>/dev/null || netstat -tlnp 2>/dev/null")
    92→    92→        if code != 0 or not out:
    93→    93→            findings.append(Finding(
    94→    94→                title="Could not enumerate open ports",
    95→    95→                severity="low", category="ports",
    96→    96→                module="host",
    97→    97→                description="Unable to run ss or netstat. Try running with sudo.",
    98→    98→                evidence="ss/netstat returned no output",
    99→    99→                asset=hostname, points_deducted=1,
   100→   100→                remediation="Run with sudo for full port visibility.",
   101→   101→            ))
   102→   102→            return findings
   103→   103→
   104→   104→    ports = []
   105→   105→    for line in out.splitlines():
   106→   106→        # Parse ss output: 0.0.0.0:3000 or [::]:8080
   107→   107→        match = re.search(r':(\d+)\s', line)
   108→   108→        if match:
   109→   109→            port = int(match.group(1))
   110→   110→            ports.append(port)
   111→   111→
   112→   112→    # Categorize risky ports
   113→   113→    risky_ports = {
   114→   114→        23: ("high", "Telnet (unencrypted)"),
   115→   115→        21: ("high", "FTP (unencrypted)"),
   116→   116→        445: ("high", "SMB/CIFS"),
   117→   117→        3389: ("high", "RDP"),
   118→   118→        5900: ("high", "VNC"),
   119→   119→        6379: ("high", "Redis (no auth by default)"),
   120→   120→        27017: ("high", "MongoDB (no auth by default)"),
   121→   121→        11211: ("high", "Memcached"),
   122→   122→        9200: ("medium", "Elasticsearch"),
   123→   123→        5432: ("medium", "PostgreSQL"),
   124→   124→        3306: ("medium", "MySQL"),
   125→   125→        8080: ("low", "HTTP alt"),
   126→   126→        8443: ("low", "HTTPS alt"),
   127→   127→        3000: ("low", "Node.js dev server"),
   128→   128→        5173: ("low", "Vite dev server"),
   129→   129→        8000: ("low", "Python dev server"),
   130→   130→    }
   131→   131→
   132→   132→    for port in sorted(set(ports)):
   133→   133→        if port in risky_ports:
   134→   134→            sev, desc = risky_ports[port]
   135→   135→            findings.append(Finding(
   136→   136→                title=f"Risky open port: {port} ({desc})",
   137→   137→                severity=sev, category="ports",
   138→   138→                module="host",
   139→   139→                description=f"Port {port} ({desc}) is open and listening. This could expose services to network attackers.",
   140→   140→                evidence=f"Port {port} found in ss/netstat output",
   141→   141→                asset=hostname, points_deducted=8 if sev == "high" else 4,
   142→   142→                remediation=f"Close port {port} if not needed, or bind to 127.0.0.1 only.",
   143→   143→            ))
   144→   144→
   145→   145→    if not findings and ports:
   146→   146→        findings.append(Finding(
   147→   147→            title=f"{len(set(ports))} open ports detected",
   148→   148→            severity="info", category="ports",
   149→   149→            module="host",
   150→   150→            description=f"Found {len(set(ports))} open ports. None are known high-risk.",
   151→   151→            evidence=", ".join(str(p) for p in sorted(set(ports))[:20]),
   152→   152→            asset=hostname, points_deducted=0,
   153→   153→            remediation="Review all open ports periodically.",
   154→   154→        ))
   155→   155→
   156→   156→    return findings

   294→   294→                    evidence=f"Administrators: {', '.join(admins[:5])}",
   295→   295→                    asset=hostname, points_deducted=5,
   296→   296→                    remediation="Remove unnecessary users from the Administrators group via Computer Management.",
   297→   297→                ))
   298→   298→        return findings

   300→   300→    # Non-Windows: check sudo users
   301→   301→    code, out = _run("getent group sudo 2>/dev/null")

   311→   311→                asset=hostname, points_deducted=5,
   312→   312→                remediation="Remove unnecessary users from the sudo group. Use specific capabilities instead.",
   313→   313→            ))

   334→   334→            severity="critical", category="users",

   338→   338→            asset=hostname, points_deducted=15,
   339→   339→            remediation="Set passwords for all users immediately: sudo passwd <username>",
   340→   340→        ))
   341→   341→
   342→   342→    return findings

   345→   345→def _check_ssh() -> List[Finding]:
   346→   346→    """Check SSH configuration hardening."""
   347→   347→    findings: List[Finding] = []
   348→   348→    hostname = _hostname()
   349→   349→    sshd_config = "/etc/ssh/sshd_config"
   350→   350→
   351→   351→    if not _file_exists(sshd_config):
   352→   352→        return findings
   353→   353→
   354→   354→    try:
   355→   355→        with open(sshd_config) as f:
   356→   356→            config = f.read()
   357→   357→    except Exception:
   358→   358→        return findings
   359→   359→
   360→   360→    ssh_checks = [
   361→   361→        (r'^#?PermitRootLogin\s+yes', "high", "SSH root login is permitted", "Set PermitRootLogin no"),
   362→   362→        (r'^#?PasswordAuthentication\s+yes', "medium", "SSH password authentication enabled", "Use SSH keys only: PasswordAuthentication no"),
   363→   363→        (r'^#?PermitEmptyPasswords\s+yes', "critical", "SSH allows empty passwords", "Set PermitEmptyPasswords no"),
   364→   364→        (r'^#?X11Forwarding\s+yes', "low", "SSH X11 forwarding enabled", "Set X11Forwarding no if not needed"),
   365→   365→        (r'^#?MaxAuthTries\s+(\d+)', None, "", ""),  # just read value
   366→   366→        (r'^#?Protocol\s+1', "high", "SSH Protocol 1 in use (insecure)", "Use Protocol 2"),
   367→   367→        (r'^#?AllowAgentForwarding\s+yes', "low", "SSH agent forwarding enabled", "Disable if not needed: AllowAgentForwarding no"),
   368→   368→    ]
   369→   369→
   370→   370→    for pattern, sev, title, remediation in ssh_checks:
   371→   371→        if re.search(pattern, config, re.MULTILINE | re.IGNORECASE):
   372→   372→            if sev is None:
   373→   373→                # MaxAuthTries check
   374→   374→                match = re.search(r'^#?MaxAuthTries\s+(\d+)', config, re.MULTILINE)
   375→   375→                if match and int(match.group(1)) > 6:
   376→   376→                    findings.append(Finding(
   377→   377→                        title=f"SSH MaxAuthTries is {match.group(1)} (too high)",
   378→   378→                        severity="low", category="ssh",
   379→   379→                        module="host",
   380→   380→                        description=f"MaxAuthTries={match.group(1)} allows more brute-force attempts.",
   381→   381→                        evidence=f"MaxAuthTries {match.group(1)}",
   382→   382→                        asset=hostname, points_deducted=2,
   383→   383→                        remediation="Set MaxAuthTries 4",
   384→   384→                    ))
   385→   385→            else:
   386→   386→                findings.append(Finding(
   387→   387→                    title=title,
   388→   388→                    severity=sev, category="ssh",
   389→   389→                    module="host",
   390→   390→                    description=f"SSH config issue: {title}",
   391→   391→                    evidence=f"Found in {sshd_config}",
   392→   392→                    asset=hostname, points_deducted=8 if sev == "high" else (15 if sev == "critical" else 3),
   393→   393→                    remediation=remediation,
   394→   394→                ))
   395→   395→
   396→   396→    return findings

   438→   438→    if code == 0 and out:

   446→   446→                    evidence=line[:200],
   447→   447→                    asset=hostname, points_deducted=15,
   448→   448→                    remediation="Remove --privileged flag. Use specific capabilities instead.",

   568→   568→def _check_cron_jobs() -> List[Finding]:
