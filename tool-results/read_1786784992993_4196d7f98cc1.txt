    12→
    13→function mulberry32(seed: number) {
    14→  return function () {
    15→    let t = (seed += 0x6d2b79f5);
    16→    t = Math.imul(t ^ (t >>> 15), t | 1);
    17→    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    18→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
    19→  };
    20→}

    22→function hashCode(s: string): number {
    23→  let h = 0;
    24→  for (let i = 0; i < s.length; i++) {
    25→    h = (Math.imul(31, h) + s.charCodeAt(i)) | 0;
    26→  }
    27→  return Math.abs(h);
    28→}

   444→function pickSeverity(rng: () => number): Severity {
   445→  const r = rng() * 100;
   446→  let cum = 0;
   447→  for (const [sev, weight] of SEVERITY_WEIGHTS) {
   448→    cum += weight;
   449→    if (r < cum) return sev;
   450→  }
   451→  return 'low';
   452→}
   453→
   454→function pickIndustry(rng: () => number): Industry {
   455→  return INDUSTRIES[Math.floor(rng() * INDUSTRIES.length)];
   456→}
   457→
   458→function pickFindingType(rng: () => number): FindingType {
   459→  return FINDING_TYPES[Math.floor(rng() * FINDING_TYPES.length)];
   460→}
   461→
   462→function pickRegion(rng: () => number): Region {
   463→  return REGIONS[Math.floor(rng() * REGIONS.length)];
   464→}
   465→
   466→function pickEntity(rng: () => number, industry: Industry): string {
   467→  const entities = ENTITY_TEMPLATES[industry];
   468→  return entities[Math.floor(rng() * entities.length)];
   469→}
   470→
   471→function pickDescription(rng: () => number, ft: FindingType, entity: string): string {
   472→  const templates = FINDING_TEMPLATES[ft];
   473→  const tpl = templates[Math.floor(rng() * templates.length)];
   474→  return tpl(entity);
   475→}
   476→
   477→function pickAssetType(rng: () => number, ft: FindingType): string {
   478→  const types = ASSET_TYPES[ft];
   479→  return types[Math.floor(rng() * types.length)];
   480→}

   482→function generateIncident(rng: () => number, timestamp: Date, id: string): Incident {
   483→  const industry = pickIndustry(rng);
   484→  const findingType = pickFindingType(rng);
   485→  const severity = pickSeverity(rng);
   486→  const entity = pickEntity(rng, industry);
   487→  const region = pickRegion(rng);
   488→
   489→  return {
   490→    id,
   491→    timestamp: timestamp.toISOString(),
   492→    severity,
   493→    industry,
   494→    findingType,
   495→    anonymizedDescription: pickDescription(rng, findingType, entity),
   496→    assetType: pickAssetType(rng, findingType),
   497→    region,
   498→    verifiable: rng() > 0.25,
   499→  };
   500→}

   502→/**
   503→ * Generate incidents for a given time window.
   504→ * Uses minute-level seed so it's deterministic per-minute (no flicker on refresh).
   505→ * Average rate: ~4.2 incidents/hour weighted by business hours.
   506→ */
   507→function generateIncidents(
   508→  since: Date,
   509→  until: Date,
   510→  seedPrefix: string = 'wos',
   511→): Incident[] {
   512→  const incidents: Incident[] = [];
   513→  const minute = 60 * 1000;
   514→  let cursor = new Date(since);
   515→  cursor.setSeconds(0, 0);
   516→  cursor.setMinutes(cursor.getMinutes() + 1);
   517→
   518→  while (cursor < until) {
   519→    const minuteKey = `wos:${Math.floor(cursor.getTime() / minute)}`;
   520→    const seed = hashCode(minuteKey);
   521→    const rng = mulberry32(seed);
   522→
   523→    const hour = cursor.getUTCHours();
   524→    const hourWeight = HOUR_WEIGHTS[hour];
   525→    const expectedIncidents = 0.07 * hourWeight;
   526→
   527→    if (rng() < expectedIncidents) {
   528→      const id = `inc-${Math.floor(cursor.getTime() / minute)}-${Math.floor(rng() * 1000)}`;
   529→      const jitteredTs = new Date(cursor.getTime() + Math.floor(rng() * 60000));
   530→      incidents.push(generateIncident(rng, jitteredTs, id));
   531→    }
   532→
   533→    cursor = new Date(cursor.getTime() + minute);
   534→  }
   535→
   536→  return incidents;
   537→}

   541→function computeStats(weekIncidents: Incident[]): Stats {
   542→  const now = new Date();
   543→  const todayStart = new Date(now.getUTCFullYear(), now.getUTCMonth(), now.getUTCDate());
   544→
   545→  const todayIncidents = weekIncidents.filter(i => new Date(i.timestamp) >= todayStart);
   546→
   547→  const byIndustry = {} as Record<Industry, number>;
   548→  const byType = {} as Record<FindingType, number>;
   549→
   550→  for (const ind of INDUSTRIES) byIndustry[ind] = 0;
   551→  for (const ft of FINDING_TYPES) byType[ft] = 0;
   552→
   553→  for (const inc of weekIncidents) {
   554→    byIndustry[inc.industry]++;
   555→    byType[inc.findingType]++;
   556→  }
   557→
   558→  return {
   559→    totalToday: todayIncidents.length,
   560→    totalWeek: weekIncidents.length,
   561→    criticalThisWeek: weekIncidents.filter(i => i.severity === 'critical').length,
   562→    byIndustry,
   563→    byType,
   564→  };
   565→}

   567→// ── Route Handler ──────────────────────────────────────────────────────
   568→
   569→export async function GET(request: NextRequest) {
   570→  const { error } = await withProtection(request, {
   571→    rateLimit: { maxRequests: 30, windowMs: 60_000 },
   572→  });
   573→  if (error) return error;
   574→
   575→  const { searchParams } = request.nextUrl;
   576→  const limitParam = searchParams.get('limit');
   577→  const severityFilter = searchParams.get('severity') as Severity | null;
   578→  const industryFilter = searchParams.get('industry') as Industry | null;
   579→
   580→  const limit = limitParam ? Math.min(Math.max(parseInt(limitParam, 10) || 50, 1), 200) : 50;
   581→
   582→  const now = new Date();
   583→  const weekAgo = new Date(now.getTime() - 7 * 24 * 60 * 60 * 1000);
   584→
   585→  const allIncidents = generateIncidents(weekAgo, now);
   586→
   587→  let filtered = allIncidents;
   588→  if (severityFilter && ['critical', 'high', 'medium', 'low'].includes(severityFilter)) {
   589→    filtered = filtered.filter(i => i.severity === severityFilter);
   590→  }
   591→  if (industryFilter && INDUSTRIES.includes(industryFilter)) {
   592→    filtered = filtered.filter(i => i.industry === industryFilter);
   593→  }
   594→
   595→  filtered.sort((a, b) => new Date(b.timestamp).getTime() - new Date(a.timestamp).getTime());
   596→
   597→  const incidents = filtered.slice(0, limit);
   598→  const stats = computeStats(allIncidents);
   599→
   600→  return NextResponse.json({ incidents, stats, simulated: true });
   601→}
