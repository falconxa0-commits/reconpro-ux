    40→def _hostname() -> str:
    41→    """Cross-platform hostname (works on Linux, macOS, Windows)."""
    42→    try:
    43→        return os.uname().nodename
    44→    except AttributeError:
    45→        return os.environ.get("COMPUTERNAME", os.environ.get("HOSTNAME", "localhost"))

    48→def _run(cmd: str, timeout: int = 10) -> Tuple[int, str]:
    49→    """Run a shell command, return (exit_code, stdout+stderr)."""
    50→    try:
    51→        r = subprocess.run(
    52→            cmd, shell=True, capture_output=True, text=True, timeout=timeout
    53→        )
    54→        return r.returncode, (r.stdout + r.stderr).strip()
    55→    except Exception:
    56→        return -1, ""

    80→def _check_open_ports() -> List[Finding]:
    81→    """Find open listening ports via ss/netstat."""
    82→    findings: List[Finding] = []
    83→    hostname = _hostname()
    84→
    85→    # Platform-specific port enumeration
    86→    if _is_windows():
    87→        code, out = _run('netstat -an 2>NUL | findstr LISTENING')
    88→        if code != 0 or not out:
    89→            findings.append(Finding(
    90→                title="Could not enumerate open ports",
    91→                severity="low", category="ports",
    92→                module="host",
    93→                description="Unable to enumerate open ports.",
    94→                evidence="netstat returned no output",
    95→                asset=hostname, points_deducted=1,
    96→                remediation="Run as Administrator for full port visibility.",
    97→            ))
    98→            return findings
    99→    else:
   100→        code, out = _run("ss -tlnp 2>/dev/null || netstat -tlnp 2>/dev/null")
   101→        if code != 0 or not out:
   102→            findings.append(Finding(
   103→                title="Could not enumerate open ports",
   104→                severity="low", category="ports",
   105→                module="host",
   106→                description="Unable to run ss or netstat. Try running with sudo.",
   107→                evidence="ss/netstat returned no output",
   108→                asset=hostname, points_deducted=1,
   109→                remediation="Run with sudo for full port visibility.",
   110→            ))
   111→            return findings
   112→
   113→    ports = []
   114→    for line in out.splitlines():
   115→        # Parse ss output: 0.0.0.0:3000 or [::]:8080
   116→        match = re.search(r':(\d+)\s', line)
   117→        if match:
   118→            port = int(match.group(1))
   119→            ports.append(port)
   120→
   121→    # Categorize risky ports
   122→    risky_ports = {
   123→        23: ("high", "Telnet (unencrypted)"),
   124→        21: ("high", "FTP (unencrypted)"),
   125→        445: ("high", "SMB/CIFS"),
   126→        3389: ("high", "RDP"),
   127→        5900: ("high", "VNC"),
   128→        6379: ("high", "Redis (no auth by default)"),
   129→        27017: ("high", "MongoDB (no auth by default)"),
   130→        11211: ("high", "Memcached"),
   131→        9200: ("medium", "Elasticsearch"),
   132→        5432: ("medium", "PostgreSQL"),
   133→        3306: ("medium", "MySQL"),
   134→        8080: ("low", "HTTP alt"),
   135→        8443: ("low", "HTTPS alt"),
   136→        3000: ("low", "Node.js dev server"),
   137→        5173: ("low", "Vite dev server"),
   138→        8000: ("low", "Python dev server"),
   139→    }
   140→
   141→    for port in sorted(set(ports)):
   142→        if port in risky_ports:
   143→            sev, desc = risky_ports[port]
   144→            findings.append(Finding(
   145→                title=f"Risky open port: {port} ({desc})",
   146→                severity=sev, category="ports",
   147→                module="host",
   148→                description=f"Port {port} ({desc}) is open and listening. This could expose services to network attackers.",
   149→                evidence=f"Port {port} found in ss/netstat output",
   150→                asset=hostname, points_deducted=8 if sev == "high" else 4,
   151→                remediation=f"Close port {port} if not needed, or bind to 127.0.0.1 only.",
   152→            ))
   153→
   154→    if not findings and ports:
   155→        findings.append(Finding(
   156→            title=f"{len(set(ports))} open ports detected",
   157→            severity="info", category="ports",
   158→            module="host",
   159→            description=f"Found {len(set(ports))} open ports. None are known high-risk.",
   160→            evidence=", ".join(str(p) for p in sorted(set(ports))[:20]),
   161→            asset=hostname, points_deducted=0,
   162→            remediation="Review all open ports periodically.",
   163→        ))
   164→
   165→    return findings

   239→                severity="info", category="firewall",

   243→                asset=hostname, points_deducted=0,

   252→            findings.append(Finding(
   253→                title="iptables has no DROP/REJECT rules",
   254→                severity="high", category="firewall",

   257→                evidence="No DROP or REJECT rules found",
   258→                asset=hostname, points_deducted=10,
   259→                remediation="Configure iptables rules or install UFW: sudo apt install ufw && sudo ufw enable",
   260→            ))
   261→        else:
   262→            findings.append(Finding(
   263→                title="iptables rules found",
   264→                severity="info", category="firewall",
   265→                module="host",
   266→                description="iptables has filtering rules configured.",
   267→                evidence="Rules present",
   268→                asset=hostname, points_deducted=0,
   269→                remediation="",
   270→            ))
   271→        return findings
   272→
   273→    findings.append(Finding(
   274→        title="No firewall detected",
   275→        severity="high", category="firewall",
   276→        module="host",
   277→        description="No firewall detected. Your machine has no network firewall.",
   278→        evidence="No firewall found",
   279→        asset=hostname, points_deducted=10,
   280→        remediation="Install and enable a firewall: sudo apt install ufw && sudo ufw enable",
   281→    ))

   285→def _check_users() -> List[Finding]:
   286→    """Check user accounts and sudo access."""
   287→    findings: List[Finding] = []
   288→    hostname = _hostname()
   289→
   290→    if _is_windows():
   291→        # Check administrator accounts on Windows
   292→        code, out = _run('net localgroup Administrators 2>NUL')
   293→        if code == 0 and out:
   294→            lines = [l.strip() for l in out.splitlines() if l.strip()]
   295→            # First two lines are header, last line is "The command completed..."
   296→            admins = [l for l in lines[2:] if l and "command completed" not in l.lower()]
   297→            if len(admins) > 2:
   298→                findings.append(Finding(
   299→                    title=f"{len(admins)} administrator accounts found",
   300→                    severity="medium", category="users",
   301→                    module="host",
   302→                    description=f"{len(admins)} users have administrator privileges: {', '.join(admins[:5])}. Minimize admin access to reduce risk.",
   303→                    evidence=f"Administrators: {', '.join(admins[:5])}",
   304→                    asset=hostname, points_deducted=5,
   305→                    remediation="Remove unnecessary users from the Administrators group via Computer Management.",
   306→                ))
   307→        return findings
   308→
   309→    # Non-Windows: check sudo users
   310→    code, out = _run("getent group sudo 2>/dev/null")
   311→    if code == 0 and out:
   312→        users = [u for u in out.split(":")[1].split(",") if u.strip()]
   313→        if len(users) > 2:
   314→            findings.append(Finding(
   315→                title=f"{len(users)} users have sudo privileges",
   316→                severity="medium", category="users",
   317→                module="host",
   318→                description=f"{len(users)} users can run sudo: {', '.join(users[:5])}. Minimize sudo access to reduce risk.",
   319→                evidence=f"sudo group: {out[:200]}",
   320→                asset=hostname, points_deducted=5,
   321→                remediation="Remove unnecessary users from the sudo group. Use specific capabilities instead.",
   322→            ))
   323→
   324→    # Check for passwordless sudo (non-Windows only)
   325→    code, out = _run("sudo -n true 2>&1")
   326→    if code == 0:
   327→        findings.append(Finding(
   328→            title="Passwordless sudo is enabled",
   329→            severity="high", category="users",
   330→            module="host",
   331→            description="Current user can run sudo without a password. Any code running as your user has root access.",
   332→            evidence="sudo -n returned exit code 0",
   333→            asset=hostname, points_deducted=10,
   334→            remediation="Require password for sudo: remove NOPASSWD from /etc/sudoers",
   335→        ))
   336→
   337→    # Check for empty password users (Linux only)
   338→    code, out = _run("sudo awk -F: '($2 == \"\") {print $1}' /etc/shadow 2>/dev/null")
   339→    if code == 0 and out.strip():
   340→        users = out.strip().splitlines()
   341→        findings.append(Finding(
   342→            title=f"{len(users)} user(s) with empty password",
   343→            severity="critical", category="users",
   344→            module="host",
   345→            description=f"Users with no password: {', '.join(users[:5])}. This is a critical security risk.",
   346→            evidence=out[:200],
   347→            asset=hostname, points_deducted=15,
   348→            remediation="Set passwords for all users immediately: sudo passwd <username>",
   349→        ))
   350→
   351→    return findings

   354→def _check_ssh() -> List[Finding]:
   355→    """Check SSH configuration hardening."""
   356→    findings: List[Finding] = []
   357→    hostname = _hostname()
   358→    sshd_config = "/etc/ssh/sshd_config"
   359→
   360→    if not _file_exists(sshd_config):
   361→        return findings
   362→
   363→    try:
   364→        with open(sshd_config) as f:
   365→            config = f.read()
   366→    except Exception:
   367→        return findings
   368→
   369→    ssh_checks = [
   370→        (r'^#?PermitRootLogin\s+yes', "high", "SSH root login is permitted", "Set PermitRootLogin no"),
   371→        (r'^#?PasswordAuthentication\s+yes', "medium", "SSH password authentication enabled", "Use SSH keys only: PasswordAuthentication no"),
   372→        (r'^#?PermitEmptyPasswords\s+yes', "critical", "SSH allows empty passwords", "Set PermitEmptyPasswords no"),
   373→        (r'^#?X11Forwarding\s+yes', "low", "SSH X11 forwarding enabled", "Set X11Forwarding no if not needed"),
   374→        (r'^#?MaxAuthTries\s+(\d+)', None, "", ""),  # just read value
   375→        (r'^#?Protocol\s+1', "high", "SSH Protocol 1 in use (insecure)", "Use Protocol 2"),
   376→        (r'^#?AllowAgentForwarding\s+yes', "low", "SSH agent forwarding enabled", "Disable if not needed: AllowAgentForwarding no"),
   377→    ]
   378→
   379→    for pattern, sev, title, remediation in ssh_checks:
   380→        if re.search(pattern, config, re.MULTILINE | re.IGNORECASE):
   381→            if sev is None:
   382→                # MaxAuthTries check
   383→                match = re.search(r'^#?MaxAuthTries\s+(\d+)', config, re.MULTILINE)
   384→                if match and int(match.group(1)) > 6:
   385→                    findings.append(Finding(
   386→                        title=f"SSH MaxAuthTries is {match.group(1)} (too high)",
   387→                        severity="low", category="ssh",
   388→                        module="host",
   389→                        description=f"MaxAuthTries={match.group(1)} allows more brute-force attempts.",
   390→                        evidence=f"MaxAuthTries {match.group(1)}",
   391→                        asset=hostname, points_deducted=2,
   392→                        remediation="Set MaxAuthTries 4",
   393→                    ))
   394→            else:
   395→                findings.append(Finding(
   396→                    title=title,
   397→                    severity=sev, category="ssh",
   398→                    module="host",
   399→                    description=f"SSH config issue: {title}",
   400→                    evidence=f"Found in {sshd_config}",
   401→                    asset=hostname, points_deducted=8 if sev == "high" else (15 if sev == "critical" else 3),
   402→                    remediation=remediation,
   403→                ))
   404→
   405→    return findings

   502→def _check_file_permissions() -> List[Finding]:

   504→    findings: List[Finding] = []
   505→    hostname = _hostname()
   506→
   507→    if _is_windows():
   508→        sensitive_files = [
   509→            ("~/.ssh/id_rsa", 0o600, "critical", "SSH private key"),
   510→            ("~/.ssh/id_ed25519", 0o600, "critical", "Ed25519 private key"),
   511→            ("~/.ssh/authorized_keys", 0o600, "high", "SSH authorized keys"),

   518→            (os.path.join(os.environ.get('APPDATA', ''), 'credentials'), 0o600, "critical", "Windows Credential Manager backup"),
   519→        ]
   520→    else:
   521→        sensitive_files = [
   522→            ("/etc/shadow", 0o640, "critical", "Password hash file"),
   523→            ("/etc/passwd", 0o644, "medium", "User account file"),
   524→            ("/etc/sudoers", 0o440, "high", "Sudoers file"),
   525→            ("/etc/ssh/sshd_config", 0o600, "medium", "SSH server config"),
   526→            ("~/.ssh/id_rsa", 0o600, "critical", "SSH private key"),
   527→            ("~/.ssh/id_ed25519", 0o600, "critical", "Ed25519 private key"),
   528→            ("~/.ssh/authorized_keys", 0o600, "high", "SSH authorized keys"),
   529→            ("~/.ssh/config", 0o600, "medium", "SSH client config"),
   530→            ("~/.gnupg", 0o700, "high", "GPG directory"),
   531→            ("~/.aws/credentials", 0o600, "critical", "AWS credentials file"),
   532→            ("~/.netrc", 0o600, "high", "netrc credentials file"),
   533→            ("~/.pgpass", 0o600, "high", "PostgreSQL password file"),
   534→        ]
   535→
   536→    for path, expected, sev, desc in sensitive_files:
   537→        full_path = os.path.expanduser(path)
   538→        if not _file_exists(full_path):
   539→            continue
   540→        perms = _file_perms(full_path)
   541→        if perms is None:
   542→            continue

   550→            findings.append(Finding(
   551→                title=f"World-readable: {desc} ({path})",

   815→def _check_suid_sgid() -> List[Finding]:
   816→    """Find world-writable SUID/SGID binaries (Linux) or auto-elevatable binaries (Windows)."""
   817→    findings: List[Finding] = []
   818→    hostname = _hostname()
   819→
   820→    if _is_windows():
   821→        # Windows: check for known auto-elevatable binaries in PATH

   823→            "compmgmtlauncher.exe", "dcomcnfg.exe", "eventvwr.exe",
   824→            "fxcopcmd.exe", "msconfig.exe", "regedit.exe", "resmon.exe",
   825→            "taskmgr.exe", "comexp.exe", "mmc.exe",
   826→        ]
   827→        code, out = _run("echo %PATH%")

   829→            return findings

   831→        found_elevatable = []
   832→        for d in path_dirs[:30]:
   833→            if not os.path.isdir(d):
   834→                continue
   835→            try:
   836→                for entry in os.listdir(d):
   837→                    if entry.lower() in auto_elevatable:
   838→                        found_elevatable.append(os.path.join(d, entry))
   839→            except PermissionError:
   840→                continue
   841→        if found_elevatable:
   842→            findings.append(Finding(
   843→                title=f"{len(found_elevatable)} auto-elevatable binaries found in PATH",
   844→                severity="medium", category="suid_sgid",
   845→                module="host",
   846→                description="Auto-elevatable binaries can escalate privileges without UAC prompt. Ensure only trusted binaries are in PATH.",
   847→                evidence="; ".join(found_elevatable[:5]),
   848→                asset=hostname, points_deducted=5,
   849→                remediation="Review PATH entries and remove unnecessary directories. Restrict file permissions on auto-elevatable binaries.",
   850→            ))

   853→    if _is_macos():

   855→        code, out = _run("find /usr/bin /usr/sbin /bin /sbin -perm -4000 2>/dev/null")

   859→            for b in suid_bins:

   861→                    if os.stat(b).st_mode & stat.S_IWOTH:
   862→                        world_writable.append(b)
   863→                except OSError:
   864→                    continue

   870→                    description="SUID binaries that are world-writable can be replaced by any user to run code as root.",
   871→                    evidence="; ".join(world_writable[:5]),
   872→                    asset=hostname, points_deducted=15,
   873→                    remediation="Remove world-write: sudo chmod o-w <binary> for each listed file.",
   874→                ))
   875→        return findings
   876→
   877→    # Linux: full SUID/SGID audit
   878→    code, out = _run("find /usr -perm -4000 -o -perm -2000 2>/dev/null")

   893→                description="SUID/SGID binaries that are world-writable can be replaced by any user to run code as elevated user/root.",
   894→                evidence="; ".join(world_writable[:5]),
   895→                asset=hostname, points_deducted=15,
   896→                remediation="Remove world-write: sudo chmod o-w <binary> for each listed file.",
   897→            ))
   898→        else:

   900→                findings.append(Finding(
   901→                    title=f"{len(suid_sgid_bins)} SUID/SGID binaries found (large count)",
   902→                    severity="low", category="suid_sgid",
   903→                    module="host",
   904→                    description=f"{len(suid_sgid_bins)} SUID/SGID binaries found. Review for unnecessary privilege escalation paths.",

   994→                asset=hostname, points_deducted=0,
   995→                remediation="",
   996→            ))
   997→        return findings
   998→
   999→    # Neither SELinux nor AppArmor detected

  1001→        title="No Mandatory Access Control (SELinux/AppArmor) detected",
  1002→        severity="medium", category="mac",
  1003→        module="host",
  1004→        description="No SELinux or AppArmor MAC system detected. Consider enabling one for additional security.",
  1005→        evidence="Neither getenforce nor aa-status succeeded",
  1006→        asset=hostname, points_deducted=5,
  1007→        remediation="Install AppArmor: sudo apt install apparmor apparmor-profiles",
  1008→    ))
  1009→    return findings

  1041→def _parse_kernel_version(ver: str) -> Optional[Tuple[int, ...]]:
  1042→    """Parse kernel version string into tuple of ints for comparison."""
  1043→    # Extract version part, stripping distro suffixes
  1044→    match = re.match(r'(\d+\.\d+(?:\.\d+)?)', ver)
  1045→    if not match:
  1046→        return None
  1047→    parts = []
  1048→    for p in match.group(1).split("."):
  1049→        try:
  1050→            parts.append(int(p))
  1051→        except ValueError:
  1052→            break
  1053→    return tuple(parts) if parts else None
  1054→

  1056→def _version_in_range(ver: Tuple[int, ...], min_v: str, max_v: str) -> bool:
  1057→    """Check if ver falls within [min_v, max_v] inclusive."""
  1058→    v_min = _parse_kernel_version(min_v)
  1059→    v_max = _parse_kernel_version(max_v)
  1060→    if v_min is None or v_max is None:
  1061→        return False
  1062→    return v_min <= ver <= v_max
