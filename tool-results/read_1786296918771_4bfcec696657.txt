   280→# ════════════════════════════════════════════════════════════════════════
   281→# REGEX PATTERNS — credential detection, API key detection, secrets
   282→# ════════════════════════════════════════════════════════════════════════
   283→
   284→# ── Email:password combos ───────────────────────────────────────────────
   285→RE_CRED_COMBO = re.compile(
   286→    r'[\w.+-]+@[\w.-]+\s*[:=]\s*['\"\x27]?\S{4,128}['\"\x27]?',
   287→    re.I,
   288→)
   289→
   290→# ── Generic email pattern (for counting domain-specific emails) ─────────
   291→RE_EMAIL = re.compile(r'[\w.+-]+@([\w.-]+)', re.I)
   292→
   293→# ── AWS Access Key IDs ──────────────────────────────────────────────────
   294→RE_AWS_KEY_ID = re.compile(r'(?:AKIA|ASIA)[0-9A-Z]{16}')
   295→
   296→# ── AWS Secret Access Keys (heuristic: 40-char base64-ish after key id) ─
   297→RE_AWS_SECRET = re.compile(
   298→    r'(?:aws_secret_access_key|AWS_SECRET_ACCESS_KEY)\s*[=:[:]]\s*['\"\x27]?([A-Za-z0-9/+=]{40})['\"\x27]?',
   299→    re.I,
   300→)
   301→
   302→# ── Generic API keys ────────────────────────────────────────────────────
   303→RE_GENERIC_API_KEY = re.compile(
   304→    r'(?:api[_-]?key|apikey|api[_-]?secret)\s*[=:]\s*['\"\x27]?([A-Za-z0-9_\-]{16,128})['\"\x27]?',
   305→    re.I,
   306→)
   307→
   308→# ── Slack tokens ────────────────────────────────────────────────────────
   309→RE_SLACK_TOKEN = re.compile(r'xox[baprs]-[0-9a-zA-Z-]{10,}')
   310→
   311→# ── Slack Webhook ───────────────────────────────────────────────────────
   312→RE_SLACK_WEBHOOK = re.compile(
   313→    r'https://hooks\.slack\.com/services/T[A-Z0-9]+/B[A-Z0-9]+/[A-Za-z0-9]+',
   314→)
   315→
   316→# ── GitHub Personal Access Tokens ───────────────────────────────────────
   317→RE_GITHUB_TOKEN = re.compile(r'gh[pousr]_[A-Za-z0-9_]{36,}')
   318→
   319→# ── Google API keys ─────────────────────────────────────────────────────
   320→RE_GOOGLE_API_KEY = re.compile(r'AIza[0-9A-Za-z\-_]{35}')
   321→
   322→# ── Firebase ─────────────────────────────────────────────────────────────
   323→RE_FIREBASE_KEY = re.compile(r'AAAA[A-Za-z0-9_-]{7}[A-Za-z0-9_-]{38}')
   324→
   325→# ── Stripe (publishable or secret) ──────────────────────────────────────
   326→RE_STRIPE_KEY = re.compile(r'(?:pk|sk|rk)_live_[0-9a-zA-Z]{24,}')
   327→
   328→# ── JWT tokens ──────────────────────────────────────────────────────────
   329→RE_JWT = re.compile(
   330→    r'eyJ[A-Za-z0-9_-]{10,}\.eyJ[A-Za-z0-9_-]{10,}\.[A-Za-z0-9_-]{10,}',
   331→)
   332→
   333→# ── Private keys (PEM blocks) ──────────────────────────────────────────
   334→RE_PRIVATE_KEY = re.compile(
   335→    r'-----BEGIN\s+(?:RSA\s+)?PRIVATE\s+KEY-----',
   336→    re.I,
   337→)
   338→
   339→# ── Database connection strings ─────────────────────────────────────────
   340→RE_DB_CONN = re.compile(
   341→    r'(?:mongodb|mysql|postgres|redis|amqp)://[^\s'\"\x27]+',
   342→    re.I,
   343→)
   344→
   345→# ── Generic token/secret assignment ─────────────────────────────────────
   346→RE_TOKEN_ASSIGNMENT = re.compile(
   347→    r'(?:token|secret|password|passwd|pwd)\s*[=:]\s*['\"\x27]?(\S{8,128})['\"\x27]?',
   348→    re.I,
   349→)
   350→
   351→# ── IPv4 addresses ──────────────────────────────────────────────────────
   352→RE_IPV4 = re.compile(r'\b(?:(?:25[0-5]|2[0-4]\d|[01]?\d\d?)\.){3}(?:25[0-5]|2[0-4]\d|[01]?\d\d?)\b')
   353→
   354→# ── ISO-8601 / common date-time patterns for temporal analysis ───────────
   355→RE_DATETIME = re.compile(
   356→    r'\b(\d{4}[-/]\d{2}[-/]\d{2}(?:[T\s]\d{2}:\d{2}(?::\d{2})?(?:Z|[+-]\d{2}:?\d{2})?)?)\b',
   357→)
   358→
   359→# ── Breach dump indicators ─────────────────────────────────────────────
   360→RE_BREACH_INDICATOR = re.compile(
   361→    r'(?:breach|dump|leak|leaked|compromised|pwned|exposed|database.?dump|combolist)\b',
   362→    re.I,
   363→)
   364→
   365→# ── SHA-1 hash (commonly used in HIBP k-anonymity) ──────────────────────
   366→RE_SHA1_HASH = re.compile(r'\b[0-9a-fA-F]{40}\b')

   370→# MINIMAL HTML TEXT EXTRACTOR (stdlib html.parser)
   371→# ════════════════════════════════════════════════════════════════════════
   372→
   373→class _TextExtractor(HTMLParser):
   374→    """Strip all HTML tags and return visible text."""
   375→
   376→    def __init__(self) -> None:
   377→        super().__init__()
   378→        self._chunks: List[str] = []
   379→
   380→    def handle_data(self, data: str) -> None:
   381→        self._chunks.append(data)
   382→
   383→    def get_text(self) -> str:
   384→        return " ".join(self._chunks)
   385→
   386→
   387→def _strip_html(raw: str) -> str:
   388→    ext = _TextExtractor()
   389→    ext.feed(raw)
   390→    return ext.get_text()

   393→# ════════════════════════════════════════════════════════════════════════
   394→# HELPER — extract domain root from a target string
   395→# ════════════════════════════════════════════════════════════════════════
   396→
   397→def _extract_domain(target: str) -> str:
   398→    """Return a clean, lower-case registrable domain from *target*.
   399→
   400→    Strips leading protocol, path, port, userinfo.  Falls back to
   401→    the raw string lower-cased if no dot is found (IP or single label).
   402→    """
   403→    t = target.strip().lower()
   404→    if t.startswith(("http://", "https://")):
   405→        t = t.split("/", 3)[2]
   406→    if "@" in t:
   407→        t = t.split("@", 1)[1]
   408→    if ":" in t:
   409→        t = t.split(":", 1)[0]
   410→    if "/" in t:
   411→        t = t.split("/", 1)[0]
   412→    return t
   413→

   415→def _extract_email_prefix(domain: str) -> str:
   416→    """Derive plausible email prefixes to search for (e.g. info@, admin@, etc.)."""
   417→    return [f"{p}@{domain}" for p in ("info", "admin", "support", "contact", "test", "root", "webmaster")]
   418→
   419→
   420→def _target_mentions(text: str, domain: str) -> int:
   421→    """Count non-overlapping occurrences of *domain* in *text*."""
   422→    return text.lower().count(domain.lower())
   423→
   424→
   425→def _truncate(s: str, max_len: int = 400) -> str:
   426→    """Truncate *s* to *max_len* chars, appending ellipsis when clipped."""
   427→    if len(s) <= max_len:
   428→        return s
   429→    return s[: max_len - 3] + "..."
   430→
   431→
   432→def _safe_json(body: str) -> Optional[Dict[str, Any]]:
   433→    """Try to parse *body* as JSON; return None on failure."""
   434→    try:
   435→        return json.loads(body)
   436→    except (json.JSONDecodeError, TypeError, ValueError):
   437→        return None
   438→
   439→
   440→def _redact_secret(secret: str, visible: int = 4) -> str:
   441→    """Show only the first *visible* chars of a secret, mask the rest."""
   442→    if len(secret) <= visible:
   443→        return "****"
   444→    return secret[:visible] + "*" * (len(secret) - visible)

   451→def _monitor_pastebin(target: str, domain: str, timeout: int, verify_tls: bool) -> List[Finding]:
   452→    """Poll PasteBin scrape API, filter recent pastes for domain mentions."""
   453→    findings: List[Finding] = []
   454→    cfg = PASTE_SITES["pastebin_scrape"]
   455→    resp = http_probe(cfg["url"], method=cfg["method"],
   456→                      headers=cfg.get("extra_headers"),
   457→                      timeout=timeout, verify_tls=verify_tls,
   458→                      limiter=default_limiter)
   459→    if not resp["ok"] or resp["status"] != 200:
   460→        return findings
   461→
   462→    data = _safe_json(resp["body"])
   463→    if not data or not isinstance(data, list):
   464→        return findings
   465→
   466→    hit_count = 0
   467→    for paste in data:
   468→        key = paste.get("key", "")
   469→        title = paste.get("title", "") or "(untitled)"
   470→        date = paste.get("date", "")
   471→        size = paste.get("size", 0)
   472→        scrape_url = paste.get("scrape_url", "")
   473→
   474→        # Fetch paste content for deeper analysis
   475→        if not scrape_url:
   476→            continue
   477→        paste_resp = http_probe(scrape_url, timeout=timeout,
   478→                                verify_tls=verify_tls, limiter=default_limiter)
   479→        body = paste_resp.get("body", "")
   480→        if _target_mentions(body, domain) == 0:
   481→            continue
   482→
   483→        hit_count += 1
   484→        # Detect credentials inside the paste
   485→        creds = RE_CRED_COMBO.findall(body)
   486→        secrets = (RE_AWS_KEY_ID.findall(body) + RE_GENERIC_API_KEY.findall(body) +
   487→                   RE_JWT.findall(body) + RE_PRIVATE_KEY.findall(body))
   488→
   489→        severity = "critical" if (creds or secrets) else "high"
   490→        desc = (f"PasteBin paste references target domain \"{domain}\". "
   491→                f"Title: \"{title}\" | Date: {date} | Size: {size} bytes")
   492→        if creds:
   493→            desc += f" | {len(creds)} credential combo(s) detected"
   494→        if secrets:
   495→            desc += f" | {len(secrets)} secret/key pattern(s) detected"
   496→
   497→        evidence_parts = [f"Paste: {scrape_url}"]
   498→        if creds:
   499→            evidence_parts.append(f"Sample creds: {_redact_secret(creds[0].split(':')[1].strip())}")
   500→        if secrets:
   501→            evidence_parts.append(f"Secret types: {', '.join(set('AWS' if s.startswith('AKIA') else 'JWT' if s.startswith('eyJ') else 'API_KEY' for s in secrets[:5]))}")
   502→
   503→        findings.append(Finding(
   504→            title=f"PasteBin Mention — {title}",
   505→            severity=severity,
   506→            category="Paste Site Exposure",
   507→            module=MODULE_NAME,
   508→            description=desc,
   509→            evidence=_truncate("; ".join(evidence_parts)),
   510→            asset=domain,
   511→            points_deducted=SEV_POINTS.get(severity, 0),
   512→            remediation="Review the paste content. Request removal from PasteBin if it contains sensitive data. Rotate any exposed credentials immediately.",
   513→            dread_score=_dread(9, 7, 8, 6, 7),
   514→        ))
   515→        time.sleep(cfg.get("rate_delay", 1.0))
   516→
   517→    if hit_count == 0:
   518→        findings.append(Finding(
   519→            title="PasteBin Monitoring — Clean",
   520→            severity="info",
   521→            category="Paste Site Exposure",
   522→            module=MODULE_NAME,
   523→            description=f"No mentions of \"{domain}\" found in recent PasteBin public pastes (last 50).",
   524→            evidence="Scanned recent 50 PasteBin listings.",
   525→            asset=domain,
   526→            points_deducted=0,
   527→            remediation="",
   528→            dread_score=0.0,
   529→        ))
   530→    return findings

   533→def _monitor_pastebin_search(target: str, domain: str, timeout: int, verify_tls: bool) -> List[Finding]:
   534→    """Extended PasteBin listing search (up to 100 recent) for domain/email keywords."""
   535→    findings: List[Finding] = []
   536→    cfg = PASTE_SITES["pastebin_search"]
   537→    url = cfg["url_template"]
   538→    resp = http_probe(url, method=cfg["method"],
   539→                      headers=cfg.get("extra_headers"),
   540→                      timeout=timeout, verify_tls=verify_tls,
   541→                      limiter=default_limiter)
   542→    if not resp["ok"] or resp["status"] != 200:
   543→        return findings
   544→
   545→    data = _safe_json(resp["body"])
   546→    if not data or not isinstance(data, list):
   547→        return findings
   548→
   549→    for paste in data:
   550→        key = paste.get("key", "")
   551→        scrape_url = paste.get("scrape_url", "")
   552→        if not scrape_url:
   553→            continue
   554→        paste_resp = http_probe(scrape_url, timeout=timeout,
   555→                                verify_tls=verify_tls, limiter=default_limiter)
   556→        body = paste_resp.get("body", "")
   557→        if _target_mentions(body, domain) == 0:
   558→            continue
   559→        # Already found by _monitor_pastebin — skip duplicates
   560→        time.sleep(1.0)
   561→    # Results are folded into _monitor_pastebin findings to avoid duplicates.
   562→    return findings

   565→def _monitor_github_gists(target: str, domain: str, timeout: int, verify_tls: bool) -> List[Finding]:
   566→    """Search GitHub code/gists for the target domain."""
   567→    findings: List[Finding] = []
   568→    cfg = PASTE_SITES["github_gists"]
   569→    query = urllib.parse.quote_plus(domain)
   570→    url = cfg["url_template"].format(query=query)
   571→
   572→    resp = http_probe(url, method=cfg["method"],
   573→                      headers=cfg.get("extra_headers"),
   574→                      timeout=timeout, verify_tls=verify_tls,
   575→                      limiter=default_limiter)
   576→
   577→    if not resp["ok"] or resp["status"] not in (200, 403):
   578→        if resp["status"] == 403:
   579→            findings.append(Finding(
   580→                title="GitHub Gist Search — Rate Limited",
   581→                severity="info",
   582→                category="Paste Site Exposure",
   583→                module=MODULE_NAME,
   584→                description="GitHub Search API returned 403 (rate limit or authentication required). "
   585→                            "Provide a GITHUB_TOKEN env var for higher limits.",
   586→                evidence=f"Status {resp['status']}: {resp['reason']}",
   587→                asset=domain,
   588→                points_deducted=0,
   589→                remediation="Set GITHUB_TOKEN environment variable for unauthenticated rate limit bypass.",
   590→                dread_score=0.0,
   591→            ))
   592→        return findings
   593→
   594→    data = _safe_json(resp["body"])
   595→    if not data:
   596→        return findings
   597→
   598→    total_count = data.get("total_count", 0)
   599→    items = data.get("items", [])
   600→
   601→    if total_count == 0:
   602→        findings.append(Finding(
   603→            title="GitHub Gist/Code Search — Clean",
   604→            severity="info",
   605→            category="Paste Site Exposure",
   606→            module=MODULE_NAME,
   607→            description=f"No public GitHub code/gist mentions of \"{domain}\" found.",
   608→            evidence=f"GitHub search returned 0 results for \"{domain}\".",
   609→            asset=domain,
   610→            points_deducted=0,
   611→            remediation="",
   612→            dread_score=0.0,
   613→        ))
   614→        return findings
   615→
   616→    for item in items[:10]:  # cap at 10 to avoid flooding
   617→        name = item.get("name", "(unnamed)")
   618→        html_url = item.get("html_url", "")
   619→        repo_name = item.get("repository", {}).get("full_name", "")
   620→        path = item.get("path", "")
   621→
   622→        severity = "high"
   623→        desc = (f"GitHub code/gist result references \"{domain}\". "
   624→                f"Repo: {repo_name} | File: {path}")
   625→
   626→        findings.append(Finding(
   627→            title=f"GitHub Code Mention — {name}",
   628→            severity=severity,
   629→            category="Paste Site Exposure",
   630→            module=MODULE_NAME,
   631→            description=desc,
   632→            evidence=_truncate(f"URL: {html_url} | Repo: {repo_name} | Path: {path}"),
   633→            asset=domain,
   634→            points_deducted=SEV_POINTS[severity],
   635→            remediation="Review the exposed code for hardcoded credentials or secrets. Contact the repository owner for removal if needed.",
   636→            dread_score=_dread(8, 6, 7, 5, 8),
   637→        ))
   638→        time.sleep(cfg.get("rate_delay", 2.0))
   639→
   640→    if total_count > 10:
   641→        findings.append(Finding(
   642→            title=f"GitHub Code Search — {total_count} Total Results",
   643→            severity="medium",
   644→            category="Paste Site Exposure",
   645→            module=MODULE_NAME,
   646→            description=f"GitHub search returned {total_count} total results for \"{domain}\". "
   647→                        f"Top 10 shown above.",
   648→            evidence=f"Total results: {total_count}",
   649→            asset=domain,
   650→            points_deducted=SEV_POINTS["medium"],
   651→            remediation="Perform manual review of all GitHub results. Use GitHub secret scanning to detect and revoke exposed credentials.",
   652→            dread_score=_dread(7, 5, 6, 4, 7),
   653→        ))
   654→
   655→    return findings

   658→def _monitor_html_paste_site(
   659→    site_key: str,
   660→    target: str,
   661→    domain: str,
   662→    timeout: int,
   663→    verify_tls: bool,
   664→) -> List[Finding]:
   665→    """Generic HTML paste-site monitor (PasteSite, Rentry, Ghostbin, Paste.ee, JustPaste)."""
   666→    findings: List[Finding] = []
   667→    cfg = PASTE_SITES[site_key]
   668→    query = urllib.parse.quote_plus(domain)
   669→    url = cfg["url_template"].format(query=query)
   670→
   671→    resp = http_probe(url, method=cfg["method"],
   672→                      headers=cfg.get("extra_headers"),
   673→                      timeout=timeout, verify_tls=verify_tls,
   674→                      limiter=default_limiter)
   675→
   676→    if not resp["ok"]:
   677→        # Not an error per se — site may be down or block bots
   678→        return findings
   679→
   680→    body = resp.get("body", "")
   681→
   682→    # For JSON-parseable sites (Rentry)
   683→    if cfg["parse"] == "rentry":
   684→        data = _safe_json(body)
   685→        if data and isinstance(data, dict):
   686→            results = data.get("results", data.get("data", []))
   687→            for entry in (results if isinstance(results, list) else []):
   688→                text = json.dumps(entry) if isinstance(entry, dict) else str(entry)
   689→                if _target_mentions(text, domain) > 0:
   690→                    findings.append(Finding(
   691→                        title=f"{cfg['name']} — Domain Mention",
   692→                        severity="high",
   693→                        category="Paste Site Exposure",
   694→                        module=MODULE_NAME,
   695→                        description=f"{cfg['name']} contains a paste referencing \"{domain}\".",
   696→                        evidence=_truncate(text[:400]),
   697→                        asset=domain,
   698→                        points_deducted=SEV_POINTS["high"],
   699→                        remediation=f"Review the paste on {cfg['name']} and request removal if it contains sensitive data.",
   700→                        dread_score=_dread(8, 6, 7, 5, 6),
   701→                    ))
   702→        if not findings:
   703→            findings.append(Finding(
   704→                title=f"{cfg['name']} — Clean",
   705→                severity="info",
   706→                category="Paste Site Exposure",
   707→                module=MODULE_NAME,
   708→                description=f"No mentions of \"{domain}\" on {cfg['name']}.",
   709→                evidence=f"Queried {cfg['name']} search endpoint.",
   710→                asset=domain,
   711→                points_deducted=0,
   712→                remediation="",
   713→                dread_score=0.0,
   714→            ))
   715→        return findings
   716→
   717→    # For HTML sites — extract text and search
   718→    text = _strip_html(body)
   719→    if _target_mentions(text, domain) > 0:
   720→        # Check for credentials in the results page
   721→        creds = RE_CRED_COMBO.findall(text)
   722→        severity = "critical" if creds else "high"
   723→        desc = (f"{cfg['name']} search returned results mentioning \"{domain}\".")
   724→        if creds:
   725→            desc += f" {len(creds)} credential combo(s) detected in results."
   726→
   727→        findings.append(Finding(
   728→            title=f"{cfg['name']} — Domain Mention",
   729→            severity=severity,
   730→            category="Paste Site Exposure",
   731→            module=MODULE_NAME,
   732→            description=desc,
   733→            evidence=_truncate(text[:400]),
   734→            asset=domain,
   735→            points_deducted=SEV_POINTS.get(severity, 0),
   736→            remediation=f"Review the paste content on {cfg['name']}. Request removal if credentials are exposed.",
   737→            dread_score=_dread(8, 6, 7, 5, 6),
   738→        ))
   739→    else:
   740→        findings.append(Finding(
   741→            title=f"{cfg['name']} — Clean",
   742→            severity="info",
   743→            category="Paste Site Exposure",
   744→            module=MODULE_NAME,
   745→            description=f"No mentions of \"{domain}\" on {cfg['name']}.",
   746→            evidence=f"Queried: {url}",
   747→            asset=domain,
   748→            points_deducted=0,
   749→            remediation="",
   750→            dread_score=0.0,
   751→        ))
   752→
   753→    return findings

   756→def _run_paste_monitoring(target: str, domain: str, timeout: int, verify_tls: bool) -> List[Finding]:
   757→    """Orchestrate all paste site monitors."""
   758→    findings: List[Finding] = []
   759→
   760→    # PasteBin (primary)
   761→    findings.extend(_monitor_pastebin(target, domain, timeout, verify_tls))
   762→    time.sleep(1.0)
   763→
   764→    # PasteBin extended
   765→    findings.extend(_monitor_pastebin_search(target, domain, timeout, verify_tls))
   766→    time.sleep(1.0)
   767→
   768→    # GitHub Gists
   769→    findings.extend(_monitor_github_gists(target, domain, timeout, verify_tls))
   770→    time.sleep(1.0)
   771→
   772→    # HTML paste sites
   773→    html_sites = ["pastesite", "rentry_co", "ghostbin", "paste_ee", "justpaste"]
   774→    for site_key in html_sites:
   775→        findings.extend(
   776→            _monitor_html_paste_site(site_key, target, domain, timeout, verify_tls)
   777→        )
   778→        time.sleep(PASTE_SITES[site_key].get("rate_delay", 1.5))
   779→
   780→    return findings

   784→# 2. CREDENTIAL LEAK DETECTION
   785→# ════════════════════════════════════════════════════════════════════════
   786→
   787→def _scan_for_credentials(text: str, domain: str) -> List[Dict[str, Any]]:
   788→    """Scan *text* for credential patterns associated with *domain*.
   789→
   790→    Returns a list of dicts with keys: type, match, redacted.
   791→    """
   792→    hits: List[Dict[str, Any]] = []
   793→    seen: set = set()
   794→
   795→    # Email:password combos
   796→    for m in RE_CRED_COMBO.finditer(text):
   797→        raw = m.group(0)
   798→        if raw not in seen:
   799→            seen.add(raw)
   800→            parts = raw.split(":", 1)
   801→            email_part = parts[0].strip().strip('"\'')
   802→            # Only flag if the email belongs to the target domain
   803→            if domain.lower() in email_part.lower():
   804→                pwd_part = parts[1].strip().strip('"\'') if len(parts) > 1 else "(password)"
   805→                hits.append({
   806→                    "type": "email_password",
   807→                    "match": raw,
   808→                    "redacted": f"{email_part}:{_redact_secret(pwd_part)}",
   809→                })
   810→
   811→    # API keys
   812→    for pattern, label in [
   813→        (RE_AWS_KEY_ID, "AWS Access Key ID"),
   814→        (RE_GENERIC_API_KEY, "Generic API Key"),
   815→        (RE_SLACK_TOKEN, "Slack Token"),
   816→        (RE_SLACK_WEBHOOK, "Slack Webhook"),
   817→        (RE_GITHUB_TOKEN, "GitHub Token"),
   818→        (RE_GOOGLE_API_KEY, "Google API Key"),
   819→        (RE_FIREBASE_KEY, "Firebase Key"),
   820→        (RE_STRIPE_KEY, "Stripe Key"),
   821→    ]:
   822→        for m in pattern.finditer(text):
   823→            raw = m.group(0)
   824→            if raw not in seen:
   825→                seen.add(raw)
   826→                hits.append({
   827→                    "type": label,
   828→                    "match": raw,
   829→                    "redacted": _redact_secret(raw),
   830→                })
   831→
   832→    # JWT tokens
   833→    for m in RE_JWT.finditer(text):
   834→        raw = m.group(0)
   835→        if raw not in seen:
   836→            seen.add(raw)
   837→            # Try to decode the header to see if it references the target
   838→            try:
   839→                header_b64 = raw.split(".")[0]
   840→                # Add padding
   841→                header_b64 += "=" * (4 - len(header_b64) % 4)
   842→                header_json = base64.urlsafe_b64decode(header_b64)
   843→                header = json.loads(header_json)
   844→                iss = header.get("iss", "") or ""
   845→                if domain.lower() in iss.lower():
   846→                    hits.append({
   847→                        "type": "JWT Token (target issuer)",
   848→                        "match": raw,
   849→                        "redacted": _redact_secret(raw, 20),
   850→                    })
   851→            except Exception:
   852→                # Generic JWT found near domain context
   853→                context_start = max(0, m.start() - 100)
   854→                context = text[context_start:m.start() + len(raw)]
   855→                if domain.lower() in context.lower():
   856→                    hits.append({
   857→                        "type": "JWT Token (near domain context)",
   858→                        "match": raw,
   859→                        "redacted": _redact_secret(raw, 20),
   860→                    })
   861→
   862→    # Private keys
   863→    for m in RE_PRIVATE_KEY.finditer(text):
   864→        raw = m.group(0)
   865→        if raw not in seen:
   866→            seen.add(raw)
   867→            hits.append({
   868→                "type": "Private Key (PEM)",
   869→                "match": raw,
   870→                "redacted": "-----BEGIN PRIVATE KEY----- [REDACTED]",
   871→            })
   872→
   873→    # Database connection strings
   874→    for m in RE_DB_CONN.finditer(text):
   875→        raw = m.group(0)
   876→        if raw not in seen and domain.lower() in raw.lower():
   877→            seen.add(raw)
   878→            # Redact password in connection string
   879→            redacted = re.sub(r'(://[^:]+:)[^@]+(@)', r'\1****\2', raw)
   880→            hits.append({
   881→                "type": "Database Connection String",
   882→                "match": raw,
   883→                "redacted": redacted,
   884→            })
   885→
   886→    # Generic token assignments near domain context
   887→    for m in RE_TOKEN_ASSIGNMENT.finditer(text):
   888→        raw = m.group(0)
   889→        if raw not in seen:
   890→            seen.add(raw)
   891→            context_start = max(0, m.start() - 80)
   892→            context = text[context_start:m.end()]
   893→            if domain.lower() in context.lower():
   894→                hits.append({
   895→                    "type": "Token/Secret Assignment",
   896→                    "match": raw,
   897→                    "redacted": _redact_secret(m.group(1), 4),
   898→                })
   899→
   900→    return hits

   910→    """Run credential leak detection across collected paste bodies.

   958→    # Group by type for reporting
   959→    type_counts: Dict[str, int] = {}
   960→    for c in all_creds:
   961→        type_counts[c["type"]] = type_counts.get(c["type"], 0) + 1
   962→
   963→    severity = "critical"
   964→    desc_parts = [f"{len(all_creds)} credential/secret pattern(s) found associated with \"{domain}\":"]
   965→    for ctype, count in sorted(type_counts.items(), key=lambda x: -x[1]):
   966→        desc_parts.append(f"  - {ctype}: {count}")
   967→
   968→    evidence_parts = []
   969→    for c in all_creds[:10]:  # Show first 10
   970→        evidence_parts.append(f"[{c['type']}] {c['redacted']}")
   971→
   972→    findings.append(Finding(
   973→        title=f"Credential Leak Detected — {len(all_creds)} Pattern(s)",
   974→        severity=severity,
   975→        category="Credential Leak",
   976→        module=MODULE_NAME,
   977→        description="\n".join(desc_parts),
   978→        evidence=_truncate(" | ".join(evidence_parts), 600),
   979→        asset=domain,
   980→        points_deducted=SEV_POINTS[severity],
   981→        remediation="1. Immediately rotate all exposed credentials. 2. Force password resets for affected accounts. 3. Revoke exposed API keys and tokens. 4. Investigate the source of the leak. 5. Enable secret scanning in CI/CD pipelines.",
   982→        dread_score=_dread(10, 8, 9, 8, 7),
   983→    ))
   984→
   985→    return findings

  1105→# 4. LEAK TEMPORAL ANALYSIS
  1106→# ════════════════════════════════════════════════════════════════════════
  1107→
  1108→def _analyse_temporal_patterns(leak_dates: List[str]) -> Dict[str, Any]:
  1109→    """Analyse a list of date strings to classify the leak as ongoing or historical.
  1110→
  1111→    Returns dict with: classification, date_range, peak_period, burst_count.
  1112→    """
  1113→    parsed: List[float] = []
  1114→    for d in leak_dates:
  1115→        for m in RE_DATETIME.finditer(d):
  1116→            try:
  1117→                ts = time.mktime(time.strptime(m.group(1)[:10], "%Y-%m-%d"))
  1118→                parsed.append(ts)
  1119→            except ValueError:
  1120→                pass
  1121→
  1122→    if not parsed:
  1123→        return {"classification": "undetermined", "date_range": "N/A", "peak_period": "N/A", "burst_count": 0}
  1124→
  1125→    parsed.sort()
  1126→    earliest = time.strftime("%Y-%m-%d", time.gmtime(parsed[0]))
  1127→    latest = time.strftime("%Y-%m-%d", time.gmtime(parsed[-1]))
  1128→    now = time.time()
  1129→
  1130→    # Calculate inter-arrival gaps
  1131→    gaps: List[float] = []
  1132→    bursts = 0
  1133→    for i in range(1, len(parsed)):
  1134→        gap_days = (parsed[i] - parsed[i - 1]) / 86400.0
  1135→        gaps.append(gap_days)
  1136→        # A "burst" is multiple leaks within 3 days
  1137→        if gap_days <= 3.0:
  1138→            bursts += 1
  1139→
  1140→    days_since_last = (now - parsed[-1]) / 86400.0
  1141→    avg_gap = sum(gaps) / len(gaps) if gaps else 9999
  1142→
  1143→    if days_since_last <= 30:
  1144→        if bursts > len(gaps) * 0.5:
  1145→            classification = "active_breach"
  1146→        else:
  1147→            classification = "ongoing_leak"
  1148→    elif days_since_last <= 180:
  1149→        classification = "recent_historical"
  1150→    else:
  1151→        classification = "historical"
  1152→
  1153→    peak_start = parsed[len(parsed) // 2] if parsed else 0
  1154→    peak_end = parsed[-1] if parsed else 0
  1155→    peak_period = f"{time.strftime('%Y-%m-%d', time.gmtime(peak_start))} to {time.strftime('%Y-%m-%d', time.gmtime(peak_end))}"
  1156→
  1157→    return {
  1158→        "classification": classification,
  1159→        "date_range": f"{earliest} to {latest}",
  1160→        "peak_period": peak_period,
  1161→        "burst_count": bursts,
  1162→        "days_since_last": round(days_since_last, 1),
  1163→        "avg_gap_days": round(avg_gap, 1),
  1164→    }
  1165→
  1166→

  1219→                    leak_dates.append(created)
  1220→

  1222→

  1230→        "ongoing_leak": ("Ongoing Leak", "high",
  1231→                          "Leak activity detected within the last 30 days but not in burst pattern. "

  1233→                          "1. Investigate the leak source immediately. 2. Check for unauthorized access. "

  1260→

  1766→def _scan_github_secrets(domain: str, timeout: int, verify_tls: bool) -> List[Finding]:
  1767→    """Search GitHub code for hardcoded secrets associated with the target org/domain.
  1768→
  1769→    Runs multiple targeted queries covering passwords, API keys, AWS secrets,
  1770→    JWT tokens, DB connection strings, private keys, webhooks, env files, and
  1771→    config secrets.
  1772→    """
  1773→    findings: List[Finding] = []
  1774→    total_results = 0
  1775→    queries_run = 0
  1776→
  1777→    headers = {
  1778→        "Accept": "application/vnd.github.v3+json",
  1779→    }
  1780→
  1781→    for query_name, query_template in GITHUB_SEARCH_QUERIES.items():
  1782→        query = urllib.parse.quote_plus(query_template.format(domain=domain))
  1783→        url = f"https://api.github.com/search/code?q={query}&per_page=5"
  1784→
  1785→        resp = http_probe(url, method="GET", headers=headers,
  1786→                          timeout=timeout, verify_tls=verify_tls,
  1787→                          limiter=default_limiter)
  1788→
  1789→        queries_run += 1
  1790→
  1791→        if resp["status"] == 403:
  1792→            findings.append(Finding(
  1793→                title="GitHub Secret Scan — Rate Limited",
  1794→                severity="info",
  1795→                category="Secret Exposure",
  1796→                module=MODULE_NAME,
  1797→                description="GitHub Search API rate limit reached during secret scanning. "
  1798→                            "Set GITHUB_TOKEN for higher limits.",
  1799→                evidence=f"Completed {queries_run}/{len(GITHUB_SEARCH_QUERIES)} queries before rate limit.",
  1800→                asset=domain,
  1801→                points_deducted=0,
  1802→                remediation="Set GITHUB_TOKEN environment variable. Use GitHub secret scanning and push protection in repository settings.",
  1803→                dread_score=0.0,
  1804→            ))
  1805→            break
  1806→
  1807→        if not resp["ok"] or resp["status"] != 200:
  1808→            time.sleep(2.0)
  1809→            continue
  1810→
  1811→        data = _safe_json(resp["body"])
  1812→        if not data:
  1813→            time.sleep(2.0)
  1814→            continue
  1815→
  1816→        count = data.get("total_count", 0)
  1817→        items = data.get("items", [])
  1818→
  1819→        if count > 0:
  1820→            total_results += count
  1821→            # Generate a finding for each query that returned results
  1822→            human_name = query_name.replace("_", " ").title()
  1823→            severity = "critical" if query_name in ("passwords_in_code", "aws_secrets", "private_keys") else "high"
  1824→
  1825→            repo_list = []
  1826→            for item in items[:5]:
  1827→                repo = item.get("repository", {}).get("full_name", "")
  1828→                path = item.get("path", "")
  1829→                html_url = item.get("html_url", "")
  1830→                repo_list.append(f"{repo}:{path}")
  1831→
  1832→            findings.append(Finding(
  1833→                title=f"GitHub Secret Exposure — {human_name} ({count} results)",
  1834→                severity=severity,
  1835→                category="Secret Exposure",
  1836→                module=MODULE_NAME,
  1837→                description=f"GitHub code search found {count} result(s) for {human_name} "
  1838→                            f"associated with \"{domain}\". "
  1839→                            f"Top repositories: {', '.join(repo_list[:5])}",
  1840→                evidence=_truncate(f"Query: {query_template.format(domain=domain)[:200]} | "
  1841→                                  f"Results: {', '.join(repo_list[:3])}"),
  1842→                asset=domain,
  1843→                points_deducted=SEV_POINTS[severity],
  1844→                remediation=("1. Review and remove exposed secrets immediately. 2. Rotate all exposed credentials. "
  1845→                             "3. Enable GitHub secret scanning and push protection. 4. Use .gitignore for sensitive files. "
  1846→                             "5. Implement pre-commit hooks for secret detection. 6. Use environment variables or secret managers."),
  1847→                dread_score=_dread(9, 7, 8, 6, 8),
  1848→            ))
  1849→
  1850→        time.sleep(2.0)  # Respect GitHub rate limits
  1851→
  1852→    if total_results == 0 and queries_run == len(GITHUB_SEARCH_QUERIES):
  1853→        findings.append(Finding(
  1854→            title="GitHub Secret Exposure — Clean",
  1855→            severity="info",
  1856→            category="Secret Exposure",
  1857→            module=MODULE_NAME,
  1858→            description=f"No exposed secrets found in GitHub code search for \"{domain}\" across {queries_run} query categories.",
  1859→            evidence=f"Queries run: {queries_run}",
  1860→            asset=domain,
  1861→            points_deducted=0,
  1862→            remediation="",
  1863→            dread_score=0.0,
  1864→        ))
  1865→
  1866→    return findings

  1873→def run_dark_web_monitor(
  1874→    target: str,
  1875→    base_url: str,
  1876→    timeout: int = 8,
  1877→    verify_tls: bool = True,
  1878→) -> List[Finding]:
  1879→    """Run the full dark-web / leak-directory monitoring suite against *target*.
  1880→
  1881→    Parameters
  1882→    ----------
  1883→    target : str
  1884→        The target domain, IP, or hostname to monitor.
  1885→    base_url : str
  1886→        The base URL of the target (unused internally but kept for API
  1887→        consistency with other ReconPro modules).
  1888→    timeout : int
  1889→        HTTP timeout in seconds for each probe (default 8).
  1890→    verify_tls : bool
  1891→        Whether to verify TLS certificates (default True).
  1892→
  1893→    Returns
  1894→    -------
  1895→    List[Finding]
  1896→        All findings from the 7 monitoring capabilities.
  1897→    """
  1898→    domain = _extract_domain(target)
  1899→    all_findings: List[Finding] = []
  1900→
  1901→    # ── 1. Paste Site Monitoring ─────────────────────────────────────────
  1902→    all_findings.extend(_run_paste_monitoring(target, domain, timeout, verify_tls))
  1903→
  1904→    # ── 2. Credential Leak Detection ─────────────────────────────────────
  1905→    # Collect paste bodies from step 1 for deeper scanning
  1906→    paste_bodies: List[str] = []
  1907→    cfg = PASTE_SITES["pastebin_scrape"]
  1908→    resp = http_probe(cfg["url"], method=cfg["method"],
  1909→                      headers=cfg.get("extra_headers"),
  1910→                      timeout=timeout, verify_tls=verify_tls,
  1911→                      limiter=default_limiter)
  1912→    if resp["ok"] and resp["status"] == 200:
  1913→        data = _safe_json(resp["body"])
  1914→        if data and isinstance(data, list):
  1915→            for paste in data:
  1916→                scrape_url = paste.get("scrape_url", "")
  1917→                if not scrape_url:
  1918→                    continue
  1919→                pr = http_probe(scrape_url, timeout=timeout,
  1920→                                verify_tls=verify_tls, limiter=default_limiter)
  1921→                if pr["ok"] and _target_mentions(pr.get("body", ""), domain) > 0:
  1922→                    paste_bodies.append(pr["body"])
  1923→                time.sleep(1.0)

  1925→        _run_credential_leak_detection(target, domain, timeout, verify_tls, paste_bodies)
