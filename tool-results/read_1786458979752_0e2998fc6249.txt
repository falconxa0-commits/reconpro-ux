    69→class _FallbackDiGraph:
    70→    """Minimal directed-graph implementation backed by plain dicts."""
    71→
    72→    def __init__(self) -> None:
    73→        self._nodes: Dict[str, Dict[str, Any]] = {}
    74→        self._succ: Dict[str, Dict[str, Dict[str, Any]]] = defaultdict(dict)
    75→        self._pred: Dict[str, Dict[str, Dict[str, Any]]] = defaultdict(dict)
    76→
    77→    # -- mutation ---------------------------------------------------------
    78→    def add_node(self, node: str, **attr: Any) -> None:
    79→        if node not in self._nodes:
    80→            self._nodes[node] = {}
    81→        self._nodes[node].update(attr)
    82→
    83→    def add_edge(self, u: str, v: str, **attr: Any) -> None:
    84→        self.add_node(u)
    85→        self.add_node(v)
    86→        self._succ[u][v] = attr
    87→        self._pred[v][u] = attr
    88→
    89→    def remove_node(self, node: str) -> None:
    90→        self._nodes.pop(node, None)
    91→        for neighbour in list(self._succ.get(node, {})):
    92→            self._pred[neighbour].pop(node, None)
    93→        self._succ.pop(node, None)
    94→        for neighbour in list(self._pred.get(node, {})):
    95→            self._succ[neighbour].pop(node, None)
    96→        self._pred.pop(node, None)
    97→
    98→    # -- queries ----------------------------------------------------------
    99→    def nodes(self, data: bool = False):
   100→        if data:
   101→            return list(self._nodes.items())
   102→        return list(self._nodes.keys())
   103→
   104→    def edges(self, data: bool = False):
   105→        result = []
   106→        for u, nbrs in self._succ.items():
   107→            for v, attr in nbrs.items():
   108→                if data:
   109→                    result.append((u, v, attr))
   110→                else:
   111→                    result.append((u, v))
   112→        return result
   113→
   114→    def successors(self, node: str):
   115→        return list(self._succ.get(node, {}).keys())
   116→
   117→    def predecessors(self, node: str):
   118→        return list(self._pred.get(node, {}).keys())
   119→
   120→    def out_edges(self, node: str, data: bool = False):
   121→        if data:
   122→            return [(node, v, a) for v, a in self._succ.get(node, {}).items()]
   123→        return [(node, v) for v in self._succ.get(node, {}).keys()]
   124→
   125→    def in_edges(self, node: str, data: bool = False):
   126→        if data:
   127→            return [(u, node, a) for u, a in self._pred.get(node, {}).items()]
   128→        return [(u, node) for u in self._pred.get(node, {}).keys()]
   129→
   130→    def has_node(self, node: str) -> bool:
   131→        return node in self._nodes
   132→
   133→    def get_node_data(self, node: str) -> Dict[str, Any]:
   134→        return dict(self._nodes.get(node, {}))
   135→
   136→    def subgraph(self, nodes: Set[str]):
   137→        g = _FallbackDiGraph()
   138→        for n in nodes:
   139→            if n in self._nodes:
   140→                g.add_node(n, **self._nodes[n])
   141→        for u, nbrs in self._succ.items():
   142→            if u in nodes:
   143→                for v, attr in nbrs.items():
   144→                    if v in nodes:
   145→                        g.add_edge(u, v, **attr)
   146→        return g
   147→
   148→    def to_dict(self) -> Dict[str, Any]:
   149→        return {
   150→            "nodes": dict(self._nodes),
   151→            "edges": [
   152→                {"source": u, "target": v, **a}
   153→                for u, nbrs in self._succ.items()
   154→                for v, a in nbrs.items()
   155→            ],
   156→        }
   157→
   158→    @classmethod
   159→    def from_dict(cls, data: Dict[str, Any]):
   160→        g = cls()
   161→        for node, attr in data.get("nodes", {}).items():
   162→            g.add_node(node, **attr)
   163→        for edge in data.get("edges", []):
   164→            src = edge.pop("source")
   165→            tgt = edge.pop("target")
   166→            g.add_edge(src, tgt, **edge)
   167→        return g
   168→
   169→
   170→def _make_graph():
   171→    """Return a NetworkX DiGraph if available, else the fallback."""
   172→    if HAS_NX:
   173→        return nx.DiGraph()
   174→    return _FallbackDiGraph()

   178→# SecurityKnowledgeGraph

   180→
   181→class SecurityKnowledgeGraph:
   182→    """Directed knowledge graph for security scan results.
   183→
   184→    Wraps a NetworkX :class:`DiGraph` when *networkx* is installed;
   185→    otherwise falls back to a lightweight pure-Python implementation.
   186→    """
   187→
   188→    def __init__(self) -> None:
   189→        self._g = _make_graph()
   190→        self._node_type_index: Dict[str, Set[str]] = defaultdict(set)
   191→        self._edge_type_index: Dict[str, List[Tuple[str, str]]] = defaultdict(list)
   192→
   193→    # ------------------------------------------------------------------
   194→    # Internal helpers
   195→    # ------------------------------------------------------------------
   196→
   197→    def _node_id(self, ntype: str, label: str) -> str:
   198→        """Deterministic node id: ``<type>:<label>``."""
   199→        return f"{ntype}:{label}"
   200→
   201→    def _add_indexed_node(self, ntype: str, label: str, **attrs: Any) -> str:
   202→        nid = self._node_id(ntype, label)
   203→        self._g.add_node(nid, node_type=ntype, label=label, **attrs)
   204→        self._node_type_index[ntype].add(nid)
   205→        return nid
   206→
   207→    def _add_indexed_edge(self, src: str, dst: str, etype: str, **attrs: Any) -> None:
   208→        self._g.add_edge(src, dst, edge_type=etype, **attrs)
   209→        self._edge_type_index[etype].append((src, dst))
   210→
   211→    def _safe_sev(self, raw) -> str:
   212→        """Normalise a severity value to lowercase string."""
   213→        if raw is None:
   214→            return "info"
   215→        s = str(raw).strip().lower()
   216→        if s in SEVERITY_ORDER:
   217→            return s
   218→        return "info"

   221→    # Public mutation API

   223→
   224→    def add_scan_result(self, result_dict: Dict[str, Any]) -> None:
   225→        """Ingest a full ``ReconProResult.to_dict()`` payload.
   226→
   227→        Creates nodes/edges for the target, every finding, associated
   228→        technologies, and open ports.
   229→        """
   230→        target = result_dict.get("target") or result_dict.get("host", "unknown")
   231→        self._add_indexed_node("target", target)
   232→
   233→        # Subdomains
   234→        for sub in result_dict.get("subdomains", []):
   235→            self.add_subdomain(sub, target)
   236→
   237→        # Ports & services
   238→        for port_entry in result_dict.get("ports", []):
   239→            if isinstance(port_entry, dict):
   240→                port_num = port_entry.get("port", port_entry.get("number"))
   241→                service = port_entry.get("service", port_entry.get("name", "unknown"))
   242→                state = port_entry.get("state", "open")
   243→            else:
   244→                port_num = port_entry
   245→                service = "unknown"
   246→                state = "open"
   247→
   248→            if port_num is None:
   249→                continue
   250→
   251→            port_id = self._add_indexed_node("port", f"{target}:{port_num}", port=int(port_num), state=state)
   252→            target_id = self._node_id("target", target)
   253→            self._add_indexed_edge(target_id, port_id, "has_port")
   254→
   255→            svc_id = self._add_indexed_node("service", f"{target}:{port_num}:{service}", name=service, port=int(port_num))
   256→            self._add_indexed_edge(port_id, svc_id, "runs_on")
   257→
   258→        # Technologies
   259→        for tech in result_dict.get("technologies", []):
   260→            tech_name = tech if isinstance(tech, str) else tech.get("name", str(tech))
   261→            tech_id = self._add_indexed_node("technology", tech_name)
   262→            target_id = self._node_id("target", target)
   263→            self._add_indexed_edge(target_id, tech_id, "uses_tech")
   264→
   265→        # Findings
   266→        for finding in result_dict.get("findings", []):
   267→            self.add_finding(finding)
   268→
   269→        # Certificates
   270→        for cert in result_dict.get("certificates", []):
   271→            if isinstance(cert, dict):
   272→                cn = cert.get("common_name", cert.get("cn", "unknown"))
   273→                issuer = cert.get("issuer", "")
   274→                expires = cert.get("expires", "")
   275→            else:
   276→                cn = str(cert)
   277→                issuer = ""
   278→                expires = ""
   279→            cert_id = self._add_indexed_node("certificate", f"{target}:{cn}", common_name=cn, issuer=issuer, expires=expires)
   280→            target_id = self._node_id("target", target)
   281→            self._add_indexed_edge(target_id, cert_id, "has_cert")
   282→
   283→        # Endpoints
   284→        for ep in result_dict.get("endpoints", []):
   285→            if isinstance(ep, dict):
   286→                path = ep.get("path", ep.get("url", str(ep)))
   287→                method = ep.get("method", "GET")
   288→            else:
   289→                path = str(ep)
   290→                method = "GET"
   291→            ep_id = self._add_indexed_node("endpoint", f"{target}:{method}:{path}", path=path, method=method)
   292→            target_id = self._node_id("target", target)
   293→            self._add_indexed_edge(target_id, ep_id, "exposes")
   294→
   295→    def add_finding(self, finding_dict: Dict[str, Any]) -> None:
   296→        """Add a finding node and connect it to its parent target."""
   297→        title = finding_dict.get("title", finding_dict.get("name", finding_dict.get("description", "unnamed")))
   298→        severity = self._safe_sev(finding_dict.get("severity"))
   299→        target = finding_dict.get("target", finding_dict.get("host", "unknown"))
   300→        finding_type = finding_dict.get("type", finding_dict.get("category", "generic"))
   301→        description = finding_dict.get("description", "")
   302→        evidence = finding_dict.get("evidence", "")
   303→        cvss_score = finding_dict.get("cvss_score", finding_dict.get("cvss", None))
   304→
   305→        fid = self._add_indexed_node(
   306→            "finding",
   307→            f"{target}:{title}",
   308→            title=title,
   309→            severity=severity,
   310→            finding_type=finding_type,
   311→            description=description,
   312→            evidence=evidence,
   313→            cvss_score=cvss_score,
   314→        )
   315→
   316→        # Ensure target node exists
   317→        self._add_indexed_node("target", target)
   318→        target_id = self._node_id("target", target)
   319→        self._add_indexed_edge(target_id, fid, "has_vuln")
   320→
   321→        # Link to port if specified
   322→        port_num = finding_dict.get("port")
   323→        if port_num is not None:
   324→            port_id = self._node_id("port", f"{target}:{port_num}")
   325→            if self._g.has_node(port_id):
   326→                self._add_indexed_edge(port_id, fid, "has_vuln")
   327→
   328→        # Link related CVEs
   329→        for cve_ref in finding_dict.get("cves", finding_dict.get("references", [])):
   330→            if isinstance(cve_ref, dict):
   331→                cve_id = cve_ref.get("id", cve_ref.get("cve_id", ""))
   332→                cve_sev = cve_ref.get("severity", severity)
   333→                cve_desc = cve_ref.get("description", "")
   334→                affected = cve_ref.get("affected", "")
   335→            else:
   336→                cve_id = str(cve_ref)
   337→                cve_sev = severity
   338→                cve_desc = ""
   339→                affected = ""
   340→            if cve_id:
   341→                self.add_cve(cve_id, cve_sev, cve_desc, affected)
   342→                self._add_indexed_edge(fid, self._node_id("cve", cve_id), "related_cve")
   343→
   344→        # Link related technologies
   345→        for tech in finding_dict.get("technologies", []):
   346→            tech_name = tech if isinstance(tech, str) else tech.get("name", str(tech))
   347→            tech_id = self._add_indexed_node("technology", tech_name)
   348→            self._add_indexed_edge(tech_id, fid, "exposes")
   349→
   350→    def add_cve(
   351→        self,
   352→        cve_id: str,
   353→        severity: str = "info",
   354→        description: str = "",
   355→        affected: str = "",
   356→    ) -> None:
   357→        """Add a CVE node (idempotent) and return its node id."""
   358→        sev = self._safe_sev(severity)
   359→        nid = self._add_indexed_node(
   360→            "cve",
   361→            cve_id,
   362→            severity=sev,
   363→            description=description,
   364→            affected=affected,
   365→        )
   366→        return nid

   368→    def add_subdomain(self, domain: str, parent: str) -> None:
   369→        """Link *domain* as a subdomain of *parent*."""
   370→        sub_id = self._add_indexed_node("subdomain", domain)
   371→        parent_id = self._node_id("target", parent)
   372→        # Parent might be a subdomain itself
   373→        if not self._g.has_node(parent_id):
   374→            parent_id = self._node_id("subdomain", parent)
   375→            if not self._g.has_node(parent_id):
   376→                self._add_indexed_node("target", parent)
   377→                parent_id = self._node_id("target", parent)
   378→        self._add_indexed_edge(parent_id, sub_id, "has_subdomain")

   384→    def get_attack_surface(self, target: str) -> Dict[str, List[Dict[str, Any]]]:
   385→        """Return all reachable endpoints, open ports, and vulns for *target*.
   386→
   387→        Returns a dict with keys ``"endpoints"``, ``"ports"``,
   388→        ``"vulnerabilities"``, and ``"technologies"``.
   389→        """
   390→        target_id = self._node_id("target", target)
   391→        if not self._g.has_node(target_id):
   392→            # Check subdomains
   393→            target_id = self._node_id("subdomain", target)
   394→            if not self._g.has_node(target_id):
   395→                return {"endpoints": [], "ports": [], "vulnerabilities": [], "technologies": []}
   396→
   397→        visited = self._bfs_collect(target_id, max_depth=3)
   398→        result: Dict[str, List[Dict[str, Any]]] = {
   399→            "endpoints": [],
   400→            "ports": [],
   401→            "vulnerabilities": [],
   402→            "technologies": [],
   403→        }
   404→
   405→        for nid in visited:
   406→            if HAS_NX:
   407→                data = self._g.nodes[nid]
   408→            else:
   409→                data = self._g.get_node_data(nid)
   410→            ntype = data.get("node_type", "")
   411→            if ntype == "endpoint":
   412→                result["endpoints"].append(data)
   413→            elif ntype == "port":
   414→                if data.get("state", "open") == "open":
   415→                    result["ports"].append(data)
   416→            elif ntype in ("vulnerability", "finding"):
   417→                result["vulnerabilities"].append(data)
   418→            elif ntype == "technology":
   419→                result["technologies"].append(data)
   420→
   421→        return result

   423→    def get_blast_radius(self, asset: str) -> List[Dict[str, Any]]:
   424→        """BFS from *asset* to find all connected vulnerabilities.
   425→
   426→        *asset* can be a node label, an IP, or a partial node id.
   427→        """
   428→        start = self._resolve_node(asset)
   429→        if start is None:
   430→            return []
   431→
   432→        visited = self._bfs_collect(start, max_depth=10)
   433→        vulns: List[Dict[str, Any]] = []
   434→        for nid in visited:
   435→            if HAS_NX:
   436→                data = self._g.nodes[nid]
   437→            else:
   438→                data = self._g.get_node_data(nid)
   439→            if data.get("node_type") in ("vulnerability", "finding", "cve"):
   440→                vulns.append(data)
   441→        return vulns

   443→    def find_chains(self, target: str) -> List[List[Dict[str, Any]]]:
   444→        """Find multi-hop attack chains originating from *target*.
   445→
   446→        Each chain follows a pattern like::
   447→
   448→            target → port → service → vulnerability → CVE
   449→        """
   450→        target_id = self._resolve_node(target)
   451→        if target_id is None:
   452→            return []
   453→
   454→        chains: List[List[Dict[str, Any]]] = []
   455→        # DFS to find paths that end at CVE or finding nodes
   456→        stack: List[Tuple[str, List[str]]] = [(target_id, [target_id])]
   457→        seen_chains: Set[tuple] = set()
   458→
   459→        while stack:
   460→            current, path = stack.pop()
   461→            succs = self._g.successors(current) if HAS_NX else self._g.successors(current)
   462→            for nxt in succs:
   463→                if nxt in path:
   464→                    continue
   465→                new_path = path + [nxt]
   466→                if HAS_NX:
   467→                    nd = self._g.nodes[nxt]
   468→                else:
   469→                    nd = self._g.get_node_data(nxt)
   470→                ntype = nd.get("node_type", "")
   471→                # A chain is complete if it reaches a CVE or high-sev finding
   472→                if ntype == "cve" or (ntype == "finding" and SEVERITY_ORDER.get(self._safe_sev(nd.get("severity")), 0) >= 2):
   473→                    chain_key = tuple(new_path)
   474→                    if chain_key not in seen_chains:
   475→                        seen_chains.add(chain_key)
   476→                        chain_data = []
   477→                        for nid in new_path:
   478→                            if HAS_NX:
   479→                                chain_data.append(dict(self._g.nodes[nid]))
   480→                            else:
   481→                                chain_data.append(self._g.get_node_data(nid))
   482→                        chains.append(chain_data)
   483→                elif len(new_path) < 8:
   484→                    stack.append((nxt, new_path))
   485→
   486→        # Sort by severity of the terminal node
   487→        def _chain_sort_key(chain):
   488→            last = chain[-1]
   489→            return SEVERITY_ORDER.get(self._safe_sev(last.get("severity")), 0)
   490→
   491→        chains.sort(key=_chain_sort_key, reverse=True)
   492→        return chains
   493→
   494→    def get_vulnerable_assets(self) -> List[Dict[str, Any]]:
   495→        """Return all assets (targets/subdomains) with HIGH or CRITICAL findings."""
   496→        vulnerable: List[Dict[str, Any]] = []
   497→        seen: Set[str] = set()
   498→
   499→        for nid in self._node_type_index.get("finding", set()):
   500→            if HAS_NX:
   501→                data = self._g.nodes[nid]
   502→            else:
   503→                data = self._g.get_node_data(nid)
   504→            sev = self._safe_sev(data.get("severity"))
   505→            if SEVERITY_ORDER.get(sev, 0) < 3:
   506→                continue
   507→            # Walk predecessors to find the owning target/subdomain
   508→            for pred in (self._g.predecessors(nid) if HAS_NX else self._g.predecessors(nid)):
   509→                if HAS_NX:
   510→                    pdata = self._g.nodes[pred]
   511→                else:
   512→                    pdata = self._g.get_node_data(pred)
   513→                ptype = pdata.get("node_type", "")
   514→                if ptype in ("target", "subdomain", "asset") and pred not in seen:
   515→                    seen.add(pred)
   516→                    vulnerable.append({"node_id": pred, **pdata, "critical_finding": data})
   517→        return vulnerable

   519→    def correlate(self) -> List[Dict[str, Any]]:
   520→        """Cross-reference findings to suggest related vulnerabilities.
   521→
   522→        Returns a list of correlation suggestions, each containing
   523→        the two related node ids, the relationship type, and a
   524→        confidence score.
   525→        """
   526→        suggestions: List[Dict[str, Any]] = []
   527→        # Group findings by technology
   528→        tech_findings: Dict[str, List[str]] = defaultdict(list)
   529→        for nid in self._node_type_index.get("finding", set()):
   530→            if HAS_NX:
   531→                data = self._g.nodes[nid]
   532→            else:
   533→                data = self._g.get_node_data(nid)
   534→            for pred in (self._g.predecessors(nid) if HAS_NX else self._g.predecessors(nid)):
   535→                if HAS_NX:
   536→                    pdata = self._g.nodes[pred]
   537→                else:
   538→                    pdata = self._g.get_node_data(pred)
   539→                if pdata.get("node_type") == "technology":
   540→                    tech_findings[pred].append(nid)
   541→
   542→        for tech_id, finding_ids in tech_findings.items():
   543→            if len(finding_ids) < 2:
   544→                continue
   545→            for i in range(len(finding_ids)):
   546→                for j in range(i + 1, len(finding_ids)):
   547→                    f1 = finding_ids[i]
   548→                    f2 = finding_ids[j]
   549→                    if HAS_NX:
   550→                        d1 = self._g.nodes[f1]
   551→                        d2 = self._g.nodes[f2]
   552→                    else:
   553→                        d1 = self._g.get_node_data(f1)
   554→                        d2 = self._g.get_node_data(f2)
   555→                    # If both share a technology, suggest they may be related
   556→                    conf = 0.6
   557→                    if d1.get("severity") == d2.get("severity"):
   558→                        conf += 0.2
   559→                    if d1.get("finding_type") == d2.get("finding_type"):
   560→                        conf += 0.2
   561→                    suggestions.append({
   562→                        "source": f1,
   563→                        "target": f2,
   564→                        "relationship": "shared_technology",
   565→                        "technology": tech_id,
   566→                        "confidence": min(conf, 1.0),
   567→                        "source_severity": d1.get("severity"),
   568→                        "target_severity": d2.get("severity"),
   569→                    })
   570→
   571→        # Also correlate by port: findings on the same port may be related
   572→        port_findings: Dict[str, List[str]] = defaultdict(list)
   573→        for nid in self._node_type_index.get("finding", set()):
   574→            for pred in (self._g.predecessors(nid) if HAS_NX else self._g.predecessors(nid)):
   575→                if HAS_NX:
   576→                    pdata = self._g.nodes[pred]
   577→                else:
   578→                    pdata = self._g.get_node_data(pred)
   579→                if pdata.get("node_type") == "port":
   580→                    port_findings[pred].append(nid)
   581→
   582→        for port_id, fids in port_findings.items():
   583→            if len(fids) < 2:
   584→                continue
   585→            for i in range(len(fids)):
   586→                for j in range(i + 1, len(fids)):
   587→                    f1, f2 = fids[i], fids[j]
   588→                    suggestions.append({
   589→                        "source": f1,
   590→                        "target": f2,
   591→                        "relationship": "shared_port",
   592→                        "port": port_id,
   593→                        "confidence": 0.7,
   594→                    })
   595→
   596→        suggestions.sort(key=lambda s: s["confidence"], reverse=True)
   597→        return suggestions

   599→    def to_cypher(self) -> str:
   600→        """Export the graph as Cypher-like CREATE statements."""
   601→        lines: List[str] = []
   602→        # Nodes
   603→        for nid in (self._g.nodes() if HAS_NX else self._g.nodes()):
   604→            if HAS_NX:
   605→                data = self._g.nodes[nid]
   606→            else:
   607→                data = self._g.get_node_data(nid)
   608→            ntype = data.pop("node_type", "Unknown")
   609→            label = data.pop("label", nid)
   610→            props = json.dumps(data, default=str)
   611→            safe_id = nid.replace(":", "__").replace("-", "_")
   612→            lines.append(f'CREATE ({safe_id}:{ntype} {{label: "{label}", {props[1:-1]}}})')
   613→            # Restore for edge generation below
   614→            data["node_type"] = ntype
   615→            data["label"] = label
   616→
   617→        # Edges
   618→        for u, v, attr in (self._g.edges(data=True) if HAS_NX else self._g.edges(data=True)):
   619→            etype = attr.get("edge_type", "RELATED_TO")
   620→            props = {k: v for k, v in attr.items() if k != "edge_type"}
   621→            props_str = json.dumps(props, default=str)
   622→            safe_u = u.replace(":", "__").replace("-", "_")
   623→            safe_v = v.replace(":", "__").replace("-", "_")
   624→            if props_str == "{}":
   625→                lines.append(f'CREATE ({safe_u})-[:{etype}]->({safe_v})')
   626→            else:
   627→                lines.append(f'CREATE ({safe_u})-[:{etype} {{{props_str[1:-1]}}}]->({safe_v})')
   628→
   629→        return "\n".join(lines)

   631→    def stats(self) -> Dict[str, Any]:
   632→        """Return a summary dict with node/edge counts broken down by type."""
   633→        node_counts: Dict[str, int] = {}
   634→        for ntype, nids in self._node_type_index.items():
   635→            node_counts[ntype] = len(nids)
   636→
   637→        edge_counts: Dict[str, int] = {}
   638→        for etype, pairs in self._edge_type_index.items():
   639→            edge_counts[etype] = len(pairs)
   640→
   641→        total_nodes = len(self._g.nodes()) if HAS_NX else len(self._g.nodes())
   642→        total_edges = len(self._g.edges()) if HAS_NX else len(self._g.edges())
   643→
   644→        # Severity distribution
   645→        sev_dist: Dict[str, int] = defaultdict(int)
   646→        for nid in self._node_type_index.get("finding", set()).union(self._node_type_index.get("cve", set())):
   647→            if HAS_NX:
   648→                data = self._g.nodes[nid]
   649→            else:
   650→                data = self._g.get_node_data(nid)
   651→            sev_dist[self._safe_sev(data.get("severity"))] += 1
   652→
   653→        return {
   654→            "total_nodes": total_nodes,
   655→            "total_edges": total_edges,
   656→            "nodes_by_type": node_counts,
   657→            "edges_by_type": edge_counts,
   658→            "severity_distribution": dict(sev_dist),
   659→            "backing_engine": "networkx" if HAS_NX else "fallback_dict",
   660→        }

   700→    def load(self, path: Optional[str] = None) -> None:
   701→        """Load a previously saved graph from JSON.
   702→
   703→        Parameters
   704→        ----------

   707→            ``~/.reconpro/memory/graph.json``.
   708→        """

   710→            path = str(DEFAULT_GRAPH_DIR / "graph.json")

   712→        with open(path, "r") as fh:

   714→

   721→        self._node_type_index = defaultdict(set)
   722→        for k, v in payload.get("node_type_index", {}).items():

   725→        self._edge_type_index = defaultdict(list)
   726→        for k, v in payload.get("edge_type_index", {}).items():

   733→    def _resolve_node(self, label: str) -> Optional[str]:
   734→        """Try to resolve a human-readable label to a full node id."""
   735→        # Exact node id
   736→        if self._g.has_node(label):
   737→            return label
   738→        # Try as target
   739→        candidate = self._node_id("target", label)
   740→        if self._g.has_node(candidate):
   741→            return candidate
   742→        # Try as subdomain
   743→        candidate = self._node_id("subdomain", label)
   744→        if self._g.has_node(candidate):
   745→            return candidate
   746→        # Try as asset
   747→        candidate = self._node_id("asset", label)
   748→        if self._g.has_node(candidate):
   749→            return candidate
   750→        # Fuzzy: search all node labels
   751→        for nid in self._g.nodes():
   752→            if HAS_NX:
   753→                data = self._g.nodes[nid]
   754→            else:
   755→                data = self._g.get_node_data(nid)
   756→            if data.get("label") == label or nid.endswith(f":{label}"):
   757→                return nid
   758→        return None

   760→    def _bfs_collect(self, start: str, max_depth: int = 5) -> Set[str]:
   761→        """BFS from *start* up to *max_depth* hops, return visited nodes."""
   762→        visited: Set[str] = set()
   763→        queue: deque = deque()
   764→        queue.append((start, 0))
   765→        visited.add(start)
   766→        while queue:
   767→            node, depth = queue.popleft()
   768→            if depth >= max_depth:
   769→                continue
   770→            for neighbour in (self._g.successors(node) if HAS_NX else self._g.successors(node)):
   771→                if neighbour not in visited:
   772→                    visited.add(neighbour)
   773→                    queue.append((neighbour, depth + 1))
   774→        return visited
