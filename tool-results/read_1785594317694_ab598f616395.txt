   117→function timeAgo(dateStr: string): string {
   118→  const diff = Date.now() - new Date(dateStr).getTime();
   119→  const mins = Math.floor(diff / 60000);
   120→  if (mins < 1) return 'just now';
   121→  if (mins < 60) return `${mins}m ago`;
   122→  const hrs = Math.floor(mins / 60);
   123→  if (hrs < 24) return `${hrs}h ago`;
   124→  const days = Math.floor(hrs / 24);
   125→  if (days < 30) return `${days}d ago`;
   126→  const months = Math.floor(days / 30);
   127→  return `${months}mo ago`;
   128→}
   129→
   130→function expiresIn(expiresAt: string): string {
   131→  const diff = new Date(expiresAt).getTime() - Date.now();
   132→  if (diff <= 0) return 'Expired';
   133→  const days = Math.floor(diff / 86400000);
   134→  const hrs = Math.floor((diff % 86400000) / 3600000);
   135→  if (days > 0) return `${days}d ${hrs}h`;
   136→  return `${hrs}h`;
   137→}
   138→
   139→function isExpired(expiresAt: string): boolean {
   140→  return new Date(expiresAt).getTime() < Date.now();
   141→}
   142→
   143→function parseSafeJson<T>(val: unknown, fallback: T): T {
   144→  if (typeof val === 'string') {
   145→    try { return JSON.parse(val) as T; } catch { return fallback; }
   146→  }
   147→  if (typeof val === 'object' && val !== null) return val as T;
   148→  return fallback;
   149→}

   153→function CopyButton({ text, label }: { text: string; label?: string }) {
   154→  const [copied, setCopied] = useState(false);
   155→  const handleCopy = async () => {
   156→    await navigator.clipboard.writeText(text);
   157→    setCopied(true);
   158→    setTimeout(() => setCopied(false), 2000);
   159→  };
   160→  return (
   161→    <button
   162→      onClick={handleCopy}
   163→      className="inline-flex items-center gap-1.5 px-2.5 py-1 rounded-md bg-gray-800 hover:bg-gray-700 border border-gray-700 text-xs text-gray-300 hover:text-white transition-colors"
   164→      title={label ?? 'Copy'}
   165→    >
   166→      {copied ? <Check className="w-3.5 h-3.5 text-[#00ff88]" /> : <Copy className="w-3.5 h-3.5" />}
   167→      {label && <span>{copied ? 'Copied!' : label}</span>}
   168→    </button>
   169→  );
   170→}

   192→  // Verify

   195→  const [verifyResult, setVerifyResult] = useState<VerificationResult | null>(null);
   196→  const [verifyError, setVerifyError] = useState('');
   197→
   198→  // Embed preview
   199→  const [embedId, setEmbedId] = useState('');
   200→  const [embedHtml, setEmbedHtml] = useState('');
   201→  const [embedLoading, setEmbedLoading] = useState(false);
   202→

   205→    open: false, stampId: '', domain: '',

   208→  const [revoking, setRevoking] = useState(false);
   209→
   210→  // Domain filter for listing
   211→  const [filterDomain, setFilterDomain] = useState('');
   212→

   214→  const loadStamps = useCallback(async () => {
   215→    setLoading(true);
   216→    try {
   217→      const params = new URLSearchParams();
   218→      if (filterDomain) params.set('domain', filterDomain);
   219→      else params.set('domain', '*');
   220→      const res = await fetch(`/api/genesis?${params.toString()}`);
   221→      if (!res.ok) throw new Error('Failed to fetch');
   222→      const data = await res.json();
   223→      const raw: Stamp[] = data.stamps ?? [];

   225→      const parsed = raw.map((s) => ({
   226→        ...s,
   227→        frameworks: parseSafeJson<string[]>(s.frameworks, []),
   228→        complianceScores: parseSafeJson<Record<string, number>>(s.complianceScores, {}),
   229→        findingsSummary: parseSafeJson<Record<string, number>>(s.findingsSummary, {}),
   230→      }));
   231→      setStamps(parsed);
   232→      // Compute stats from the list
   233→      const now = Date.now();
   234→      const monthAgo = now - 30 * 86400000;
   235→      setStats({
   236→        totalIssued: parsed.length,
   237→        activeStamps: parsed.filter((s) => s.status === 'active' && !isExpired(s.expiresAt)).length,
   238→        expiredThisMonth: parsed.filter(
   239→          (s) => s.status === 'expired' || (s.expiresAt && new Date(s.expiresAt).getTime() < now && new Date(s.expiresAt).getTime() > monthAgo)
   240→        ).length,
   241→        totalVerifications: parsed.reduce((acc, s) => acc + (s.verifiedCount ?? 0), 0),
   242→        totalEmbedViews: parsed.reduce((acc, s) => acc + (s.embedViews ?? 0), 0),
   243→      });
   244→    } catch {
   245→      /* silent */
   246→    } finally {
   247→      setLoading(false);
   248→    }
   249→  }, [filterDomain]);
   250→
   251→  useEffect(() => {
   252→    void loadStamps();
   253→  }, [loadStamps]);
   254→
   255→  // ── Issue Stamp ─────────────────────────────────────────────────
   256→  const handleIssue = async () => {

   258→    setIssuing(true);
   259→    setIssueError('');

   261→    try {
   262→      const res = await fetch('/api/genesis', {
   263→        method: 'POST',
   264→        headers: { 'Content-Type': 'application/json' },
   265→        body: JSON.stringify({
   266→          domain: issueDomain.trim(),
   267→          tier: issueTier,
   268→          scanId: issueScanId.trim() || undefined,
   269→        }),
   270→      });
   271→      const data = await res.json();
   272→      if (!res.ok) throw new Error(data.error ?? 'Failed to issue stamp');
   273→      setIssuedStamp({
   274→        stamp: {
   275→          ...data.stamp,
   276→          frameworks: parseSafeJson<string[]>(data.stamp.frameworks, []),
   277→          complianceScores: parseSafeJson<Record<string, number>>(data.stamp.complianceScores, {}),
   278→          findingsSummary: parseSafeJson<Record<string, number>>(data.stamp.findingsSummary, {}),
   279→        },
   280→        embedCode: data.embedCode,
   281→      });
   282→      setIssueDomain('');
   283→      setIssueScanId('');
   284→      void loadStamps();
   285→    } catch (err) {
   286→      setIssueError(err instanceof Error ? err.message : 'Unknown error');
   287→    } finally {
   288→      setIssuing(false);
   289→    }
   290→  };

   302→      setVerifyResult(data);
   303→    } catch (err) {
   304→      setVerifyError(err instanceof Error ? err.message : 'Unknown error');
   305→    } finally {
   306→      setVerifying(false);
   307→    }
   308→  };

   316→        method: 'POST',

   326→      /* silent */

   328→      setRevoking(false);

   365→
