   274→def _extract_domain(target: str) -> str:
   275→    """Extract the root domain from a target."""
   276→    host = target.replace("https://", "").replace("http://", "").split("/")[0]
   277→    host = host.split(":")[0]
   278→    parts = host.rsplit(".", 1)
   279→    if len(parts) >= 2:
   280→        # Check for common TLDs with two-part SLDs
   281→        tld = parts[-1]
   282→        if tld in ("co.uk", "com.au", "org.uk", "ac.uk", "gov.uk", "co.jp"):
   283→            sub_parts = host.rsplit(".", 2)
   284→            return ".".join(sub_parts[-2:]) if len(sub_parts) >= 2 else host
   285→    return host
   286→

   288→def _extract_email_domains(target: str) -> List[str]:
   289→    """Generate likely email domains for the target."""
   290→    domain = _extract_domain(target)
   291→    domains = [domain]
   292→    # Common email domain patterns
   293→    domains.append(f"mail.{domain}")
   294→    # Strip www
   295→    if domain.startswith("www."):
   296→        domains.append(domain[4:])
   297→    return list(set(domains))

   300→def _safe_probe(url: str, timeout: int = 5, verify_tls: bool = True) -> Dict[str, Any]:
   301→    """Safe HTTP probe with error handling."""
   302→    try:
   303→        return http_probe(
   304→            url, timeout=timeout, verify_tls=verify_tls,
   305→            limiter=default_limiter,
   306→        )
   307→    except Exception:
   308→        return {"ok": False, "status": 0, "reason": "error", "headers": {}, "body": ""}

   311→def _parse_json(body: str) -> Optional[Any]:
   312→    try:
   313→        return json.loads(body)
   314→    except Exception:
   315→        return None

   318→def _count_pattern(body: str, pattern: re.Pattern) -> int:
   319→    """Count non-overlapping pattern matches."""
   320→    return len(pattern.findall(body))

   323→def _extract_emails_from_domain(domain: str) -> List[str]:
   324→    """Extract likely email addresses from a domain name."""
   325→    prefixes = ["admin", "info", "support", "contact", "root", "test",
   326→                "webmaster", "noreply", "security", "dev", "staging"]
   327→    return [f"{p}@{domain}" for p in prefixes]

   329→
   330→def _decode_jwt(token: str) -> Optional[Dict[str, Any]]:
   331→    """Decode JWT header and payload without verification."""
   332→    parts = token.split(".")
   333→    if len(parts) != 3:
   334→        return None
   335→    result = {}
   336→    for i, label in enumerate(("header", "payload")):
   337→        try:
   338→            padded = parts[i] + "=" * (-len(parts[i]) % 4)
   339→            decoded = base64.urlsafe_b64decode(padded)
   340→            parsed = json.loads(decoded)
   341→            result[label] = parsed
   342→        except Exception:
   343→            result[label] = None
   344→    return result

   348→# Cat 1: Paste Site Monitoring
   349→# ═══════════════════════════════════════════════════════════════════════════
   350→
   351→def _check_paste_sites(
   352→    target: str, base_url: str, timeout: int, verify_tls: bool,
   353→) -> List[Finding]:
   354→    findings: List[Finding] = []
   355→    host = _extract_domain(target)
   356→    queries = [host]
   357→
   358→    for site_id, site in PASTE_SITES.items():
   359→        for query in queries:
   360→            url = site["search_url"].format(query=urllib.parse.quote(query))
   361→            resp = _safe_probe(url, timeout=min(timeout, 5), verify_tls=verify_tls)
   362→
   363→            if resp.get("ok") and resp.get("status") == 200:
   364→                body = resp.get("body", "")
   365→
   366→                # Check for result indicators
   367→                result_indicators = 0
   368→                # Common patterns indicating search results
   369→                if "result" in body.lower():
   370→                    result_indicators += 1
   371→                if "paste" in body.lower():
   372→                    result_indicators += 1
   373→                if host.lower() in body.lower():
   374→                    result_indicators += 3  # Domain mention is strongest signal
   375→                if re.search(rf'{re.escape(host)}[\s/<]', body, re.IGNORECASE):
   376→                    result_indicators += 2
   377→
   378→                # Count occurrences
   379→                domain_count = body.lower().count(host.lower())
   380→
   381→                if result_indicators >= 3:
   382→                    sev = site.get("severity", "high")
   383→                    pts = 18 if sev == "critical" else 12
   384→                    findings.append(_make(
   385→                        f"Paste Site Exposure — {site['name']}: domain found",
   386→                        sev, "paste_exposure",
   387→                        f"Target domain '{host}' found on {site['name']}. "
   388→                        f"Indicators: {result_indicators}, domain mentions: {domain_count}. "
   389→                        f"This may indicate leaked credentials, configuration data, or source code.",
   390→                        f"Source: {site['name']}; Mentions: {domain_count}; URL: {url}",
   391→                        host, pts,
   392→                        "Review the paste content. Request removal if credentials/secrets are exposed.",
   393→                        0.7,
   394→                    ))
   395→
   396→    return findings

   399→# ═══════════════════════════════════════════════════════════════════════════
   400→# Cat 2: Credential Pattern Detection
   401→# ═══════════════════════════════════════════════════════════════════════════
   402→
   403→def _check_credential_patterns(
   404→    target: str, base_url: str, timeout: int, verify_tls: bool,
   405→) -> List[Finding]:
   406→    """Check the target's own responses for leaked credentials."""
   407→    findings: List[Finding] = []
   408→    host = _extract_domain(target)
   409→
   410→    # Fetch the target's own pages for credential leakage
   411→    paths_to_check = ["/", "/robots.txt", "/sitemap.xml", "/config.js",
   412→                       "/.env", "/package.json", "/webpack.config.js"]
   413→    for path in paths_to_check:
   414→        url = base_url.rstrip("/") + path
   415→        resp = _safe_probe(url, timeout=min(timeout, 5), verify_tls=verify_tls)
   416→        if not resp.get("ok") or resp.get("status") in (404, 403):
   417→            continue
   418→
   419→        body = resp.get("body", "")
   420→        if len(body) < 50:
   421→            continue
   422→
   423→        for cred_id, cred_info in CREDENTIAL_PATTERNS.items():
   424→            matches = cred_info["pattern"].findall(body)
   425→            if matches:
   426→                count = len(matches)
   427→                sev = cred_info.get("severity", "high")
   428→                pts = 20 if sev == "critical" else 12
   429→
   430→                # Decode JWT tokens for extra evidence
   431→                extra = ""
   432→                if cred_id == "jwt_token" and matches:
   433→                    jwt_data = _decode_jwt(matches[0])
   434→                    if jwt_data and jwt_data.get("header"):
   435→                        extra = f" JWT header: {json.dumps(jwt_data['header'])}"
   436→
   437→                evidence_str = (
   438→                    f"Path: {path}; Pattern: {cred_id}; "
   439→                    f"Matches: {count}; Sample: {str(matches[0])[:80]}{extra}"
   440→                )
   441→                findings.append(_make(
   442→                    f"Credential Leak — {cred_info['description']} on {path}",
   443→                    sev, "credential_leak",
   444→                    f"{cred_info['description']} detected {count} time(s) at {path}. "
   445→                    f"Immediate credential rotation recommended.{extra}",
   446→                    evidence_str, host, pts,
   447→                    "Rotate any exposed credentials immediately. Remove sensitive data from source code.",
   448→                    0.9 if sev == "critical" else 0.6,
   449→                ))
   450→
   451→    return findings

   463→
   464→    # Resolve IPs for the target
   465→    try:
   466→        ips = socket.getaddrinfo(host, None)
   467→        ip_list = list(set(addr[4][0] for addr in ips))
   468→    except Exception:
   469→        ip_list = []
   470→
   471→    # Check threat intel sources for each IP
   472→    for source_id, source in THREAT_INTEL_SOURCES.items():
   473→        if source.get("type") != "json_api":
   474→            continue
   475→
   476→        for ip in ip_list[:3]:
   477→            url = source["search_url"].format(query=ip)
   478→            resp = _safe_probe(url, timeout=min(timeout, 5), verify_tls=verify_tls)
   479→
   480→            if resp.get("ok") and resp.get("status") == 200:
   481→                body = resp.get("body", "")
   482→                data = _parse_json(body)

   484→                    # Check for malicious indicators
   485→                    if data.get("query_status") == "ok":
   486→                        count = len(data.get("data", []))
   487→                        if count > 0:
   488→                            findings.append(_make(
   489→                                f"Infrastructure Exposure — {ip} on {source['name']}",
   490→                                "critical", "infrastructure_exposure",
   491→                                f"IP {ip} has {count} malicious indicators on {source['name']}. "
   492→                                f"Target infrastructure may be compromised or hosting malware.",
   493→                                f"IP: {ip}; Source: {source['name']}; Indicators: {count}",
   494→                                host, 20,
   495→                                "Investigate server for compromise. Check for malware, unauthorized access.",
   496→                                0.9,
   497→                            ))
   498→                    elif data.get("response_code") == 1:
   499→                        # VirusTotal-style positive detection
   500→                        positives = data.get("positives", 0)
   501→                        total = data.get("total", 0)
   502→                        if positives > 0:
   503→                            findings.append(_make(
   504→                                f"Infrastructure Exposure — {ip}: {positives}/{total} vendors flag",
   505→                                "critical", "infrastructure_exposure",
   506→                                f"IP {ip} flagged by {positives}/{total} security vendors on "
   507→                                f"{source['name']}. Infrastructure may be malicious.",
   508→                                f"IP: {ip}; Positives: {positives}/{total}; Source: {source['name']}",
   509→                                host, 20,
   510→                                "Immediate investigation required. Check all services on this IP.",
   511→                                0.9,
   512→                            ))
   513→
   514→    return findings

   520→
   521→def _check_ct_logs(
   522→    target: str, base_url: str, timeout: int, verify_tls: bool,

   526→
   527→    resp = _safe_probe(

   532→
   533→    if resp.get("ok") and resp.get("status") == 200:
   534→        data = _parse_json(resp.get("body", ""))

   545→                        subdomains.add(n)
   546→                issuer = cert.get("issuer_name", "")

   548→                    issuers.add(issuer)
   549→
   550→            if cert_count > 50:

   571→                    0.2,
   572→                ))
   573→
   574→            # Check for suspicious issuer patterns (self-signed, unknown CAs)
