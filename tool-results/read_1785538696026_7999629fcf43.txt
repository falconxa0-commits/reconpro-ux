   178→   178→// --- Circular Progress Ring ---
   179→   179→function CircularProgressRing({ score, size = 140, strokeWidth = 8 }: { score: number; size?: number; strokeWidth?: number }) {
   180→   180→  const radius = (size - strokeWidth * 2) / 2;
   181→   181→  const circumference = 2 * Math.PI * radius;
   182→   182→  const offset = circumference - (score / 100) * circumference;
   183→   183→
   184→   184→  const color = score >= 80 ? '#f85149' : score >= 60 ? '#f97316' : score >= 40 ? '#eab308' : '#00ff88';
   185→   185→  const glowColor = color + '50';
   186→   186→
   187→   187→  return (
   188→   188→    <div className="relative" style={{ width: size, height: size }}>
   189→   189→      <svg viewBox={`0 0 ${size} ${size}`} className="w-full h-full -rotate-90">
   190→   190→        <circle
   191→   191→          cx={size / 2}
   192→   192→          cy={size / 2}
   193→   193→          r={radius}
   194→   194→          fill="none"
   195→   195→          stroke="rgba(255,255,255,0.06)"
   196→   196→          strokeWidth={strokeWidth}
   197→   197→        />
   198→   198→        <motion.circle
   199→   199→          cx={size / 2}
   200→   200→          cy={size / 2}
   201→   201→          r={radius}
   202→   202→          fill="none"
   203→   203→          stroke={color}
   204→   204→          strokeWidth={strokeWidth}
   205→   205→          strokeLinecap="round"
   206→   206→          strokeDasharray={circumference}
   207→   207→          initial={{ strokeDashoffset: circumference }}
   208→   208→          animate={{ strokeDashoffset: offset }}
   209→   209→          transition={{ duration: 2, ease: 'easeOut', delay: 0.3 }}
   210→   210→          style={{ filter: `drop-shadow(0 0 12px ${glowColor})` }}
   211→   211→        />
   212→   212→      </svg>
   213→   213→      <div className="absolute inset-0 flex flex-col items-center justify-center">
   214→   214→        <motion.span
   215→   215→          className="font-bold font-mono text-3xl"
   216→   216→          style={{ color }}
   217→   217→          initial={{ opacity: 0, scale: 0.5 }}
   218→   218→          animate={{ opacity: 1, scale: 1 }}
   219→   219→          transition={{ delay: 0.8, type: 'spring', stiffness: 200 }}
   220→   220→        >
   221→   221→          {score}
   222→   222→        </motion.span>
   223→   223→        <span className="text-[10px] font-semibold uppercase tracking-widest text-[#8b949e]">
   224→   224→          / 100
   225→   225→        </span>
   226→   226→      </div>
   227→   227→    </div>
   228→   228→  );
   229→   229→}
