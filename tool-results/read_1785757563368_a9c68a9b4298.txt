   358→   710→    def on_key(self, event: Key) -> None:
   359→   711→        """Handle key events — boot skip + history navigation."""
   360→   712→        # Skip boot on ANY key press during boot sequence
   361→   713→        if not self._boot_done and not self._boot_skipped:
   362→   714→            self._boot_skipped = True
   363→   715→            try:
   364→   716→                boot = self.query_one("#boot-screen", BootScreen)
   365→   717→                if boot._timer:
   366→   718→                    boot._timer.stop()
   367→   719→            except NoMatches:
   368→   720→                pass
   369→   721→            self._finish_boot()
   370→   722→            return
   371→   723→
   372→   724→        # Command history navigation
   373→   725→        if event.key == "up" and self._history_index > 0:
   374→   726→            try:
   375→   727→                inp = self.query_one("#command-input", Input)
   376→   728→                if inp.has_focus:
   377→   729→                    self._history_index -= 1
   378→   730→                    inp.value = self._command_history[self._history_index]
   379→   731→            except NoMatches:
   380→   732→                pass
   381→   733→        elif event.key == "down":
   382→   734→            try:
   383→   735→                inp = self.query_one("#command-input", Input)
   384→   736→                if inp.has_focus:
   385→   737→                    if self._history_index < len(self._command_history) - 1:
   386→   738→                        self._history_index += 1
   387→   739→                        inp.value = self._command_history[self._history_index]
   388→   740→                    else:
   389→   741→                        self._history_index = len(self._command_history)
   390→   742→                        inp.value = ""
   391→   743→            except NoMatches:
   392→   744→                pass

   866→  1218→    @work(exclusive=False, thread=True)
   867→  1219→    def _run_scan_worker(
   868→  1220→        self,
   869→  1221→        target: str,
   870→  1222→        modules: Optional[List[str]] = None,
   871→  1223→        is_local: bool = False,
   872→  1224→    ) -> None:
   873→  1225→        """Run a scan in a background thread."""
   874→  1226→        self.call_from_thread(self.is_scanning.set, True)
   875→  1227→
   876→  1228→        # Set modules to scanning
   877→  1229→        if modules:
   878→  1230→            for m in modules:
   879→  1231→                if m in self._module_cells:
   880→  1232→                    self.call_from_thread(self._module_cells[m].set_status, "scanning")
   881→  1233→        else:
   882→  1234→            all_mods = (
   883→  1235→                list(MODULE_NAMES.keys())
   884→  1236→                if is_local
   885→  1237→                else [k for k in MODULE_NAMES if k not in ("host", "dev", "doctor")]
   886→  1238→            )
   887→  1239→            for m in all_mods:
   888→  1240→                if m in self._module_cells:
   889→  1241→                    self.call_from_thread(self._module_cells[m].set_status, "scanning")
   890→  1242→
   891→  1243→        try:
   892→  1244→            from .scanner import scan, audit_scan
   893→  1245→            from .history import save_scan
   894→  1246→
   895→  1247→            self.call_from_thread(
   896→  1248→                self._chat, f"[{DIM_CYAN}]  ⏳ Scan in progress...[/]"
   897→  1249→            )
   898→  1250→
   899→  1251→            if is_local:
   900→  1252→                result = audit_scan(target=target, modules=modules)
   901→  1253→            else:
   902→  1254→                result = scan(target=target, modules=modules)
   903→  1255→
   904→  1256→            # Process findings
   905→  1257→            data = result.to_dict()
   906→  1258→            self.call_from_thread(self._on_scan_complete, data, modules, is_local)
   907→  1259→
   908→  1260→            # Save to history
   909→  1261→            try:
   910→  1262→                save_scan(data, label="nexus")
   911→  1263→            except Exception:
   912→  1264→                pass
   913→  1265→
   914→  1266→        except Exception as e:
   915→  1267→            self.call_from_thread(self._chat, f"[{RED}]  ✗ Scan error: {e}[/]")
   916→  1268→            if modules:
   917→  1269→                for m in modules:
   918→  1270→                    if m in self._module_cells:
   919→  1271→                        self.call_from_thread(self._module_cells[m].set_status, "error")
   920→  1272→            else:
   921→  1273→                self.call_from_thread(self._set_all_modules_status, "error")
   922→  1274→
   923→  1275→        finally:
   924→  1276→            self.call_from_thread(self.is_scanning.set, False)
   925→  1277→
   926→  1278→    @work(exclusive=False, thread=True)
   927→  1279→    def _run_blitz_worker(self, targets: List[str]) -> None:
   928→  1280→        """Run parallel blitz scan."""
   929→  1281→        self.call_from_thread(self.is_scanning.set, True)
   930→  1282→        self.call_from_thread(self._set_all_modules_status, "scanning")
   931→  1283→
   932→  1284→        try:
   933→  1285→            from .parallel import blitz_scan
   934→  1286→
   935→  1287→            self.call_from_thread(
   936→  1288→                self._chat, f"[{DIM_CYAN}]  ⏳ Blitz scanning {len(targets)} targets...[/]"
   937→  1289→            )
   938→  1290→
   939→  1291→            result = blitz_scan(targets, save=True)
   940→  1292→
   941→  1293→            self.call_from_thread(

  1277→  1629→    def on_click(self, event: Click) -> None:
  1278→  1630→        """Open finding detail modal for the clicked finding.
  1279→  1631→
  1280→  1632→        Uses the vertical click position to determine which finding index
  1281→  1633→        the user clicked on, mapping it to _findings_list.
  1282→  1634→        """
  1283→  1635→        try:
  1284→  1636→            if (
  1285→  1637→                isinstance(event.widget, RichLog)
  1286→  1638→                and event.widget.id == "findings-feed"
  1287→  1639→                and self._findings_list
  1288→  1640→            ):
  1289→  1641→                # Get the line offset to determine which finding was clicked.
  1290→  1642→                # Each finding entry is 1 line in the RichLog.
  1291→  1643→                # We use the y coordinate relative to the widget to estimate the index.
  1292→  1644→                region = event.widget.region
  1293→  1645→                if region is not None:
  1294→  1646→                    # Calculate which line was clicked relative to the scroll position
  1295→  1647→                    scroll_offset = event.widget.scroll_y
  1296→  1648→                    click_y = event.y - region.y
  1297→  1649→                    # Each finding takes 1 line; estimate index from click position
  1298→  1650→                    # Account for scroll position
  1299→  1651→                    line_height = 1
  1300→  1652→                    if hasattr(event.widget, "_line_height"):
  1301→  1653→                        line_height = event.widget._line_height
  1302→  1654→                    clicked_index = int((click_y + scroll_offset * line_height) / line_height)
  1303→  1655→                    # Clamp to valid range
  1304→  1656→                    clicked_index = max(0, min(clicked_index, len(self._findings_list) - 1))
  1305→  1657→                    finding = self._findings_list[clicked_index]
  1306→  1658→                else:
  1307→  1659→                    finding = self._findings_list[-1]
  1308→  1660→                self.push_screen(FindingDetailModal(finding))
  1309→  1661→        except (NoMatches, Exception):
