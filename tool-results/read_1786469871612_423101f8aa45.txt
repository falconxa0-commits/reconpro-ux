   988→    def _check_memory_usage(self) -> int:
   989→        """Return approximate current memory usage in bytes.
   990→
   991→        Uses resource module if available (Unix), otherwise falls back
   992→        to a basic tracking mechanism.
   993→        """
   994→        try:
   995→            import resource as _resource  # type: ignore[import-untyped]
   996→            return _resource.getrusage(_resource.RUSAGE_SELF).ru_maxrss * 1024
   997→        except (ImportError, AttributeError):
   998→            # Fallback: track based on sys.getsizeof of known objects
   999→            return 0

  1001→    def _enforce_cpu_limit(self) -> None:
  1002→        """Set CPU time limit on platforms that support it (Unix)."""
  1003→        try:
  1004→            import resource as _resource  # type: ignore[import-untyped]
  1005→            _resource.setrlimit(
  1006→                _resource.RLIMIT_CPU,
  1007→                (int(self.cpu_time_seconds), int(self.cpu_time_seconds * 1.5)),
  1008→            )
  1009→        except (ImportError, AttributeError, OSError, ValueError):
  1010→            pass  # Not available on this platform
  1011→
  1012→    def _restore_cpu_limit(self) -> None:
  1013→        """Restore CPU time limit after sandboxed execution."""
  1014→        try:
  1015→            import resource as _resource  # type: ignore[import-untyped]
  1016→            # Set to a very high value (effectively unlimited)
  1017→            _resource.setrlimit(_resource.RLIMIT_CPU, (0x7FFFFFFF, 0x7FFFFFFF))
  1018→        except (ImportError, AttributeError, OSError, ValueError):
  1019→            pass

  1021→    def _validate_output(self, result: Any) -> List[Dict[str, Any]]:
  1022→        """Validate plugin output is safe and within limits.
  1023→
  1024→        Args:
  1025→            result: The return value from the plugin.
  1026→
  1027→        Returns:
  1028→            List of finding dicts, or raises on invalid output.
  1029→        """
  1030→        if not isinstance(result, list):
  1031→            raise SandboxViolation(
  1032→                "output_validation",
  1033→                f"Plugin must return a list, got {type(result).__name__}",
  1034→            )
  1035→
  1036→        output_str = json.dumps(result, default=str)
  1037→        if len(output_str) > self.max_output_size:
  1038→            raise ResourceLimitExceeded(
  1039→                "output_size",
  1040→                self.max_output_size,
  1041→                len(output_str),
  1042→            )
  1043→
  1044→        validated: List[Dict[str, Any]] = []
  1045→        required_keys = {"title", "severity", "category"}
  1046→        for i, item in enumerate(result):
  1047→            if not isinstance(item, dict):
  1048→                raise SandboxViolation(
  1049→                    f"output_item_{i}",
  1050→                    f"Each finding must be a dict, got {type(item).__name__}",
  1051→                )
  1052→            if not required_keys.issubset(item.keys()):
  1053→                missing = required_keys - item.keys()
  1054→                raise SandboxViolation(
  1055→                    f"output_item_{i}",
  1056→                    f"Finding missing required keys: {missing}",
  1057→                )
  1058→            validated.append(item)
  1059→
  1060→        return validated

  1062→    def execute(
  1063→        self,
  1064→        plugin_func: Callable,
  1065→        plugin_name: str = "unknown",
  1066→        **kwargs: Any,
  1067→    ) -> Dict[str, Any]:
  1068→        """Execute a plugin function inside the sandbox.
  1069→
  1070→        The plugin function is called with **kwargs as arguments.
  1071→        Its execution is wrapped with:
  1072→        - Restricted builtins
  1073→        - CPU time limits (where supported)
  1074→        - Memory usage checks
  1075→        - Output validation
  1076→        - Full audit logging
  1077→
  1078→        Args:
  1079→            plugin_func: The plugin's callable (typically its ``run`` function).
  1080→            plugin_name: Human-readable plugin name for logging.
  1081→            **kwargs: Arguments forwarded to the plugin function.
  1082→
  1083→        Returns:
  1084→            Dict with keys:
  1085→            - success: bool
  1086→            - findings: list of validated finding dicts (on success)
  1087→            - error: str (on failure)
  1088→            - execution_time: float (seconds)
  1089→            - memory_used: int (bytes, 0 if unmeasurable)
  1090→            - operations: count of sandbox-tracked operations
  1091→        """
  1092→        self._log_operation("sandbox_enter", {"plugin": plugin_name})
  1093→        start_time = time.monotonic()
  1094→        memory_before = self._check_memory_usage()
  1095→
  1096→        try:
  1097→            # Set CPU time limit
  1098→            self._enforce_cpu_limit()
  1099→            self._log_operation("cpu_limit_set", {
  1100→                "limit_seconds": self.cpu_time_seconds,
  1101→            })
  1102→
  1103→            # Execute the plugin function
  1104→            # Note: Python does not provide true sandboxing. The restricted
  1105→            # builtins are applied when the plugin code uses exec/eval.
  1106→            # For direct function calls, we wrap and monitor.
  1107→            self._log_operation("plugin_execute_start", {
  1108→                "plugin": plugin_name,
  1109→                "args": list(kwargs.keys()),
  1110→            })
  1111→
  1112→            try:
  1113→                result = plugin_func(**kwargs)
  1114→            except TypeError as e:
  1115→                raise SandboxViolation(
  1116→                    "plugin_call",
  1117→                    f"Plugin call failed: {e}",
  1118→                )
  1119→
  1120→            self._log_operation("plugin_execute_end", {
  1121→                "plugin": plugin_name,
  1122→                "result_type": type(result).__name__,
  1123→            })
  1124→
  1125→            # Validate output
  1126→            findings = self._validate_output(result)
  1127→            self._log_operation("output_validated", {
  1128→                "plugin": plugin_name,
  1129→                "finding_count": len(findings),
  1130→            })
  1131→
  1132→            memory_after = self._check_memory_usage()
  1133→            if (memory_after > 0
  1134→                    and memory_before > 0):
  1135→                memory_delta = memory_after - memory_before
  1136→                if memory_delta > self.memory_limit_bytes:
  1137→                    raise ResourceLimitExceeded(
  1138→                        "memory",
  1139→                        self.memory_limit_bytes,
  1140→                        memory_delta,
  1141→                    )
  1142→
  1143→            return {
  1144→                "success": True,
  1145→                "findings": findings,
  1146→                "execution_time": time.monotonic() - start_time,
  1147→                "memory_used": max(0, memory_after - memory_before),
  1148→                "operations": len(self._operation_log),
  1149→            }
  1150→
  1151→        except SandboxViolation as exc:
  1152→            self._log_operation("sandbox_violation", {
  1153→                "plugin": plugin_name,
  1154→                "operation": exc.operation,
  1155→                "reason": exc.reason,
  1156→            })
  1157→            return {
  1158→                "success": False,
  1159→                "error": str(exc),
  1160→                "execution_time": time.monotonic() - start_time,
  1161→                "memory_used": 0,
  1162→                "operations": len(self._operation_log),
  1163→            }
  1164→
  1165→        except ResourceLimitExceeded as exc:
  1166→            self._log_operation("resource_limit_exceeded", {
  1167→                "plugin": plugin_name,
  1168→                "resource": exc.resource,
  1169→                "limit": exc.limit,
  1170→                "actual": exc.actual,
  1171→            })
  1172→            return {
  1173→                "success": False,
  1174→                "error": str(exc),
  1175→                "execution_time": time.monotonic() - start_time,
  1176→                "memory_used": 0,
  1177→                "operations": len(self._operation_log),
  1178→            }
  1179→
  1180→        except Exception as exc:
  1181→            self._log_operation("plugin_error", {
  1182→                "plugin": plugin_name,
  1183→                "error_type": type(exc).__name__,
  1184→                "error": str(exc)[:500],
  1185→            })
  1186→            return {
  1187→                "success": False,
  1188→                "error": f"Plugin execution error: {type(exc).__name__}: {exc}",
  1189→                "execution_time": time.monotonic() - start_time,
  1190→                "memory_used": 0,
  1191→                "operations": len(self._operation_log),
  1192→            }
  1193→
  1194→        finally:
  1195→            # Restore CPU limits
  1196→            self._restore_cpu_limit()
  1197→            self._log_operation("sandbox_exit", {"plugin": plugin_name})
  1198→
  1199→    def get_operation_log(self) -> List[Dict[str, Any]]:

  1209→    @staticmethod
  1210→    def check_code_for_violations(code: str) -> List[Dict[str, Any]]:
  1211→        """Statically analyze code for sandbox violations.
  1212→
  1213→        Checks for:
  1214→        - Use of forbidden builtins (eval, exec, compile, open, __import__)
  1215→        - Import of forbidden modules
  1216→        - Use of dangerous attributes (os.system, subprocess.call, etc.)
  1217→
  1218→        Args:
  1219→            code: Python source code to analyze.
  1220→
  1221→        Returns:
  1222→            List of violation dicts with keys:
  1223→            - type: violation type
  1224→            - match: the matching text
  1225→            - line: line number
  1226→            - severity: "critical" or "high"
  1227→        """
  1228→        violations: List[Dict[str, Any]] = []
  1229→        lines = code.split("\n")
  1230→
  1231→        # Check for forbidden builtins
  1232→        forbidden_builtins_pattern = re.compile(
  1233→            r'\b(' + '|'.join(re.escape(b) for b in _FORBIDDEN_BUILTINS) + r')\b'
  1234→        )
  1235→        for line_num, line in enumerate(lines, start=1):
  1236→            # Skip comments
  1237→            stripped = line.strip()
  1238→            if stripped.startswith("#"):
  1239→                continue
  1240→            # Check for string contexts (simple heuristic)
  1241→            # Remove string literals to avoid false positives
  1242→            clean_line = re.sub(r'"[^"]*"', '""', line)
  1243→            clean_line = re.sub(r"'[^']*'", "''", clean_line)
  1244→
  1245→            for match in forbidden_builtins_pattern.finditer(clean_line):
  1246→                builtin = match.group(1)
  1247→                # Avoid flagging __import__ when it's just a string reference
  1248→                if builtin == "__import__" and "__import__" not in clean_line.split("=")[-1]:
  1249→                    violations.append({
  1250→                        "type": "forbidden_builtin",
  1251→                        "match": builtin,
  1252→                        "line": line_num,
  1253→                        "severity": "critical",
  1254→                    })
  1255→                elif builtin != "__import__":
  1256→                    violations.append({
  1257→                        "type": "forbidden_builtin",
  1258→                        "match": builtin,
  1259→                        "line": line_num,
  1260→                        "severity": "critical",
  1261→                    })
  1262→
  1263→        # Check for forbidden imports
  1264→        import_pattern = re.compile(
  1265→            r'\b(?:import|from)\s+([a-zA-Z_][a-zA-Z0-9_]*)'
  1266→        )
  1267→        for line_num, line in enumerate(lines, start=1):
  1268→            stripped = line.strip()
  1269→            if stripped.startswith("#"):
  1270→                continue
  1271→            for match in import_pattern.finditer(line):
  1272→                module = match.group(1)
  1273→                if module in _FORBIDDEN_MODULES:
  1274→                    violations.append({
  1275→                        "type": "forbidden_module",
  1276→                        "match": module,
  1277→                        "line": line_num,
  1278→                        "severity": "high",
  1279→                    })
  1280→
  1281→        # Check for dangerous attribute access patterns
  1282→        dangerous_patterns = [
  1283→            (r'\bos\.system\b', "os.system", "critical"),
  1284→            (r'\bos\.popen\b', "os.popen", "critical"),
  1285→            (r'\bsubprocess\.', "subprocess", "critical"),
  1286→            (r'\bshutil\.', "shutil", "high"),
  1287→            (r'\bctypes\.', "ctypes", "critical"),
  1288→            (r'\b__builtins__\b', "__builtins__", "high"),
  1289→            (r'\bglobals\(\)', "globals()", "high"),
  1290→            (r'\blocals\(\)', "locals()", "medium"),
  1291→            (r'\bgetattr\s*\(\s*__builtins__', "getattr(__builtins__", "critical"),
  1292→        ]
  1293→        for line_num, line in enumerate(lines, start=1):
  1294→            stripped = line.strip()
  1295→            if stripped.startswith("#"):
  1296→                continue
  1297→            for pattern, name, severity in dangerous_patterns:
  1298→                if re.search(pattern, line):
  1299→                    violations.append({
  1300→                        "type": "dangerous_pattern",
  1301→                        "match": name,
  1302→                        "line": line_num,
  1303→                        "severity": severity,
  1304→                    })
  1305→
  1306→        return violations

  1419→    Usage::
  1420→
  1421→        sm = SecretsManager()
  1422→        findings = sm.scan_content("some text with AKIAIOSFODNN7EXAMPLE key")
  1423→        report = sm.get_report()
  1424→    """
  1425→
  1426→    SEVERITY_ORDER: Dict[str, int] = {
  1427→        "critical": 0,
  1428→        "high": 1,
  1429→        "medium": 2,
  1430→        "low": 3,
  1431→    }
  1432→
  1433→    def __init__(self, audit_logger: Optional[SecurityAuditLogger] = None) -> None:
  1434→        self._findings: List[SecretFinding] = []
  1435→        self._audit = audit_logger
  1436→        self._scan_count = 0
  1437→        self._files_scanned = 0
  1438→        self._lock = threading.Lock()
  1439→
  1440→    def scan_for_secrets(
  1441→        self,
  1442→        content: str,
  1443→        source: str = "string_input",
  1444→    ) -> List[Dict[str, Any]]:
  1445→        """Scan content for secrets using both built-in and extended patterns.
  1446→
  1447→        This method first uses the existing detect_secrets_in_text() from
  1448→        security.py, then applies additional patterns from this module.
  1449→
  1450→        Args:
  1451→            content: Text content to scan.
  1452→            source: Identifier for where the content came from.
  1453→
  1454→        Returns:
  1455→            List of dicts with keys:
  1456→            - secret_type: str
  1457→            - match: str (redacted for critical/high)
  1458→            - line: int
  1459→            - offset: int
  1460→            - severity: str
  1461→            - description: str
  1462→            - source: str
  1463→            - detector: str ("core" or "extended")
  1464→        """
  1465→        if not content:
  1466→            return []
  1467→
  1468→        self._scan_count += 1
  1469→        results: List[Dict[str, Any]] = []
  1470→        lines = content.split("\n")
  1471→
  1472→        # Phase 1: Use existing core detection from security.py
  1473→        core_findings = detect_secrets_in_text(content)
  1474→        for finding in core_findings:
  1475→            redacted = self._redact_match(finding["match"], finding["confidence"])
  1476→            results.append({
  1477→                "secret_type": finding["type"],
  1478→                "match": redacted,
  1479→                "line": finding["line"],
  1480→                "offset": finding["offset"],
  1481→                "severity": finding["confidence"],
  1482→                "description": self._describe_finding(finding["type"]),
  1483→                "source": source,
  1484→                "detector": "core",
  1485→            })
  1486→
  1487→        # Phase 2: Extended patterns
  1488→        for secret_type, pattern, severity, description in _EXTENDED_SECRET_PATTERNS:
  1489→            for line_num, line in enumerate(lines, start=1):
  1490→                for match in pattern.finditer(line):
  1491→                    matched_text = match.group(0)
  1492→                    redacted = self._redact_match(matched_text, severity)
  1493→                    results.append({
  1494→                        "secret_type": secret_type,
  1495→                        "match": redacted,
  1496→                        "line": line_num,
  1497→                        "offset": match.start(),
  1498→                        "severity": severity,
  1499→                        "description": description,
  1500→                        "source": source,
  1501→                        "detector": "extended",
  1502→                    })
  1503→
  1504→        # Store findings
  1505→        with self._lock:
  1506→            for r in results:
  1507→                self._findings.append(SecretFinding(
  1508→                    secret_type=r["secret_type"],
  1509→                    match=r["match"],
  1510→                    line=r["line"],
  1511→                    offset=r["offset"],
  1512→                    severity=r["severity"],
  1513→                    description=r["description"],
  1514→                    source=source,
  1515→                ))
  1516→
  1517→        # Audit log
  1518→        if results and self._audit:
  1519→            severity_counts: Dict[str, int] = {}
  1520→            for r in results:
  1521→                s = r["severity"]
  1522→                severity_counts[s] = severity_counts.get(s, 0) + 1
  1523→            self._audit.warn(
  1524→                "SECRETS_DETECTED",
  1525→                target=source,
  1526→                details={
  1527→                    "total": len(results),
  1528→                    "severity_counts": severity_counts,
  1529→                    "types": list(set(r["secret_type"] for r in results)),
  1530→                },
  1531→            )
  1532→
  1533→        return results
  1534→
  1535→    def scan_findings(
  1536→        self,
  1537→        findings: List[Any],
  1538→        source_prefix: str = "scan_finding",
  1539→    ) -> List[Dict[str, Any]]:
  1540→        """Scan a list of findings for leaked secrets.
  1541→
  1542→        Each finding is serialized to text and scanned.  Accepts both
  1543→        Finding objects (with a ``to_dict()`` method) and plain dicts.
  1544→
  1545→        Args:
  1546→            findings: List of Finding objects or dicts.
  1547→            source_prefix: Prefix for the source field (e.g.
  1548→                ``"scan_finding"`` → ``"scan_finding:recon"``).
  1549→
  1550→        Returns:
  1551→            Combined list of all secret detection results.
  1552→        """
  1553→        all_results: List[Dict[str, Any]] = []
  1554→        for f in findings:
  1555→            try:
  1556→                if hasattr(f, "to_dict"):
  1557→                    text = json.dumps(f.to_dict(), default=str)
  1558→                    module = getattr(f, "module", "unknown")
  1559→                elif isinstance(f, dict):
  1560→                    text = json.dumps(f, default=str)
  1561→                    module = f.get("module", "unknown")
  1562→                else:
  1563→                    text = str(f)
  1564→                    module = "unknown"
  1565→                source = f"{source_prefix}:{module}"
  1566→                results = self.scan_for_secrets(text, source=source)
  1567→                all_results.extend(results)
  1568→            except Exception as exc:
  1569→                logger.debug(
  1570→                    "Secret scan failed for finding: %s", exc, exc_info=True,
  1571→                )
  1572→        return all_results
  1573→
  1574→    def scan_file(self, filepath: str) -> List[Dict[str, Any]]:
  1575→        """Scan a file for secrets.
  1576→
  1577→        Args:
  1578→            filepath: Path to the file to scan.
  1579→
  1580→        Returns:
  1581→            Same format as scan_for_secrets().
  1582→        """
  1583→        if not os.path.isfile(filepath):
  1584→            return []
  1585→
  1586→        try:
  1587→            with open(filepath, "r", encoding="utf-8", errors="ignore") as f:
  1588→                content = f.read()
  1589→        except OSError as exc:
  1590→            logger.warning("Cannot read file for secret scanning: %s: %s", filepath, exc)
  1591→            return []
  1592→
  1593→        self._files_scanned += 1
  1594→        return self.scan_for_secrets(content, source=filepath)
  1595→
  1596→    def scan_directory(
  1597→        self,
  1598→        dirpath: str,
  1599→        extensions: Optional[Set[str]] = None,
  1600→        max_files: int = 1000,
  1601→    ) -> List[Dict[str, Any]]:
  1602→        """Recursively scan a directory for secrets.
  1603→
  1604→        Args:
  1605→            dirpath: Root directory to scan.
  1606→            extensions: Set of file extensions to scan (e.g., {".py", ".js"}).
  1607→                      If None, scans all files.
  1608→            max_files: Maximum number of files to scan.
  1609→
  1610→        Returns:
  1611→            Combined list of all findings.
  1612→        """
  1613→        all_results: List[Dict[str, Any]] = []
  1614→        files_scanned = 0
  1615→
  1616→        if not os.path.isdir(dirpath):
  1617→            return []
  1618→
  1619→        for root, _dirs, files in os.walk(dirpath):
  1620→            if files_scanned >= max_files:
  1621→                break
  1622→            for fname in sorted(files):
  1623→                if files_scanned >= max_files:
  1624→                    break
  1625→                # Skip hidden files and common non-text files
  1626→                if fname.startswith("."):
  1627→                    continue
  1628→                if fname.endswith((".pyc", ".pyo", ".so", ".dll", ".exe",
  1629→                                  ".png", ".jpg", ".jpeg", ".gif", ".ico",
  1630→                                  ".zip", ".tar", ".gz", ".bz2", ".xz")):
  1631→                    continue
  1632→
  1633→                filepath = os.path.join(root, fname)
  1634→                if extensions is not None:
  1635→                    _, ext = os.path.splitext(fname)
  1636→                    if ext.lower() not in extensions:
  1637→                        continue
  1638→
  1639→                results = self.scan_file(filepath)
  1640→                all_results.extend(results)
  1641→                files_scanned += 1
  1642→
  1643→        return all_results
  1644→
  1645→    def get_report(self) -> Dict[str, Any]:
  1646→        """Generate a comprehensive secrets scanning report.
  1647→
  1648→        Returns:
  1649→            Dict with keys:
  1650→            - total_findings: int
  1651→            - by_severity: dict mapping severity -> count
  1652→            - by_type: dict mapping secret_type -> count
  1653→            - by_source: dict mapping source -> count
  1654→            - by_detector: dict mapping detector -> count
  1655→            - critical_findings: list of critical/high severity findings
  1656→            - scan_stats: scanning statistics
  1657→        """
  1658→        with self._lock:
  1659→            findings = list(self._findings)
  1660→
  1661→        by_severity: Dict[str, int] = {}
  1662→        by_type: Dict[str, int] = {}
  1663→        by_source: Dict[str, int] = {}
  1664→        by_detector: Dict[str, int] = {}
  1665→        critical_findings: List[Dict[str, Any]] = []
  1666→
  1667→        for f in findings:
  1668→            by_severity[f.severity] = by_severity.get(f.severity, 0) + 1
  1669→            by_type[f.secret_type] = by_type.get(f.secret_type, 0) + 1
  1670→            by_source[f.source] = by_source.get(f.source, 0) + 1
  1671→            _CORE_TYPES = {
  1672→                "aws_key", "github_token", "github_token_fine_grained",
  1673→                "generic_api_key", "private_key_rsa", "private_key_ec",
  1674→                "private_key_generic", "password_in_url", "db_connection_string",
  1675→                "jwt_token",
  1676→            }
  1677→            detector = "core" if f.secret_type in _CORE_TYPES else "extended"
  1678→            by_detector[detector] = by_detector.get(detector, 0) + 1
  1679→
  1680→            if f.severity in ("critical", "high"):
  1681→                critical_findings.append({
  1682→                    "type": f.secret_type,
  1683→                    "line": f.line,
  1684→                    "severity": f.severity,
  1685→                    "description": f.description,
  1686→                    "source": f.source,
  1687→                })
  1688→
  1689→        return {
  1690→            "total_findings": len(findings),
  1691→            "by_severity": by_severity,
  1692→            "by_type": by_type,
  1693→            "by_source": by_source,
  1694→            "by_detector": by_detector,
  1695→            "critical_findings": critical_findings,
  1696→            "scan_stats": {
  1697→                "content_scans": self._scan_count,
  1698→                "files_scanned": self._files_scanned,
  1699→            },
  1700→            "generated_at": datetime.now(timezone.utc).isoformat(),
  1701→        }
  1702→
  1703→    def clear_findings(self) -> None:
  1704→        """Clear all stored findings."""
  1705→        with self._lock:
  1706→            self._findings.clear()
  1707→            self._scan_count = 0
  1708→            self._files_scanned = 0

  1710→    @staticmethod

  1715→        Medium/low: show full match.

  1717→        if severity in ("critical", "high"):
  1718→            if len(match) > 12:
  1719→                return match[:4] + "***REDACTED***" + match[-4:]
  1720→            return "***REDACTED***"
  1721→        return match

  1726→        descriptions: Dict[str, str] = {

  1729→            "github_token_fine_grained": "GitHub Fine-Grained PAT",
  1730→            "generic_api_key": "Generic API Key",
  1731→            "private_key_rsa": "RSA Private Key",
  1732→            "private_key_ec": "EC Private Key",

  1741→# ══════════════════════════════════════════════════════════════════════════════
  1742→# 4. TAMPER-EVIDENT SECURITY AUDIT LOGGER
  1743→# ══════════════════════════════════════════════════════════════════════════════
  1744→
  1745→
  1746→class TamperEvidenceLogger:
  1747→    """Tamper-evident security event logger.
  1748→
  1749→    Extends the concept of SecurityAuditLogger with hash-chained entries
  1750→    that make it detectable if log entries have been added, removed, or
  1751→    modified.
  1752→
  1753→    Each log entry includes:
  1754→    - sequence number (monotonically increasing)
  1755→    - timestamp (ISO 8601 UTC)
  1756→    - level (INFO, WARN, ALERT)
  1757→    - event_type
  1758→    - module
  1759→    - target
  1760→    - details (dict)
  1761→    - previous_hash: SHA-256 of the previous entry's canonical form
  1762→    - entry_hash: SHA-256 of this entry (including previous_hash)
  1763→
  1764→    The chain can be verified at any time to detect tampering.
  1765→    """
  1766→
  1767→    def __init__(
  1768→        self,
  1769→        log_dir: Optional[str] = None,
  1770→        module_name: str = "security_hardening",
  1771→        chain_file: Optional[str] = None,
  1772→    ) -> None:
  1773→        self.module_name = module_name
  1774→        if log_dir is None:
  1775→            log_dir = os.path.join(os.path.expanduser("~"), ".reconpro")
  1776→        self.log_dir = log_dir
  1777→        os.makedirs(log_dir, exist_ok=True)
  1778→
  1779→        if chain_file is None:
  1780→            chain_file = os.path.join(log_dir, "security_chain.log")
  1781→        self.chain_file = chain_file
  1782→
  1783→        self._entries: List[Dict[str, Any]] = []
  1784→        self._last_hash: str = "GENESIS"  # Seed for the chain
  1785→        self._sequence: int = 0
  1786→        self._lock = threading.Lock()
  1787→        self._event_counts: Dict[str, int] = {}
  1788→        self._level_counts: Dict[str, int] = {"INFO": 0, "WARN": 0, "ALERT": 0}
  1789→
  1790→        # Try to load existing chain
  1791→        self._load_chain()
  1792→
  1793→    def _load_chain(self) -> None:
  1794→        """Load existing chain from disk to continue the hash sequence."""
  1795→        if not os.path.isfile(self.chain_file):
  1796→            return
  1797→        try:
  1798→            with open(self.chain_file, "r", encoding="utf-8") as f:
  1799→                for line in f:
  1800→                    line = line.strip()
  1801→                    if not line:
  1802→                        continue
  1803→                    try:
  1804→                        entry = json.loads(line)
  1805→                        self._entries.append(entry)
  1806→                        self._sequence = entry.get("sequence", 0)
  1807→                        self._last_hash = entry.get("entry_hash", "GENESIS")
  1808→                        level = entry.get("level", "INFO")
  1809→                        if level in self._level_counts:
  1810→                            self._level_counts[level] += 1
  1811→                        event_type = entry.get("event_type", "")
  1812→                        self._event_counts[event_type] = self._event_counts.get(event_type, 0) + 1
  1813→                    except (json.JSONDecodeError, KeyError):
  1814→                        continue
  1815→        except OSError:
  1816→            pass
  1817→
  1818→    def _compute_hash(self, canonical: str) -> str:
  1819→        """Compute SHA-256 hash of a canonical string."""
  1820→        return hashlib.sha256(canonical.encode("utf-8")).hexdigest()
  1821→
  1822→    def _make_canonical(self, entry: Dict[str, Any]) -> str:
  1823→        """Create a canonical string representation for hashing.
  1824→
  1825→        The canonical form includes all fields in a deterministic order.
  1826→        """
  1827→        # Build a deterministic representation
  1828→        parts = [
  1829→            str(entry.get("sequence", "")),
  1830→            str(entry.get("timestamp", "")),
  1831→            str(entry.get("level", "")),
  1832→            str(entry.get("event_type", "")),
  1833→            str(entry.get("module", "")),
  1834→            sanitize_log(str(entry.get("target", ""))),
  1835→            str(entry.get("previous_hash", "")),
  1836→            json.dumps(entry.get("details", {}), sort_keys=True, default=str),
  1837→        ]
  1838→        return "|".join(parts)
  1839→
  1840→    def log(
  1841→        self,
  1842→        level: str,
  1843→        event_type: str,
  1844→        target: str = "",
  1845→        details: Optional[Dict[str, Any]] = None,
  1846→    ) -> Dict[str, Any]:
  1847→        """Append a tamper-evident log entry.
  1848→
  1849→        Args:
  1850→            level: "INFO", "WARN", or "ALERT".
  1851→            event_type: Category of security event.
  1852→            target: Target associated with the event.
  1853→            details: Optional dict of additional details.
  1854→
  1855→        Returns:
  1856→            The complete log entry dict including hashes.
  1857→        """
  1858→        if level not in ("INFO", "WARN", "ALERT"):
  1859→            level = "INFO"
  1860→
  1861→        with self._lock:
  1862→            self._sequence += 1
  1863→            entry: Dict[str, Any] = {
  1864→                "sequence": self._sequence,
  1865→                "timestamp": datetime.now(timezone.utc).isoformat(),
  1866→                "level": level,
  1867→                "event_type": event_type,
  1868→                "module": self.module_name,
  1869→                "target": sanitize_log(target),
  1870→                "details": details or {},
  1871→                "previous_hash": self._last_hash,
  1872→            }
  1873→
  1874→            # Compute hash of this entry
  1875→            canonical = self._make_canonical(entry)
  1876→            entry["entry_hash"] = self._compute_hash(canonical)
  1877→
  1878→            self._last_hash = entry["entry_hash"]
  1879→            self._entries.append(entry)
  1880→
  1881→            # Update statistics
  1882→            self._level_counts[level] = self._level_counts.get(level, 0) + 1
  1883→            self._event_counts[event_type] = self._event_counts.get(event_type, 0) + 1
  1884→
  1885→            # Persist to disk
  1886→            self._persist_entry(entry)
  1887→
  1888→            return dict(entry)
  1889→
  1890→    def _persist_entry(self, entry: Dict[str, Any]) -> None:
  1891→        """Append a single entry to the chain file on disk."""
  1892→        try:
  1893→            with open(self.chain_file, "a", encoding="utf-8") as f:
  1894→                f.write(json.dumps(entry, default=str) + "\n")
  1895→        except OSError as exc:
  1896→            logger.error("Failed to persist security log entry: %s", exc)
  1897→
  1898→    def verify_chain(self) -> Dict[str, Any]:
  1899→        """Verify the integrity of the entire log chain.
  1900→
  1901→        Checks that:
  1902→        1. Sequence numbers are monotonically increasing.
  1903→        2. Each entry's hash matches its content.
  1904→        3. Each entry's previous_hash matches the prior entry's hash.
  1905→
  1906→        Returns:
  1907→            Dict with keys:
  1908→            - valid: bool (True if chain is intact)
  1909→            - entries_checked: int
  1910→            - first_violation: dict describing the first issue found (if any)
  1911→            - violations: list of all violation descriptions
  1912→        """
  1913→        violations: List[str] = []
  1914→        first_violation: Optional[Dict[str, Any]] = None
  1915→        prev_hash = "GENESIS"
  1916→        expected_sequence = 1
  1917→
  1918→        for i, entry in enumerate(self._entries):
  1919→            entry_issues: List[str] = []
  1920→
  1921→            # Check sequence
  1922→            if entry.get("sequence") != expected_sequence:
  1923→                issue = (
  1924→                    f"Entry {i}: expected sequence {expected_sequence}, "
  1925→                    f"got {entry.get('sequence')}"
  1926→                )
  1927→                entry_issues.append(issue)
  1928→
  1929→            # Check previous_hash
  1930→            if entry.get("previous_hash") != prev_hash:
  1931→                issue = (
  1932→                    f"Entry {i}: previous_hash mismatch. "
  1933→                    f"Expected {prev_hash[:16]}..., got {str(entry.get('previous_hash', ''))[:16]}..."
  1934→                )
  1935→                entry_issues.append(issue)
  1936→
  1937→            # Check entry_hash
  1938→            entry_copy = {k: v for k, v in entry.items() if k != "entry_hash"}
  1939→            canonical = self._make_canonical(entry_copy)
  1940→            computed_hash = self._compute_hash(canonical)
  1941→            if entry.get("entry_hash") != computed_hash:
  1942→                issue = (
  1943→                    f"Entry {i}: hash mismatch. "
  1944→                    f"Stored {str(entry.get('entry_hash', ''))[:16]}..., "
  1945→                    f"computed {computed_hash[:16]}..."
  1946→                )
  1947→                entry_issues.append(issue)
  1948→
  1949→            if entry_issues:
  1950→                if first_violation is None:
  1951→                    first_violation = {
  1952→                        "entry_index": i,
  1953→                        "sequence": entry.get("sequence"),
  1954→                        "issues": entry_issues,
  1955→                    }
  1956→                violations.extend(entry_issues)
  1957→
  1958→            prev_hash = entry.get("entry_hash", "")
  1959→            expected_sequence += 1
  1960→
  1961→        return {
  1962→            "valid": len(violations) == 0,
  1963→            "entries_checked": len(self._entries),
  1964→            "first_violation": first_violation,
  1965→            "violations": violations,
  1966→            "verified_at": datetime.now(timezone.utc).isoformat(),
  1967→        }
  1968→
  1969→    def get_statistics(self) -> Dict[str, Any]:
  1970→        """Return statistics about logged security events.
  1971→
  1972→        Returns:
  1973→            Dict with keys:
