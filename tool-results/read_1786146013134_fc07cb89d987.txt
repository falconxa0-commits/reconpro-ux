    95→    95→class ChainPattern:
    96→    96→    """A rule that describes how specific finding categories combine
    97→    97→    into a compound attack path.
    98→    98→
    99→    99→    Parameters
   100→   100→    ----------
   101→   101→    name:
   102→   102→        Human-readable name for the compound attack.
   103→   103→    description:
   104→   104→        One-line description of the attack scenario.
   105→   105→    severity:
   106→   106→        Resulting severity of the compound attack (uppercase).
   107→   107→    required_categories:
   108→   108→        The finding categories that must all be present to trigger this chain.
   109→   109→        A finding matches if its ``category`` or ``type`` field contains
   110→   110→        the pattern as a substring (case-insensitive).
   111→   111→    required_tags:
   112→   112→        Optional additional tags that at least one finding must carry.
   113→   113→        Tags are checked against the ``tags`` list field of findings.
   114→   114→    severity_multiplier:
   115→   115→        Score multiplier applied on top of the base severity.
   116→   116→    narrative_template:
   117→   117→        A template string describing the compound attack.  May contain
   118→   118→        ``{findings}`` placeholder (replaced with finding titles).
   119→   119→    steps:
   120→   120→        Ordered attack steps describing the exploit sequence.
   121→   121→    remediation:
   122→   122→        How to break this specific chain.
   123→   123→    """
   124→   124→
   125→   125→    name: str
   126→   126→    description: str
   127→   127→    severity: str
   128→   128→    required_categories: List[str]
   129→   129→    required_tags: List[str] = field(default_factory=list)
   130→   130→    severity_multiplier: float = 1.5
   131→   131→    narrative_template: str = ""
   132→   132→    steps: List[str] = field(default_factory=list)
   133→   133→    remediation: str = ""
   134→   134→
   135→   135→
   136→   136→@dataclass
   137→   137→class ChainReport:
   138→   138→    """A realized compound attack path, produced when a ChainPattern
   139→   139→    matches against actual findings.
   140→   140→
   141→   141→    Parameters
   142→   142→    ----------
   143→   143→    name:
   144→   144→        Chain name (from pattern or LLM).
   145→   145→    severity:
   146→   146→        Computed severity (one of info/low/medium/high/critical).
   147→   147→    score:
   148→   148→        Numeric risk score for ranking.
   149→   149→    findings:
   150→   150→        The individual finding dicts that form this chain.
   151→   151→    narrative:
   152→   152→        Human-readable attack story.
   153→   153→    steps:
   154→   154→        Ordered attack steps.
   155→   155→    remediation:
   156→   156→        How to break the chain.
   157→   157→    confidence:
   158→   158→        0.0 – 1.0 confidence that this chain is real and exploitable.
   159→   159→    """
   160→   160→
   161→   161→    name: str
   162→   162→    severity: str
   163→   163→    score: float
   164→   164→    findings: List[Dict[str, Any]]
   165→   165→    narrative: str
   166→   166→    steps: List[str]
   167→   167→    remediation: str
   168→   168→    confidence: float = 0.8
   169→   169→
   170→   170→    def to_dict(self) -> Dict[str, Any]:
   171→   171→        """Serialise to a plain dict for JSON output / display."""
   172→   172→        return {
   173→   173→            "name": self.name,
   174→   174→            "severity": self.severity,
   175→   175→            "score": round(self.score, 2),
   176→   176→            "finding_count": len(self.findings),
   177→   177→            "finding_titles": [f.get("title", f.get("name", "")) for f in self.findings],
   178→   178→            "narrative": self.narrative,
   179→   179→            "steps": self.steps,
   180→   180→            "remediation": self.remediation,
   181→   181→            "confidence": round(self.confidence, 2),
   182→   182→            "is_chain": True,
   183→   183→        }

  1416→  1416→class AttackPathFinder:
  1417→  1417→    """Discovers compound attack paths by combining rule-based pattern
  1418→  1418→    matching, graph-based traversal, and optional LLM-assisted analysis.
  1419→  1419→
  1420→  1420→    Parameters
  1421→  1421→    ----------
  1422→  1422→    memory_store :
  1423→  1423→        Optional :class:`~reconpro.memory.UnifiedMemoryStore` instance for
  1424→  1424→        graph-based chain discovery.  When ``None``, only rule-based
  1425→  1425→        analysis is available.
  1426→  1426→    """
  1427→  1427→
  1428→  1428→    def __init__(self, memory_store: UnifiedMemoryStore | None = None) -> None:
  1429→  1429→        self._memory = memory_store
  1430→  1430→
  1431→  1431→    # ------------------------------------------------------------------
  1432→  1432→    # Public API
  1433→  1433→    # ------------------------------------------------------------------
  1434→  1434→
  1435→  1435→    def find_chains(
  1436→  1436→        self,
  1437→  1437→        target: str,
  1438→  1438→        findings: List[Dict[str, Any]],
  1439→  1439→    ) -> List[ChainReport]:
  1440→  1440→        """Run rule-based chain discovery against *findings* for *target*.
  1441→  1441→
  1442→  1442→        Groups findings by category, matches against all ``CHAIN_RULES``,
  1443→  1443→        and returns a ranked list of :class:`ChainReport` objects.
  1444→  1444→        """
  1445→  1445→        # Index findings by normalised category
  1446→  1446→        by_cat: Dict[str, List[Dict[str, Any]]] = defaultdict(list)
  1447→  1447→        for f in findings:
  1448→  1448→            cat = _finding_category(f)
  1449→  1449→            if cat:
  1450→  1450→                by_cat[cat].append(f)
  1451→  1451→
  1452→  1452→        # Also index by tags
  1453→  1453→        by_tag: Dict[str, List[Dict[str, Any]]] = defaultdict(list)
  1454→  1454→        for f in findings:
  1455→  1455→            for tag in _finding_tags(f):
  1456→  1456→                by_tag[tag].append(f)
  1457→  1457→
  1458→  1458→        reports: List[ChainReport] = []
  1459→  1459→        seen_rule_names: Set[str] = set()
  1460→  1460→
  1461→  1461→        for pattern in CHAIN_RULES:
  1462→  1462→            if pattern.name in seen_rule_names:
  1463→  1463→                continue
  1464→  1464→
  1465→  1465→            # Check all required categories are present
  1466→  1466→            matched_findings: List[Dict[str, Any]] = []
  1467→  1467→            all_cats_present = True
  1468→  1468→            for req_cat in pattern.required_categories:
  1469→  1469→                req_lower = req_cat.lower()
  1470→  1470→                # Check if any finding category contains the required category as substring
  1471→  1471→                found = False
  1472→  1472→                for existing_cat, cat_findings in by_cat.items():
  1473→  1473→                    if req_lower in existing_cat or existing_cat in req_lower:
  1474→  1474→                        if cat_findings:
  1475→  1475→                            matched_findings.append(cat_findings[0])
  1476→  1476→                            found = True
  1477→  1477→                            break
  1478→  1478→                if not found:
  1479→  1479→                    all_cats_present = False
  1480→  1480→                    break
  1481→  1481→
  1482→  1482→            if not all_cats_present:
  1483→  1483→                continue
  1484→  1484→
  1485→  1485→            # Check required tags (at least one finding must have at least one tag)
  1486→  1486→            if pattern.required_tags:
  1487→  1487→                req_tags_lower = [t.lower() for t in pattern.required_tags]
  1488→  1488→                tag_match = False
  1489→  1489→                for f in matched_findings:
  1490→  1490→                    f_tags = _finding_tags(f)
  1491→  1491→                    for rt in req_tags_lower:
  1492→  1492→                        if rt in f_tags or any(rt in ft for ft in f_tags):
  1493→  1493→                            tag_match = True
  1494→  1494→                            break
  1495→  1495→                    if tag_match:
  1496→  1496→                        break
  1497→  1497→                if not tag_match:
  1498→  1498→                    # Also check all findings, not just matched ones
  1499→  1499→                    for f in findings:
  1500→  1500→                        f_tags = _finding_tags(f)
  1501→  1501→                        for rt in req_tags_lower:
  1502→  1502→                            if rt in f_tags or any(rt in ft for ft in f_tags):
  1503→  1503→                                tag_match = True
  1504→  1504→                                break
  1505→  1505→                        if tag_match:
  1506→  1506→                            break
  1507→  1507→                if not tag_match:
  1508→  1508→                    continue
  1509→  1509→
  1510→  1510→            # Compute score
  1511→  1511→            max_child_sev = max(_sev_score(f.get("severity", "info")) for f in matched_findings)
  1512→  1512→            path_length_bonus = (len(matched_findings) - 2) * 1.5  # bonus for longer chains
  1513→  1513→            score = max_child_sev * pattern.severity_multiplier + max(path_length_bonus, 0.0)
  1514→  1514→
  1515→  1515→            # Build confidence based on how specific the match is
  1516→  1516→            confidence = 0.7 + (0.05 * len(pattern.required_categories))
  1517→  1517→            if pattern.required_tags:
  1518→  1518→                confidence += 0.1
  1519→  1519→            confidence = min(confidence, 0.95)
  1520→  1520→
  1521→  1521→            # Build the narrative
  1522→  1522→            narrative = _format_narrative(pattern.narrative_template, matched_findings)
  1523→  1523→
  1524→  1524→            # Build steps (use pattern steps if available)
  1525→  1525→            steps = pattern.steps if hasattr(pattern, "steps") and pattern.steps else []
  1526→  1526→            if not steps:
  1527→  1527→                steps = [f"Step {i+1}: Exploit {_finding_title(f)}" for i, f in enumerate(matched_findings)]
  1528→  1528→
  1529→  1529→            # Build remediation
  1530→  1530→            remediation = pattern.remediation if hasattr(pattern, "remediation") and pattern.remediation else (
  1531→  1531→                "Address each individual finding in the chain to break the compound attack path."
  1532→  1532→            )
  1533→  1533→
  1534→  1534→            report = ChainReport(
  1535→  1535→                name=pattern.name,
  1536→  1536→                severity=_normalise_severity(pattern.severity),
  1537→  1537→                score=round(score, 2),
  1538→  1538→                findings=list(matched_findings),
  1539→  1539→                narrative=narrative,
  1540→  1540→                steps=steps,
  1541→  1541→                remediation=remediation,
  1542→  1542→                confidence=round(confidence, 2),
  1543→  1543→            )
  1544→  1544→            reports.append(report)
  1545→  1545→            seen_rule_names.add(pattern.name)
  1546→  1546→
  1547→  1547→        # Sort by score descending
  1548→  1548→        reports.sort(key=lambda r: r.score, reverse=True)
  1549→  1549→        return reports
  1550→  1550→
  1551→  1551→    def find_compound_paths(self, target: str) -> List[ChainReport]:
  1552→  1552→        """Graph-based chain discovery using the memory store's knowledge graph.
  1553→  1553→
  1554→  1554→        Combines structural graph chains with rule-based matching to
  1555→  1555→        produce compound path reports.
  1556→  1556→
  1557→  1557→        Returns an empty list if no memory store is available.
  1558→  1558→        """
  1559→  1559→        if self._memory is None:
  1560→  1560→            return []
  1561→  1561→
  1562→  1562→        reports: List[ChainReport] = []
  1563→  1563→
  1564→  1564→        # 1. Get graph-based chains from the knowledge graph
  1565→  1565→        try:
  1566→  1566→            graph_chains = self._memory.find_chains(target)
  1567→  1567→        except Exception:
  1568→  1568→            graph_chains = []
  1569→  1569→
  1570→  1570→        # 2. Get all findings for the target
  1571→  1571→        try:
  1572→  1572→            all_findings = self._memory.get_findings(target)
  1573→  1573→        except Exception:
  1574→  1574→            all_findings = []
  1575→  1575→
  1576→  1576→        # Filter to actual vulnerability findings (not scan summaries)
  1577→  1577→        vuln_findings = [
  1578→  1578→            f for f in all_findings
  1579→  1579→            if f.get("severity", "info") not in ("info",)
  1580→  1580→            and f.get("title", "").find("Scan completed") == -1
  1581→  1581→        ]
  1582→  1582→
  1583→  1583→        if not vuln_findings:
  1584→  1584→            # Try rule-based analysis on whatever we have
  1585→  1585→            return self.find_chains(target, all_findings)
  1586→  1586→
  1587→  1587→        # 3. Rule-based chains
  1588→  1588→        rule_reports = self.find_chains(target, vuln_findings)
  1589→  1589→        reports.extend(rule_reports)
  1590→  1590→
  1591→  1591→        # 4. Graph-based chain enrichment
  1592→  1592→        seen_rule_names = {r.name for r in rule_reports}
  1593→  1593→        for chain in graph_chains:
  1594→  1594→            if not chain or len(chain) < 2:
  1595→  1595→                continue
  1596→  1596→
  1597→  1597→            # Extract finding nodes from the chain
  1598→  1598→            chain_findings = [
  1599→  1599→                step for step in chain
  1600→  1600→                if step.get("node_type") in ("finding", "vulnerability", "cve")
  1601→  1601→            ]
  1602→  1602→
  1603→  1603→            if len(chain_findings) < 2:
  1604→  1604→                continue
  1605→  1605→
  1606→  1606→            # Check if this chain is already covered by a rule
  1607→  1607→            chain_title_parts = [f.get("title", "") for f in chain_findings]
  1608→  1608→            covered = False
  1609→  1609→            for existing_name in seen_rule_names:
  1610→  1610→                # Simple check: if any two titles contain words from an existing rule's categories
  1611→  1611→                for part in chain_title_parts[:2]:
  1612→  1612→                    part_lower = part.lower()
  1613→  1613→                    if any(cat.lower() in part_lower for rule in CHAIN_RULES if rule.name == existing_name for cat in rule.required_categories):
  1614→  1614→                        covered = True
  1615→  1615→                        break
  1616→  1616→                if covered:
  1617→  1617→                    break
  1618→  1618→
  1619→  1619→            if covered:
  1620→  1620→                continue
  1621→  1621→
  1622→  1622→            # Build a generic graph-based chain report
  1623→  1623→            max_sev = max(_sev_score(f.get("severity", "info")) for f in chain_findings)
  1624→  1624→            path_len = len(chain_findings)
  1625→  1625→            score = max_sev * 1.3 + (path_len - 2) * 1.0
  1626→  1626→
  1627→  1627→            # Determine chain type from the path
  1628→  1628→            types_in_chain = [
  1629→  1629→                f.get("finding_type", f.get("type", "unknown"))
  1630→  1630→                for f in chain_findings
  1631→  1631→            ]
  1632→  1632→
  1633→  1633→            report = ChainReport(
  1634→  1634→                name=f"Graph Path: {' → '.join(types_in_chain[:3])}",
  1635→  1635→                severity=_normalise_severity(chain_findings[0].get("severity", "medium")),
  1636→  1636→                score=round(score, 2),
  1637→  1637→                findings=chain_findings,
  1638→  1638→                narrative=(
  1639→  1639→                    "Graph analysis discovered a multi-hop attack path: "
  1640→  1640→                    + " → ".join(f.get("title", "unknown") for f in chain_findings)
  1641→  1641→                    + ". These vulnerabilities are connected through shared assets "
  1642→  1642→                    "and can be chained for greater impact."
  1643→  1643→                ),
  1644→  1644→                steps=[
  1645→  1645→                    f"Step {i+1}: Exploit {_finding_title(f)} ({f.get('severity', '?')} severity)"
  1646→  1646→                    for i, f in enumerate(chain_findings)
  1647→  1647→                ],
  1648→  1648→                remediation=(
  1649→  1649→                    "Break this attack path by remediating the highest-severity finding "
  1650→  1650→                    f"in the chain: {chain_findings[0].get('title', 'unknown')}."
  1651→  1651→                ),
  1652→  1652→                confidence=0.5,  # Graph-based chains have lower confidence
  1653→  1653→            )
  1654→  1654→            reports.append(report)
  1655→  1655→
  1656→  1656→        # Deduplicate and sort
  1657→  1657→        reports.sort(key=lambda r: r.score, reverse=True)
  1658→  1658→        return reports
  1659→  1659→
  1660→  1660→    def llm_assisted_chains(
  1661→  1661→        self,
  1662→  1662→        target: str,
  1663→  1663→        findings: List[Dict[str, Any]],
  1664→  1664→        llm_client: Any = None,
  1665→  1665→    ) -> List[ChainReport]:
  1666→  1666→        """Use an LLM (OpenAI or Anthropic) to discover additional chains.
  1667→  1667→
  1668→  1668→        Parameters
  1669→  1669→        ----------
  1670→  1670→        target :
  1671→  1671→            The target being analysed.
  1672→  1672→        findings :
  1673→  1673→            List of finding dicts to analyse.
  1674→  1674→        llm_client :
  1675→  1675→            Optional pre-configured LLM client.  If ``None``, the method
  1676→  1676→            attempts to create one from environment variables
  1677→  1677→            (``OPENAI_API_KEY`` or ``ANTHROPIC_API_KEY``).  Falls back to
  1678→  1678→            rule-based analysis if no LLM is available.
  1679→  1679→
  1680→  1680→        Returns
  1681→  1681→        -------
  1682→  1682→        List of :class:`ChainReport` — LLM chains first, then rule-based.
  1683→  1683→        """
  1684→  1684→        rule_reports = self.find_chains(target, findings)
  1685→  1685→
  1686→  1686→        if not findings:
  1687→  1687→            return rule_reports
  1688→  1688→
  1689→  1689→        llm_reports = self._call_llm(target, findings, llm_client)
  1690→  1690→        if llm_reports:
  1691→  1691→            # Merge: LLM chains first, then rule-based chains that aren't duplicates
  1692→  1692→            llm_names = {r.name.lower() for r in llm_reports}
  1693→  1693→            for rr in rule_reports:
  1694→  1694→                if rr.name.lower() not in llm_names:
  1695→  1695→                    llm_reports.append(rr)
  1696→  1696→            return llm_reports
  1697→  1697→
  1698→  1698→        return rule_reports

  1704→  1704→    def _call_llm(
  1705→  1705→        self,
  1706→  1706→        target: str,
  1707→  1707→        findings: List[Dict[str, Any]],
  1708→  1708→        llm_client: Any = None,
  1709→  1709→    ) -> List[ChainReport]:
  1710→  1710→        """Attempt to call an LLM and parse the response into ChainReport objects.
  1711→  1711→
  1712→  1712→        Returns an empty list on any failure (silently falls back to rules).
  1713→  1713→        """
  1714→  1714→        # Prepare findings for the prompt (truncate to avoid token limits)
  1715→  1715→        trimmed = []
  1716→  1716→        for f in findings[:30]:  # cap at 30 findings
  1717→  1717→            trimmed.append({
  1718→  1718→                "title": _finding_title(f),
  1719→  1719→                "severity": _normalise_severity(f.get("severity", "info")),
  1720→  1720→                "category": _finding_category(f),
  1721→  1721→                "description": str(f.get("description", ""))[:200],
  1722→  1722→            })
  1723→  1723→
  1724→  1724→        prompt = _LLIM_PROMPT.format(
  1725→  1725→            target=target,
  1726→  1726→            findings_json=json.dumps(trimmed, indent=2),
  1727→  1727→        )
  1728→  1728→
  1729→  1729→        response_text = ""
  1730→  1730→
  1731→  1731→        # Try OpenAI
  1732→  1732→        if llm_client is not None and hasattr(llm_client, "chat"):  # type: ignore[union-attr]
  1733→  1733→            try:
  1734→  1734→                resp = llm_client.chat.completions.create(  # type: ignore[union-attr]
  1735→  1735→                    model="gpt-4o-mini",
  1736→  1736→                    messages=[{"role": "user", "content": prompt}],
  1737→  1737→                    temperature=0.3,
  1738→  1738→                    max_tokens=2000,
  1739→  1739→                )
  1740→  1740→                response_text = resp.choices[0].message.content or ""
  1741→  1741→            except Exception:
  1742→  1742→                response_text = ""
  1743→  1743→        elif HAS_OPENAI:
  1744→  1744→            try:
  1745→  1745→                import os
  1746→  1746→                api_key = os.environ.get("OPENAI_API_KEY")
  1747→  1747→                if not api_key:
  1748→  1748→                    return []
  1749→  1749→                client = _openai_mod.OpenAI(api_key=api_key)
  1750→  1750→                resp = client.chat.completions.create(
  1751→  1751→                    model="gpt-4o-mini",
  1752→  1752→                    messages=[{"role": "user", "content": prompt}],
  1753→  1753→                    temperature=0.3,
  1754→  1754→                    max_tokens=2000,
  1755→  1755→                )
  1756→  1756→                response_text = resp.choices[0].message.content or ""
  1757→  1757→            except Exception:
  1758→  1758→                response_text = ""
  1759→  1759→
  1760→  1760→        # Try Anthropic if OpenAI didn't work
  1761→  1761→        if not response_text and HAS_ANTHROPIC:
  1762→  1762→            try:
  1763→  1763→                import os
  1764→  1764→                api_key = os.environ.get("ANTHROPIC_API_KEY")
  1765→  1765→                if not api_key:

  1769→  1769→                    model="claude-haiku-4-5-20241022",
  1770→  1770→                    max_tokens=2000,
  1771→  1771→                    messages=[{"role": "user", "content": prompt}],
  1772→  1772→                )
  1773→  1773→                response_text = resp.content[0].text if resp.content else ""
  1774→  1774→            except Exception:
  1775→  1775→                response_text = ""
  1776→  1776→
  1777→  1777→        if not response_text:
  1778→  1778→            return []
  1779→  1779→
  1780→  1780→        return self._parse_llm_response(response_text, findings)

  1782→  1782→    @staticmethod
  1783→  1783→    def _parse_llm_response(
  1784→  1784→        response_text: str,
  1785→  1785→        findings: List[Dict[str, Any]],
  1786→  1786→    ) -> List[ChainReport]:
  1787→  1787→        """Parse an LLM JSON response into ChainReport objects.
  1788→  1788→
  1789→  1789→        Gracefully handles malformed JSON, missing fields, and out-of-range indices.
  1790→  1790→        """
  1791→  1791→        reports: List[ChainReport] = []
  1792→  1792→
  1793→  1793→        # Try to extract JSON from the response (handle markdown code blocks)
  1794→  1794→        json_match = re.search(r"```(?:json)?\s*\n?(.*?)\n?```", response_text, re.DOTALL)
  1795→  1795→        if json_match:
  1796→  1796→            json_str = json_match.group(1).strip()
  1797→  1797→        else:
  1798→  1798→            # Try to find a top-level JSON object
  1799→  1799→            brace_match = re.search(r"\{.*\}", response_text, re.DOTALL)
  1800→  1800→            json_str = brace_match.group(0) if brace_match else response_text.strip()
  1801→  1801→
  1802→  1802→        try:
  1803→  1803→            data = json.loads(json_str)
  1804→  1804→        except (json.JSONDecodeError, AttributeError):
  1805→  1805→            return reports
  1806→  1806→
  1807→  1807→        chains = data.get("chains", [])
  1808→  1808→        if not isinstance(chains, list):
  1809→  1809→            return reports
  1810→  1810→
  1811→  1811→        for chain in chains:
  1812→  1812→            if not isinstance(chain, dict):
  1813→  1813→                continue
  1814→  1814→
  1815→  1815→            try:
  1816→  1816→                name = str(chain.get("name", "LLM-Discovered Chain"))
  1817→  1817→                severity = _normalise_severity(chain.get("severity", "medium"))
  1818→  1818→                confidence = float(chain.get("confidence", 0.6))
  1819→  1819→                confidence = max(0.0, min(1.0, confidence))
  1820→  1820→                narrative = str(chain.get("narrative", "LLM-identified compound attack path."))
  1821→  1821→                steps = chain.get("steps", [])
  1822→  1822→                if not isinstance(steps, list):
  1823→  1823→                    steps = []
  1824→  1824→                steps = [str(s) for s in steps]
  1825→  1825→                remediation = str(chain.get("remediation", "Address the identified vulnerability chain."))
  1826→  1826→
  1827→  1827→                # Resolve findings by index
  1828→  1828→                indices = chain.get("findings_indices", [])
  1829→  1829→                if not isinstance(indices, list):
  1830→  1830→                    indices = []
  1831→  1831→                matched_findings = []
  1832→  1832→                for idx in indices:
  1833→  1833→                    if isinstance(idx, (int, float)) and 0 <= int(idx) < len(findings):
  1834→  1834→                        matched_findings.append(findings[int(idx)])
  1835→  1835→
  1836→  1836→                if not matched_findings:
  1837→  1837→                    continue
  1838→  1838→
  1839→  1839→                score = max(_sev_score(f.get("severity", "info")) for f in matched_findings) * 1.4
  1840→  1840→
  1841→  1841→                reports.append(ChainReport(
  1842→  1842→                    name=name,
  1843→  1843→                    severity=severity,
  1844→  1844→                    score=round(score, 2),
  1845→  1845→                    findings=matched_findings,
  1846→  1846→                    narrative=narrative,
  1847→  1847→                    steps=steps if steps else [
  1848→  1848→                        f"Step {i+1}: Exploit {_finding_title(f)}"
  1849→  1849→                        for i, f in enumerate(matched_findings)
  1850→  1850→                    ],
  1851→  1851→                    remediation=remediation,
  1852→  1852→                    confidence=round(confidence, 2),
  1853→  1853→                ))
  1854→  1854→            except Exception:
  1855→  1855→                continue
  1856→  1856→
  1857→  1857→        return reports

  1883→  1883→def chains_to_findings(chains: List[ChainReport]) -> List[Dict[str, Any]]:

  1885→  1885→

  1893→  1893→        d["chain_findings"] = chain.findings
  1894→  1894→        d["chain_steps"] = chain.steps
  1895→  1895→        d["type"] = "attack_chain"

  1900→  1900→        d["severity"] = chain.severity

  1902→  1902→        result.append(d)
  1903→  1903→    return result
