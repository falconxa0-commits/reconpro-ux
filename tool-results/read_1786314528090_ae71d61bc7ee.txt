   701→def _check_whitespace_steganography(
   702→    body: str, content_type: str
   703→) -> List[Finding]:
   704→    """Cat 1: Detect anomalous whitespace patterns in text responses."""
   705→    findings: List[Finding] = []
   706→    if not body or content_type not in ("text/html", "application/json", "text/plain"):
   707→        return findings
   708→
   709→    lines = body.split("\n")
   710→    total_lines = len(lines)
   711→    if total_lines < 3:
   712→        return findings
   713→
   714→    # ── Zero-width character detection ──
   715→    zwc_counts: Dict[str, int] = Counter()
   716→    zwc_positions: Dict[str, List[int]] = {}
   717→    for idx, line in enumerate(lines):
   718→        for ch, info in ZERO_WIDTH_CHARS.items():
   719→            count = line.count(ch)
   720→            if count > 0:
   721→                zwc_counts[ch] += count
   722→                zwc_positions.setdefault(ch, []).append(idx)
   723→
   724→    if zwc_counts:
   725→        total_zwc = sum(zwc_counts.values())
   726→        zwc_ratio = total_zwc / max(len(body), 1)
   727→        details = []
   728→        for ch, count in zwc_counts.most_common(5):
   729→            info = ZERO_WIDTH_CHARS[ch]
   730→            details.append(
   731→                f"{info['unicode']} ({info['name']}): {count} occurrences "
   732→                f"on lines {zwc_positions[ch][:10]}"
   733→            )
   734→        evidence = "; ".join(details)
   735→        severity = "high" if any(c in HIGH_RISK_ZWC for c in zwc_counts) else "medium"
   736→        if zwc_ratio > 0.01:
   737→            severity = "critical"
   738→        findings.append(Finding(
   739→            title="Whitespace Steganography — Zero-Width Characters Detected",
   740→            severity=severity,
   741→            category="steganography-whitespace",
   742→            module="steganography_detector",
   743→            description=(
   744→                f"Response body contains {total_zwc} invisible Unicode zero-width characters "
   745→                f"(ratio: {zwc_ratio:.4f}). These characters are invisible to users but can "
   746→                f"encode binary data. This is a known steganographic technique (sig: WS-002)."
   747→            ),
   748→            evidence=evidence,
   749→            asset="",
   750→            points_deducted=15 if severity in ("critical", "high") else 8,
   751→            remediation=(
   752→                "Audit the application for steganography libraries or custom encoding logic. "
   753→                "Strip zero-width characters from user-generated content and responses."
   754→            ),
   755→            dread_score=0.7 if severity in ("critical", "high") else 0.4,
   756→        ))
   757→
   758→    # ── Trailing whitespace binary encoding ──
   759→    trailing_space_lines = 0
   760→    trailing_tab_lines = 0
   761→    trailing_mixed_lines = 0
   762→    for line in lines:
   763→        stripped = line.rstrip()
   764→        trailing = line[len(stripped):]
   765→        if not trailing:
   766→            continue
   767→        has_space = ' ' in trailing
   768→        has_tab = '\t' in trailing
   769→        if has_space and has_tab:
   770→            trailing_mixed_lines += 1
   771→        elif has_space:
   772→            trailing_space_lines += 1
   773→        elif has_tab:
   774→            trailing_tab_lines += 1
   775→
   776→    trailing_total = trailing_space_lines + trailing_tab_lines + trailing_mixed_lines
   777→    if trailing_total > total_lines * 0.3 and trailing_total > 5:
   778→        findings.append(Finding(
   779→            title="Whitespace Steganography — Trailing Whitespace Binary Encoding",
   780→            severity="medium",
   781→            category="steganography-whitespace",
   782→            module="steganography_detector",
   783→            description=(
   784→                f"{trailing_total}/{total_lines} lines have trailing whitespace "
   785→                f"(spaces: {trailing_space_lines}, tabs: {trailing_tab_lines}, "
   786→                f"mixed: {trailing_mixed_lines}). This pattern is consistent with "
   787→                f"trailing-whitespace binary encoding (sig: WS-001)."
   788→            ),
   789→            evidence=f"Trailing whitespace on {trailing_total} of {total_lines} lines",
   790→            asset="",
   791→            points_deducted=8,
   792→            remediation="Review content generation pipeline for whitespace steganography.",
   793→            dread_score=0.4,
   794→        ))
   795→
   796→    # ── CRLF/LF line ending steganography ──
   797→    crlf_count = len(_MIXED_LINE_ENDINGS_RE.findall(body))
   798→    if crlf_count > 0:
   799→        # Check if there's a mix of CRLF and bare LF
   800→        bare_lf_lines = body.count("\n") - crlf_count
   801→        if 0 < bare_lf_lines < len(lines) and crlf_count < len(lines):
   802→            mix_ratio = crlf_count / max(len(lines), 1)
   803→            if 0.1 < mix_ratio < 0.9:
   804→                findings.append(Finding(
   805→                    title="Whitespace Steganography — Mixed Line Ending Encoding",
   806→                    severity="low",
   807→                    category="steganography-whitespace",
   808→                    module="steganography_detector",
   809→                    description=(
   810→                        f"Response has mixed line endings: {crlf_count} CRLF, "
   811→                        f"{bare_lf_lines} bare LF out of {len(lines)} total lines. "
   812→                        f"This could encode binary data via CRLF/LF alternation (sig: WS-003)."
   813→                    ),
   814→                    evidence=f"CRLF: {crlf_count}, bare LF: {bare_lf_lines}",
   815→                    asset="",
   816→                    points_deducted=3,
   817→                    remediation="Normalize line endings in the response generation pipeline.",
   818→                    dread_score=0.2,
   819→                ))
   820→
   821→    return findings

   824→def _check_base64_anomalies(body: str) -> List[Finding]:
   825→    """Cat 2: Analyze base64-encoded content for anomalies."""
   826→    findings: List[Finding] = []
   827→    if not body:
   828→        return findings
   829→
   830→    # ── Find all base64-like strings ──
   831→    b64_candidates = set(_BASE64_RE.findall(body) + _BASE64_URL_RE.findall(body))
   832→
   833→    for candidate in b64_candidates:
   834→        if len(candidate) < 20:
   835→            continue
   836→
   837→        # Check for non-standard padding
   838→        pad_count = candidate.count("=")
   839→        expected_pad = len(candidate) % 4
   840→        if expected_pad != 0:
   841→            expected_pad = 4 - expected_pad
   842→        if pad_count > expected_pad and pad_count > 0:
   843→            findings.append(Finding(
   844→                title="Base64 Anomaly — Suspicious Padding",
   845→                severity="medium",
   846→                category="steganography-base64",
   847→                module="steganography_detector",
   848→                description=(
   849→                    f"Base64 string has {pad_count} padding characters but expects {expected_pad} "
   850→                    f"based on length {len(candidate)}. Extra padding can encode hidden data (sig: B64-003)."
   851→                ),
   852→                evidence=f"Candidate (first 80 chars): {candidate[:80]}...",
   853→                asset="",
   854→                points_deducted=6,
   855→                remediation="Review base64 encoding logic for correct padding implementation.",
   856→                dread_score=0.4,
   857→            ))
   858→
   859→        # Check character frequency distribution
   860→        freq = Counter(candidate.replace("=", ""))
   861→        if len(freq) >= 10:
   862→            counts = [freq.get(chr(c), 0) for c in range(65, 91)] + \
   863→                     [freq.get(chr(c), 0) for c in range(97, 123)] + \
   864→                     [freq.get(chr(c), 0) for c in range(48, 58)] + \
   865→                     [freq.get('+', 0), freq.get('/', 0)]
   866→            present_counts = [c for c in counts if c > 0]
   867→            if len(present_counts) >= 10:
   868→                chi2 = _chi_squared_uniform(present_counts)
   869→                # For truly random base64, chi2 should be moderate
   870→                # Very low chi2 = suspiciously uniform (encrypted/compressed data encoded as b64)
   871→                # Very high chi2 = non-random content encoded as b64
   872→                if chi2 > len(present_counts) * 3:
   873→                    decoded = _safe_b64_decode(candidate)
   874→                    decoded_preview = ""
   875→                    sensitive = []
   876→                    if decoded:
   877→                        try:
   878→                            decoded_text = decoded.decode("utf-8", errors="replace")
   879→                            decoded_preview = decoded_text[:200]
   880→                            sensitive = _contains_sensitive_keywords(decoded_text)
   881→                        except Exception:
   882→                            pass
   883→                    severity = "high" if sensitive else "medium"
   884→                    findings.append(Finding(
   885→                        title="Base64 Anomaly — Non-Uniform Character Distribution",
   886→                        severity=severity,
   887→                        category="steganography-base64",
   888→                        module="steganography_detector",
   889→                        description=(
   890→                            f"Base64 string (len={len(candidate)}) has highly non-uniform character "
   891→                            f"distribution (chi2={chi2:.1f}). This suggests the content is not random "
   892→                            f"base64 but may encode structured or cipher text (sig: B64-004)."
   893→                            + (f" Decoded content contains sensitive keywords: {sensitive}" if sensitive else "")
   894→                        ),
   895→                        evidence=f"Chi-squared: {chi2:.1f}; Decoded preview: {decoded_preview[:150]}",
   896→                        asset="",
   897→                        points_deducted=10 if severity == "high" else 6,
   898→                        remediation=(
   899→                            "Audit the source of this base64 content. "
   900→                            "Ensure it is legitimate and not used for data exfiltration."
   901→                        ),
   902→                        dread_score=0.6 if severity == "high" else 0.4,
   903→                    ))
   904→
   905→        # Check for nested base64
   906→        depth, final_decoded = _safe_b64_decode_recursive(candidate)
   907→        if depth >= 3:
   908→            sensitive = _contains_sensitive_keywords(final_decoded or "")
   909→            severity = "high" if sensitive else "medium"
   910→            findings.append(Finding(
   911→                title=f"Base64 Anomaly — Nested Encoding (Depth {depth})",
   912→                severity=severity,
   913→                category="steganography-base64",
   914→                module="steganography_detector",
   915→                description=(
   916→                    f"Base64 string is encoded {depth} layers deep. "
   917→                    f"Nested base64 encoding is a strong indicator of obfuscation (sig: B64-001)."
   918→                    + (f" Inner content contains sensitive keywords: {sensitive}" if sensitive else "")
   919→                ),
   920→                evidence=f"Depth: {depth}; Inner content (first 200 chars): {final_decoded[:200]}",
   921→                asset="",
   922→                points_deducted=12 if severity == "high" else 8,
   923→                remediation="Investigate the source of multi-layered base64 encoding.",
   924→                dread_score=0.6 if severity == "high" else 0.4,
   925→            ))
   926→
   927→    return findings

  1096→def _check_image_lsb(raw_body: bytes, content_type: str) -> List[Finding]:
  1097→    """Cat 4: Analyze image bytes for LSB steganography indicators."""
  1098→    findings: List[Finding] = []
  1099→    if not raw_body or len(raw_body) < 100:
  1100→        return findings
  1101→
  1102→    img_type = _detect_image_type(raw_body)
  1103→    if img_type is None and "image/" not in content_type:
  1104→        return findings
  1105→
  1106→    # Extract payload data (skip file headers)
  1107→    payload_start = 0
  1108→    if img_type == "jpeg":
  1109→        # Skip JPEG SOI + APP markers to find SOS (Start of Scan)
  1110→        payload_start = 2  # Skip SOI marker
  1111→        idx = 2
  1112→        while idx < len(raw_body) - 1:
  1113→            if raw_body[idx] != 0xFF:
  1114→                payload_start = idx
  1115→                break
  1116→            marker = raw_body[idx + 1]
  1117→            if marker == 0xDA:  # SOS marker
  1118→                payload_start = idx + 2
  1119→                break
  1120→            if marker == 0xD8 or marker == 0xD9:
  1121→                idx += 2
  1122→                continue
  1123→            if idx + 3 < len(raw_body):
  1124→                seg_len = struct.unpack(">H", raw_body[idx + 2:idx + 4])[0]
  1125→                idx += 2 + seg_len
  1126→            else:
  1127→                break
  1128→    elif img_type == "png":
  1129→        payload_start = 8  # Skip PNG signature
  1130→        # Skip IHDR chunk
  1131→        if len(raw_body) >= 33:
  1132→            ihdr_len = struct.unpack(">I", raw_body[8:12])[0]
  1133→            payload_start = 8 + 12 + ihdr_len  # length(4) + type(4) + data + crc(4)
  1134→    elif img_type == "gif":
  1135→        payload_start = 13  # Skip GIF header + LSD + GCT
  1136→        if len(raw_body) > 13:
  1137→            gct_flag = raw_body[10] & 0x80
  1138→            if gct_flag:
  1139→                gct_size = 3 * (2 ** ((raw_body[10] & 0x07) + 1))
  1140→                payload_start = 13 + gct_size
  1141→
  1142→    payload = raw_body[payload_start:payload_start + 65536]  # Analyze first 64KB
  1143→    if len(payload) < 256:
  1144→        return findings
  1145→
  1146→    # ── LSB plane analysis ──
  1147→    lsb_bits = [b & 1 for b in payload]
  1148→    ones_count = sum(lsb_bits)
  1149→    zeros_count = len(lsb_bits) - ones_count
  1150→
  1151→    # For natural images, LSBs are roughly 50/50 but not perfectly uniform
  1152→    # For LSB steganography, the distribution becomes suspiciously uniform
  1153→    total = len(lsb_bits)
  1154→    ratio = ones_count / total if total > 0 else 0
  1155→    expected = 0.5
  1156→    deviation = abs(ratio - expected)
  1157→
  1158→    # Chi-squared test on LSB distribution
  1159→    chi2 = ((ones_count - total * expected) ** 2 / (total * expected) +
  1160→            (zeros_count - total * expected) ** 2 / (total * expected))
  1161→
  1162→    # Also check byte value distribution for even/odd bias
  1163→    even_count = sum(1 for b in payload if b % 2 == 0)
  1164→    odd_count = len(payload) - even_count
  1165→    byte_ratio = even_count / len(payload) if payload else 0
  1166→    byte_deviation = abs(byte_ratio - 0.5)
  1167→
  1168→    # Check for unusual pairing (RS analysis approximation)
  1169→    # Count adjacent byte pairs with specific patterns
  1170→    smooth_pairs = 0  # Both bytes have similar LSB
  1171→    for i in range(0, len(payload) - 1, 2):
  1172→        if (payload[i] & 1) == (payload[i + 1] & 1):
  1173→            smooth_pairs += 1
  1174→    smooth_ratio = smooth_pairs / (len(payload) // 2) if len(payload) >= 2 else 0
  1175→
  1176→    # Scoring: natural images have chi2 around 0.1-4.0 for LSB
  1177→    # LSB-stego tends toward very low chi2 (approaching 0)
  1178→    is_suspicious = False
  1179→    reasons = []
  1180→
  1181→    if chi2 < 0.01 and total > 1000:
  1182→        is_suspicious = True
  1183→        reasons.append(f"LSB distribution is perfectly uniform (chi2={chi2:.6f})")
  1184→    elif deviation < 0.002 and total > 1000:
  1185→        is_suspicious = True

  1223→def _check_css_steganography(body: str, content_type: str) -> List[Finding]:
  1224→    """Cat 5: Detect encoded data in CSS."""
  1225→    findings: List[Finding] = []
  1226→    if not body:
  1227→        return findings
  1228→    is_css = ("text/css" in content_type or "<style" in body.lower() or
  1229→              body.lstrip().startswith("/*") or
  1230→              re.search(r'[.#][\w-]+\s*\{', body[:500]))

  1238→        body_without_comments = _CSS_COMMENT_RE.sub("", body)
  1239→        comment_ratio = total_comment_len / max(len(body), 1)

  1307→def _check_timing(target: str, base_url: str, timeout: int, verify_tls: bool) -> List[Finding]:
  1308→    """Cat 10: Timing channel detection."""
  1309→    findings: List[Finding] = []

  1311→    for _ in range(6):
  1312→        t0 = time.monotonic()
  1313→        http_probe(base_url, timeout=timeout, verify_tls=verify_tls, limiter=default_limiter)
  1314→        timings.append(time.monotonic() - t0)
  1315→    if len(timings) >= 4:
  1316→        mean_t = sum(timings) / len(timings)
  1317→        variance = sum((t - mean_t) ** 2 for t in timings) / len(timings)
  1318→        std_t = variance ** 0.5
  1319→        cv = (std_t / mean_t * 100) if mean_t > 0 else 0
  1320→        if cv > 80 and mean_t > 0.05:

  1323→                severity="high", category="steganography-timing",
  1324→                module="steganography_detector",
  1325→                description=f"Response timing shows high variability (CV={cv:.0f}%) "
  1326→                            f"across {len(timings)} requests. May encode binary data.",
  1327→                evidence=f"Timings: {[f'{t*1000:.0f}ms' for t in timings]}",
  1328→                asset=target, points_deducted=15,
  1329→                remediation="Investigate server-side timing behavior.",
  1330→                dread_score=0.8,
  1331→            ))

  1334→
  1335→def run_steganography_detector(
  1336→    target: str,
  1337→    base_url: str,
  1338→    timeout: int = 8,
  1339→    verify_tls: bool = True,
  1340→) -> List[Finding]:
  1341→    """Run all steganography detection checks against the target.
  1342→
  1343→    10 detection categories: whitespace, base64, headers, image LSB,
  1344→    CSS, JS variables, response size, charset, metadata, timing.
  1345→    """
  1346→    findings: List[Finding] = []
  1347→    host = target.replace("https://", "").replace("http://", "").split("/")[0]
  1348→
  1349→    resp = http_probe(base_url, timeout=timeout, verify_tls=verify_tls, limiter=default_limiter)
  1350→    body = resp.get("body", "")
  1351→    headers = resp.get("headers", {})
  1352→    ct = _get_content_type(headers)
  1353→
  1354→    # Get raw bytes for image analysis
  1355→    raw_body = b""
  1356→    try:
  1357→        import ssl as _ssl
  1358→        import urllib.request as _ur
  1359→        req = _ur.Request(base_url, headers={"User-Agent": "ReconPro/2.0"})
  1360→        ctx = _ssl.create_default_context() if verify_tls else _ssl._create_unverified_context()
  1361→        with _ur.urlopen(req, timeout=timeout, context=ctx) as r:
  1362→            raw_body = r.read(262144)
  1363→    except Exception:
  1364→        raw_body = body.encode("utf-8", errors="replace")
  1365→
  1366→    # All 10 detection categories
  1367→    findings.extend(_check_whitespace_steganography(body, ct, host))
  1368→    findings.extend(_check_base64_anomalies(body, host))
  1369→    findings.extend(_check_header_steganography(headers, host))
  1370→    findings.extend(_check_image_lsb(raw_body, ct))
  1371→    findings.extend(_check_css_steganography(body, ct, host))
  1372→    findings.extend(_check_js_variables(body, ct, host))
  1373→    findings.extend(_check_response_size(body, headers))
  1374→    findings.extend(_check_charset(body, headers, host))
  1375→    findings.extend(_check_metadata(raw_body, ct, host))
  1376→    findings.extend(_check_timing(host, base_url, timeout, verify_tls))
  1377→
  1378→    for f in findings:
  1379→        if not f.asset:
  1380→            f.asset = host
  1381→
  1382→    return findings
