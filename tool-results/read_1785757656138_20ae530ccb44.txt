   871→  1220→        self,
   872→  1221→        target: str,
   873→  1222→        modules: Optional[List[str]] = None,
   874→  1223→        is_local: bool = False,

   879→  1228→        # Set modules to scanning

   884→  1233→        else:

   886→  1235→                list(MODULE_NAMES.keys())
   887→  1236→                if is_local
   888→  1237→                else [k for k in MODULE_NAMES if k not in ("host", "dev", "doctor")]
   889→  1238→            )

   893→  1242→

   895→  1244→            from .scanner import scan, audit_scan
   896→  1245→            from .history import save_scan

   902→  1251→            if is_local:

   916→  1265→
   917→  1266→        except Exception as e:

   922→  1271→                        self.call_from_thread(self._module_cells[m].set_status, "error")
   923→  1272→            else:
   924→  1273→                self.call_from_thread(self._set_all_modules_status, "error")
   925→  1274→
   926→  1275→        finally:
