  1090→        indent: str,

  1094→        """Replace mktemp() with mkstemp() or NamedTemporaryFile()."""

  1099→                    r"mktemp\s*\([^)]*\)",
  1100→                    "NamedTemporaryFile(delete=False).name",
  1101→                    line,

  1105→                    f"{indent}from tempfile import NamedTemporaryFile  # noqa: safe replacement\n"

  1218→    def visit_Call(self, node: ast.Call) -> None:
  1219→        fname = _get_call_name(node)
  1220→        func_full = _get_call_full_name(node)
  1221→
  1222→        if self.category == "dangerous_eval" and fname == "eval":
  1223→            self.matched_nodes.append(node)
  1224→        elif self.category == "dangerous_exec" and fname == "exec":
  1225→            self.matched_nodes.append(node)
  1226→        elif self.category == "command_injection":
  1227→            if fname in ("run", "call", "Popen", "check_output", "check_call"):
  1228→                for kw in node.keywords:
  1229→                    if kw.arg == "shell" and isinstance(kw.value, ast.Constant) and kw.value.value is True:
  1230→                        self.matched_nodes.append(node)
  1231→                        break
  1232→        elif self.category == "unsafe_deserialization":
  1233→            if func_full and ("pickle.loads" in func_full or "marshal.loads" in func_full):
  1234→                self.matched_nodes.append(node)
  1235→            elif func_full and "yaml.load" in func_full:
  1236→                self.matched_nodes.append(node)
  1237→        elif self.category == "jwt_no_verify":
  1238→            if func_full and "jwt.decode" in func_full:
  1239→                self.matched_nodes.append(node)
  1240→        elif self.category == "temp_file_race":
  1241→            if func_full and "mktemp" in func_full:
  1242→                self.matched_nodes.append(node)
  1243→
  1244→        self.generic_visit(node)

  1246→    def visit_Assign(self, node: ast.Assign) -> None:
  1247→        if self.category == "hardcoded_secret":
  1248→            # Get the source text for the line
  1249→            start = getattr(node, "lineno", 0)
  1250→            end = getattr(node, "end_lineno", start)

  1256→                                    "api_key", "apikey", "private_key")

  1259→                        if isinstance(node.value, (ast.Constant,)) and isinstance(node.value.value, str):

  1304→def _get_call_name(node: ast.Call) -> str:
  1305→    """Get the simple function name from a Call node."""
  1306→    func = node.func
  1307→    if isinstance(func, ast.Name):
  1308→        return func.id
  1309→    if isinstance(func, ast.Attribute):
  1310→        return func.attr
  1311→    return ""

  1313→
  1314→def _get_call_full_name(node: ast.Call) -> str:
  1315→    """Get the full dotted function name from a Call node."""
  1316→    func = node.func
  1317→    if isinstance(func, ast.Name):
  1318→        return func.id
  1319→    if isinstance(func, ast.Attribute):
  1320→        parts = [func.attr]
  1321→        current = func.value
  1322→        while isinstance(current, ast.Attribute):
  1323→            parts.append(current.attr)
  1324→            current = current.value
  1325→        if isinstance(current, ast.Name):
  1326→            parts.append(current.id)
  1327→        parts.reverse()
  1328→        return ".".join(parts)
  1329→    return ""

  1345→        self._path = store_path or (MEMORY_DIR / "fix_proposals.json")

  1352→    def _ensure_dir(self) -> None:
  1353→        """Ensure the storage directory exists."""
  1354→        try:
  1355→            self._path.parent.mkdir(parents=True, exist_ok=True)
  1356→        except OSError as exc:
  1357→            logger.error("Cannot create fix store directory %s: %s", self._path.parent, exc)

  1359→    def _load(self) -> None:
  1360→        """Load proposals from disk."""
  1361→        if not self._path.exists():
  1362→            self._proposals = {}
  1363→            self._history = []
  1364→            return
  1365→        try:
  1366→            raw = self._path.read_text(encoding="utf-8")
  1367→            data = json.loads(raw)
  1368→            proposals_data = data.get("proposals", {})
  1369→            self._proposals = {
  1370→                pid: FixProposal.from_dict(pdata)
  1371→                for pid, pdata in proposals_data.items()
  1372→            }
  1373→            self._history = data.get("history", [])
  1374→        except (json.JSONDecodeError, OSError, KeyError) as exc:
  1375→            logger.warning("Failed to load fix store from %s: %s", self._path, exc)
  1376→            self._proposals = {}
  1377→            self._history = []
  1378→
  1379→    def _save(self) -> None:
  1380→        """Persist proposals to disk."""
  1381→        self._ensure_dir()
  1382→        try:
  1383→            data = {
  1384→                "proposals": {
  1385→                    pid: p.to_dict() for pid, p in self._proposals.items()
  1386→                },
  1387→                "history": self._history[-500:],  # Keep last 500 history entries
  1388→                "updated_at": datetime.now(timezone.utc).isoformat(),
  1389→            }
  1390→            tmp_path = self._path.with_suffix(".tmp")
  1391→            tmp_path.write_text(json.dumps(data, indent=2, ensure_ascii=False), encoding="utf-8")
  1392→            tmp_path.replace(self._path)
  1393→        except OSError as exc:
  1394→            logger.error("Failed to save fix store to %s: %s", self._path, exc)

  1522→class AutoFixEngine:
  1523→    """Top-level engine that analyses scan results and proposes automated fixes.
  1524→
  1525→    The engine is **dry-run by default** — it only *proposes* fixes.
  1526→    Applying a fix requires explicit approval via :meth:`apply_fix` with
  1527→    ``dry_run=False``.

  1529→    Usage::
  1530→
  1531→        engine = AutoFixEngine()
  1532→        proposals = engine.propose_fixes(scan_result)
  1533→        for p in proposals:
  1534→            print(f"[{p.severity}] {p.title} (confidence={p.confidence})")
  1535→

  1537→        result = engine.apply_fix(proposals[0], dry_run=False)
  1538→        print(result)
  1539→    """
  1540→
  1541→    def __init__(
  1542→        self,
  1543→        store: Optional[FixStore] = None,
  1544→        dry_run: bool = True,
  1545→    ) -> None:
  1546→        self._analyzer = FixAnalyzer()
  1547→        self._store = store or FixStore()
  1548→        self._dry_run = dry_run
  1549→        logger.info("AutoFixEngine initialised (dry_run=%s)", dry_run)

  1551→    # ── Public API ────────────────────────────────────────────────────
  1552→
  1553→    def analyze_issue(self, issue: Dict[str, Any]) -> Optional[FixProposal]:
  1554→        """Analyse a single issue and return a fix proposal."""
  1555→        try:
  1556→            proposal = self._analyzer.analyze_issue(issue)
  1557→            if proposal is not None:
  1558→                self._store.add(proposal)
  1559→                logger.info(
  1560→                    "Proposed fix %s: %s (confidence=%.2f)",
  1561→                    proposal.id, proposal.title, proposal.confidence,
  1562→                )
  1563→            return proposal
  1564→        except Exception as exc:
  1565→            logger.error("Failed to analyze issue: %s", exc, exc_info=True)
  1566→            return None

  1568→    def propose_fixes(self, scan_result: Dict[str, Any]) -> List[FixProposal]:
  1569→        """Analyse a full scan result and propose fixes for all findings.
  1570→
  1571→        The scan_result dict can contain:
  1572→        - ``findings``: List of finding dicts (from http_layer.Finding.to_dict())
  1573→        - ``diagnostics``: Dict from diagnostics.run_diagnostics()
  1574→        - ``config_issues``: List from diagnostics.validate_config()
  1575→        - ``cross_validation``: Dict from cross_validator
  1576→
  1577→        Any combination of the above keys is supported.
  1578→        """
  1579→        proposals: List[FixProposal] = []
  1580→
  1581→        # 1. Process scan findings
  1582→        findings = scan_result.get("findings", [])
  1583→        if findings:
  1584→            proposals.extend(self._process_findings(findings))
  1585→
  1586→        # 2. Process diagnostics failures
  1587→        diagnostics = scan_result.get("diagnostics", {})
  1588→        if diagnostics:
  1589→            proposals.extend(self._process_diagnostics(diagnostics))
  1590→
  1591→        # 3. Process config issues
  1592→        config_issues = scan_result.get("config_issues", [])
  1593→        if config_issues:

  1641→    def apply_fix(
  1642→        self,
  1643→        fix_proposal: FixProposal,
  1644→        dry_run: Optional[bool] = None,
  1645→    ) -> Dict[str, Any]:
  1646→        """Apply an approved fix.
  1647→
  1648→        Args:
  1649→            fix_proposal: The proposal to apply.
  1650→            dry_run: Override the engine's dry_run setting. Defaults to the
  1651→                    engine's setting. Pass False to actually apply.
  1652→
  1653→        Returns:
  1654→            Dict with keys: success, message, dry_run, proposal_id, changes.
  1655→        """
  1656→        is_dry = dry_run if dry_run is not None else self._dry_run
  1657→
  1658→        result: Dict[str, Any] = {
  1659→            "success": False,
  1660→            "message": "",
  1661→            "dry_run": is_dry,
  1662→            "proposal_id": fix_proposal.id,
  1663→            "changes": [],
  1664→        }
  1665→
  1666→        # Validate proposal state
  1667→        if fix_proposal.status == ProposalStatus.APPLIED.value:
  1668→            result["message"] = "Proposal has already been applied."
  1669→            return result
  1670→
  1671→        if fix_proposal.status != ProposalStatus.APPROVED.value and not is_dry:
  1672→            result["message"] = (
  1673→                f"Proposal must be in 'approved' status to apply. "
  1674→                f"Current status: {fix_proposal.status}. Use fix_store.approve(id) first."
  1675→            )
  1676→            return result
  1677→
  1678→        # Dry-run: just describe what would happen
  1679→        if is_dry:
  1680→            result["success"] = True
  1681→            result["message"] = self._describe_dry_run(fix_proposal)
  1682→            result["changes"].append({
  1683→                "type": "dry_run",
  1684→                "file": fix_proposal.affected_file,
  1685→                "lines": fix_proposal.affected_lines,
  1686→                "description": f"Would apply: {fix_proposal.title}",
  1687→            })
  1688→            logger.info("[DRY-RUN] Would apply fix %s: %s", fix_proposal.id, fix_proposal.title)
  1689→            return result
  1690→
  1691→        # Actual application
  1692→        try:
  1693→            applied = self._do_apply(fix_proposal)
  1694→            if applied:
  1695→                fix_proposal.status = ProposalStatus.APPLIED.value
  1696→                fix_proposal.applied_at = datetime.now(timezone.utc).isoformat()
  1697→                self._store.update(fix_proposal)
  1698→                result["success"] = True
  1699→                result["message"] = f"Fix applied successfully: {fix_proposal.title}"
  1700→                result["changes"].append({
  1701→                    "type": "applied",
  1702→                    "file": fix_proposal.affected_file,
  1703→                    "lines": fix_proposal.affected_lines,
  1704→                    "description": fix_proposal.title,
  1705→                })
  1706→                logger.info("Applied fix %s: %s", fix_proposal.id, fix_proposal.title)
  1707→            else:
  1708→                fix_proposal.status = ProposalStatus.FAILED.value
  1709→                fix_proposal.error_message = "No applicable transformation found"
  1710→                self._store.update(fix_proposal)
  1711→                result["message"] = "No applicable transformation found for this proposal."
  1712→        except Exception as exc:
  1713→            fix_proposal.status = ProposalStatus.FAILED.value
  1714→            fix_proposal.error_message = str(exc)
  1715→            self._store.update(fix_proposal)
  1716→            result["message"] = f"Failed to apply fix: {exc}"
  1717→            logger.error("Failed to apply fix %s: %s", fix_proposal.id, exc, exc_info=True)
  1718→
  1719→        return result

  1721→    def batch_fix(
  1722→        self,
  1723→        fix_proposals: List[FixProposal],
  1724→        dry_run: Optional[bool] = None,
  1725→    ) -> Dict[str, Any]:
  1726→        """Apply multiple fixes.
  1727→
  1728→        Args:
  1729→            fix_proposals: List of proposals to apply.
  1730→            dry_run: Override the engine's dry_run setting.
  1731→
  1732→        Returns:
  1733→            Dict with keys: total, succeeded, failed, skipped, results.
  1734→        """
  1735→        results: List[Dict[str, Any]] = []
  1736→        succeeded = 0
  1737→        failed = 0
  1738→        skipped = 0
  1739→
  1740→        for proposal in fix_proposals:
  1741→            # Skip non-automated in non-dry-run mode unless pre-approved
  1742→            is_dry = dry_run if dry_run is not None else self._dry_run
  1743→            if not is_dry and not proposal.automated and proposal.status != ProposalStatus.APPROVED.value:
  1744→                results.append({
  1745→                    "proposal_id": proposal.id,
  1746→                    "title": proposal.title,
  1747→                    "success": False,
  1748→                    "message": "Skipped: non-automated and not pre-approved",
  1749→                    "dry_run": is_dry,
  1750→                })
  1751→                skipped += 1
  1752→                continue
  1753→
  1754→            res = self.apply_fix(proposal, dry_run=dry_run)
  1755→            results.append(res)
  1756→            if res["success"]:
  1757→                succeeded += 1
  1758→            else:
  1759→                failed += 1
  1760→
  1761→        return {
  1762→            "total": len(fix_proposals),
  1763→            "succeeded": succeeded,
  1764→            "failed": failed,
  1765→            "skipped": skipped,
  1766→            "results": results,
  1767→        }

  1773→    def verify_fix(
  1774→        self,
  1775→        fix_proposal: FixProposal,
  1776→        post_state: Dict[str, Any],
  1777→    ) -> Dict[str, Any]:
  1778→        """Verify that a fix was effective.
  1779→
  1780→        Args:
  1781→            fix_proposal: The applied proposal to verify.
  1782→            post_state: The state after fix application (e.g., re-scan results).
  1783→
  1784→        Returns:
  1785→            Dict with keys: verified, details, remaining_issues.
  1786→        """
  1787→        result: Dict[str, Any] = {
  1788→            "verified": False,
  1789→            "proposal_id": fix_proposal.id,
  1790→            "details": "",
  1791→            "remaining_issues": [],
  1792→        }
  1793→
  1794→        if fix_proposal.status != ProposalStatus.APPLIED.value:
  1795→            result["details"] = f"Proposal not in applied status (current: {fix_proposal.status})"
  1796→            return result
  1797→
  1798→        # Check if the original issue is still present
  1799→        source_issue = fix_proposal.source_issue
  1800→        original_category = source_issue.get("category", "")
  1801→        original_title = source_issue.get("title", "")
  1802→        original_message = source_issue.get("message", "")
  1803→
  1804→        # Look for the issue in the post_state
  1805→        post_findings = post_state.get("findings", [])
  1806→        post_issues = post_state.get("issues", post_findings)
  1807→
  1808→        still_present = False
  1809→        for issue in post_issues:
  1810→            issue_cat = issue.get("category", "")
  1811→            issue_title = issue.get("title", "")
  1812→            issue_msg = issue.get("message", "")
  1813→
  1814→            if issue_cat == original_category:
  1815→                if issue_title == original_title or issue_msg == original_message:
  1816→                    still_present = True
  1817→                    result["remaining_issues"].append(issue)
  1818→
  1819→        if not still_present:
  1820→            fix_proposal.status = ProposalStatus.VERIFIED.value
  1821→            fix_proposal.verified_at = datetime.now(timezone.utc).isoformat()
  1822→            self._store.update(fix_proposal)
  1823→            result["verified"] = True
  1824→            result["details"] = f"Fix verified: {fix_proposal.title} — original issue no longer detected."
  1825→            logger.info("Verified fix %s: %s", fix_proposal.id, fix_proposal.title)
  1826→        else:
  1827→            result["details"] = (
  1828→                f"Fix NOT verified: {fix_proposal.title} — issue still detected. "
  1829→                f"{len(result['remaining_issues'])} remaining instance(s)."
  1830→            )
  1831→            logger.warning("Fix %s not verified: %s", fix_proposal.id, fix_proposal.title)
  1832→
  1833→        return result

  1933→    def _do_apply(self, proposal: FixProposal) -> bool:
  1934→        """Actually apply a fix to a file.
  1935→
  1936→        Creates a backup before modification. Returns True on success.
  1937→        """
  1938→        file_path = proposal.affected_file
  1939→        if not file_path:
  1940→            logger.warning("No file path in proposal %s — cannot apply", proposal.id)
  1941→            return False
  1942→
  1943→        path = Path(file_path)
  1944→        if not path.is_file():
  1945→            logger.warning("File not found: %s", file_path)
  1946→            return False
  1947→
  1948→        if not proposal.proposed_code:
  1949→            logger.warning("No proposed code in proposal %s — nothing to apply", proposal.id)
  1950→            return False
  1951→
  1952→        # Read current file
  1953→        try:
  1954→            original_content = path.read_text(encoding="utf-8")
  1955→        except OSError as exc:
  1956→            raise RuntimeError(f"Cannot read {file_path}: {exc}") from exc
  1957→
  1958→        # Create backup
  1959→        backup_path = path.with_suffix(path.suffix + ".bak")
  1960→        try:
  1961→            shutil.copy2(str(path), str(backup_path))
  1962→        except OSError as exc:
  1963→            raise RuntimeError(f"Cannot create backup at {backup_path}: {exc}") from exc
  1964→
  1965→        # Apply the fix
  1966→        try:
  1967→            lines = original_content.splitlines()
  1968→            affected = proposal.affected_lines
  1969→
  1970→            if affected:
  1971→                # Replace specific line ranges
  1972→                start = affected[0] - 1  # Convert to 0-based
  1973→                end = affected[-1]      # End stays 1-based for slice
  1974→
  1975→                proposed_lines = proposal.proposed_code.splitlines()
  1976→                new_lines = lines[:start] + proposed_lines + lines[end:]
  1977→                new_content = "\n".join(new_lines)
  1978→
  1979→                # Preserve original line ending
  1980→                if original_content.endswith("\n"):
  1981→                    new_content += "\n"
  1982→            else:
  1983→                # No line info — append proposed code as a comment
  1984→                new_content = original_content.rstrip("\n") + "\n"
  1985→                new_content += f"\n# AUTO-FIX ({proposal.id}): {proposal.title}\n"
  1986→                new_content += f"# {proposal.description}\n"
  1987→                new_content += f"# TODO: {proposal.proposed_code}\n"
  1988→
  1989→            # Write the modified file
  1990→            path.write_text(new_content, encoding="utf-8")
  1991→            logger.info("Applied fix %s to %s", proposal.id, file_path)
  1992→            return True
  1993→
  1994→        except Exception as exc:
  1995→ # Restore backup on failure
  1996→            try:
  1997→                shutil.copy2(str(backup_path), str(path))
  1998→            except OSError:
  1999→                logger.error("CRITICAL: Failed to restore backup for %s", file_path)
  2000→            raise RuntimeError(f"Failed to apply fix: {exc}") from exc
