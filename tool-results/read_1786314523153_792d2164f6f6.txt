   617→def _safe_b64_decode(s: str) -> Optional[bytes]:
   618→    """Attempt base64 decode, returning None on failure."""
   619→    try:
   620→        # Try standard base64
   621→        padded = s + "=" * (-len(s) % 4)
   622→        return base64.b64decode(padded, validate=True)
   623→    except Exception:
   624→        try:
   625→            # Try URL-safe base64
   626→            padded = s + "=" * (-len(s) % 4)
   627→            return base64.urlsafe_b64decode(padded)
   628→        except Exception:
   629→            return None

   632→def _safe_b64_decode_recursive(s: str, max_depth: int = 5) -> tuple[int, Optional[str]]:
   633→    """Recursively base64-decode a string. Returns (depth, final_decoded_str)."""
   634→    depth = 0
   635→    current = s.strip()
   636→    for _ in range(max_depth):
   637→        decoded = _safe_b64_decode(current)
   638→        if decoded is None:
   639→            break
   640→        try:
   641→            text = decoded.decode("utf-8")
   642→        except Exception:
   643→            break
   644→        if not text or not all(32 <= ord(c) < 127 or c in "\n\r\t" for c in text):
   645→            break
   646→        current = text
   647→        depth += 1
   648→    return depth, current

   651→def _chi_squared_uniform(observed_counts: List[int]) -> float:
   652→    """Compute chi-squared statistic against uniform distribution."""
   653→    if not observed_counts:
   654→        return 0.0
   655→    n = sum(observed_counts)
   656→    if n == 0:
   657→        return 0.0
   658→    expected = n / len(observed_counts)
   659→    return sum((o - expected) ** 2 / expected for o in observed_counts if expected > 0)
