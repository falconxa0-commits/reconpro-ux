     9→
    10→function mulberry32(seed: number) {
    11→  return function () {
    12→    let t = (seed += 0x6d2b79f5);
    13→    t = Math.imul(t ^ (t >>> 15), t | 1);
    14→    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    15→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
    16→  };
    17→}

    19→function hashCode(s: string): number {
    20→  let h = 0;
    21→  for (let i = 0; i < s.length; i++) {
    22→    h = (Math.imul(31, h) + s.charCodeAt(i)) | 0;
    23→  }
    24→  return Math.abs(h);
    25→}

   441→function pickSeverity(rng: () => number): Severity {
   442→  const r = rng() * 100;
   443→  let cum = 0;
   444→  for (const [sev, weight] of SEVERITY_WEIGHTS) {
   445→    cum += weight;
   446→    if (r < cum) return sev;
   447→  }
   448→  return 'low';
   449→}

   451→function pickIndustry(rng: () => number): Industry {
   452→  return INDUSTRIES[Math.floor(rng() * INDUSTRIES.length)];
   453→}
   454→
   455→function pickFindingType(rng: () => number): FindingType {
   456→  return FINDING_TYPES[Math.floor(rng() * FINDING_TYPES.length)];
   457→}
   458→
   459→function pickRegion(rng: () => number): Region {
   460→  return REGIONS[Math.floor(rng() * REGIONS.length)];
   461→}
   462→
   463→function pickEntity(rng: () => number, industry: Industry): string {
   464→  const entities = ENTITY_TEMPLATES[industry];
   465→  return entities[Math.floor(rng() * entities.length)];
   466→}
   467→
   468→function pickDescription(rng: () => number, ft: FindingType, entity: string): string {
   469→  const templates = FINDING_TEMPLATES[ft];
   470→  const tpl = templates[Math.floor(rng() * templates.length)];
   471→  return tpl(entity);
   472→}
   473→
   474→function pickAssetType(rng: () => number, ft: FindingType): string {
   475→  const types = ASSET_TYPES[ft];
   476→  return types[Math.floor(rng() * types.length)];
   477→}

   479→function generateIncident(rng: () => number, timestamp: Date, id: string): Incident {
   480→  const industry = pickIndustry(rng);
   481→  const findingType = pickFindingType(rng);
   482→  const severity = pickSeverity(rng);
   483→  const entity = pickEntity(rng, industry);
   484→  const region = pickRegion(rng);
   485→
   486→  return {
   487→    id,
   488→    timestamp: timestamp.toISOString(),
   489→    severity,
   490→    industry,
   491→    findingType,
   492→    anonymizedDescription: pickDescription(rng, findingType, entity),
   493→    assetType: pickAssetType(rng, findingType),
   494→    region,
   495→    verifiable: rng() > 0.25,
   496→  };
   497→}

   499→/**
   500→ * Generate incidents for a given time window.
   501→ * Uses minute-level seed so it's deterministic per-minute (no flicker on refresh).
   502→ * Average rate: ~4.2 incidents/hour weighted by business hours.
   503→ */
   504→function generateIncidents(
   505→  since: Date,
   506→  until: Date,
   507→  seedPrefix: string = 'wos',
   508→): Incident[] {
   509→  const incidents: Incident[] = [];
   510→  const minute = 60 * 1000;
   511→  let cursor = new Date(since);
   512→  cursor.setSeconds(0, 0);
   513→  cursor.setMinutes(cursor.getMinutes() + 1);
   514→
   515→  while (cursor < until) {
   516→    const minuteKey = `wos:${Math.floor(cursor.getTime() / minute)}`;
   517→    const seed = hashCode(minuteKey);
   518→    const rng = mulberry32(seed);
   519→
   520→    const hour = cursor.getUTCHours();
   521→    const hourWeight = HOUR_WEIGHTS[hour];
   522→    const expectedIncidents = 0.07 * hourWeight;
   523→
   524→    if (rng() < expectedIncidents) {
   525→      const id = `inc-${Math.floor(cursor.getTime() / minute)}-${Math.floor(rng() * 1000)}`;
   526→      const jitteredTs = new Date(cursor.getTime() + Math.floor(rng() * 60000));
   527→      incidents.push(generateIncident(rng, jitteredTs, id));
   528→    }
   529→
   530→    cursor = new Date(cursor.getTime() + minute);
   531→  }
   532→
   533→  return incidents;
   534→}

   536→// ── Stats Calculation ──────────────────────────────────────────────────
   537→
   538→function computeStats(weekIncidents: Incident[]): Stats {
   539→  const now = new Date();
   540→  const todayStart = new Date(now.getUTCFullYear(), now.getUTCMonth(), now.getUTCDate());
   541→
   542→  const todayIncidents = weekIncidents.filter(i => new Date(i.timestamp) >= todayStart);
   543→
   544→  const byIndustry = {} as Record<Industry, number>;
   545→  const byType = {} as Record<FindingType, number>;
   546→
   547→  for (const ind of INDUSTRIES) byIndustry[ind] = 0;
   548→  for (const ft of FINDING_TYPES) byType[ft] = 0;
   549→
   550→  for (const inc of weekIncidents) {
   551→    byIndustry[inc.industry]++;
   552→    byType[inc.findingType]++;
   553→  }
   554→
   555→  return {
   556→    totalToday: todayIncidents.length,
   557→    totalWeek: weekIncidents.length,
   558→    criticalThisWeek: weekIncidents.filter(i => i.severity === 'critical').length,
   559→    byIndustry,
   560→    byType,
   561→  };
   562→}

   564→// ── Route Handler ──────────────────────────────────────────────────────
   565→
   566→export async function GET(request: NextRequest) {
   567→  const { searchParams } = request.nextUrl;
   568→  const limitParam = searchParams.get('limit');
   569→  const severityFilter = searchParams.get('severity') as Severity | null;
   570→  const industryFilter = searchParams.get('industry') as Industry | null;
   571→
   572→  const limit = limitParam ? Math.min(Math.max(parseInt(limitParam, 10) || 50, 1), 200) : 50;
   573→
   574→  const now = new Date();
   575→  const weekAgo = new Date(now.getTime() - 7 * 24 * 60 * 60 * 1000);
   576→
   577→  const allIncidents = generateIncidents(weekAgo, now);
   578→
   579→  let filtered = allIncidents;
   580→  if (severityFilter && ['critical', 'high', 'medium', 'low'].includes(severityFilter)) {
   581→    filtered = filtered.filter(i => i.severity === severityFilter);
   582→  }
   583→  if (industryFilter && INDUSTRIES.includes(industryFilter)) {
   584→    filtered = filtered.filter(i => i.industry === industryFilter);
   585→  }
   586→
   587→  filtered.sort((a, b) => new Date(b.timestamp).getTime() - new Date(a.timestamp).getTime());
   588→
   589→  const incidents = filtered.slice(0, limit);
   590→  const stats = computeStats(allIncidents);
   591→
   592→  return NextResponse.json({ incidents, stats });
   593→}
