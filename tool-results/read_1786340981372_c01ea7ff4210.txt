   200→        console.print(BANNER)
   201→
   202→
   203→def _spinner_wrap(msg: str, fn, *a, _json_mode: bool | None = None, **kw):
   204→    """Run a function with a spinner (suppressed in JSON mode)."""
   205→    if _json_mode is None:
   206→        _json_mode = getattr(_cli_args, 'json_output', False) if _cli_args else False
   207→    if _json_mode:
   208→        return fn(*a, **kw)
   209→    with Progress(SpinnerColumn(), TextColumn("[bold]{task.description}[/]"),
   210→                   TimeElapsedColumn(), console=console) as progress:
   211→        task = progress.add_task(msg, total=None)
   212→        result = fn(*a, **kw)
   213→        progress.update(task, completed=True)
   214→    return result

   217→def _print_findings(findings, args, title: str = "SCAN") -> None:
   218→    """Print findings from a module run, respecting JSON mode."""
   219→    if getattr(args, 'json_output', False):
   220→        out = []
   221→        for f in findings:
   222→            out.append({
   223→                "title": f.title,
   224→                "severity": f.severity,
   225→                "category": f.category,
   226→                "module": f.module,
   227→                "description": f.description,
   228→                "evidence": f.evidence,
   229→                "asset": f.asset,
   230→                "points_deducted": f.points_deducted,
   231→                "remediation": f.remediation,
   232→            })
   233→        console.print_json(json.dumps(out, indent=2, default=str))
   234→    else:
   235→        if not findings:
   236→            console.print("  [dim]No findings.[/]")
   237→        else:
   238→            for f in findings:
   239→                c = SEV_COLORS.get(f.severity, "white")
   240→                console.print(f"  [{c}]{f.severity.upper():8}[/{c}]  {f.title}")
   241→                if f.evidence:
   242→                    console.print(f"       [dim]{f.evidence[:200]}[/]")
   243→            console.print(f"  [bold]{len(findings)} finding(s).[/]")
   244→    console.print()
