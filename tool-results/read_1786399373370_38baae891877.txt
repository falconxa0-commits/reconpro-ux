   206→def _shannon_entropy(data: str) -> float:
   207→    """Calculate Shannon entropy of a string in bits per character.
   208→
   209→    Higher entropy means more random/unpredictable data — a strong
   210→    indicator of encoded or encrypted payloads.
   211→
   212→    Formula: H(X) = -Σ p(x) * log2(p(x))
   213→    """
   214→    if not data:
   215→        return 0.0
   216→    freq: Dict[str, int] = {}
   217→    for ch in data:
   218→        freq[ch] = freq.get(ch, 0) + 1
   219→    length = len(data)
   220→    entropy = 0.0
   221→    for count in freq.values():
   222→        p = count / length
   223→        if p > 0:
   224→            entropy -= p * math.log2(p)
   225→    return entropy
   226→

   228→def _entropy_for_channel(channel_type: str, data: str) -> Tuple[float, bool]:
   229→    """Calculate entropy and compare against channel-specific threshold.
   230→
   231→    Returns (entropy_value, is_above_threshold).
   232→    """
   233→    sig = CHANNEL_SIGNATURES.get(channel_type, {})
   234→    threshold = sig.get("entropy_threshold", 3.5)
   235→    entropy = _shannon_entropy(data)
   236→    return entropy, entropy >= threshold

   239→def _byte_entropy(data: bytes) -> float:
   240→    """Shannon entropy over raw bytes (0-255 alphabet)."""
   241→    if not data:
   242→        return 0.0
   243→    freq = [0] * 256
   244→    for b in data:
   245→        freq[b] += 1
   246→    length = len(data)
   247→    entropy = 0.0
   248→    for count in freq:
   249→        if count > 0:
   250→            p = count / length
   251→            entropy -= p * math.log2(p)
   252→    return entropy
   253→

   255→def _index_of_coincidence(data: str) -> float:
   256→    """Calculate the Index of Coincidence (IoC) for a string.
   257→
   258→    English text: ~0.0667. Random: ~0.0385.
   259→    Low IoC with high entropy = strong indicator of encoding.
   260→    """
   261→    if len(data) < 2:
   262→        return 0.0
   263→    freq: Dict[str, int] = {}
   264→    for ch in data.lower():
   265→        if ch.isalnum():
   266→            freq[ch] = freq.get(ch, 0) + 1
   267→    n = sum(freq.values())
   268→    if n < 2:
   269→        return 0.0
   270→    ic = sum(c * (c - 1) for c in freq.values()) / (n * (n - 1))
   271→    return ic

   478→def _internet_checksum(data: bytes) -> int:
   479→    """Calculate the Internet checksum (RFC 1071)."""
   480→    if len(data) % 2:
   481→        data += b"\x00"
   482→    s = 0
   483→    for i in range(0, len(data), 2):
   484→        s += (data[i] << 8) + data[i + 1]
   485→    while s >> 16:
   486→        s = (s & 0xFFFF) + (s >> 16)
   487→    return ~s & 0xFFFF

   636→def _simulate_websocket_frame() -> Dict[str, Any]:
   637→    """Demonstrate WebSocket frame covert channel."""
   638→    secret = b"WS SECRET"
   639→    # WebSocket frame format: FIN(1) RSV(3) Opcode(4) MASK(1) LEN(7) [MASK_KEY(32)] PAYLOAD(...)
   640→    # Method 1: Frame length encodes data
   641→    frame_details = []
   642→    for byte_val in secret:
   643→        # FIN=1, RSV=000, Opcode=0001 (text), MASK=0, LEN=7 bits
   644→        # We encode the secret byte in the payload length (after a base offset)
   645→        payload_len = 128 + byte_val
   646→        # Frame header: 0x81 (FIN+text opcode) + payload_len
   647→        header = bytes([0x81, payload_len])
   648→        frame_details.append({
   649→            "secret_byte": chr(byte_val) if 32 <= byte_val < 127 else f"0x{byte_val:02x}",
   650→            "byte_hex": f"0x{byte_val:02x}",
   651→            "frame_header_hex": header.hex(),
   652→            "opcode": "0x1 (text)",
   653→            "payload_length": payload_len,
   654→            "covert_delta": byte_val,
   655→        })
   656→
   657→    # Method 2: Opcode switching
   658→    # Opcodes: 0x1=text, 0x2=binary, 0x8=close, 0x9=ping, 0xA=pong
   659→    # Encode 2 bits per frame via opcode selection
   660→    opcode_map = {
   661→        0x1: "00", 0x2: "01", 0x8: "10", 0x9: "11",
   662→    }
   663→    opcode_frames = []
   664→    binary_secret = "".join(f"{b:08b}" for b in secret)
   665→    for i in range(0, len(binary_secret), 2):
   666→        pair = binary_secret[i:i+2].ljust(2, "0")
   667→        for opcode, bits in opcode_map.items():
   668→            if bits == pair:
   669→                opcode_frames.append({
   670→                    "bits": pair,
   671→                    "opcode": f"0x{opcode:x}",
   672→                    "opcode_name": {0x1: "text", 0x2: "binary", 0x8: "close", 0x9: "ping"}[opcode],
   673→                })
   674→                break

   677→        "technique": "Covert data in WebSocket frame structure",
   678→        "methods": [
   679→            {
   680→                "name": "Frame Length Encoding",
   681→                "description": "Payload length = base_offset + secret_byte",
   682→                "base_offset": 128,
   683→                "bits_per_frame": 8,
   684→                "frame_details": frame_details,
   685→            },
   686→            {
   687→                "name": "Opcode Switching",
   688→                "description": "Frame opcode encodes 2 bits (text=00, binary=01, close=10, ping=11)",

   692→                "note": "Frequent opcode switching is anomalous in normal WebSocket traffic",

   906→def _detect_timing_channel(
   907→    target: str, base_url: str, timeout: int, verify_tls: bool
   908→) -> List[Finding]:
   909→    """Analyze response timing for timing-channel indicators.
   910→
   911→    Sends multiple requests and measures inter-response time deltas.
   912→    Consistent, non-natural timing patterns indicate encoding.
   913→    """
   914→    findings: List[Finding] = []
   915→    sig = CHANNEL_SIGNATURES["timing_channel"]
   916→
   917→    # Collect timing samples
   918→    num_samples = 10
   919→    timings: List[float] = []
   920→
   921→    for i in range(num_samples):
   922→        start = time.monotonic()
   923→        resp = http_probe(
   924→            f"{base_url}/?_t={int(time.time() * 1000)}_{i}",
   925→            timeout=timeout, verify_tls=verify_tls, limiter=default_limiter,
   926→        )
   927→        elapsed = time.monotonic() - start
   928→        timings.append(elapsed)
   929→
   930→    if len(timings) < sig["min_observations"]:
   931→        return findings
   932→
   933→    # Calculate timing statistics
   934→    deltas = [timings[i+1] - timings[i] for i in range(len(timings) - 1)]
   935→    mean_delta = sum(deltas) / len(deltas)
   936→    variance = sum((d - mean_delta) ** 2 for d in deltas) / len(deltas)
   937→    std_dev = math.sqrt(variance)
   938→
   939→    if std_dev == 0:
   940→        coeff_of_var = 0.0
   941→    else:
   942→        coeff_of_var = std_dev / mean_delta if mean_delta > 0 else float("inf")
   943→
   944→    # Very consistent timing is suspicious (indicates server-side encoding)
   945→    # Check if deltas cluster around specific values (discrete levels)
   946→    quantized_deltas = [round(d * 100) / 100 for d in deltas]  # quantize to 10ms
   947→    unique_deltas = set(quantized_deltas)
   948→
   949→    # If we see very few unique delta values relative to samples, that's suspicious
   950→    delta_diversity = len(unique_deltas) / len(quantized_deltas) if quantized_deltas else 0
   951→
   952→    is_suspicious = False
   953→    indicators = []
   954→
   955→    if coeff_of_var < sig["std_dev_threshold"] and mean_delta > 0.01:
   956→        is_suspicious = True
   957→        indicators.append(
   958→            f"Extremely consistent timing (CV={coeff_of_var:.4f}, "
   959→            f"mean={mean_delta*1000:.1f}ms, σ={std_dev*1000:.2f}ms)"
   960→        )
   961→
   962→    if delta_diversity < 0.4 and len(deltas) >= 5:
   963→        is_suspicious = True
   964→        indicators.append(
   965→            f"Low timing diversity ({len(unique_deltas)} unique values in "
   966→            f"{len(quantized_deltas)} deltas = {delta_diversity:.2f})"
   967→        )
   968→
   969→    if is_suspicious:
   970→        findings.append(Finding(
   971→            title="Suspicious Response Timing Patterns (Timing Channel Indicator)",
   972→            severity=sig["severity"],
   973→            category="covert_channel",
   974→            module="covert_channel",
   975→            description=(
   976→                f"Response timing analysis reveals patterns consistent with "
   977→                f"timing-based covert channels. {'; '.join(indicators)}. "
   978→                f"Timing channels encode data in inter-request/response delays."
   979→            ),
   980→            evidence=(
   981→                f"Timings (ms): {[f'{t*1000:.1f}' for t in timings]}\n"
   982→                f"Deltas (ms): {[f'{d*1000:.1f}' for d in deltas]}\n"
   983→                f"Mean delta: {mean_delta*1000:.1f}ms, StdDev: {std_dev*1000:.2f}ms, "
   984→                f"CV: {coeff_of_var:.4f}, Diversity: {delta_diversity:.2f}"
   985→            ),
   986→            asset=target,
   987→            points_deducted=12,
   988→            dread_score=sig["dread_score"],
   989→            remediation=(
   990→                "Implement request jitter/randomization. Monitor inter-request timing "
   991→                "distributions for anomalies. Use network flow analysis to detect "
   992→                "periodic or clustered timing patterns."
   993→            ),
   994→        ))
   995→    else:
   996→        findings.append(Finding(
   997→            title="Timing Channel Analysis (No Anomalies Detected)",
   998→            severity="info",
   999→            category="covert_channel",
  1000→            module="covert_channel",
  1001→            description=(
  1002→                f"Analyzed {num_samples} response timings. Mean delta: "
  1003→                f"{mean_delta*1000:.1f}ms, CV: {coeff_of_var:.4f}. "
  1004→                f"No suspicious patterns detected."
  1005→            ),
  1006→            evidence=f"CV: {coeff_of_var:.4f}, Diversity: {delta_diversity:.2f}",
  1007→            asset=target,
  1008→            points_deducted=0,
  1009→            dread_score=0.0,
