   765→    def on_key(self, event: Key) -> None:
   766→        """Handle key events — boot skip + completer + history + vim-nav."""
   767→        # Skip boot on ANY key press during boot sequence
   768→        if not self._boot_done and not self._boot_skipped:
   769→            self._boot_skipped = True
   770→            try:
   771→                boot = self.query_one("#boot-screen", BootScreen)
   772→                if boot._timer:
   773→                    boot._timer.stop()
   774→            except NoMatches:
   775→                pass
   776→            self._finish_boot()
   777→            return
   778→
   779→        # ── Number keys: quick jump to panels (0=input, 1=chat, 2=findings, 3=modules) ──
   780→        if event.key in ("0", "1", "2", "3") and not self._input_focused():
   781→            jump_map = {"0": "#command-input", "1": "#chat-log", "2": "#findings-feed", "3": "#module-grid"}
   782→            self._focus_widget(jump_map[event.key])
   783→            event.stop()
   784→            return
   785→
   786→        # ── Tab: completer accept/cycle / quick-pick on empty / cycle focus ──
   787→        if event.key == "tab":
   788→            try:
   789→                inp = self.query_one("#command-input", Input)
   790→                comp = self.query_one("#cmd-completer", CommandCompleter)
   791→                if inp.has_focus:
   792→                    # Completer visible → accept top suggestion
   793→                    if comp.visible:
   794→                        accepted = comp.accept_top()
   795→                        if accepted:
   796→                            self._apply_completion(inp, comp, accepted)
   797→                        event.stop()
   798→                        return
   799→                    # Empty input → quick-pick menu
   800→                    if not inp.value.strip():
   801→                        comp.show_quick_pick()
   802→                        event.stop()
   803→                        return
   804→            except NoMatches:
   805→                pass
   806→            # Fall through to default Tab behavior (cycle focus)
   807→            return
   808→
   809→        # ── Escape: dismiss completer, close modal, or jump to input ──
   810→        if event.key == "escape":
   811→            try:
   812→                comp = self.query_one("#cmd-completer", CommandCompleter)
   813→                if comp.visible:
   814→                    comp.hide()
   815→                    event.stop()
   816→                    return
   817→            except NoMatches:
   818→                pass
   819→            # If any modal screen is open, Escape will dismiss it via Textual
   820→            # Otherwise jump back to input
   821→            if not self._input_focused():
   822→                self._focus_input()
   823→                event.stop()
   824→                return
   825→
   826→        # ── Up/Down in completer: cycle suggestions ──
   827→        if event.key in ("up", "down"):
   828→            try:
   829→                comp = self.query_one("#cmd-completer", CommandCompleter)
   830→                inp = self.query_one("#command-input", Input)
   831→                if comp.visible and inp.has_focus:
   832→                    direction = -1 if event.key == "up" else 1
   833→                    comp.cycle_selection(direction)
   834→                    event.stop()
   835→                    return
   836→            except NoMatches:
   837→                pass
   838→
   839→        # ── j/k vim-style navigation: findings feed + chat log ──
   840→        if event.key in ("j", "k"):
   841→            # Findings feed navigation
   842→            try:
   843→                feed = self.query_one("#findings-feed", RichLog)
   844→                if feed.has_focus and self._findings_list:
   845→                    self._vim_navigate_findings(event.key)
   846→                    event.stop()
   847→                    return
   848→            except NoMatches:
   849→                pass
   850→            # Chat log scroll (j=down, k=up)
   851→            try:
   852→                chat = self.query_one("#chat-log", RichLog)
   853→                if chat.has_focus:
   854→                    if event.key == "j":
   855→                        chat.scroll_relative(3)
   856→                    else:
   857→                        chat.scroll_relative(-3)
   858→                    event.stop()
   859→                    return
   860→            except NoMatches:
   861→                pass
   862→
   863→        # ── Arrow keys on module grid: navigate cells ──
   864→        if event.key in ("up", "down", "left", "right"):
   865→            try:
   866→                grid = self.query_one("#module-grid", Container)
   867→                if grid.has_focus:
   868→                    self._arrow_navigate_modules(event.key)
   869→                    event.stop()
   870→                    return
   871→            except NoMatches:
   872→                pass
   873→
   874→        # ── Enter on findings: open detail modal ──
   875→        if event.key == "enter":
   876→            try:
   877→                feed = self.query_one("#findings-feed", RichLog)
   878→                if feed.has_focus and self._findings_list:
   879→                    idx = self._findings_cursor if 0 <= self._findings_cursor < len(self._findings_list) else len(self._findings_list) - 1
   880→                    self.push_screen(FindingDetailModal(self._findings_list[idx]))
   881→                    event.stop()
   882→                    return
   883→            except NoMatches:
   884→                pass
   885→
   886→        # Command history navigation
   887→        if event.key == "up" and self._history_index > 0:
   888→            try:
   889→                inp = self.query_one("#command-input", Input)
   890→                if inp.has_focus:
   891→                    self._history_index -= 1
   892→                    inp.value = self._command_history[self._history_index]
   893→            except NoMatches:
   894→                pass
   895→        elif event.key == "down":
   896→            try:
   897→                inp = self.query_one("#command-input", Input)
   898→                if inp.has_focus:
   899→                    if self._history_index < len(self._command_history) - 1:
   900→                        self._history_index += 1
   901→                        inp.value = self._command_history[self._history_index]
   902→                    else:
   903→                        self._history_index = len(self._command_history)
   904→                        inp.value = ""
   905→            except NoMatches:
   906→                pass

   969→    def _apply_completion(self, inp: Input, comp: CommandCompleter, accepted: str) -> None:
   970→        """Apply a completion suggestion to the input field.
   971→
   972→        Phase C: handles both command-level and arg-level completions.
   973→        For arg mode, appends/replaces the last word instead of the first.
   974→        """
   975→        parts = inp.value.split()
   976→
   977→        if comp._mode == "command":
   978→            # Replace the first word with the accepted command
   979→            rest = parts[1:] if len(parts) > 1 else []
   980→            inp.value = accepted + (" " + " ".join(rest) if rest else "")
   981→        else:
   982→            # Arg-level: replace the last word
   983→            if len(parts) >= 2:
   984→                parts[-1] = accepted
   985→                inp.value = " ".join(parts)
   986→            else:
   987→                inp.value = accepted
   988→
   989→        inp.cursor_position = len(inp.value)

  1669→    def _handle_scan(self, args: List[str]) -> None:
  1670→        if not args:
  1671→            self._chat(f"[{RED}]Usage: scan <target> [with <mod1> <mod2> ...][/]")
  1672→            return
  1673→
  1674→        # Parse "with" clause
  1675→        modules: Optional[List[str]] = None
  1676→        target_parts: List[str] = []
  1677→        found_with = False
  1678→        mod_parts: List[str] = []
  1679→        for a in args:
  1680→            if a.lower() == "with":
  1681→                found_with = True
  1682→                continue
  1683→            if found_with:
  1684→                mod_parts.append(a)
  1685→            else:
  1686→                target_parts.append(a)
  1687→
  1688→        target = " ".join(target_parts)
  1689→        if not target:
  1690→            self._chat(f"[{RED}]No target specified.[/]")
  1691→            return
  1692→
  1693→        if mod_parts:
  1694→            modules = [m.strip().lower() for m in mod_parts]
  1695→            self._chat(f"[{DIM_CYAN}]Scanning [cyan bold]{target}[/] with modules: {', '.join(modules)}[/]")
  1696→        else:
  1697→            self._chat(f"[{DIM_CYAN}]Scanning [cyan bold]{target}[/]...[/]")
  1698→
  1699→        self.current_target = target
  1700→        self._first_scan_done = True
  1701→        self._run_scan_worker(target, modules=modules)

  1875→    @work(exclusive=False, thread=True)
  1876→    def _run_scan_worker(
  1877→        self,
  1878→        target: str,
  1879→        modules: Optional[List[str]] = None,
  1880→        is_local: bool = False,
  1881→    ) -> None:
  1882→        """Run a scan in a background thread."""
  1883→        self.call_from_thread(self.is_scanning.set, True)
  1884→
  1885→        # Set modules to scanning + push module-specific hints
  1886→        if modules:
  1887→            for m in modules:
  1888→                if m in self._module_cells:
  1889→                    self.call_from_thread(self._module_cells[m].set_status, "scanning")
  1890→                    self.call_from_thread(self._push_module_scanning_hint, m)
  1891→        else:
  1892→            all_mods = (
  1893→                list(MODULE_NAMES.keys())
  1894→                if is_local
  1895→                else [k for k in MODULE_NAMES if k not in ("host", "dev", "doctor")]
  1896→            )
  1897→            for m in all_mods:
  1898→                if m in self._module_cells:
  1899→                    self.call_from_thread(self._module_cells[m].set_status, "scanning")
  1900→
  1901→        try:
  1902→            from .scanner import scan, audit_scan
  1903→            from .history import save_scan
  1904→
  1905→            self.call_from_thread(
  1906→                self._chat, f"[{DIM_CYAN}]  ⏳ Scan in progress...[/]"
  1907→            )
  1908→
  1909→            if is_local:
  1910→                result = audit_scan(target=target, modules=modules)
  1911→            else:
  1912→                result = scan(target=target, modules=modules)
  1913→
  1914→            # Process findings
  1915→            data = result.to_dict()
  1916→            self.call_from_thread(self._on_scan_complete, data, modules, is_local)
  1917→
  1918→            # Save to history
  1919→            try:
  1920→                save_scan(data, label="nexus")
  1921→            except Exception:
  1922→                pass
  1923→
  1924→        except Exception as e:
  1925→            self.call_from_thread(self._chat, f"[{RED}]  ✗ Scan error: {e}[/]")
  1926→            # Phase C: push error_state hints
  1927→            self.call_from_thread(self._push_error_hint)
  1928→            if modules:
  1929→                for m in modules:
  1930→                    if m in self._module_cells:
  1931→                        self.call_from_thread(self._module_cells[m].set_status, "error")
  1932→            else:
  1933→                self.call_from_thread(self._set_all_modules_status, "error")
  1934→
  1935→        finally:
  1936→            self.call_from_thread(self.is_scanning.set, False)

  1972→        label: str,
  1973→        modules: Optional[List[str]] = None,
  1974→        target: str = "",
  1975→        is_local: bool = False,
  1976→    ) -> None:
  1977→        """Generic background worker for Phase C extended commands.
  1978→
  1979→        Reuses the scan infrastructure for commands like passive, fuzzer,

  2004→
  2005→                data = result.to_dict()
  2006→                self.call_from_thread(self._process_scan_data, data, modules_list=modules)
  2007→            elif target and target not in ("defense", "compliance", "delta", "benchmark"):
  2008→                # Full scan with target
  2009→                from .scanner import scan, audit_scan
  2010→
  2011→                if is_local:
  2012→                    result = audit_scan(target=target)
  2013→                else:
  2014→                    result = scan(target=target)
  2015→
  2016→                data = result.to_dict()
  2017→                self.call_from_thread(self._process_scan_data, data, modules_list=None)
  2018→            else:
  2019→                # Non-scan operations (defense, compliance, delta, benchmark)
  2020→                self.call_from_thread(
  2021→                    self._chat,
  2022→                    f"[{GREEN}]  ✓ {label} operation complete.[/]",
  2023→                )
  2024→
  2025→        except ImportError:
  2026→            self.call_from_thread(
  2027→                self._chat,
  2028→                f"[{YELLOW}]  ⚠ {label} requires additional modules. Use [cyan]pip install reconpro[full][/][/]",
  2029→            )
  2030→        except Exception as e:
  2031→            self.call_from_thread(self._chat, f"[{RED}]  ✗ {label} error: {e}[/]")
  2032→            self.call_from_thread(self._push_error_hint)
  2033→            if modules:
  2034→                for m in modules:
  2035→                    if m in self._module_cells:
  2036→                        self.call_from_thread(self._module_cells[m].set_status, "error")
  2037→
  2038→        finally:
  2039→            self.call_from_thread(self.is_scanning.set, False)
  2040→            if modules:
  2041→                for m in modules:
  2042→                    if m in self._module_cells:

  2368→    def on_click(self, event: Click) -> None:
  2369→        """Open finding detail modal for the clicked finding.
  2370→
  2371→        Uses the vertical click position to determine which finding index
  2372→        the user clicked on, mapping it to _findings_list.
  2373→        """
  2374→        try:
  2375→            if (
  2376→                isinstance(event.widget, RichLog)
  2377→                and event.widget.id == "findings-feed"
  2378→                and self._findings_list
  2379→            ):
  2380→                # Get the line offset to determine which finding was clicked.
  2381→                # Each finding entry is 1 line in the RichLog.
  2382→                # We use the y coordinate relative to the widget to estimate the index.
  2383→                region = event.widget.region
  2384→                if region is not None:
  2385→                    # Calculate which line was clicked relative to the scroll position
  2386→                    scroll_offset = event.widget.scroll_y
  2387→                    click_y = event.y - region.y
  2388→                    # Each finding takes 1 line; estimate index from click position
  2389→                    # Account for scroll position
  2390→                    line_height = 1
  2391→                    if hasattr(event.widget, "_line_height"):
  2392→                        line_height = event.widget._line_height
  2393→                    clicked_index = int((click_y + scroll_offset * line_height) / line_height)
  2394→                    # Clamp to valid range
  2395→                    clicked_index = max(0, min(clicked_index, len(self._findings_list) - 1))
  2396→                    finding = self._findings_list[clicked_index]
  2397→                else:
  2398→                    finding = self._findings_list[-1]
  2399→                self.push_screen(FindingDetailModal(finding))
  2400→        except (NoMatches, Exception):
  2401→            pass
