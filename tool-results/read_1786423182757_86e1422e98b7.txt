    99→   118→   217→   466→    def analyze_drift_trend(self, target: str) -> Dict[str, Any]:
   100→   119→   218→   467→        """Analyze drift patterns over time.
   101→   120→   219→   468→
   102→   121→   220→   469→        Returns trend analysis: increasing, decreasing, cyclical,
   103→   122→   221→   470→        or stable, along with category breakdowns.
   104→   123→   222→   471→        """
   105→   124→   223→   472→        snapshots = self.load_snapshots(target)
   106→   125→   224→   473→        if len(snapshots) < 2:
   107→   126→   225→   474→            return {
   108→   127→   226→   475→                "target": target,
   109→   128→   227→   476→                "trend": "insufficient_data",
   110→   129→   228→   477→                "snapshot_count": len(snapshots),
   111→   130→   229→   478→                "category_breakdown": {},
   112→   131→   230→   479→                "details": [],
   113→   132→   231→   480→            }
   114→   133→   232→   481→
   115→   134→   233→   482→        all_events: List[DriftEvent] = []
   116→   135→   234→   483→        for i in range(len(snapshots) - 1):
   117→   136→   235→   484→            events = self.compare_snapshots(snapshots[i + 1], snapshots[i])
   118→   137→   236→   485→            all_events.extend(events)
   119→   138→   237→   486→
   120→   139→   238→   487→        # Category breakdown
   121→   140→   239→   488→        category_counts: Dict[str, int] = {}
   122→   141→   240→   489→        for ev in all_events:
   123→   142→   241→   490→            cat = ev.field_changed
   124→   143→   242→   491→            category_counts[cat] = category_counts.get(cat, 0) + 1
   125→   144→   243→   492→
   126→   145→   244→   493→        # Group events by snapshot pair to detect trends
   127→   146→   245→   494→        pairs: List[int] = []
   128→   147→   246→   495→        for i in range(len(snapshots) - 1):
   129→   148→   247→   496→            pair_events = self.compare_snapshots(
   130→   149→   248→   497→                snapshots[i + 1], snapshots[i]
   131→   150→   249→   498→            )
   132→   151→   250→   499→            pairs.append(len(pair_events))
   133→   152→   251→   500→
   134→   153→   252→   501→        # Determine trend
   135→   154→   253→   502→        trend = self._determine_trend(pairs)
   136→   155→   254→   503→
   137→   156→   255→   504→        # Time between changes
   138→   157→   256→   505→        change_intervals: List[str] = []
   139→   158→   257→   506→        for i in range(len(snapshots) - 1):
   140→   159→   258→   507→            events = self.compare_snapshots(snapshots[i + 1], snapshots[i])
   141→   160→   259→   508→            if events:
   142→   161→   260→   509→                try:
   143→   162→   261→   510→                    ts_a = datetime.fromisoformat(snapshots[i + 1].timestamp)
   144→   163→   262→   511→                    ts_b = datetime.fromisoformat(snapshots[i].timestamp)
   145→   164→   263→   512→                    delta = abs((ts_b - ts_a).total_seconds())
   146→   165→   264→   513→                    if delta < 3600:
   147→   166→   265→   514→                        interval = f"{int(delta // 60)} minutes"
   148→   167→   266→   515→                    elif delta < 86400:
   149→   168→   267→   516→                        interval = f"{delta / 3600:.1f} hours"
   150→   169→   268→   517→                    else:
   151→   170→   269→   518→                        interval = f"{delta / 86400:.1f} days"
   152→   171→   270→   519→                    change_intervals.append(interval)
   153→   172→   271→   520→                except (ValueError, TypeError):
   154→   173→   272→   521→                    pass
   155→   174→   273→   522→
   156→   175→   274→   523→        return {
   157→   176→   275→   524→            "target": target,
   158→   177→   276→   525→            "trend": trend,
   159→   178→   277→   526→            "snapshot_count": len(snapshots),
   160→   179→   278→   527→            "total_drift_events": len(all_events),
   161→   180→   279→   528→            "category_breakdown": category_counts,
   162→   181→   280→   529→            "events_per_snapshot_pair": pairs,
   163→   182→   281→   530→            "change_intervals": change_intervals,
   164→   183→   282→   531→            "risk_trend": self._risk_trend(snapshots),
   165→   184→   283→   532→            "details": [e.to_dict() for e in all_events],
   166→   185→   284→   533→        }

   444→   463→   562→   811→    @staticmethod
   445→   464→   563→   812→    def _determine_trend(pairs: List[int]) -> str:
   446→   465→   564→   813→        """Classify the drift trend from a list of event counts."""
   447→   466→   565→   814→        if len(pairs) < 2:
   448→   467→   566→   815→            return "stable"
   449→   468→   567→   816→
   450→   469→   568→   817→        first_half = pairs[: len(pairs) // 2]
   451→   470→   569→   818→        second_half = pairs[len(pairs) // 2 :]
   452→   471→   570→   819→        avg_first = sum(first_half) / len(first_half)
   453→   472→   571→   820→        avg_second = sum(second_half) / len(second_half)
   454→   473→   572→   821→
   455→   474→   573→   822→        if avg_first == 0 and avg_second == 0:
   456→   475→   574→   823→            return "stable"
   457→   476→   575→   824→        if avg_first == 0:
   458→   477→   576→   825→            return "increasing"
   459→   478→   577→   826→
   460→   479→   578→   827→        ratio = avg_second / avg_first
   461→   480→   579→   828→        if ratio > 1.5:
   462→   481→   580→   829→            return "increasing"
   463→   482→   581→   830→        elif ratio < 0.5:
   464→   483→   582→   831→            return "decreasing"
   465→   484→   583→   832→        else:
   466→   485→   584→   833→            # Check for cyclical pattern (alternating high/low)
   467→   486→   585→   834→            if len(pairs) >= 4:
   468→   487→   586→   835→                alternating = 0
   469→   488→   587→   836→                for i in range(1, len(pairs)):
   470→   489→   588→   837→                    if pairs[i] == pairs[i - 1]:
   471→   490→   589→   838→                        continue
   472→   491→   590→   839→                    if i == 1:
   473→   492→   591→   840→                        alternating += 1
   474→   493→   592→   841→                        continue
   475→   494→   593→   842→                    curr_up = pairs[i] > pairs[i - 1]
   476→   495→   594→   843→                    prev_up = pairs[i - 1] > pairs[i - 2]
   477→   496→   595→   844→                    if prev_up != curr_up:
   478→   497→   596→   845→                        alternating += 1
   479→   498→   597→   846→                if alternating > len(pairs) * 0.6:
   480→   499→   598→   847→                    return "cyclical"
   481→   500→   599→   848→            return "stable"

   483→   502→   601→   850→    def _risk_trend(self, snapshots: List[DriftSnapshot]) -> str:
   484→   503→   602→   851→        """Determine if overall risk is trending up or down."""
   485→   504→   603→   852→        if len(snapshots) < 3:
   486→   505→   604→   853→            return "insufficient_data"
   487→   506→   605→   854→
   488→   507→   606→   855→        scores: List[float] = []
   489→   508→   607→   856→        for i in range(len(snapshots) - 1):
   490→   509→   608→   857→            events = self.compare_snapshots(snapshots[i + 1], snapshots[i])
   491→   510→   609→   858→            scores.append(self._scorer.score_drift(events))
   492→   511→   610→   859→
   493→   512→   611→   860→        recent = scores[: max(1, len(scores) // 2)]
   494→   513→   612→   861→        older = scores[max(1, len(scores) // 2) :]
   495→   514→   613→   862→        avg_recent = sum(recent) / len(recent)
   496→   515→   614→   863→        avg_older = sum(older) / len(older) if older else 0.0
   497→   516→   615→   864→
   498→   517→   616→   865→        if avg_recent > avg_older * 1.3:
   499→   518→   617→   866→            return "increasing_risk"
   500→   519→   618→   867→        elif avg_recent < avg_older * 0.7:
   501→   520→   619→   868→            return "decreasing_risk"
   502→   521→   620→   869→        return "stable_risk"
