    90→// ── Mapping helpers ──────────────────────────────────────────────────────────
    91→
    92→function mapFrameworkStatus(
    93→  apiStatus: ApiFramework['status']
    94→): ComplianceFramework['status'] {
    95→  switch (apiStatus) {
    96→    case 'pass':
    97→      return 'Compliant';
    98→    case 'warn':
    99→    case 'needs_review':
   100→      return 'Needs Attention';
   101→    case 'fail':
   102→      return 'Non-Compliant';
   103→    default:
   104→      return 'Needs Attention';
   105→  }
   106→}
   107→
   108→function mapControlStatus(
   109→  apiStatus: ApiControl['status']
   110→): FrameworkControl['status'] {
   111→  switch (apiStatus) {
   112→    case 'pass':
   113→      return 'pass';
   114→    case 'fail':
   115→      return 'fail';
   116→    case 'warn':
   117→      return 'pending';
   118→    default:
   119→      return 'pending';
   120→  }
   121→}
   122→
   123→function mapApiToFramework(api: ApiFramework): ComplianceFramework {
   124→  return {
   125→    id: api.id,
   126→    name: api.name,
   127→    icon: api.icon,
   128→    score: api.score,
   129→    status: mapFrameworkStatus(api.status),
   130→    controlsPassed: api.controlsPassed,
   131→    controlsTotal: api.controlsTotal,
   132→    lastAssessed: api.lastAssessed,
   133→    controls: api.controls.map((c) => ({
   134→      id: c.id,
   135→      description: c.name,
   136→      status: mapControlStatus(c.status),
   137→      category: c.category,
   138→      evidenceLink: c.evidence && c.evidence !== 'No relevant findings detected' && !c.evidence.startsWith('Not assessable') ? c.evidence : undefined,
   139→    })),
   140→  };
   141→}

   145→function getScoreColor(score: number): string {
   146→  if (score >= 90) return '#00ff88';
   147→  if (score >= 75) return '#d29922';
   148→  return '#ff3355';

   151→function getScoreGradient(score: number): string {
   152→  if (score >= 90) return 'from-[#00ff88]/20 to-[#00ff88]/5';
   153→  if (score >= 75) return 'from-[#d29922]/20 to-[#d29922]/5';
   154→  return 'from-[#ff3355]/20 to-[#ff3355]/5';
   155→}

   170→// ── Circular Gauge SVG ──────────────────────────────────────────────────────
   171→
   172→function ComplianceGauge({ score }: { score: number }) {
   173→  const radius = 70;
   174→  const circumference = 2 * Math.PI * radius;
   175→  const offset = circumference - (score / 100) * circumference;
   176→  const color = getScoreColor(score);
   177→
   178→  return (
   179→    <div className="relative inline-flex items-center justify-center">
   180→      <svg width="180" height="180" viewBox="0 0 180 180" className="transform -rotate-90">
   181→        {/* Background circle */}
   182→        <circle cx="90" cy="90" r={radius} fill="none" stroke="rgba(255,255,255,0.06)" strokeWidth="10" />
   183→        {/* Progress arc */}
   184→        <motion.circle
   185→          cx="90"
   186→          cy="90"
   187→          r={radius}
   188→          fill="none"
   189→          stroke={color}
   190→          strokeWidth="10"
   191→          strokeLinecap="round"
   192→          strokeDasharray={circumference}
   193→          initial={{ strokeDashoffset: circumference }}
   194→          animate={{ strokeDashoffset: offset }}
   195→          transition={{ duration: 1.5, ease: 'easeOut' as const, delay: 0.3 }}
   196→          style={{
   197→            filter: `drop-shadow(0 0 8px ${color}50)`,
   198→          }}
   199→        />
   200→      </svg>
   201→      <div className="absolute flex flex-col items-center justify-center">
   202→        <motion.span
   203→          className="text-3xl font-black text-white"
   204→          initial={{ opacity: 0, scale: 0.5 }}
   205→          animate={{ opacity: 1, scale: 1 }}
   206→          transition={{ delay: 0.5, duration: 0.5 }}
   207→        >
   208→          {score}%
   209→        </motion.span>
   210→        <span className="text-[10px] uppercase tracking-widest text-[#444444] font-medium">Overall</span>
   211→      </div>
   212→    </div>
   213→  );
   214→}

   218→function TrendSparkline() {
   219→  const points = [62, 65, 68, 72, 70, 75, 78, 82, 85, 84, 87, 89];
   220→  const w = 200;
   221→  const h = 40;
   222→  const step = w / (points.length - 1);
   223→
   224→  const pathData = points
   225→    .map((p, i) => `${i === 0 ? 'M' : 'L'} ${i * step} ${h - (p / 100) * h}`)
   226→    .join(' ');
   227→
   228→  const areaData = `${pathData} L ${w} ${h} L 0 ${h} Z`;
   229→
   230→  return (
   231→    <svg width={w} height={h} viewBox={`0 0 ${w} ${h}`} className="overflow-visible">
   232→      <defs>
   233→        <linearGradient id="sparkGrad" x1="0" y1="0" x2="0" y2="1">
   234→          <stop offset="0%" stopColor="#00ff88" stopOpacity="0.3" />
   235→          <stop offset="100%" stopColor="#00ff88" stopOpacity="0" />
   236→        </linearGradient>
   237→      </defs>
   238→      <motion.path
   239→        d={areaData}
   240→        fill="url(#sparkGrad)"
   241→        initial={{ opacity: 0 }}
   242→        animate={{ opacity: 1 }}
   243→        transition={{ delay: 0.8, duration: 0.6 }}
   244→      />
   245→      <motion.path
   246→        d={pathData}
   247→        fill="none"
   248→        stroke="#00ff88"
   249→        strokeWidth="2"
   250→        strokeLinecap="round"
   251→        strokeLinejoin="round"
   252→        initial={{ pathLength: 0 }}
   253→        animate={{ pathLength: 1 }}
   254→        transition={{ delay: 0.5, duration: 1.2, ease: 'easeOut' as const }}
   255→      />
   256→      <motion.circle
   257→        cx={(points.length - 1) * step}
   258→        cy={h - (points[points.length - 1] / 100) * h}
   259→        r="3"
   260→        fill="#00ff88"
   261→        initial={{ opacity: 0, scale: 0 }}
   262→        animate={{ opacity: 1, scale: 1 }}
   263→        transition={{ delay: 1.5, duration: 0.3 }}
   264→      />
   265→    </svg>
   266→  );
   267→}

   374→export function CompliancePanel({ framework = 'all' }: CompliancePanelProps) {
   375→  const authHeaders = useAuthHeaders();
   376→  const [frameworks, setFrameworks] = useState<ComplianceFramework[]>([]);
   377→  const [overallScore, setOverallScore] = useState(0);
   378→  const [loading, setLoading] = useState(true);
   379→  const [regenerating, setRegenerating] = useState(false);
   380→  const [selectedFramework, setSelectedFramework] = useState(framework === 'all' ? 'soc2' : framework);
   381→  const [expandedControl, setExpandedControl] = useState<string | null>(null);
   382→  const [controlStates, setControlStates] = useState<Record<string, boolean>>({});
   383→  const selectedFrameworkRef = useRef(selectedFramework);
   384→  selectedFrameworkRef.current = selectedFramework;
   385→
   386→  const fetchCompliance = useCallback(async () => {
   387→    try {
   388→      const res = await fetch('/api/compliance', { headers: authHeaders });
   389→      if (!res.ok) throw new Error('Failed to fetch compliance data');
   390→      const data: ApiResponse = await res.json();
   391→      const mapped = data.frameworks.map(mapApiToFramework);
   392→      setFrameworks(mapped);
   393→      setOverallScore(data.overallScore);
   394→      // Auto-select the first framework if the selected one isn't available
   395→      const current = selectedFrameworkRef.current;
   396→      if (mapped.length > 0 && !mapped.find((f) => f.id === current)) {
   397→        setSelectedFramework(mapped[0].id);
   398→      }
   399→    } catch (err) {
   400→      console.error('Failed to load compliance data:', err);
   401→    } finally {
   402→      setLoading(false);
   403→      setRegenerating(false);
   404→    }
   405→  }, [authHeaders]);
   406→
   407→  useEffect(() => {
   408→    fetchCompliance();
   409→  }, [fetchCompliance]);
   410→
   411→  const handleRegenerate = () => {
   412→    setRegenerating(true);
   413→    fetchCompliance();
   414→  };
   415→
   416→  if (loading) {
   417→    return <LoadingSkeleton />;
   418→  }
   419→
   420→  if (frameworks.length === 0) {
   421→    return (
   422→      <div className="w-full flex flex-col items-center justify-center py-20 text-center">

   427→          <RefreshCw className={`w-3.5 h-3.5 ${regenerating ? 'animate-spin' : ''}`} />
   428→          Refresh
   429→        </Button>
   430→      </div>
   431→    );
   432→  }
   433→
   434→  const selected = frameworks.find((f) => f.id === selectedFramework) || frameworks[0];
   435→
   436→  const toggleControlStatus = (controlId: string) => {
   437→    setControlStates((prev) => ({ ...prev, [controlId]: !prev[controlId] }));
   438→  };
   439→
   440→  const passedCount = selected.controls.filter((c) => c.status === 'pass').length;
   441→  const failedCount = selected.controls.filter((c) => c.status === 'fail').length;
   442→  const controlPassRate = selected.controls.length > 0
   443→    ? Math.round((passedCount / selected.controls.length) * 100)
   444→    : 0;

   447→    <motion.div

   449→      initial="hidden"

   457→            <Shield className="w-5 h-5 text-[#00ff88]" />
   458→          </div>
   459→          <div>
   460→            <h2 className="text-xl font-bold text-white">Compliance Framework</h2>
