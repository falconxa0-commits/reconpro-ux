   577→   866→   866→  1218→    @work(exclusive=False, thread=True)
   578→   867→   867→  1219→    def _run_scan_worker(
   579→   868→   868→  1220→        self,
   580→   869→   869→  1221→        target: str,
   581→   870→   870→  1222→        modules: Optional[List[str]] = None,
   582→   871→   871→  1223→        is_local: bool = False,
   583→   872→   872→  1224→    ) -> None:
   584→   873→   873→  1225→        """Run a scan in a background thread."""
   585→   874→   874→  1226→        self.call_from_thread(self.is_scanning.set, True)
   586→   875→   875→  1227→
   587→   876→   876→  1228→        # Set modules to scanning
   588→   877→   877→  1229→        if modules:
   589→   878→   878→  1230→            for m in modules:
   590→   879→   879→  1231→                if m in self._module_cells:
   591→   880→   880→  1232→                    self.call_from_thread(self._module_cells[m].set_status, "scanning")
   592→   881→   881→  1233→        else:
   593→   882→   882→  1234→            all_mods = (
   594→   883→   883→  1235→                list(MODULE_NAMES.keys())
   595→   884→   884→  1236→                if is_local
   596→   885→   885→  1237→                else [k for k in MODULE_NAMES if k not in ("host", "dev", "doctor")]
   597→   886→   886→  1238→            )
   598→   887→   887→  1239→            for m in all_mods:
   599→   888→   888→  1240→                if m in self._module_cells:
   600→   889→   889→  1241→                    self.call_from_thread(self._module_cells[m].set_status, "scanning")
   601→   890→   890→  1242→
   602→   891→   891→  1243→        try:
   603→   892→   892→  1244→            from .scanner import scan, audit_scan
   604→   893→   893→  1245→            from .history import save_scan
   605→   894→   894→  1246→
   606→   895→   895→  1247→            self.call_from_thread(
   607→   896→   896→  1248→                self._chat, f"[{DIM_CYAN}]  ⏳ Scan in progress...[/]"
   608→   897→   897→  1249→            )
   609→   898→   898→  1250→
   610→   899→   899→  1251→            if is_local:
   611→   900→   900→  1252→                result = audit_scan(target=target, modules=modules)
   612→   901→   901→  1253→            else:
   613→   902→   902→  1254→                result = scan(target=target, modules=modules)
   614→   903→   903→  1255→
   615→   904→   904→  1256→            # Process findings
   616→   905→   905→  1257→            data = result.to_dict()
   617→   906→   906→  1258→            self.call_from_thread(self._on_scan_complete, data, modules, is_local)
   618→   907→   907→  1259→
   619→   908→   908→  1260→            # Save to history
   620→   909→   909→  1261→            try:
   621→   910→   910→  1262→                save_scan(data, label="nexus")
   622→   911→   911→  1263→            except Exception:
   623→   912→   912→  1264→                pass
   624→   913→   913→  1265→
   625→   914→   914→  1266→        except Exception as e:
   626→   915→   915→  1267→            self.call_from_thread(self._chat, f"[{RED}]  ✗ Scan error: {e}[/]")
   627→   916→   916→  1268→            if modules:
   628→   917→   917→  1269→                for m in modules:
   629→   918→   918→  1270→                    if m in self._module_cells:
   630→   919→   919→  1271→                        self.call_from_thread(self._module_cells[m].set_status, "error")
   631→   920→   920→  1272→            else:
   632→   921→   921→  1273→                self.call_from_thread(self._set_all_modules_status, "error")
   633→   922→   922→  1274→
   634→   923→   923→  1275→        finally:
   635→   924→   924→  1276→            self.call_from_thread(self.is_scanning.set, False)

   988→  1277→  1277→  1629→    def on_click(self, event: Click) -> None:
   989→  1278→  1278→  1630→        """Open finding detail modal for the clicked finding.
   990→  1279→  1279→  1631→
   991→  1280→  1280→  1632→        Uses the vertical click position to determine which finding index
   992→  1281→  1281→  1633→        the user clicked on, mapping it to _findings_list.
   993→  1282→  1282→  1634→        """
   994→  1283→  1283→  1635→        try:
   995→  1284→  1284→  1636→            if (
   996→  1285→  1285→  1637→                isinstance(event.widget, RichLog)

   998→  1287→  1287→  1639→                and self._findings_list
   999→  1288→  1288→  1640→            ):
  1000→  1289→  1289→  1641→                # Get the line offset to determine which finding was clicked.
  1001→  1290→  1290→  1642→                # Each finding entry is 1 line in the RichLog.
  1002→  1291→  1291→  1643→                # We use the y coordinate relative to the widget to estimate the index.
  1003→  1292→  1292→  1644→                region = event.widget.region
  1004→  1293→  1293→  1645→                if region is not None:

  1006→  1295→  1295→  1647→                    scroll_offset = event.widget.scroll_y
  1007→  1296→  1296→  1648→                    click_y = event.y - region.y
  1008→  1297→  1297→  1649→                    # Each finding takes 1 line; estimate index from click position
  1009→  1298→  1298→  1650→                    # Account for scroll position
  1010→  1299→  1299→  1651→                    line_height = 1
  1011→  1300→  1300→  1652→                    if hasattr(event.widget, "_line_height"):
  1012→  1301→  1301→  1653→                        line_height = event.widget._line_height
  1013→  1302→  1302→  1654→                    clicked_index = int((click_y + scroll_offset * line_height) / line_height)
  1014→  1303→  1303→  1655→                    # Clamp to valid range
  1015→  1304→  1304→  1656→                    clicked_index = max(0, min(clicked_index, len(self._findings_list) - 1))
  1016→  1305→  1305→  1657→                    finding = self._findings_list[clicked_index]
  1017→  1306→  1306→  1658→                else:
  1018→  1307→  1307→  1659→                    finding = self._findings_list[-1]
  1019→  1308→  1308→  1660→                self.push_screen(FindingDetailModal(finding))
  1020→  1309→  1309→  1661→        except (NoMatches, Exception):
