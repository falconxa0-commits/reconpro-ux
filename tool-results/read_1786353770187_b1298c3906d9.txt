   380→    "beaconing": ["T1071", "T1573"],
   381→    "steganography": ["T1001", "T1041"],
   382→    "supply_chain": ["T1585", "T1195.002"],
   383→    "cve": ["T1588.004"],
   384→}
   385→
   386→_CATEGORY_PATTERNS: List[Tuple[str, str, float]] = [
   387→    (r"\bsql\b|\bsqli\b", "sqli", 0.95),
   388→    (r"\bxss\b|cross.?site", "xss", 0.95),
   389→    (r"\bssrf\b", "ssrf", 0.95),
   390→    (r"\brce\b|remote.?code", "rce", 0.95),
   391→    (r"auth.*bypass|unauthorized", "auth_bypass", 0.90),
   392→    (r"privilege.*escal", "privilege_escalation", 0.90),
   393→    (r"credential.*theft|password.*leak|default.*cred", "credential_theft", 0.90),
   394→    (r"data.*expos|data.*leak", "data_exposure", 0.85),
   395→    (r"misconfigur|security.*header", "misconfiguration", 0.85),
   396→    (r"weak.*encrypt|crypto.*fail", "crypto_failure", 0.85),
   397→    (r"denial.*service|\bdos\b", "dos", 0.85),
   398→    (r"\bxxe\b", "xxe", 0.95),
   399→    (r"default.*password|default.*cred", "default_credentials", 0.90),
   400→    (r"open.?redirect", "open_redirect", 0.85),
   401→    (r"jwt.*expos|token.*leak", "jwt_exposure", 0.85),
   402→    (r"beacon|c2.*channel", "beaconing", 0.85),
   403→    (r"steganograph|hidden.*data", "steganography", 0.90),
   404→    (r"supply.?chain|dependency.*vuln", "supply_chain", 0.85),
   405→    (r"injection", "injection", 0.80),
   406→    (r"cve-\d{4}-\d+", "cve", 0.90),
   407→]

   409→
   410→def _quick_classify(finding: Dict[str, Any]) -> str:
   411→    """Quick classification of a finding."""
   412→    title = finding.get("title", "").lower()
   413→    desc = finding.get("description", "").lower()
   414→    evidence = finding.get("evidence", "").lower()
   415→    combined = f"{title} {desc} {evidence}"
   416→    best = "misc"
   417→    best_score = 0.0
   418→    for pattern, cat, weight in _CATEGORY_PATTERNS:
   419→        if re.search(pattern, combined, re.IGNORECASE):
   420→            if weight > best_score:
   421→                best_score = weight
   422→                best = cat
   423→    return best

   447→    def enrich_scan(self, findings: List[Dict[str, Any]], target: str = "") -> ThreatIntelReport:
   448→        """Enrich all findings from a scan with threat intelligence.
   449→
   450→        Parameters:
   451→            findings: List of finding dicts from ReconProResult
   452→            target: Target hostname
   453→
   454→        Returns:
   455→            ThreatIntelReport with enriched findings and summary.
   456→        """
   457→        t0 = time.monotonic()
   458→        report = ThreatIntelReport(target=target)
   459→
   460→        if not findings:
   461→            report.threat_summary = f"No findings to enrich for {target}."
   462→            report.analysis_duration_ms = round((time.monotonic() - t0) * 1000, 2)
   463→            return report
   464→
   465→        all_cves: set = set()
   466→        all_cwes: set = set()
   467→        all_capecs: set = set()
   468→        all_mitre: Dict[str, str] = {}
   469→        tech_counts: Dict[str, int] = defaultdict(int)
   470→        known_exploited = 0
   471→        active_exploited = 0
   472→        severity_trends: Dict[str, str] = {}
   473→
   474→        for f in findings:
   475→            enriched = self._enrich_finding(f)
   476→            report.enriched_findings.append(enriched.to_dict())
   477→
   478→            all_cves.update(enriched.cve_matches)
   479→            all_cwes.update(enriched.cwe_matches)
   480→            all_capecs.update(enriched.capec_matches)
   481→            for t in enriched.mitre_techniques:
   482→                all_mitre[t.get("technique", "")] = t.get("name", "")
   483→
   484→            for tech in enriched.affected_technologies:
   485→                tech_counts[tech] += 1
   486→
   487→            if enriched.known_exploitation:
   488→                known_exploited += 1
   489→            if enriched.active_exploitation:
   490→                active_exploited += 1
   491→
   492→            severity = f.get("severity", "info").lower()
   493→            severity_trends[f.get("title", "")[:50]] = enriched.severity_evolution
   494→
   495→        report.unique_cves = sorted(all_cves)
   496→        report.unique_cwes = sorted(all_cwes)
   497→        report.unique_capecs = sorted(all_capecs)
   498→        report.unique_mitre = [{"technique": k, "name": v} for k, v in sorted(all_mitre.items())]
   499→        report.known_exploited_count = known_exploited
   500→        report.active_exploitation_count = active_exploited
   501→        report.technology_profile = dict(tech_counts)
   502→        report.severity_trends = severity_trends
   503→        report.cache_hits = self._cache_hits
   504→        report.cache_misses = self._cache_misses
   505→        report.online_queries = self._online_queries
   506→
   507→        # Generate summary
   508→        report.threat_summary = self._generate_summary(
   509→            findings, all_cves, all_cwes, all_capecs,
   510→            known_exploited, active_exploited, target,
   511→        )
   512→
   513→        report.analysis_duration_ms = round((time.monotonic() - t0) * 1000, 2)
   514→        return report

   516→    def lookup_cve(self, cve_id: str) -> Optional[Dict[str, Any]]:
   517→        """Look up a CVE in the built-in database."""
   518→        cve_upper = cve_id.upper()
   519→        cached = self._cache_get(f"cve:{cve_upper}")
   520→        if cached is not None:
   521→            self._cache_hits += 1
   522→            return cached
   523→
   524→        self._cache_misses += 1
   525→        record = _CVE_KNOWN_PATTERNS.get(cve_upper)
   526→
   527→        if record:
   528→            result = {
   529→                "cve_id": cve_upper,
   530→                "name": record["name"],
   531→                "description": record["description"],
   532→                "severity": record["severity"],
   533→                "affected": record.get("affected", []),
   534→                "known_exploited": record.get("known_exploited", False),
   535→                "in_cisa_kev": cve_upper in _CISA_KEV_CVE_LIST,
   536→            }
   537→            self._cache_set(f"cve:{cve_upper}", result)
   538→            return result
   539→
   540→        # Try online lookup if enabled
   541→        if self._enable_online:
   542→            result = self._online_cve_lookup(cve_upper)
   543→            if result:
   544→                self._cache_set(f"cve:{cve_upper}", result)
   545→                return result
   546→
   547→        self._cache_set(f"cve:{cve_upper}", None)
   548→        return None

   636→    def _enrich_finding(self, finding: Dict[str, Any]) -> EnrichedFinding:
   637→        """Enrich a single finding with threat intelligence."""
   638→        cat = _quick_classify(finding)
   639→        enriched = EnrichedFinding(original=finding)
   640→
   641→        # Extract CVE references from evidence/title
   642→        cve_refs = self._extract_cve_refs(finding)
   643→        for cve in cve_refs:
   644→            enriched.cve_matches.append(cve)
   645→            record = self.lookup_cve(cve)
   646→            if record:
   647→                enriched.threat_records.append(ThreatRecord(
   648→                    source="CVE", source_id=cve,
   649→                    name=record.get("name", cve),
   650→                    description=record.get("description", ""),
   651→                    severity=record.get("severity", "info"),
   652→                    confidence=0.9,
   653→                    metadata=record,
   654→                    evidence_chain=[f"CVE reference found in finding: {cve}"],
   655→                ).to_dict())
   656→                if record.get("known_exploited"):
   657→                    enriched.known_exploitation = True
   658→                if record.get("in_cisa_kev"):
   659→                    enriched.active_exploitation = True
   660→                    enriched.severity_evolution = "increasing"
   661→
   662→        # CWE mapping
   663→        cwe_ids = _CATEGORY_TO_CWE.get(cat, [])
   664→        for cwe in cwe_ids:
   665→            enriched.cwe_matches.append(cwe)
   666→            record = self.lookup_cwe(cwe)
   667→            if record:
   668→                enriched.threat_records.append(ThreatRecord(
   669→                    source="CWE", source_id=cwe,
   670→                    name=record.get("name", cwe),
   671→                    description=record.get("description", ""),
   672→                    severity=record.get("risk", "info"),
   673→                    confidence=0.8,
   674→                    metadata=record,
   675→                    evidence_chain=[f"Category '{cat}' maps to {cwe}"],
   676→                ).to_dict())
   677→
   678→        # CAPEC mapping
   679→        capec_ids = _CATEGORY_TO_CAPEC.get(cat, [])
   680→        for capec in capec_ids:
   681→            enriched.capec_matches.append(capec)
   682→            record = self.lookup_capec(capec)
   683→            if record:
   684→                enriched.threat_records.append(ThreatRecord(
   685→                    source="CAPEC", source_id=capec,
   686→                    name=record.get("name", capec),
   687→                    description=record.get("description", ""),
   688→                    severity=record.get("risk", "info"),
   689→                    confidence=0.75,
   690→                    metadata=record,
   691→                    evidence_chain=[f"Category '{cat}' maps to {capec}"],
   692→                ).to_dict())
   693→
   694→        # MITRE mapping
   695→        mitre_ids = _CATEGORY_TO_MITRE.get(cat, [])
   696→        for tid in mitre_ids:
   697→            record = self.lookup_mitre(tid)
   698→            if record:
   699→                enriched.mitre_techniques.append({
   700→                    "technique": tid,
   701→                    "name": record.get("name", ""),
   702→                    "tactic": record.get("tactic", ""),
   703→                })
   704→                enriched.threat_records.append(ThreatRecord(
   705→                    source="MITRE", source_id=tid,
   706→                    name=record.get("name", ""),
   707→                    description=f"ATT&CK technique {tid}",
   708→                    severity=record.get("risk", "info"),
   709→                    confidence=0.8,
   710→                    metadata=record,
   711→                    evidence_chain=[f"Category '{cat}' maps to technique {tid}"],
   712→                ).to_dict())
   713→
   714→        # OWASP mapping
   715→        owasp_items = self.get_owasp_mapping(cat)
   716→        for item in owasp_items:
   717→            enriched.threat_records.append(ThreatRecord(
   718→                source="OWASP", source_id=item["owasp_id"],
   719→                name=item["name"],
   720→                description=f"OWASP Top 10 - {item['owasp_id']}: {item['name']}",
   721→                severity="medium",
   722→                confidence=0.7,
   723→                metadata=item,
   724→                evidence_chain=[f"Category '{cat}' maps to {item['owasp_id']}"],
   725→            ).to_dict())
   726→
   727→        # Determine affected technologies
   728→        enriched.affected_technologies = self._detect_technologies(finding)
   729→
   730→        # Severity evolution
   731→        if not enriched.severity_evolution:
   732→            sev = finding.get("severity", "info").lower()
   733→            if sev in ("critical", "high"):
   734→                enriched.severity_evolution = "stable"  # Already severe
   735→            else:
   736→                enriched.severity_evolution = "stable"
   737→
   738→        # Enrichment confidence
   739→        total_records = len(enriched.threat_records)
   740→        evidence_len = len(finding.get("evidence", ""))
   741→        enriched.enrichment_confidence = min(1.0,
   742→            (0.3 if total_records > 0 else 0.0) +
   743→            (0.3 if evidence_len > 50 else 0.1) +
   744→            (0.2 if enriched.cve_matches else 0.0) +
   745→            (0.2 if enriched.mitre_techniques else 0.0)
   746→        )
   747→
   748→        # Threat summary
   749→        enriched.threat_summary = self._finding_threat_summary(enriched, cat)
   750→
   751→        return enriched

   804→    def _finding_threat_summary(self, enriched: EnrichedFinding, cat: str) -> str:
   805→        """Generate a threat summary for a single finding."""
   806→        parts = [f"Finding maps to category '{cat}'."]
   807→        if enriched.cve_matches:
   808→            parts.append(f"References {len(enriched.cve_matches)} CVE(s): {', '.join(enriched.cve_matches[:3])}")
   809→        if enriched.known_exploitation:
   810→            parts.append("Known exploitation detected — CRITICAL")
   811→        if enriched.active_exploitation:
   812→            parts.append("Active exploitation reported in CISA KEV")
   813→        if enriched.mitre_techniques:
   814→            techniques = [t.get("name", "") for t in enriched.mitre_techniques[:3]]
   815→            parts.append(f"MITRE ATT&CK: {', '.join(techniques)}")
   816→        if enriched.cwe_matches:
   817→            parts.append(f"{len(enriched.cwe_matches)} CWE(s) mapped")
   818→        if enriched.affected_technologies:
   819→            parts.append(f"Affects: {', '.join(enriched.affected_technologies[:5])}")
   820→        return ". ".join(parts) + "."

   822→    def _generate_summary(self, findings: List[Dict[str, Any]],
   823→                           cves: set, cwes: set, capecs: set,
   824→                           known_exploited: int, active_exploited: int,
   825→                           target: str) -> str:
   826→        """Generate overall threat intelligence summary."""
   827→        parts = [
   828→            f"Threat intelligence analysis for {target} covered {len(findings)} findings.",
   829→        ]
   830→
   831→        if cves:
   832→            parts.append(f"Identified {len(cves)} unique CVE references: {', '.join(sorted(cves)[:5])}")
   833→        if cwes:
   834→            parts.append(f"Mapped to {len(cwes)} CWE categories")
   835→        if capecs:
   836→            parts.append(f"Covered {len(capecs)} CAPEC attack patterns")
   837→
   838→        if known_exploited > 0:
   839→            parts.append(f"WARNING: {known_exploited} findings reference known exploited vulnerabilities")
   840→        if active_exploited > 0:
   841→            parts.append(f"CRITICAL: {active_exploited} findings are in CISA Known Exploited Vulnerabilities catalog")
   842→
   843→        if not known_exploited and not active_exploited:
   844→            parts.append("No known exploited vulnerabilities detected")
   845→
   846→        return ". ".join(parts) + "."

   848→    # -- Cache methods -------------------------------------------------------
   849→
   850→    def _cache_get(self, key: str) -> Optional[Any]:
   851→        """Get cached value if not expired."""
   852→        with self._cache_lock:
   853→            if key in self._cache:
   854→                ts, val = self._cache[key]
   855→                if time.time() - ts < self._cache_ttl:
   856→                    return val
   857→                del self._cache[key]
   858→        return None
   859→
   860→    def _cache_set(self, key: str, value: Any) -> None:
   861→        """Set cache value with timestamp."""
   862→        with self._cache_lock:
   863→            self._cache[key] = (time.time(), value)
   864→
   865→    def clear_cache(self) -> None:
   866→        """Clear all cached entries."""
   867→        with self._cache_lock:
   868→            self._cache.clear()

   872→    def _online_cve_lookup(self, cve_id: str) -> Optional[Dict[str, Any]]:
   873→        """Attempt online CVE lookup via NVD API (stdlib HTTP only)."""
   874→        try:
   875→            url = f"https://services.nvd.nist.gov/rest/json/cves/2.0?cveId={cve_id}"
   876→            ctx = ssl.create_default_context()
   877→            req = urllib.request.Request(url, headers={
   878→                "User-Agent": "ReconPro/10.0 (Threat Intel Engine)",
   879→                "Accept": "application/json",
   880→            })
   881→            with urllib.request.urlopen(req, timeout=10, context=ctx) as resp:
