    31→  1031→    def prioritize(self, extended_findings: List[Dict[str, Any]]) -> List[Dict[str, Any]]:

    33→  1033→
    34→  1034→        Considers: severity, exploitability, business impact, blast radius, fix complexity.
    35→  1035→        Returns ordered list with priority_rank and reasoning.

    37→  1037→        if not extended_findings:
    38→  1038→            return []

    40→  1040→        scored: List[Tuple[float, int, Dict[str, Any]]] = []
    41→  1041→        for i, ef in enumerate(extended_findings):
    42→  1042→            priority = self._compute_priority(ef)
    43→  1043→            scored.append((priority, i, ef))
    44→  1044→

    46→  1046→
    47→  1047→        result: List[Dict[str, Any]] = []
    48→  1048→        for rank, (score, orig_idx, ef) in enumerate(scored, 1):
    49→  1049→            entry = dict(ef)
    50→  1050→            entry["priority_rank"] = rank
    51→  1051→            entry["priority_score"] = round(score, 2)

    53→  1053→            result.append(entry)
    54→  1054→
    55→  1055→        return result

    57→  1057→    def _compute_priority(self, ef: Dict[str, Any]) -> float:

    63→  1063→
    64→  1064→        sev_weight = _sev_val(severity) / 10.0
    65→  1065→        return (sev_weight * 0.35 + exploitability * 0.25 +
    66→  1066→                impact * 0.25 + confidence * 0.15)

    75→  1075→        parts = [f"#{rank} priority"]

   111→  1111→def generate_cvss(severity: str, exploitability: float = 0.5, impact: float = 0.5) -> Dict[str, Any]:
   112→  1112→    """Generate a CVSS v3 interpretation from severity and metrics.
   113→  1113→
   114→  1114→    Returns dict with base_score, severity, vector_string.
   115→  1115→    """
   116→  1116→    base = CVSS_BASE_SCORES.get(severity.lower(), 5.0)
   117→  1117→
   118→  1118→    # Adjust based on exploitability and impact
   119→  1119→    adj = base + (exploitability - 0.5) * 1.5 + (impact - 0.5) * 2.0
   120→  1120→    base_score = round(max(0.0, min(10.0, adj)), 1)
   121→  1121→
   122→  1122→    if base_score >= 9.0:
   123→  1123→        sev = "CRITICAL"
   124→  1124→    elif base_score >= 7.0:
   125→  1125→        sev = "HIGH"
   126→  1126→    elif base_score >= 4.0:
   127→  1127→        sev = "MEDIUM"
   128→  1128→    elif base_score > 0.0:
   129→  1129→        sev = "LOW"
   130→  1130→    else:
   131→  1131→        sev = "NONE"
   132→  1132→
   133→  1133→    # AV:N/AC:L/PR:N/UI:N/S:U/C:H/I:H/A:H (typical remote vuln)
   134→  1134→    if exploitability >= 0.7:
   135→  1135→        av, ac, pr, ui = "N", "L", "N", "N"
   136→  1136→    elif exploitability >= 0.4:
   137→  1137→        av, ac, pr, ui = "N", "L", "L", "N"
   138→  1138→    else:
   139→  1139→        av, ac, pr, ui = "N", "H", "L", "R"
   140→  1140→
   141→  1141→    c = "H" if impact >= 0.7 else ("L" if impact >= 0.3 else "N")
   142→  1142→    i = "H" if impact >= 0.7 else ("L" if impact >= 0.4 else "N")
   143→  1143→    a = "H" if impact >= 0.8 else ("L" if impact >= 0.3 else "N")
   144→  1144→
   145→  1145→    vector = f"CVSS:3.1/AV:{av}/AC:{ac}/PR:{pr}/UI:{ui}/S:U/C:{c}/I:{i}/A:{a}"
   146→  1146→
   147→  1147→    return {
   148→  1148→        "base_score": base_score,
   149→  1149→        "severity": sev,
   150→  1150→        "vector_string": vector,
   151→  1151→    }

   540→  1540→    def _reduce_false_positives(self, findings: List[Dict]) -> List[Dict]:
   541→  1541→        """Apply heuristic rules to identify likely false positives.
   542→  1542→        
   543→  1543→        Rules:
   544→  1544→        1. Findings with very low severity and generic descriptions
   545→  1545→        2. Duplicate findings with different titles but same asset+category
   546→  1546→        3. Findings with empty or very short evidence
   547→  1547→        4. Findings whose descriptions match common false-positive patterns
   548→  1548→        """
   549→  1549→        fp_indicators = [
   550→  1550→            "example", "test", "sample", "demo", "placeholder",
   551→  1551→            "localhost", "127.0.0.1", "0.0.0.0", "::1",
   552→  1552→            "staging", "development", "dev-", "test-",
   553→  1553→        ]
   554→  1554→        
   555→  1555→        result = []
   556→  1556→        seen_signatures = set()
   557→  1557→        
   558→  1558→        for f in findings:
   559→  1559→            fp_score = 0  # 0 = definitely real, higher = more likely FP
   560→  1560→            
   561→  1561→            # Rule 1: Check for FP indicator terms in evidence
   562→  1562→            evidence = f.get("evidence", "").lower()
   563→  1563→            asset = f.get("asset", "").lower()
   564→  1564→            title = f.get("title", "").lower()
   565→  1565→            
   566→  1566→            for indicator in fp_indicators:
   567→  1567→                if indicator in evidence or indicator in asset:
   568→  1568→                    fp_score += 2
   569→  1569→                    break
   570→  1570→            
   571→  1571→            # Rule 2: Deduplicate by asset+category signature
   572→  1572→            sig = f"{f.get('asset', '')}:{f.get('category', '')}:{str(f.get('severity', ''))}"
   573→  1573→            if sig in seen_signatures and f.get("severity", "").lower() in ("info", "low"):
   574→  1574→                fp_score += 3
   575→  1575→            seen_signatures.add(sig)
   576→  1576→            
   577→  1577→            # Rule 3: Very short or empty evidence
   578→  1578→            if len(evidence.strip()) < 10:
   579→  1579→                fp_score += 2
   580→  1580→            
   581→  1581→            # Rule 4: Generic description patterns
   582→  1582→            generic_patterns = [
   583→  1583→                r'^a .* (was|is) (found|detected|discovered)$',
   584→  1584→                r'^possible .*$',
   585→  1585→                r'^potential .*$',
   586→  1586→            ]
   587→  1587→            import re
   588→  1588→            desc = f.get("description", "").lower().strip()
   589→  1589→            for pattern in generic_patterns:
   590→  1590→                if re.match(pattern, desc):
   591→  1591→                    fp_score += 1
   592→  1592→                    break
   593→  1593→            
   594→  1594→            # Mark the finding with FP probability
   595→  1595→            f["false_positive_probability"] = min(1.0, fp_score / 10.0)
   596→  1596→            
   597→  1597→            # Only exclude if very high FP score AND low severity
   598→  1598→            if fp_score >= 6 and f.get("severity", "").lower() in ("info",):
   599→  1599→                continue  # Skip likely false positive
   600→  1600→            
   601→  1601→            result.append(f)
   602→  1602→        
   603→  1603→        return result

   605→  1605→    def _analyze_asset_criticality(self, findings: List[Dict]) -> Dict[str, float]:
   606→  1606→        """Analyze asset criticality based on finding density and severity.
   607→  1607→        
   608→  1608→        Returns dict mapping asset names to criticality scores (0-100).
   609→  1609→        Assets with more critical/high findings are more critical.
   610→  1610→        """
   611→  1611→        asset_data: Dict[str, Dict] = {}
   612→  1612→        
   613→  1613→        sev_weights = {"critical": 10, "high": 8, "medium": 5, "low": 2, "info": 0.5}
   614→  1614→        
   615→  1615→        for f in findings:
   616→  1616→            asset = f.get("asset", "unknown")
   617→  1617→            if asset not in asset_data:
   618→  1618→                asset_data[asset] = {"findings": 0, "weighted_score": 0.0, "max_sev": 0}
   619→  1619→            
   620→  1620→            asset_data[asset]["findings"] += 1
   621→  1621→            sev = f.get("severity", "info").lower()
   622→  1622→            asset_data[asset]["weighted_score"] += sev_weights.get(sev, 0.5)
   623→  1623→            asset_data[asset]["max_sev"] = max(
   624→  1624→                asset_data[asset]["max_sev"],
   625→  1625→                sev_weights.get(sev, 0.5)
   626→  1626→            )
   627→  1627→        
   628→  1628→        # Normalize to 0-100
   629→  1629→        max_weighted = max((d["weighted_score"] for d in asset_data.values()), default=1)
   630→  1630→        
   631→  1631→        criticality = {}
   632→  1632→        for asset, data in asset_data.items():
   633→  1633→            raw = (data["weighted_score"] / max(max_weighted, 1)) * 100
   634→  1634→            # Boost if has critical findings
   635→  1635→            if data["max_sev"] >= 10:
   636→  1636→                raw = min(100, raw + 10)
   637→  1637→            criticality[asset] = round(raw, 1)
   638→  1638→        
   639→  1639→        return criticality
