     1→'use client';
     2→
     3→import { useEffect, useRef, useState, useMemo, useCallback } from 'react';
     4→import { motion } from 'framer-motion';
     5→import { Badge } from '@/components/ui/badge';
     6→
     7→// ─── Types ───────────────────────────────────────────────────────────
     8→
     9→interface RadarBlip {
    10→  id: string;
    11→  label: string;
    12→  fullLabel: string;
    13→  angle: number;          // degrees 0-360, 0 = north
    14→  distance: number;       // 0-1 fraction of max radius
    15→  severity: string;
    16→  category: string;
    17→  brightness: number;     // 0-1 peaks on sweep pass
    18→  size: number;
    19→  discovered: boolean;
    20→  discoveredAt: number;   // timestamp when first hit by sweep
    21→  pingPhase: number;      // ripple animation phase
    22→  signalStrength: number; // 0-1 simulated signal
    23→}
    24→
    25→interface RadarMapProps {
    26→  findings: Array<{
    27→    id: string;
    28→    title: string;
    29→    severity: string;
    30→    category: string;
    31→    description?: string;
    32→    asset: string;
    33→  }>;
    34→  domain: string;
    35→  isScanning?: boolean;
    36→  height?: number;
    37→}
    38→
    39→// ─── Constants ───────────────────────────────────────────────────────
    40→
    41→const SEV_COLORS: Record<string, string> = {
    42→  critical: '#ef4444',
    43→  high: '#f97316',
    44→  medium: '#eab308',
    45→  low: '#22c55e',
    46→  info: '#6b7280',
    47→};
    48→
    49→const CAT_DISTANCE: Record<string, number> = {
    50→  vulnerability: 0.18,
    51→  header: 0.32,
    52→  port: 0.45,
    53→  ssl: 0.58,
    54→  dns: 0.68,
    55→  subdomain: 0.78,
    56→  technology: 0.92,
    57→};
    58→
    59→const CAT_SYMBOL: Record<string, string> = {
    60→  vulnerability: '\u25C6', // diamond
    61→  port: '\u25A0',         // square
    62→  subdomain: '\u25CF',    // circle
    63→  ssl: '\u25B2',          // triangle
    64→  dns: '\u25C7',          // diamond outline
    65→  header: '\u25CB',       // circle outline
    66→  technology: '\u25A1',   // square outline
    67→};
    68→
    69→const DEG = Math.PI / 180;
    70→
    71→// ─── Component ───────────────────────────────────────────────────────
    72→
    73→export function RadarMap({ findings, domain, isScanning = false, height = 560 }: RadarMapProps) {
    74→  const canvasRef = useRef<HTMLCanvasElement>(null);
    75→  const containerRef = useRef<HTMLDivElement>(null);
    76→  const [dimensions, setDimensions] = useState({ width: 0, height: 0 });
    77→  const [hoveredBlip, setHoveredBlip] = useState<RadarBlip | null>(null);
    78→  const [tooltipPos, setTooltipPos] = useState({ x: 0, y: 0 });
    79→
    80→  const blipsRef = useRef<RadarBlip[]>([]);
    81→  const animRef = useRef<number>(0);
    82→  const sweepRef = useRef(0);
    83→  const timeRef = useRef(0);
    84→  const discoveredOrderRef = useRef<string[]>([]);
    85→  const pingsRef = useRef<Array<{ x: number; y: number; r: number; alpha: number; color: string }>>([]);
    86→
    87→  // ── Seed-based deterministic placement ──
    88→  const blipData = useMemo(() => {
    89→    if (findings.length === 0) return [];
    90→
    91→    let hash = 0;
    92→    for (let i = 0; i < domain.length; i++) {
    93→      hash = ((hash << 5) - hash) + domain.charCodeAt(i);
    94→      hash |= 0;
    95→    }
    96→    const rand = () => {
    97→      hash = (hash * 1103515245 + 12345) & 0x7fffffff;
    98→      return hash / 0x7fffffff;
    99→    };
   100→
   101→    // Group by category for sector placement
   102→    const byCategory: Record<string, typeof findings> = {};
   103→    for (const f of findings) {
   104→      (byCategory[f.category] ??= []).push(f);
   105→    }
   106→
   107→    const result: RadarBlip[] = [];
   108→    const catKeys = Object.keys(CAT_DISTANCE);
   109→    let globalIdx = 0;
   110→
   111→    for (const cat of catKeys) {
   112→      const items = byCategory[cat] || [];
   113→      const baseDistance = CAT_DISTANCE[cat];
   114→      const sectorStart = catKeys.indexOf(cat) * (360 / catKeys.length);
   115→
   116→      for (let i = 0; i < items.length; i++) {
   117→        const f = items[i];
   118→        const sectorAngle = sectorStart + (i / Math.max(items.length, 1)) * (360 / catKeys.length);
   119→        const jitter = (rand() - 0.5) * 18;
   120→        const angle = ((sectorAngle + jitter) % 360 + 360) % 360;
   121→        const distJitter = (rand() - 0.5) * 0.1;
   122→        const distance = Math.max(0.08, Math.min(0.97, baseDistance + distJitter));
   123→
   124→        result.push({
   125→          id: f.id,
   126→          label: f.asset.length > 22 ? f.asset.slice(0, 22) + '..' : f.asset,
   127→          fullLabel: f.asset,
   128→          angle,
   129→          distance,
   130→          severity: f.severity,
   131→          category: f.category,
   132→          brightness: 0,
   133→          size: f.severity === 'critical' ? 5.5 : f.severity === 'high' ? 4.5 : f.severity === 'medium' ? 4 : 3.2,
   134→          discovered: false,
   135→          discoveredAt: 0,
   136→          pingPhase: 0,
   137→          signalStrength: 0.5 + rand() * 0.5,
   138→        });
   139→        globalIdx++;
   140→      }
   141→    }
   142→    return result;
   143→  }, [findings, domain]);
   144→
   145→  useEffect(() => {
   146→    blipsRef.current = blipData.map(b => ({ ...b }));
   147→    discoveredOrderRef.current = [];
   148→  }, [blipData]);
   149→
   150→  // ── Resize ──
   151→  useEffect(() => {
   152→    const el = containerRef.current;
   153→    if (!el) return;
   154→    const ro = new ResizeObserver(entries => {
   155→      for (const entry of entries) {
   156→        setDimensions({ width: entry.contentRect.width, height: entry.contentRect.height });
   157→      }
   158→    });
   159→    ro.observe(el);
   160→    return () => ro.disconnect();
   161→  }, []);
   162→
   163→  // ── Main render loop ──
   164→  useEffect(() => {
   165→    if (dimensions.width === 0 || dimensions.height === 0) return;
   166→    const canvas = canvasRef.current;
   167→    if (!canvas) return;
   168→    const ctx = canvas.getContext('2d');
   169→    if (!ctx) return;
   170→
   171→    const dpr = window.devicePixelRatio || 1;
   172→    canvas.width = dimensions.width * dpr;
   173→    canvas.height = dimensions.height * dpr;
   174→    ctx.scale(dpr, dpr);
   175→
   176→    const W = dimensions.width;
   177→    const H = dimensions.height;
   178→    const cx = W / 2;
   179→    const cy = H / 2;
   180→    const maxR = Math.min(cx, cy) - 36;
   181→    const blips = blipsRef.current;
   182→    const pings = pingsRef.current;
   183→
   184→    let lastTs = 0;
   185→
   186→    function draw(ts: number) {
   187→      const dt = lastTs ? Math.min((ts - lastTs) / 1000, 0.05) : 0.016;
   188→      lastTs = ts;
   189→      timeRef.current += dt;
   190→      const t = timeRef.current;
   191→
   192→      // Sweep: one rotation every 5 seconds
   193→      sweepRef.current = (sweepRef.current + dt * 72) % 360;
   194→      const sweepRad = sweepRef.current * DEG;
   195→
   196→      ctx.clearRect(0, 0, W, H);
   197→
   198→      // ──────────────────────────────────
   199→      // 1. RADAR SCREEN BACKGROUND
   200→      // ──────────────────────────────────
   201→
   202→      // Dark CRT-style background with vignette
   203→      const bgGrad = ctx.createRadialGradient(cx, cy, 0, cx, cy, maxR + 40);
   204→      bgGrad.addColorStop(0, '#040a06');
   205→      bgGrad.addColorStop(0.85, '#020804');
   206→      bgGrad.addColorStop(1, '#010302');
   207→      ctx.fillStyle = bgGrad;
   208→      ctx.beginPath();
   209→      ctx.arc(cx, cy, maxR + 6, 0, Math.PI * 2);
   210→      ctx.fill();
   211→
   212→      // Outer ring bezel
   213→      ctx.beginPath();
   214→      ctx.arc(cx, cy, maxR + 6, 0, Math.PI * 2);
   215→      ctx.strokeStyle = 'rgba(0, 255, 136, 0.12)';
   216→      ctx.lineWidth = 2;
   217→      ctx.stroke();
   218→
   219→      ctx.beginPath();
   220→      ctx.arc(cx, cy, maxR + 3, 0, Math.PI * 2);
   221→      ctx.strokeStyle = 'rgba(0, 255, 136, 0.06)';
   222→      ctx.lineWidth = 1;
   223→      ctx.stroke();
   224→
   225→      // ──────────────────────────────────
   226→      // 2. SWEEP TRAIL (phosphor afterglow)
   227→      // ──────────────────────────────────
   228→      // We draw the trail as many thin arc slices going backwards
   229→      const trailSpan = 60 * DEG; // 60 degree trail
   230→      const trailSteps = 40;
   231→      for (let i = 0; i < trailSteps; i++) {
   232→        const frac0 = i / trailSteps;
   233→        const frac1 = (i + 1) / trailSteps;
   234→        const a0 = sweepRad - trailSpan * (1 - frac0);
   235→        const a1 = sweepRad - trailSpan * (1 - frac1);
   236→        const alpha = frac1 * 0.18;
   237→        ctx.beginPath();
   238→        ctx.moveTo(cx, cy);
   239→        ctx.arc(cx, cy, maxR, a0, a1);
   240→        ctx.closePath();
   241→        ctx.fillStyle = `rgba(0, 255, 136, ${alpha})`;
   242→        ctx.fill();
   243→      }
   244→
   245→      // ──────────────────────────────────
   246→      // 3. RANGE RINGS
   247→      // ──────────────────────────────────
   248→      const rings = 5;
   249→      for (let i = 1; i <= rings; i++) {
   250→        const r = (i / rings) * maxR;
   251→        ctx.beginPath();
   252→        ctx.arc(cx, cy, r, 0, Math.PI * 2);
   253→        ctx.strokeStyle = i === rings ? 'rgba(0, 255, 136, 0.2)' : 'rgba(0, 255, 136, 0.07)';
   254→        ctx.lineWidth = i === rings ? 1.5 : 0.7;
   255→        ctx.stroke();
   256→
   257→        // Tick marks every 10 degrees on outer ring
   258→        if (i === rings) {
   259→          for (let deg = 0; deg < 360; deg += 10) {
   260→            const rad = deg * DEG;
   261→            const inner = maxR - (deg % 30 === 0 ? 8 : 4);
   262→            ctx.beginPath();
   263→            ctx.moveTo(cx + inner * Math.cos(rad), cy + inner * Math.sin(rad));
   264→            ctx.lineTo(cx + r * Math.cos(rad), cy + r * Math.sin(rad));
   265→            ctx.strokeStyle = deg % 30 === 0 ? 'rgba(0, 255, 136, 0.2)' : 'rgba(0, 255, 136, 0.08)';
   266→            ctx.lineWidth = deg % 30 === 0 ? 1 : 0.5;
   267→            ctx.stroke();
   268→          }
   269→        }
   270→      }
   271→
   272→      // ──────────────────────────────────
   273→      // 4. RANGE LABELS
   274→      // ──────────────────────────────────
   275→      ctx.fillStyle = 'rgba(0, 255, 136, 0.22)';
   276→      ctx.font = '8px "Geist Mono", monospace';
   277→      ctx.textAlign = 'left';
   278→      ctx.textBaseline = 'bottom';
   279→      for (let i = 1; i <= rings; i++) {
   280→        const r = (i / rings) * maxR;
   281→        ctx.fillText(`${i * 20}%`, cx + r + 5, cy - 3);
   282→      }
   283→
   284→      // ──────────────────────────────────
   285→      // 5. CROSS-HAIR & SECTOR LINES
   286→      // ──────────────────────────────────
   287→      ctx.strokeStyle = 'rgba(0, 255, 136, 0.1)';
   288→      ctx.lineWidth = 0.7;
   289→      // Cardinal lines
   290→      for (let a = 0; a < 360; a += 90) {
   291→        const rad = a * DEG;
   292→        ctx.beginPath();
   293→        ctx.moveTo(cx, cy);
   294→        ctx.lineTo(cx + maxR * Math.cos(rad), cy + maxR * Math.sin(rad));
   295→        ctx.stroke();
   296→      }
   297→      // Diagonal lines (faint)
   298→      ctx.strokeStyle = 'rgba(0, 255, 136, 0.04)';
   299→      for (let a = 45; a < 360; a += 90) {
   300→        const rad = a * DEG;
   301→        ctx.beginPath();
   302→        ctx.moveTo(cx, cy);
   303→        ctx.lineTo(cx + maxR * Math.cos(rad), cy + maxR * Math.sin(rad));
   304→        ctx.stroke();
   305→      }
   306→
   307→      // ──────────────────────────────────
   308→      // 6. BEARING LABELS (compass rose)
   309→      // ──────────────────────────────────
   310→      ctx.font = 'bold 11px "Geist Mono", monospace';
   311→      ctx.textBaseline = 'middle';
   312→      const labelR = maxR + 20;
   313→      const directions = [
   314→        { text: 'N', angle: 0, align: 'center' as CanvasTextAlign, base: 'alphabetic' as CanvasTextBaseline },
   315→        { text: 'E', angle: 90, align: 'left' as CanvasTextAlign, base: 'middle' as CanvasTextBaseline },
   316→        { text: 'S', angle: 180, align: 'center' as CanvasTextAlign, base: 'hanging' as CanvasTextBaseline },
   317→        { text: 'W', angle: 270, align: 'right' as CanvasTextAlign, base: 'middle' as CanvasTextBaseline },
   318→      ];
   319→      for (const d of directions) {
   320→        const rad = d.angle * DEG;
   321→        const lx = cx + labelR * Math.cos(rad);
   322→        const ly = cy + labelR * Math.sin(rad);
   323→        ctx.textAlign = d.align;
   324→        ctx.textBaseline = d.base;
   325→        ctx.fillStyle = d.text === 'N' ? '#00ff88' : 'rgba(0, 255, 136, 0.45)';
   326→        ctx.fillText(d.text, lx, ly);
   327→      }
   328→
   329→      // Degree labels around the outside
   330→      ctx.font = '7px "Geist Mono", monospace';
   331→      ctx.fillStyle = 'rgba(0, 255, 136, 0.18)';
   332→      ctx.textBaseline = 'middle';
   333→      for (let deg = 30; deg < 360; deg += 30) {
   334→        if (deg % 90 === 0) continue;
   335→        const rad = deg * DEG;
   336→        const lx = cx + (maxR + 18) * Math.cos(rad);
   337→        const ly = cy + (maxR + 18) * Math.sin(rad);
   338→        ctx.textAlign = Math.abs(Math.cos(rad)) < 0.01 ? 'center' : Math.cos(rad) > 0 ? 'left' : 'right';
   339→        ctx.fillText(`${String(deg).padStart(3, '0')}`, lx, ly);
   340→      }
   341→
   342→      // ──────────────────────────────────
   343→      // 7. SWEEP BEAM LINE
   344→      // ──────────────────────────────────
   345→      const beamEndX = cx + maxR * Math.cos(sweepRad);
   346→      const beamEndY = cy + maxR * Math.sin(sweepRad);
   347→
   348→      // Beam glow (wide, faint)
   349→      const beamGlow = ctx.createLinearGradient(cx, cy, beamEndX, beamEndY);
   350→      beamGlow.addColorStop(0, 'rgba(0, 255, 136, 0.0)');
   351→      beamGlow.addColorStop(0.3, 'rgba(0, 255, 136, 0.08)');
   352→      beamGlow.addColorStop(1, 'rgba(0, 255, 136, 0.02)');
   353→      ctx.beginPath();
   354→      ctx.moveTo(cx, cy);
   355→      ctx.lineTo(beamEndX, beamEndY);
   356→      ctx.strokeStyle = beamGlow;
   357→      ctx.lineWidth = 12;
   358→      ctx.stroke();
   359→
   360→      // Main beam (sharp)
   361→      const beamSharp = ctx.createLinearGradient(cx, cy, beamEndX, beamEndY);
   362→      beamSharp.addColorStop(0, 'rgba(0, 255, 136, 0.8)');
   363→      beamSharp.addColorStop(0.4, 'rgba(0, 255, 136, 0.45)');
   364→      beamSharp.addColorStop(1, 'rgba(0, 255, 136, 0.1)');
   365→      ctx.beginPath();
   366→      ctx.moveTo(cx, cy);
   367→      ctx.lineTo(beamEndX, beamEndY);
   368→      ctx.strokeStyle = beamSharp;
   369→      ctx.lineWidth = 1.5;
   370→      ctx.stroke();
   371→
   372→      // Beam tip glow
   373→      const tipGlow = ctx.createRadialGradient(beamEndX, beamEndY, 0, beamEndX, beamEndY, 20);
   374→      tipGlow.addColorStop(0, 'rgba(0, 255, 136, 0.25)');
   375→      tipGlow.addColorStop(1, 'transparent');
   376→      ctx.beginPath();
   377→      ctx.arc(beamEndX, beamEndY, 20, 0, Math.PI * 2);
   378→      ctx.fillStyle = tipGlow;
   379→      ctx.fill();
   380→
   381→      // ──────────────────────────────────
   382→      // 8. BLIPS (radar contacts)
   383→      // ──────────────────────────────────
   384→      for (const blip of blips) {
   385→        const blipRad = blip.angle * DEG;
   386→        const bx = cx + blip.distance * maxR * Math.cos(blipRad);
   387→        const by = cy + blip.distance * maxR * Math.sin(blipRad);
   388→
   389→        // Angular diff between sweep and blip
   390→        let angleDiff = sweepRef.current - blip.angle;
   391→        if (angleDiff < 0) angleDiff += 360;
   392→        if (angleDiff > 360) angleDiff -= 360;
   393→
   394→        // Blip illuminates when sweep passes (within 25 degrees behind)
   395→        const hitWindow = 25;
   396→        if (angleDiff < hitWindow && angleDiff >= 0) {
   397→          const wasDiscovered = blip.discovered;
   398→          blip.brightness = 1 - (angleDiff / hitWindow);
   399→          blip.discovered = true;
   400→
   401→          // First discovery: trigger ping + add to order
   402→          if (!wasDiscovered) {
   403→            blip.discoveredAt = t;
   404→            blip.pingPhase = 1;
   405→            discoveredOrderRef.current.push(blip.id);
   406→            const color = SEV_COLORS[blip.severity] || '#00ff88';
   407→            pings.push({ x: bx, y: by, r: 0, alpha: 0.8, color });
   408→          }
   409→        } else {
   410→          // Slow decay
   411→          blip.brightness = Math.max(0, blip.brightness - dt * 0.25);
   412→        }
   413→
   414→        // Ping ripple animation
   415→        if (blip.pingPhase > 0) {
   416→          blip.pingPhase = Math.max(0, blip.pingPhase - dt * 1.2);
   417→          const pingR = (1 - blip.pingPhase) * 30;
   418→          const pingAlpha = blip.pingPhase * 0.5;
   419→          const color = SEV_COLORS[blip.severity] || '#00ff88';
   420→          const pr = parseInt(color.slice(1, 3), 16);
   421→          const pg = parseInt(color.slice(3, 5), 16);
   422→          const pb = parseInt(color.slice(5, 7), 16);
   423→
   424→          ctx.beginPath();
   425→          ctx.arc(bx, by, pingR, 0, Math.PI * 2);
   426→          ctx.strokeStyle = `rgba(${pr},${pg},${pb},${pingAlpha})`;
   427→          ctx.lineWidth = 1.5;
   428→          ctx.stroke();
   429→
   430→          // Second ring
   431→          if (blip.pingPhase < 0.7) {
   432→            const ping2R = (1 - blip.pingPhase * 1.3) * 25;
   433→            if (ping2R > 0) {
   434→              ctx.beginPath();
   435→              ctx.arc(bx, by, ping2R, 0, Math.PI * 2);
   436→              ctx.strokeStyle = `rgba(${pr},${pg},${pb},${pingAlpha * 0.4})`;
   437→              ctx.lineWidth = 1;
   438→              ctx.stroke();
   439→            }
   440→          }
   441→        }
   442→
   443→        // Only draw if discovered
   444→        if (!blip.discovered) continue;
   445→
   446→        const alpha = Math.max(0.12, blip.brightness);
   447→        const color = SEV_COLORS[blip.severity] || '#00ff88';
   448→        const cr = parseInt(color.slice(1, 3), 16);
   449→        const cg = parseInt(color.slice(3, 5), 16);
   450→        const cb = parseInt(color.slice(5, 7), 16);
   451→        const r = blip.size * (0.7 + blip.brightness * 0.5);
   452→
   453→        // Signal noise: slight random position jitter for low-signal blips
   454→        const noiseAmp = (1 - blip.signalStrength) * 1.5;
   455→        const jx = blip.brightness > 0.3 ? 0 : (Math.sin(t * 20 + blip.angle) * noiseAmp);
   456→        const jy = blip.brightness > 0.3 ? 0 : (Math.cos(t * 18 + blip.angle * 2) * noiseAmp);
   457→
   458→        const fx = bx + jx;
   459→        const fy = by + jy;
   460→
   461→        // Outer glow (when bright)
   462→        if (blip.brightness > 0.2) {
   463→          const glowR = r + 12 + blip.brightness * 8;
   464→          const glowGrad = ctx.createRadialGradient(fx, fy, 0, fx, fy, glowR);
   465→          glowGrad.addColorStop(0, `rgba(${cr},${cg},${cb},${blip.brightness * 0.3})`);
   466→          glowGrad.addColorStop(1, 'transparent');
   467→          ctx.beginPath();
   468→          ctx.arc(fx, fy, glowR, 0, Math.PI * 2);
   469→          ctx.fillStyle = glowGrad;
   470→          ctx.fill();
   471→        }
   472→
   473→        // Blip body
   474→        ctx.beginPath();
   475→        ctx.arc(fx, fy, r, 0, Math.PI * 2);
   476→        ctx.fillStyle = `rgba(${cr},${cg},${cb},${alpha})`;
   477→        ctx.fill();
   478→
   479→        // White-hot core (bright blips)
   480→        if (blip.brightness > 0.4) {
   481→          ctx.beginPath();
   482→          ctx.arc(fx, fy, r * 0.35, 0, Math.PI * 2);
   483→          ctx.fillStyle = `rgba(255,255,255,${blip.brightness * 0.7})`;
   484→          ctx.fill();
   485→        }
   486→
   487→        // Critical: pulsing warning ring
   488→        if (blip.severity === 'critical' && blip.brightness > 0.05) {
   489→          const pulseR = r + 8 + Math.sin(t * 5) * 4;
   490→          ctx.beginPath();
   491→          ctx.arc(fx, fy, pulseR, 0, Math.PI * 2);
   492→          ctx.strokeStyle = `rgba(239,68,68,${Math.max(0.08, blip.brightness * 0.5)})`;
   493→          ctx.lineWidth = 1;
   494→          ctx.stroke();
   495→
   496→          // Second pulse ring offset
   497→          const pulse2R = r + 12 + Math.sin(t * 5 + 1) * 3;
   498→          ctx.beginPath();
   499→          ctx.arc(fx, fy, pulse2R, 0, Math.PI * 2);
   500→          ctx.strokeStyle = `rgba(239,68,68,${Math.max(0.04, blip.brightness * 0.25)})`;
   501→          ctx.lineWidth = 0.5;
   502→          ctx.stroke();
   503→        }
   504→
   505→        // High severity: single pulse
   506→        if (blip.severity === 'high' && blip.brightness > 0.1) {
   507→          const hPulseR = r + 6 + Math.sin(t * 3.5) * 3;
   508→          ctx.beginPath();
   509→          ctx.arc(fx, fy, hPulseR, 0, Math.PI * 2);
   510→          ctx.strokeStyle = `rgba(249,115,22,${Math.max(0.05, blip.brightness * 0.3)})`;
   511→          ctx.lineWidth = 0.7;
   512→          ctx.stroke();
   513→        }
   514→
   515→        // Label (visible when bright enough)
   516→        if (blip.brightness > 0.35) {
   517→          const labelAlpha = (blip.brightness - 0.35) * 1.2;
   518→          ctx.fillStyle = `rgba(230,237,243,${Math.min(1, labelAlpha)})`;
   519→          ctx.font = '8px "Geist Mono", monospace';
   520→          ctx.textAlign = 'left';
   521→          ctx.textBaseline = 'middle';
   522→
   523→          // Background for readability
   524→          const labelText = blip.label;
   525→          const metrics = ctx.measureText(labelText);
   526→          ctx.fillStyle = `rgba(4,10,6,${labelAlpha * 0.6})`;
   527→          ctx.fillRect(fx + r + 5, fy - 6, metrics.width + 4, 12);
   528→
   529→          ctx.fillStyle = `rgba(230,237,243,${Math.min(1, labelAlpha * 0.9)})`;
   530→          ctx.fillText(labelText, fx + r + 7, fy);
   531→        }
   532→
   533→        // Connecting line to center for very bright blips
   534→        if (blip.brightness > 0.6) {
   535→          const lineAlpha = (blip.brightness - 0.6) * 0.5;
   536→          ctx.beginPath();
   537→          ctx.moveTo(cx, cy);
   538→          ctx.lineTo(fx, fy);
   539→          ctx.strokeStyle = `rgba(${cr},${cg},${cb},${lineAlpha})`;
   540→          ctx.lineWidth = 0.4;
   541→          ctx.setLineDash([2, 6]);
   542→          ctx.lineDashOffset = -t * 30;
   543→          ctx.stroke();
   544→          ctx.setLineDash([]);
   545→          ctx.lineDashOffset = 0;
   546→        }
   547→      }
   548→
   549→      // ──────────────────────────────────
   550→      // 9. PING RIPPLES (global)
   551→      // ──────────────────────────────────
   552→      for (let i = pings.length - 1; i >= 0; i--) {
   553→        const p = pings[i];
   554→        p.r += dt * 80;
   555→        p.alpha -= dt * 0.6;
   556→        if (p.alpha <= 0) { pings.splice(i, 1); continue; }
   557→        const pr = parseInt(p.color.slice(1, 3), 16);
   558→        const pg = parseInt(p.color.slice(3, 5), 16);
   559→        const pb = parseInt(p.color.slice(5, 7), 16);
   560→        ctx.beginPath();
   561→        ctx.arc(p.x, p.y, p.r, 0, Math.PI * 2);
   562→        ctx.strokeStyle = `rgba(${pr},${pg},${pb},${p.alpha * 0.3})`;
   563→        ctx.lineWidth = 1;
   564→        ctx.stroke();
   565→      }
   566→
   567→      // ──────────────────────────────────
   568→      // 10. CENTER POINT & DOMAIN LABEL
   569→      // ──────────────────────────────────
   570→      const centerGlow = ctx.createRadialGradient(cx, cy, 0, cx, cy, 25);
   571→      centerGlow.addColorStop(0, 'rgba(0, 255, 136, 0.2)');
   572→      centerGlow.addColorStop(1, 'transparent');
   573→      ctx.beginPath();
   574→      ctx.arc(cx, cy, 25, 0, Math.PI * 2);
   575→      ctx.fillStyle = centerGlow;
   576→      ctx.fill();
   577→
   578→      // Center cross
   579→      ctx.strokeStyle = 'rgba(0, 255, 136, 0.6)';
   580→      ctx.lineWidth = 1;
   581→      ctx.beginPath(); ctx.moveTo(cx - 6, cy); ctx.lineTo(cx + 6, cy); ctx.stroke();
   582→      ctx.beginPath(); ctx.moveTo(cx, cy - 6); ctx.lineTo(cx, cy + 6); ctx.stroke();
   583→
   584→      // Center dot
   585→      ctx.beginPath();
   586→      ctx.arc(cx, cy, 2, 0, Math.PI * 2);
   587→      ctx.fillStyle = '#00ff88';
   588→      ctx.fill();
   589→
   590→      // Domain label below center
   591→      ctx.font = 'bold 9px "Geist Mono", monospace';
   592→      ctx.textAlign = 'center';
   593→      ctx.textBaseline = 'top';
   594→      ctx.fillStyle = 'rgba(0, 255, 136, 0.5)';
   595→      ctx.fillText(domain.toUpperCase(), cx, cy + 12);
   596→
   597→      // ──────────────────────────────────
   598→      // 11. SCANNING INDICATOR
   599→      // ──────────────────────────────────
   600→      if (isScanning) {
   601→        const scanPulse = Math.sin(t * 4) * 0.5 + 0.5;
   602→        ctx.beginPath();
   603→        ctx.arc(cx, cy, maxR + 8, 0, Math.PI * 2);
   604→        ctx.strokeStyle = `rgba(0, 255, 136, ${0.08 + scanPulse * 0.08})`;
   605→        ctx.lineWidth = 1.5;
   606→        ctx.setLineDash([6, 6]);
   607→        ctx.lineDashOffset = -t * 40;
   608→        ctx.stroke();
   609→        ctx.setLineDash([]);
   610→        ctx.lineDashOffset = 0;
   611→
   612→        // "SCANNING" text flash
   613→        if (Math.sin(t * 6) > 0) {
   614→          ctx.font = 'bold 10px "Geist Mono", monospace';
   615→          ctx.textAlign = 'center';
   616→          ctx.textBaseline = 'middle';
   617→          ctx.fillStyle = `rgba(0, 255, 136, ${0.4 + scanPulse * 0.3})`;
   618→          ctx.fillText('SCANNING', cx, cy + maxR + 28);
   619→        }
   620→      }
   621→
   622→      // ──────────────────────────────────
   623→      // 12. HUD OVERLAY
   624→      // ──────────────────────────────────
   625→
   626→      // Corner brackets
   627→      const cLen = 22;
   628→      const cOff = 10;
   629→      ctx.strokeStyle = 'rgba(0, 255, 136, 0.18)';
   630→      ctx.lineWidth = 1.5;
   631→      // TL
   632→      ctx.beginPath(); ctx.moveTo(cOff, cOff + cLen); ctx.lineTo(cOff, cOff); ctx.lineTo(cOff + cLen, cOff); ctx.stroke();
   633→      // TR
   634→      ctx.beginPath(); ctx.moveTo(W - cOff - cLen, cOff); ctx.lineTo(W - cOff, cOff); ctx.lineTo(W - cOff, cOff + cLen); ctx.stroke();
   635→      // BL
   636→      ctx.beginPath(); ctx.moveTo(cOff, H - cOff - cLen); ctx.lineTo(cOff, H - cOff); ctx.lineTo(cOff + cLen, H - cOff); ctx.stroke();
   637→      // BR
   638→      ctx.beginPath(); ctx.moveTo(W - cOff - cLen, H - cOff); ctx.lineTo(W - cOff, H - cOff); ctx.lineTo(W - cOff, H - cOff - cLen); ctx.stroke();
   639→
   640→      // Top-left HUD data
   641→      ctx.font = '8px "Geist Mono", monospace';
   642→      ctx.textAlign = 'left';
   643→      ctx.textBaseline = 'top';
   644→      ctx.fillStyle = 'rgba(0, 255, 136, 0.4)';
   645→      ctx.fillText(isScanning ? 'MODE: ACTIVE SCAN' : 'MODE: PASSIVE MONITOR', 18, 18);
   646→      ctx.fillText(`BRG: ${String(Math.floor(sweepRef.current)).padStart(3, '0')}\u00B0`, 18, 32);
   647→
   648→      const discoveredCount = blips.filter(b => b.discovered).length;
   649→      ctx.fillText(`CONTACTS: ${discoveredCount}/${blips.length}`, 18, 46);
   650→
   651→      // Bottom-left HUD
   652→      ctx.textBaseline = 'bottom';
   653→      ctx.fillText(`FREQ: ${(2.4 + Math.sin(t * 0.1) * 0.01).toFixed(3)} GHz`, 18, H - 14);
   654→      ctx.fillText(`GAIN: ${(38 + Math.sin(t * 0.3) * 2).toFixed(1)} dB`, 18, H - 28);
   655→
   656→      // Top-right HUD
   657→      ctx.textAlign = 'right';
   658→      ctx.textBaseline = 'top';
   659→      ctx.fillStyle = 'rgba(0, 255, 136, 0.4)';
   660→      ctx.fillText(`TARGET: ${domain.toUpperCase()}`, W - 18, 18);
   661→      ctx.fillText(`T+ ${t.toFixed(1)}s`, W - 18, 32);
   662→
   663→      // Bottom-right HUD
   664→      ctx.textBaseline = 'bottom';
   665→      ctx.fillStyle = 'rgba(0, 255, 136, 0.3)';
   666→      const rotCount = Math.floor(sweepRef.current / 360);
   667→      ctx.fillText(`ROT: ${rotCount + 1} | SWEEP: ${sweepRef.current.toFixed(0)}\u00B0`, W - 18, H - 14);
   668→      const critBlips = blips.filter(b => b.discovered && b.severity === 'critical').length;
   669→      if (critBlips > 0) {
   670→        ctx.fillStyle = 'rgba(239, 68, 68, 0.5)';

   672→      }
   673→
   674→      // ──────────────────────────────────
   675→      // 13. NOISE / SCAN LINES (CRT effect)
   676→      // ──────────────────────────────────
   677→      // Very subtle horizontal scan lines
   678→      ctx.fillStyle = 'rgba(0, 0, 0, 0.03)';
   679→      for (let y = 0; y < H; y += 3) {
   680→        ctx.fillRect(0, y, W, 1);
   681→      }
   682→
   683→      // Occasional noise flicker
   684→      if (Math.random() < 0.02) {
   685→        const noiseY = Math.random() * H;
   686→        const noiseH = 1 + Math.random() * 2;
   687→        ctx.fillStyle = `rgba(0, 255, 136, ${0.02 + Math.random() * 0.03})`;
   688→        ctx.fillRect(0, noiseY, W, noiseH);
   689→      }
   690→
   691→      animRef.current = requestAnimationFrame(draw);
   692→    }
   693→
   694→    function safeDraw(ts: number) {
   695→      try { draw(ts); } catch (e) { console.error('Radar error:', e); }
   696→    }
   697→
   698→    animRef.current = requestAnimationFrame(safeDraw);
   699→    return () => cancelAnimationFrame(animRef.current);
   700→  }, [dimensions, domain, isScanning]);

   703→  const handleMouseMove = useCallback((e: React.MouseEvent) => {
   704→    if (!canvasRef.current) return;
   705→    const rect = canvasRef.current.getBoundingClientRect();
   706→    const mx = e.clientX - rect.left;

   711→
   712→    let found: RadarBlip | null = null;
   713→    for (const blip of blipsRef.current) {
   714→      if (!blip.discovered) continue;
   715→      const rad = blip.angle * DEG;
   716→      const bx = cx + blip.distance * maxR * Math.cos(rad);
   717→      const by = cy + blip.distance * maxR * Math.sin(rad);
   718→      const dx = mx - bx;
   719→      const dy = my - by;
   720→      if (dx * dx + dy * dy < 120) {
   721→        found = blip;
   722→        setTooltipPos({ x: e.clientX - rect.left, y: e.clientY - rect.top });
   723→        break;
   724→      }
   725→    }
   726→    setHoveredBlip(found);
   727→  }, [dimensions]);
   728→

   748→              LIVE SCAN
   749→            </Badge>
   750→          )}
   751→        </div>
   752→        <div className="flex items-center gap-4 text-xs text-muted-foreground">
   753→          <span className="font-mono">{discoveredCount} contacts</span>
   754→          {critCount > 0 && <span className="text-[#ef4444] font-mono">{critCount} crit</span>}

   772→            style={{

   774→              top: Math.max(tooltipPos.y - 50, 8),
