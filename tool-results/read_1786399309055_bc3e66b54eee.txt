    22→
    23→_SEV_WEIGHT: Dict[str, float] = {
    24→    "critical": 10.0,
    25→    "high": 7.5,
    26→    "medium": 5.0,
    27→    "low": 2.5,
    28→    "info": 1.0,
    29→}

    31→_PTS_MAP: Dict[str, int] = {
    32→    "critical": 15,
    33→    "high": 10,
    34→    "medium": 6,
    35→    "low": 3,
    36→    "info": 1,

    52→# ── Simple YAML fallback parser ──────────────────────────────────────
    53→
    54→def _simple_yaml_load(text: str) -> Any:
    55→    """Attempt to import yaml; if unavailable, parse top-level JSON-like structures.
    56→
    57→    This fallback handles most CloudFormation and Kubernetes manifests that use
    58→    simple key-value or list-of-dicts structures.
    59→    """
    60→    try:
    61→        import yaml  # type: ignore[import-untyped]
    62→        return yaml.safe_load(text)
    63→    except Exception:
    64→        pass
    65→
    66→    # Fallback: try JSON first (many YAML files are valid JSON subsets)
    67→    try:
    68→        return json.loads(text)
    69→    except Exception:
    70→        pass
    71→
    72→    # Ultra-simple line-based extractor for key: value and - key: value
    73→    result: Dict[str, Any] = {}
    74→    current_list_key: Optional[str] = None
    75→    list_items: List[Dict[str, Any]] = []
    76→
    77→    for line in text.splitlines():
    78→        stripped = line.strip()
    79→        if not stripped or stripped.startswith("#"):
    80→            continue
    81→
    82→        # List item
    83→        if stripped.startswith("- "):
    84→            item = stripped[2:]
    85→            if ":" in item:
    86→                k, v = item.split(":", 1)
    87→                entry = {k.strip(): v.strip().strip('"\'')}
    88→                if current_list_key:
    89→                    list_items.append(entry)
    90→                else:
    91→                    result.setdefault(k.strip(), []).append(v.strip().strip('"\''))
    92→            continue
    93→
    94→        # Key: Value
    95→        if ":" in stripped and not stripped.endswith(":"):
    96→            k, v = stripped.split(":", 1)
    97→            k = k.strip()
    98→            v = v.strip()
    99→            if v:
   100→                if v.lower() in ("true", "false"):
   101→                    result[k] = v.lower() == "true"
   102→                elif v.isdigit():
   103→                    result[k] = int(v)
   104→                else:
   105→                    result[k] = v.strip('"\'')
   106→            else:
   107→                result[k] = None
   108→                current_list_key = k
   109→                list_items = []
   110→                result[f"_{k}_items"] = list_items
   111→
   112→    if list_items and current_list_key:
   113→        result[current_list_key] = list_items
   114→
   115→    return result
   116→
   117→
   118→def _yaml_load_all(text: str) -> List[Any]:
   119→    """Load all YAML documents from a multi-doc stream."""
   120→    try:
   121→        import yaml  # type: ignore[import-untyped]
   122→        docs = list(yaml.safe_load_all(text))
   123→        return [d for d in docs if d is not None]
   124→    except Exception:
   125→        pass
   126→
   127→    # Fallback: split on "---" and parse individually
   128→    parts = re.split(r"^---\s*$", text, flags=re.MULTILINE)
   129→    results: List[Any] = []
   130→    for part in parts:
   131→        part = part.strip()
   132→        if not part:
   133→            continue
   134→        try:
   135→            results.append(json.loads(part))
   136→        except Exception:
   137→            parsed = _simple_yaml_load(part)
   138→            if parsed:
   139→                results.append(parsed)
   140→    return results

   148→def _tf_block_ranges(text: str) -> List[Tuple[str, int, int, List[str]]]:
   149→    """Extract top-level HCL blocks as (block_type, start, end, lines)."""
   150→    lines = text.splitlines()
   151→    blocks: List[Tuple[str, int, int, List[str]]] = []
   152→    i = 0
   153→    while i < len(lines):
   154→        m = re.match(r'^(\w+)\s+(?:"([^"]+)")?\s*\{', lines[i])
   155→        if m:
   156→            block_type = m.group(1)
   157→            label = m.group(2) or ""
   158→            depth = 1
   159→            start = i + 1
   160→            block_lines = [lines[i]]
   161→            i += 1
   162→            while i < len(lines) and depth > 0:
   163→                block_lines.append(lines[i])
   164→                for ch in lines[i]:
   165→                    if ch == "{":
   166→                        depth += 1
   167→                    elif ch == "}":
   168→                        depth -= 1
   169→                i += 1
   170→            blocks.append((block_type, start, start + len(block_lines) - 1, block_lines))
   171→        else:
   172→            i += 1
   173→    return blocks
   174→
   175→
   176→def _tf_in_block(block_lines: List[str], pattern: str) -> Optional[int]:
   177→    """Return 1-based line number within *block_lines* matching *pattern*, or None."""
   178→    for idx, line in enumerate(block_lines):
   179→        if re.search(pattern, line, re.I):
   180→            return idx + 1
   181→    return None
   182→
   183→
   184→def _tf_not_in_block(block_lines: List[str], pattern: str) -> bool:
   185→    """Return True if *pattern* is NOT found anywhere in *block_lines*."""
   186→    return not any(re.search(pattern, line, re.I) for line in block_lines)

   189→def parse_tf_file(path: str) -> List[Dict[str, Any]]:
   190→    """Parse a .tf / .tfstate file and return a list of issue dicts."""
   191→    issues: List[Dict[str, Any]] = []
   192→    try:
   193→        text = Path(path).read_text(encoding="utf-8", errors="replace")
   194→    except OSError:
   195→        return issues
   196→
   197→    blocks = _tf_block_ranges(text)
   198→
   199→    for block_type, start, end, blines in blocks:
   200→        label_match = re.search(r'"([^"]+)"', blines[0])
   201→        label = label_match.group(1) if label_match else ""
   202→        joined = "\n".join(blines)
   203→        fname = os.path.basename(path)
   204→
   205→        # ── S3 Bucket Checks ─────────────────────────────────────────
   206→
   207→        if "aws_s3_bucket" in block_type and "policy" not in block_type:
   208→            # Check 1: No versioning block
   209→            if _tf_not_in_block(blines, r'versioning'):
   210→                ln = _tf_in_block(blines, r'aws_s3_bucket') or start
   211→                issues.append({
   212→                    "title": "S3 Bucket Missing Versioning",
   213→                    "severity": "medium", "category": "s3_misconfiguration",
   214→                    "description": f"S3 bucket '{label}' does not enable versioning, making it vulnerable to accidental or malicious data deletion.",
   215→                    "evidence": _ev(fname, start + ln - 1),
   216→                    "remediation": 'Add a versioning block: versioning { enabled = true }',
   217→                    "dread_score": _dread("medium"), "points_deducted": _pts("medium"),
   218→                })
   219→
   220→            # Check 2: public-read ACL
   221→            acl_ln = _tf_in_block(blines, r'acl\s*=\s*"public-read"')
   222→            if acl_ln:
   223→                issues.append({
   224→                    "title": "S3 Bucket with Public-Read ACL",
   225→                    "severity": "high", "category": "s3_misconfiguration",
   226→                    "description": f"S3 bucket '{label}' has ACL set to 'public-read', exposing objects to the internet.",
   227→                    "evidence": _ev(fname, start + acl_ln - 1),
   228→                    "remediation": 'Set acl = "private" and use bucket policies for controlled access.',
   229→                    "dread_score": _dread("high"), "points_deducted": _pts("high"),
   230→                })
   231→
   232→            # Check 3: public-read-write ACL
   233→            acl_ln2 = _tf_in_block(blines, r'acl\s*=\s*"public-read-write"')
   234→            if acl_ln2:
   235→                issues.append({
   236→                    "title": "S3 Bucket with Public-Read-Write ACL",
   237→                    "severity": "critical", "category": "s3_misconfiguration",
   238→                    "description": f"S3 bucket '{label}' has ACL set to 'public-read-write', allowing anyone to read AND write.",
   239→                    "evidence": _ev(fname, start + acl_ln2 - 1),
   240→                    "remediation": 'Set acl = "private" immediately. Use IAM policies for access control.',
   241→                    "dread_score": _dread("critical"), "points_deducted": _pts("critical"),
   242→                })
   243→
   244→            # Check 4: No server-side encryption
   245→            if _tf_not_in_block(blines, r'(server_side_encryption_configuration|aws_kms_key)'):
   246→                ln = _tf_in_block(blines, r'aws_s3_bucket') or start
   247→                issues.append({
   248→                    "title": "S3 Bucket Missing Encryption",
   249→                    "severity": "medium", "category": "s3_misconfiguration",
   250→                    "description": f"S3 bucket '{label}' has no server-side encryption configured.",
   251→                    "evidence": _ev(fname, start + ln - 1),
   252→                    "remediation": 'Add aws_s3_bucket_server_side_encryption_configuration with SSE-S3 or SSE-KMS.',
   253→                    "dread_score": _dread("medium"), "points_deducted": _pts("medium"),
   254→                })
   255→
   256→            # Check 5: No logging
   257→            if _tf_not_in_block(blines, r'aws_s3_bucket_logging'):
   258→                ln = _tf_in_block(blines, r'aws_s3_bucket') or start
   259→                issues.append({
   260→                    "title": "S3 Bucket Missing Access Logging",
   261→                    "severity": "low", "category": "s3_misconfiguration",
   262→                    "description": f"S3 bucket '{label}' has no access logging enabled, reducing audit capability.",
   263→                    "evidence": _ev(fname, start + ln - 1),
   264→                    "remediation": 'Add an aws_s3_bucket_logging resource targeting this bucket.',
   265→                    "dread_score": _dread("low"), "points_deducted": _pts("low"),
   266→                })
   267→
   268→        # Check 6: S3 Bucket Policy with Principal: "*"
   269→        if "aws_s3_bucket_policy" in block_type:
   270→            if re.search(r'Principal\s*=\s*"\*"', joined):
   271→                ln = _tf_in_block(blines, r'Principal') or start
   272→                issues.append({
   273→                    "title": "S3 Bucket Policy with Wildcard Principal",
   274→                    "severity": "critical", "category": "s3_misconfiguration",
   275→                    "description": "Bucket policy allows access from any principal ('*'), potentially exposing all objects.",
   276→                    "evidence": _ev(fname, start + ln - 1),
   277→                    "remediation": "Restrict Principal to specific AWS accounts, users, or roles.",
   278→                    "dread_score": _dread("critical"), "points_deducted": _pts("critical"),
   279→                })
   280→
   281→        # ── Security Group Checks ─────────────────────────────────────
   282→
   283→        if "aws_security_group" in block_type or "aws_security_group_rule" in block_type:
   284→            # Check 7: 0.0.0.0/0 ingress
   285→            if re.search(r'cidr_blocks\s*=\s*\[\s*"0\.0\.0\.0/0"', joined):
   286→                ln = _tf_in_block(blines, r'0\.0\.0\.0/0') or start
   287→                port_ln = _tf_in_block(blines, r'from_port|to_port')
   288→                port_info = ""
   289→                if port_ln and port_ln - 1 < len(blines):
   290→                    port_info = " " + blines[port_ln - 1].strip()
   291→                issues.append({
   292→                    "title": f"Security Group Open to 0.0.0.0/0{port_info}",
   293→                    "severity": "high", "category": "network_exposure",
   294→                    "description": f"Security group '{label}' allows ingress from 0.0.0.0/0 (entire internet).{port_info}",
   295→                    "evidence": _ev(fname, start + ln - 1),
   296→                    "remediation": "Restrict cidr_blocks to specific IP ranges. Use VPN/security groups for admin access.",
   297→                    "dread_score": _dread("high"), "points_deducted": _pts("high"),
   298→                })
   299→
   300→            # Check 8: SSH (port 22) open to the world
   301→            if re.search(r'from_port\s*=\s*22', joined) and re.search(r'cidr_blocks\s*=\s*\[\s*"0\.0\.0\.0/0"', joined):
   302→                ln = _tf_in_block(blines, r'from_port\s*=\s*22') or start
   303→                issues.append({
   304→                    "title": "SSH (Port 22) Open to Internet",
   305→                    "severity": "critical", "category": "network_exposure",
   306→                    "description": "SSH access (port 22) is exposed to 0.0.0.0/0, allowing brute-force attacks.",
   307→                    "evidence": _ev(fname, start + ln - 1),
   308→                    "remediation": "Restrict SSH ingress to known IP ranges or a bastion host.",
   309→                    "dread_score": _dread("critical"), "points_deducted": _pts("critical"),
   310→                })
   311→
   312→            # Check 9: RDP (port 3389) open to the world
   313→            if re.search(r'from_port\s*=\s*3389', joined) and re.search(r'cidr_blocks\s*=\s*\[\s*"0\.0\.0\.0/0"', joined):
   314→                ln = _tf_in_block(blines, r'from_port\s*=\s*3389') or start
   315→                issues.append({
   316→                    "title": "RDP (Port 3389) Open to Internet",
   317→                    "severity": "critical", "category": "network_exposure",
   318→                    "description": "RDP access (port 3389) is exposed to 0.0.0.0/0, a prime target for attackers.",
   319→                    "evidence": _ev(fname, start + ln - 1),
   320→                    "remediation": "Restrict RDP to known IP ranges. Use AWS Systems Manager Session Manager instead.",
   321→                    "dread_score": _dread("critical"), "points_deducted": _pts("critical"),
   322→                })
   323→
   324→            # Check 10: No egress rules (default allow all)
   325→            if _tf_not_in_block(blines, r'egress') and "aws_security_group" in block_type:
   326→                ln = _tf_in_block(blines, r'aws_security_group') or start
   327→                issues.append({
   328→                    "title": "Security Group Missing Egress Rules",
   329→                    "severity": "medium", "category": "network_exposure",
   330→                    "description": f"Security group '{label}' has no explicit egress rules, defaulting to allow all outbound.",
   331→                    "evidence": _ev(fname, start + ln - 1),
   332→                    "remediation": "Add an explicit egress rule restricting outbound traffic to required endpoints.",
   333→                    "dread_score": _dread("medium"), "points_deducted": _pts("medium"),
   334→                })
   335→
   336→        # ── IAM Policy Checks ─────────────────────────────────────────
   337→
   338→        if "aws_iam_policy" in block_type or "aws_iam_policy_document" in block_type or "aws_iam_role_policy" in block_type:
   339→            # Check 11: Full admin Action: "*"
   340→            has_star_action = bool(re.search(r'"Action"\s*=\s*"\*"', joined))
   341→            has_star_action2 = bool(re.search(r'actions\s*=\s*\[\s*"\*"', joined))
   342→            has_star_resource = bool(re.search(r'"Resource"\s*=\s*"\*"', joined))
   343→            has_star_resource2 = bool(re.search(r'resources\s*=\s*\[\s*"\*"', joined))
   344→
   345→            if (has_star_action or has_star_action2) and (has_star_resource or has_star_resource2):
   346→                ln = _tf_in_block(blines, r'"\*"') or start
   347→                issues.append({
   348→                    "title": "IAM Policy with Full Admin Access",
   349→                    "severity": "critical", "category": "iam_misconfiguration",
   350→                    "description": "IAM policy grants Action: '*' on Resource: '*' — full administrative access.",
   351→                    "evidence": _ev(fname, start + ln - 1),
   352→                    "remediation": "Apply least-privilege: specify explicit actions and restrict resources to ARN patterns.",
   353→                    "dread_score": _dread("critical"), "points_deducted": _pts("critical"),
   354→                })
   355→
   356→            # Check 12: Wildcard action without wildcard resource (still risky)
   357→            if (has_star_action or has_star_action2) and not (has_star_resource or has_star_resource2):
   358→                ln = _tf_in_block(blines, r'"Action"\s*=\s*"\*"') or _tf_in_block(blines, r'actions\s*=\s*\[\s*"\*"') or start
   359→                issues.append({
   360→                    "title": "IAM Policy with Wildcard Action",
   361→                    "severity": "high", "category": "iam_misconfiguration",
   362→                    "description": "IAM policy uses Action: '*' which grants all actions on specified resources.",
   363→                    "evidence": _ev(fname, start + ln - 1),
   364→                    "remediation": "Replace '*' with explicit action names (e.g., 's3:GetObject', 's3:PutObject').",
   365→                    "dread_score": _dread("high"), "points_deducted": _pts("high"),
   366→                })
   367→
   368→        # ── RDS / Database Checks ─────────────────────────────────────
   369→
   370→        if "aws_db_instance" in block_type or "aws_rds_cluster" in block_type:
   371→            # Check 13: No encryption
   372→            if _tf_not_in_block(blines, r'storage_encrypted\s*=\s*true'):
   373→                ln = _tf_in_block(blines, r'aws_db_instance|aws_rds_cluster') or start
   374→                issues.append({
   375→                    "title": "RDS Instance Missing Encryption",
   376→                    "severity": "high", "category": "encryption",
   377→                    "description": f"Database instance '{label}' does not have storage encryption enabled.",
   378→                    "evidence": _ev(fname, start + ln - 1),
   379→                    "remediation": "Set storage_encrypted = true and optionally specify kms_key_id.",
   380→                    "dread_score": _dread("high"), "points_deducted": _pts("high"),
   381→                })
   382→
   383→            # Check 14: Publicly accessible
   384→            pub_ln = _tf_in_block(blines, r'publicly_accessible\s*=\s*true')
   385→            if pub_ln:
   386→                issues.append({
   387→                    "title": "RDS Instance Publicly Accessible",
   388→                    "severity": "critical", "category": "network_exposure",
   389→                    "description": f"Database '{label}' is publicly accessible, exposing data to the internet.",
   390→                    "evidence": _ev(fname, start + pub_ln - 1),
   391→                    "remediation": "Set publicly_accessible = false and use VPC security groups for access.",
   392→                    "dread_score": _dread("critical"), "points_deducted": _pts("critical"),
   393→                })
   394→
   395→            # Check 15: No IAM DB authentication
   396→            if _tf_not_in_block(blines, r'iam_database_authentication_enabled\s*=\s*true'):
   397→                ln = _tf_in_block(blines, r'aws_db_instance') or start
   398→                issues.append({
   399→                    "title": "RDS Instance Missing IAM Authentication",
   400→                    "severity": "low", "category": "iam_misconfiguration",
   401→                    "description": f"Database '{label}' does not use IAM database authentication.",
   402→                    "evidence": _ev(fname, start + ln - 1),
   403→                    "remediation": "Enable iam_database_authentication_enabled = true to use IAM for DB credentials.",
   404→                    "dread_score": _dread("low"), "points_deducted": _pts("low"),
   405→                })
   406→
   407→        # ── EC2 Instance Checks ───────────────────────────────────────
   408→
   409→        if "aws_instance" in block_type:
   410→            # Check 16: No IAM instance profile
   411→            if _tf_not_in_block(blines, r'iam_instance_profile'):
   412→                ln = _tf_in_block(blines, r'aws_instance') or start
   413→                issues.append({
   414→                    "title": "EC2 Instance Without IAM Profile",
   415→                    "severity": "medium", "category": "iam_misconfiguration",
   416→                    "description": f"EC2 instance '{label}' has no IAM instance profile, preventing fine-grained AWS access control.",
   417→                    "evidence": _ev(fname, start + ln - 1),
   418→                    "remediation": "Attach an IAM instance profile with least-privilege permissions.",
   419→                    "dread_score": _dread("medium"), "points_deducted": _pts("medium"),
   420→                })
   421→
   422→            # Check 17: No monitoring / detailed monitoring
   423→            if _tf_not_in_block(blines, r'monitoring\s*=\s*true'):
   424→                ln = _tf_in_block(blines, r'aws_instance') or start
   425→                issues.append({
   426→                    "title": "EC2 Instance Missing Detailed Monitoring",
   427→                    "severity": "low", "category": "monitoring",
   428→                    "description": f"EC2 instance '{label}' does not have detailed monitoring enabled.",
   429→                    "evidence": _ev(fname, start + ln - 1),
   430→                    "remediation": "Set monitoring = true for enhanced metrics (additional cost applies).",
   431→                    "dread_score": _dread("low"), "points_deducted": _pts("low"),
   432→                })
   433→
   434→            # Check 18: No user data (potentially no hardening)
   435→            if _tf_not_in_block(blines, r'user_data'):
   436→                ln = _tf_in_block(blines, r'aws_instance') or start
   437→                issues.append({
   438→                    "title": "EC2 Instance Without Hardening User Data",
   439→                    "severity": "info", "category": "hardening",
   440→                    "description": f"EC2 instance '{label}' has no user_data script for OS-level hardening.",
   441→                    "evidence": _ev(fname, start + ln - 1),
   442→                    "remediation": "Add a user_data script to install security updates and configure the OS.",
   443→                    "dread_score": _dread("info"), "points_deducted": _pts("info"),
   444→                })
   445→
   446→        # ── EBS Volume Checks ─────────────────────────────────────────
   447→
   448→        if "aws_ebs_volume" in block_type:
   449→            # Check 19: No encryption
   450→            if _tf_not_in_block(blines, r'encrypted\s*=\s*true'):
   451→                ln = _tf_in_block(blines, r'aws_ebs_volume') or start
   452→                issues.append({
   453→                    "title": "EBS Volume Unencrypted",
   454→                    "severity": "high", "category": "encryption",
   455→                    "description": f"EBS volume '{label}' is not encrypted at rest.",
   456→                    "evidence": _ev(fname, start + ln - 1),
   457→                    "remediation": "Set encrypted = true. For existing volumes, create a snapshot, copy with encryption, and replace.",
   458→                    "dread_score": _dread("high"), "points_deducted": _pts("high"),
   459→                })
   460→
   461→        # ── Lambda Checks ─────────────────────────────────────────────

   797→def _cf_check_security_group(resource: Dict[str, Any], fname: str) -> List[Dict[str, Any]]:
   798→    """Check CloudFormation SecurityGroup resources."""
   799→    issues: List[Dict[str, Any]] = []
   800→    res_name = resource.get("_cf_name", "unknown")
   801→
   802→    props = resource.get("Properties", {})
   803→    if not isinstance(props, dict):
   804→        return issues
   805→
   806→    ingress_rules = props.get("SecurityGroupIngress", [])
   807→    if not isinstance(ingress_rules, list):
   808→        ingress_rules = [ingress_rules]
   809→
   810→    for rule in ingress_rules:
   811→        if not isinstance(rule, dict):
   812→            continue
   813→        cidr = rule.get("CidrIp", "")
   814→        if isinstance(cidr, str) and cidr == "0.0.0.0/0":
   815→            port = rule.get("FromPort", "any")
   816→            issues.append({
   817→                "title": f"CloudFormation SecurityGroup Open to Internet (port {port})",
   818→                "severity": "high", "category": "network_exposure",
   819→                "description": f"SecurityGroup '{res_name}' allows ingress from 0.0.0.0/0 on port {port}.",
   820→                "evidence": f"{fname}: {res_name}.SecurityGroupIngress CidrIp=0.0.0.0/0",
   821→                "remediation": "Restrict CidrIp to specific IP ranges or use VPC CIDR.",
   822→                "dread_score": _dread("high"), "points_deducted": _pts("high"),
   823→            })
   824→
   825→            # SSH specifically
   826→            if str(port) == "22":
   827→                issues.append({
   828→                    "title": "CloudFormation: SSH (Port 22) Open to Internet",
   829→                    "severity": "critical", "category": "network_exposure",
   830→                    "description": f"SecurityGroup '{res_name}' exposes SSH to the entire internet.",
   831→                    "evidence": f"{fname}: {res_name}.SecurityGroupIngress port 22 CidrIp=0.0.0.0/0",
   832→                    "remediation": "Restrict SSH CidrIp to bastion host or VPN IP range.",
   833→                    "dread_score": _dread("critical"), "points_deducted": _pts("critical"),
   834→                })
   835→
   836→            # RDP specifically
   837→            if str(port) == "3389":
   838→                issues.append({
   839→                    "title": "CloudFormation: RDP (Port 3389) Open to Internet",
   840→                    "severity": "critical", "category": "network_exposure",
   841→                    "description": f"SecurityGroup '{res_name}' exposes RDP to the entire internet.",
   842→                    "evidence": f"{fname}: {res_name}.SecurityGroupIngress port 3389 CidrIp=0.0.0.0/0",
   843→                    "remediation": "Use AWS Systems Manager Session Manager instead of RDP.",
   844→                    "dread_score": _dread("critical"), "points_deducted": _pts("critical"),
   845→                })
   846→
   847→    return issues

   850→def _cf_check_bucket_policy(resource: Dict[str, Any], fname: str) -> List[Dict[str, Any]]:
   851→    """Check CloudFormation S3 BucketPolicy resources."""
   852→    issues: List[Dict[str, Any]] = []
   853→    res_name = resource.get("_cf_name", "unknown")
   854→    props = resource.get("Properties", {})
   855→    if not isinstance(props, dict):
   856→        return issues
   857→
   858→    policy_doc = props.get("PolicyDocument", {})
   859→    if not isinstance(policy_doc, dict):
   860→        return issues
   861→
   862→    statements = policy_doc.get("Statement", [])
   863→    if not isinstance(statements, list):
   864→        statements = [statements]
   865→
   866→    for stmt in statements:
   867→        if not isinstance(stmt, dict):
   868→            continue
   869→        principal = stmt.get("Principal", {})
   870→        if isinstance(principal, dict) and principal.get("AWS") == "*":
   871→            issues.append({
   872→                "title": "CloudFormation: S3 BucketPolicy with Wildcard Principal",
   873→                "severity": "critical", "category": "s3_misconfiguration",
   874→                "description": f"BucketPolicy '{res_name}' has Principal: '*' allowing any AWS account access.",
   875→                "evidence": f"{fname}: {res_name} Principal: *",
   876→                "remediation": "Restrict Principal to specific AWS account ARNs.",
   877→                "dread_score": _dread("critical"), "points_deducted": _pts("critical"),
   878→            })
   879→
   880→        action = stmt.get("Action", "")
   881→        resource = stmt.get("Resource", "")

  1014→                    resource = stmt.get("Resource", "")

  1020→                            "evidence": f"{fname}: {res_name} PolicyDocument Action: * Resource: *",

  1022→                            "dread_score": _dread("critical"), "points_deducted": _pts("critical"),
  1023→                        })
  1024→
  1025→    return issues

  1028→def parse_cf_file(path: str) -> List[Dict[str, Any]]:
  1029→    """Parse a CloudFormation YAML/JSON template and return a list of issue dicts."""
  1030→    issues: List[Dict[str, Any]] = []
  1031→    fname = os.path.basename(path)
  1032→
  1033→    try:
  1034→        text = Path(path).read_text(encoding="utf-8", errors="replace")
  1035→    except OSError:
  1036→        return issues
  1037→
  1038→    template = _simple_yaml_load(text)
  1039→    if not isinstance(template, dict):
  1040→        return issues
  1041→
  1042→    resources = template.get("Resources", {})
  1043→    if not isinstance(resources, dict):
  1044→        return issues
  1045→
  1046→    for res_name, resource in resources.items():
  1047→        if not isinstance(resource, dict):
  1048→            continue
  1049→        resource["_cf_name"] = res_name
  1050→        res_type = str(resource.get("Type", ""))
  1051→
  1052→        if "SecurityGroup" in res_type:
  1053→            issues.extend(_cf_check_security_group(resource, fname))
  1054→        if "BucketPolicy" in res_type:
  1055→            issues.extend(_cf_check_bucket_policy(resource, fname))
  1056→        if "DBInstance" in res_type or "RDS" in res_type:
  1057→            issues.extend(_cf_check_rds(resource, fname))
  1058→        if res_type == "AWS::S3::Bucket":
  1059→            issues.extend(_cf_check_s3_bucket(resource, fname))
  1060→        if "IAM" in res_type and ("Role" in res_type or "Policy" in res_type or "User" in res_type):
  1061→            issues.extend(_cf_check_iam(resource, fname))
  1062→
  1063→        # Generic: Lambda without VPC
  1064→        if "Lambda" in res_type and "Function" in res_type:
  1065→            props = resource.get("Properties", {})
  1066→            if isinstance(props, dict) and not props.get("VpcConfig"):
  1067→                issues.append({
  1068→                    "title": f"CloudFormation: Lambda '{res_name}' Not in VPC",
  1069→                    "severity": "low", "category": "network_exposure",
  1070→                    "description": f"Lambda function '{res_name}' has no VpcConfig, running with direct internet access.",
  1071→                    "evidence": f"{fname}: {res_name} missing VpcConfig",
  1072→                    "remediation": "Add VpcConfig with SubnetIds and SecurityGroupIds.",
  1073→                    "dread_score": _dread("low"), "points_deducted": _pts("low"),
  1074→                })
  1075→
  1076→        # EC2 Instance without IAM instance profile
  1077→        if res_type == "AWS::EC2::Instance":
  1078→            props = resource.get("Properties", {})
  1079→            if isinstance(props, dict) and not props.get("IamInstanceProfile"):
  1080→                issues.append({
  1081→                    "title": f"CloudFormation: EC2 '{res_name}' Without IAM Profile",
  1082→                    "severity": "medium", "category": "iam_misconfiguration",
  1083→                    "description": f"EC2 instance '{res_name}' has no IamInstanceProfile for fine-grained AWS access.",
  1084→                    "evidence": f"{fname}: {res_name} missing IamInstanceProfile",
  1085→                    "remediation": "Add an IamInstanceProfile property referencing an IAM instance profile.",
  1086→                    "dread_score": _dread("medium"), "points_deducted": _pts("medium"),
  1087→                })
  1088→
  1089→        # SecurityGroup missing egress
  1090→        if "SecurityGroup" in res_type and "Ingress" not in res_type:
  1091→            props = resource.get("Properties", {})
  1092→            if isinstance(props, dict):
  1093→                has_egress = props.get("SecurityGroupEgress", [])
  1094→                if not has_egress:
  1095→                    issues.append({
  1096→                        "title": f"CloudFormation: SecurityGroup '{res_name}' Missing Egress Rules",
  1097→                        "severity": "medium", "category": "network_exposure",
  1098→                        "description": f"SecurityGroup '{res_name}' has no egress rules, defaulting to allow all outbound.",
  1099→                        "evidence": f"{fname}: {res_name} missing SecurityGroupEgress",
  1100→                        "remediation": "Add explicit SecurityGroupEgress rules to restrict outbound traffic.",
  1101→                        "dread_score": _dread("medium"), "points_deducted": _pts("medium"),
  1102→                    })
  1103→
  1104→    return issues

  1112→# Sensitive file patterns that should never be COPY/ADDed
  1113→_SENSITIVE_FILE_PATTERNS: List[Tuple[re.Pattern, str]] = [
  1114→    (re.compile(r"\.env", re.I), ".env file"),
  1115→    (re.compile(r"\.aws", re.I), ".aws credentials directory"),
  1116→    (re.compile(r"(id_rsa|id_dsa|id_ecdsa|id_ed25519)", re.I), "SSH private key"),
  1117→    (re.compile(r"\.ssh", re.I), ".ssh directory"),
  1118→    (re.compile(r"\.npmrc", re.I), ".npmrc (may contain auth tokens)"),
  1119→    (re.compile(r"\.pem", re.I), ".pem certificate/key file"),
  1120→    (re.compile(r"\.key", re.I), ".key private key file"),
  1121→    (re.compile(r"\.p12|\.pfx", re.I), "PKCS#12 certificate bundle"),
  1122→    (re.compile(r"credentials\.json|service-account", re.I), "GCP service account credentials"),
  1123→]
  1124→
  1125→# Secret environment variable patterns
  1126→_SECRET_ENV_PATTERNS = re.compile(
  1127→    r'(?:PASSWORD|SECRET|TOKEN|API_KEY|PRIVATE_KEY|AUTH_TOKEN|ACCESS_KEY)'
  1128→    r'\s*=\s*["\x27]\S{1,}["\x27]',
  1129→    re.I,
  1130→)
  1131→
  1132→# Dangerous ports to expose
  1133→_DANGEROUS_PORTS: Dict[str, str] = {
  1134→    "22": "SSH",
  1135→    "2375": "Docker daemon (unencrypted)",
  1136→    "2376": "Docker daemon (TLS)",
  1137→    "6379": "Redis",
  1138→    "27017": "MongoDB",
  1139→    "5432": "PostgreSQL",
  1140→    "3306": "MySQL",
  1141→    "9200": "Elasticsearch",
  1142→    "11211": "Memcached",
  1143→}

  1146→def parse_dockerfile(path: str) -> List[Dict[str, Any]]:
  1147→    """Parse a Dockerfile and return a list of issue dicts."""
  1148→    issues: List[Dict[str, Any]] = []
  1149→    fname = os.path.basename(path)
  1150→
  1151→    try:
  1152→        text = Path(path).read_text(encoding="utf-8", errors="replace")
  1153→    except OSError:
  1154→        return issues
  1155→
  1156→    lines = text.splitlines()
  1157→    has_user = False
  1158→    has_healthcheck = False
  1159→    has_from = False
  1160→    from_line = 0
  1161→    user_line_num = 0
  1162→
  1163→    for idx, line in enumerate(lines):
  1164→        stripped = line.strip()
  1165→        if not stripped or stripped.startswith("#"):
  1166→            continue
  1167→
  1168→        ln = idx + 1
  1169→
  1170→        # ── FROM checks ───────────────────────────────────────────────
  1171→        if stripped.upper().startswith("FROM "):
  1172→            has_from = True
  1173→            from_line = ln
  1174→            from_val = stripped[5:].strip()
  1175→
  1176→            # Check: FROM scratch
  1177→            if from_val.lower() == "scratch":
  1178→                issues.append({
  1179→                    "title": "Dockerfile Uses FROM scratch",
  1180→                    "severity": "medium", "category": "container_image",
  1181→                    "description": "FROM scratch produces a minimal image but provides no package management or shell for debugging.",
  1182→                    "evidence": _ev(fname, ln),
  1183→                    "remediation": "Use a minimal base like alpine or distroless if possible.",
  1184→                    "dread_score": _dread("medium"), "points_deducted": _pts("medium"),
  1185→                })
  1186→            # Check: FROM without tag (mutable image)
  1187→            elif ":" not in from_val.split("@")[0]:
  1188→                issues.append({
  1189→                    "title": "Dockerfile FROM Without Specific Tag",
  1190→                    "severity": "medium", "category": "container_image",
  1191→                    "description": f"Base image '{from_val}' has no specific tag, using 'latest' which is mutable.",
  1192→                    "evidence": _ev(fname, ln),
  1193→                    "remediation": f"Pin the image: {from_val}:<specific-tag> or use a digest @{from_val}@sha256:...",
  1194→                    "dread_score": _dread("medium"), "points_deducted": _pts("medium"),
  1195→                })
  1196→            # Check: FROM using :latest tag
  1197→            elif from_val.endswith(":latest"):
  1198→                issues.append({
  1199→                    "title": "Dockerfile FROM Uses :latest Tag",
  1200→                    "severity": "medium", "category": "container_image",
  1201→                    "description": f"Base image uses ':latest' tag which is mutable and non-reproducible.",
  1202→                    "evidence": _ev(fname, ln),
  1203→                    "remediation": f"Pin to a specific version tag instead of :latest.",
  1204→                    "dread_score": _dread("medium"), "points_deducted": _pts("medium"),
  1205→                })
  1206→
  1207→        # ── USER checks ───────────────────────────────────────────────
  1208→        if stripped.upper().startswith("USER "):
  1209→            user_val = stripped[5:].strip()
  1210→            has_user = True
  1211→            user_line_num = ln
  1212→
  1213→            if user_val in ("root", "0"):
  1214→                issues.append({
  1215→                    "title": "Dockerfile Explicitly Runs as Root",
  1216→                    "severity": "high", "category": "container_privilege",
  1217→                    "description": "USER root/0 is explicitly set, running the container with full root privileges.",
  1218→                    "evidence": _ev(fname, ln),
  1219→                    "remediation": "Create a non-root user and set USER <username>.",
  1220→                    "dread_score": _dread("high"), "points_deducted": _pts("high"),
  1221→                })
  1222→
  1223→        # ── COPY / ADD checks ─────────────────────────────────────────
  1224→        if stripped.upper().startswith(("COPY ", "ADD ")):
  1225→            # Check for sensitive files
  1226→            for pat, desc in _SENSITIVE_FILE_PATTERNS:
  1227→                if pat.search(stripped):
  1228→                    issues.append({
  1229→                        "title": f"Dockerfile Copies Sensitive File ({desc})",
  1230→                        "severity": "critical", "category": "secrets_leak",
  1231→                        "description": f"COPY or ADD instruction includes {desc} in the image.",
  1232→                        "evidence": _ev(fname, ln),
  1233→                        "remediation": f"Do not copy {desc} into the image. Use runtime secrets or multi-stage builds.",
  1234→                        "dread_score": _dread("critical"), "points_deducted": _pts("critical"),
  1235→                    })
  1236→                    break
  1237→
  1238→            # Check: ADD with remote URL (supply chain risk)
  1239→            if re.search(r'ADD\s+https?://', stripped, re.I):
  1240→                issues.append({
  1241→                    "title": "Dockerfile ADD from Remote URL",
  1242→                    "severity": "high", "category": "supply_chain",
  1243→                    "description": "ADD instruction fetches from a remote URL, creating a supply chain risk.",
  1244→                    "evidence": _ev(fname, ln),
  1245→                    "remediation": "Use RUN curl/wget with checksum verification instead of ADD for remote URLs.",
  1246→                    "dread_score": _dread("high"), "points_deducted": _pts("high"),
  1247→                })
  1248→
  1249→        # ── ENV checks ────────────────────────────────────────────────
  1250→        if stripped.upper().startswith("ENV "):
  1251→            env_val = stripped[4:].strip()
  1252→            if _SECRET_ENV_PATTERNS.search(env_val):
  1253→                issues.append({
  1254→                    "title": "Dockerfile ENV Contains Secret",
  1255→                    "severity": "high", "category": "hardcoded_secret",
  1256→                    "description": "An ENV variable appears to contain a hardcoded secret/password/token.",
  1257→                    "evidence": _ev(fname, ln),
  1258→                    "remediation": "Use Docker secrets, Kubernetes secrets, or runtime environment injection.",
  1259→                    "dread_score": _dread("high"), "points_deducted": _pts("high"),
  1260→                })
  1261→
  1262→        # ── RUN checks ────────────────────────────────────────────────
  1263→        if stripped.upper().startswith("RUN "):
  1264→            run_val = stripped[4:].strip()
  1265→
  1266→            # curl|wget piped to bash/sh
  1267→            if re.search(r'(?:curl|wget)\s+.*\|\s*(?:ba)?sh', run_val):
  1268→                issues.append({
  1269→                    "title": "Dockerfile RUN Pipes Remote Script to Shell",
  1270→                    "severity": "high", "category": "supply_chain",
  1271→                    "description": "RUN instruction pipes curl/wget output directly to shell — supply chain risk.",
  1272→                    "evidence": _ev(fname, ln),
  1273→                    "remediation": "Download the script first, verify its checksum, then execute it.",
  1274→                    "dread_score": _dread("high"), "points_deducted": _pts("high"),
  1275→                })
  1276→
  1277→            # --privileged in RUN (docker run inside docker)
  1278→            if "--privileged" in run_val:
  1279→                issues.append({
  1280→                    "title": "Dockerfile RUN with --privileged Flag",
  1281→                    "severity": "critical", "category": "container_privilege",
  1282→                    "description": "RUN instruction uses --privileged, granting full host access to the build container.",
  1283→                    "evidence": _ev(fname, ln),
  1284→                    "remediation": "Remove --privileged flag. Grant only specific capabilities.",
  1285→                    "dread_score": _dread("critical"), "points_deducted": _pts("critical"),
  1286→                })
  1287→
  1288→            # Running as root in RUN (sudo/su)
  1289→            if re.search(r'\b(?:sudo|su\s)\b', run_val):
  1290→                issues.append({
  1291→                    "title": "Dockerfile RUN Uses sudo/su",
  1292→                    "severity": "medium", "category": "container_privilege",
  1293→                    "description": "RUN uses sudo/su, indicating the container may need root privileges at runtime.",
  1294→                    "evidence": _ev(fname, ln),
  1295→                    "remediation": "Avoid sudo/su. If root is needed for install, switch back to non-root user.",
  1296→                    "dread_score": _dread("medium"), "points_deducted": _pts("medium"),
  1297→                })
  1298→
  1299→            # apt-get without no-install-recommends
  1300→            if re.search(r'apt-get\s+install', run_val) and "--no-install-recommends" not in run_val:
  1301→                issues.append({
  1302→                    "title": "Dockerfile apt-get Without --no-install-recommends",
  1303→                    "severity": "low", "category": "image_size",
  1304→                    "description": "apt-get install without --no-install-recommends increases image size and attack surface.",
  1305→                    "evidence": _ev(fname, ln),
  1306→                    "remediation": "Add --no-install-recommends to apt-get install and combine RUN layers.",
  1307→                    "dread_score": _dread("low"), "points_deducted": _pts("low"),
  1308→                })
  1309→
  1310→            # apt-get update without cache cleanup in same layer
  1311→            if re.search(r'apt-get\s+update', run_val) and not re.search(r'rm\s+-rf\s+/var/lib/apt', run_val):
  1312→                issues.append({
  1313→                    "title": "Dockerfile apt-get Cache Not Cleaned",
  1314→                    "severity": "low", "category": "image_size",
  1315→                    "description": "apt-get update without cache cleanup in the same layer increases image size.",
  1316→                    "evidence": _ev(fname, ln),
  1317→                    "remediation": "Add '&& rm -rf /var/lib/apt/lists/*' in the same RUN instruction.",
  1318→                    "dread_score": _dread("low"), "points_deducted": _pts("low"),
  1319→                })
  1320→
  1321→        # ── EXPOSE checks ─────────────────────────────────────────────
  1322→        if stripped.upper().startswith("EXPOSE "):
  1323→            ports = stripped[7:].strip().split()
  1324→            for port in ports:
  1325→                port_clean = port.split("/")[0]
  1326→                if port_clean in _DANGEROUS_PORTS:

  1834→def run(target: str, base_url: str, **kwargs) -> Tuple[List[Finding], int, str, str]:
  1835→    """Scan a directory for IaC security issues.
  1836→
  1837→    Recursively discovers Terraform, CloudFormation, Dockerfile, and
  1838→    Kubernetes manifest files, runs the appropriate parser, and
  1839→    converts all issues to Finding objects.
  1840→
  1841→    Returns:
  1842→        (findings, max_points, module_name, category)
  1843→    """
  1844→    findings: List[Finding] = []
  1845→    target_path = Path(target)
  1846→
  1847→    max_points = 100
  1848→    module_name = "iac_audit"
  1849→    category = "infrastructure_as_code"
  1850→
  1851→    if not target_path.is_dir():
  1852→        # Single file mode
  1853→        _scan_file(target_path, findings)
  1854→    else:
  1855→        # Recursive directory scan
  1856→        for p in sorted(target_path.rglob("*")):
  1857→            if not p.is_file():
  1858→                continue
  1859→            # Skip hidden dirs and common non-IaC dirs
  1860→            parts = p.parts
  1861→            if any(part.startswith(".") for part in parts if part != "."):
  1862→                continue
  1863→            if any(part in ("node_modules", "vendor", "__pycache__", ".git", "venv") for part in parts):
  1864→                continue
  1865→            _scan_file(p, findings)
  1866→
  1867→    # Deduplicate by title + evidence
  1868→    seen: set = set()
  1869→    deduped: List[Finding] = []
  1870→    for f in findings:
  1871→        key = (f.title, f.evidence)
  1872→        if key not in seen:
  1873→            seen.add(key)
  1874→            deduped.append(f)
  1875→
  1876→    return (deduped, max_points, module_name, category)
