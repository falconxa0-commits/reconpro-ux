    44→@dataclass
    45→class BenchResult:
    46→    """Single benchmark measurement."""
    47→    name: str
    48→    finding_count: int
    49→    wall_ms: float
    50→    peak_mem_kb: float = 0.0
    51→    cpu_time_ms: float = 0.0
    52→    error: str = ""
    53→    notes: str = ""
    54→
    55→    def to_dict(self) -> Dict[str, Any]:
    56→        return {
    57→            "name": self.name,
    58→            "finding_count": self.finding_count,
    59→            "wall_ms": round(self.wall_ms, 2),
    60→            "peak_mem_kb": round(self.peak_mem_kb, 1),
    61→            "cpu_time_ms": round(self.cpu_time_ms, 2),
    62→            "error": self.error,

    86→SEVERITIES = ["critical", "high", "medium", "low", "info"]
    87→SEV_WEIGHTS = {"critical": 0.1, "high": 0.2, "medium": 0.3, "low": 0.25, "info": 0.15}

    89→CATEGORIES = [

    92→    "weak_cipher", "missing_xfo", "cookie_flags", "cors_misconfig",

    94→    "sensitive_data_in_url", "jwt_weak_signing", "xml_external_entity",
    95→    "ssrf", "command_injection",

    98→MODULES = [
    99→    "recon", "auth", "chain", "bot", "gorgon", "oblivion",
   100→    "vibesec", "nhi", "pegasus", "cloud_recon", "team",

   104→def _rand_sev() -> str:
   105→    r = random.random()
   106→    cumulative = 0.0
   107→    for sev, w in SEV_WEIGHTS.items():
   108→        cumulative += w
   109→        if r < cumulative:
   110→            return sev
   111→    return "info"
   112→
   113→
   114→def _make_finding(idx: int) -> Dict[str, Any]:
   115→    """Generate a single synthetic finding dict."""
   116→    sev = _rand_sev()
   117→    pts = {"critical": 25, "high": 15, "medium": 8, "low": 3, "info": 1}[sev]
   118→    cat = random.choice(CATEGORIES)
   119→    mod = random.choice(MODULES)
   120→    return {
   121→        "title": f"Synthetic finding #{idx}: {cat.replace('_', ' ').title()}",
   122→        "severity": sev,
   123→        "category": cat,
   124→        "module": mod,
   125→        "description": f"This is a synthetic {sev} severity finding in the {cat} category from the {mod} module. " * 3,
   126→        "evidence": f"Evidence string for finding {idx}: some-response-header-value-or-body-snippet " * 2,
   127→        "asset": f"asset-{idx % 50}.example.com",
   128→        "points_deducted": pts,
   129→        "remediation": f"Apply fix for {cat}: configure proper security headers and validate input sanitization.",
   130→        "dread_score": round(random.uniform(0.1, 1.0), 2),
   131→    }
   132→
   133→
   134→def generate_findings(n: int) -> List[Dict[str, Any]]:
   135→    """Generate n synthetic findings."""
   136→    random.seed(42)  # deterministic
   137→    return [_make_finding(i) for i in range(n)]

   139→
   140→def make_scan_data(findings: List[Dict[str, Any]]) -> Dict[str, Any]:
   141→    """Wrap findings into a full scan data dict suitable for report generators."""
   142→    sev_counts: Dict[str, int] = defaultdict(int)
   143→    total_pts = 0
   144→    for f in findings:
   145→        s = f.get("severity", "info").lower()
   146→        sev_counts[s] += 1
   147→        total_pts += f.get("points_deducted", 0)
   148→    score = max(0, 100 - total_pts)
   149→    grade = "F"
   150→    for threshold, g in [(90, "A+"), (80, "A"), (65, "B"), (50, "C"), (35, "D"), (0, "F")]:
   151→        if score >= threshold:
   152→            grade = g
   153→            break
   154→    return {
   155→        "target": "benchmark-target.example.com",
   156→        "modules_run": list(set(f.get("module", "recon") for f in findings)),
   157→        "findings": findings,
   158→        "severity_counts": dict(sev_counts),
   159→        "total_score": score,
   160→        "grade": grade,
   161→        "badge_markdown": f"![ReconPro {grade}](https://img.shields.io/badge/ReconPro-{grade}-red)",
   162→        "module_results": {},
   163→        "intelligence": None,
   164→    }

   167→def timed_mem(func, *args, **kwargs) -> Tuple[Any, float, float, Optional[float]]:
   168→    """Run func(*args, **kwargs), return (result, wall_s, peak_mem_kb, cpu_s).
   169→    
   170→    Uses tracemalloc for peak memory measurement.
   171→    """
   172→    gc.collect()
   173→    tracemalloc.start()
   174→    t0 = time.perf_counter()
   175→    t_cpu = time.process_time()
   176→    try:
   177→        result = func(*args, **kwargs)
   178→    except Exception as e:
   179→        t1 = time.perf_counter()
   180→        t_cpu_end = time.process_time()
   181→        _, peak = tracemalloc.get_traced_memory()
   182→        tracemalloc.stop()
   183→        return None, t1 - t0, peak / 1024, t_cpu_end - t_cpu
   184→    t1 = time.perf_counter()
   185→    t_cpu_end = time.process_time()
   186→    _, peak = tracemalloc.get_traced_memory()
   187→    tracemalloc.stop()
   188→    return result, t1 - t0, peak / 1024, t_cpu_end - t_cpu

   291→    """Benchmark attack graph construction, path finding, cycle detection."""

   294→    print("=" * 70)
   295→
   296→    from reconpro.attack_graph import AttackGraphEngine
   297→
   298→    suite = BenchSuite("attack_graph")
   299→    node_counts = [50, 200, 500, 1000]
   300→
   301→    for n in node_counts:
   302→        findings = generate_findings(n)
   303→        engine = AttackGraphEngine()
   304→
   305→        # Full analyze()
   306→        result, wall_s, mem_kb, cpu_s = timed_mem(engine.analyze, findings, "benchmark.local")
   307→        if result is None:
   308→            suite.add(BenchResult(
   309→                name="full_analyze", finding_count=n,
   310→                wall_ms=wall_s * 1000, peak_mem_kb=mem_kb, cpu_time_ms=cpu_s * 1000,
   311→                error="exception during analyze",
   312→            ))
   313→        else:
   314→            graph_res = result
   315→            suite.add(BenchResult(
   316→                name="full_analyze", finding_count=n,
   317→                wall_ms=wall_s * 1000, peak_mem_kb=mem_kb, cpu_time_ms=cpu_s * 1000,
   318→                notes=f"nodes={len(graph_res.nodes)},edges={len(graph_res.edges)},chains={len(graph_res.attack_chains)}",
   319→            ))
   320→
   321→            # Path finding benchmark: build graph first, then time path ops
   322→            engine2 = AttackGraphEngine()
   323→            engine2.build_graph(findings, "benchmark.local")
   324→            g = engine2.graph
   325→
   326→            # Find paths between first and last node
   327→            node_ids = list(g.nodes.keys())
   328→            if len(node_ids) >= 2:
   329→                start, end = node_ids[0], node_ids[-1]
   330→                t0 = time.perf_counter()
   331→                paths = g.bfs_paths(start, end, max_depth=10)
   332→                t1 = time.perf_counter()
   333→                suite.add(BenchResult(
   334→                    name="bfs_path_finding", finding_count=n,
   335→                    wall_ms=(t1 - t0) * 1000,
   336→                    notes=f"paths_found={len(paths)}",
   337→                ))
   338→
   339→                # Shortest path
   340→                t0 = time.perf_counter()
   341→                sp = g.shortest_path(start, end)
   342→                t1 = time.perf_counter()
   343→                suite.add(BenchResult(
   344→                    name="shortest_path", finding_count=n,
   345→                    wall_ms=(t1 - t0) * 1000,
   346→                    notes=f"found={sp is not None},len={len(sp) if sp else 0}",
   347→                ))
   348→
   349→                # Betweenness proxy
   350→                t0 = time.perf_counter()
   351→                betw = g.betweenness_proxy()
   352→                t1 = time.perf_counter()
   353→                suite.add(BenchResult(
   354→                    name="betweenness_proxy", finding_count=n,
   355→                    wall_ms=(t1 - t0) * 1000,
   356→                    notes=f"nodes_scored={len(betw)}",
   357→                ))
   358→
   359→            # Cycle detection (via DFS in _extract_attack_chains)
   360→            # Kill chain mapping
   361→            t0 = time.perf_counter()
   362→            kc = engine2._map_kill_chain()
   363→            t1 = time.perf_counter()
   364→            suite.add(BenchResult(
   365→                name="kill_chain_mapping", finding_count=n,
   366→                wall_ms=(t1 - t0) * 1000,
   367→                notes=f"phases={len(kc.get('phases', {}))}",
   368→            ))
   369→
   370→        print(f"    n={n:>6} nodes: completed")
   371→
   372→    all_results["attack_graph"] = suite

   676→def bench_concurrent_scaling(all_results: Dict[str, BenchSuite]) -> None:
   677→    """Test concurrent execution scaling with varying worker counts."""
   678→    print("\n" + "=" * 70)
   679→    print("  BENCHMARK 7: Concurrent Execution Scaling")
   680→    print("=" * 70)
   681→
   682→    from reconpro.ai_analyst import AIAnalystEngine
   683→    from reconpro.attack_graph import AttackGraphEngine
   684→    from reconpro.threat_intel import ThreatIntelEngine
   685→
   686→    suite = BenchSuite("concurrent_scaling")
   687→    worker_counts = [1, 2, 4, 8]
   688→    batch_sizes = [50, 200]
   689→
   690→    for batch_size in batch_sizes:
   691→        for workers in worker_counts:
   692→            # Simulate parallel analysis of multiple independent finding sets
   693→            def _analyze_batch(findings_batch):
   694→                engine = AIAnalystEngine()
   695→                return engine.analyze_scan(findings_batch, "bench.local")
   696→
   697→            # Create independent batches
   698→            all_batches = [generate_findings(batch_size) for _ in range(workers)]
   699→
   700→            t0 = time.perf_counter()
   701→            with ThreadPoolExecutor(max_workers=workers) as executor:
   702→                futures = [executor.submit(_analyze_batch, batch) for batch in all_batches]
   703→                results = [f.result() for f in as_completed(futures)]
   704→            t1 = time.perf_counter()
   705→
   706→            total_findings = batch_size * workers
   707→            suite.add(BenchResult(
   708→                name="ai_analyst_concurrent",
   709→                finding_count=total_findings,
   710→                wall_ms=(t1 - t0) * 1000,
   711→                notes=f"workers={workers},batch_size={batch_size}",
   712→            ))
   713→
   714→            # Threat intel concurrent
   715→            def _enrich_batch(findings_batch):
   716→                engine = ThreatIntelEngine(enable_online=False)
   717→                return engine.enrich_scan(findings_batch, "bench.local")
   718→
   719→            all_batches2 = [generate_findings(batch_size) for _ in range(workers)]
   720→            t0 = time.perf_counter()
   721→            with ThreadPoolExecutor(max_workers=workers) as executor:
   722→                futures = [executor.submit(_enrich_batch, batch) for batch in all_batches2]
   723→                results = [f.result() for f in as_completed(futures)]
   724→            t1 = time.perf_counter()
   725→
   726→            suite.add(BenchResult(
   727→                name="threat_intel_concurrent",
   728→                finding_count=total_findings,
   729→                wall_ms=(t1 - t0) * 1000,
   730→                notes=f"workers={workers},batch_size={batch_size}",
   731→            ))
   732→
   733→            print(f"    workers={workers}, batch={batch_size}: completed")
   734→
   735→    # Scaling efficiency analysis
   736→    print("    Computing scaling efficiency...")
   737→    # Run sequential baseline for comparison
   738→    for batch_size in [200]:
   739→        batch = generate_findings(batch_size)
   740→        engine = AIAnalystEngine()
   741→        t0 = time.perf_counter()
   742→        for _ in range(4):  # same total work as 4 workers x 1 batch
   743→            engine.analyze_scan(batch, "bench.local")
   744→        t1 = time.perf_counter()
   745→        sequential_ms = (t1 - t0) * 1000
   746→        suite.add(BenchResult(
   747→            name="ai_analyst_sequential_baseline",
   748→            finding_count=batch_size * 4,
   749→            wall_ms=sequential_ms,
   750→            notes=f"batch_size={batch_size},iterations=4",
   751→        ))
   752→
   753→    all_results["concurrent_scaling"] = suite
