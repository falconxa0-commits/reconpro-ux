   167→    def _probe_one(self, url: str, path: str) -> Optional[Dict[str, Any]]:

   376→class AIVendorFingerprinter:
   377→    """Fingerprint the AI vendor behind a base_url.
   378→
   379→    Usage:
   380→        fp = AIVendorFingerprinter()
   381→        vendors = fp.fingerprint("https://api.example.com")
   382→    """
   383→
   384→    TIMEOUT = 10
   385→    USER_AGENT = "ReconPro-AI-RedTeam/9.1.0"
   386→    PROBE_PATHS = ["/v1/models", "/", "/health", "/api/docs", "/v1/chat/completions"]
   387→
   388→    def fingerprint(self, base_url: str) -> List[Dict[str, Any]]:
   389→        """Return list of {vendor, confidence, evidence} for detected vendors."""
   390→        base_url = base_url.rstrip("/")
   391→        all_headers: Dict[str, str] = {}
   392→        all_bodies: List[str] = []
   393→        found_paths: List[str] = []
   394→
   395→        for path in self.PROBE_PATHS:
   396→            try:
   397→                url = f"{base_url}{path}"
   398→                req = urllib.request.Request(url, method="GET")
   399→                req.add_header("User-Agent", self.USER_AGENT)
   400→                req.add_header("Accept", "application/json")
   401→                with urllib.request.urlopen(req, timeout=self.TIMEOUT) as resp:
   402→                    all_headers.update(dict(resp.headers))
   403→                    body = resp.read(16384).decode("utf-8", errors="replace")
   404→                    all_bodies.append(body)
   405→                    found_paths.append(path)
   406→            except urllib.error.HTTPError as exc:
   407→                if exc.headers:
   408→                    all_headers.update(dict(exc.headers))
   409→                try:
   410→                    body = exc.read(16384).decode("utf-8", errors="replace")
   411→                    all_bodies.append(body)
   412→                    found_paths.append(path)
   413→                except Exception:
   414→                    pass
   415→            except Exception:
   416→                continue
   417→
   418→        combined_body = " ".join(all_bodies).lower()
   419→        combined_headers = {k.lower(): v.lower() for k, v in all_headers.items()}
   420→
   421→        return self._match_vendors(combined_headers, combined_body, found_paths)
   422→
   423→    def _match_vendors(self, headers: Dict[str, str], body: str,
   424→                       paths: List[str]) -> List[Dict[str, Any]]:
   425→        """Score each vendor against collected evidence."""
   426→        results: List[Dict[str, Any]] = []
   427→
   428→        for sig in _VENDOR_SIGNATURES:
   429→            score = 0.0
   430→            evidence: List[str] = []
   431→
   432→            # Header clues (high weight)
   433→            for hc in sig["header_clues"]:
   434→                if hc.lower() in headers:
   435→                    score += 0.35
   436→                    evidence.append(f"header:{hc}")
   437→
   438→            # Body clues (medium weight)
   439→            for bc in sig["body_clues"]:
   440→                if bc.lower() in body:
   441→                    score += 0.15
   442→                    evidence.append(f"body:{bc}")
   443→
   444→            # Endpoint clues (medium weight)
   445→            for ec in sig["endpoint_clues"]:
   446→                if ec in paths:
   447→                    score += 0.20
   448→                    evidence.append(f"endpoint:{ec}")
   449→
   450→            # Cap at 1.0
   451→            score = min(round(score, 2), 1.0)
   452→
   453→            if score >= 0.15:
   454→                results.append({
   455→                    "vendor": sig["name"],
   456→                    "confidence": score,
   457→                    "evidence": evidence,
   458→                })
   459→
   460→        results.sort(key=lambda r: r["confidence"], reverse=True)
   461→        return results

   465→# Watermark Analyzer
   466→# ────────────────────────────────────────────────────────────────────────
   467→
   468→# Known watermark-related header keys and body patterns
   469→_WATERMARK_HEADERS = {
   470→    "x-watermark", "x-content-watermark", "x-ai-watermark",
   471→    "x-signature-sha256", "x-hash-chain", "x-provenance",
   472→}
   473→
   474→# Raw strings with \x22 for embedded double-quote in char classes
   475→_WATERMARK_BODY_PATTERNS = [
   476→    (r"watermark[\s_-]?id[:\s]+[\w-]+", "watermark-id"),
   477→    (r"watermark[\s_-]?token[:\s]+[\w+/=]+", "watermark-token"),
   478→    (r"[0-9a-f]{64}", "sha256-hex"),
   479→    (r'x-watermark[^"]*', "x-watermark-header-leak"),
   480→    (r'provenance[\x22:\s]+[\w./-]+', "provenance-chain"),
   481→]
   482→
   483→# Token-frequency bias indicators: short, highly repeated n-grams
   484→_BIAS_NGRAM_RE = re.compile(r"\b(\w{3,8})\b")
   485→
   486→
   487→class WatermarkAnalyzer:
   488→    """Detect cryptographic watermark indicators in LLM responses.
   489→
   490→    Usage:
   491→        wa = WatermarkAnalyzer()
   492→        result = wa.analyze(response_text, response_headers)
   493→    """
   494→
   495→    def analyze(self, response_text: str,
   496→                response_headers: Optional[Dict[str, str]] = None) -> Dict[str, Any]:
   497→        """Return watermark analysis dict."""
   498→        response_headers = response_headers or {}
   499→        detected = False
   500→        confidence = 0.0
   501→        wm_type = "none"
   502→        indicators: List[str] = []
   503→
   504→        # --- Header checks ---
   505→        for hdr in _WATERMARK_HEADERS:
   506→            val = response_headers.get(hdr, response_headers.get(hdr.lower(), ""))
   507→            if val:
   508→                detected = True
   509→                confidence += 0.4
   510→                indicators.append(f"header:{hdr}={val[:60]}")
   511→                wm_type = "header-injected"
   512→
   513→        # --- Body pattern checks ---
   514→        text_sample = response_text[:8192]
   515→        for pattern, label in _WATERMARK_BODY_PATTERNS:
   516→            try:
   517→                matches = re.findall(pattern, text_sample, re.IGNORECASE)
   518→            except re.error:
   519→                matches = []
   520→            if matches:
   521→                detected = True
   522→                confidence = min(confidence + 0.25, 1.0)
   523→                indicators.append(f"body:{label} ({len(matches)} hit(s))")
   524→                if wm_type == "none":
   525→                    wm_type = label
   526→
   527→        # --- Bias analysis: check for unusually uniform token frequency ---
   528→        bias_result = self._check_token_bias(text_sample)
   529→        if bias_result["suspicious"]:
   530→            detected = True
   531→            confidence = min(confidence + bias_result["score"], 1.0)
   532→            indicators.append(f"bias:{bias_result['reason']}")
   533→            if wm_type == "none":
   534→                wm_type = "token-bias"
   535→
   536→        return {
   537→            "watermark_detected": detected,
   538→            "confidence": round(min(confidence, 1.0), 3),
   539→            "type": wm_type,
   540→            "indicators": indicators,
   541→        }
   542→
   543→    @staticmethod
   544→    def _check_token_bias(text: str) -> Dict[str, Any]:
   545→        """Look for suspiciously uniform short-token distributions.
   546→
   547→        Watermarking schemes like Kirchenbauer et al. introduce subtle
   548→        token-frequency bias.  We approximate by checking if a short
   549→        n-gram repeats far more than expected.
   550→        """
   551→        tokens = _BIAS_NGRAM_RE.findall(text.lower())
   552→        if len(tokens) < 50:
   553→            return {"suspicious": False, "score": 0.0, "reason": ""}
   554→
   555→        freq: Dict[str, int] = {}
   556→        for t in tokens:
   557→            freq[t] = freq.get(t, 0) + 1
   558→
   559→        total = len(tokens)
   560→        unique = len(freq)
   561→        if total == 0:
   562→            return {"suspicious": False, "score": 0.0, "reason": ""}
   563→
   564→        # Type-token ratio: normal English ~0.4-0.6; watermarked may skew
   565→        ttr = unique / total
   566→
   567→        # Top-token concentration
   568→        top_count = max(freq.values())
   569→        top_ratio = top_count / total
   570→
   571→        suspicious = False
   572→        reason = ""
   573→        score = 0.0
   574→
   575→        if ttr < 0.15 and total > 200:
   576→            suspicious = True
   577→            score += 0.3
   578→            reason = f"low-ttr({ttr:.3f})"
   579→
   580→        if top_ratio > 0.08:
   581→            suspicious = True
   582→            score += 0.2
   583→            reason = (reason + "; " if reason else "") + f"top-token-concentration({top_ratio:.3f})"
   584→
   585→        # Check for hex-block patterns (cryptographic residue)
   586→        hex_blocks = re.findall(r"(?:^|\s)[0-9a-f]{32,}(?:\s|$)", text)
   587→        if hex_blocks:
   588→            suspicious = True
   589→            score += 0.3
   590→            reason = (reason + "; " if reason else "") + f"hex-blocks({len(hex_blocks)})"
   591→
   592→        return {"suspicious": suspicious, "score": score, "reason": reason}

   625→        total_score = 0.0
   626→
   627→        for pattern, label, weight in _COLLAPSE_PATTERNS:
   628→            try:
   629→                matches = re.findall(pattern, response_text, re.IGNORECASE | re.MULTILINE)
   630→            except re.error:
   631→                matches = []
   632→            if matches:
   633→                total_score += weight
   634→                indicators.append({
   635→                    "type": label,
   636→                    "matches": len(matches),
   637→                    "weight": weight,
   638→                    "sample": str(matches[0])[:120],
   639→                })
   640→
   641→        # Additional heuristic: entropy of character distribution
   642→        entropy_score = self._char_entropy_score(response_text)
   643→        if entropy_score < 3.0 and len(response_text) > 200:
   644→            total_score += 0.2
   645→            indicators.append({
   646→                "type": "low-entropy",
   647→                "matches": 1,
   648→                "weight": 0.2,
   649→                "sample": f"char-entropy={entropy_score:.2f}",
   650→            })
   651→
   652→        collapse_probability = min(round(total_score, 3), 1.0)
   653→        return {
   654→            "collapse_probability": collapse_probability,
   655→            "indicators_found": indicators,
   656→            "collapsed": collapse_probability >= 0.5,
   657→        }
   658→
   659→    @staticmethod
   660→    def _char_entropy_score(text: str) -> float:
   661→        """Shannon entropy of character distribution in *text*."""
   662→        import math
   663→        if not text:
   664→            return 0.0
   665→        freq: Dict[str, int] = {}
   666→        for ch in text:
   667→            freq[ch] = freq.get(ch, 0) + 1
   668→        length = len(text)
   669→        entropy = 0.0
   670→        for count in freq.values():
   671→            p = count / length
   672→            if p > 0:
   673→                entropy -= p * math.log2(p)
   674→        return entropy

   767→
   768→class SecretExtractor:
   769→    """Extract AI-specific secrets from response text and headers.
   770→
   771→    Usage:
   772→        se = SecretExtractor()
   773→        secrets = se.extract(response_text, response_headers)
   774→    """
   775→
   776→    def extract(self, response_text: str,
   777→                response_headers: Optional[Dict[str, str]] = None) -> List[Dict[str, Any]]:
   778→        """Scan *response_text* and *response_headers* for secrets."""
   779→        response_headers = response_headers or {}
   780→        found: List[Dict[str, Any]] = []
   781→        seen: set = set()
   782→
   783→        # Scan text body
   784→        self._scan(response_text, "response_body", found, seen)
   785→
   786→        # Scan header values
   787→        header_blob = json.dumps(response_headers, separators=("", ":"))
   788→        self._scan(header_blob, "response_header", found, seen)
   789→
   790→        # Deduplicate by value hash
   791→        deduped: List[Dict[str, Any]] = []
   792→        seen_hashes: set = set()
   793→        for s in found:
   794→            h = hashlib.sha256(s["value_masked"].encode()).hexdigest()[:16]
   795→            if h not in seen_hashes:
   796→                seen_hashes.add(h)
   797→                deduped.append(s)
   798→
   799→        return deduped
   800→
   801→    def _scan(self, text: str, source: str, results: List[Dict[str, Any]],
   802→              seen: set) -> None:
   803→        """Run all patterns against *text*, appending to *results*."""
   804→        for pattern, label, description in _SECRET_PATTERNS:
   805→            try:
   806→                compiled = re.compile(pattern)
   807→            except re.error:
   808→                continue
   809→            for m in compiled.finditer(text):
   810→                value = m.group(0)
   811→                # Mask middle of sensitive values for safe reporting
   812→                masked = self._mask(value)
   813→                key = (label, masked)
   814→                if key not in seen:
   815→                    seen.add(key)
   816→                    results.append({
   817→                        "type": label,
   818→                        "description": description,
   819→                        "value_masked": masked,
   820→                        "source": source,
   821→                        "start": m.start(),
   822→                        "end": m.end(),
   823→                    })
   824→
   825→    @staticmethod
   826→    def _mask(value: str) -> str:
   827→        """Mask the middle portion of a secret for safe display."""
   828→        if len(value) <= 12:
   829→            return value[:4] + "****" + value[-4:] if len(value) >= 8 else "****"
   830→        return value[:6] + "****" + value[-4:]
