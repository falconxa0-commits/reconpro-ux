    47→def _render_engineering_panel(result) -> None:
    48→    """Display engineering pipeline results in a Rich panel.
    49→
    50→    Shows engineering score, quality intelligence, stage summary, and
    51→    top recommendations.  Used by audit, scan, and implicit scan commands.
    52→    """
    53→    if result.engineering:
    54→        eng = result.engineering
    55→        eng_metrics = eng.get('metrics', {})
    56→        passed = eng_metrics.get('passed', 0)
    57→        total = eng_metrics.get('total_stages', 0)
    58→        failed = eng_metrics.get('failed', 0)
    59→        eng_score = eng.get('engineering_score', result.engineering_score)
    60→        qi_score = eng.get('quality_score', 0)
    61→        rec_count = eng.get('recommendations_count', 0)
    62→        score_color = "bright_green" if eng_score >= 70 else "yellow" if eng_score >= 40 else "bright_red"
    63→
    64→        lines = [
    65→            f"Status: [bold]{eng.get('overall_status', 'unknown').upper()}[/bold]  |  "
    66→            f"Engineering Score: [bold {score_color}]{eng_score:.1f}/100[/]  |  "
    67→            f"Duration: {eng.get('total_duration_s', 0):.2f}s",
    68→            f"Stages: {passed}/{total} passed" + (f"  |  {failed} failed" if failed else ""),
    69→        ]
    70→        if qi_score > 0:
    71→            lines.append(f"Quality Score: {qi_score:.1f}/100  (Grade: {eng.get('quality_grade', 'N/A')})")
    72→        if rec_count > 0:
    73→            lines.append(f"Recommendations: {rec_count}")
    74→
    75→        console.print(Panel(
    76→            "\n".join(lines),
    77→            title="[bold]ENGINEERING PIPELINE[/bold]",
    78→            border_style="bright_cyan",
    79→        ))
    80→    elif result.engineering_score > 0:
    81→        # Show engineering score even without full pipeline
    82→        score_color = "bright_green" if result.engineering_score >= 70 else "yellow" if result.engineering_score >= 40 else "bright_red"
    83→        console.print(Panel(
    84→            f"Engineering Score: [bold {score_color}]{result.engineering_score:.1f}/100[/]",
    85→            title="[bold]ENGINEERING SCORE[/bold]",
    86→            border_style="bright_cyan",
    87→        ))

    90→def _render_summary(result, title: str = "RECONPRO") -> None:
    91→    score = result.total_score
    92→    grade = result.grade
    93→    grade_color = GRADE_COLORS.get(grade, "bold bright_red")
    94→    sc = result.severity_counts
    95→    total = len(result.findings)
    96→    bar_width = 50
    97→    filled = int(score / 100 * bar_width)
    98→    bar = "█" * filled + "░" * (bar_width - filled)
    99→    lines = [
   100→        Text(f"\n  {bar} [bold {grade_color}]{score}/100 ({grade})[/{grade_color}]"),
   101→        Text(f"\n  Target: [cyan]{result.target}[/]"),
   102→        Text(f"  Modules: [dim]{', '.join(result.modules_run)}[/]"),
   103→        Text(""),
   104→        Text(
   105→            f"  Findings: [bold]{total}[/]  "
   106→            f"[bright_red]{sc.get('critical', 0)} critical[/], "
   107→            f"[red]{sc.get('high', 0)} high[/], "
   108→            f"[yellow]{sc.get('medium', 0)} medium[/], "
   109→            f"[green]{sc.get('low', 0)} low[/], "
   110→            f"[dim]{sc.get('info', 0)} info[/]",
   111→        ),
   112→    ]
   113→    if result.vibesec_score is not None:
   114→        lines.append(Text(
   115→            f"\n  VibeSec Score: [bold bright_green]{result.vibesec_score}/100 ({result.vibesec_grade})[/]"
   116→        ))
   117→    console.print(Panel(Group(*lines), border_style=grade_color,
   118→                         title=f"[bold]{title}[/bold]", title_align="left", padding=(1, 2)))

   121→def _render_findings_table(result, show_remediation: bool = False) -> None:
   122→    if not result.findings:
   123→        console.print("\n  [bright_green]No findings. Target is clean.[/]")
   124→        return
   125→    table = Table(title=f"Findings — {len(result.findings)} detected",
   126→                  border_style="bright_white", header_style="bold bright_white")
   127→    table.add_column("Severity", style="bold", width=10)
   128→    table.add_column("Module", style="cyan", width=12)
   129→    table.add_column("Category", style="magenta", width=18)
   130→    table.add_column("Finding", style="white")
   131→    table.add_column("Pts", style="yellow", width=5)
   132→    if show_remediation:
   133→        table.add_column("Fix", style="bright_green", width=50)
   134→    for f in result.findings[:80]:
   135→        sev = f.get("severity", "info")
   136→        color = SEV_COLORS.get(sev, "white")
   137→        row = [
   138→            f"[{color}]{sev.upper()}[/{color}]",
   139→            f.get("module", ""), f.get("category", ""),
   140→            f.get("title", "")[:70], str(f.get("points_deducted", 0)),
   141→        ]
   142→        if show_remediation:
   143→            row.append(f.get("remediation", "")[:50])
   144→        table.add_row(*row)
   145→    console.print(table)

   148→def _render_module_breakdown(result) -> None:
   149→    if not result.module_results:
   150→        return
   151→    t = Table(title="Module Breakdown", border_style="dim", header_style="bold dim")
   152→    t.add_column("Module", style="bold", width=16)
   153→    t.add_column("Findings", style="white", width=10)
   154→    t.add_column("Critical", style="bright_red", width=10)
   155→    t.add_column("High", style="red", width=8)
   156→    t.add_column("Medium", style="yellow", width=8)
   157→    for mod_id, mod_data in result.module_results.items():
   158→        findings = mod_data.get("findings", [])
   159→        t.add_row(mod_id.upper(), str(len(findings)),
   160→                   str(sum(1 for f in findings if f.get("severity") == "critical")),
   161→                   str(sum(1 for f in findings if f.get("severity") == "high")),
   162→                   str(sum(1 for f in findings if f.get("severity") == "medium")))
   163→    console.print(t)

   174→def _render_remediations(result) -> None:

   179→        console.print("\n  [bright_green]No actionable fixes needed.[/]")
   180→        return
   181→    console.print("\n[bold bright_white]  Fix Commands:[/bold bright_white]\n")
   182→    seen = set()

   184→        fix = f.get("remediation", "").strip()

   193→def _output_result(result, args, title: str = "RECONPRO", show_remediation: bool = False) -> None:
   194→    json_output = getattr(args, "json_output", False)
   195→    output_file = getattr(args, "output_file", None)
   196→    if json_output:
   197→        text = json.dumps(result.to_dict(), indent=2, default=str)
   198→        if output_file:
   199→            with open(output_file, "w") as fp:
   200→                fp.write(text)
   201→            console.print(f"  [green]JSON saved: [cyan]{output_file}[/][/]")
   202→        else:
   203→            print(text)
   204→    else:
   205→        _render_summary(result, title=title)
   206→        console.print()
   207→        _render_module_breakdown(result)
   208→        console.print()
   209→        _render_findings_table(result, show_remediation=show_remediation)
   210→        if show_remediation:
   211→            _render_remediations(result)
   212→        _render_badge(result)
   213→        if output_file:
   214→            with open(output_file, "w") as fp:
   215→                fp.write(json.dumps(result.to_dict(), indent=2, default=str))
   216→            console.print(f"\n  [green]JSON saved: [cyan]{output_file}[/][/]")
   217→        # Auto-save to history
   218→        save_scan(result.to_dict())
   219→    console.print()

   233→def _spinner_wrap(msg: str, fn, *a, _json_mode: bool | None = None, **kw):
   234→    """Run a function with a spinner (suppressed in JSON mode)."""
   235→    if _json_mode is None:
   236→        _json_mode = getattr(_cli_args, 'json_output', False) if _cli_args else False
   237→    if _json_mode:
   238→        return fn(*a, **kw)
   239→    with Progress(SpinnerColumn(), TextColumn("[bold]{task.description}[/]"),
   240→                   TimeElapsedColumn(), console=console) as progress:
   241→        task = progress.add_task(msg, total=None)
   242→        result = fn(*a, **kw)
   243→        progress.update(task, completed=True)
   244→    return result

   280→def _collect_dashboard_data() -> dict:
   281→    """Gather all metrics for the status dashboard.
   282→
   283→    Returns a dict with keys:
   284→      security_score, security_grade, module_health_pct,
   285→      healthy_modules, total_modules, quality_score, engineering_score,
   286→      coverage, regression_status, regression_count, memory_entries,
   287→      last_scan_ago, architecture_score
   288→    """
   289→    data: dict = {}
   290→
   291→    # ── Security score from latest scan ───────────────────────────
   292→    latest = get_latest()
   293→    if latest:
   294→        data["security_score"] = latest.get("total_score", 0)
   295→        data["security_grade"] = latest.get("grade", "N/A")
   296→        data["last_scan_target"] = latest.get("target", "")
   297→        saved_at = latest.get("_saved_at", "")
   298→        if saved_at:
   299→            try:
   300→                dt = datetime.fromisoformat(saved_at)
   301→                ago = (datetime.now(timezone.utc) - dt.astimezone(timezone.utc)).total_seconds()
   302→                data["last_scan_ago"] = _format_ago(ago)
   303→                data["last_scan_seconds"] = ago
   304→            except Exception:
   305→                data["last_scan_ago"] = "unknown"
   306→                data["last_scan_seconds"] = None
   307→        else:
   308→            data["last_scan_ago"] = "unknown"
   309→            data["last_scan_seconds"] = None
   310→    else:
   311→        data["security_score"] = None
   312→        data["security_grade"] = "N/A"
   313→        data["last_scan_ago"] = "no scans"
   314→        data["last_scan_seconds"] = None
   315→        data["last_scan_target"] = ""
   316→
   317→    # ── Module health from engine ─────────────────────────────────
   318→    try:
   319→        from .engine import _module_health
   320→        all_mods = set(ALL_MODULES) | set(LOCAL_MODULES.keys())
   321→        healthy = sum(1 for m in all_mods if _module_health.is_healthy(m))
   322→        total = len(all_mods)
   323→        data["healthy_modules"] = healthy
   324→        data["total_modules"] = total
   325→        data["module_health_pct"] = round(healthy / total * 100, 0) if total else 0
   326→    except Exception:
   327→        data["healthy_modules"] = 0
   328→        data["total_modules"] = 0
   329→        data["module_health_pct"] = 0
   330→
   331→    # ── Quality score from repository memory / quality intelligence ──
   332→    data["quality_score"] = None
   333→    data["engineering_score"] = None
   334→    data["coverage"] = None
   335→    data["architecture_score"] = None
   336→    try:
   337→        from .repository_memory import RepositoryMemory
   338→        mem = RepositoryMemory()
   339→        q_fact = mem.recall("quality.composite_score")
   340→        if q_fact:
   341→            val = q_fact.value
   342→            if isinstance(val, (int, float)):
   343→                data["quality_score"] = int(val)
   344→        e_fact = mem.recall("engineering.pipeline_score")
   345→        if e_fact:
   346→            val = e_fact.value
   347→            if isinstance(val, (int, float)):
   348→                data["engineering_score"] = int(val)
   349→        c_fact = mem.recall("quality.coverage")
   350→        if c_fact:
   351→            val = c_fact.value
   352→            if isinstance(val, (int, float)):
   353→                data["coverage"] = int(val)
   354→        a_fact = mem.recall("engineering.architecture_score")
   355→        if a_fact:
   356→            val = a_fact.value
   357→            if isinstance(val, (int, float)):
   358→                data["architecture_score"] = int(val)
   359→    except Exception:
   360→        pass
   361→
   362→    # Fallback: try quality_intelligence snapshot file
   363→    if data["quality_score"] is None:
   364→        try:
   365→            from .constants import MEMORY_DIR
   366→            snap_file = MEMORY_DIR / "quality_snapshots.json"
   367→            if snap_file.exists():
   368→                with open(snap_file) as f:
   369→                    snapshots = json.load(f)
   370→                if isinstance(snapshots, list) and snapshots:
   371→                    last_snap = snapshots[-1]
   372→                    data["quality_score"] = int(last_snap.get("composite_score", 0))
   373→                    dims = last_snap.get("dimensions", {})
   374→                    cov = dims.get("coverage", {}).get("score")
   375→                    if cov:
   376→                        data["coverage"] = int(cov)
   377→        except Exception:
   378→            pass
   379→
   380→    # ── Regression status ─────────────────────────────────────────
   381→    data["regression_status"] = "STABLE"
   382→    data["regression_count"] = 0
   383→    try:
   384→        from .regression_intelligence import RegressionIntelligence
   385→        ri = RegressionIntelligence()
   386→        regressions = ri.detect_regression()
   387→        if regressions:
   388→            data["regression_count"] = len(regressions)
   389→            data["regression_status"] = "UNSTABLE"
   390→    except Exception:
   391→        pass
   392→
   393→    # ── Memory entries ─────────────────────────────────────────────
   394→    data["memory_entries"] = 0
   395→    try:
   396→        from .repository_memory import RepositoryMemory
   397→        mem = RepositoryMemory()
   398→        data["memory_entries"] = mem.count()
   399→    except Exception:
   400→        pass
   401→
   402→    return data

   405→def _format_ago(seconds: float) -> str:
   406→    """Format a duration in seconds as a human-readable string."""
   407→    if seconds < 60:
   408→        return f"{int(seconds)} seconds ago"
   409→    if seconds < 3600:
   410→        m = int(seconds // 60)
   411→        s = int(seconds % 60)
   412→        return f"{m} minute{'' if m == 1 else 's'} ago"
   413→    if seconds < 86400:
   414→        h = int(seconds // 3600)
   415→        m = int((seconds % 3600) // 60)
   416→        return f"{h}h {m}m ago"
   417→    d = int(seconds // 86400)
   418→    h = int((seconds % 86400) // 3600)
   419→    return f"{d}d {h}h ago"

   422→def _render_dashboard(data: dict) -> None:
   423→    """Render the status dashboard using rich.Panel and rich.Table."""
   424→    # ── Main status panel ─────────────────────────────────────────
   425→    score = data.get("security_score")
   426→    grade = data.get("security_grade", "N/A")
   427→    grade_color = GRADE_COLORS.get(grade, "bold bright_red")
   428→    score_str = f"{score}/100 ({grade})" if score is not None else "N/A (no scans)"
   429→
   430→    health_pct = data.get("module_health_pct", 0)
   431→    healthy = data.get("healthy_modules", 0)
   432→    total = data.get("total_modules", 0)
   433→    health_color = "green" if health_pct >= 80 else ("yellow" if health_pct >= 50 else "red")
   434→
   435→    quality = data.get("quality_score")
   436→    quality_str = f"{quality}/100" if quality is not None else "N/A"
   437→
   438→    eng = data.get("engineering_score")
   439→    eng_str = f"{eng}/100" if eng is not None else "N/A"
   440→
   441→    cov = data.get("coverage")
   442→    cov_str = f"{cov}%" if cov is not None else "N/A"
   443→
   444→    reg_status = data.get("regression_status", "STABLE")
   445→    reg_count = data.get("regression_count", 0)
   446→    reg_color = "green" if reg_status == "STABLE" else "red"
   447→    reg_str = f"{reg_status} ({reg_count} regression{'' if reg_count == 1 else 's'})"
   448→
   449→    mem_count = data.get("memory_entries", 0)
   450→    mem_str = f"{mem_count:,}"
   451→
   452→    last_scan = data.get("last_scan_ago", "no scans")
   453→
   454→    arch = data.get("architecture_score")
   455→    arch_str = f"{arch}/100" if arch is not None else "N/A"
   456→
   457→    main_table = Table(show_header=False, box=None, border_style="bright_cyan", padding=(0, 2))
   458→    main_table.add_column("Label", style="bold bright_white", width=22)
   459→    main_table.add_column("Value", style="bright_white")
   460→    main_table.add_row("Security Score:", f"[{grade_color}]{score_str}[/{grade_color}]")
   461→    main_table.add_row("Module Health:", f"[{health_color}]{health_pct:.0f}% ({healthy}/{total} healthy)[/{health_color}]")
   462→    main_table.add_row("Quality Score:", quality_str)
   463→    main_table.add_row("Engineering Score:", eng_str)
   464→    main_table.add_row("Coverage:", cov_str)
   465→    main_table.add_row("Regression Status:", f"[{reg_color}]{reg_str}[/{reg_color}]")
   466→    main_table.add_row("Memory Entries:", mem_str)
   467→    main_table.add_row("Last Scan:", last_scan)
   468→    main_table.add_row("Architecture Score:", arch_str)
   469→
   470→    console.print(Panel(
   471→        main_table,
   472→        title=f"[bold bright_white]RECONPRO v{__version__} STATUS DASHBOARD[/bold bright_white]",
   473→        border_style="bright_cyan",
   474→        title_align="center",
   475→        padding=(1, 2),
   476→    ))
   477→
   478→    # ── Module health breakdown ────────────────────────────────────
   479→    try:
   480→        from .engine import _module_health
   481→        status = _module_health.get_status()
   482→        if status:
