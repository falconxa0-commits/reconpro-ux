    68→// ── Fear Gauge (SVG Arc) ───────────────────────────────────────────────
    69→
    70→function FearGauge({ score, level }: { score: number; level: FearLevel }) {
    71→  const color = LEVEL_COLORS[level];
    72→  const isCritical = level === 'CRITICAL' || level === 'SEVERE';
    73→  const size = 280;
    74→  const strokeWidth = 24;
    75→  const radius = (size - strokeWidth) / 2;
    76→  const circumference = Math.PI * radius; // 270 degrees = 3/4 of full circle
    77→  const cx = size / 2;
    78→  const cy = size / 2 + 20;
    79→
    80→  // 270-degree arc: from 135° to 405° (in standard math angles)
    81→  const startAngle = 135;
    82→  const endAngle = 405;
    83→  const totalAngle = endAngle - startAngle;
    84→  const scoreAngle = startAngle + (score / 100) * totalAngle;
    85→
    86→  function polarToCartesian(angle: number) {
    87→    const rad = (angle * Math.PI) / 180;
    88→    return {
    89→      x: cx + radius * Math.cos(rad),
    90→      y: cy + radius * Math.sin(rad),
    91→    };
    92→  }
    93→
    94→  function describeArc(start: number, end: number) {
    95→    const s = polarToCartesian(start);
    96→    const e = polarToCartesian(end);
    97→    const largeArc = end - start > 180 ? 1 : 0;
    98→    return `M ${s.x} ${s.y} A ${radius} ${radius} 0 ${largeArc} 1 ${e.x} ${e.y}`;
    99→  }
   100→
   101→  // Background arc path
   102→  const bgPath = describeArc(startAngle, endAngle);
   103→  // Score arc path
   104→  const scorePath = score > 0 ? describeArc(startAngle, scoreAngle) : '';
   105→  // Needle position
   106→  const needlePos = polarToCartesian(scoreAngle);
   107→
   108→  // Tick marks
   109→  const ticks = [0, 20, 40, 60, 80, 100];
   110→  const tickMarks = ticks.map(t => {
   111→    const angle = startAngle + (t / 100) * totalAngle;
   112→    const inner = polarToCartesian(angle);
   113→    const outer = polarToCartesian(angle);
   114→    const ix = cx + (radius - 15) * Math.cos((angle * Math.PI) / 180);
   115→    const iy = cy + (radius - 15) * Math.sin((angle * Math.PI) / 180);
   116→    const ox = cx + (radius + 8) * Math.cos((angle * Math.PI) / 180);
   117→    const oy = cy + (radius + 8) * Math.sin((angle * Math.PI) / 180);
   118→    return { label: t, ix, iy, ox, oy };
   119→  });
   120→
   121→  return (
   122→    <div className="relative flex flex-col items-center">
   123→      {isCritical && (
   124→        <div className="absolute inset-0 rounded-full animate-pulse"
   125→          style={{
   126→            background: `radial-gradient(circle, ${color}33 0%, transparent 70%)`,
   127→            filter: 'blur(20px)',
   128→          }}
   129→        />
   130→      )}
   131→      <svg width={size} height={size} viewBox={`0 0 ${size} ${size}`} className="drop-shadow-lg">
   132→        {/* Background arc */}
   133→        <path d={bgPath} fill="none" stroke="#1f2937" strokeWidth={strokeWidth} strokeLinecap="round" />
   134→        {/* Score arc */}
   135→        <motion.path
   136→          d={scorePath}
   137→          fill="none"
   138→          stroke={color}
   139→          strokeWidth={strokeWidth}
   140→          strokeLinecap="round"
   141→          initial={{ pathLength: 0 }}
   142→          animate={{ pathLength: 1 }}
   143→          transition={{ duration: 1.5, ease: 'easeOut' as const }}
   144→          style={{ filter: `drop-shadow(0 0 8px ${color}88)` }}
   145→        />
   146→        {/* Tick marks */}
   147→        {tickMarks.map((tick) => (
   148→          <g key={tick.label}>
   149→            <line x1={tick.ix} y1={tick.iy} x2={tick.ox} y2={tick.oy} stroke="#4b5563" strokeWidth={1.5} />
   150→            <text
   151→              x={cx + (radius + 22) * Math.cos(((startAngle + (tick.label / 100) * totalAngle) * Math.PI) / 180)}
   152→              y={cy + (radius + 22) * Math.sin(((startAngle + (tick.label / 100) * totalAngle) * Math.PI) / 180)}
   153→              textAnchor="middle" dominantBaseline="middle"
   154→              fill="#6b7280" fontSize={11}
   155→            >
   156→              {tick.label}
   157→            </text>
   158→          </g>
   159→        ))}
   160→        {/* Needle dot */}
   161→        <motion.circle
   162→          cx={needlePos.x} cy={needlePos.y} r={6}
   163→          fill={color}
   164→          initial={{ opacity: 0, scale: 0 }}
   165→          animate={{ opacity: 1, scale: 1 }}
   166→          transition={{ delay: 1.2, duration: 0.3 }}
   167→          style={{ filter: `drop-shadow(0 0 6px ${color})` }}
   168→        />
   169→      </svg>
   170→      {/* Center score */}
   171→      <div className="absolute inset-0 flex flex-col items-center justify-center" style={{ paddingTop: 30 }}>
   172→        <motion.span
   173→          className="font-mono font-black tabular-nums"
   174→          style={{ color, fontSize: 72, lineHeight: 1, textShadow: `0 0 40px ${color}44` }}
   175→          initial={{ opacity: 0, scale: 0.5 }}
   176→          animate={{ opacity: 1, scale: 1 }}
   177→          transition={{ delay: 0.3, duration: 0.8, ease: 'easeOut' as const }}
   178→        >
   179→          {score.toFixed(1)}
   180→        </motion.span>
   181→        <span className="text-xs text-gray-500 mt-1 uppercase tracking-[0.3em]">out of 100</span>
   182→      </div>
   183→    </div>
   184→  );
   185→}

   187→// ── Sparkline (pure SVG) ──────────────────────────────────────────────
   188→
   189→function Sparkline({ data, color = '#60a5fa', width = 120, height = 32 }: {
   190→  data: number[]; color?: string; width?: number; height?: number;
   191→}) {
   192→  if (data.length < 2) return null;
   193→  const min = Math.min(...data);
   194→  const max = Math.max(...data);
   195→  const range = max - min || 1;
   196→  const points = data.map((v, i) => {
   197→    const x = (i / (data.length - 1)) * width;
   198→    const y = height - ((v - min) / range) * (height - 4) - 2;
   199→    return `${x},${y}`;
   200→  }).join(' ');
   201→
   202→  return (
   203→    <svg width={width} height={height} className="overflow-visible">
   204→      <polyline points={points} fill="none" stroke={color} strokeWidth={1.5} strokeLinecap="round" strokeLinejoin="round" />
   205→    </svg>
   206→  );
   207→}

   209→// ── 90-Day Trend Chart (pure SVG) ─────────────────────────────────────
   210→
   211→function TrendChart({ data }: { data: HistoricalDataPoint[] }) {
   212→  const width = 900;
   213→  const height = 280;
   214→  const padding = { top: 20, right: 20, bottom: 40, left: 45 };
   215→  const chartW = width - padding.left - padding.right;
   216→  const chartH = height - padding.top - padding.bottom;
   217→
   218→  const scores = data.map(d => d.score);
   219→  const movingAvg = calculateMovingAverage(scores, 7);
   220→  const minScore = 0;
   221→  const maxScore = 100;
   222→
   223→  function xScale(i: number) { return padding.left + (i / (data.length - 1)) * chartW; }
   224→  function yScale(v: number) { return padding.top + chartH - ((v - minScore) / (maxScore - minScore)) * chartH; }
   225→
   226→  // Level zone backgrounds
   227→  const zones = [
   228→    { from: 0, to: 20, color: '#22c55e08' },
   229→    { from: 20, to: 40, color: '#ffaa0008' },
   230→    { from: 40, to: 60, color: '#ff884408' },
   231→    { from: 60, to: 80, color: '#ff335508' },
   232→    { from: 80, to: 100, color: '#7f1d1d08' },
   233→  ];
   234→
   235→  // Line path
   236→  const linePoints = data.map((d, i) => `${xScale(i)},${yScale(d.score)}`).join(' ');
   237→  // Area path
   238→  const areaPath = `M ${xScale(0)},${yScale(minScore)} ` +
   239→    data.map((d, i) => `L ${xScale(i)},${yScale(d.score)}`).join(' ') +
   240→    ` L ${xScale(data.length - 1)},${yScale(minScore)} Z`;
   241→
   242→  // Moving average path
   243→  const maPoints = movingAvg.map((v, i) => `${xScale(i)},${yScale(v)}`).join(' ');
   244→
   245→  // Find spike points (score > 15 above 7-day MA)
   246→  const spikes: number[] = [];
   247→  for (let i = 7; i < data.length; i++) {
   248→    if (data[i].score - movingAvg[i] > 15) spikes.push(i);
   249→  }
   250→
   251→  // Date labels (every ~15 days)
   252→  const dateLabels: { x: number; label: string }[] = [];
   253→  const step = Math.max(1, Math.floor(data.length / 6));
   254→  for (let i = 0; i < data.length; i += step) {
   255→    const d = new Date(data[i].date);
   256→    dateLabels.push({ x: xScale(i), label: `${d.getMonth() + 1}/${d.getDate()}` });
   257→  }
   258→
   259→  const [hoveredSpike, setHoveredSpike] = useState<number | null>(null);
   260→
   261→  return (
   262→    <div className="w-full overflow-x-auto">
   263→      <svg viewBox={`0 0 ${width} ${height}`} className="w-full h-auto" style={{ minWidth: 600 }}>
   264→        {/* Zone backgrounds */}
   265→        {zones.map(z => (
   266→          <rect
   267→            key={z.to}
   268→            x={padding.left} y={yScale(z.to)} width={chartW} height={yScale(z.from) - yScale(z.to)}
   269→            fill={z.color}
   270→          />
   271→        ))}
   272→        {/* Grid lines */}
   273→        {[0, 20, 40, 60, 80, 100].map(v => (
   274→          <line key={v} x1={padding.left} y1={yScale(v)} x2={width - padding.right} y2={yScale(v)} stroke="#374151" strokeWidth={0.5} strokeDasharray="4 4" />
   275→        ))}
   276→        {/* Y-axis labels */}
   277→        {[0, 20, 40, 60, 80, 100].map(v => (
   278→          <text key={v} x={padding.left - 8} y={yScale(v) + 4} textAnchor="end" fill="#6b7280" fontSize={11}>{v}</text>
   279→        ))}
   280→        {/* Date labels */}
   281→        {dateLabels.map((d, i) => (
   282→          <text key={i} x={d.x} y={height - 8} textAnchor="middle" fill="#6b7280" fontSize={10}>{d.label}</text>
   283→        ))}
   284→        {/* Area fill */}
   285→        <path d={areaPath} fill="url(#fearGradient)" opacity={0.3} />
   286→        <defs>
   287→          <linearGradient id="fearGradient" x1="0" y1="0" x2="0" y2="1">
   288→            <stop offset="0%" stopColor="#ff3355" stopOpacity={0.4} />
   289→            <stop offset="100%" stopColor="#ff3355" stopOpacity={0} />
   290→          </linearGradient>
   291→        </defs>
   292→        {/* Score line */}
   293→        <motion.polyline
   294→          points={linePoints} fill="none" stroke="#60a5fa" strokeWidth={2} strokeLinejoin="round"
   295→          initial={{ pathLength: 0 }} animate={{ pathLength: 1 }} transition={{ duration: 2, ease: 'easeOut' as const }}
   296→        />
   297→        {/* Moving average */}
   298→        <motion.polyline
   299→          points={maPoints} fill="none" stroke="#f59e0b" strokeWidth={1.5} strokeDasharray="6 3" strokeLinejoin="round" opacity={0.7}
   300→          initial={{ pathLength: 0 }} animate={{ pathLength: 1 }} transition={{ duration: 2, delay: 0.5, ease: 'easeOut' as const }}
   301→        />
   302→        {/* Spike markers */}
   303→        {spikes.map(i => (
   304→          <g key={i}
   305→            onMouseEnter={() => setHoveredSpike(i)} onMouseLeave={() => setHoveredSpike(null)}
   306→            className="cursor-pointer"
   307→          >
   308→            <circle cx={xScale(i)} cy={yScale(data[i].score)} r={5} fill="#ff3355" stroke="#1f2937" strokeWidth={2} />
   309→            <line x1={xScale(i)} y1={yScale(data[i].score)} x2={xScale(i)} y2={yScale(0)} stroke="#ff335544" strokeWidth={1} strokeDasharray="3 3" />
   310→          </g>
   311→        ))}
   312→        {/* Spike tooltip */}
   313→        {hoveredSpike !== null && (
   314→          <g>
   315→            <rect
   316→              x={Math.min(xScale(hoveredSpike) - 60, width - 170)}
   317→              y={yScale(data[hoveredSpike].score) - 50}
   318→              width={160} height={36} rx={6} fill="#1f2937" stroke="#374151" strokeWidth={1}
   319→            />
   320→            <text
   321→              x={Math.min(xScale(hoveredSpike) - 52, width - 162)}
   322→              y={yScale(data[hoveredSpike].score) - 28}
   323→              fill="#f8fafc" fontSize={11}
   324→            >
   325→              {data[hoveredSpike].date}: {data[hoveredSpike].score.toFixed(1)} — {data[hoveredSpike].level}
   326→            </text>
   327→          </g>
   328→        )}
   329→        {/* Legend */}
   330→        <line x1={width - 180} y1={12} x2={width - 160} y2={12} stroke="#60a5fa" strokeWidth={2} />
   331→        <text x={width - 155} y={16} fill="#9ca3af" fontSize={10}>Daily Score</text>
   332→        <line x1={width - 90} y1={12} x2={width - 70} y2={12} stroke="#f59e0b" strokeWidth={1.5} strokeDasharray="6 3" />
   333→        <text x={width - 65} y={16} fill="#9ca3af" fontSize={10}>7-day MA</text>
   334→      </svg>
   335→    </div>
   336→  );
   337→}

   435→// ── Threat Feed ────────────────────────────────────────────────────────
   436→
   437→function ThreatFeed({ threats }: { threats: string[] }) {
   438→  const [activeIndex, setActiveIndex] = useState(0);
   439→  const intervalRef = useRef<NodeJS.Timeout | null>(null);
   440→
   441→  useEffect(() => {
   442→    intervalRef.current = setInterval(() => {
   443→      setActiveIndex(prev => (prev + 1) % threats.length);
   444→    }, 5000);
   445→    return () => { if (intervalRef.current) clearInterval(intervalRef.current); };
   446→  }, [threats.length]);
   447→
   448→  const severityIcons = [AlertTriangle, Zap, Shield, Activity, Radio];
   449→  const severityColors = ['#ff3355', '#ff8844', '#ffaa00', '#60a5fa', '#a78bfa'];
   450→
   451→  return (
   452→    <div className="space-y-2">
   453→      <AnimatePresence mode="wait">
   454→        {threats.map((threat, i) => (
   455→          <motion.div
   456→            key={`${i}-${activeIndex === i ? 'active' : 'inactive'}`}
   457→            initial={{ opacity: 0, x: 20 }}
   458→            animate={{ opacity: activeIndex === i ? 1 : 0.4, x: 0 }}
   459→            exit={{ opacity: 0, x: -20 }}
   460→            transition={{ duration: 0.4 }}
   461→            className={`flex items-start gap-3 p-3 rounded-lg border transition-all ${
   462→              activeIndex === i ? 'bg-gray-800/50 border-gray-700' : 'border-transparent'
   463→            }`}
   464→            onMouseEnter={() => { if (intervalRef.current) clearInterval(intervalRef.current); }}
   465→            onMouseLeave={() => {
   466→              intervalRef.current = setInterval(() => {
   467→                setActiveIndex(prev => (prev + 1) % threats.length);
   468→              }, 5000);
   469→            }}
   470→          >
   471→            <div className="mt-0.5 shrink-0">
   472→              {(() => { const C = severityIcons[i % severityIcons.length]; return <C size={16} style={{ color: severityColors[i % severityColors.length] }} />; })()}
   473→            </div>

   475→            <span className="text-[10px] text-gray-600 whitespace-nowrap ml-auto shrink-0">
   476→              {new Date(Date.now() - i * 3600000).toLocaleTimeString('en-US', { hour: '2-digit', minute: '2-digit' })}
   477→            </span>
   478→          </motion.div>
   479→        ))}
   480→      </AnimatePresence>
   481→      <div className="flex justify-center gap-1.5 pt-2">
   482→        {threats.map((_, i) => (

   485→          />
   486→        ))}
   487→      </div>
   488→    </div>
   489→  );
