   139→   139→class EngineeringMetrics:
   140→   140→    """Collect, track, and persist engineering metrics.
   141→   141→
   142→   142→    Tracks test pass rate, coverage trend, debt ratio, and complexity
   143→   143→    trend over time. Persists snapshots to JSON files at
   144→   144→    ``RECONPRO_HOME/metrics/``.
   145→   145→
   146→   146→    Parameters
   147→   147→    ----------
   148→   148→    repo_path : str | Path
   149→   149→        Root directory of the repository to analyze.
   150→   150→    metrics_dir : Path | None
   151→   151→        Directory for persisted metrics. Defaults to ``METRICS_DIR``.
   152→   152→    """
   153→   153→
   154→   154→    def __init__(
   155→   155→        self,
   156→   156→        repo_path: str | Path = ".",
   157→   157→        metrics_dir: Optional[Path] = None,
   158→   158→    ) -> None:
   159→   159→        self._repo = Path(repo_path).resolve()
   160→   160→        self._metrics_dir = metrics_dir or METRICS_DIR
   161→   161→        self._metrics_dir.mkdir(parents=True, exist_ok=True)
   162→   162→        self._history_file = self._metrics_dir / "engineering_metrics.json"
   163→   163→        self._snapshots: List[MetricsSnapshot] = []
   164→   164→        self._load_history()
   165→   165→
   166→   166→    # -- Persistence -----------------------------------------------------
   167→   167→
   168→   168→    def _load_history(self) -> None:
   169→   169→        """Load historical metrics from disk."""
   170→   170→        if not self._history_file.exists():
   171→   171→            return
   172→   172→        try:
   173→   173→            with open(self._history_file, "r", encoding="utf-8") as f:
   174→   174→                data = json.load(f)
   175→   175→            raw_list = data.get("snapshots", [])
   176→   176→            self._snapshots = [MetricsSnapshot.from_dict(s) for s in raw_list]
   177→   177→        except (json.JSONDecodeError, OSError, KeyError, TypeError):
   178→   178→            logger.debug("Could not load metrics history", exc_info=True)
   179→   179→            self._snapshots = []
   180→   180→
   181→   181→    def _save_history(self) -> None:
   182→   182→        """Persist current metrics to disk."""
   183→   183→        try:
   184→   184→            data = {
   185→   185→                "repo_path": str(self._repo),
   186→   186→                "updated": datetime.now(timezone.utc).isoformat(),
   187→   187→                "total_snapshots": len(self._snapshots),
   188→   188→                "snapshots": [s.to_dict() for s in self._snapshots],
   189→   189→            }
   190→   190→            with open(self._history_file, "w", encoding="utf-8") as f:
   191→   191→                json.dump(data, f, indent=2, ensure_ascii=False)
   192→   192→        except OSError:
   193→   193→            logger.debug("Could not save metrics history", exc_info=True)
   194→   194→
   195→   195→    # -- Metric Collection ------------------------------------------------
   196→   196→
   197→   197→    def collect_current_metrics(self) -> MetricsSnapshot:
   198→   198→        """Analyze the repository and return a fresh metrics snapshot.
   199→   199→
   200→   200→        Walks the repository tree, counts files and lines, estimates
   201→   201→        cyclomatic complexity for Python files, and calculates a
   202→   202→        synthetic debt ratio.
   203→   203→        """
   204→   204→        t0 = time.perf_counter()
   205→   205→        file_count = 0
   206→   206→        total_lines = 0
   207→   207→        source_lines = 0
   208→   208→        complexity_values: List[float] = []
   209→   209→        debt_markers = 0
   210→   210→
   211→   211→        for fpath in self._walk_source_files():
   212→   212→            file_count += 1
   213→   213→            try:
   214→   214→                content = fpath.read_text(encoding="utf-8", errors="ignore")
   215→   215→            except OSError:
   216→   216→                continue
   217→   217→
   218→   218→            lines = content.splitlines()
   219→   219→            line_count = len(lines)
   220→   220→            total_lines += line_count
   221→   221→
   222→   222→            if fpath.suffix in _SOURCE_EXTENSIONS:
   223→   223→                source_lines += line_count
   224→   224→
   225→   225→                # Estimate complexity for Python files
   226→   226→                if fpath.suffix == ".py":
   227→   227→                    cx = self._estimate_complexity(lines)
   228→   228→                    complexity_values.append(cx)
   229→   229→
   230→   230→                    # Count debt markers (TODO, FIXME, HACK, XXX)
   231→   231→                    for line in lines:
   232→   232→                        stripped = line.strip().upper()
   233→   233→                        if stripped.startswith(("# TODO", "# FIXME", "# HACK", "# XXX")):
   234→   234→                            debt_markers += 1
   235→   235→
   236→   236→        # Calculate aggregate metrics
   237→   237→        complexity_avg = (
   238→   238→            sum(complexity_values) / len(complexity_values)
   239→   239→            if complexity_values else 0.0
   240→   240→        )
   241→   241→        complexity_max = max(complexity_values) if complexity_values else 0.0
   242→   242→        debt_ratio = (
   243→   243→            (debt_markers / source_lines * 100.0)
   244→   244→            if source_lines > 0 else 0.0
   245→   245→        )
   246→   246→
   247→   247→        # Cap debt_ratio at 100
   248→   248→        debt_ratio = min(debt_ratio, 100.0)
   249→   249→
   250→   250→        elapsed_ms = (time.perf_counter() - t0) * 1000.0
   251→   251→
   252→   252→        snapshot = MetricsSnapshot(
   253→   253→            timestamp=datetime.now(timezone.utc).isoformat(),
   254→   254→            debt_ratio=debt_ratio,
   255→   255→            debt_items=debt_markers,
   256→   256→            complexity_avg=complexity_avg,
   257→   257→            complexity_max=complexity_max,
   258→   258→            file_count=file_count,
   259→   259→            total_lines=total_lines,
   260→   260→            source_lines=source_lines,
   261→   261→            pipeline_duration_ms=elapsed_ms,
   262→   262→        )
   263→   263→
   264→   264→        return snapshot
   265→   265→
   266→   266→    def record_snapshot(self, snapshot: MetricsSnapshot) -> None:
   267→   267→        """Append a snapshot to history and persist."""
   268→   268→        self._snapshots.append(snapshot)
   269→   269→        # Keep last 100 snapshots to prevent unbounded growth
   270→   270→        if len(self._snapshots) > 100:
   271→   271→            self._snapshots = self._snapshots[-100:]
   272→   272→        self._save_history()

   304→   304→    # -- Helpers ----------------------------------------------------------
   305→   305→
   306→   306→    def _walk_source_files(self) -> List[Path]:
   307→   307→        """Yield all source files in the repository, skipping _SKIP_DIRS."""
   308→   308→        files: List[Path] = []
   309→   309→        if not self._repo.is_dir():
   310→   310→            return files
   311→   311→        try:
   312→   312→            for root, dirs, filenames in os.walk(self._repo):
   313→   313→                # Prune skipped directories in-place
   314→   314→                dirs[:] = [
   315→   315→                    d for d in dirs
   316→   316→                    if d not in _SKIP_DIRS and not d.startswith(".")
   317→   317→                    if d not in {"__pycache__", "node_modules"}
   318→   318→                ]
   319→   319→                dirs[:] = [
   320→   320→                    d for d in dirs
   321→   321→                    if d not in _SKIP_DIRS
   322→   322→                ]
   323→   323→                for fname in sorted(filenames):
   324→   324→                    fpath = Path(root) / fname
   325→   325→                    if fpath.suffix in _SOURCE_EXTENSIONS:
   326→   326→                        files.append(fpath)
   327→   327→        except OSError:
   328→   328→            logger.debug("Error walking repository", exc_info=True)
   329→   329→        return files

   331→   331→    @staticmethod
   332→   332→    def _estimate_complexity(lines: List[str]) -> float:
   333→   333→        """Estimate cyclomatic complexity for a Python file.
   334→   334→
   335→   335→        Counts branch-inducing keywords: ``if``, ``elif``, ``for``,
   336→   336→        ``while``, ``except``, ``with``, ``and``, ``or``. Adds 1 for
   337→   337→        the base path.
   338→   338→
   339→   339→        Ignores lines inside comments and strings (best-effort heuristic).
   340→   340→        """
   341→   341→        complexity = 1.0  # Base path
   342→   342→        branch_keywords = re.compile(
   343→   343→            r"\b(if|elif|for|while|except|with)\b|\b(and|or)\b"
   344→   344→        )
   345→   345→        in_multiline_string = False
   346→   346→        for line in lines:
   347→   347→            stripped = line.strip()
   348→   348→            # Simple heuristic for triple-quoted strings
   349→   349→            triple_count = stripped.count('"""') + stripped.count("'''")
   350→   350→            if triple_count % 2 != 0:
   351→   351→                in_multiline_string = not in_multiline_string
   352→   352→            if in_multiline_string:
   353→   353→                continue
   354→   354→            # Skip comment-only lines
   355→   355→            if stripped.startswith("#"):
   356→   356→                continue
   357→   357→            complexity += len(branch_keywords.findall(stripped))
   358→   358→        return complexity

   488→   488→class EngineeringBaseline:
   489→   489→    """Manage engineering baselines for a repository.
   490→   490→
   491→   491→    Captures snapshots of repository state (file manifests with
   492→   492→    hashes and complexity estimates), compares current state against
   493→   493→    baselines, and tracks regression status.
   494→   494→
   495→   495→    Baselines are persisted to ``RECONPRO_HOME/baselines/<name>.json``.
   496→   496→
   497→   497→    Parameters
   498→   498→    ----------
   499→   499→    repo_path : str | Path
   500→   500→        Root directory of the repository.
   501→   501→    baseline_dir : Path | None
   502→   502→        Directory for persisted baselines.
   503→   503→    """
   504→   504→
   505→   505→    def __init__(
   506→   506→        self,
   507→   507→        repo_path: str | Path = ".",
   508→   508→        baseline_dir: Optional[Path] = None,
   509→   509→    ) -> None:
   510→   510→        self._repo = Path(repo_path).resolve()
   511→   511→        self._baseline_dir = baseline_dir or BASELINE_DIR
   512→   512→        self._baseline_dir.mkdir(parents=True, exist_ok=True)
   513→   513→
   514→   514→    # -- Public API -------------------------------------------------------
   515→   515→
   516→   516→    def capture_snapshot(self, name: str = "") -> BaselineSnapshot:
   517→   517→        """Capture a baseline snapshot of the current repository state.
   518→   518→
   519→   519→        Parameters
   520→   520→        ----------
   521→   521→        name : str
   522→   522→            Baseline name. Defaults to ``"auto_<timestamp>"``.
   523→   523→
   524→   524→        Returns
   525→   525→        -------
   526→   526→        BaselineSnapshot
   527→   527→            The captured baseline.
   528→   528→        """
   529→   529→        if not name:
   530→   530→            name = f"auto_{datetime.now(timezone.utc).strftime('%Y%m%d_%H%M%S')}"
   531→   531→
   532→   532→        fingerprints: List[FileFingerprint] = []
   533→   533→        total_lines = 0
   534→   534→        total_size = 0
   535→   535→        complexity_values: List[float] = []
   536→   536→        hash_parts: List[str] = []
   537→   537→
   538→   538→        metrics = EngineeringMetrics(repo_path=self._repo)
   539→   539→        source_files = metrics._walk_source_files()
   540→   540→
   541→   541→        for fpath in source_files:
   542→   542→            rel_path = str(fpath.relative_to(self._repo))
   543→   543→            try:
   544→   544→                content = fpath.read_bytes()
   545→   545→            except OSError:
   546→   546→                continue
   547→   547→
   548→   548→            sha = hashlib.sha256(content).hexdigest()
   549→   549→            text_content = content.decode("utf-8", errors="ignore")
   550→   550→            lines = text_content.splitlines()
   551→   551→            line_count = len(lines)
   552→   552→
   553→   553→            complexity = 0.0
   554→   554→            if fpath.suffix == ".py":
   555→   555→                complexity = EngineeringMetrics._estimate_complexity(lines)
   556→   556→
   557→   557→            fp = FileFingerprint(
   558→   558→                path=rel_path,
   559→   559→                size_bytes=len(content),
   560→   560→                line_count=line_count,
   561→   561→                sha256=sha,
   562→   562→                complexity=complexity,
   563→   563→            )
   564→   564→            fingerprints.append(fp)
   565→   565→            total_lines += line_count
   566→   566→            total_size += len(content)
   567→   567→            if complexity > 0:
   568→   568→                complexity_values.append(complexity)
   569→   569→            hash_parts.append(f"{rel_path}:{sha[:16]}")
   570→   570→
   571→   571→        avg_complexity = (
   572→   572→            sum(complexity_values) / len(complexity_values)
   573→   573→            if complexity_values else 0.0
   574→   574→        )
   575→   575→        max_complexity = max(complexity_values) if complexity_values else 0.0
   576→   576→        composite_hash = hashlib.sha256(
   577→   577→            "\n".join(sorted(hash_parts)).encode()
   578→   578→        ).hexdigest()[:32]
   579→   579→
   580→   580→        snapshot = BaselineSnapshot(
   581→   581→            name=name,
   582→   582→            timestamp=datetime.now(timezone.utc).isoformat(),
   583→   583→            repo_path=str(self._repo),
   584→   584→            files=sorted(fingerprints, key=lambda f: f.path),
   585→   585→            total_files=len(fingerprints),
   586→   586→            total_lines=total_lines,
   587→   587→            total_size_bytes=total_size,
   588→   588→            avg_complexity=avg_complexity,
   589→   589→            max_complexity=max_complexity,
   590→   590→            composite_hash=composite_hash,
   591→   591→        )
   592→   592→
   593→   593→        self._save_baseline(snapshot)
   594→   594→        return snapshot
   595→   595→
   596→   596→    def compare(
   597→   597→        self,
   598→   598→        baseline_name: Optional[str] = None,
   599→   599→        baseline: Optional[BaselineSnapshot] = None,
   600→   600→    ) -> DriftReport:
   601→   601→        """Compare current repository state against a baseline.
   602→   602→
   603→   603→        Parameters
   604→   604→        ----------
   605→   605→        baseline_name : str | None
   606→   606→            Name of a previously saved baseline.
   607→   607→        baseline : BaselineSnapshot | None
   608→   608→            An in-memory baseline object. Takes precedence over
   609→   609→            *baseline_name*.
   610→   610→
   611→   611→        Returns
   612→   612→        -------
   613→   613→        DriftReport
   614→   614→            Detailed comparison results.
   615→   615→        """
   616→   616→        if baseline is None:
   617→   617→            if baseline_name is None:
   618→   618→                # Use the most recent baseline
   619→   619→                baseline = self._load_latest_baseline()
   620→   620→            else:
   621→   621→                baseline = self.load_baseline(baseline_name)
   622→   622→
   623→   623→        if baseline is None:
   624→   624→            return DriftReport(
   625→   625→                baseline_name=baseline_name or "",
   626→   626→                current_timestamp=datetime.now(timezone.utc).isoformat(),
   627→   627→                regression_details=["No baseline found for comparison"],
   628→   628→            )
   629→   629→
   630→   630→        # Build current state index
   631→   631→        current_fps: Dict[str, FileFingerprint] = {}
   632→   632→        metrics = EngineeringMetrics(repo_path=self._repo)
   633→   633→        for fpath in metrics._walk_source_files():
   634→   634→            rel_path = str(fpath.relative_to(self._repo))
   635→   635→            try:
   636→   636→                content = fpath.read_bytes()
   637→   637→            except OSError:
   638→   638→                continue
   639→   639→            text_content = content.decode("utf-8", errors="ignore")
   640→   640→            lines = text_content.splitlines()
   641→   641→            complexity = 0.0
   642→   642→            if fpath.suffix == ".py":
   643→   643→                complexity = EngineeringMetrics._estimate_complexity(lines)
   644→   644→            current_fps[rel_path] = FileFingerprint(
   645→   645→                path=rel_path,
   646→   646→                size_bytes=len(content),
   647→   647→                line_count=len(lines),
   648→   648→                sha256=hashlib.sha256(content).hexdigest(),
   649→   649→                complexity=complexity,
   650→   650→            )
   651→   651→
   652→   652→        # Build baseline index
   653→   653→        baseline_map: Dict[str, FileFingerprint] = {
   654→   654→            f.path: f for f in baseline.files
   655→   655→        }
   656→   656→
   657→   657→        current_paths = set(current_fps.keys())
   658→   658→        baseline_paths = set(baseline_map.keys())
   659→   659→
   660→   660→        added = sorted(current_paths - baseline_paths)
   661→   661→        removed = sorted(baseline_paths - current_paths)
   662→   662→        common = current_paths & baseline_paths
   663→   663→
   664→   664→        modified: List[str] = []
   665→   665→        drift_events: List[Dict[str, Any]] = []
   666→   666→        regression_details: List[str] = []
   667→   667→
   668→   668→        for path in sorted(common):
   669→   669→            cur = current_fps[path]
   670→   670→            base = baseline_map[path]
   671→   671→
   672→   672→            if cur.sha256 != base.sha256:
   673→   673→                modified.append(path)
   674→   674→                event: Dict[str, Any] = {
   675→   675→                    "path": path,
   676→   676→                    "type": "CONTENT_CHANGE",
   677→   677→                    "severity": "low",
   678→   678→                }
   679→   679→
   680→   680→                # Detect regression: significant complexity increase
   681→   681→                if cur.complexity > 0 and base.complexity > 0:
   682→   682→                    cx_delta = cur.complexity - base.complexity
   683→   683→                    if cx_delta > 10:
   684→   684→                        event["severity"] = "high"
   685→   685→                        event["detail"] = (
   686→   686→                            f"Complexity surged by {cx_delta:.1f} "
   687→   687→                            f"(from {base.complexity:.1f} to {cur.complexity:.1f})"
   688→   688→                        )
   689→   689→                        regression_details.append(event["detail"])
   690→   690→                    elif cx_delta > 5:
   691→   691→                        event["severity"] = "medium"
   692→   692→                        event["detail"] = (
   693→   693→                            f"Complexity increased by {cx_delta:.1f} "
   694→   694→                            f"(from {base.complexity:.1f} to {cur.complexity:.1f})"
   695→   695→                        )
   696→   696→
   697→   697→                # Detect regression: significant line count increase
   698→   698→                line_delta = cur.line_count - base.line_count
   699→   699→                if line_delta > 100:
   700→   700→                    event["severity"] = max(
   701→   701→                        event.get("severity", "low"), "medium"
   702→   702→                    )
   703→   703→                    if "detail" not in event:
   704→   704→                        event["detail"] = (
   705→   705→                            f"File grew by {line_delta} lines"
   706→   706→                        )
   707→   707→
   708→   708→                drift_events.append(event)
   709→   709→
   710→   710→        # New files add complexity
   711→   711→        for path in added:
   712→   712→            cur = current_fps[path]
   713→   713→            if cur.complexity > 15:
   714→   714→                regression_details.append(
   715→   715→                    f"New file {path} has high complexity ({cur.complexity:.1f})"
   716→   716→                )
   717→   717→
   718→   718→        # Removed files might be regression (deleted tests, docs)
   719→   719→        for path in removed:
   720→   720→            base = baseline_map[path]
   721→   721→            if "test" in path.lower():
   722→   722→                regression_details.append(
   723→   723→                    f"Test file removed: {path}"
   724→   724→                )
   725→   725→
   726→   726→        # Calculate drift score
   727→   727→        total_changes = len(added) + len(removed) + len(modified)
   728→   728→        drift_score = self._calculate_drift_score(
   729→   729→            added=added,
   730→   730→            removed=removed,
   731→   731→            modified=modified,
   732→   732→            drift_events=drift_events,
   733→   733→            regression_details=regression_details,
   734→   734→        )
   735→   735→
   736→   736→        return DriftReport(
   737→   737→            baseline_name=baseline.name,
   738→   738→            baseline_timestamp=baseline.timestamp,
   739→   739→            current_timestamp=datetime.now(timezone.utc).isoformat(),
   740→   740→            files_added=added,
   741→   741→            files_removed=removed,
   742→   742→            files_modified=modified,
   743→   743→            drift_events=drift_events,
   744→   744→            drift_score=drift_score,
   745→   745→            regression_detected=len(regression_details) > 0,
   746→   746→            regression_details=regression_details,
   747→   747→        )
   748→   748→
   749→   749→    def load_baseline(self, name: str) -> Optional[BaselineSnapshot]:
   750→   750→        """Load a named baseline from disk."""
   751→   751→        path = self._baseline_dir / f"{name}.json"
   752→   752→        if not path.exists():
   753→   753→            return None
   754→   754→        try:
   755→   755→            with open(path, "r", encoding="utf-8") as f:
   756→   756→                data = json.load(f)
   757→   757→            return BaselineSnapshot.from_dict(data)
   758→   758→        except (json.JSONDecodeError, OSError, KeyError, TypeError):
   759→   759→            logger.debug("Could not load baseline %s", name, exc_info=True)
   760→   760→            return None
   761→   761→
   762→   762→    def list_baselines(self) -> List[Dict[str, str]]:
   763→   763→        """List all saved baselines with name and timestamp."""
   764→   764→        results: List[Dict[str, str]] = []
   765→   765→        if not self._baseline_dir.exists():
   766→   766→            return results
   767→   767→        for fpath in sorted(self._baseline_dir.glob("*.json")):
   768→   768→            baseline = self.load_baseline(fpath.stem)
   769→   769→            if baseline:
   770→   770→                results.append({
   771→   771→                    "name": baseline.name,
   772→   772→                    "timestamp": baseline.timestamp,
   773→   773→                    "total_files": str(baseline.total_files),
   774→   774→                    "total_lines": str(baseline.total_lines),
   775→   775→                })
   776→   776→        return results
   777→   777→
   778→   778→    def delete_baseline(self, name: str) -> bool:
   779→   779→        """Delete a saved baseline. Returns True on success."""
   780→   780→        path = self._baseline_dir / f"{name}.json"
   781→   781→        if not path.exists():
   782→   782→            return False
   783→   783→        try:
   784→   784→            path.unlink()
   785→   785→            return True
   786→   786→        except OSError:
   787→   787→            return False
   788→   788→
   789→   789→    # -- Internal ---------------------------------------------------------
   790→   790→
   791→   791→    def _save_baseline(self, snapshot: BaselineSnapshot) -> None:
   792→   792→        """Persist a baseline to disk."""
   793→   793→        path = self._baseline_dir / f"{snapshot.name}.json"
   794→   794→        try:
   795→   795→            with open(path, "w", encoding="utf-8") as f:
   796→   796→                json.dump(snapshot.to_dict(), f, indent=2, ensure_ascii=False)
   797→   797→        except OSError:
   798→   798→            logger.debug("Could not save baseline %s", snapshot.name, exc_info=True)
   799→   799→
   800→   800→    def _load_latest_baseline(self) -> Optional[BaselineSnapshot]:
   801→   801→        """Load the most recently created baseline."""
   802→   802→        baselines = self.list_baselines()
   803→   803→        if not baselines:
   804→   804→            return None
   805→   805→        # list_baselines returns sorted by filename; get the last one
   806→   806→        latest = baselines[-1]
   807→   807→        return self.load_baseline(latest["name"])
   808→   808→
   809→   809→    @staticmethod
   810→   810→    def _calculate_drift_score(
   811→   811→        added: List[str],
   812→   812→        removed: List[str],
   813→   813→        modified: List[str],
   814→   814→        drift_events: List[Dict[str, Any]],
   815→   815→        regression_details: List[str],
   816→   816→    ) -> float:
   817→   817→        """Calculate a 0-100 drift score.
   818→   818→
   819→   819→        Higher score = more drift. Regression indicators boost the score.
   820→   820→        """
   821→   821→        if not added and not removed and not modified:
   822→   822→            return 0.0
   823→   823→
   824→   824→        # Base score from change volume
   825→   825→        total_changes = len(added) + len(removed) + len(modified)
   826→   826→        volume_score = min(total_changes * 2.0, 40.0)
   827→   827→
   828→   828→        # Severity weighting
   829→   829→        severity_weights = {"low": 1.0, "medium": 2.0, "high": 3.0, "critical": 5.0}
   830→   830→        event_score = 0.0
   831→   831→        for event in drift_events:
   832→   832→            sev = event.get("severity", "low")
   833→   833→            event_score += severity_weights.get(sev, 1.0)
   834→   834→        event_score = min(event_score, 30.0)
   835→   835→
   836→   836→        # Regression penalty
   837→   837→        regression_penalty = min(len(regression_details) * 5.0, 30.0)
   838→   838→
   839→   839→        return min(volume_score + event_score + regression_penalty, 100.0)

   910→   910→    def health_check(self) -> Dict[str, Any]:
   911→   911→        """Run repository health diagnostics.
   912→   912→
   913→   913→        Delegates to ``diagnostics.health_check()``,
   914→   914→        ``diagnostics.validate_config()``, and
   915→   915→        ``diagnostics.run_diagnostics()`` for comprehensive results.
   916→   916→
   917→   917→        Returns
   918→   918→        -------
   919→   919→        dict
   920→   920→            Structured health report with status per category,
   921→   921→            config validation issues, and summary statistics.
   922→   922→        """
   923→   923→        result: Dict[str, Any] = {
   924→   924→            "timestamp": datetime.now(timezone.utc).isoformat(),
   925→   925→            "repo_path": str(self._repo),
   926→   926→        }
   927→   927→
   928→   928→        # 1. Quick health check from diagnostics module
   929→   929→        try:
   930→   930→            from .diagnostics import health_check as diag_health
   931→   931→            result["health"] = diag_health()
   932→   932→        except Exception as exc:
   933→   933→            logger.debug("Health check error: %s", exc, exc_info=True)
   934→   934→            result["health"] = {"error": str(exc)}
   935→   935→
   936→   936→        # 2. Config validation from diagnostics module
   937→   937→        try:
   938→   938→            from .diagnostics import validate_config
   939→   939→            result["config_issues"] = validate_config()

   941→   941→            logger.debug("Config validation error: %s", exc, exc_info=True)

   966→   966→    def drift_detection(
   967→   967→        self,
   968→   968→        baseline_name: Optional[str] = None,
   969→   969→        create_baseline_if_missing: bool = True,
   970→   970→    ) -> Dict[str, Any]:
   971→   971→        """Detect code drift from baselines.
   972→   972→
   973→   973→        Compares the current repository state against the most recent
   974→   974→        baseline (or a named one). If no baseline exists and
   975→   975→        *create_baseline_if_missing* is True, captures one automatically.
   976→   976→
   977→   977→        Parameters
   978→   978→        ----------
   979→   979→        baseline_name : str | None
   980→   980→            Specific baseline to compare against.

   982→   982→            Auto-capture a baseline if none exists.
   983→   983→
   984→   984→        Returns
   985→   985→        -------

   988→   988→            and regression details.
   989→   989→        """
   990→   990→        result: Dict[str, Any] = {
   991→   991→            "timestamp": datetime.now(timezone.utc).isoformat(),
   992→   992→            "repo_path": str(self._repo),
   993→   993→        }
   994→   994→
   995→   995→        # Check infrastructure drift monitor availability
   996→   996→        try:
   997→   997→            from .drift_monitor import InfrastructureDriftMonitor
   998→   998→            result["infra_drift_available"] = True
   999→   999→        except ImportError:
  1000→  1000→            result["infra_drift_available"] = False
  1001→  1001→
  1002→  1002→        # Check if we have any baselines

  1005→  1005→            snapshot = self._baseline.capture_snapshot(
  1006→  1006→                name=f"initial_{datetime.now(timezone.utc).strftime('%Y%m%d_%H%M%S')}"
  1007→  1007→            )
  1008→  1008→            result["auto_baseline_created"] = True
  1009→  1009→            result["baseline"] = snapshot.to_dict()
  1010→  1010→            result["drift"] = {
  1011→  1011→                "drift_score": 0.0,
  1012→  1012→                "message": "Initial baseline captured; no drift to report",
  1013→  1013→            }
  1014→  1014→            return result
  1015→  1015→
  1016→  1016→        # Run drift comparison
  1017→  1017→        drift_report = self._baseline.compare(baseline_name=baseline_name)
  1018→  1018→        result["drift"] = drift_report.to_dict()
  1019→  1019→
  1020→  1020→        return result

  1022→  1022→    def quality_gate(
  1023→  1023→        self,
  1024→  1024→        metrics_snapshot: Optional[MetricsSnapshot] = None,
  1025→  1025→        drift_report: Optional[DriftReport] = None,
  1026→  1026→        health_result: Optional[Dict[str, Any]] = None,
  1027→  1027→    ) -> QualityGateResult:
  1028→  1028→        """Enforce quality thresholds before merge.
  1029→  1029→
  1030→  1030→        Evaluates metrics, drift, and health against configured
  1031→  1031→        thresholds. Returns a detailed pass/fail result.
  1032→  1032→
  1033→  1033→        Parameters
  1034→  1034→        ----------
  1035→  1035→        metrics_snapshot : MetricsSnapshot | None
  1036→  1036→            Current metrics. Collected automatically if None.
  1037→  1037→        drift_report : DriftReport | None
  1038→  1038→            Drift comparison result. Run automatically if None.
  1039→  1039→        health_result : dict | None
  1040→  1040→            Health check result. Run automatically if None.
  1041→  1041→
  1042→  1042→        Returns
  1043→  1043→        -------
  1044→  1044→        QualityGateResult
  1045→  1045→            Pass/fail result with per-threshold evaluations.
  1046→  1046→        """
  1047→  1047→        # Collect data if not provided
  1048→  1048→        if metrics_snapshot is None:
  1049→  1049→            metrics_snapshot = self._metrics.collect_current_metrics()
  1050→  1050→
  1051→  1051→        if drift_report is None:
  1052→  1052→            drift_report = self._baseline.compare()
  1053→  1053→
  1054→  1054→        if health_result is None:
  1055→  1055→            health_result = self.health_check()
  1056→  1056→
  1057→  1057→        evaluations: Dict[str, Dict[str, Any]] = {}
  1058→  1058→        failures: List[str] = []
  1059→  1059→        warnings: List[str] = []
  1060→  1060→        score_deductions = 0.0
  1061→  1061→
  1062→  1062→        # 1. Test pass rate gate
  1063→  1063→        test_rate = metrics_snapshot.test_pass_rate
  1064→  1064→        min_test_rate = self._thresholds.get("min_test_pass_rate", 80.0)
  1065→  1065→        test_pass = test_rate >= min_test_rate or test_rate == 0.0  # 0 = no tests run, not a failure
  1066→  1066→        evaluations["test_pass_rate"] = {
  1067→  1067→            "value": test_rate,
  1068→  1068→            "threshold": min_test_rate,
  1069→  1069→            "passed": test_pass,
  1070→  1070→        }
  1071→  1071→        if not test_pass and test_rate > 0:
  1072→  1072→            failures.append(
  1073→  1073→                f"Test pass rate {test_rate:.1f}% below threshold {min_test_rate:.1f}%"
  1074→  1074→            )
  1075→  1075→            score_deductions += 25.0
  1076→  1076→
  1077→  1077→        # 2. Debt ratio gate
  1078→  1078→        debt = metrics_snapshot.debt_ratio
  1079→  1079→        max_debt = self._thresholds.get("max_debt_ratio", 30.0)
  1080→  1080→        debt_pass = debt <= max_debt
  1081→  1081→        evaluations["debt_ratio"] = {
  1082→  1082→            "value": debt,
  1083→  1083→            "threshold": max_debt,
  1084→  1084→            "passed": debt_pass,
  1085→  1085→        }
  1086→  1086→        if not debt_pass:
  1087→  1087→            failures.append(
  1088→  1088→                f"Debt ratio {debt:.1f}% exceeds threshold {max_debt:.1f}%"
  1089→  1089→            )
  1090→  1090→            score_deductions += 20.0
  1091→  1091→        elif debt > max_debt * 0.7:
  1092→  1092→            warnings.append(
  1093→  1093→                f"Debt ratio {debt:.1f}% approaching threshold {max_debt:.1f}%"
  1094→  1094→            )
  1095→  1095→
  1096→  1096→        # 3. Complexity gate
  1097→  1097→        complexity = metrics_snapshot.complexity_max
  1098→  1098→        max_cx = self._thresholds.get("max_complexity", 20.0)
  1099→  1099→        cx_pass = complexity <= max_cx
  1100→  1100→        evaluations["complexity"] = {
  1101→  1101→            "value": complexity,
  1102→  1102→            "threshold": max_cx,
  1103→  1103→            "passed": cx_pass,
  1104→  1104→        }
  1105→  1105→        if not cx_pass:
  1106→  1106→            failures.append(
  1107→  1107→                f"Max complexity {complexity:.1f} exceeds threshold {max_cx:.1f}"
  1108→  1108→            )
  1109→  1109→            score_deductions += 15.0
  1110→  1110→
  1111→  1111→        # 4. Drift score gate
  1112→  1112→        drift_score = drift_report.drift_score
  1113→  1113→        max_drift = self._thresholds.get("max_drift_score", 50.0)
  1114→  1114→        drift_pass = drift_score <= max_drift
  1115→  1115→        evaluations["drift_score"] = {
  1116→  1116→            "value": drift_score,
  1117→  1117→            "threshold": max_drift,
  1118→  1118→            "passed": drift_pass,
  1119→  1119→        }
  1120→  1120→        if not drift_pass:
  1121→  1121→            failures.append(
  1122→  1122→                f"Drift score {drift_score:.1f} exceeds threshold {max_drift:.1f}"
  1123→  1123→            )
  1124→  1124→            score_deductions += 20.0
  1125→  1125→        elif drift_score > max_drift * 0.7:
  1126→  1126→            warnings.append(
  1127→  1127→                f"Drift score {drift_score:.1f} approaching threshold {max_drift:.1f}"
  1128→  1128→            )
  1129→  1129→
  1130→  1130→        # 5. Drift events volume gate
  1131→  1131→        total_drift_events = (
  1132→  1132→            len(drift_report.files_added)
  1133→  1133→            + len(drift_report.files_removed)
  1134→  1134→            + len(drift_report.files_modified)
  1135→  1135→        )
  1136→  1136→        max_events = self._thresholds.get("max_drift_events", 10)
  1137→  1137→        events_pass = total_drift_events <= max_events
  1138→  1138→        evaluations["drift_events"] = {
  1139→  1139→            "value": total_drift_events,
  1140→  1140→            "threshold": max_events,
  1141→  1141→            "passed": events_pass,
  1142→  1142→        }
  1143→  1143→        if not events_pass:
  1144→  1144→            failures.append(
  1145→  1145→                f"Drift events ({total_drift_events}) exceed threshold ({max_events})"
  1146→  1146→            )
  1147→  1147→            score_deductions += 10.0
  1148→  1148→
  1149→  1149→        # 6. Health check gate
  1150→  1150→        health_score = health_result.get("health_score", 0.0)
  1151→  1151→        min_health = self._thresholds.get("min_health_checks_passed", 0.7)
  1152→  1152→        health_pass = health_score >= min_health
  1153→  1153→        evaluations["health_checks"] = {
  1154→  1154→            "value": health_score,
  1155→  1155→            "threshold": min_health,
  1156→  1156→            "passed": health_pass,
  1157→  1157→        }
  1158→  1158→        if not health_pass:
  1159→  1159→            failures.append(
  1160→  1160→                f"Health score {health_score:.1f}% below threshold {min_health:.1f}%"
  1161→  1161→            )
  1162→  1162→            score_deductions += 10.0
  1163→  1163→
  1164→  1164→        # 7. Regression check
  1165→  1165→        if drift_report.regression_detected:
  1166→  1166→            failures.append(
  1167→  1167→                f"Regression detected: {len(drift_report.regression_details)} issue(s)"
  1168→  1168→            )
  1169→  1169→            score_deductions += 15.0
  1170→  1170→            for detail in drift_report.regression_details:
  1171→  1171→                warnings.append(f"Regression: {detail}")
  1172→  1172→
  1173→  1173→        # Calculate aggregate score
  1174→  1174→        score = max(0.0, 100.0 - score_deductions)
  1175→  1175→        passed = len(failures) == 0
  1176→  1176→
  1177→  1177→        return QualityGateResult(
  1178→  1178→            passed=passed,
  1179→  1179→            score=score,
  1180→  1180→            thresholds=dict(self._thresholds),
  1181→  1181→            evaluations=evaluations,
  1182→  1182→            failures=failures,
  1183→  1183→            warnings=warnings,
  1184→  1184→        )

  1186→  1186→    def engineering_workflow(
  1187→  1187→        self,

  1190→  1190→    ) -> Dict[str, Any]:
  1191→  1191→        """Run the full engineering pipeline.
  1192→  1192→
  1193→  1193→        Executes health check, drift detection, metrics collection,
  1194→  1194→        and quality gate evaluation in sequence.
  1195→  1195→
  1196→  1196→        Parameters
  1197→  1197→        ----------

  1199→  1199→            Auto-capture a baseline if none exists during drift detection.
  1200→  1200→        record_metrics : bool

  1202→  1202→
  1203→  1203→        Returns

  1210→  1210→        result: Dict[str, Any] = {
  1211→  1211→            "timestamp": datetime.now(timezone.utc).isoformat(),
  1212→  1212→            "repo_path": str(self._repo),
  1213→  1213→            "pipeline": "engineering_workflow",
  1214→  1214→        }
  1215→  1215→
  1216→  1216→        # Step 1: Health check
  1217→  1217→        t0 = time.perf_counter()

  1228→  1228→        )
  1229→  1229→        result["drift_detection"] = {
  1230→  1230→            "duration_ms": round((time.perf_counter() - t0) * 1000, 2),
  1231→  1231→            "result": drift,
  1232→  1232→        }
  1233→  1233→
  1234→  1234→        # Step 3: Metrics collection
  1235→  1235→        t0 = time.perf_counter()
  1236→  1236→        metrics_snapshot = self._metrics.collect_current_metrics()
  1237→  1237→        if record_metrics:
  1238→  1238→            self._metrics.record_snapshot(metrics_snapshot)

  1242→  1242→            "trends": {
  1243→  1243→                "coverage": self._metrics.get_coverage_trend(),
  1244→  1244→                "debt": self._metrics.get_debt_trend(),
  1245→  1245→                "complexity": self._metrics.get_complexity_trend(),

  1247→  1247→            },
  1248→  1248→        }
  1249→  1249→
  1250→  1250→        # Step 4: Quality gate
  1251→  1251→        t0 = time.perf_counter()
  1252→  1252→        drift_report = DriftReport()
  1253→  1253→        drift_data = drift.get("drift", {})
  1254→  1254→        if isinstance(drift_data, dict):
  1255→  1255→            drift_report = DriftReport(
  1256→  1256→                baseline_name=drift_data.get("baseline_name", ""),
  1257→  1257→                baseline_timestamp=drift_data.get("baseline_timestamp", ""),
  1258→  1258→                current_timestamp=drift_data.get("current_timestamp", ""),

  1264→  1264→                regression_detected=drift_data.get("regression_detected", False),
  1265→  1265→                regression_details=drift_data.get("regression_details", []),
  1266→  1266→            )
  1267→  1267→
  1268→  1268→        gate = self.quality_gate(
  1269→  1269→            metrics_snapshot=metrics_snapshot,
  1270→  1270→            drift_report=drift_report,

  1273→  1273→        result["quality_gate"] = {

  1286→  1286→            "regression_detected": drift_report.regression_detected,

  1289→  1289→
  1290→  1290→        # Emit observability event via plugin hook if available

  1317→  1317→    def _compute_health_score(diag_result: Dict[str, Any]) -> float:
  1318→  1318→        """Compute a 0-100 health score from diagnostics results."""
  1319→  1319→        total = diag_result.get("total_checks", 0)
  1320→  1320→        passed = diag_result.get("passed", 0)

  1325→  1325→    def _check_repository_health(self) -> Dict[str, Any]:
  1326→  1326→        """Run repository-specific health checks.
  1327→  1327→
  1328→  1328→        Checks:
  1329→  1329→        - Repository directory exists and is accessible
  1330→  1330→        - Source file count
  1331→  1331→        - Presence of test files
  1332→  1332→        - Presence of configuration files
  1333→  1333→        - Presence of documentation
  1334→  1334→        """
  1335→  1335→        checks: Dict[str, Any] = {}
  1336→  1336→
  1337→  1337→        # Directory exists
  1338→  1338→        checks["repo_exists"] = self._repo.is_dir()
  1339→  1339→
  1340→  1340→        # Count source files
  1341→  1341→        source_files = self._metrics._walk_source_files()
  1342→  1342→        checks["source_file_count"] = len(source_files)
  1343→  1343→
  1344→  1344→        # Test file detection
  1345→  1345→        test_files = [
  1346→  1346→            f for f in source_files
  1347→  1347→            if "test" in f.name.lower()
  1348→  1348→            or "tests" in f.parts
  1349→  1349→            or "test_" in f.name
  1350→  1350→            or f.name.startswith("test_")
  1351→  1351→            or f.name.endswith("_test.py")
  1352→  1352→        ]
  1353→  1353→        checks["test_file_count"] = len(test_files)
  1354→  1354→        checks["has_tests"] = len(test_files) > 0
  1355→  1355→
  1356→  1356→        # Configuration file detection
  1357→  1357→        config_patterns = [
  1358→  1358→            "setup.py", "setup.cfg", "pyproject.toml",
  1359→  1359→            "package.json", "Cargo.toml", "go.mod",
  1360→  1360→            "Makefile", "CMakeLists.txt", "build.gradle",
  1361→  1361→            ".github", ".gitlab-ci.yml", "Jenkinsfile",
  1362→  1362→        ]
  1363→  1363→        found_configs = []
  1364→  1364→        for pattern in config_patterns:
  1365→  1365→            if (self._repo / pattern).exists():
  1366→  1366→                found_configs.append(pattern)
  1367→  1367→        checks["config_files"] = found_configs
  1368→  1368→        checks["has_build_config"] = len(found_configs) > 0
  1369→  1369→
  1370→  1370→        # Documentation detection
  1371→  1371→        doc_patterns = ["README.md", "README.rst", "README.txt", "docs/"]
  1372→  1372→        found_docs = []
  1373→  1373→        for pattern in doc_patterns:
  1374→  1374→            if (self._repo / pattern).exists():
  1375→  1375→                found_docs.append(pattern)
  1376→  1376→        checks["documentation_files"] = found_docs
