   727→    def on_key(self, event: Key) -> None:
   728→        """Handle key events — boot skip + history navigation."""
   729→        # Skip boot on ANY key press during boot sequence
   730→        if not self._boot_done and not self._boot_skipped:
   731→            self._boot_skipped = True
   732→            try:
   733→                boot = self.query_one("#boot-screen", BootScreen)
   734→                if boot._timer:
   735→                    boot._timer.stop()
   736→            except NoMatches:
   737→                pass
   738→            self._finish_boot()
   739→            return
   740→
   741→        # Command history navigation
   742→        if event.key == "up" and self._history_index > 0:
   743→            try:
   744→                inp = self.query_one("#command-input", Input)
   745→                if inp.has_focus:
   746→                    self._history_index -= 1
   747→                    inp.value = self._command_history[self._history_index]
   748→            except NoMatches:
   749→                pass
   750→        elif event.key == "down":
   751→            try:
   752→                inp = self.query_one("#command-input", Input)
   753→                if inp.has_focus:
   754→                    if self._history_index < len(self._command_history) - 1:
   755→                        self._history_index += 1
   756→                        inp.value = self._command_history[self._history_index]
   757→                    else:
   758→                        self._history_index = len(self._command_history)
   759→                        inp.value = ""
   760→            except NoMatches:
   761→                pass

   822→    def watch_finding_count(self, old: int, new: int) -> None:

   824→        crit = sum(
   825→            1 for f in self._findings_list
   826→            if f.get("severity", "").lower() == "critical"

   833→            1 for f in self._findings_list
   834→            if f.get("severity", "").lower() == "medium"
   835→        )

   839→            self.query_one("#stat-findings", StatCounter).set(new, delta=delta)
   840→        except NoMatches:
   841→            pass

   849→            pass
   850→        try:
   851→            self.query_one("#stat-medium", StatCounter).set(med)
   852→        except NoMatches:
   853→            pass

   856→        now = time.monotonic()
   857→        for _ in range(delta):
   858→            self._finding_timestamps.append(now)

   860→        # Push to findings sparkline (true findings-per-minute)
   861→        try:
   862→            spark = self.query_one("#spark-findings", Sparkline)
   863→            # Calculate actual FPM from timestamps in the last 60s

   869→            pass

   871→        # Wire into velocity meter
   872→        try:
   873→            vm = self.query_one("#velocity-meter", VelocityMeter)

   875→                vm.record_finding(delta)

   880→        try:
   881→            el = self.query_one("#bottom-info", Label)

   883→                f"[{GREEN}]●[/] {new} findings  [{RED}]{crit}[/]C [{YELLOW}]{high}[/]H [{MUTED}]{med}[/]M"

  1031→    def _process_command(self, raw: str) -> None:
  1032→        """Parse and execute a user command."""
  1033→        self._chat(f"[{CYAN} bold]▸ {raw}[/]")
  1034→
  1035→        parts = raw.split()
  1036→        cmd = parts[0].lower()
  1037→        args = parts[1:]
  1038→
  1039→        if cmd in ("quit", "q", "exit"):
  1040→            self.exit()
  1041→            return
  1042→
  1043→        if cmd == "clear":
  1044→            self._clear_all_feeds()
  1045→            return
  1046→
  1047→        if cmd == "help":
  1048→            self._show_help()
  1049→            return
  1050→
  1051→        if cmd == "scan":
  1052→            self._handle_scan(args)
  1053→            return
  1054→
  1055→        if cmd == "audit":
  1056→            self._handle_audit(args)
  1057→            return
  1058→
  1059→        if cmd == "dev":
  1060→            self._handle_dev(args)
  1061→            return
  1062→
  1063→        if cmd == "doctor":
  1064→            self._handle_doctor()
  1065→            return
  1066→
  1067→        if cmd == "blitz":
  1068→            self._handle_blitz(args)
  1069→            return
  1070→
  1071→        if cmd == "subdomains":
  1072→            self._handle_subdomains(args)
  1073→            return
  1074→
  1075→        if cmd == "agent":
  1076→            self._handle_agent(args)
  1077→            return
  1078→
  1079→        if cmd == "swarm":
  1080→            self._handle_swarm(args)
  1081→            return
  1082→
  1083→        if cmd == "adversarial":
  1084→            self._handle_adversarial(args)
  1085→            return
  1086→
  1087→        if cmd == "graph":
  1088→            self._handle_graph()
  1089→            return
  1090→
  1091→        if cmd == "export":
  1092→            self._handle_export(args)
  1093→            return
  1094→
  1095→        if cmd == "history":
  1096→            self._handle_history()
  1097→            return
  1098→
  1099→        if cmd == "theme":
  1100→            # Already handled in on_input_submitted before _process_command
  1101→            return
  1102→
  1103→        self._chat(f"[{RED}]Unknown command: {cmd}[/]")
  1104→        self._chat(f"[{DIM_CYAN}]Type [cyan]help[/] for available commands.[/]")

  1106→    def _handle_theme(self, args: List[str]) -> None:
  1107→        """Handle theme switching command."""
  1108→        if not args or args[0].lower() in ("list", "ls"):
  1109→            available = Theme.available_themes()
  1110→            current = Theme.current_name()
  1111→            self._chat(f"[{CYAN} bold]  THEMES[/]")
  1112→            for name in available:
  1113→                marker = f" [{GREEN}]●[/]" if name == current else "  [{DIM_CYAN}]○[/]"
  1114→                display_name = THEMES[name].get("name", name)
  1115→                self._chat(f"  {marker} [{CYAN}]{name}[/] [{MUTED}]- {display_name}[/]")
  1116→            return
  1117→
  1118→        name = args[0].lower()

  1290→    @work(exclusive=False, thread=True)
  1291→    def _run_scan_worker(
  1292→        self,
  1293→        target: str,
  1294→        modules: Optional[List[str]] = None,
  1295→        is_local: bool = False,
  1296→    ) -> None:
  1297→        """Run a scan in a background thread."""
  1298→        self.call_from_thread(self.is_scanning.set, True)
  1299→
  1300→        # Set modules to scanning
  1301→        if modules:
  1302→            for m in modules:
  1303→                if m in self._module_cells:
  1304→                    self.call_from_thread(self._module_cells[m].set_status, "scanning")
  1305→        else:
  1306→            all_mods = (
  1307→                list(MODULE_NAMES.keys())
  1308→                if is_local
  1309→                else [k for k in MODULE_NAMES if k not in ("host", "dev", "doctor")]
  1310→            )
  1311→            for m in all_mods:
  1312→                if m in self._module_cells:
  1313→                    self.call_from_thread(self._module_cells[m].set_status, "scanning")
  1314→
  1315→        try:
  1316→            from .scanner import scan, audit_scan
  1317→            from .history import save_scan
  1318→
  1319→            self.call_from_thread(
  1320→                self._chat, f"[{DIM_CYAN}]  ⏳ Scan in progress...[/]"
  1321→            )
  1322→
  1323→            if is_local:
  1324→                result = audit_scan(target=target, modules=modules)
  1325→            else:
  1326→                result = scan(target=target, modules=modules)
  1327→
  1328→            # Process findings
  1329→            data = result.to_dict()
  1330→            self.call_from_thread(self._on_scan_complete, data, modules, is_local)
  1331→
  1332→            # Save to history
  1333→            try:
  1334→                save_scan(data, label="nexus")
  1335→            except Exception:
  1336→                pass
  1337→
  1338→        except Exception as e:
  1339→            self.call_from_thread(self._chat, f"[{RED}]  ✗ Scan error: {e}[/]")
  1340→            if modules:
  1341→                for m in modules:
  1342→                    if m in self._module_cells:
  1343→                        self.call_from_thread(self._module_cells[m].set_status, "error")
  1344→            else:
  1345→                self.call_from_thread(self._set_all_modules_status, "error")
  1346→
  1347→        finally:
  1348→            self.call_from_thread(self.is_scanning.set, False)

  1701→    def on_click(self, event: Click) -> None:
  1702→        """Open finding detail modal for the clicked finding.
  1703→
  1704→        Uses the vertical click position to determine which finding index
  1705→        the user clicked on, mapping it to _findings_list.
  1706→        """
  1707→        try:
  1708→            if (
  1709→                isinstance(event.widget, RichLog)
  1710→                and event.widget.id == "findings-feed"
  1711→                and self._findings_list
  1712→            ):
  1713→                # Get the line offset to determine which finding was clicked.
  1714→                # Each finding entry is 1 line in the RichLog.
  1715→                # We use the y coordinate relative to the widget to estimate the index.
  1716→                region = event.widget.region
  1717→                if region is not None:
  1718→                    # Calculate which line was clicked relative to the scroll position
  1719→                    scroll_offset = event.widget.scroll_y
  1720→                    click_y = event.y - region.y
  1721→                    # Each finding takes 1 line; estimate index from click position
  1722→                    # Account for scroll position
  1723→                    line_height = 1
  1724→                    if hasattr(event.widget, "_line_height"):
  1725→                        line_height = event.widget._line_height
  1726→                    clicked_index = int((click_y + scroll_offset * line_height) / line_height)
  1727→                    # Clamp to valid range
  1728→                    clicked_index = max(0, min(clicked_index, len(self._findings_list) - 1))
  1729→                    finding = self._findings_list[clicked_index]
  1730→                else:
  1731→                    finding = self._findings_list[-1]
  1732→                self.push_screen(FindingDetailModal(finding))
  1733→        except (NoMatches, Exception):
  1734→            pass
