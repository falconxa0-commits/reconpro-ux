   419→/**
   420→ * Project quantum hardware qubit count for a given year.
   421→ * Uses compound growth from current baseline.
   422→ */
   423→export function estimateQuantumAdvancement(targetYear: number): number {
   424→  const { currentBestQubits, projectedGrowthRate, baselineYear } = QUANTUM_HARDWARE;
   425→  const yearsElapsed = targetYear - baselineYear;
   426→  if (yearsElapsed <= 0) return currentBestQubits;
   427→
   428→  // Compound growth with slight acceleration (quantum advantage curve)
   429→  // Early years grow slightly faster, later years slow as engineering challenges increase
   430→  const accelerationFactor = 1 + (yearsElapsed * 0.003);
   431→  const effectiveGrowth = Math.pow(projectedGrowthRate, yearsElapsed) * accelerationFactor;
   432→
   433→  return Math.floor(currentBestQubits * effectiveGrowth);
   434→}

   436→/**
   437→ * Determine when quantum hardware will have enough qubits to break a given algorithm.
   438→ */
   439→export function calculateBreakYear(
   440→  algorithm: string,
   441→  keySize: number
   442→): { breakYear: number; milestone: QuantumMilestone; confidence: string } {
   443→  // Map common TLS identifiers to our milestone keys
   444→  const algorithmKey = mapAlgorithmToMilestone(algorithm, keySize);
   445→
   446→  if (algorithmKey.startsWith('pqc_')) {
   447→    // Already post-quantum
   448→    const milestone = QUANTUM_MILESTONES[algorithmKey];
   449→    return {
   450→      breakYear: milestone.estimatedBreakYear,
   451→      milestone,
   452→      confidence: 'very_low',
   453→    };
   454→  }
   455→
   456→  const milestone = QUANTUM_MILESTONES[algorithmKey];
   457→
   458→  if (!milestone) {
   459→    // Fallback: estimate based on key size
   460→    const fallbackQubits = keySize * 2;
   461→    let year = new Date().getFullYear();
   462→    let qubits = estimateQuantumAdvancement(year);
   463→    while (qubits < fallbackQubits && year < 2100) {
   464→      year++;
   465→      qubits = estimateQuantumAdvancement(year);
   466→    }
   467→    return {
   468→      breakYear: year,
   469→      milestone: {
   470→        qubitsRequired: fallbackQubits,
   471→        estimatedBreakYear: year,
   472→        confidence: 'low',
   473→        description: `Estimated for ${algorithm} with ${keySize}-bit key`,
   474→      },
   475→      confidence: 'low',
   476→    };
   477→  }
   478→
   479→  // Refine break year based on actual hardware projection
   480→  let projectedBreakYear = milestone.estimatedBreakYear;
   481→  let qubits = estimateQuantumAdvancement(projectedBreakYear);
   482→
   483→  // If hardware projection suggests earlier or later break
   484→  if (qubits >= milestone.qubitsRequired) {
   485→    // Walk backwards to find when qubits first meet the requirement
   486→    while (projectedBreakYear > new Date().getFullYear() + 1) {
   487→      projectedBreakYear--;
   488→      if (estimateQuantumAdvancement(projectedBreakYear) < milestone.qubitsRequired) {
   489→        projectedBreakYear++;
   490→        break;
   491→      }
   492→    }
   493→  } else {
   494→    // Walk forwards
   495→    while (qubits < milestone.qubitsRequired && projectedBreakYear < 2100) {
   496→      projectedBreakYear++;
   497→      qubits = estimateQuantumAdvancement(projectedBreakYear);
   498→    }
   499→  }
   500→
   501→  return {
   502→    breakYear: projectedBreakYear,
   503→    milestone,
   504→    confidence: milestone.confidence,
   505→  };
   506→}

   508→/**
   509→ * Map TLS algorithm identifiers to internal milestone keys.
   510→ */
   511→function mapAlgorithmToMilestone(algorithm: string, keySize: number): string {
   512→  const algo = algorithm.toUpperCase().trim();
   513→
   514→  // Post-quantum detection
   515→  if (algo.includes('KYBER') || algo.includes('ML-KEM')) {
   516→    if (keySize >= 1024) return 'pqc_kyber_1024';
   517→    if (keySize >= 768) return 'pqc_kyber_768';
   518→    return 'pqc_kyber_512';
   519→  }
   520→  if (algo.includes('DILITHIUM') || algo.includes('ML-DSA')) {
   521→    if (keySize >= 5) return 'pqc_dilithium5';
   522→    if (keySize >= 3) return 'pqc_dilithium3';
   523→    return 'pqc_dilithium2';
   524→  }
   525→  if (algo.includes('SPHINCS') || algo.includes('SLH-DSA')) {
   526→    return 'pqc_sphincsplus';
   527→  }
   528→
   529→  // RSA
   530→  if (algo.includes('RSA')) {
   531→    if (keySize >= 8192) return 'rsa_8192';
   532→    if (keySize >= 4096) return 'rsa_4096';
   533→    if (keySize >= 2048) return 'rsa_2048';
   534→    if (keySize >= 1024) return 'rsa_1024';
   535→    return 'rsa_2048'; // default assumption
   536→  }
   537→
   538→  // Elliptic Curve
   539→  if (algo.includes('ECDH') || algo.includes('ECDHE') || algo.includes('X25519') || algo.includes('X448')) {
   540→    if (algo.includes('P-384') || algo.includes('SECP384') || keySize >= 384) return 'ec_p384';
   541→    if (algo.includes('P-521') || algo.includes('SECP521') || keySize >= 521) return 'ec_p521';
   542→    if (algo.includes('25519')) return 'ec_curve25519';
   543→    if (keySize >= 384) return 'ec_p384';
   544→    return 'ec_p256'; // default assumption
   545→  }
   546→
   547→  if (algo.includes('ECDSA') || algo.includes('ED25519') || algo.includes('EDDSA')) {
   548→    if (algo.includes('P-384') || algo.includes('SECP384') || keySize >= 384) return 'ec_p384';
   549→    if (algo.includes('P-521') || algo.includes('SECP521') || keySize >= 521) return 'ec_p521';
   550→    if (algo.includes('25519')) return 'ec_ed25519';
   551→    if (keySize >= 384) return 'ec_p384';
   552→    return 'ec_p256';
   553→  }
   554→
   555→  // AES / Symmetric
   556→  if (algo.includes('AES')) {
   557→    if (keySize >= 256) return 'aes_256';
   558→    return 'aes_128';
   559→  }
   560→
   561→  if (algo.includes('3DES') || algo.includes('DES')) {
   562→    return 'aes_128'; // 3DES effectively worse than AES-128
   563→  }
   564→
   565→  if (algo.includes('SHA')) {
   566→    if (algo.includes('384') || keySize >= 384) return 'sha_384';
   567→    return 'sha_256';
   568→  }
   569→
   570→  // Default fallback — assume RSA-2048 as most common
   571→  return 'rsa_2048';
   572→}

   574→/**
   575→ * Calculate algorithm family from key.
   576→ */
   577→function getAlgorithmFamily(algorithmKey: string): AlgorithmFamily {
   578→  if (algorithmKey.startsWith('pqc_')) return 'pqc';
   579→  if (algorithmKey.startsWith('rsa_')) return 'rsa';
   580→  if (algorithmKey.startsWith('ec_')) return 'ec';
   581→  if (algorithmKey.startsWith('aes_') || algorithmKey.startsWith('sha_')) return 'aes';
   582→  return 'rsa'; // default
   583→}

   630→  ) {
   631→    const migration = PQC_MIGRATIONS.signatures;

   635→      nistLevel: migration.nistLevel,
   636→    };
   637→  }
   638→
   639→  // Symmetric
   640→  if (family === 'aes') {
   641→    const migration = PQC_MIGRATIONS.symmetric_upgrade;
   642→    return {
   643→      recommendation: `Upgrade to ${migration.to} (${migration.status})`,
   644→      pqcAlgorithm: migration.to,
   645→      nistLevel: migration.nistLevel,
   646→    };
   647→  }
   648→
   649→  // Default
   650→  const migration = PQC_MIGRATIONS.key_exchange;

   658→/**
   659→ * Harvest Now, Decrypt Later (HNDL) risk assessment.
   660→ * Attackers are ALREADY capturing encrypted traffic to decrypt later.
   661→ */
   662→export function calculateHNDLRisk(
   663→  assets: AssetAssessment[],
   664→  industry?: string,
   665→  companyName?: string
   666→): HNDLRiskAssessment {
   667→  const now = new Date();
   668→  const nonPQCAssets = assets.filter(a => a.algorithmFamily !== 'pqc');
   669→
   670→  if (nonPQCAssets.length === 0) {
   671→    return {
   672→      score: 0,
   673→      description: 'All assets use post-quantum cryptography. No HNDL risk detected.',
   674→      dataAtRisk: 'None',
   675→      recommendation: 'Continue monitoring quantum hardware developments.',
   676→      captureWindowYears: 0,
   677→      estimatedRecords: '0',
   678→      regulatoryExposure: 'None',
   679→    };
   680→  }
   681→
   682→  // Calculate capture window — the gap between now and when quantum can decrypt
   683→  const earliestBreakYear = Math.min(...nonPQCAssets.map(a => a.estimatedBreakYear));
   684→  const captureWindow = earliestBreakYear - now.getFullYear();
   685→
   686→  // Score based on capture window, number of assets, and data sensitivity
   687→  let baseScore = 0;
   688→
   689→  // Capture window component (shorter window = higher urgency)
   690→  if (captureWindow <= 3) baseScore += 35;
   691→  else if (captureWindow <= 5) baseScore += 25;
   692→  else if (captureWindow <= 7) baseScore += 15;
   693→  else baseScore += 5;
   694→
   695→  // Asset count component
   696→  baseScore += Math.min(25, nonPQCAssets.length * 5);
   697→
   698→  // Industry sensitivity component
   699→  const industryMultiplier: Record<string, number> = {
   700→    finance: 1.3,
   701→    healthcare: 1.35,
   702→    government: 1.25,
   703→    retail: 1.15,
   704→    energy: 1.2,
   705→    telecom: 1.15,
   706→    technology: 1.0,
   707→    education: 1.1,
   708→  };
   709→  baseScore *= industryMultiplier[industry || 'technology'];
   710→
   711→  // Critical assets bonus
   712→  const criticalCount = nonPQCAssets.filter(a => a.riskLevel === 'CRITICAL').length;
   713→  baseScore += criticalCount * 8;
   714→
   715→  const score = Math.min(100, Math.round(baseScore));
   716→
   717→  // Estimate data at risk
   718→  const estimatedRecords = estimateRecordsAtRisk(industry, nonPQCAssets.length);
   719→
   720→  // Regulatory exposure
   721→  const regulatoryExposure = getRegulatoryExposure(industry);
   722→
   723→  // Generate description
   724→  const descriptions: Record<number, string> = {
   725→    80: `CRITICAL HNDL THREAT: ${companyName ? `${companyName}'s` : 'Your'} encrypted traffic is actively being captured by state-sponsored adversaries. With quantum decryption expected within ${captureWindow} years, intercepted data will be fully readable. Immediate PQC migration is non-negotiable.`,
   726→    60: `SEVERE HNDL RISK: ${companyName ? `${companyName}'s` : 'Your'} encrypted communications are vulnerable to "store now, decrypt later" attacks. Nation-state threat actors routinely bulk-capture TLS traffic. With ${captureWindow} years until quantum decryption, time is running out.`,
   727→    40: `MODERATE HNDL EXPOSURE: ${companyName ? `${companyName}` : 'Your organization'} has ${nonPQCAssets.length} assets using quantum-vulnerable cryptography. Adversaries may be stockpiling encrypted data for future quantum decryption within ${captureWindow} years.`,
   728→    20: `LOW HNDL RISK: While ${companyName ? `${companyName}` : 'your organization'} has some quantum-vulnerable assets, the timeline provides adequate runway for migration. HNDL attacks remain a theoretical concern.`,
   729→    0: `MINIMAL HNDL RISK: ${companyName ? `${companyName}'s` : 'Your'} cryptography posture provides reasonable protection against harvest-now-decrypt-later attacks in the near term.`,
   730→  };
   731→
   732→  const descKey = score >= 80 ? 80 : score >= 60 ? 60 : score >= 40 ? 40 : score >= 20 ? 20 : 0;
   733→
   734→  return {
   735→    score,
   736→    description: descriptions[descKey],
   737→    dataAtRisk: `estimated ${estimatedRecords} records`,
   738→    recommendation: score >= 60
   739→      ? 'URGENT: Begin PQC migration immediately. Implement hybrid key exchange as interim measure. Audit all long-term data stores.'
   740→      : score >= 40
   741→        ? 'Begin PQC migration planning. Prioritize highest-risk assets. Test hybrid deployments in staging.'
   742→        : 'Continue monitoring NIST PQC transition timelines. Begin inventory of quantum-vulnerable systems.',
   743→    captureWindowYears: Math.max(0, captureWindow),
   744→    estimatedRecords,
   745→    regulatoryExposure,

   804→/**
   805→ * Generate prioritized PQC migration roadmap.
   806→ */
   807→export function generateMigrationPlan(assets: AssetAssessment[]): MigrationStep[] {
   808→  // Group assets by migration category
   809→  const categories: Record<string, AssetAssessment[]> = {
   810→    'Key Exchange': [],
   811→    'Digital Signatures': [],
   812→    'Symmetric Encryption': [],
   813→    'Post-Quantum Ready': [],
   814→  };
   815→
   816→  for (const asset of assets) {
   817→    if (asset.algorithmFamily === 'pqc') {
   818→      categories['Post-Quantum Ready'].push(asset);
   819→    } else if (
   820→      asset.algorithmFamily === 'rsa' ||
   821→      (asset.algorithmFamily === 'ec' && asset.keyExchange.includes('ECDH'))
   822→    ) {
   823→      categories['Key Exchange'].push(asset);
   824→    } else if (asset.algorithmFamily === 'ec') {
   825→      categories['Digital Signatures'].push(asset);
   826→    } else {
   827→      categories['Symmetric Encryption'].push(asset);
   828→    }
   829→  }
   830→
   831→  const steps: MigrationStep[] = [];
   832→  let priority = 1;
   833→
   834→  // Key exchange migrations (highest priority — direct HNDL target)
   835→  for (const asset of categories['Key Exchange']) {
   836→    const migration = PQC_MIGRATIONS.key_exchange;
   837→    steps.push({
   838→      priority: priority++,
   839→      asset: `${asset.domain}:${asset.port}`,
   840→      currentCrypto: `${asset.currentAlgorithm} (${asset.keySize}-bit)`,
   841→      recommendedCrypto: migration.to,
   842→      nistLevel: migration.nistLevel,
   843→      estimatedEffort: asset.riskLevel === 'CRITICAL' ? '2-4 weeks' : '4-8 weeks',
   844→      costEstimate: asset.riskLevel === 'CRITICAL' ? '$25K-$75K' : '$50K-$150K',
   845→      category: 'Key Exchange',
   846→      urgency: migration.urgency,
   847→    });
   848→  }
   849→
   850→  // Digital signature migrations
   851→  for (const asset of categories['Digital Signatures']) {
   852→    const migration = PQC_MIGRATIONS.signatures;
   853→    steps.push({
   854→      priority: priority++,
   855→      asset: `${asset.domain}:${asset.port}`,
   856→      currentCrypto: `${asset.currentAlgorithm} (${asset.keySize}-bit)`,
   857→      recommendedCrypto: migration.to,
   858→      nistLevel: migration.nistLevel,
   859→      estimatedEffort: '2-6 months',
   860→      costEstimate: '$75K-$250K',
   861→      category: 'Digital Signatures',
   862→      urgency: migration.urgency,
   863→    });
   864→  }
   865→
   866→  // Symmetric upgrades
   867→  for (const asset of categories['Symmetric Encryption']) {
   868→    const migration = PQC_MIGRATIONS.symmetric_upgrade;
   869→    steps.push({
   870→      priority: priority++,
   871→      asset: `${asset.domain}:${asset.port}`,
   872→      currentCrypto: `${asset.currentAlgorithm} (${asset.keySize}-bit)`,
   873→      recommendedCrypto: migration.to,
   874→      nistLevel: migration.nistLevel,
   875→      estimatedEffort: '2-4 weeks',
   876→      costEstimate: '$10K-$50K',
   877→      category: 'Symmetric Encryption',
   878→      urgency: migration.urgency,
   879→    });
   880→  }
   881→
   882→  // Sort by priority (earliest break year first within same category)
   883→  steps.sort((a, b) => {
   884→    const aAsset = assets.find(x => `${x.domain}:${x.port}` === a.asset);
   885→    const bAsset = assets.find(x => `${x.domain}:${x.port}` === b.asset);
   886→    if (aAsset && bAsset) {
   887→      return aAsset.estimatedBreakYear - bAsset.estimatedBreakYear;
   888→    }
   889→    return a.priority - b.priority;
   890→  });
   891→
   892→  // Re-number priorities after sort
   893→  steps.forEach((step, i) => {
   894→    step.priority = i + 1;
   895→  });
   896→
   897→  return steps;
   898→}

   980→/**
   981→ * Determine the display name for a cipher/key exchange combination.
   982→ */
   983→function getAlgorithmDisplayName(
   984→  keyExchange: string,
   985→  cipherSuite: string,
   986→  keySize: number
   987→): string {
   988→  const ke = keyExchange.toUpperCase();
   989→  if (ke.includes('ECDHE') && ke.includes('ECDSA')) return 'ECDHE-ECDSA';
   990→  if (ke.includes('ECDHE') && ke.includes('RSA')) return 'ECDHE-RSA';
   991→  if (ke.includes('ECDHE')) return 'ECDHE';
   992→  if (ke.includes('RSA')) return 'RSA';
   993→  if (ke.includes('X25519') || ke.includes('X448')) return 'X25519';
   994→  if (cipherSuite.includes('AES_256')) return 'AES-256-GCM';
   995→  if (cipherSuite.includes('AES_128')) return 'AES-128-GCM';
   996→  if (ke.includes('DHE')) return 'DHE';
   997→  return ke || cipherSuite.split('_')[0] || 'UNKNOWN';
   998→}
