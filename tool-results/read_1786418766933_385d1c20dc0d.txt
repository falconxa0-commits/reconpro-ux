   167→@dataclass
   168→class AssetProfile:
   169→    """Profile of a single discovered asset with decay analysis."""
   170→    url: str
   171→    asset_type: str
   172→    status_code: Optional[int]
   173→    decay_score: int
   174→    indicators: List[str] = field(default_factory=list)
   175→    last_modified: Optional[str] = None
   176→    risk_level: str = "low"
   177→    server_header: Optional[str] = None
   178→    title: Optional[str] = None
   179→    cert_info: Optional[Dict[str, Any]] = None
   180→    tech_stack: List[str] = field(default_factory=list)
   181→    discovered_by: str = ""
   182→    timestamp: str = ""
   183→
   184→    def __post_init__(self) -> None:
   185→        if not self.timestamp:
   186→            self.timestamp = datetime.now(timezone.utc).isoformat()
   187→        if self.decay_score >= 75:
   188→            self.risk_level = "critical"
   189→        elif self.decay_score >= 55:
   190→            self.risk_level = "high"
   191→        elif self.decay_score >= 35:
   192→            self.risk_level = "medium"
   193→        elif self.decay_score >= 15:
   194→            self.risk_level = "low"
   195→        else:
   196→            self.risk_level = "info"
   197→
   198→    def to_dict(self) -> Dict[str, Any]:
   199→        """Serialize to a plain dictionary."""
   200→        return {
   201→            "url": self.url,
   202→            "asset_type": self.asset_type,

   221→def _make_request(
   222→    url: str,
   223→    timeout: int = 8,
   224→    method: str = "GET",
   225→    follow_redirects: bool = True,
   226→    headers: Optional[Dict[str, str]] = None,
   227→) -> Tuple[Optional[Dict[str, str]], Optional[int], Optional[str], Optional[str]]:
   228→    """
   229→    Perform an HTTP request using stdlib urllib and return
   230→    (response_headers, status_code, body_snippet, final_url).
   231→
   232→    Returns (None, None, None, None) on any failure.
   233→    """
   234→    default_headers = {
   235→        "User-Agent": "ReconPro/9.2.0 (Security Scanner; https://reconpro.dev)",
   236→        "Accept": "text/html,application/xhtml+xml,application/xml;q=0.9,*/*;q=0.8",
   237→        "Accept-Language": "en-US,en;q=0.5",
   238→        "Connection": "close",
   239→    }
   240→    if headers:
   241→        default_headers.update(headers)
   242→
   243→    try:
   244→        req = Request(url, headers=default_headers, method=method)
   245→        resp = urlopen(req, timeout=timeout, context=None)
   246→        resp_headers = {k.lower(): v for k, v in resp.getheaders()}
   247→        status_code = resp.getcode()
   248→        final_url = resp.geturl()
   249→        body = resp.read(65536)
   250→        try:
   251→            body_text = body.decode("utf-8", errors="replace")[:16384]
   252→        except Exception:
   253→            body_text = ""
   254→        return resp_headers, status_code, body_text, final_url
   255→    except HTTPError as exc:
   256→        resp_headers = {k.lower(): v for k, v in (exc.headers or {}).items()}
   257→        try:
   258→            body = exc.read(65536)
   259→            body_text = body.decode("utf-8", errors="replace")[:16384]
   260→        except Exception:
   261→            body_text = ""
   262→        return resp_headers, exc.code, body_text, url
   263→    except (URLError, UrlLibURLError, OSError, Exception):
   264→        return None, None, None, None

   267→def _probe_path(base_url: str, path: str, timeout: int = 8) -> AssetProfile:
   268→    """Probe a single path under *base_url* and return an AssetProfile."""
   269→    url = urljoin(base_url.rstrip("/") + "/", path.lstrip("/"))
   270→    resp_headers, status_code, body, final_url = _make_request(url, timeout=timeout)
   271→    cert_info = _check_certificate_age(urlparse(url).hostname or "")
   272→    title = _extract_title(body or "")
   273→    indicators: List[str] = []
   274→    tech_stack: List[str] = _detect_tech_stack(resp_headers, body or "")
   275→
   276→    asset_type = _classify_asset(resp_headers, body or "", title, cert_info)
   277→    indicators.extend(_assess_indicators(resp_headers, body or "", title, cert_info))
   278→    last_mod = resp_headers.get("last-modified") if resp_headers else None
   279→
   280→    decay = _calculate_decay_from_indicators(indicators, resp_headers, cert_info)
   281→
   282→    profile = AssetProfile(
   283→        url=url,
   284→        asset_type=asset_type,
   285→        status_code=status_code,
   286→        decay_score=decay,
   287→        indicators=indicators,
   288→        last_modified=last_mod,
   289→        server_header=(resp_headers.get("server", "") if resp_headers else None),
   290→        title=title,
   291→        cert_info=cert_info,
   292→        tech_stack=tech_stack,
   293→        discovered_by="path_probe",
   294→    )
   295→    return profile

   298→def _extract_title(html: str) -> str:
   299→    """Extract <title> content from HTML."""
   300→    m = re.search(r"<title[^>]*>(.*?)</title>", html, re.I | re.S)
   301→    if m:
   302→        return m.group(1).strip()[:256]
   303→    return ""

   306→def _detect_tech_stack(headers: Optional[Dict[str, str]], body: str) -> List[str]:
   307→    """Detect technology stack from headers and body."""
   308→    stack: List[str] = []
   309→    if headers:
   310→        server = headers.get("server", "")
   311→        if "nginx" in server.lower():
   312→            stack.append("nginx")
   313→        if "apache" in server.lower():
   314→            stack.append("Apache")
   315→        if "iis" in server.lower() or "Microsoft-HTTPAPI" in server:
   316→            stack.append("IIS")
   317→        if "tomcat" in server.lower() or "Apache-Coyote" in server:
   318→            stack.append("Tomcat")
   319→        powered_by = headers.get("x-powered-by", "")
   320→        if "php" in powered_by.lower():
   321→            stack.append("PHP")
   322→        if "express" in powered_by.lower():
   323→            stack.append("Express.js")
   324→        if "asp.net" in powered_by.lower():
   325→            stack.append("ASP.NET")
   326→        if "next" in powered_by.lower():
   327→            stack.append("Next.js")
   328→    if body:
   329→        if "wp-content" in body or "wp-includes" in body:
   330→            stack.append("WordPress")
   331→        if re.search(r'wp-content/themes/\w+', body):
   332→            stack.append("WordPress (themed)")
   333→        if "drupal" in body.lower():
   334→            stack.append("Drupal")
   335→        if "Joomla" in body:
   336→            stack.append("Joomla")
   337→        if "django" in body.lower() or "csrfmiddlewaretoken" in body:
   338→            stack.append("Django")
   339→        if "laravel" in body.lower() or "laravel_session" in body:
   340→            stack.append("Laravel")
   341→        if "react" in body.lower() and "__NEXT_DATA__" in body:
   342→            stack.append("React/Next.js")
   343→        if re.search(r'angular[^.]', body, re.I):
   344→            stack.append("Angular")
   345→        if "vue" in body.lower() or "__vue" in body:
   346→            stack.append("Vue.js")
   347→        if "jquery" in body.lower():
   348→            stack.append("jQuery")
   349→        if "bootstrap" in body.lower():
   350→            stack.append("Bootstrap")
   351→        if "/api-docs" in body or "swagger" in body.lower():
   352→            stack.append("Swagger")
   353→        if "graphql" in body.lower():
   354→            stack.append("GraphQL")
   355→    return list(dict.fromkeys(stack))

   358→def _check_certificate_age(host: str) -> Optional[Dict[str, Any]]:
   359→    """Check TLS certificate details using stdlib ssl. Returns cert info dict."""
   360→    if not host:
   361→        return None
   362→    ctx = ssl.create_default_context()
   363→    ctx.check_hostname = False
   364→    ctx.verify_mode = ssl.CERT_NONE
   365→    port = 443
   366→    try:
   367→        with socket.create_connection((host, port), timeout=6) as sock:
   368→            with ctx.wrap_socket(sock, server_hostname=host) as tls_sock:
   369→                cert_der = tls_sock.getpeercert(binary_form=True)
   370→                cert_dict = tls_sock.getpeercert()
   371→                if not cert_dict:
   372→                    return {"host": host, "error": "no_certificate"}
   373→                now = datetime.now(timezone.utc)
   374→                not_before = None
   375→                not_after = None
   376→                if "notBefore" in cert_dict:
   377→                    not_before = datetime.strptime(cert_dict["notBefore"], "%b %d %H:%M:%S %Y %Z").replace(tzinfo=timezone.utc)
   378→                if "notAfter" in cert_dict:
   379→                    not_after = datetime.strptime(cert_dict["notAfter"], "%b %d %H:%M:%S %Y %Z").replace(tzinfo=timezone.utc)
   380→                days_until_expiry = None
   381→                expired = False
   382→                near_expiry = False
   383→                if not_after:
   384→                    delta = not_after - now
   385→                    days_until_expiry = delta.days
   386→                    expired = delta.total_seconds() < 0
   387→                    near_expiry = 0 < delta.days <= 30
   388→
   389→                age_days = None
   390→                if not_before:
   391→                    age_days = (now - not_before).days
   392→
   393→                issuer = ""
   394→                subject = ""
   395→                for pair in cert_dict.get("issuer", []):
   396→                    if pair[0][0] == "organizationName":
   397→                        issuer = pair[0][1]
   398→                for pair in cert_dict.get("subject", []):
   399→                    if pair[0][0] == "commonName":
   400→                        subject = pair[0][1]
   401→
   402→                hostname_match = False
   403→                for san_type, san_value in cert_dict.get("subjectAltName", []):
   404→                    if san_type == "DNS" and san_value.lower() == host.lower():
   405→                        hostname_match = True
   406→                        break
   407→                if not hostname_match and subject.lower() == host.lower():
   408→                    hostname_match = True
   409→
   410→                self_signed = issuer == subject or "Let's Encrypt" not in issuer and not any(
   411→                    known in issuer for known in ["DigiCert", "Sectigo", "GlobalSign", "Cloudflare", "Amazon", "Google"]
   412→                )
   413→
   414→                serial_hex = ""
   415→                if cert_der:
   416→                    serial_hex = hashlib.sha256(cert_der).hexdigest()[:16]
   417→
   418→                return {
   419→                    "host": host,
   420→                    "subject": subject,
   421→                    "issuer": issuer,
   422→                    "not_before": not_before.isoformat() if not_before else None,
   423→                    "not_after": not_after.isoformat() if not_after else None,
   424→                    "days_until_expiry": days_until_expiry,
   425→                    "age_days": age_days,
   426→                    "expired": expired,
   427→                    "near_expiry": near_expiry,
   428→                    "self_signed": self_signed,
   429→                    "hostname_match": hostname_match,
   430→                    "serial_fingerprint": serial_hex,
   431→                }
   432→    except (socket.timeout, socket.gaierror, OSError, ssl.SSLError, Exception):
   433→        return {"host": host, "error": "connection_failed"}

   436→def _classify_asset(
   437→    headers: Optional[Dict[str, str]],
   438→    body: str,
   439→    title: str,
   440→    cert_info: Optional[Dict[str, Any]],
   441→) -> str:
   442→    """Classify an asset into a type string based on response signals."""
   443→    if not headers and not body:
   444→        return "unreachable"
   445→    if body:
   446→        bl = body.lower()
   447→        if "phpmyadmin" in bl or "pma" in bl:
   448→            return "database_admin"
   449→        if "jenkins" in bl:
   450→            return "ci_cd"
   451→        if "grafana" in bl:
   452→            return "monitoring"
   453→        if "kibana" in bl:
   454→            return "monitoring"
   455→        if "redis commander" in bl or "redis-commander" in bl:
   456→            return "database_admin"
   457→        if "pgadmin" in bl:
   458→            return "database_admin"
   459→        if "mongo express" in bl or "mongo-express" in bl:
   460→            return "database_admin"
   461→        if "solr" in bl and "admin" in bl:
   462→            return "search_admin"
   463→        if "tomcat manager" in bl or "/manager/html" in bl:
   464→            return "app_server"
   465→        if "swagger" in bl or "openapi" in bl or "/api-docs" in bl:
   466→            return "api_documentation"
   467→        if "graphql" in bl:
   468→            return "api_graphql"
   469→        if "/api/" in bl or re.search(r'api[_-]?key|apikey|api-key', bl):
   470→            return "api_endpoint"
   471→        if "sentry" in bl and ("dsn" in bl or "project" in bl):
   472→            return "error_tracking"
   473→        if "aws_access_key" in bl or "AKIA" in body:
   474→            return "cloud_credentials"
   475→        if "firebase" in bl and "config" in bl:
   476→            return "cloud_config"
   477→        if "wordpress" in bl or "wp-content" in bl:
   478→            return "cms_wordpress"
   479→        if "drupal" in bl:
   480→            return "cms_drupal"
   481→        if "joomla" in bl:
   482→            return "cms_joomla"
   483→        for default in DecayIndicators.DEFAULT_PAGE_TITLES:
   484→            if default.lower() in title.lower():
   485→                return "default_install"
   486→        if "directory listing" in bl or "index of /" in title.lower():
   487→            return "directory_listing"
   488→        if "staging" in bl or "development" in title.lower() or "dev" in title.lower():
   489→            return "staging_environment"
   490→    if headers:
   491→        ct = headers.get("content-type", "")
   492→        if "application/json" in ct:
   493→            return "api_response"
   494→        if "text/xml" in ct or "application/xml" in ct:
   495→            return "xml_endpoint"
   496→    return "web_application"

   499→def _assess_indicators(
   500→    headers: Optional[Dict[str, str]],
   501→    body: str,
   502→    title: str,
   503→    cert_info: Optional[Dict[str, Any]],
   504→) -> List[str]:
   505→    """Assess decay indicators from response signals."""
   506→    indicators: List[str] = []
   507→    if headers:
   508→        server = headers.get("server", "")
   509→        for old in DecayIndicators.OUTDATED_SERVER_HEADERS:
   510→            if old.lower() in server.lower():
   511→                indicators.append("server_header_outdated")
   512→                break
   513→        header_names = {k.lower() for k in headers}
   514→        if "strict-transport-security" not in header_names:
   515→            indicators.append("no_hsts")
   516→        if "content-security-policy" not in header_names:
   517→            indicators.append("no_csp")
   518→        for sec_h in DecayIndicators.MISSING_SECURITY_HEADERS:
   519→            if sec_h not in header_names:
   520→                pass  # already captured the most critical
   521→        if "last-modified" not in header_names:
   522→            indicators.append("no_last_modified")
   523→    if title:
   524→        for default_title in DecayIndicators.DEFAULT_PAGE_TITLES:
   525→            if default_title.lower() in title.lower():
   526→                indicators.append("default_install_page")
   527→                break
   528→        if "index of" in title.lower():
   529→            indicators.append("directory_listing")
   530→    if body:
   531→        if "wp-content" in body or "wp-includes" in body:
   532→            for pat in DecayIndicators.WORDPRESS_OLD_SIGNATURES:
   533→                if pat.search(body):
   534→                    indicators.append("old_wordpress")
   535→                    break
   536→        for pat in DecayIndicators.OUTDATED_JS_SIGNATURES:
   537→            if pat.search(body):
   538→                indicators.append("outdated_js_libs")
   539→                break
   540→        if re.search(r'<[^>]+(?:notice|warning|error|fatal|deprecated)[^>]*>.*?(?:on line \d+|in .*\.php)', body, re.I | re.S):
   541→            indicators.append("php_errors_visible")
   542→        if "stack trace" in body.lower() or "traceback (most recent" in body.lower():
   543→            indicators.append("stack_trace_exposed")
   544→        if "debug" in body.lower() and ("debug_mode" in body.lower() or "DEBUG" in body):
   545→            indicators.append("debug_mode_on")
   546→        if not re.search(r'<link[^>]+favicon', body, re.I):
   547→            indicators.append("no_favicon")
   548→    if cert_info:
   549→        if cert_info.get("expired"):
   550→            indicators.append("expired_cert")
   551→        if cert_info.get("self_signed"):
   552→            indicators.append("self_signed_cert")
   553→        if cert_info.get("near_expiry"):
   554→            indicators.append("near_expiry_cert")
   555→        if not cert_info.get("hostname_match", False) and cert_info.get("subject"):
   556→            indicators.append("cert_hostname_mismatch")
   557→    return list(dict.fromkeys(indicators))

   559→
   560→def _calculate_decay_from_indicators(
   561→    indicators: List[str],
   562→    headers: Optional[Dict[str, str]],
   563→    cert_info: Optional[Dict[str, Any]],
   564→) -> int:
   565→    """Calculate a 0-100 decay score based on collected indicators."""
   566→    score = 0
   567→    weight_map: Dict[str, int] = {
   568→        "expired_cert": 25,
   569→        "self_signed_cert": 15,
   570→        "near_expiry_cert": 12,
   571→        "cert_hostname_mismatch": 10,
   572→        "default_install_page": 20,
   573→        "directory_listing": 18,
   574→        "server_header_outdated": 15,
   575→        "old_wordpress": 18,
   576→        "outdated_js_libs": 10,
   577→        "no_hsts": 5,
   578→        "no_csp": 5,
   579→        "no_favicon": 3,
   580→        "no_last_modified": 2,
   581→        "no_robots_txt": 3,
   582→        "http_only_no_redirect": 8,
   583→        "php_errors_visible": 15,
   584→        "stack_trace_exposed": 20,
   585→        "debug_mode_on": 18,
   586→    }
   587→    for ind in indicators:
   588→        score += weight_map.get(ind, 5)
   589→    if cert_info and cert_info.get("days_until_expiry") is not None:
   590→        days = cert_info["days_until_expiry"]
   591→        if days < 0:
   592→            score = max(score, 60)
   593→        elif days <= 7:
   594→            score += 20
   595→        elif days <= 30:
   596→            score += 10
   597→    if cert_info and cert_info.get("age_days") is not None:
   598→        age = cert_info["age_days"]
   599→        if age > 730:
   600→            score += 8
   601→        elif age > 365:
   602→            score += 5
   603→    return min(score, 100)

   837→
   838→        Returns a dictionary with keys for every scan category, each containing
   839→        a list of :class:`AssetProfile` dicts.

   841→        self.timeout = timeout
   842→        timestamp = datetime.now(timezone.utc).isoformat()
   843→
   844→        results: Dict[str, Any] = {
   845→            "target": target,
   846→            "base_url": base_url,

   849→            "summary": {
   850→                "total_assets": 0,
   851→                "critical": 0,
   852→                "high": 0,

   865→            ("shadow_apis", self.discover_shadow_apis, [base_url]),
   866→            ("orphaned_dns", self.discover_orphaned_dns, [target]),
   867→            ("forgotten_services", self.discover_forgotten_services, [base_url]),
   868→            ("cloud_leaks", self.discover_cloud_leaks, [base_url]),
   869→            ("internal_exposure", self.discover_internal_exposure, [base_url]),
   870→        ]
   871→
   872→        for cat_name, cat_func, cat_args in categories:
   873→            try:
   874→                assets = cat_func(*cat_args)
   875→                if isinstance(assets, list):
   876→                    results["findings"][cat_name] = [a.to_dict() if isinstance(a, AssetProfile) else a for a in assets]
   877→                elif isinstance(assets, dict):
   878→                    results["findings"][cat_name] = assets
   879→                else:
   880→                    results["findings"][cat_name] = assets
   881→            except Exception as exc:
   882→                results["findings"][cat_name] = {"error": str(exc)}
   883→
   884→        # Build summary counts

   888→        for cat_key, cat_val in results["findings"].items():
   889→            if isinstance(cat_val, list):
   890→                for item in cat_val:
   891→                    if isinstance(item, dict):

   901→                        total += 1
   902→                        rl = item.get("risk_level", "info")
   903→                        if rl in risk_counts:

   905→                if cat_val["assets"]:
   906→                    cats_with_findings += 1
   907→
   908→        results["summary"]["total_assets"] = total
   909→        results["summary"]["critical"] = risk_counts["critical"]
   910→        results["summary"]["high"] = risk_counts["high"]
   911→        results["summary"]["medium"] = risk_counts["medium"]
   912→        results["summary"]["low"] = risk_counts["low"]
   913→        results["summary"]["info"] = risk_counts["info"]
   914→        results["summary"]["categories_with_findings"] = cats_with_findings

   922→    def discover_forgotten_subdomains(self, target: str) -> List[AssetProfile]:
   923→        """
   924→        Enumerate subdomains via cert transparency (crt.sh) and common-prefix
   925→        brute-force DNS resolution.
   926→        """
   927→        found: List[AssetProfile] = []
   928→        resolved_hosts: Dict[str, str] = {}
   929→
   930→        # --- Phase 1: crt.sh certificate transparency ---
   931→        ct_hosts = self._query_crtsh(target)
   932→        for host in ct_hosts:
   933→            if host not in resolved_hosts:
   934→                ip = self._resolve_host(host)
   935→                if ip:
   936→                    resolved_hosts[host] = ip
   937→
   938→        # --- Phase 2: Brute-force common prefixes ---
   939→        for prefix in self.COMMON_SUBDOMAIN_PREFIXES:
   940→            fqdn = f"{prefix}.{target}"
   941→            if fqdn not in resolved_hosts:
   942→                ip = self._resolve_host(fqdn)
   943→                if ip:
   944→                    resolved_hosts[fqdn] = ip
   945→
   946→        # --- Phase 3: Profile each discovered subdomain ---
   947→        for host, ip in resolved_hosts.items():
   948→            if host == target or host == f"www.{target}":
   949→                continue
   950→            profile = self._profile_host(host, ip)
   951→            profile.discovered_by = "subdomain_enum"
   952→            found.append(profile)
   953→
   954→        return found

   956→    def _query_crtsh(self, domain: str) -> List[str]:
   957→        """Query crt.sh for certificate transparency subdomains."""
   958→        hosts: List[str] = []
   959→        url = f"https://crt.sh/?q=%.{domain}&output=json"
   960→        try:
   961→            resp_headers, status, body, final_url = _make_request(
   962→                url, timeout=12, headers=self._session_headers
   963→            )
   964→            if body and status == 200:
   965→                try:
   966→                    entries = json.loads(body)
   967→                    for entry in entries:
   968→                        name_value = entry.get("name_value", "")
   969→                        for name in name_value.split("\n"):
   970→                            name = name.strip().lstrip("*.")
   971→                            if name and name.endswith(f".{domain}") and name != domain:
   972→                                if name not in hosts:
   973→                                    hosts.append(name)
   974→                except (json.JSONDecodeError, KeyError, TypeError):
   975→                    pass
   976→                # Fallback: regex extraction
   977→                if not hosts:
   978→                    pattern = re.compile(r'([a-zA-Z0-9][a-zA-Z0-9-]*\.' + re.escape(domain) + r')')
   979→                    matches = pattern.findall(body)
   980→                    for m in matches:
   981→                        m = m.lstrip("*.")
   982→                        if m not in hosts and m != domain:
   983→                            hosts.append(m)
   984→        except Exception:
   985→            pass
   986→        return hosts

   988→    def _resolve_host(self, host: str) -> Optional[str]:
   989→        """Resolve a hostname to an IP address via DNS. Returns IP or None."""
   990→        try:
   991→            result = socket.getaddrinfo(host, None, socket.AF_INET, socket.SOCK_STREAM)
   992→            if result:
   993→                return result[0][4][0]
   994→        except (socket.gaierror, socket.herror, OSError):
   995→            pass
   996→        return None

  1232→        }

  1234→        # Resolve A / AAAA

  1236→        findings["records"]["A"] = ipv4

  1252→        findings["records"]["CNAME"] = cname_records

  1257→            ns_host = ns.rstrip(".")

  1259→                orphaned_ns.append(ns_host)

  1270→                if not smtp_open:
  1271→                    stale_mx.append({"host": mx_host, "ip": ip, "issue": "smtp_port_closed"})
  1272→            else:
  1273→                stale_mx.append({"host": mx_host, "ip": None, "issue": "unresolvable"})

  1289→                status_code=None,

  1314→                discovered_by="dns_analysis",

  1318→        return findings

  1320→    def _dns_lookup(self, domain: str, record_type: str) -> List[str]:
  1321→        """Perform a DNS lookup using system resolver (dig fallback or getaddrinfo)."""
  1322→        results: List[str] = []
  1323→        if record_type in ("A", "AAAA"):
  1324→            try:
  1325→                family = socket.AF_INET if record_type == "A" else socket.AF_INET6
  1326→                addrs = socket.getaddrinfo(domain, None, family, socket.SOCK_STREAM)
  1327→                for addr in addrs:
  1328→                    ip = addr[4][0]
  1329→                    if ip not in results:
  1330→                        results.append(ip)
  1331→            except (socket.gaierror, OSError):
  1332→                pass
  1333→            return results
  1334→
  1335→        # For NS, MX, TXT, CNAME — try system dig/host commands
  1336→        try:
  1337→            cmd = ["dig", "+short", record_type, domain]
  1338→            proc = subprocess.run(
  1339→                cmd, capture_output=True, text=True, timeout=8,
  1340→                stdin=subprocess.DEVNULL,
  1341→            )
  1342→            if proc.returncode == 0 and proc.stdout.strip():
  1343→                for line in proc.stdout.strip().split("\n"):
  1344→                    line = line.strip()
  1345→                    if line and line not in results:
  1346→                        results.append(line)
  1347→                return results
  1348→        except (FileNotFoundError, subprocess.TimeoutExpired, OSError):
  1349→            pass
  1350→
  1351→        # Fallback: try host command
  1352→        try:
  1353→            cmd = ["host", "-t", record_type, domain]
  1354→            proc = subprocess.run(
  1355→                cmd, capture_output=True, text=True, timeout=8,
  1356→                stdin=subprocess.DEVNULL,
  1357→            )
  1358→            if proc.returncode == 0 and proc.stdout.strip():
  1359→                for line in proc.stdout.strip().split("\n"):
  1360→                    line = line.strip()
  1361→                    # Extract the value from host output (typically after "is an alias for" or "mail is handled by")
  1362→                    parts = line.split()
  1363→                    if len(parts) >= 2:
  1364→                        value = parts[-1].rstrip(".")
  1365→                        if value and value not in results:
  1366→                            results.append(value + ".")
  1367→                return results
  1368→        except (FileNotFoundError, subprocess.TimeoutExpired, OSError):
  1369→            pass
  1370→
  1371→        return results
  1372→
  1373→    def _check_port_open(self, host: str, port: int, timeout: int = 4) -> bool:
  1374→        """Check if a TCP port is open on a host."""
  1375→        try:
  1376→            with socket.create_connection((host, port), timeout=timeout):
  1377→                return True
  1378→        except (socket.timeout, socket.gaierror, OSError):
  1379→            return False

  1463→            decay = _calculate_decay_from_indicators(indicators, resp_headers, cert_info)
  1464→            # Cloud leaks are inherently high risk
  1465→            if credential_hits:
  1466→                decay = max(decay, 80)
  1467→
  1468→            profile = AssetProfile(
  1469→                url=url,
  1470→                asset_type=asset_type,
  1471→                status_code=status_code,
  1472→                decay_score=decay,
  1473→                indicators=indicators,
  1474→                last_modified=resp_headers.get("last-modified") if resp_headers else None,
  1475→                server_header=resp_headers.get("server") if resp_headers else None,
  1476→                title=title,
  1477→                cert_info=cert_info,
  1478→                tech_stack=tech_stack,
  1479→                discovered_by="cloud_leak_scan",
  1480→            )
  1481→            found.append(profile)
  1482→
  1483→        # Phase 2: Scan the main page for leaked credentials in JS/HTML
  1484→        resp_headers, status_code, body, _ = _make_request(base_url, self.timeout, headers=self._session_headers)
  1485→        if body and status_code in (200, 301, 302):
  1486→            credential_hits = self._scan_for_credentials(body)
  1487→            if credential_hits:
  1488→                cert_info = _check_certificate_age(urlparse(base_url).hostname or "")
  1489→                title = _extract_title(body)
  1490→                profile = AssetProfile(
  1491→                    url=base_url,
  1492→                    asset_type="cloud_credentials",
  1493→                    status_code=status_code,
  1494→                    decay_score=max(80, len(credential_hits) * 15),
  1495→                    indicators=credential_hits[:10],
  1496→                    title=title,
  1497→                    cert_info=cert_info,
  1498→                    discovered_by="cloud_leak_body_scan",
  1499→                )
  1500→                found.append(profile)
  1501→
  1502→        return found

  1504→    def _scan_for_credentials(self, body: str) -> List[str]:
  1505→        """Scan response body for known credential/secret patterns."""
  1506→        hits: List[str] = []
  1507→        seen_types: set = set()
  1508→        for pattern in self.CLOUD_CREDENTIAL_PATTERNS:
  1509→            matches = pattern.findall(body)
  1510→            if matches:
  1511→                # Determine credential type from pattern
  1512→                match_str = matches[0][:20]  # truncated for safety
  1513→                # Classify the type
  1514→                for label, detect_pat in [
  1515→                    ("aws_access_key_detected", re.compile(r'AKIA')),
  1516→                    ("gcp_service_account_detected", re.compile(r'"type"\s*:\s*"service_account"')),
  1517→                    ("firebase_api_key_detected", re.compile(r'AIza')),
  1518→                    ("google_oauth_detected", re.compile(r'ya29\\.')),
  1519→                    ("slack_token_detected", re.compile(r'xox[bpras]-')),
  1520→                    ("github_pat_detected", re.compile(r'ghp_')),
  1521→                    ("github_oauth_detected", re.compile(r'gho_')),
  1522→                    ("gitlab_pat_detected", re.compile(r'glpat-')),
  1523→                    ("jwt_token_detected", re.compile(r'^eyJ', re.M)),
  1524→                    ("private_key_detected", re.compile(r'BEGIN.*PRIVATE KEY')),
  1525→                    ("stripe_key_detected", re.compile(r'sk_live_')),

  1533→                    ("openai_key_detected", re.compile(r'sk-[0-9a-zA-Z]{48}')),
  1534→                ]:
  1535→                    if detect_pat.search(body) and label not in seen_types:

  1544→    def discover_internal_exposure(self, base_url: str) -> List[AssetProfile]:
  1545→        """
  1546→        Find accidentally exposed internal tools: internal dashboards, monitoring
  1547→        panels, error tracking (Sentry), logging (ELK), CI/CD dashboards.
  1548→        """
  1549→        found: List[AssetProfile] = []
  1550→        probed_urls: set = set()
  1551→
  1552→        for path in self.INTERNAL_EXPOSURE_PATHS:
  1553→            url = urljoin(base_url.rstrip("/") + "/", path.lstrip("/"))
  1554→            if url in probed_urls:
  1555→                continue
  1556→            probed_urls.add(url)
  1557→            resp_headers, status_code, body, final_url = _make_request(url, self.timeout, headers=self._session_headers)
  1558→            if status_code is None or status_code == 404:
  1559→                continue
  1560→            if status_code not in (200, 201, 202, 301, 302, 303, 307, 308, 401, 403):
  1561→                continue
  1562→            cert_info = _check_certificate_age(urlparse(url).hostname or "")
  1563→            title = _extract_title(body or "")
  1564→            tech_stack = _detect_tech_stack(resp_headers, body or "")
  1565→            indicators = _assess_indicators(resp_headers, body or "", title, cert_info)
  1566→            asset_type = _classify_asset(resp_headers, body or "", title, cert_info)
  1567→            decay = _calculate_decay_from_indicators(indicators, resp_headers, cert_info)
  1568→
  1569→            # Internal tools exposed publicly are high risk regardless of decay
  1570→            if status_code in (200, 401, 403):
  1571→                decay = max(decay, 50)
  1572→                if asset_type == "web_application":
  1573→                    # Try to infer internal tool type from path/title
  1574→                    bl = (body or "").lower()
  1575→                    tl = title.lower()
  1576→                    if any(kw in bl or kw in tl for kw in ["monitoring", "grafana", "kibana", "prometheus", "alert"]):
  1577→                        asset_type = "monitoring_exposed"
  1578→                        indicators.append("internal_monitoring_exposed")
  1579→                        decay = max(decay, 70)
  1580→                    elif any(kw in bl or kw in tl for kw in ["jenkins", "gitlab", "ci", "pipeline", "build"]):
  1581→                        asset_type = "cicd_exposed"
  1582→                        indicators.append("internal_cicd_exposed")
  1583→                        decay = max(decay, 70)
  1584→                    elif any(kw in bl or kw in tl for kw in ["sentry", "rollbar", "bugsnag", "datadog", "newrelic"]):
  1585→                        asset_type = "error_tracking_exposed"
  1586→                        indicators.append("internal_error_tracking_exposed")
  1587→                        decay = max(decay, 60)
  1588→                    elif any(kw in bl or kw in tl for kw in ["admin", "dashboard", "backoffice", "intranet"]):
  1589→                        asset_type = "internal_dashboard_exposed"
  1590→                        indicators.append("internal_dashboard_exposed")
  1591→                        decay = max(decay, 65)
  1592→                    elif any(kw in bl or kw in tl for kw in ["log", "elk", "elasticsearch", "logstash"]):
  1593→                        asset_type = "logging_exposed"
  1594→                        indicators.append("internal_logging_exposed")
  1595→                        decay = max(decay, 65)
  1596→
  1597→            profile = AssetProfile(
  1598→                url=url,
  1599→                asset_type=asset_type,
  1600→                status_code=status_code,
  1601→                decay_score=min(decay, 100),
  1602→                indicators=indicators,
  1603→                last_modified=resp_headers.get("last-modified") if resp_headers else None,
  1604→                server_header=resp_headers.get("server") if resp_headers else None,
  1605→                title=title,
  1606→                cert_info=cert_info,
  1607→                tech_stack=tech_stack,
  1608→                discovered_by="internal_exposure_scan",
  1609→            )
  1610→            found.append(profile)

  1612→        return found

  1616→    # ---------------------------------------------------------------------

  1618→    def calculate_decay_score(self, asset_info: Dict[str, Any]) -> int:
  1619→        """
  1620→        Calculate how "abandoned" an asset is on a 0-100 scale.
  1621→
  1622→        Factors:
  1623→        - Certificate expiry proximity (expired = +25, near expiry = +12)
  1624→        - Outdated server headers (+15)
  1625→        - Default content pages (+20)
  1626→        - Missing security headers (HSTS, CSP) (+5 each)
  1627→        - HTTP-only with no redirect (+8)
  1628→        - No robots.txt (+3), no favicon (+3)
  1629→        - Old WordPress version signatures (+18)
  1630→        - Outdated JS libraries (jQuery 1.x, Bootstrap 3.x) (+10)
  1631→        - Visible PHP errors (+15)
  1632→        - Stack traces exposed (+20)
  1633→        - Debug mode indicators (+18)
  1634→        - Directory listings (+18)
  1635→        - Self-signed certificates (+15)
  1636→        - Certificate age > 1 year (+5), > 2 years (+8)
  1637→        - No Last-Modified header (+2)
  1638→
  1639→        :param asset_info: Dictionary with keys like ``headers``, ``body``,
  1640→            ``title``, ``cert_info``, ``status_code``.
  1641→        :returns: Integer 0-100 where 100 means fully abandoned/decayed.
  1642→        """
  1643→        headers = asset_info.get("headers") or {}
  1644→        body = asset_info.get("body", "")
  1645→        title = asset_info.get("title", "")
  1646→        cert_info = asset_info.get("cert_info")
  1647→        status_code = asset_info.get("status_code")
  1648→
  1649→        indicators = _assess_indicators(headers, body, title, cert_info)
  1650→        score = _calculate_decay_from_indicators(indicators, headers, cert_info)
  1651→
  1652→        # Additional: check for HTTP-only
  1653→        if asset_info.get("http_only") and not asset_info.get("has_https_redirect"):
  1654→            score += 8
  1655→
  1656→        # Check for no robots.txt
  1657→        if asset_info.get("no_robots_txt"):
  1658→            score += 3
  1659→
  1660→        # High status codes suggest degradation
  1661→        if status_code and status_code >= 500:
  1662→            score += 10
  1663→        elif status_code and status_code == 403:
  1664→            score += 5
  1665→
  1666→        return min(score, 100)
