    21→    21→# 1. Shannon Entropy
    22→    22→# ──────────────────────────────────────────────────────────────────────────────
    23→    23→
    24→    24→def shannon_entropy(s: str) -> float:
    25→    25→    """Calculate Shannon entropy of *s* in bits.
    26→    26→
    27→    27→    Returns 0.0 for empty or single-character strings.
    28→    28→    Formula: H = -sum(p_i * log2(p_i) for each unique character frequency p_i).
    29→    29→    """
    30→    30→    if not s or len(s) <= 1:
    31→    31→        return 0.0
    32→    32→
    33→    33→    freq: Dict[str, int] = {}
    34→    34→    for ch in s:
    35→    35→        freq[ch] = freq.get(ch, 0) + 1
    36→    36→
    37→    37→    length = len(s)
    38→    38→    entropy = 0.0
    39→    39→    for count in freq.values():
    40→    40→        p = count / length
    41→    41→        entropy -= p * math.log2(p)
    42→    42→
    43→    43→    return entropy

   129→   129→# 5. ContextAnalyzer
   130→   130→# ──────────────────────────────────────────────────────────────────────────────
   131→   131→
   132→   132→class ContextAnalyzer:
   133→   133→    """Extract contextual signals from variable names, file paths, and code."""
   134→   134→
   135→   135→    SECRET_KEYWORDS: List[str] = [
   136→   136→        "password", "passwd", "pass",
   137→   137→        "secret", "token", "apikey", "api_key", "api-key",
   138→   138→        "access_key", "accesskey", "private_key", "privatekey",
   139→   139→        "credentials", "creds", "credential",
   140→   140→        "auth", "authorization", "authenticate",
   141→   141→        "private", "cert", "certificate",
   142→   142→    ]
   143→   143→
   144→   144→    SENSITIVE_FILENAMES: List[str] = [
   145→   145→        ".env", ".env.local", ".env.production", ".env.development",
   146→   146→        ".env.staging", ".env.test", ".env.backup",
   147→   147→        "credentials", "credential", "creds",
   148→   148→        "config", "configuration", "settings", "settings.local",
   149→   149→        "secrets", "secret",
   150→   150→        "id_rsa", "id_ed25519", "id_ecdsa", "id_dsa",
   151→   151→        ".netrc", ".pgpass", ".my.cnf",
   152→   152→    ]
   153→   153→
   154→   154→    TEMPORARY_COMMENTS: List[str] = [
   155→   155→        "todo: remove", "fixme", "temporary", "hack", "workaround",
   156→   156→        "remove before", "delete this", "not for production",
   157→   157→        "debug only", "testing only", "do not commit", "don't commit",
   158→   158→        "wip", "xxx",
   159→   159→    ]
   160→   160→
   161→   161→    def analyze_variable_name(self, name: str) -> List[str]:
   162→   162→        """Return signal strings found in *name* (e.g. 'contains:password')."""
   163→   163→        signals: List[str] = []
   164→   164→        lower = name.lower().replace("-", "_").replace(" ", "_")
   165→   165→        for kw in self.SECRET_KEYWORDS:
   166→   166→            if kw in lower:
   167→   167→                signals.append(f"var:{kw}")
   168→   168→        return signals
   169→   169→
   170→   170→    def analyze_file_path(self, path: str) -> List[str]:
   171→   171→        """Return signal strings for sensitive file/directory names in *path*."""
   172→   172→        signals: List[str] = []
   173→   173→        lower = path.lower().replace(os.sep, "/")
   174→   174→        parts = lower.split("/")
   175→   175→        for part in parts:
   176→   176→            for sf in self.SENSITIVE_FILENAMES:
   177→   177→                if part == sf or part.startswith(sf + "."):
   178→   178→                    signals.append(f"file:{sf}")
   179→   179→                    break
   180→   180→        return signals
   181→   181→
   182→   182→    def analyze_surrounding_code(self, code: str) -> List[str]:
   183→   183→        """Return signal strings for temporary / debug comments in *code*."""
   184→   184→        signals: List[str] = []
   185→   185→        upper = code.upper()
   186→   186→        for tc in self.TEMPORARY_COMMENTS:
   187→   187→            if tc.upper() in upper:
   188→   188→                signals.append(f"comment:{tc.lower()}")
   189→   189→        return signals

   192→   192→# ──────────────────────────────────────────────────────────────────────────────
   193→   193→# 3. SecretClassifier
   194→   194→# ──────────────────────────────────────────────────────────────────────────────
   195→   195→
   196→   196→class SecretClassifier:
   197→   197→    """Classify a string as a real secret, weak secret, or benign."""
   198→   198→
   199→   199→    def __init__(self) -> None:
   200→   200→        self._ctx = ContextAnalyzer()
   201→   201→        self._prefix_lookup = self._build_prefix_lookup()
   202→   202→
   203→   203→    # -- prefix lookup is ordered longest-prefix-first so substrings don't
   204→   204→    # -- steal matches from longer, more specific prefixes (e.g. sk_live_ before s)
   205→   205→    @staticmethod
   206→   206→    def _build_prefix_lookup() -> List[tuple[str, str, float]]:
   207→   207→        items = [(prefix, info["type"], float(info["confidence"]))
   208→   208→                 for prefix, info in KNOWN_PREFIXES.items()]
   209→   209→        items.sort(key=lambda x: len(x[0]), reverse=True)
   210→   210→        return items
   211→   211→
   212→   212→    # ── public API ──────────────────────────────────────────────────────────
   213→   213→
   214→   214→    def classify(
   215→   215→        self,
   216→   216→        value: str,
   217→   217→        variable_name: str = "",
   218→   218→        file_path: str = "",
   219→   219→        surrounding_code: str = "",
   220→   220→    ) -> SecretMatch:
   221→   221→        """Classify *value* and return a :class:`SecretMatch`.
   222→   222→
   223→   223→        Parameters
   224→   224→        ----------
   225→   225→        value : str
   226→   226→            The candidate secret string.
   227→   227→        variable_name : str
   228→   228→            Name of the variable/field holding the value (e.g. "AWS_SECRET_KEY").
   229→   229→        file_path : str
   230→   230→            Path of the file where the value was found.
   231→   231→        surrounding_code : str
   232→   232→            A few lines of code around the match for comment analysis.
   233→   233→        """
   234→   234→        entropy = shannon_entropy(value)
   235→   235→
   236→   236→        # --- a) Match known prefix (longest first) ---
   237→   237→        matched_prefix: Optional[str] = None
   238→   238→        prefix_type = ""
   239→   239→        prefix_confidence = 0.0
   240→   240→        value_upper = value.upper()
   241→   241→        for pfx, ptype, pconf in self._prefix_lookup:
   242→   242→            if value.startswith(pfx) or value_upper.startswith(pfx.upper()):
   243→   243→                matched_prefix = pfx
   244→   244→                prefix_type = ptype
   245→   245→                prefix_confidence = pconf
   246→   246→                break
   247→   247→
   248→   248→        # --- b) Context signals ---
   249→   249→        context_signals: List[str] = []
   250→   250→        context_signals.extend(self._ctx.analyze_variable_name(variable_name))
   251→   251→        context_signals.extend(self._ctx.analyze_file_path(file_path))
   252→   252→        context_signals.extend(self._ctx.analyze_surrounding_code(surrounding_code))
   253→   253→
   254→   254→        has_secret_var = any(s.startswith("var:") for s in context_signals)
   255→   255→        has_sensitive_file = any(s.startswith("file:") for s in context_signals)
   256→   256→        has_temp_comment = any(s.startswith("comment:") for s in context_signals)
   257→   257→
   258→   258→        # --- c) Format validation ---
   259→   259→        format_valid = self._validate_format(value, matched_prefix)
   260→   260→
   261→   261→        # --- d) Classification matrix ---
   262→   262→        severity, confidence, recommendation = self._classify_matrix(
   263→   263→            entropy=entropy,
   264→   264→            has_prefix=bool(matched_prefix),
   265→   265→            prefix_confidence=prefix_confidence,
   266→   266→            has_secret_var=has_secret_var,
   267→   267→            has_sensitive_file=has_sensitive_file,
   268→   268→            has_temp_comment=has_temp_comment,
   269→   269→            format_valid=format_valid,
   270→   270→        )
   271→   271→
   272→   272→        secret_type = prefix_type if prefix_type else self._infer_type(context_signals, value)
   273→   273→
   274→   274→        # Downgrade confidence slightly if marked as temporary
   275→   275→        if has_temp_comment and confidence > 0.3:
   276→   276→            confidence *= 0.6
   277→   277→
   278→   278→        return SecretMatch(
   279→   279→            value=value,
   280→   280→            secret_type=secret_type,
   281→   281→            severity=severity,
   282→   282→            confidence=round(confidence, 3),
   283→   283→            entropy=round(entropy, 3),
   284→   284→            matched_prefix=matched_prefix,
   285→   285→            context_signals=context_signals,
   286→   286→            recommendation=recommendation,
   287→   287→        )
   288→   288→
   289→   289→    # ── internal helpers ────────────────────────────────────────────────────
   290→   290→
   291→   291→    def _classify_matrix(
   292→   292→        self,
   293→   293→        entropy: float,
   294→   294→        has_prefix: bool,
   295→   295→        prefix_confidence: float,
   296→   296→        has_secret_var: bool,
   297→   297→        has_sensitive_file: bool,
   298→   298→        has_temp_comment: bool,
   299→   299→        format_valid: bool,
   300→   300→    ) -> tuple[str, float, str]:
   301→   301→        """Apply the classification rules and return (severity, confidence, rec)."""
   302→   302→
   303→   303→        # Known prefix + high entropy + secret variable name → CRITICAL
   304→   304→        if has_prefix and entropy > 3.5 and has_secret_var:
   305→   305→            return (
   306→   306→                "CRITICAL",
   307→   307→                min(prefix_confidence * 1.04, 0.99),  # cap at 99%
   308→   308→                "Real secret detected with high confidence. Rotate immediately.",
   309→   309→            )
   310→   310→
   311→   311→        # Known prefix + high entropy → HIGH
   312→   312→        if has_prefix and entropy > 3.5:
   313→   313→            conf = prefix_confidence if format_valid else prefix_confidence * 0.85
   314→   314→            return (
   315→   315→                "HIGH",
   316→   316→                conf,
   317→   317→                "Likely a real secret based on known prefix and high entropy."
   318→   318→                + (" Format is valid." if format_valid else " Format is unexpected."),
   319→   319→            )
   320→   320→
   321→   321→        # High entropy + secret variable name → HIGH
   322→   322→        if entropy >= 3.5 and has_secret_var:
   323→   323→            return (
   324→   324→                "HIGH",
   325→   325→                0.85,
   326→   326→                "High-entropy value assigned to a secret-related variable. Likely a real secret.",
   327→   327→            )
   328→   328→
   329→   329→        # High entropy + sensitive file → HIGH
   330→   330→        if entropy >= 3.5 and has_sensitive_file:
   331→   331→            return (
   332→   332→                "HIGH",
   333→   333→                0.80,
   334→   334→                "High-entropy value found in a sensitive file. Likely a real secret.",
   335→   335→            )
   336→   336→
   337→   337→        # High entropy + no context → MEDIUM (could be UUID, session ID, hash)
   338→   338→        if entropy >= 3.5:
   339→   339→            return (
   340→   340→                "MEDIUM",
   341→   341→                0.50,
   342→   342→                "High-entropy string but no secret-related context. Could be a UUID, hash, or session ID.",
   343→   343→            )
   344→   344→
   345→   345→        # Low entropy + known prefix (e.g. password123) → HIGH (weak but real)
   346→   346→        if has_prefix and entropy <= 3.5:
   347→   347→            return (
   348→   348→                "HIGH",
   349→   349→                prefix_confidence * 0.7,
   350→   350→                "Known secret prefix with low entropy — likely a weak or placeholder secret. Still a real leak.",
   351→   351→            )
   352→   352→
   353→   353→        # Low entropy + secret variable name → MEDIUM
   354→   354→        if entropy < 3.5 and has_secret_var:
   355→   355→            return (
   356→   356→                "MEDIUM",
   357→   357→                0.55,
   358→   358→                "Secret-related variable holds a low-entropy value. Could be a weak or default credential.",
   359→   359→            )
   360→   360→
   361→   361→        # Low entropy + no context → INFO
   362→   362→        return (
   363→   363→            "INFO",
   364→   364→            0.10,
   365→   365→            "Low entropy with no secret context. Probably not a secret.",
   366→   366→        )
   367→   367→
   368→   368→    def _validate_format(self, value: str, prefix: Optional[str]) -> bool:
   369→   369→        """Check value format against expectations for its prefix type."""
   370→   370→        if not prefix:
   371→   371→            return self._validate_generic_api_key(value)
   372→   372→
   373→   373→        pu = prefix.upper()
   374→   374→
   375→   375→        # AWS access key: exactly 20 chars uppercase+digits after AKIA/ASIA
   376→   376→        if pu in ("AKIA", "ASIA"):
   377→   377→            suffix = value[len(prefix):]
   378→   378→            return len(suffix) == 16 and suffix.isalnum() and suffix.isupper()
   379→   379→
   380→   380→        # GitHub tokens: exactly 36 hex chars after prefix
   381→   381→        if pu in ("GHP_", "GHO_", "GHU_", "GHS_", "GHR_"):
   382→   382→            suffix = value[len(prefix):]
   383→   383→            return len(suffix) == 36 and all(c in "0123456789abcdefABCDEF" for c in suffix)
   384→   384→
   385→   385→        # GitLab PAT
   386→   386→        if pu in ("GLPAT-", "GLPTT-"):
   387→   387→            suffix = value[len(prefix):]
   388→   388→            return len(suffix) >= 20 and bool(re.match(r"^[\w-]+$", suffix))
   389→   389→
   390→   390→        # Slack tokens
   391→   391→        if pu in ("XOXB-", "XOXP-", "XOXO-"):
   392→   392→            suffix = value[len(prefix):]
   393→   393→            return len(suffix) >= 10 and bool(re.match(r"^[\w-]+$", suffix))
   394→   394→
   395→   395→        # Stripe keys
   396→   396→        if pu in ("SK_LIVE_", "PK_LIVE_", "RK_LIVE_"):
   397→   397→            suffix = value[len(prefix):]
   398→   398→            return len(suffix) >= 20 and bool(re.match(r"^[\w]+$", suffix))
   399→   399→
   400→   400→        # Twilio: SK followed by hex
   401→   401→        if pu == "SK" and len(prefix) == 2 and len(value) > 2:
   402→   402→            suffix = value[2:]
   403→   403→            return suffix and all(c in "0123456789abcdef" for c in suffix)
   404→   404→
   405→   405→        # Google API key: AIza + 33 chars
   406→   406→        if prefix == "AIza":
   407→   407→            suffix = value[4:]
   408→   408→            return len(suffix) == 33 and bool(re.match(r"^[\w-]+$", suffix))
   409→   409→
   410→   410→        # Google OAuth: ya29. + token chars
   411→   411→        if value.startswith("ya29."):
   412→   412→            suffix = value[5:]
   413→   413→            return len(suffix) >= 10 and bool(re.match(r"^[\w._-]+$", suffix))
   414→   414→
   415→   415→        # JWT: three base64url segments
   416→   416→        if prefix == "eyJ":
   417→   417→            parts = value.split(".")
   418→   418→            if len(parts) != 3:
   419→   419→                return False
   420→   420→            b64_pattern = re.compile(r"^[A-Za-z0-9_-]+$")
   421→   421→            return all(b64_pattern.match(p) for p in parts)
   422→   422→
   423→   423→        # Azure prefixes
   424→   424→        if pu in ("DEFAULTAZURE", "AZURE_CLIENT"):
   425→   425→            return len(value) > len(prefix)
   426→   426→
   427→   427→        return self._validate_generic_api_key(value)
   428→   428→
   429→   429→    @staticmethod
   430→   430→    def _validate_generic_api_key(value: str) -> bool:
   431→   431→        """Generic API key heuristic: 20-100 chars, mix of upper + lower + digit."""
   432→   432→        if not (20 <= len(value) <= 100):
   433→   433→            return False
   434→   434→        has_upper = any(c.isupper() for c in value)
   435→   435→        has_lower = any(c.islower() for c in value)
   436→   436→        has_digit = any(c.isdigit() for c in value)
   437→   437→        return has_upper and has_lower and has_digit
   438→   438→
   439→   439→    @staticmethod
   440→   440→    def _infer_type(signals: List[str], value: str) -> str:
   441→   441→        """Guess the secret type from context signals when no prefix matched."""
   442→   442→        for sig in signals:
   443→   443→            kw = sig.split(":", 1)[1] if ":" in sig else sig
   444→   444→            kw = kw.replace("_", " ").title()
   445→   445→        # Use the first variable signal if present
   446→   446→        for sig in signals:
   447→   447→            if sig.startswith("var:"):
   448→   448→                kw = sig[4:].replace("_", " ").title()
   449→   449→                return f"Potential {kw}"
   450→   450→        return "Unknown Secret"
   451→   451→
   452→   452→
   453→   453→# ──────────────────────────────────────────────────────────────────────────────
   454→   454→# 7 & 8. Convenience scanning functions
   455→   455→# ──────────────────────────────────────────────────────────────────────────────
   456→   456→
   457→   457→# Regex to extract string literals from source files (covers Python, JS, TS, etc.)
   458→   458→_STRING_LITERAL_RE = re.compile(
   459→   459→    r'"""(?:[^"\\]|\\.)*?"""'
   460→   460→    r"|'''(?:[^'\\]|\\.)*?'''"
   461→   461→    r'|"(?:[^"\\]|\\.)*"'
   462→   462→    r"|'(?:[^'\\]|\\.)*'"
   463→   463→    r'|`(?:[^`\\]|\\.)*`',
   464→   464→)
   465→   465→
   466→   466→# Regex to extract variable = "value" or KEY = value patterns from config/env files
   467→   467→# Uses \x5c for backslash to avoid quoting issues inside character classes.
   468→   468→_ASSIGNMENT_RE = re.compile(
   469→   469→    r'(?:^|(?<=[\s,;{]))'
   470→   470→    r'([A-Za-z_][\w]*)'
   471→   471→    r'\s*(?:=|:)\s*'
   472→   472→    r'(["\x27](?:[^"\x27\x5c]|\x5c.)*["\x27])',
   473→   473→    re.MULTILINE,
   474→   474→)
   475→   475→
   476→   476→# Minimum entropy to even bother classifying (below this, classify returns INFO anyway)
   477→   477→_ENTROPY_FLOOR = 2.5
   478→   478→
   479→   479→
   480→   480→def scan_string(
   481→   481→    value: str,
   482→   482→    variable_name: str = "",
   483→   483→    file_path: str = "",
   484→   484→    surrounding_code: str = "",
   485→   485→) -> Optional[SecretMatch]:
   486→   486→    """Classify a single candidate string.
   487→   487→
   488→   488→    Returns ``None`` when the value is clearly not a secret (very low entropy,
   489→   489→    too short, empty, or classified as INFO).
   490→   490→    """
   491→   491→    if not value or len(value) < 6:
   492→   492→        return None
   493→   493→
   494→   494→    classifier = SecretClassifier()
   495→   495→    result = classifier.classify(
   496→   496→        value=value,
   497→   497→        variable_name=variable_name,
   498→   498→        file_path=file_path,
   499→   499→        surrounding_code=surrounding_code,
   500→   500→    )
   501→   501→
   502→   502→    if result.severity == "INFO":
   503→   503→        return None
   504→   504→
   505→   505→    return result

   508→   508→def scan_file(path: str) -> List[SecretMatch]:
   509→   509→    """Read a file, extract string literals, and classify each as a potential secret.
   510→   510→
   511→   511→    Skips binary files and files larger than 256 KB.
   512→   512→    """
   513→   513→    results: List[SecretMatch] = []
   514→   514→    classifier = SecretClassifier()
   515→   515→
   516→   516→    try:
   517→   517→        file_size = os.path.getsize(path)
   518→   518→        if file_size > 256 * 1024 or file_size == 0:
   519→   519→            return results
   520→   520→    except OSError:
   521→   521→        return results
   522→   522→
   523→   523→    try:
   524→   524→        with open(path, errors="replace") as fh:
   525→   525→            content = fh.read()
   526→   526→    except OSError:
   527→   527→        return results
   528→   528→
   529→   529→    # Quick binary detection: if >5% of first 4 KB are null bytes, skip
   530→   530→    sample = content[:4096]
   531→   531→    if sample.count("\x00") > len(sample) * 0.05:
   532→   532→        return results
   533→   533→
   534→   534→    seen_values: set[str] = set()
   535→   535→    lines = content.splitlines()
   536→   536→
   537→   537→    # Strategy 1: Extract assignment patterns (VAR = "value")
   538→   538→    for m in _ASSIGNMENT_RE.finditer(content):
   539→   539→        var_name = m.group(1)
   540→   540→        raw_val = m.group(2)
   541→   541→        # Strip surrounding quotes
   542→   542→        if (raw_val.startswith('"') and raw_val.endswith('"')) or \
   543→   543→           (raw_val.startswith("'") and raw_val.endswith("'")):
   544→   544→            val = raw_val[1:-1]
   545→   545→        else:
   546→   546→            val = raw_val
   547→   547→
   548→   548→        if len(val) < 6 or val in seen_values:
   549→   549→            continue
   550→   550→        seen_values.add(val)
   551→   551→
   552→   552→        # Get surrounding code (3 lines before and after the match)
   553→   553→        start_line = content[:m.start()].count("\n")
   554→   554→        ctx_start = max(0, start_line - 3)
   555→   555→        ctx_end = min(len(lines), start_line + 4)
   556→   556→        surrounding = "\n".join(lines[ctx_start:ctx_end])
   557→   557→
   558→   558→        match = classifier.classify(
   559→   559→            value=val,
   560→   560→            variable_name=var_name,
   561→   561→            file_path=path,
   562→   562→            surrounding_code=surrounding,
   563→   563→        )
   564→   564→        if match.severity != "INFO":
   565→   565→            results.append(match)
   566→   566→
   567→   567→    # Strategy 2: Extract bare string literals that look high-entropy
   568→   568→    for m in _STRING_LITERAL_RE.finditer(content):
   569→   569→        raw_val = m.group(0)
   570→   570→        # Strip quotes
   571→   571→        if (raw_val.startswith('"""') and raw_val.endswith('"""')) or \
   572→   572→           (raw_val.startswith("'''") and raw_val.endswith("'''")):
   573→   573→            val = raw_val[3:-3]
   574→   574→        elif raw_val and raw_val[0] in ("'", '"', '`') and raw_val[-1] == raw_val[0]:
   575→   575→            val = raw_val[1:-1]
   576→   576→        else:
   577→   577→            val = raw_val
   578→   578→
   579→   579→        if len(val) < 8 or val in seen_values:
   580→   580→            continue
   581→   581→
   582→   582→        # Quick entropy pre-filter to avoid classifying every string
   583→   583→        ent = shannon_entropy(val)
   584→   584→        if ent < _ENTROPY_FLOOR:
   585→   585→            continue
   586→   586→
   587→   587→        seen_values.add(val)
   588→   588→
   589→   589→        start_line = content[:m.start()].count("\n")
   590→   590→        ctx_start = max(0, start_line - 2)
   591→   591→        ctx_end = min(len(lines), start_line + 3)
   592→   592→        surrounding = "\n".join(lines[ctx_start:ctx_end])
   593→   593→
   594→   594→        match = classifier.classify(
   595→   595→            value=val,
   596→   596→            file_path=path,
   597→   597→            surrounding_code=surrounding,
   598→   598→        )
   599→   599→        if match.severity != "INFO":
   600→   600→            results.append(match)
   601→   601→
   602→   602→    return results
   603→   603→
   604→   604→
   605→   605→def scan_directory(
   606→   606→    path: str,
   607→   607→    extensions: Optional[List[str]] = None,
   608→   608→) -> List[SecretMatch]:
   609→   609→    """Recursively scan a directory for secrets in files.
   610→   610→
   611→   611→    Parameters
   612→   612→    ----------
   613→   613→    path : str
   614→   614→        Root directory to scan.
   615→   615→    extensions : list[str] | None
   616→   616→        File extensions to include (e.g. [".py", ".js"]).
   617→   617→        Defaults to common source/config extensions if *None*.
   618→   618→    """
   619→   619→    if extensions is None:
   620→   620→        extensions = [
   621→   621→            ".py", ".js", ".ts", ".tsx", ".jsx", ".mjs", ".cjs",
   622→   622→            ".rb", ".go", ".rs", ".java", ".php", ".c", ".cpp", ".h",
   623→   623→            ".yml", ".yaml", ".json", ".toml", ".cfg", ".ini",
   624→   624→            ".env", ".sh", ".bash", ".zsh", ".ps1",
   625→   625→            ".tf", ".hcl",                      # Terraform
   626→   626→            ".properties", ".xml", ".conf",     # Java / Nginx
   627→   627→            ".sql",                                # DB scripts
   628→   628→            ".gradle", ".cmake",                  # Build files
   629→   629→        ]
   630→   630→
   631→   631→    # Normalise extensions to lowercase with dot
   632→   632→    ext_set = {e.lower() if e.startswith(".") else f".{e.lower()}" for e in extensions}
   633→   633→
   634→   634→    root = Path(path)
   635→   635→    if not root.is_dir():
   636→   636→        return []
   637→   637→
   638→   638→    results: List[SecretMatch] = []
   639→   639→    # Directories to skip
   640→   640→    skip_dirs = {
   641→   641→        "node_modules", ".venv", "venv", "env", ".git", "__pycache__",
   642→   642→        ".tox", ".mypy_cache", ".pytest_cache", ".next", "dist", "build",
   643→   643→        ".cache", ".idea", ".vscode", "target", "vendor", ".bundle",
   644→   644→        "coverage", ".nyc_output", ".turbo",
   645→   645→    }
   646→   646→
   647→   647→    for filepath in root.rglob("*"):
   648→   648→        # Skip directories
   649→   649→        if filepath.is_dir():
   650→   650→            continue
   651→   651→        # Skip blacklisted directories
   652→   652→        if any(part in skip_dirs for part in filepath.parts):
   653→   653→            continue
   654→   654→        # Skip symlinks
   655→   655→        if filepath.is_symlink():
   656→   656→            continue
   657→   657→        # Check extension
   658→   658→        if ext_set and filepath.suffix.lower() not in ext_set:
   659→   659→            # Also allow extensionless sensitive filenames like .env, Dockerfile
   660→   660→            basename_lower = filepath.name.lower()
   661→   661→            if not any(basename_lower.startswith(sf) for sf in [".env", "dockerfile", "makefile", "credentials", "id_rsa"]):
   662→   662→                continue
   663→   663→
   664→   664→        results.extend(scan_file(str(filepath)))
   665→   665→
   666→   666→    return results
   667→   667→

   669→   669→# ── File Entropy Analysis ─────────────────────────────────────────────────
   670→   670→
   671→   671→def file_entropy(filepath: str) -> Dict[str, Any]:
   672→   672→    """Calculate Shannon entropy of a file. Returns {entropy, bits_per_byte, size, path, assessment}."""
   673→   673→    try:
   674→   674→        with open(filepath, "rb") as f:
   675→   675→            data = f.read()
   676→   676→        if not data:
   677→   677→            return {"entropy": 0.0, "bits_per_byte": 0.0, "size": 0, "path": filepath, "assessment": "empty"}
   678→   678→        ent = shannon_entropy(data.decode("latin-1"))
   679→   679→        return {
   680→   680→            "entropy": round(ent, 4),
   681→   681→            "bits_per_byte": round(ent / 8.0 * 100, 2),
   682→   682→            "size": len(data),
   683→   683→            "path": filepath,
   684→   684→            "assessment": "high_entropy" if ent > 7.0 else "medium_entropy" if ent > 5.0 else "low_entropy",
   685→   685→        }
   686→   686→    except Exception as e:
   687→   687→        return {"entropy": 0.0, "bits_per_byte": 0.0, "size": 0, "path": filepath, "assessment": f"error: {e}"}
   688→   688→

   690→   690→def directory_entropy_scan(directory: str, extensions: Optional[List[str]] = None) -> List[Dict[str, Any]]:
   691→   691→    """Scan all files in a directory for high entropy (potential encrypted/compiled secrets)."""
   692→   692→    results = []
   693→   693→    exts = set(extensions) if extensions else None
   694→   694→    for root, dirs, files in os.walk(directory):
   695→   695→        for fname in sorted(files):
   696→   696→            fpath = os.path.join(root, fname)
   697→   697→            if exts:
   698→   698→                _, ext = os.path.splitext(fname)
   699→   699→                if ext.lstrip(".").lower() not in exts:
   700→   700→                    continue
   701→   701→            info = file_entropy(fpath)
   702→   702→            if info["entropy"] > 5.0:
   703→   703→                results.append(info)
   704→   704→    return sorted(results, key=lambda x: -x["entropy"])

   707→   707→def crypto_material_detector(data: str, context: str = "") -> List[Dict[str, Any]]:
   708→   708→    """Detect cryptographic material (keys, certs, tokens) using entropy + patterns."""
   709→   709→    from .entropy import KNOWN_PREFIXES
   710→   710→    findings = []
   711→   711→    lines = data.split("\n")
   712→   712→    for i, line in enumerate(lines):
   713→   713→        stripped = line.strip()
   714→   714→        for prefix, info in KNOWN_PREFIXES.items():
   715→   715→            if stripped.startswith(prefix):
   716→   716→                ent = shannon_entropy(stripped)
   717→   717→                findings.append({
   718→   718→                    "line": i + 1,
   719→   719→                    "prefix": prefix,
   720→   720→                    "type": info["type"],
   721→   721→                    "entropy": round(ent, 4),
   722→   722→                    "value_preview": stripped[:20] + "..." if len(stripped) > 20 else stripped,
   723→   723→                    "confidence": info["confidence"],
   724→   724→                    "context": context,
   725→   725→                })
   726→   726→    return findings
