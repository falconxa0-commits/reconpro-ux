   463→
   464→    def classify(self, finding: Dict[str, Any]) -> ClassificationResult:
   465→        """Classify a single finding into an attack category.
   466→
   467→        Returns ClassificationResult with attack_category, confidence, and
   468→        matching indicators.
   469→        """
   470→        title = finding.get("title", "").lower()
   471→        description = finding.get("description", "").lower()
   472→        evidence = finding.get("evidence", "").lower()
   473→        category = finding.get("category", "").lower()
   474→        module = finding.get("module", "").lower()
   475→        severity = finding.get("severity", "info").lower()
   476→
   477→        scores: Dict[str, float] = defaultdict(float)
   478→        indicators: Dict[str, List[str]] = defaultdict(list)
   479→
   480→        # Phase 1: Title pattern matching
   481→        combined = f"{title} {description}"
   482→        for pattern, cat, weight in self._TITLE_RULES:
   483→            if re.search(pattern, combined, re.IGNORECASE):
   484→                scores[cat] += weight
   485→                indicators[cat].append(f"title_match:{pattern[:20]}")
   486→
   487→        # Phase 2: Module-based classification
   488→        if module in self._MODULE_RULES:
   489→            cat = self._MODULE_RULES[module]
   490→            scores[cat] += 0.75
   491→            indicators[cat].append(f"module_source:{module}")
   492→
   493→        # Phase 3: Category hint
   494→        if category:
   495→            for pattern, cat, weight in self._TITLE_RULES:
   496→                if re.search(pattern, category, re.IGNORECASE):
   497→                    scores[cat] += 0.5
   498→                    indicators[cat].append(f"category_hint:{category}")
   499→
   500→        # Phase 4: Evidence enrichment
   501→        for pattern, cat, weight in self._TITLE_RULES:
   502→            if re.search(pattern, evidence, re.IGNORECASE):
   503→                scores[cat] += 0.3
   504→                indicators[cat].append(f"evidence_match:{pattern[:20]}")
   505→
   506→        if not scores:
   507→            return ClassificationResult(
   508→                attack_category="misc",
   509→                confidence=0.3,
   510→                indicators=["no_patterns_matched"],
   511→            )
   512→
   513→        # Select best category
   514→        best_cat = max(scores, key=scores.get)
   515→        best_score = scores[best_cat]
   516→
   517→        # Normalize confidence: weight by evidence strength
   518→        has_evidence = len(evidence) > 10
   519→        has_description = len(description) > 20
   520→        evidence_boost = 0.15 if has_evidence else 0.0
   521→        desc_boost = 0.1 if has_description else 0.0
   522→        severity_boost = _sev_val(severity) / 100.0
   523→
   524→        raw_conf = min(1.0, best_score / 2.0 + evidence_boost + desc_boost + severity_boost)
   525→
   526→        return ClassificationResult(
   527→            attack_category=best_cat,
   528→            confidence=round(raw_conf, 2),
   529→            indicators=indicators[best_cat][:5],
   530→        )
   531→
   532→    def classify_batch(self, findings: List[Dict[str, Any]]) -> List[ClassificationResult]:
   533→        """Classify a batch of findings."""
   534→        return [self.classify(f) for f in findings]

   563→
   564→class FindingCorrelator:
   565→    """Correlates findings across modules to detect relationships.
   566→
   567→    Groups findings by shared assets, detects duplicates, and identifies
   568→    attack chains.
   569→    """
   570→
   571→    def correlate(self, findings: List[Dict[str, Any]]) -> Tuple[List[CorrelationGroup], List[Dict[str, Any]]]:
   572→        """Correlate findings. Returns (correlation_groups, deduplicated_findings)."""
   573→        if not findings:
   574→            return [], []
   575→
   576→        # Phase 1: Deduplicate
   577→        unique, dedup_count = self._deduplicate(findings)
   578→
   579→        # Phase 2: Group by shared asset
   580→        asset_groups = self._group_by_asset(unique)
   581→
   582→        # Phase 3: Detect attack chains
   583→        chain_groups = self._detect_chains(unique)
   584→
   585→        # Merge asset groups with chain groups (avoid duplicates)
   586→        all_groups = self._merge_groups(asset_groups, chain_groups)
   587→
   588→        return all_groups, unique
   589→
   590→    def _deduplicate(self, findings: List[Dict[str, Any]]) -> Tuple[List[Dict[str, Any]], int]:
   591→        """Remove duplicate findings based on title + asset + category."""
   592→        seen: Dict[str, Dict[str, Any]] = {}
   593→        dup_count = 0
   594→        for f in findings:
   595→            key = self._finding_hash(f)
   596→            if key in seen:
   597→                dup_count += 1
   598→                # Keep the one with more evidence
   599→                existing = seen[key]
   600→                if len(f.get("evidence", "")) > len(existing.get("evidence", "")):
   601→                    seen[key] = f
   602→            else:
   603→                seen[key] = f
   604→        return list(seen.values()), dup_count
   605→
   606→    @staticmethod
   607→    def _finding_hash(f: Dict[str, Any]) -> str:
   608→        """Create a deterministic hash for dedup."""
   609→        raw = f"{f.get('title', '')}:{f.get('asset', '')}:{f.get('category', '')}".lower().strip()
   610→        return hashlib.sha256(raw.encode()).hexdigest()[:16]
   611→
   612→    def _group_by_asset(self, findings: List[Dict[str, Any]]) -> List[CorrelationGroup]:
   613→        """Group findings that share the same asset."""
   614→        groups: Dict[str, List[Dict[str, Any]]] = defaultdict(list)
   615→        for f in findings:
   616→            asset = f.get("asset", "unknown")
   617→            groups[asset].append(f)
   618→
   619→        result: List[CorrelationGroup] = []
   620→        for asset, items in groups.items():
   621→            if len(items) < 2:
   622→                continue
   623→            sevs = [f.get("severity", "info") for f in items]
   624→            combined_risk = self._compute_combined_risk(sevs)
   625→            result.append(CorrelationGroup(
   626→                findings=items,
   627→                correlation_type="shared_asset",
   628→                combined_risk=combined_risk,
   629→                combined_confidence=0.7,
   630→            ))
   631→        return result
   632→
   633→    def _detect_chains(self, findings: List[Dict[str, Any]]) -> List[CorrelationGroup]:
   634→        """Detect attack chains using templates and category analysis."""
   635→        # Classify all findings
   636→        classifier = FindingClassifier()
   637→        classified: List[Tuple[Dict[str, Any], ClassificationResult]] = []
   638→        for f in findings:
   639→            cr = classifier.classify(f)
   640→            classified.append((f, cr))
   641→
   642→        chains: List[CorrelationGroup] = []
   643→        # Check all pairs against chain templates
   644→        for i, j in combinations(range(len(classified)), 2):
   645→            f_a, cr_a = classified[i]
   646→            f_b, cr_b = classified[j]
   647→            for cat_a, cat_b, chain_name in _ATTACK_CHAIN_TEMPLATES:
   648→                if (cr_a.attack_category == cat_a and cr_b.attack_category == cat_b) or \
   649→                   (cr_b.attack_category == cat_a and cr_a.attack_category == cat_b):
   650→                    # Verify shared asset or related
   651→                    shared = (f_a.get("asset", "") == f_b.get("asset", "") or
   652→                              self._assets_related(f_a.get("asset", ""), f_b.get("asset", "")))
   653→                    if shared or True:  # Even unrelated assets can form chains
   654→                        conf = min(cr_a.confidence, cr_b.confidence) * 0.8
   655→                        chains.append(CorrelationGroup(
   656→                            findings=[f_a, f_b],
   657→                            correlation_type="attack_chain",
   658→                            attack_chain=[chain_name],
   659→                            combined_risk=self._max_severity([f_a, f_b]),
   660→                            combined_confidence=round(conf, 2),
   661→                        ))
   662→        return chains
   663→
   664→    @staticmethod
   665→    def _assets_related(a: str, b: str) -> bool:
   666→        """Check if two assets are related (same domain, subdomain)."""
   667→        a, b = a.lower(), b.lower()
   668→        if a == b:
   669→            return True
   670→        # Same base domain
   671→        parts_a = a.replace("www.", "").split(".")
   672→        parts_b = b.replace("www.", "").split(".")
   673→        if len(parts_a) >= 2 and len(parts_b) >= 2:
   674→            return parts_a[-2:] == parts_b[-2:]
   675→        return False
   676→
   677→    @staticmethod
   678→    def _compute_combined_risk(severities: List[str]) -> str:
   679→        """Compute combined risk from multiple severities."""
   680→        if "critical" in severities:
   681→            return "critical"
   682→        if "high" in severities:
   683→            return "high"
   684→        if any(s == "medium" for s in severities) and len(severities) >= 2:
   685→            return "high"
   686→        return max(severities, key=lambda s: _SEVERITY_WEIGHTS.get(s, 1.0)) if severities else "low"
   687→
   688→    @staticmethod
   689→    def _max_severity(findings: List[Dict[str, Any]]) -> str:
   690→        sevs = [f.get("severity", "info") for f in findings]
   691→        for s in ["critical", "high", "medium", "low", "info"]:
   692→            if s in sevs:
   693→                return s
   694→        return "info"
   695→
   696→    def _merge_groups(self, asset_groups: List[CorrelationGroup],
   697→                      chain_groups: List[CorrelationGroup]) -> List[CorrelationGroup]:
   698→        """Merge asset groups and chain groups, preferring chains."""
   699→        seen_sets: set = set()
   700→        result = list(chain_groups)
   701→        chain_keys = set()
   702→        for cg in chain_groups:
   703→            for f in cg.findings:
   704→                chain_keys.add(self._finding_hash(f))
   705→
   706→        for ag in asset_groups:
   707→            # Only add asset groups that don't overlap with existing chain findings
   708→            ag_keys = {self._finding_hash(f) for f in ag.findings}
   709→            if not ag_keys & chain_keys:
   710→                result.append(ag)
   711→
   712→        return result

   719→class ExploitabilityEstimator:
   720→    """Estimates exploit difficulty based on finding characteristics."""
   721→
   722→    # Category-based exploitability baselines
   723→    _CATEGORY_EXPLOITABILITY: Dict[str, float] = {
   724→        "sqli": 0.85, "xss": 0.80, "rce": 0.90, "ssrf": 0.75,
   725→        "auth_bypass": 0.70, "privilege_escalation": 0.75,
   726→        "credential_theft": 0.65, "data_exposure": 0.70,
   727→        "misconfiguration": 0.60, "crypto_failure": 0.55,
   728→        "dos": 0.80, "default_credentials": 0.90,
   729→        "open_redirect": 0.85, "jwt_exposure": 0.75,
   730→        "sensitive_api": 0.70, "injection": 0.80,
   731→        "xxe": 0.70, "supply_chain": 0.65,
   732→    }
   733→
   734→    def estimate(self, finding: Dict[str, Any]) -> Dict[str, Any]:
   735→        """Estimate exploitability metrics for a finding.
   736→
   737→        Returns dict with exploitability, attacker_effort, likelihood, confidence.
   738→        """
   739→        severity = finding.get("severity", "info").lower()
   740→        category = finding.get("category", "").lower()
   741→        evidence = finding.get("evidence", "")
   742→        title = finding.get("title", "").lower()
   743→        description = finding.get("description", "").lower()
   744→        dread = finding.get("dread_score", 0.0)
   745→        module = finding.get("module", "").lower()
   746→
   747→        # Classify for category-based baseline
   748→        classifier = FindingClassifier()
   749→        cls = classifier.classify(finding)
   750→        cat = cls.attack_category
   751→
   752→        # Base exploitability from severity + category
   753→        sev_exploit = _sev_val(severity) / 10.0
   754→        cat_exploit = self._CATEGORY_EXPLOITABILITY.get(cat, 0.5)
   755→        base_exploitability = (sev_exploit * 0.4 + cat_exploit * 0.4 + min(dread / 10.0, 1.0) * 0.2)
   756→
   757→        # Evidence strength boost
   758→        evidence_strength = min(len(evidence) / 200.0, 1.0)
   759→        description_strength = min(len(description) / 150.0, 1.0)
   760→        evidence_boost = (evidence_strength * 0.3 + description_strength * 0.2)
   761→
   762→        exploitability = min(1.0, base_exploitability + evidence_boost * 0.15)
   763→
   764→        # Attacker effort (inverse of exploitability)
   765→        if exploitability >= 0.8:
   766→            effort = "low"
   767→        elif exploitability >= 0.6:
   768→            effort = "medium"
   769→        elif exploitability >= 0.4:
   770→            effort = "high"
   771→        else:
   772→            effort = "very_high"
   773→
   774→        # Likelihood
   775→        likelihood = min(1.0, exploitability * 0.8 + evidence_boost * 0.2)
   776→
   777→        # Confidence
   778→        confidence = min(1.0, cls.confidence * 0.6 + evidence_strength * 0.3 + 0.1)
   779→
   780→        return {
   781→            "exploitability": round(exploitability, 2),
   782→            "attacker_effort": effort,
   783→            "likelihood": round(likelihood, 2),
   784→            "confidence": round(confidence, 2),
   785→        }

   885→
   886→class AttackPathDetector:
   887→    """Detects attack paths across findings.
   888→
   889→    Chains individual findings into multi-step attack sequences
   890→    and estimates path complexity, likelihood, and impact.
   891→    """
   892→
   893→    def detect(self, findings: List[Dict[str, Any]]) -> List[AttackPath]:
   894→        """Detect attack paths from a list of findings.
   895→
   896→        Returns a list of AttackPath objects sorted by confidence.
   897→        """
   898→        if len(findings) < 2:
   899→            return []
   900→
   901→        classifier = FindingClassifier()
   902→        classified: List[Tuple[Dict[str, Any], ClassificationResult]] = []
   903→        for f in findings:
   904→            cr = classifier.classify(f)
   905→            classified.append((f, cr))
   906→
   907→        paths: List[AttackPath] = []
   908→
   909→        # Check multi-step chain templates
   910→        for chain_cats, name, risk in _MULTI_STEP_CHAINS:
   911→            matching = self._find_matching_sequence(chain_cats, classified)
   912→            if matching and len(matching) >= 2:
   913→                path = self._build_path(matching, name, risk)
   914→                if path:
   915→                    paths.append(path)
   916→
   917→        # Detect two-step pairs
   918→        for cat_a, cat_b, chain_name in _ATTACK_CHAIN_TEMPLATES:
   919→            matching = self._find_pair(cat_a, cat_b, classified)
   920→            if matching:
   921→                path = self._build_two_step_path(matching, chain_name)
   922→                if path:
   923→                    paths.append(path)
   924→
   925→        # Sort by confidence
   926→        paths.sort(key=lambda p: p.confidence, reverse=True)
   927→        return paths
   928→
   929→    def _find_matching_sequence(
   930→        self, chain_cats: List[str],
   931→        classified: List[Tuple[Dict[str, Any], ClassificationResult]],
   932→    ) -> List[Dict[str, Any]]:
   933→        """Find findings that match a chain template sequence."""
   934→        matched: List[Dict[str, Any]] = []
   935→        for cat in chain_cats:
   936→            for f, cr in classified:
   937→                if cr.attack_category == cat and f not in matched:
   938→                    matched.append(f)
   939→                    break
   940→        return matched
   941→
   942→    def _find_pair(
   943→        self, cat_a: str, cat_b: str,
   944→        classified: List[Tuple[Dict[str, Any], ClassificationResult]],
   945→    ) -> Optional[Tuple[Dict[str, Any], Dict[str, Any]]]:
   946→        """Find a pair of findings matching a two-step chain."""
   947→        a_match = None
   948→        b_match = None
   949→        for f, cr in classified:
   950→            if cr.attack_category == cat_a and a_match is None:
   951→                a_match = f
   952→            elif cr.attack_category == cat_b and b_match is None:
   953→                b_match = f
   954→            if a_match and b_match:
   955→                return (a_match, b_match)
   956→        return None
   957→
   958→    def _build_path(self, findings: List[Dict[str, Any]], name: str, risk: str) -> Optional[AttackPath]:
   959→        """Build an AttackPath from a matched finding sequence."""
   960→        if not findings:
   961→            return None
   962→
   963→        steps = [
   964→            {
   965→                "title": f.get("title", ""),
   966→                "category": f.get("category", ""),
   967→                "severity": f.get("severity", "info"),
   968→                "asset": f.get("asset", ""),
   969→            }
   970→            for f in findings
   971→        ]
   972→
   973→        # Compute aggregate metrics
   974→        severities = [f.get("severity", "info") for f in findings]
   975→        avg_exploit = sum(_sev_val(s) for s in severities) / len(severities) / 10.0
   976→        likelihood = min(1.0, avg_exploit * 0.7 + 0.2)
   977→        impact = _sev_val(risk if risk in _SEVERITY_WEIGHTS else "high") / 10.0
   978→
   979→        # Complexity based on chain length
   980→        if len(findings) >= 4:
   981→            complexity = "very_high"
   982→        elif len(findings) >= 3:
   983→            complexity = "high"
   984→        elif len(findings) >= 2:
   985→            complexity = "medium"
   986→        else:
   987→            complexity = "low"
   988→
   989→        narrative = (
   990→            f"Attack path '{name}': An attacker could progress through {len(findings)} "
   991→            f"stages — starting with '{steps[0]['title']}' against {steps[0].get('asset', 'target')}, "
   992→            f"progressing through intermediate vulnerabilities, "
   993→            f"and achieving '{steps[-1]['title']}' as the final objective. "
   994→            f"Overall risk: {risk}. Complexity: {complexity}."
   995→        )
   996→
   997→        return AttackPath(
   998→            steps=steps,
   999→            entry_point=steps[0].get("asset", ""),
  1000→            objective=name,
  1001→            complexity=complexity,
  1002→            likelihood=round(likelihood, 2),
  1003→            impact=round(impact, 2),
  1004→            confidence=round(min(1.0, likelihood * impact), 2),
  1005→            narrative=narrative,
  1006→        )
  1007→
  1008→    def _build_two_step_path(self, pair: Tuple[Dict[str, Any], Dict[str, Any]],
  1009→                             name: str) -> Optional[AttackPath]:
  1010→        """Build a two-step AttackPath."""
  1011→        return self._build_path(list(pair), name, "high")

  1018→
  1019→class RemediationPrioritizer:
  1020→    """Prioritizes remediation based on multiple risk factors."""
  1021→
  1022→    def prioritize(self, extended_findings: List[Dict[str, Any]]) -> List[Dict[str, Any]]:
  1023→        """Sort findings by remediation priority.
  1024→
  1025→        Considers: severity, exploitability, business impact, blast radius, fix complexity.
  1026→        Returns ordered list with priority_rank and reasoning.
  1027→        """
  1028→        if not extended_findings:
  1029→            return []
  1030→
  1031→        scored: List[Tuple[float, int, Dict[str, Any]]] = []
  1032→        for i, ef in enumerate(extended_findings):
  1033→            priority = self._compute_priority(ef)
  1034→            scored.append((priority, i, ef))
  1035→
  1036→        scored.sort(key=lambda x: x[0], reverse=True)
  1037→
  1038→        result: List[Dict[str, Any]] = []
  1039→        for rank, (score, orig_idx, ef) in enumerate(scored, 1):
  1040→            entry = dict(ef)
  1041→            entry["priority_rank"] = rank
  1042→            entry["priority_score"] = round(score, 2)
  1043→            entry["priority_reasoning"] = self._reasoning(ef, score, rank)
  1044→            result.append(entry)
  1045→
  1046→        return result
  1047→
  1048→    def _compute_priority(self, ef: Dict[str, Any]) -> float:
  1049→        """Compute a priority score (higher = fix first)."""
  1050→        severity = ef.get("severity", "info").lower()
  1051→        exploitability = ef.get("exploitability", 0.5)
  1052→        impact = ef.get("business_impact_score", 0.5)
  1053→        confidence = ef.get("confidence_score", 0.5)
  1054→
  1055→        sev_weight = _sev_val(severity) / 10.0
  1056→        return (sev_weight * 0.35 + exploitability * 0.25 +
  1057→                impact * 0.25 + confidence * 0.15)
  1058→
  1059→    def _reasoning(self, ef: Dict[str, Any], score: float, rank: int) -> str:
  1060→        """Generate a human-readable reasoning for the priority."""
  1061→        severity = ef.get("severity", "info").upper()
  1062→        exploit = ef.get("exploitability", 0.5)
  1063→        impact = ef.get("business_impact_score", 0.5)
  1064→        title = ef.get("title", "Unknown")
  1065→
  1066→        parts = [f"#{rank} priority"]
  1067→        if severity == "CRITICAL":
  1068→            parts.append("critical severity")
  1069→        elif severity == "HIGH":
  1070→            parts.append("high severity")
  1071→        if exploit >= 0.7:
  1072→            parts.append("easily exploitable")
  1073→        if impact >= 0.7:
  1074→            parts.append("high business impact")
  1075→        parts.append(f"(score: {score:.2f})")
  1076→
  1077→        return f"{'; '.join(parts)} — {title}"

  1085→def resolve_mappings(attack_category: str) -> Tuple[Dict[str, str], List[str], List[str]]:
  1086→    """Resolve MITRE ATT&CK, CWE, and CAPEC mappings for a category.
  1087→
  1088→    Returns (mitre_primary, cwe_list, capec_list).

  1090→    mitre_entries = MITRE_ATTACK_MAP.get(attack_category, [])

  1092→    if mitre_entries:
  1093→        m = mitre_entries[0]
  1094→        mitre_primary = {"tactic": m["tactic"], "technique": m["technique"], "name": m["name"]}

  1096→    cwe_list = CWE_MAP.get(attack_category, [])
  1097→    capec_list = CAPEC_MAP.get(attack_category, [])
  1098→
  1099→    return mitre_primary, cwe_list, capec_list

  1102→def generate_cvss(severity: str, exploitability: float = 0.5, impact: float = 0.5) -> Dict[str, Any]:
  1103→    """Generate a CVSS v3 interpretation from severity and metrics.
  1104→
  1105→    Returns dict with base_score, severity, vector_string.
  1106→    """
  1107→    base = CVSS_BASE_SCORES.get(severity.lower(), 5.0)
  1108→
  1109→    # Adjust based on exploitability and impact
  1110→    adj = base + (exploitability - 0.5) * 1.5 + (impact - 0.5) * 2.0
  1111→    base_score = round(max(0.0, min(10.0, adj)), 1)
  1112→
  1113→    if base_score >= 9.0:
  1114→        sev = "CRITICAL"
  1115→    elif base_score >= 7.0:
  1116→        sev = "HIGH"
  1117→    elif base_score >= 4.0:
  1118→        sev = "MEDIUM"
  1119→    elif base_score > 0.0:
  1120→        sev = "LOW"
  1121→    else:
  1122→        sev = "NONE"
  1123→
  1124→    # AV:N/AC:L/PR:N/UI:N/S:U/C:H/I:H/A:H (typical remote vuln)
  1125→    if exploitability >= 0.7:
  1126→        av, ac, pr, ui = "N", "L", "N", "N"
  1127→    elif exploitability >= 0.4:
  1128→        av, ac, pr, ui = "N", "L", "L", "N"
  1129→    else:
  1130→        av, ac, pr, ui = "N", "H", "L", "R"
  1131→
  1132→    c = "H" if impact >= 0.7 else ("L" if impact >= 0.3 else "N")
  1133→    i = "H" if impact >= 0.7 else ("L" if impact >= 0.4 else "N")
  1134→    a = "H" if impact >= 0.8 else ("L" if impact >= 0.3 else "N")
  1135→
  1136→    vector = f"CVSS:3.1/AV:{av}/AC:{ac}/PR:{pr}/UI:{ui}/S:U/C:{c}/I:{i}/A:{a}"
  1137→
  1138→    return {
  1139→        "base_score": base_score,
  1140→        "severity": sev,
  1141→        "vector_string": vector,
  1142→    }
