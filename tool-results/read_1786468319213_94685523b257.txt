   203→            "fix_proposals_count": self.fix_proposals_count,

   208→    def save(self, directory: Optional[Path] = None) -> Path:
   209→        """Persist cycle results as JSON.
   210→
   211→        Args:
   212→            directory: Override directory. Defaults to ENGINEERING_CYCLES_DIR.
   213→
   214→        Returns:
   215→            Path to the saved JSON file.

   217→        save_dir = _ensure_dir(directory or ENGINEERING_CYCLES_DIR)
   218→        filename = f"cycle_{self.cycle_id}_{datetime.now(timezone.utc).strftime('%Y%m%d%H%M%S')}.json"
   219→        filepath = save_dir / filename
   220→        try:
   221→            with open(filepath, "w", encoding="utf-8") as f:
   222→                json.dump(self.to_dict(), f, indent=2, ensure_ascii=False, default=str)
   223→            logger.info("Cycle results saved to %s", filepath)
   224→        except OSError:
   225→            logger.warning("Failed to save cycle results to %s", filepath, exc_info=True)
   226→        return filepath

   228→    @classmethod
   229→    def load_latest(cls, directory: Optional[Path] = None) -> Optional["EngineeringCycleResult"]:
   230→        """Load the most recent cycle result from JSON files.
   231→
   232→        Args:
   233→            directory: Override directory.
   234→
   235→        Returns:
   236→            EngineeringCycleResult or None if no files found.
   237→        """
   238→        search_dir = directory or ENGINEERING_CYCLES_DIR
   239→        if not search_dir.exists():
   240→            return None
   241→        try:
   242→            files = sorted(search_dir.glob("cycle_*.json"), key=os.path.getmtime)
   243→            if not files:
   244→                return None
   245→            with open(files[-1], "r", encoding="utf-8") as f:
   246→                data = json.load(f)
   247→            result = cls()
   248→            result.cycle_id = data.get("cycle_id", "")
   249→            result.repository_path = data.get("repository_path", "")
   250→            result.mode = data.get("mode", "full")
   251→            result.overall_status = data.get("overall_status", "pass")
   252→            result.started_at = data.get("started_at", "")
   253→            result.finished_at = data.get("finished_at", "")
   254→            result.total_duration_s = data.get("total_duration_s", 0.0)
   255→            result.metrics = data.get("metrics", {})
   256→            result.recommendations_count = data.get("recommendations_count", 0)
   257→            result.fix_proposals_count = data.get("fix_proposals_count", 0)
   258→            result.anomaly_count = data.get("anomaly_count", 0)
   259→            result.regression_count = data.get("regression_count", 0)
   260→            for sdata in data.get("stages", []):
   261→                sr = StageResult(
   262→                    stage=EngineeringStage(sdata.get("stage", "unknown")),
   263→                    status=sdata.get("status", "pass"),
   264→                    duration_s=sdata.get("duration_s", 0.0),
   265→                    data=sdata.get("data", {}),
   266→                    error=sdata.get("error", ""),
   267→                )
   268→                result.stages.append(sr)
   269→            return result
   270→        except (OSError, json.JSONDecodeError, KeyError, ValueError):
   271→            logger.warning("Failed to load latest cycle result", exc_info=True)
   272→            return None

   403→    # ── Stage execution helpers ────────────────────────────────────────
   404→
   405→    def _execute_stage(
   406→        self,
   407→        stage: EngineeringStage,
   408→        stage_fn: Any,
   409→        result: EngineeringCycleResult,
   410→    ) -> StageResult:
   411→        """Execute a single stage with timing and error handling.
   412→
   413→        Args:
   414→            stage: The stage enum value.
   415→            stage_fn: Callable that performs the stage work.
   416→            result: The cycle result to append this stage result to.
   417→
   418→        Returns:
   419→            StageResult with outcome data.
   420→        """
   421→        stage_result = StageResult(stage=stage)
   422→        t0 = time.monotonic()
   423→        try:
   424→            data = stage_fn()
   425→            if isinstance(data, dict):
   426→                stage_result.data = data
   427→            elif data is not None:
   428→                stage_result.data = {"result": data}
   429→            stage_result.status = "pass"
   430→            logger.info("Stage %s completed successfully", stage.value)
   431→        except Exception as exc:
   432→            stage_result.status = "fail"
   433→            stage_result.error = f"{type(exc).__name__}: {exc}"
   434→            logger.warning(
   435→                "Stage %s failed (continuing): %s", stage.value, exc, exc_info=True
   436→            )
   437→        finally:
   438→            stage_result.duration_s = time.monotonic() - t0
   439→        result.stages.append(stage_result)
   440→        return stage_result
   441→
   442→    def _compute_overall_status(self, result: EngineeringCycleResult) -> str:
   443→        """Compute overall pass/fail/warn from stage statuses.
   444→
   445→        Rules:
   446→            - Any 'fail' → overall 'fail'
   447→            - Any 'warn' (and no 'fail') → overall 'warn'
   448→            - All 'pass' or 'skipped' → overall 'pass'
   449→        """
   450→        has_fail = any(s.status == "fail" for s in result.stages)
   451→        has_warn = any(s.status == "warn" for s in result.stages)
   452→        if has_fail:
   453→            return "fail"
   454→        if has_warn:
   455→            return "warn"
   456→        return "pass"

   461→        """a. Digital Twin: capture current state."""
   462→        twin = self._get_digital_twin()
   463→        state = twin.capture_state()

   470→    def _stage_digital_twin_anomaly(self) -> Dict[str, Any]:
   471→        """a. Digital Twin: detect anomalies."""
   472→        twin = self._get_digital_twin()
   473→        anomalies = twin.detect_anomaly()
   474→        anomaly_list = anomalies.get("anomalies", [])
   475→        return {
   476→            "anomaly_count": anomalies.get("anomaly_count", len(anomaly_list)),
   477→            "anomalies": anomaly_list,

   480→    def _stage_repository_memory_recall(self) -> Dict[str, Any]:
   481→        """b. Repository Memory: recall previous engineering knowledge."""
   482→        mem = self._get_repository_memory()
   483→        # Search for all engineering-cycle-related facts
   484→        facts = mem.search(query_text="engineering")
   485→        if not facts:
   486→            facts = mem.search(query_text="pipeline")
   487→        return {
   488→            "recall_count": len(facts),
   489→            "facts": [
   490→                {"key": f.key, "confidence": f.confidence, "source": f.source}
   491→                for f in facts[:50]  # Cap at 50 to avoid excessive data
   492→            ],
   493→        }

   500→        if hasattr(report, "to_dict"):

   502→        elif hasattr(report, "summary"):
   503→            data = {"summary": report.summary()}
   504→        else:

   526→    def _stage_regression_intelligence(self) -> Dict[str, Any]:
   527→        """g. Regression Intelligence: detect regressions."""
   528→        ri = self._get_regression_intelligence()
   529→        # Use a synthetic "current state" for comparison
   530→        current_results: Dict[str, Any] = {
   531→            "timestamp": datetime.now(timezone.utc).isoformat(),
   532→            "repository_path": self._repo_path,
   533→        }
   534→        regressions = ri.detect_regression(current_results)
   535→        reg_list = regressions if isinstance(regressions, list) else []
   536→        return {
   537→            "regression_count": len(reg_list),
   538→            "regressions": [

   668→    def _stage_quality_intelligence(self) -> Dict[str, Any]:
   669→        """k2. Quality Intelligence: multi-dimensional code quality analysis."""
   670→        try:
   671→            from .quality_intelligence import QualityIntelligence
   672→
   673→            qi = QualityIntelligence()
   674→            repo_path = self._repo_path or "."
   675→            # Analyze key package files
   676→            target_files = []
   677→            pkg = Path(repo_path)
   678→            if pkg.is_dir():
   679→                for py_file in sorted(pkg.glob("*.py"))[:20]:
   680→                    target_files.append(str(py_file))
   681→                # Also analyze subdirectories
   682→                for subdir in ["reconpro", "modules", "integrations"]:
   683→                    sub_path = pkg / subdir
   684→                    if sub_path.is_dir():
   685→                        for py_file in sorted(sub_path.glob("*.py"))[:10]:
   686→                            target_files.append(str(py_file))
   687→
   688→            if not target_files:
   689→                return {"analyzed": 0, "reason": "no Python files found"}
   690→
   691→            results = {}
   692→            for fpath in target_files:
   693→                try:
   694→                    snapshot = qi.analyze_file(fpath)
   695→                    results[fpath] = snapshot.to_dict() if hasattr(snapshot, "to_dict") else str(snapshot)
   696→                except Exception:
   697→                    continue
   698→
   699→            overall = qi.analyze_repository(repo_path) if pkg.is_dir() else None
   700→            data = {
   701→                "files_analyzed": len(results),
   702→                "results": results,
   703→            }
   704→            if overall and hasattr(overall, "to_dict"):
   705→                data["overall"] = overall.to_dict()
   706→            return data
   707→        except Exception as exc:
   708→            return {"analyzed": 0, "error": str(exc)}

   721→    def _stage_repository_memory_persist(self) -> Dict[str, Any]:
   722→        """m. Repository Memory: persist results for next cycle."""
   723→        mem = self._get_repository_memory()
   724→        if not self._last_result:
   725→            return {"persisted": False, "reason": "no result to persist"}
   726→
   727→        facts_stored = 0
   728→
   729→        # Persist cycle summary
   730→        try:
   731→            mem.remember(
   732→                key=f"engineering_cycle/{self._last_result.cycle_id}/summary",
   733→                value={
   734→                    "overall_status": self._last_result.overall_status,

   739→                    "anomaly_count": self._last_result.anomaly_count,
   740→                    "regression_count": self._last_result.regression_count,
   741→                },
   742→                metadata={"source": "engineering_workflow", "tags": ["cycle", "summary"]},
   743→            )
   744→            facts_stored += 1
   745→        except Exception:
   746→            logger.debug("Failed to persist cycle summary", exc_info=True)
   747→
   748→        # Persist per-stage results
   749→        for sr in self._last_result.stages:
   750→            try:
   751→                mem.remember(

   758→                    metadata={"source": "engineering_workflow", "tags": ["cycle", "stage"]},

   761→            except Exception:
   762→                logger.debug(
   763→                    "Failed to persist stage %s", sr.stage.value, exc_info=True
   764→                )
   765→
   766→        return {"persisted": True, "facts_stored": facts_stored}

   790→    def run_full_cycle(
   791→        self,
   792→        repository_path: Optional[str] = None,
   793→    ) -> EngineeringCycleResult:
   794→        """Execute the complete engineering cycle (all 14 stages, a–m).
   795→
   796→        Steps:
   797→            a. Digital Twin: capture state, detect anomalies
   798→            b. Repository Memory: recall previous knowledge
   799→            c. Auto Validation: run full validation pipeline
   800→            d. Diagnostics: health checks
   801→            e. Drift Detection: detect code drift
   802→            f. Benchmark: quick benchmarks, performance regressions
   803→            g. Regression Intelligence: detect regressions
   804→            h. Learning: learn from collected data
   805→            i. Recommendations: generate from findings
   806→            j. Auto Fix: propose fixes
   807→            k. Quality Gate: enforce thresholds
   808→            l. Reporting: comprehensive report
   809→            m. Repository Memory: persist results
   810→
   811→        Args:
   812→            repository_path: Override repository path for this cycle.
   813→
   814→        Returns:
   815→            EngineeringCycleResult with all stage results.
   816→        """
   817→        if repository_path:
   818→            self._repo_path = str(repository_path)
   819→            self._repo_resolved = Path(repository_path).resolve()
   820→
   821→        result = EngineeringCycleResult(
   822→            cycle_id=datetime.now(timezone.utc).strftime("%Y%m%d%H%M%S%f"),
   823→            repository_path=self._repo_path,
   824→            mode="full",
   825→            started_at=datetime.now(timezone.utc).isoformat(),
   826→        )
   827→
   828→        self._last_result = result
   829→        cycle_start = time.monotonic()
   830→        logger.info("Starting full engineering cycle %s for %s", result.cycle_id, self._repo_path)
   831→
   832→        for stage in _FULL_CYCLE_STAGES:
   833→            method_name = self._STAGE_DISPATCH.get(stage)
   834→            if method_name is None:
   835→                logger.warning("No dispatch for stage %s — skipping", stage.value)
   836→                result.stages.append(StageResult(stage=stage, status="skipped"))
   837→                continue
   838→            stage_fn = getattr(self, method_name, None)
   839→            if stage_fn is None:
   840→                result.stages.append(StageResult(stage=stage, status="skipped"))
   841→                continue
   842→            sr = self._execute_stage(stage, stage_fn, result)
   843→            # Track aggregate counts
   844→            if sr.status == "pass" and sr.data:
   845→                if sr.stage == EngineeringStage.DIGITAL_TWIN_ANOMALY:
   846→                    result.anomaly_count = sr.data.get("anomaly_count", 0)
   847→                if sr.stage == EngineeringStage.REGRESSION_INTELLIGENCE:
   848→                    result.regression_count = sr.data.get("regression_count", 0)
   849→                if sr.stage == EngineeringStage.RECOMMENDATIONS:
   850→                    result.recommendations_count = sr.data.get("recommendations_count", 0)
   851→                if sr.stage == EngineeringStage.AUTO_FIX:
   852→                    result.fix_proposals_count = sr.data.get("fix_proposals_count", 0)
   853→
   854→        result.finished_at = datetime.now(timezone.utc).isoformat()
   855→        result.total_duration_s = time.monotonic() - cycle_start
   856→        result.overall_status = self._compute_overall_status(result)
   857→        result.metrics = {
   858→            "total_stages": len(result.stages),
   859→            "passed": sum(1 for s in result.stages if s.status == "pass"),
   860→            "failed": sum(1 for s in result.stages if s.status == "fail"),
   861→            "warned": sum(1 for s in result.stages if s.status == "warn"),
   862→            "skipped": sum(1 for s in result.stages if s.status == "skipped"),
   863→        }
   864→
   865→        # Persist results
   866→        result.save()
   867→
   868→        self._cycle_count += 1
   869→        logger.info(
   870→            "Engineering cycle %s complete: status=%s duration=%.2fs",
   871→            result.cycle_id,
   872→            result.overall_status,
   873→            result.total_duration_s,
   874→        )
   875→        return result
