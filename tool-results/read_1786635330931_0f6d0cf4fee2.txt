   244→// ── Color Picker ─────────────────────────────────────────────────────────────
   245→
   246→function ColorPickerField({
   247→  label,
   248→  value,
   249→  onChange,
   250→  presets,
   251→}: {
   252→  label: string;
   253→  value: string;
   254→  onChange: (v: string) => void;
   255→  presets: string[];
   256→}) {
   257→  const [hexInput, setHexInput] = useState(value);
   258→  const handleHexSubmit = () => {
   259→    if (/^#[0-9a-fA-F]{6}$/.test(hexInput)) {
   260→      onChange(hexInput);
   261→    }
   262→  };
   263→  return (
   264→    <div className="space-y-2.5">
   265→      <Label className="text-[13px] font-medium text-[#444444]">{label}</Label>
   266→      <div className="flex items-center gap-3">
   267→        <div className="relative">
   268→          <div
   269→            className="h-10 w-10 rounded-lg border border-white/10 shadow-inner cursor-pointer"
   270→            style={{ backgroundColor: value }}
   271→          />
   272→        </div>
   273→        <div className="flex-1 flex items-center gap-2">
   274→          <Input
   275→            value={hexInput}
   276→            onChange={(e) => setHexInput(e.target.value)}
   277→            onBlur={() => setHexInput(value)}
   278→            onKeyDown={(e) => e.key === 'Enter' && handleHexSubmit()}
   279→            className="h-10 bg-[#000000] border-white/[0.08] text-[#f0f0f0] font-mono text-sm placeholder:text-[#333333]"
   280→            placeholder="#00ff88"
   281→          />
   282→          <Button
   283→            type="button"
   284→            variant="outline"
   285→            size="sm"
   286→            className="h-10 border-white/[0.08] text-[#444444] hover:text-[#f0f0f0] hover:bg-white/[0.04] shrink-0"
   287→            onClick={handleHexSubmit}
   288→          >
   289→            Apply
   290→          </Button>
   291→        </div>
   292→      </div>
   293→      <div className="flex items-center gap-2 flex-wrap">
   294→        {presets.map((p) => (
   295→          <ColorSwatch
   296→            key={p}
   297→            color={p}
   298→            selected={p.toLowerCase() === value.toLowerCase()}
   299→            onClick={() => {
   300→              onChange(p);
   301→              setHexInput(p);
   302→            }}
   303→          />
   304→        ))}
   305→      </div>
   306→    </div>
   307→  );
   308→}

   310→// ── DNS Record Row ──────────────────────────────────────────────────────────
   311→
   312→function DnsRecordRow({ record }: { record: DnsRecord }) {
   313→  const [copied, setCopied] = useState(false);
   314→  const handleCopy = () => {
   315→    navigator.clipboard.writeText(record.value);
   316→    setCopied(true);
   317→    setTimeout(() => setCopied(false), 2000);
   318→  };
   319→  return (
   320→    <div className="flex items-center gap-3 rounded-xl bg-[#030407]/60 border border-white/[0.04] p-3">
   321→      <Badge
   322→        variant="outline"
   323→        className="shrink-0 font-mono text-xs px-2 py-0.5 border-[#00ff88]/30 text-[#00ff88] bg-[#00ff88]/5"
   324→      >
   325→        {record.type}
   326→      </Badge>
   327→      <div className="flex-1 min-w-0">
   328→        <p className="text-[13px] font-medium text-[#f0f0f0] truncate">
   329→          {record.name}
   330→        </p>
   331→        <p className="text-xs text-[#444444] font-mono truncate">
   332→          {record.value}
   333→        </p>
   334→      </div>
   335→      <div className="flex items-center gap-2 shrink-0">
   336→        {record.status === 'verified' ? (
   337→          <Badge className="bg-[#00ff88]/10 text-[#00ff88] border-[#00ff88]/20 text-xs px-2 py-0.5">
   338→            <CheckCircle2 className="h-3 w-3 mr-1" />
   339→            Verified
   340→          </Badge>
   341→        ) : (
   342→          <Badge className="bg-yellow-500/10 text-yellow-400 border-yellow-500/20 text-xs px-2 py-0.5">
   343→            <Clock className="h-3 w-3 mr-1" />
   344→            Pending
   345→          </Badge>
   346→        )}
   347→        <Button
   348→          variant="ghost"
   349→          size="sm"
   350→          className="h-8 w-8 p-0 text-[#444444] hover:text-[#f0f0f0] hover:bg-white/[0.06]"
   351→          onClick={handleCopy}
   352→        >
   353→          {copied ? (
   354→            <Check className="h-3.5 w-3.5 text-[#00ff88]" />
   355→          ) : (
   356→            <Copy className="h-3.5 w-3.5" />
   357→          )}
   358→        </Button>
   359→      </div>
   360→    </div>
   361→  );
   362→}
