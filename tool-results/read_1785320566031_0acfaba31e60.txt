    85→class _RateLimiter:
    86→    """Token-bucket rate limiter — safe for concurrent use.
    87→
    88→    Default: 10 requests per second. Use throttle.acquire() before any
    89→    network call; it blocks until a token is available.
    90→    """
    91→
    92→    def __init__(self, max_per_second: float = 10.0):
    93→        self._rate = max_per_second
    94→        self._min_interval = 1.0 / max_per_second
    95→        self._lock = Lock()
    96→        self._last = 0.0
    97→
    98→    def acquire(self) -> None:
    99→        """Block until we're allowed to make the next request."""
   100→        with self._lock:
   101→            now = time.monotonic()
   102→            elapsed = now - self._last
   103→            if elapsed < self._min_interval:
   104→                time.sleep(self._min_interval - elapsed)
   105→            self._last = time.monotonic()
   106→

   121→def classify_ipv6(addr: str) -> str:
   122→    """Classify an IPv6 address. Returns: 'global', 'link_local', 'unique_local',
   123→    'loopback', 'mapped_v4', 'documentation', 'multicast', 'unspecified'."""
   124→    if addr == "::":
   125→        return "unspecified"
   126→    if _IPV6_LOOPBACK_RE.match(addr):
   127→        return "loopback"
   128→    if _IPV6_LINK_LOCAL_RE.match(addr):
   129→        return "link_local"
   130→    if _IPV6_UNIQUE_LOCAL_RE.match(addr):
   131→        return "unique_local"
   132→    if _IPV6_MAPPED_V4_RE.match(addr):
   133→        return "mapped_v4"
   134→    if _IPV6_DOC_RE.match(addr):
   135→        return "documentation"
   136→    if _IPV6_MULTICAST_RE.match(addr):
   137→        return "multicast"
   138→    return "global"

   175→def run_argv(argv: list, timeout: int = 8) -> Tuple[str, str]:
   176→    """Execute a pre-parsed argv list without any shell interpretation."""
   177→    try:
   178→        r = subprocess.run(argv, shell=False, capture_output=True, text=True, timeout=timeout)
   179→        return r.stdout.strip(), r.stderr.strip()
   180→    except Exception:
   181→        return "", ""

   198→def http_probe(url: str, method: str = "GET", body: Optional[bytes] = None,
   199→               headers: Optional[Dict[str, str]] = None, timeout: int = 8) -> Dict[str, Any]:
   200→    """Hardened single HTTP layer — all modules delegate through this."""
   201→    RATE_LIMITER.acquire()  # m5: thread-safe rate limiting
   202→    h = {
   203→        "User-Agent": "ReconPro-Unified/1.0 (Six-Blades-One-Target; +https://reconpro.security)",
   204→        "X-ReconPro-Signature": RECONPRO_SIGNATURE,
   205→        "Accept": "application/json,text/plain,*/*",
   206→    }
   207→    if headers:
   208→        h.update(headers)
   209→    req = urllib.request.Request(url, data=body, method=method, headers=h)
   210→    try:
   211→        ctx = ssl.create_default_context()
   212→        if CONFIG.insecure:
   213→            ctx.check_hostname = False
   214→            ctx.verify_mode = ssl.CERT_NONE
   215→        with urllib.request.urlopen(req, timeout=timeout, context=ctx) as resp:
   216→            raw = resp.read(8192)
   217→            return {
   218→                "ok": True, "status": resp.status, "reason": resp.reason,
   219→                "headers": dict(resp.headers.items()),
   220→                "body": raw.decode("utf-8", errors="replace")[:8192],
   221→            }
   222→    except urllib.error.HTTPError as e:
   223→        try:
   224→            raw = e.read(8192)
   225→            body_str = raw.decode("utf-8", errors="replace")[:8192]
   226→        except Exception:
   227→            body_str = ""
   228→        return {
   229→            "ok": False, "status": e.code, "reason": e.reason,
   230→            "headers": dict(e.headers.items()) if e.headers else {},
   231→            "body": body_str,
   232→        }
   233→    except Exception as e:
   234→        return {"ok": False, "status": 0, "reason": str(e), "headers": {}, "body": ""}

   437→def _sign_report(report: Dict[str, Any]) -> str:
   438→    """HMAC-SHA256 sign a report for tamper-proof telemetry."""
   439→    creds = _load_credentials()
   440→    api_key = creds.get("api_key", "")
   441→    canonical = json.dumps(report, sort_keys=True, separators=(",", ":"), ensure_ascii=False)
   442→    signature = hmac.new(api_key.encode(), canonical.encode(), hashlib.sha256).hexdigest()
   443→    return signature

   975→# ── Filename safety (m4) ────────────────────────────────────────────────────
   976→_FILENAME_UNSAFE_RE = re.compile(r'[\x00-\x1f\x7f\x80-\x9f/\\:\n\r\t]')
   977→_FILENAME_MAX_LEN = 120
   978→
   979→def _safe_filename(raw: str) -> str:
   980→    """Sanitize a user-supplied string for use as a filename component.
   981→
   982→    Strips: null bytes, control chars, path separators, backslashes.
   983→    Truncates to _FILENAME_MAX_LEN. Rejects empty results (returns 'unnamed').
   984→    """
   985→    cleaned = _FILENAME_UNSAFE_RE.sub('_', raw)
   986→    # Collapse consecutive underscores
   987→    cleaned = re.sub(r'_{2,}', '_', cleaned)
   988→    cleaned = cleaned.strip('._-')
   989→    cleaned = cleaned[:_FILENAME_MAX_LEN]
   990→    if not cleaned or cleaned in ('.', '..'):
   991→        return 'unnamed'
   992→    return cleaned

   994→def _find_cache(prefix: str, host: str) -> Optional[str]:
   995→    """Find a cached JSON for this host with the given prefix."""
   996→    if not os.path.isdir(CACHE_DIR):
   997→        return None
   998→    safe = _safe_filename(host)
   999→    candidates = [
  1000→        f"{CACHE_DIR}/{prefix}_{safe}.json",
  1001→        f"{CACHE_DIR}/{prefix}_{safe}_v2.json",
  1002→    ]
  1003→    # also any *{host}*.json
  1004→    for f in sorted(os.listdir(CACHE_DIR), reverse=True):
  1005→        if f.startswith(prefix) and safe in f and f.endswith(".json"):
  1006→            candidates.append(f"{CACHE_DIR}/{f}")
  1007→    for c in candidates:
  1008→        if os.path.exists(c) and os.path.getsize(c) > 100:
  1009→            return c
  1010→    return None

  1097→def module_vibesec(host: str) -> Dict[str, Any]:

  1099→
  1100→    Checks four categories:
  1101→      1. Exposed env/config endpoints
  1102→      2. Unauthenticated API/webhook routes
  1103→      3. Permissive CORS policies
  1104→      4. Exposed anon/public backend keys (Supabase, Firebase, S3, R2)
  1105→
  1106→    Returns a 100-point VibeSec score with A+ to F grade mapping.
  1107→    Every finding is verified via native HTTP response validation (zero fabrication).
  1108→    """
  1109→    findings: List[Dict[str, Any]] = []
  1110→    deductions = 0  # each vulnerability deducts from 100
  1111→
  1112→    base_url = host if host.startswith("http") else f"https://{host}"
  1113→
  1114→    def add(title, severity, category, description, evidence, points_deducted):
  1115→        nonlocal deductions
  1116→        deductions += points_deducted
  1117→        findings.append({
  1118→            "title": title, "severity": severity, "category": category,
  1119→            "description": description, "evidence": evidence, "asset": host,
  1120→            "points_deducted": points_deducted,
  1121→        })
  1122→
  1123→    # ── Category 1: Exposed Environment/Config Files ──────────────────────
  1124→    audit_log("vibesec.config.start", detail=host)
  1125→    for path in VIBESEC_SENSITIVE_PATHS:
  1126→        url = base_url.rstrip("/") + path
  1127→        resp = http_probe(url, timeout=5)
  1128→        status = resp.get("status", 0)
  1129→        body = resp.get("body", "")[:2048]
  1130→        if status == 200 and len(body) > 10:
  1131→            # Verify: real content, not just a redirect page
  1132→            has_secrets = any(kw in body.lower() for kw in [
  1133→                "api_key", "secret", "password", "token", "database_url",
  1134→                "private_key", "supabase", "firebase", "aws_access",
  1135→            ])
  1136→            if has_secrets:
  1137→                add(f"Exposed config — {path}", "critical", "exposed_config",
  1138→                    f"Sensitive configuration endpoint accessible: {path} (200 OK, {len(body)} bytes, contains secrets)",
  1139→                    f"GET {path} → {status} ({len(body)}B, secrets detected)", 15)
  1140→            else:
  1141→                add(f"Exposed config — {path}", "high", "exposed_config",
  1142→                    f"Configuration endpoint accessible: {path} (200 OK, {len(body)} bytes)",
  1143→                    f"GET {path} → {status} ({len(body)}B)", 10)
  1144→        elif status in (200, 201, 301, 302, 307, 308) and status != 404:
  1145→            add(f"Config path accessible — {path}", "medium", "exposed_config",
  1146→                f"Endpoint returned {status} (may redirect or serve partial content)",
  1147→                f"GET {path} → {status}", 5)
  1148→    audit_log("vibesec.config.done", detail=f"{len([f for f in findings if f['category'] == 'exposed_config'])} findings")
  1149→

  1152→    for path in VIBESEC_API_PATHS:
  1153→        url = base_url.rstrip("/") + path
  1154→        resp = http_probe(url, timeout=5)
  1155→        status = resp.get("status", 0)
  1156→        body = resp.get("body", "")[:2048]

  1161→                "unauthorized", "401", "forbidden", "authentication required",

  1424→def compute_unified_verdict(report: Dict[str, Any]) -> Dict[str, Any]:
  1425→    """Combine all module scores into a unified verdict."""
  1426→    scores = {}
  1427→    # RECON: weighted by severity counts
  1428→    recon = report.get("recon", {})
  1429→    sc = recon.get("severity_counts", {})
  1430→    scores["recon"] = min(100, sc.get("critical", 0) * 25 + sc.get("high", 0) * 15 +
  1431→                          sc.get("medium", 0) * 8 + sc.get("low", 0) * 3 + sc.get("info", 0))
  1432→    # AUTH: bypasses / 75 * 100
  1433→    auth = report.get("auth_bypass", {})
  1434→    scores["auth"] = min(100, int((auth.get("bypasses_successful", 0) / max(1, auth.get("total_attempts", 1))) * 400))
  1435→    # CHAIN: SSRF + open redirects
  1436→    chain = report.get("chain_hunter", {})
  1437→    scores["chain"] = min(100, chain.get("ssrf_detected", 0) * 25 + chain.get("open_redirects", 0) * 15)
  1438→    # BOT: bot hits
  1439→    bot = report.get("bot_hunter", {})
  1440→    scores["bot"] = min(100, bot.get("bot_hits", 0) * 20)
  1441→    # GORGON
  1442→    gorgon = report.get("gorgon", {})
  1443→    scores["gorgon"] = gorgon.get("threatScore", gorgon.get("verdict", {}).get("threatScore", 0)) if isinstance(gorgon, dict) else 0
  1444→    # OBLIVION
  1445→    oblivion = report.get("oblivion", {})
  1446→    scores["oblivion"] = oblivion.get("verdict", {}).get("threatScore", 0) if isinstance(oblivion, dict) else 0
  1447→    # VIBESEC: inverted score (low vibesec = more dangerous for unified)
  1448→    vibesec = report.get("vibesec", {})
  1449→    vibesec_score = vibesec.get("vibesec_score", 100) if isinstance(vibesec, dict) else 100
  1450→    scores["vibesec"] = 100 - vibesec_score  # invert: more vibe-vulns = higher unified threat
  1451→
  1452→    # Weighted unified score
  1453→    weights = {"recon": 0.08, "auth": 0.13, "chain": 0.13, "bot": 0.08, "gorgon": 0.18, "oblivion": 0.28, "vibesec": 0.12}
  1454→    unified = int(sum(scores.get(k, 0) * w for k, w in weights.items()))
  1455→
  1456→    if unified >= 90: verdict = "OMNIPOTENT VERDICT — The target has been comprehensively dissolved."
  1457→    elif unified >= 75: verdict = "DEVASTATING VERDICT — The target's defenses are thoroughly compromised."
  1458→    elif unified >= 50: verdict = "SUBSTANTIAL VERDICT — Multiple critical exposures confirmed."
  1459→    elif unified >= 25: verdict = "NOTABLE VERDICT — Some exposures detected."
  1460→    else: verdict = "MUNDANE VERDICT — The target resisted most probes."
  1461→
  1462→    return {
  1463→        "unified_score": unified,
  1464→        "module_scores": scores,
  1465→        "verdict_text": verdict,
  1466→        "verdict_level": "OMNIPOTENT" if unified >= 90 else "DEVASTATING" if unified >= 75 else
  1467→                          "SUBSTANTIAL" if unified >= 50 else "NOTABLE" if unified >= 25 else "MUNDANE",
  1468→    }
