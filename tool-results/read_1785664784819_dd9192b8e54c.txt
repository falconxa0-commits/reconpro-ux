   588→def _match_category(finding_title: str, finding_category: str) -> Optional[str]:
   589→    """Try to map a finding to a remediation rule key."""
   590→    combined = f"{finding_category} {finding_title}".lower()
   591→    best_key: Optional[str] = None
   592→    best_len = 0
   593→    for alias, rule_key in _CATEGORY_ALIAS.items():
   594→        if alias.lower() in combined:
   595→            if len(alias) > best_len:
   596→                best_len = len(alias)
   597→                best_key = rule_key
   598→    return best_key

   683→                for r in self.rounds
   684→            ],
   685→            "remediation_report": self.remediation_report,
   686→        }
   687→
   688→
   689→# ── Adversarial Loop ──────────────────────────────────────────────────────
   690→
   691→class AdversarialLoop:
   692→    """Runs HACKER vs CODER vs GUARDIAN in an iterative remediation loop."""
   693→
   694→    def __init__(
   695→        self,
   696→        target: str,
   697→        max_rounds: int = 3,
   698→        modules: Optional[List[str]] = None,
   699→        is_local: bool = False,
   700→    ) -> None:
   701→        self.target = target
   702→        self.max_rounds = max_rounds
   703→        self.modules = modules
   704→        self.is_local = is_local
   705→        self._all_actions: List[RemediationAction] = []
   706→        self._fixed_titles: set = set()
   707→        self._failed_titles: set = set()
   708→        self._feedback_context: Dict[str, List[str]] = {}
   709→        self._converged = False
   710→
   711→    # ── HACKER phase ──────────────────────────────────────────────────
   712→
   713→    def _run_hacker(self) -> Tuple[List[Dict[str, Any]], int, str]:
   714→        """Run a deep scan and return (findings, score, grade)."""
   715→        print(f"  {red('⚔  HACKER')}  Scanning {white(self.target)} …")
   716→        try:
   717→            if self.is_local:
   718→                result = audit_scan(target=self.target, modules=self.modules)
   719→            else:
   720→                result = scan(target=self.target, modules=self.modules, all_modules=False)
   721→            findings = result.findings
   722→            score = result.total_score
   723→            grade = result.grade
   724→            mods = result.modules_run
   725→        except Exception as exc:
   726→            print(f"    {dim(f'Scan error: {exc}')}")
   727→            return [], 100, "A"
   728→        print(f"    {red(str(len(findings)))} finding(s) discovered")
   729→        return findings, score, grade
   730→
   731→    # ── CODER phase ───────────────────────────────────────────────────
   732→
   733→    def _run_coder(
   734→        self,
   735→        findings: List[Dict[str, Any]],
   736→        round_num: int,
   737→    ) -> List[RemediationAction]:
   738→        """Generate remediation actions for each finding."""
   739→        print(f"  {green('💻  CODER')}  Generating fixes …")
   740→        actions: List[RemediationAction] = []
   741→        for f in findings:
   742→            title = f.get("title", "")
   743→            category = f.get("category", "")
   744→            severity = f.get("severity", "info")
   745→
   746→            # Skip already-fixed or already-attempted findings
   747→            if title in self._fixed_titles:
   748→                continue
   749→            if title in self._failed_titles:
   750→                continue
   751→
   752→            rule_key = _match_category(title, category)
   753→            if rule_key and rule_key in REMEDIATION_RULES:
   754→                rule = REMEDIATION_RULES[rule_key]
   755→                # Enrich with feedback from previous rounds if available
   756→                fix_cmds = list(rule["fix_commands"])
   757→                if title in self._feedback_context:
   758→                    prior_notes = self._feedback_context[title]
   759→                    fix_cmds.append("")
   760→                    fix_cmds.append("# ── GUARDIAN feedback from prior round(s) ──")
   761→                    for note in prior_notes:
   762→                        fix_cmds.append(f"#   > {note}")
   763→                    fix_cmds.append("#   Ensure the above commands are applied before re-verification.")
   764→
   765→                action = RemediationAction(
   766→                    finding_title=title,
   767→                    finding_category=category,
   768→                    severity=severity,
   769→                    rule_key=rule_key,
   770→                    rule_description=rule["description"],
   771→                    fix_commands=fix_cmds,
   772→                    confidence=rule["confidence"],
   773→                    round_applied=round_num,
   774→                )
   775→            else:
   776→                # Generic remediation for unmapped categories
   777→                action = RemediationAction(
   778→                    finding_title=title,
   779→                    finding_category=category,
   780→                    severity=severity,
   781→                    rule_key=None,
   782→                    rule_description=f"Manual remediation needed for category '{category}'",
   783→                    fix_commands=[
   784→                        f"# Manual investigation required:",
   785→                        f"#   Finding : {title}",
   786→                        f"#   Category: {category}",
   787→                        f"#   Severity: {severity}",
   788→                        f"# Review the evidence and apply appropriate fixes.",
   789→                        f"# Re-scan after fixing to verify closure.",
   790→                    ],
   791→                    confidence=0.30,
   792→                    round_applied=round_num,
   793→                )
   794→            actions.append(action)
   795→        print(f"    {green(str(len(actions)))} fix(es) generated")
   796→        return actions
   797→
   798→    # ── GUARDIAN phase ─────────────────────────────────────────────────
   799→
   800→    def _run_guardian(
   801→        self,
   802→        actions: List[RemediationAction],
   803→        all_findings: List[Dict[str, Any]],
   804→        round_num: int,
   805→    ) -> Tuple[int, int, List[str]]:
   806→        """Rule-based verification of proposed fixes.
   807→
   808→        Returns (verified_fixed_count, still_vulnerable_count, remaining_titles).
   809→        """
   810→        print(f"  {yellow('🛡  GUARDIAN')} Verifying fixes …")
   811→        verified = 0
   812→        still_open = 0
   813→        remaining: List[str] = []
   814→
   815→        # Build a lookup of current finding titles
   816→        finding_titles = {f.get("title", "") for f in all_findings}
   817→
   818→        for action in actions:
   819→            title = action.finding_title
   820→            rule_key = action.rule_key
   821→
   822→            if rule_key and rule_key in REMEDIATION_RULES:
   823→                rule = REMEDIATION_RULES[rule_key]
   824→                verif_keys = rule.get("verification_keys", [])
   825→                confidence = rule.get("confidence", 0.5)
   826→
   827→                # Check severity – high-confidence rules for low-severity findings auto-verify
   828→                if confidence >= 0.90:
   829→                    action.verified = True
   830→                    action.verification_note = (
   831→                        f"High-confidence rule ({confidence:.0%}) — "
   832→                        f"fix commands should resolve '{title}'."
   833→                    )
   834→                    verified += 1
   835→                    self._fixed_titles.add(title)
   836→                    self._failed_titles.discard(title)
   837→                elif confidence >= 0.80:
   838→                    # Medium-high confidence: verify if the finding is still present
   839→                    if title in finding_titles:
   840→                        # Still present after fix — mark as still vulnerable
   841→                        action.verified = False
   842→                        action.verification_note = (
   843→                            f"Finding '{title}' still present after remediation attempt. "
   844→                            f"Commands target: {', '.join(verif_keys[:3])}."
   845→                        )
   846→                        still_open += 1
   847→                        remaining.append(title)
   848→                        self._failed_titles.add(title)
   849→                        # Feed back context for next round
   850→                        self._feedback_context.setdefault(title, []).append(
   851→                            f"Round {round_num}: Fix not effective. Try applying all commands."
   852→                        )
   853→                    else:
   854→                        action.verified = True
   855→                        action.verification_note = f"Finding '{title}' no longer detected."
   856→                        verified += 1
   857→                        self._fixed_titles.add(title)
   858→                        self._failed_titles.discard(title)
   859→                else:
   860→                    # Lower confidence — always flag as needing verification
   861→                    action.verified = False
   862→                    action.verification_note = (
   863→                        f"Rule confidence {confidence:.0%} is below threshold. "
   864→                        f"Manual verification required for '{title}'."
   865→                    )
   866→                    still_open += 1
   867→                    remaining.append(title)
   868→                    self._feedback_context.setdefault(title, []).append(
   869→                        f"Round {round_num}: Low confidence rule. Escalate to manual review."
   870→                    )
   871→            else:
   872→                # No matching rule
   873→                action.verified = False
   874→                action.verification_note = (
   875→                    f"No verification rule for category '{action.finding_category}'. "
   876→                    f"Manual review required."
   877→                )
   878→                still_open += 1
   879→                remaining.append(title)
   880→
   881→        print(f"    {green(str(verified))} verified fixed  │  "
   882→              f"{red(str(still_open))} still vulnerable")
   883→        return verified, still_open, remaining
   884→
   885→    # ── Progress bar ───────────────────────────────────────────────────
   886→
   887→    @staticmethod
   888→    def _progress_bar(fixed: int, total: int, width: int = 30) -> str:
   889→        if total == 0:
   890→            return green("█" * width) + " 100%"
   891→        ratio = fixed / total
   892→        filled = int(width * ratio)
   893→        bar = green("█" * filled) + dim("░" * (width - filled))
   894→        return f"{bar} {ratio:.0%}"
   895→
   896→    # ── Round display ──────────────────────────────────────────────────
   897→
   898→    @staticmethod
   899→    def _print_round_banner(round_num: int, max_rounds: int) -> None:
   900→        print()
   901→        print(f"{white('═' * 60)}")
   902→        print(f"{bold(f'  ROUND {round_num} / {max_rounds}')}")
   903→        print(f"{white('═' * 60)}")
   904→
   905→    # ── Main loop ──────────────────────────────────────────────────────
   906→
   907→    def run(self) -> AdversarialResult:
   908→        """Execute the full adversarial loop and return an AdversarialResult."""
   909→        print()
   910→        print(f"{bold(cyan('  ╔══════════════════════════════════════════════╗'))}")
   911→        print(f"{bold(cyan('  ║    ADVERSARIAL  SELF-PLAY  REMEDIATION      ║'))}")
   912→        print(f"{bold(cyan('  ║    HACKER → CODER → GUARDIAN  (loop)       ║'))}")
   913→        print(f"{bold(cyan('  ╚══════════════════════════════════════════════╝'))}")
   914→        print(f"  Target: {white(self.target)}")
   915→        print(f"  Max rounds: {white(str(self.max_rounds))}")
   916→        print(f"  Mode: {white('LOCAL AUDIT' if self.is_local else 'REMOTE SCAN')}")
   917→
   918→        result = AdversarialResult(
   919→            target=self.target,
   920→            max_rounds=self.max_rounds,
   921→        )
   922→
   923→        initial_findings: List[Dict[str, Any]] = []
   924→        initial_score = 100
   925→        initial_grade = "A"
   926→        all_findings_snapshot: List[Dict[str, Any]] = []
   927→
   928→        for rnd in range(1, self.max_rounds + 1):
   929→            t0 = time.monotonic()
   930→            self._print_round_banner(rnd, self.max_rounds)
   931→
   932→            # ── HACKER ─────────────────────────────────────────────────
   933→            findings, score_before, grade_before = self._run_hacker()
   934→            if rnd == 1:
   935→                initial_findings = list(findings)
   936→                initial_score = score_before
   937→                initial_grade = grade_before
   938→
   939→            # Track new vs already-known findings
   940→            known_titles = self._fixed_titles | self._failed_titles
   941→            new_findings = [f for f in findings if f.get("title", "") not in known_titles]
   942→            hacker_new = len(new_findings)
   943→            hacker_total = len(findings)
   944→
   945→            # ── CODER ──────────────────────────────────────────────────
   946→            actions = self._run_coder(findings, round_num=rnd)
   947→            self._all_actions.extend(actions)
   948→
   949→            # ── GUARDIAN ───────────────────────────────────────────────
   950→            verified_fixed, still_vulnerable, remaining_titles = self._run_guardian(
   951→                actions, findings, round_num=rnd,
   952→            )
   953→            elapsed = time.monotonic() - t0
   954→
   955→            # Compute projected score after fixes
   956→            fixed_deduction = sum(
   957→                f.get("points_deducted", 0)
   958→                for f in findings
   959→                if f.get("title", "") in self._fixed_titles
   960→            )
   961→            projected_score = min(100, 100 - sum(
   962→                f.get("points_deducted", 0)
   963→                for f in findings
   964→                if f.get("title", "") not in self._fixed_titles
   965→            ))
   966→            projected_grade = compute_grade(projected_score)
   967→
   968→            total_open = still_vulnerable + (
   969→                len(findings) - hacker_new
   970→                if rnd == 1
   971→                else len([f for f in findings if f.get("title", "") in self._failed_titles])
   972→            )
   973→
   974→            # Simpler open count: findings not yet fixed
   975→            open_titles = {
   976→                f.get("title", "") for f in findings
   977→            } - self._fixed_titles
   978→            total_open = len(open_titles)
   979→
   980→            progress_str = self._progress_bar(
   981→                len(self._fixed_titles),
   982→                len(initial_findings) if initial_findings else 1,
   983→            )
   984→
   985→            round_summary = RoundSummary(
   986→                round_num=rnd,
   987→                hacker_findings=hacker_total,
   988→                hacker_new=hacker_new,
   989→                coder_fixes=len(actions),
   990→                guardian_verified_fixed=verified_fixed,
   991→                guardian_still_vulnerable=still_vulnerable,
   992→                total_open=total_open,
   993→                score_before=score_before,
   994→                score_after=projected_score,
   995→                grade_before=grade_before,
   996→                grade_after=projected_grade,
   997→                elapsed_seconds=elapsed,
   998→                actions=actions,
   999→                remaining_titles=remaining_titles,
  1000→            )
  1001→            result.rounds.append(round_summary)
  1002→
  1003→            # ── Round summary output ────────────────────────────────────
  1004→            print()
  1005→            print(f"  {dim('─── Round Summary ────────────────────────────')}")
  1006→            print(f"  Findings scanned    : {red(str(hacker_total))}  (new: {red(str(hacker_new))})")
  1007→            print(f"  Fixes generated     : {green(str(len(actions)))}")
  1008→            print(f"  Verified fixed      : {green(str(verified_fixed))}")
  1009→            print(f"  Still vulnerable    : {red(str(still_vulnerable))}")
  1010→            print(f"  Open issues         : {yellow(str(total_open))}")
  1011→            print(f"  Progress            : {progress_str}")
  1012→            print(f"  Score               : {grade_before} ({score_before}) → {projected_grade} ({projected_score})")
  1013→            print(f"  Elapsed             : {dim(f'{elapsed:.1f}s')}")
  1014→
  1015→            # ── Show individual fixes ──────────────────────────────────
  1016→            if actions:
  1017→                print()
  1018→                print(f"  {dim('─── Fixes Applied ────────────────────────────')}")
  1019→                for act in actions:
  1020→                    status_icon = green("✓") if act.verified else red("✗")
  1021→                    sev_color = {
  1022→                        "critical": BRIGHT_RED, "high": RED,
  1023→                        "medium": YELLOW, "low": DIM, "info": DIM,
  1024→                    }.get(act.severity, DIM)
  1025→                    print(f"    {status_icon} {_c(sev_color, act.severity.upper().ljust(8))} "
  1026→                          f"{act.finding_title[:60]}")
  1027→                    if act.rule_key:
  1028→                        print(f"      {dim(f'Rule: {act.rule_key} (confidence {act.confidence:.0%})')}")
  1029→
  1030→            # ── Check convergence ───────────────────────────────────────
  1031→            if total_open == 0:
  1032→                self._converged = True
  1033→                print()
  1034→                print(f"  {green(bold('  ✅ ALL FINDINGS REMEDIATED — loop converged!'))}")
  1035→                break
  1036→
  1037→            if rnd < self.max_rounds and still_vulnerable == 0 and hacker_new == 0:
  1038→                # No new findings and no new fixes — likely converged
  1039→                self._converged = True
  1040→                print()
  1041→                print(f"  {yellow(bold('  ⚠  No new findings or fixes — loop stalled. Stopping early.'))}")
  1042→                break
  1043→
  1044→        # ── Build final report ────────────────────────────────────────
  1045→        result.total_findings_initial = len(initial_findings)
  1046→        result.total_findings_final = len(
  1047→            {f.get("title", "") for f in initial_findings} - self._fixed_titles
  1048→        )
  1049→        result.findings_fixed = len(self._fixed_titles & {
  1050→            f.get("title", "") for f in initial_findings
  1051→        })
  1052→        result.findings_unfixed = result.total_findings_initial - result.findings_fixed
  1053→        result.final_score = result.rounds[-1].score_after if result.rounds else 100
  1054→        result.final_grade = compute_grade(result.final_score)
  1055→        result.initial_score = initial_score
  1056→        result.initial_grade = initial_grade
  1057→        result.converged = self._converged
  1058→        result.remediation_report = self._build_remediation_report(result)
  1059→
  1060→        # ── Final scorecard ────────────────────────────────────────────
  1061→        self._print_scorecard(result)
  1062→
  1063→        return result
  1064→

  1067→    def _build_remediation_report(self, result: AdversarialResult) -> str:
  1068→        lines: List[str] = []
  1069→        lines.append(f"RECONPRO ADVERSARIAL REMEDIATION REPORT")

  1081→        for rnd in result.rounds:
  1082→            lines.append(f"--- Round {rnd.round_num} ---")
  1083→            lines.append(f"  HACKER:   {rnd.hacker_findings} findings ({rnd.hacker_new} new)")
  1084→            lines.append(f"  CODER:    {rnd.coder_fixes} fixes generated")
  1085→            lines.append(f"  GUARDIAN: {rnd.guardian_verified_fixed} fixed, {rnd.guardian_still_vulnerable} still open")
  1086→            lines.append("")
  1087→
  1088→            for act in rnd.actions:
  1089→                status = "FIXED" if act.verified else "OPEN"
  1090→                lines.append(f"  [{status}] {act.finding_title}")

  1092→                if act.rule_key:
  1093→                    lines.append(f"    Rule: {act.rule_key} (confidence {act.confidence:.0%})")
  1094→                lines.append(f"    Fix commands:")
  1095→                for cmd in act.fix_commands:
  1096→                    lines.append(f"      {cmd}")
  1097→                lines.append(f"    Note: {act.verification_note}")
  1098→                lines.append("")
  1099→
  1100→        # Unfixed findings detail
  1101→        unfixed_actions = [a for a in self._all_actions if not a.verified]
  1102→        if unfixed_actions:
  1103→            lines.append("=== REMAINING VULNERABILITIES ===")
  1104→            for act in unfixed_actions:
  1105→                lines.append(f"  - {act.finding_title} ({act.severity})")
  1106→                lines.append(f"    {act.verification_note}")
  1107→            lines.append("")
  1108→
  1109→        return "\n".join(lines)

  1142→# ── Public entry point ────────────────────────────────────────────────────
  1143→
  1144→def run_adversarial(
  1145→    target: str,
  1146→    max_rounds: int = 3,
  1147→    modules: Optional[List[str]] = None,
  1148→    is_local: bool = False,
  1149→) -> AdversarialResult:
  1150→    """Run the adversarial self-play loop against a target.
  1151→
  1152→    Args:
  1153→        target:      Domain/URL for remote scan, or directory for local audit.
  1154→        max_rounds:  Maximum HACKER→CODER→GUARDIAN iterations (default 3).
  1155→        modules:     Optional list of module IDs to use during scans.
  1156→        is_local:    If True, run audit_scan instead of scan.
  1157→
  1158→    Returns:
  1159→        AdversarialResult with all round summaries, remediation report, and scores.
  1160→    """
  1161→    loop = AdversarialLoop(
  1162→        target=target,
  1163→        max_rounds=max_rounds,
  1164→        modules=modules,
  1165→        is_local=is_local,
  1166→    )
  1167→    return loop.run()
