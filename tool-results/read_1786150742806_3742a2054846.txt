   138→class HeaderRotator:
   139→    """Rotates through 50+ real browser User-Agent strings and produces a
   140→    complete, consistent set of HTTP headers for each rotation.
   141→
   142→    Maintains referer state across calls so successive requests look like
   143→    a real browsing session.
   144→    """
   145→
   146→    def __init__(self, seed: Optional[int] = None) -> None:
   147→        self._rng = random.Random(seed)
   148→        self._ua_pool = list(_USER_AGENTS)
   149→        self._rng.shuffle(self._ua_pool)
   150→        self._index = 0
   151→        self._referer: Optional[str] = None
   152→        self._lang_codes = list(_ACCEPT_LANGS.keys())
   153→
   154→    # -- public API ----------------------------------------------------------
   155→
   156→    def set_referer(self, url: str) -> None:
   157→        """Store a referer URL that will be included on the *next* rotate()."""
   158→        self._referer = url
   159→
   160→    def clear_referer(self) -> None:
   161→        """Clear stored referer."""
   162→        self._referer = None
   163→
   164→    def rotate(self, custom_headers: Optional[Dict[str, str]] = None) -> Dict[str, str]:
   165→        """Return a full set of HTTP headers with a randomised User-Agent.
   166→
   167→        Parameters
   168→        ----------
   169→        custom_headers:
   170→            Optional dict of headers that override / extend the generated set.
   171→
   172→        Returns
   173→        -------
   174→        dict  –  mapping of header name → value
   175→        """
   176→        ua = self._pick_ua()
   177→        headers: Dict[str, str] = {
   178→            "User-Agent": ua,
   179→            "Accept": self._accept_for(ua),
   180→            "Accept-Language": _ACCEPT_LANGS[self._rng.choice(self._lang_codes)],
   181→            "Accept-Encoding": "gzip, deflate, br, zstd",
   182→            "Connection": "keep-alive",
   183→            "Cache-Control": "max-age=0",
   184→            "Sec-Fetch-Dest": "document",
   185→            "Sec-Fetch-Mode": "navigate",
   186→            "Sec-Fetch-Site": "none",
   187→        }
   188→
   189→        # Add Sec-CH-UA only for Chrome / Edge
   190→        sec_ch = self._sec_ch_ua(ua)
   191→        if sec_ch is not None:
   192→            headers["Sec-CH-UA"] = sec_ch[0]
   193→            headers["Sec-CH-UA-Mobile"] = sec_ch[1]
   194→            headers["Sec-CH-UA-Platform"] = sec_ch[2]
   195→
   196→        if self._referer is not None:
   197→            headers["Referer"] = self._referer
   198→            self._referer = None
   199→
   200→        if custom_headers:
   201→            headers.update(custom_headers)
   202→
   203→        return headers
   204→
   205→    # -- internals -----------------------------------------------------------
   206→
   207→    def _pick_ua(self) -> str:
   208→        """Pick next UA in shuffled order, re-shuffle when exhausted."""
   209→        if self._index >= len(self._ua_pool):
   210→            self._rng.shuffle(self._ua_pool)
   211→            self._index = 0
   212→        ua = self._ua_pool[self._index]
   213→        self._index += 1
   214→        return ua
   215→
   216→    @staticmethod
   217→    def _accept_for(ua: str) -> str:
   218→        """Return an Accept header appropriate for the browser family."""
   219→        if "Edg/" in ua:
   220→            return (
   221→                "text/html,application/xhtml+xml,application/xml;q=0.9,"
   222→                "image/avif,image/webp,image/apng,*/*;q=0.8,"
   223→                "application/signed-exchange;v=b3;q=0.7"
   224→            )
   225→        if "Firefox/" in ua:
   226→            return (
   227→                "text/html,application/xhtml+xml,application/xml;q=0.9,"
   228→                "image/avif,image/webp,*/*;q=0.8"
   229→            )
   230→        if "Safari/" in ua and "Chrome/" not in ua:
   231→            return (
   232→                "text/html,application/xhtml+xml,application/xml;q=0.9,"
   233→                "*/*;q=0.8"
   234→            )
   235→        # Chrome / generic Chromium
   236→        return (
   237→            "text/html,application/xhtml+xml,application/xml;q=0.9,"
   238→            "image/avif,image/webp,image/apng,*/*;q=0.8,"
   239→            "application/signed-exchange;v=b3;q=0.7"
   240→        )
   241→
   242→    @staticmethod
   243→    def _sec_ch_ua(ua: str) -> Optional[tuple]:
   244→        """Return (Sec-CH-UA, Mobile, Platform) or None for non-Chrome."""
   245→        if "Edg/" in ua:
   246→            ver = _extract_chrome_ver(ua)
   247→            ch = _SEC_CH_UA_EDGE.get(ver)
   248→            platform = _extract_platform(ua)
   249→            if ch:
   250→                return (ch, "?0", f'"{platform}"')
   251→            return None
   252→        if "Chrome/" in ua and "Safari/" in ua and "Edg/" not in ua:
   253→            ver = _extract_chrome_ver(ua)
   254→            is_mobile = "Mobile" in ua
   255→            pool = _SEC_CH_UA_MOBILE if is_mobile else _SEC_CH_UA
   256→            ch = pool.get(ver)
   257→            platform = _extract_platform(ua)
   258→            if ch:
   259→                return (ch, "?1" if is_mobile else "?0", f'"{platform}"')
   260→            return None
   261→        return None
   262→
   263→
   264→# helpers for HeaderRotator
   265→
   266→
   267→def _extract_chrome_ver(ua: str) -> str:
   268→    """Extract the major Chrome version number from a UA string."""
   269→    import re
   270→    m = re.search(r"Chrome/(\d+)\.", ua)
   271→    return m.group(1) if m else "124"
   272→
   273→
   274→def _extract_platform(ua: str) -> str:
   275→    """Guess the platform string for Sec-CH-UA-Platform."""
   276→    if "Windows" in ua:
   277→        return "Windows"
   278→    if "Macintosh" in ua or "Mac OS X" in ua:
   279→        return "macOS"
   280→    if "Android" in ua:
   281→        return "Android"
   282→    if "iPhone" in ua or "iPad" in ua:
   283→        return "iOS"
   284→    if "Linux" in ua:
   285→        return "Linux"
   286→    return "Unknown"

   375→            "cipher_suites": [
   376→                "TLS_AES_128_GCM_SHA256",
   377→                "TLS_AES_256_GCM_SHA384",
   378→                "TLS_CHACHA20_POLY1305_SHA256",
   379→                "TLS_ECDHE_ECDSA_WITH_AES_128_GCM_SHA256",
   380→                "TLS_ECDHE_RSA_WITH_AES_128_GCM_SHA256",
   381→                "TLS_ECDHE_ECDSA_WITH_AES_256_GCM_SHA384",
   382→                "TLS_ECDHE_RSA_WITH_AES_256_GCM_SHA384",
   383→                "TLS_ECDHE_ECDSA_WITH_CHACHA20_POLY1305_SHA256",
   384→                "TLS_ECDHE_RSA_WITH_CHACHA20_POLY1305_SHA256",
   385→                "TLS_RSA_WITH_AES_128_GCM_SHA256",
   386→                "TLS_RSA_WITH_AES_256_GCM_SHA384",
   387→            ],
   388→            "tls_version": "TLSv1.3",
   389→            "extensions": [
   390→                "server_name", "extended_master_secret", "renegotiation_info",
   391→                "supported_versions", "psk_key_exchange_modes", "key_share",
   392→                "signature_algorithms", "supported_groups", "application_layer_protocol_negotiation",
   393→                "signed_certificate_timestamp", "status_request",
   394→            ],
   395→            "grease": True,
   396→        },
   397→    }
   398→
   399→    def __init__(self, seed: Optional[int] = None) -> None:
   400→        self._rng = random.Random(seed)
   401→
   402→    def get_profile(self, name: str = "CHROME_120") -> Dict[str, Any]:
   403→        """Return a TLS fingerprint profile dict.
   404→
   405→        Parameters
   406→        ----------
   407→        name : str
   408→            One of CHROME_120, FIREFOX_121, SAFARI_17, EDGE_120, RANDOM.
   409→
   410→        Returns
   411→        -------
   412→        dict with keys: cipher_suites, tls_version, extensions, grease
   413→        """
   414→        if name == self.RANDOM:
   415→            return self._random_profile()
   416→        profile = self._PRESETS.get(name)
   417→        if profile is None:
   418→            raise ValueError(
   419→                f"Unknown TLS profile '{name}'. "
   420→                f"Choose from: {', '.join(self._PRESETS)}, {self.RANDOM}"
   421→            )
   422→        # Return a deep-ish copy so callers can mutate safely
   423→        return {
   424→            "cipher_suites": list(profile["cipher_suites"]),
   425→            "tls_version": profile["tls_version"],
   426→            "extensions": list(profile["extensions"]),
   427→            "grease": profile["grease"],
   428→        }

   430→    def _random_profile(self) -> Dict[str, Any]:
   431→        """Create a randomized TLS profile that still looks plausible."""
   432→        base_name = self._rng.choice(list(self._PRESETS.keys()))
   433→        base = self._PRESETS[base_name]
   434→        ciphers = list(base["cipher_suites"])
   435→        self._rng.shuffle(ciphers)
   436→        # Keep AES_128_GCM near the top 60 % of the time (most browsers prefer it)
   437→        if self._rng.random() < 0.6:
   438→            aes128 = "TLS_AES_128_GCM_SHA256"
   439→            if aes128 in ciphers:
   440→                ciphers.remove(aes128)
   441→                ciphers.insert(self._rng.randint(0, 2), aes128)
   442→        extensions = list(base["extensions"])
   443→        self._rng.shuffle(extensions)
   444→        # SNI should stay first
   445→        if "server_name" in extensions:
   446→            extensions.remove("server_name")
   447→            extensions.insert(0, "server_name")
   448→        return {
   449→            "cipher_suites": ciphers,
   450→            "tls_version": base["tls_version"],
   451→            "extensions": extensions,
   452→            "grease": base["grease"],
   453→        }

   460→# ---------------------------------------------------------------------------
   461→# 3. TimingJitter
   462→# ---------------------------------------------------------------------------
   463→
   464→
   465→class TimingJitter:
   466→    """Add human-like timing jitter between requests.
   467→
   468→    Uses a Gaussian distribution with configurable stddev.  Supports
   469→    burst patterns that simulate real browsing (rapid clicks then pause).
   470→    """
   471→
   472→    MIN_DELAY: float = 0.05
   473→    MAX_DELAY: float = 5.0
   474→    DEFAULT_STDDEV: float = 0.3
   475→
   476→    def __init__(
   477→        self,
   478→        stddev: float = DEFAULT_STDDEV,
   479→        min_delay: float = MIN_DELAY,
   480→        max_delay: float = MAX_DELAY,
   481→        burst_enabled: bool = True,
   482→        seed: Optional[int] = None,
   483→    ) -> None:
   484→        self._rng = random.Random(seed)
   485→        self._stddev = stddev
   486→        self._min = min_delay
   487→        self._max = max_delay
   488→        self._burst_enabled = burst_enabled
   489→        self._burst_remaining = 0
   490→        self._in_burst = False
   491→
   492→    def add_jitter(self) -> float:
   493→        """Return a delay in seconds drawn from a Gaussian distribution,
   494→        clamped to [min_delay, max_delay].
   495→
   496→        If burst mode is enabled, some calls will return near-zero delay to
   497→        simulate rapid human clicks, followed by a longer pause.
   498→        """
   499→        if self._burst_enabled:
   500→            delay = self._burst_jitter()
   501→        else:
   502→            delay = self._gaussian_jitter()
   503→        return max(self._min, min(self._max, delay))
   504→
   505→    def _gaussian_jitter(self) -> float:
   506→        """Pure Gaussian jitter: mean=0, clamped so result is >= min."""
   507→        sample = self._rng.gauss(0, self._stddev)
   508→        # Shift so the result is a positive delay
   509→        return max(self._min, abs(sample) + self._min)
   510→
   511→    def _burst_jitter(self) -> float:
   512→        """Simulate a burst of rapid requests then a pause.
   513→
   514→        Each burst consists of 2-5 requests with minimal delay (~0.05-0.15s),
   515→        then a pause of 1.5-4.0s before the next burst.
   516→        """
   517→        if not self._in_burst:
   518→            # Decide whether to start a burst (40 % chance)
   519→            if self._rng.random() < 0.4:
   520→                self._in_burst = True
   521→                self._burst_remaining = self._rng.randint(2, 5)
   522→            else:
   523→                # Normal delay
   524→                return self._gaussian_jitter()
   525→
   526→        if self._in_burst:
   527→            if self._burst_remaining > 1:
   528→                self._burst_remaining -= 1
   529→                return self._rng.uniform(0.05, 0.15)
   530→            else:
   531→                # End of burst — return a longer pause
   532→                self._in_burst = False
   533→                self._burst_remaining = 0
   534→                return self._rng.uniform(1.5, 4.0)
   535→
   536→        return self._gaussian_jitter()
   537→

   540→# 4. PathCanonicalizer
   541→# ---------------------------------------------------------------------------
   542→
   543→
   544→class PathCanonicalizer:
   545→    """Generate variations of a URL path to test WAF / CDN normalisation.
   546→
   547→    Produces 8-12 variants using techniques like dot-segments, URL-encoding,
   548→    mixed case, trailing slashes, and overlong UTF-8 sequences.
   549→    """
   550→
   551→    def generate_variants(self, path: str) -> List[str]:
   552→        """Return a list of 8-12 path variations for WAF testing.
   553→
   554→        Parameters
   555→        ----------
   556→        path : str
   557→            The original URL path (e.g. ``/api/v1/users``).
   558→
   559→        Returns
   560→        -------
   561→        list[str]
   562→        """
   563→        if not path.startswith("/"):
   564→            path = "/" + path
   565→
   566→        variants: List[str] = []
   567→        # 1. Original
   568→        variants.append(path)
   569→        # 2. Dot-segment prefix
   570→        variants.append("/./" + path.lstrip("/"))
   571→        # 3. Double-encoding of first char after slash
   572→        variants.append(self._double_encode_char(path))
   573→        # 4. URL-encoded full first segment
   574→        variants.append(self._encode_segment(path))
   575→        # 5. Mixed case — title-case each segment
   576→        variants.append(self._mixed_case(path))
   577→        # 6. UPPERCASE
   578→        variants.append(path.upper())
   579→        # 7. Trailing slash
   580→        variants.append(path.rstrip("/") + "/")
   581→        # 8. Double trailing slash
   582→        variants.append(path.rstrip("/") + "//")
   583→        # 9. Trailing dot-segment
   584→        variants.append(path.rstrip("/") + "/.")
   585→        # 10. Overlong UTF-8 for first slash-separated segment
   586→        variants.append(self._overlong_utf8(path))
   587→        # 11. Null-byte injection (common WAF bypass)
   588→        variants.append(path.rstrip("/") + "%00")
   589→        # 12. Semicolon / path-param injection
   590→        variants.append(path.rstrip("/") + ";jsessionid=" + _rand_hex(8))
   591→
   592→        # De-duplicate while preserving order
   593→        seen: set = set()
   594→        unique: List[str] = []
   595→        for v in variants:
   596→            if v not in seen:
   597→                seen.add(v)
   598→                unique.append(v)
   599→
   600→        return unique
   601→
   602→    # -- variant generators --------------------------------------------------
   603→
   604→    @staticmethod
   605→    def _double_encode_char(path: str) -> str:
   606→        """Double-encode the first character of each path segment.
   607→
   608→        E.g. '/a/b' → '/%2561/%2562'
   609→        """
   610→        parts = path.strip("/").split("/")
   611→        encoded_parts: List[str] = []
   612→        for part in parts:
   613→            if not part:
   614→                continue
   615→            first = part[0]
   616→            double = "%25" + ("%02X" % ord(first))
   617→            encoded_parts.append(double + part[1:])
   618→        return "/" + "/".join(encoded_parts)
   619→
   620→    @staticmethod
   621→    def _encode_segment(path: str) -> str:
   622→        """Fully URL-encode the first path segment."""
   623→        parts = path.strip("/").split("/")
   624→        if not parts or not parts[0]:
   625→            return path
   626→        encoded_first = urllib.parse.quote(parts[0], safe="")
   627→        return "/" + encoded_first + ("/" + "/".join(parts[1:]) if len(parts) > 1 else "")
   628→
   629→    @staticmethod
   630→    def _mixed_case(path: str) -> str:
   631→        """Title-case each segment: /api/v1/users → /Api/V1/Users."""
   632→        parts = path.strip("/").split("/")
   633→        titled = [p[:1].upper() + p[1:].lower() if p else "" for p in parts]
   634→        return "/" + "/".join(titled)
   635→
   636→    @staticmethod
   637→    def _overlong_utf8(path: str) -> str:
   638→        """Replace the first character of the first segment with an overlong
   639→        UTF-8 2-byte sequence (e.g. 'a' → %C0%AF for '/').
   640→
   641→        Specifically this targets the forward-slash character to produce
   642→        /path%C0%AFrest style bypasses.
   643→        """
   644→        parts = path.strip("/").split("/")
   645→        if not parts:
   646→            return path
   647→        # Use a valid overlong encoding of 'a' (U+0061) as 2-byte: 0xC1 0xA1
   648→        # Note: many modern systems reject overlong UTF-8, but some WAFs still
   649→        # mishandle it. We encode the first char of the first segment.
   650→        first = parts[0][0] if parts[0] else "a"
   651→        cp = ord(first)
   652→        if cp < 0x80:
   653→            # Overlong 2-byte: 110xxxxx 10xxxxxx → 0xC0|cp>>6, 0x80|cp&0x3F
   654→            b1 = 0xC0 | (cp >> 6)
   655→            b2 = 0x80 | (cp & 0x3F)
   656→            overlong = f"%{b1:02X}%{b2:02X}"
   657→            parts[0] = overlong + parts[0][1:]
   658→        return "/" + "/".join(parts)
   659→
   660→
   661→def _rand_hex(n: int) -> str:
   662→    """Return a random hex string of length *n*."""
   663→    return "".join(random.choice("0123456789abcdef") for _ in range(n))

   776→    def apply_to_request(
   777→        self,
   778→        url: str,
   779→        method: str = "GET",
   780→        headers: Optional[Dict[str, str]] = None,
   781→    ) -> Dict[str, Any]:
   782→        """Build a fully evasive request configuration.
   783→
   784→        Returns
   785→        -------
   786→        dict with keys:
   787→            url     – possibly canonicalised URL
   788→            headers – rotated header dict
   789→            delay   – jitter delay in seconds (caller should ``time.sleep``)
   790→            method  – HTTP method
   791→        """

   793→        final_url = url
   794→        if self._config.get("rotate_path") and self._config.get("path_variant_chance", 0) > 0:
   795→            if self._rng.random() < self._config["path_variant_chance"]:

   797→                variants = self._path_canon.generate_variants(parsed.path)
   798→                chosen = self._rng.choice(variants)
   799→                final_url = urllib.parse.urlunparse(
   800→                    (parsed.scheme, parsed.netloc, chosen, parsed.params, parsed.query, parsed.fragment)
   801→                )
   802→
   803→        # Rotate headers
   804→        rotated = self._header_rotator.rotate(custom_headers=headers)
   805→
   806→        # Compute delay
   807→        delay = self._timing.add_jitter()
   808→
   809→        return {
   810→            "url": final_url,
   811→            "headers": rotated,
   812→            "delay": delay,
   813→            "method": method.upper(),
   814→        }

   830→# 6. EvasionSession

   833→
   834→class EvasionSession:
   835→    """Stateful evasion across multiple requests to the same target.
   836→
   837→    Maintains a consistent User-Agent, cookies, and timing pattern so that
   838→    a sequence of requests looks like one real human browsing a site.
   839→    """
   840→
   841→    def __init__(
   842→        self,
   843→        target_base: str,
   844→        profile_name: str = EvasionProfile.BALANCED,
   845→        seed: Optional[int] = None,
   846→    ) -> None:
   847→        self._target_base = target_base.rstrip("/")
   848→        self._profile = EvasionProfile(name=profile_name, seed=seed)
   849→        self._rng = random.Random(seed)
   850→
   851→        # Session state
   852→        self._request_count = 0
   853→        self._current_ua: Optional[str] = None
   854→        self._current_headers: Dict[str, str] = {}
   855→        self._cookies: Dict[str, str] = {}
   856→        self._visited_urls: List[str] = []
   857→        self._timing = self._profile._timing
   858→        self._tls = TlsFingerprint(seed=seed)
   859→        self._tls_profile_name = self._rng.choice(["CHROME_120", "FIREFOX_121", "SAFARI_17", "EDGE_120"])
   860→        self._tls_profile = self._tls.get_profile(self._tls_profile_name)
   861→
   862→        # Initialise with one consistent set of headers
   863→        self._current_headers = self._profile._header_rotator.rotate()
   864→        self._current_ua = self._current_headers.get("User-Agent", "")
   865→
   866→    def next_request(self, url: str) -> Dict[str, Any]:
   867→        """Return a complete request configuration for the next request.
   868→
   869→        The returned dict has keys: url, headers, delay, method, cookies,
   870→        tls_profile.
   871→
   872→        The session keeps the same User-Agent across all requests (to look
   873→        like one user) while still varying other headers slightly.
   874→        """
   875→        self._request_count += 1
   876→
   877→        # Keep the same UA, but lightly rotate other headers
   878→        headers = dict(self._current_headers)
   879→        headers["User-Agent"] = self._current_ua
   880→
   881→        # Set referer to the last visited URL (simulates navigation)
   882→        if self._visited_urls:
   883→            headers["Referer"] = self._visited_urls[-1]
   884→
   885→        # Vary Sec-Fetch-* headers based on navigation pattern
   886→        if self._request_count == 1:
   887→            headers["Sec-Fetch-Site"] = "none"
   888→            headers["Sec-Fetch-Mode"] = "navigate"
   889→        elif url.startswith(self._target_base):
   890→            headers["Sec-Fetch-Site"] = "same-origin"
   891→            # Every few requests simulate a sub-resource fetch
   892→            if self._rng.random() < 0.2:
   893→                headers["Sec-Fetch-Dest"] = "script"
   894→                headers["Sec-Fetch-Mode"] = "no-cors"
   895→            else:
   896→                headers["Sec-Fetch-Dest"] = "document"
   897→                headers["Sec-Fetch-Mode"] = "navigate"
   898→        else:
   899→            headers["Sec-Fetch-Site"] = "cross-site"
   900→            headers["Sec-Fetch-Mode"] = "navigate"
   901→
   902→        # Slightly vary Accept-Language on ~10 % of requests (user might
   903→        # have changed language setting, or we simulate a shared IP)
   904→        if self._rng.random() < 0.1:
   905→            lang_keys = list(_ACCEPT_LANGS.keys())
   906→            headers["Accept-Language"] = _ACCEPT_LANGS[self._rng.choice(lang_keys)]
   907→
   908→        # Compute delay
   909→        delay = self._timing.add_jitter()
   910→
   911→        # Track visited URL
   912→        self._visited_urls.append(url)
   913→
   914→        # Build cookie header if we have cookies
   915→        cookie_header = "; ".join(f"{k}={v}" for k, v in self._cookies.items())
   916→        if cookie_header:
   917→            headers["Cookie"] = cookie_header
   918→
   919→        return {
   920→            "url": url,
   921→            "headers": headers,
   922→            "delay": delay,
   923→            "method": "GET",
   924→            "cookies": dict(self._cookies),
   925→            "tls_profile": self._tls_profile,
   926→            "request_count": self._request_count,
   927→        }
   928→
   929→    def update_cookies(self, cookies: Dict[str, str]) -> None:
   930→        """Update the session cookie jar (e.g. from a Set-Cookie response)."""
   931→        self._cookies.update(cookies)
