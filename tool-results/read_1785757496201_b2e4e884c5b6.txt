  1170→  1219→  1219→    def _run_scan_worker(
  1171→  1220→  1220→        self,
  1172→  1221→  1221→        target: str,
  1173→  1222→  1222→        modules: Optional[List[str]] = None,
  1174→  1223→  1223→        is_local: bool = False,
  1175→  1224→  1224→    ) -> None:
  1176→  1225→  1225→        """Run a scan in a background thread."""

  1178→  1227→  1227→
  1179→  1228→  1228→        # Set modules to scanning
  1180→  1229→  1229→        if modules:
  1181→  1230→  1230→            for m in modules:
  1182→  1231→  1231→                if m in self._module_cells:
  1183→  1232→  1232→                    self.call_from_thread(self._module_cells[m].set_status, "scanning")
  1184→  1233→  1233→        else:
  1185→  1234→  1234→            all_mods = (
  1186→  1235→  1235→                list(MODULE_NAMES.keys())
  1187→  1236→  1236→                if is_local
  1188→  1237→  1237→                else [k for k in MODULE_NAMES if k not in ("host", "dev", "doctor")]
  1189→  1238→  1238→            )
  1190→  1239→  1239→            for m in all_mods:
  1191→  1240→  1240→                if m in self._module_cells:
  1192→  1241→  1241→                    self.call_from_thread(self._module_cells[m].set_status, "scanning")
  1193→  1242→  1242→
  1194→  1243→  1243→        try:
  1195→  1244→  1244→            from .scanner import scan, audit_scan
  1196→  1245→  1245→            from .history import save_scan
  1197→  1246→  1246→
  1198→  1247→  1247→            self.call_from_thread(
  1199→  1248→  1248→                self._chat, f"[{DIM_CYAN}]  ⏳ Scan in progress...[/]"
  1200→  1249→  1249→            )
  1201→  1250→  1250→
  1202→  1251→  1251→            if is_local:
  1203→  1252→  1252→                result = audit_scan(target=target, modules=modules)
  1204→  1253→  1253→            else:
  1205→  1254→  1254→                result = scan(target=target, modules=modules)
  1206→  1255→  1255→
  1207→  1256→  1256→            # Process findings
  1208→  1257→  1257→            data = result.to_dict()
  1209→  1258→  1258→            self.call_from_thread(self._on_scan_complete, data, modules, is_local)
  1210→  1259→  1259→
  1211→  1260→  1260→            # Save to history
  1212→  1261→  1261→            try:
  1213→  1262→  1262→                save_scan(data, label="nexus")
  1214→  1263→  1263→            except Exception:
  1215→  1264→  1264→                pass
  1216→  1265→  1265→
  1217→  1266→  1266→        except Exception as e:
  1218→  1267→  1267→            self.call_from_thread(self._chat, f"[{RED}]  ✗ Scan error: {e}[/]")
  1219→  1268→  1268→            if modules:
  1220→  1269→  1269→                for m in modules:
  1221→  1270→  1270→                    if m in self._module_cells:
  1222→  1271→  1271→                        self.call_from_thread(self._module_cells[m].set_status, "error")
  1223→  1272→  1272→            else:
  1224→  1273→  1273→                self.call_from_thread(self._set_all_modules_status, "error")
  1225→  1274→  1274→
  1226→  1275→  1275→        finally:
  1227→  1276→  1276→            self.call_from_thread(self.is_scanning.set, False)

  1581→  1630→  1630→        """Open finding detail modal for the clicked finding.
  1582→  1631→  1631→
  1583→  1632→  1632→        Uses the vertical click position to determine which finding index
  1584→  1633→  1633→        the user clicked on, mapping it to _findings_list.
  1585→  1634→  1634→        """
  1586→  1635→  1635→        try:
  1587→  1636→  1636→            if (
  1588→  1637→  1637→                isinstance(event.widget, RichLog)
  1589→  1638→  1638→                and event.widget.id == "findings-feed"
  1590→  1639→  1639→                and self._findings_list
  1591→  1640→  1640→            ):

  1593→  1642→  1642→                # Each finding entry is 1 line in the RichLog.
  1594→  1643→  1643→                # We use the y coordinate relative to the widget to estimate the index.
  1595→  1644→  1644→                region = event.widget.region

  1599→  1648→  1648→                    click_y = event.y - region.y
  1600→  1649→  1649→                    # Each finding takes 1 line; estimate index from click position
  1601→  1650→  1650→                    # Account for scroll position
  1602→  1651→  1651→                    line_height = 1

  1604→  1653→  1653→                        line_height = event.widget._line_height
  1605→  1654→  1654→                    clicked_index = int((click_y + scroll_offset * line_height) / line_height)
  1606→  1655→  1655→                    # Clamp to valid range
  1607→  1656→  1656→                    clicked_index = max(0, min(clicked_index, len(self._findings_list) - 1))
  1608→  1657→  1657→                    finding = self._findings_list[clicked_index]
  1609→  1658→  1658→                else:
  1610→  1659→  1659→                    finding = self._findings_list[-1]
  1611→  1660→  1660→                self.push_screen(FindingDetailModal(finding))
