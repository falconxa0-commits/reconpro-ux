    89→function getSeverityConfig(severity: string) {
    90→  switch (severity) {
    91→    case 'critical':
    92→      return {
    93→        color: '#ff3355',
    94→        bg: 'rgba(244,63,94,0.15)',
    95→        border: 'rgba(244,63,94,0.3)',
    96→        icon: XCircle,
    97→        label: 'CRITICAL',
    98→      };
    99→    case 'high':
   100→      return {
   101→        color: '#ff8844',
   102→        bg: 'rgba(251,191,36,0.15)',
   103→        border: 'rgba(251,191,36,0.3)',
   104→        icon: AlertTriangle,
   105→        label: 'HIGH',
   106→      };
   107→    case 'medium':
   108→      return {
   109→        color: '#d29922',
   110→        bg: 'rgba(210,153,34,0.15)',
   111→        border: 'rgba(210,153,34,0.3)',
   112→        icon: AlertCircle,
   113→        label: 'MEDIUM',
   114→      };
   115→    case 'low':
   116→      return {
   117→        color: '#44aaff',
   118→        bg: 'rgba(88,166,255,0.15)',
   119→        border: 'rgba(88,166,255,0.3)',
   120→        icon: FileWarning,
   121→        label: 'LOW',
   122→      };
   123→    default:
   124→      return {
   125→        color: '#444444',
   126→        bg: 'rgba(139,148,158,0.15)',
   127→        border: 'rgba(139,148,158,0.3)',
   128→        icon: AlertCircle,
   129→        label: 'INFO',
   130→      };
   131→  }
   132→}
   133→
   134→function getScheduleColor(schedule: string): string {
   135→  switch (schedule) {
   136→    case 'hourly':
   137→      return 'bg-[rgba(52,211,153,0.15)] text-[#00ff88] border-[rgba(52,211,153,0.3)]';
   138→    case 'daily':
   139→      return 'bg-[rgba(88,166,255,0.15)] text-[#44aaff] border-[rgba(88,166,255,0.3)]';
   140→    case 'weekly':
   141→      return 'bg-[rgba(210,153,34,0.15)] text-[#d29922] border-[rgba(210,153,34,0.3)]';
   142→    case 'monthly':
   143→      return 'bg-[rgba(168,85,247,0.15)] text-[#888888] border-[rgba(168,85,247,0.3)]';
   144→    default:
   145→      return 'bg-[rgba(139,148,158,0.15)] text-[#444444] border-[rgba(139,148,158,0.3)]';
   146→  }
   147→}
   148→
   149→function getPolicyStatusConfig(status: string) {
   150→  switch (status) {
   151→    case 'active':
   152→      return { color: '#00ff88', label: 'Active' };
   153→    case 'warning':
   154→      return { color: '#d29922', label: 'Warning' };
   155→    case 'error':
   156→      return { color: '#ff3355', label: 'Error' };
   157→    case 'idle':
   158→      return { color: '#333333', label: 'Paused' };
   159→    default:
   160→      return { color: '#333333', label: 'Unknown' };
   161→  }
   162→}
   163→
   164→function getScanTypeIcon(type: string) {
   165→  switch (type) {
   166→    case 'Full Infrastructure':
   167→    case 'full':
   168→      return Server;
   169→    case 'API Surface':
   170→    case 'api':
   171→      return Globe;
   172→    case 'SSL/TLS':
   173→    case 'ssl':
   174→      return Shield;
   175→    case 'DNS Recon':
   176→    case 'dns':
   177→      return MonitorCheck;
   178→    default:
   179→      return Activity;
   180→  }
   181→}
   182→
   183→function getAlertStatusConfig(status: string) {
   184→  switch (status) {
   185→    case 'new':
   186→      return { color: '#ff3355', bg: 'rgba(244,63,94,0.15)', label: 'NEW' };
   187→    case 'acknowledged':
   188→      return { color: '#d29922', bg: 'rgba(210,153,34,0.15)', label: 'ACK' };
   189→    case 'resolved':
   190→      return { color: '#00ff88', bg: 'rgba(52,211,153,0.15)', label: 'FIXED' };
   191→    default:
   192→      return { color: '#333333', bg: 'rgba(72,79,88,0.15)', label: 'N/A' };
   193→  }
   194→}

   214→
   215→export function MonitoringPanel() {
   216→  const [data, setData] = useState<MonitoringData | null>(null);
   217→  const [loading, setLoading] = useState(true);
   218→  const [error, setError] = useState('');
   219→  const [addOpen, setAddOpen] = useState(false);
   220→  const [newName, setNewName] = useState('');
   221→  const [newDomain, setNewDomain] = useState('');
   222→  const [newSchedule, setNewSchedule] = useState<'hourly' | 'daily' | 'weekly' | 'monthly'>('daily');
   223→  const [newScanType, setNewScanType] = useState('full');
   224→
   225→  const fetchData = useCallback(async () => {
   226→    setLoading(true);
   227→    setError('');
   228→    try {
   229→      const res = await fetch('/api/monitoring');
   230→      if (!res.ok) throw new Error(`HTTP ${res.status}`);
   231→      const json = await res.json();
   232→      setData({
   233→        policies: json.policies || [],
   234→        scheduledRuns: json.scheduledRuns || [],
   235→        alerts: json.alerts || [],
   236→      });
   237→    } catch (err) {
   238→      console.error('Failed to fetch monitoring data:', err);
   239→      setError('Failed to load monitoring data. Please try again.');
   240→    } finally {
   241→      setLoading(false);
   242→    }
   243→  }, []);
   244→
   245→  useEffect(() => {
   246→    fetchData();
   247→  }, [fetchData]);
   248→
   249→  const policies = data?.policies || [];
   250→  const scheduledRuns = data?.scheduledRuns || [];
   251→  const alerts = data?.alerts || [];
   252→
   253→  const togglePolicy = async (id: string) => {
   254→    const policy = policies.find((p) => p.id === id);
   255→    if (!policy) return;
   256→    try {
   257→      await fetch('/api/monitoring', {
   258→        method: 'PATCH',
   259→        headers: { 'Content-Type': 'application/json' },
   260→        body: JSON.stringify({ id, enabled: !policy.enabled }),
   261→      });
   262→      await fetchData();
   263→    } catch (err) {
   264→      console.error('Failed to toggle policy:', err);
   265→    }
   266→  };
   267→
   268→  const deletePolicy = async (id: string) => {
   269→    try {
   270→      await fetch(`/api/monitoring?id=${id}`, { method: 'DELETE' });
   271→      await fetchData();
   272→    } catch (err) {
   273→      console.error('Failed to delete policy:', err);
   274→    }
   275→  };
   276→
   277→  const handleNewPolicy = async () => {
   278→    if (!newName || !newDomain) return;
   279→    try {
   280→      await fetch('/api/monitoring', {
   281→        method: 'POST',
   282→        headers: { 'Content-Type': 'application/json' },
   283→        body: JSON.stringify({
   284→          name: newName,
   285→          targetDomain: newDomain,
   286→          schedule: newSchedule,
   287→          scanType: newScanType,
   288→        }),
   289→      });
   290→      setAddOpen(false);
   291→      setNewName('');
   292→      setNewDomain('');
   293→      setNewSchedule('daily');
   294→      setNewScanType('full');
   295→      await fetchData();
   296→    } catch (err) {
   297→      console.error('Failed to create policy:', err);
   298→    }
   299→  };
   300→
   301→  const handleRunNow = (id: string) => {
   302→    console.log('Run policy now:', id);
   303→  };
   304→
   305→  const activePolicies = policies.filter((p) => p.enabled).length;
   306→  const totalFindings = policies.reduce((acc, p) => acc + p.findings, 0);
   307→
   308→  if (loading) {
   309→    return (
   310→      <div className="w-full flex items-center justify-center py-20">
   311→        <p className="text-[#444444]">Loading...</p>
   312→      </div>
   313→    );
   314→  }
   315→
   316→  if (error) {
   317→    return (
   318→      <div className="w-full flex flex-col items-center justify-center gap-3 py-20">
   319→        <AlertCircle className="h-8 w-8 text-red-400" />
   320→        <p className="text-sm text-red-400">{error}</p>
   321→        <Button variant="outline" size="sm" onClick={fetchData} className="border-zinc-700 text-white hover:bg-zinc-800">
   322→          <RefreshCw className="mr-2 h-3.5 w-3.5" />
   323→          Retry
   324→        </Button>
   325→      </div>
   326→    );
   327→  }
   328→
   329→  return (
   330→    <motion.div
   331→      variants={containerVariants}
   332→      initial="hidden"
   333→      animate="visible"
   334→      className="w-full space-y-6"
   335→    >
   336→      {/* ── Header ──────────────────────────────────────────────────────── */}
   337→      <motion.div variants={itemVariants} className="flex flex-col sm:flex-row sm:items-center justify-between gap-4">
   338→        <div className="flex items-center gap-3">
   339→          <div className="p-2 rounded-lg bg-[rgba(52,211,153,0.1)] border border-[rgba(52,211,153,0.2)]">
   340→            <Activity className="w-5 h-5 text-[#00ff88]" />
   341→          </div>
   342→          <div>
   343→            <h2 className="text-xl font-bold text-[#f0f0f0]">Continuous Monitoring</h2>
   344→            <p className="text-sm text-[#444444]">Automated attack surface surveillance and policy management</p>
   345→          </div>
   346→        </div>
   347→        <Dialog open={addOpen} onOpenChange={setAddOpen}>
   348→          <DialogTrigger asChild>
   349→            <Button className="bg-[#00ff88] hover:bg-[#00cc6a] text-[#0a0d14] font-semibold gap-2">
   350→              <Plus className="w-4 h-4" />
   351→              New Policy
   352→            </Button>
   353→          </DialogTrigger>
   354→          <DialogContent className="bg-[#080b14] border-[#21262d] text-[#f0f0f0]">
   355→            <DialogHeader>
   356→              <DialogTitle className="text-[#f0f0f0]">Create Monitoring Policy</DialogTitle>
   357→              <DialogDescription className="text-[#444444]">
   358→                Set up automated monitoring for a target domain.
   359→              </DialogDescription>
   360→            </DialogHeader>
   361→            <div className="space-y-4 py-4">
   362→              <div className="space-y-2">
   363→                <label className="text-sm font-medium text-[#f0f0f0]">Policy Name</label>
   364→                <Input
   365→                  placeholder="e.g. Production Infrastructure"
   366→                  value={newName}
   367→                  onChange={(e) => setNewName(e.target.value)}
   368→                  className="bg-[#050710] border-[#21262d] text-[#f0f0f0] placeholder:text-[#333333] focus:border-[#00ff88]"
   369→                />
   370→              </div>
   371→              <div className="space-y-2">
   372→                <label className="text-sm font-medium text-[#f0f0f0]">Target Domain</label>
   373→                <Input
   374→                  placeholder="e.g. example.com"
   375→                  value={newDomain}
   376→                  onChange={(e) => setNewDomain(e.target.value)}
   377→                  className="bg-[#050710] border-[#21262d] text-[#f0f0f0] placeholder:text-[#333333] focus:border-[#00ff88]"
   378→                />
   379→              </div>
   380→              <div className="space-y-2">
   381→                <label className="text-sm font-medium text-[#f0f0f0]">Schedule</label>
   382→                <div className="grid grid-cols-4 gap-2">
   383→                  {(['hourly', 'daily', 'weekly', 'monthly'] as const).map((s) => (
   384→                    <button
   385→                      key={s}
   386→                      onClick={() => setNewSchedule(s)}
   387→                      className={`px-3 py-2 rounded-lg text-sm font-medium border transition-all capitalize ${
   388→                        newSchedule === s
   389→                          ? 'bg-[rgba(52,211,153,0.15)] border-[rgba(52,211,153,0.4)] text-[#00ff88]'
   390→                          : 'bg-[#050710] border-[#21262d] text-[#444444] hover:border-[#30363d]'
   391→                      }`}
   392→                    >
   393→                      {s}
   394→                    </button>
   395→                  ))}
   396→                </div>
   397→              </div>
   398→              <div className="space-y-2">
   399→                <label className="text-sm font-medium text-[#f0f0f0]">Scan Type</label>
   400→                <div className="grid grid-cols-2 gap-2">
   401→                  {['full', 'quick', 'stealth'].map((t) => (
   402→                    <button
   403→                      key={t}
   404→                      onClick={() => setNewScanType(t)}
   405→                      className={`px-3 py-2 rounded-lg text-sm font-medium border transition-all capitalize ${
   406→                        newScanType === t
   407→                          ? 'bg-[rgba(52,211,153,0.15)] border-[rgba(52,211,153,0.4)] text-[#00ff88]'
   408→                          : 'bg-[#050710] border-[#21262d] text-[#444444] hover:border-[#30363d]'
   409→                      }`}
   410→                    >
   411→                      {t}
   412→                    </button>
   413→                  ))}
   414→                </div>
   415→              </div>
   416→            </div>
   417→            <DialogFooter>
   418→              <Button variant="outline" onClick={() => setAddOpen(false)} className="border-[#21262d] text-[#444444] hover:bg-[#050710]">
   419→                Cancel
   420→              </Button>
   421→              <Button onClick={handleNewPolicy} className="bg-[#00ff88] hover:bg-[#00cc6a] text-[#0a0d14] font-semibold">
   422→                Create Policy
   423→              </Button>
   424→            </DialogFooter>
   425→          </DialogContent>
   426→        </Dialog>
   427→      </motion.div>
   428→
   429→      {/* ── Stats Row ───────────────────────────────────────────────────── */}
   430→      <motion.div variants={itemVariants} className="grid grid-cols-2 lg:grid-cols-4 gap-4">
   431→        {[
   432→          { label: 'Active Policies', value: activePolicies, icon: Shield, color: '#00ff88', subtitle: `of ${policies.length} total` },
   433→          { label: 'Monitored Assets', value: policies.length, icon: Globe, color: '#44aaff', subtitle: 'domains & endpoints' },
   434→          { label: 'Alerts Today', value: alerts.filter((a) => a.status === 'new').length, icon: AlertTriangle, color: '#ff8844', subtitle: `${totalFindings} findings` },
   435→          { label: 'Uptime', value: policies.length > 0 ? '99.97%' : '—', icon: TrendingUp, color: '#00ff88', subtitle: 'last 30 days' },
   436→        ].map((stat) => (
   437→          <motion.div
   438→            key={stat.label}
   439→            whileHover={cardHover}
   440→            className="rounded-xl border border-[#21262d] bg-[#080b14] p-4"
   441→          >
   442→            <div className="flex items-center justify-between mb-3">
   443→              <span className="text-xs font-medium text-[#444444] uppercase tracking-wider">{stat.label}</span>
   444→              <stat.icon className="w-4 h-4" style={{ color: stat.color }} />
   445→            </div>
   446→            <p className="text-2xl font-bold text-[#f0f0f0]">{stat.value}</p>
   447→            <p className="text-[10px] text-[#333333] mt-1">{stat.subtitle}</p>
   448→          </motion.div>
   449→        ))}
   450→      </motion.div>
   451→
   452→      {/* ── Policy Cards ────────────────────────────────────────────────── */}
   453→      <motion.div variants={itemVariants}>
   454→        <div className="flex items-center justify-between mb-3">
   455→          <h3 className="text-sm font-semibold text-[#f0f0f0]">Monitoring Policies</h3>
   456→          <span className="text-[10px] text-[#333333] uppercase tracking-wider">{policies.length} policies</span>
   457→        </div>
   458→        {policies.length === 0 ? (
   459→          <div className="rounded-xl border border-[#21262d] bg-[#080b14] p-8 text-center">
   460→            <Shield className="w-8 h-8 text-[#333333] mx-auto mb-3" />
   461→            <p className="text-sm text-[#444444]">No policies yet. Create one to start monitoring.</p>
   462→          </div>
   463→        ) : (
   464→          <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
   465→            {policies.map((policy, idx) => {
   466→              const ScanIcon = getScanTypeIcon(policy.scanType);
   467→              const statusCfg = getPolicyStatusConfig(policy.status);
   468→              return (
   469→                <motion.div
   470→                  key={policy.id}
   471→                  variants={itemVariants}
   472→                  whileHover={cardHover}
   473→                  className="rounded-xl border border-[#21262d] bg-[#080b14] overflow-hidden relative"
   474→                >
   475→                  {/* Status indicator bar */}
   476→                  <div className="absolute top-0 left-0 right-0 h-0.5 opacity-60" style={{ backgroundColor: statusCfg.color }} />
   477→
   478→                  <div className="p-5 pt-6">
   479→                    {/* Top row: name + toggle */}
   480→                    <div className="flex items-start justify-between mb-4">
   481→                      <div className="flex items-center gap-3">
   482→                        <div
   483→                          className="w-10 h-10 rounded-lg flex items-center justify-center"
   484→                          style={{ backgroundColor: statusCfg.color + '15' }}
   485→                        >
   486→                          <ScanIcon className="w-5 h-5" style={{ color: statusCfg.color }} />
   487→                        </div>
   488→                        <div>
   489→                          <h4 className="text-sm font-bold text-[#f0f0f0]">{policy.name}</h4>
   490→                          <div className="flex items-center gap-2 mt-0.5">
   491→                            <span className={`inline-flex items-center px-2 py-0.5 rounded text-[10px] font-medium border ${getScheduleColor(policy.schedule)}`}>
   492→                              <Timer className="w-2.5 h-2.5 mr-1" />
   493→                              {policy.schedule}
   494→                            </span>
   495→                            <span
   496→                              className="w-1.5 h-1.5 rounded-full"
   497→                              style={{ backgroundColor: statusCfg.color }}
   498→                            />
   499→                            <span className="text-[10px] text-[#444444]">{statusCfg.label}</span>
   500→                          </div>
   501→                        </div>
   502→                      </div>
   503→                      <div className="flex items-center gap-2">
   504→                        <button
   505→                          onClick={() => handleRunNow(policy.id)}
   506→                          className="p-1.5 rounded-md border border-[#21262d] text-[#444444] hover:text-[#00ff88] hover:border-[rgba(52,211,153,0.3)] hover:bg-[rgba(52,211,153,0.1)] transition-all"
   507→                          title="Run now"
   508→                        >
   509→                          <Play className="w-3.5 h-3.5" />
   510→                        </button>
   511→                        <button
   512→                          onClick={() => deletePolicy(policy.id)}
   513→                          className="p-1.5 rounded-md border border-[#21262d] text-[#444444] hover:text-[#ff3355] hover:border-[rgba(244,63,94,0.3)] hover:bg-[rgba(244,63,94,0.1)] transition-all"
   514→                          title="Delete"
   515→                        >
   516→                          <Trash2 className="w-3.5 h-3.5" />
   517→                        </button>
   518→                        <Switch
   519→                          checked={policy.enabled}

   525→
   526→                    {/* Target & scan info */}

   528→                      <div className="space-y-1">

   531→                          <Globe className="w-3 h-3 text-[#444444]" />
   532→                          <span className="font-mono text-[11px]">{policy.targetDomain}</span>
