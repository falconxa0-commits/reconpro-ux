    45→        if score >= threshold:
    46→            return grade
    47→    return "F"

    50→# ── A. Structured Logger ────────────────────────────────────────────────
    51→
    52→
    53→class StructuredLogger:
    54→    """JSON-formatted structured logger for ReconPro.
    55→
    56→    Features:
    57→    - Consistent JSON output format
    58→    - Automatic timestamp, trace_id, module fields
    59→    - Configurable log levels
    60→    - Output to stdout (default) or file
    61→    - Scan context propagation
    62→
    63→    Log format:
    64→    {
    65→        "ts": "2024-01-15T10:30:00.000Z",
    66→        "level": "INFO",
    67→        "module": "quantum_fingerprint",
    68→        "event": "scan_start",
    69→        "target": "example.com",
    70→        "trace_id": "abc123",
    71→        "duration_ms": 1234,
    72→        "details": {}
    73→    }
    74→    """
    75→
    76→    LEVEL_ORDER: Dict[str, int] = {
    77→        "DEBUG": 10,
    78→        "INFO": 20,
    79→        "WARNING": 30,
    80→        "ERROR": 40,
    81→        "CRITICAL": 50,
    82→    }
    83→
    84→    def __init__(
    85→        self,
    86→        module: str = "root",
    87→        enabled: bool = True,
    88→        min_level: str = "DEBUG",
    89→        output_file: Optional[str] = None,
    90→    ) -> None:
    91→        self._module = module
    92→        self._enabled = enabled
    93→        self._min_level = self.LEVEL_ORDER.get(min_level.upper(), 10)
    94→        self._output_file = output_file
    95→        self._file_handle: Optional[Any] = None
    96→        self._trace_id: Optional[str] = None
    97→        self._target: Optional[str] = None
    98→        self._lock = threading.Lock()
    99→
   100→        if output_file:
   101→            os.makedirs(os.path.dirname(output_file) or ".", exist_ok=True)
   102→            self._file_handle = open(output_file, "a", encoding="utf-8")
   103→
   104→    @property
   105→    def enabled(self) -> bool:
   106→        return self._enabled
   107→
   108→    @enabled.setter
   109→    def enabled(self, value: bool) -> None:
   110→        self._enabled = value
   111→
   112→    def set_trace_context(self, trace_id: Optional[str], target: Optional[str] = None) -> None:
   113→        """Set scan context for subsequent log entries."""
   114→        self._trace_id = trace_id
   115→        if target is not None:
   116→            self._target = target
   117→
   118→    def clear_trace_context(self) -> None:
   119→        """Clear scan context."""
   120→        self._trace_id = None
   121→        self._target = None
   122→
   123→    def _should_log(self, level: str) -> bool:
   124→        if not self._enabled:
   125→            return False
   126→        return self.LEVEL_ORDER.get(level, 0) >= self._min_level
   127→
   128→    def _emit(self, level: str, event: str, **kwargs: Any) -> None:
   129→        if not self._should_log(level):
   130→            return
   131→
   132→        entry: Dict[str, Any] = {
   133→            "ts": _utc_iso(),
   134→            "level": level,
   135→            "module": self._module,
   136→            "event": event,
   137→        }
   138→        if self._trace_id:
   139→            entry["trace_id"] = self._trace_id
   140→        if self._target:
   141→            entry["target"] = self._target
   142→        entry.update(kwargs)
   143→
   144→        line = json.dumps(entry, default=str, ensure_ascii=False)
   145→
   146→        with self._lock:
   147→            print(line, flush=True)
   148→            if self._file_handle:
   149→                self._file_handle.write(line + "\n")
   150→                self._file_handle.flush()
   151→
   152→    def debug(self, event: str, **kwargs: Any) -> None:
   153→        self._emit("DEBUG", event, **kwargs)
   154→
   155→    def info(self, event: str, **kwargs: Any) -> None:
   156→        self._emit("INFO", event, **kwargs)
   157→
   158→    def warning(self, event: str, **kwargs: Any) -> None:
   159→        self._emit("WARNING", event, **kwargs)
   160→
   161→    def error(self, event: str, **kwargs: Any) -> None:
   162→        self._emit("ERROR", event, **kwargs)
   163→
   164→    def critical(self, event: str, **kwargs: Any) -> None:
   165→        self._emit("CRITICAL", event, **kwargs)
   166→
   167→    def log_exception(self, event: str, exc: Optional[BaseException] = None, **kwargs: Any) -> None:
   168→        """Log an ERROR event with full traceback."""
   169→        tb = traceback.format_exception(type(exc), exc, exc.__traceback__) if exc else traceback.format_stack()
   170→        self._emit("ERROR", event, traceback="".join(tb).strip(), **kwargs)
   171→
   172→    def child(self, module: str) -> "StructuredLogger":
   173→        """Create a child logger that inherits trace context."""
   174→        child = StructuredLogger(
   175→            module=module,
   176→            enabled=self._enabled,
   177→            min_level=[k for k, v in self.LEVEL_ORDER.items() if v == self._min_level][0],
   178→            output_file=self._output_file,
   179→        )
   180→        child._trace_id = self._trace_id
   181→        child._target = self._target
   182→        return child
   183→
   184→    def close(self) -> None:
   185→        """Close file handle if open."""
   186→        if self._file_handle:
   187→            self._file_handle.close()
   188→            self._file_handle = None
   189→
   190→    def __del__(self) -> None:
   191→        self.close()

   197→class TimerContext:
   198→    """Context manager for timing a block of code."""
   199→
   200→    def __init__(self, collector: "MetricsCollector", name: str) -> None:
   201→        self._collector = collector
   202→        self._name = name
   203→        self._start: Optional[float] = None
   204→        self._elapsed: float = 0.0
   205→
   206→    def __enter__(self) -> "TimerContext":
   207→        self._start = time.perf_counter()
   208→        return self
   209→
   210→    def __exit__(self, *args: Any) -> None:
   211→        if self._start is not None:
   212→            self._elapsed = (time.perf_counter() - self._start) * 1000.0
   213→            self._collector.histogram_observe(f"{self._name}_duration_ms", self._elapsed)
   214→
   215→    @property
   216→    def elapsed_ms(self) -> float:
   217→        return self._elapsed

   219→
   220→class MetricsCollector:
   221→    """Thread-safe metrics collection.
   222→
   223→    Metrics:
   224→    - counters: monotonically increasing values (requests, findings, errors)
   225→    - gauges: point-in-time values (memory, connections, queue_depth)
   226→    - histograms: distribution tracking (latency, response_size)
   227→    - timers: named timers with start/stop
   228→
   229→    All methods are O(1) and thread-safe.
   230→    Minimal overhead: disabled metrics cost ~50ns per call.
   231→    """
   232→
   233→    def __init__(self, enabled: bool = True) -> None:
   234→        self._enabled = enabled
   235→        self._lock = threading.Lock()
   236→        self._counters: Dict[str, float] = {}
   237→        self._gauges: Dict[str, float] = {}
   238→        self._histograms: Dict[str, List[float]] = {}
   239→        self._active_timers: Dict[str, float] = {}
   240→
   241→    @property
   242→    def enabled(self) -> bool:
   243→        return self._enabled
   244→
   245→    @enabled.setter
   246→    def enabled(self, value: bool) -> None:
   247→        self._enabled = value
   248→
   249→    def counter_increment(self, name: str, amount: float = 1) -> None:
   250→        """Increment a counter by amount (default 1)."""
   251→        if not self._enabled:
   252→            return
   253→        with self._lock:
   254→            self._counters[name] = self._counters.get(name, 0) + amount
   255→
   256→    def gauge_set(self, name: str, value: float) -> None:
   257→        """Set a gauge to a specific value."""
   258→        if not self._enabled:
   259→            return
   260→        with self._lock:
   261→            self._gauges[name] = value
   262→
   263→    def gauge_increment(self, name: str, amount: float = 1) -> None:
   264→        """Increment (or decrement) a gauge."""
   265→        if not self._enabled:
   266→            return
   267→        with self._lock:
   268→            self._gauges[name] = self._gauges.get(name, 0) + amount
   269→
   270→    def histogram_observe(self, name: str, value: float) -> None:
   271→        """Record a value in a histogram."""
   272→        if not self._enabled:
   273→            return
   274→        with self._lock:
   275→            if name not in self._histograms:
   276→                self._histograms[name] = []
   277→            self._histograms[name].append(value)
   278→
   279→    def timer_start(self, name: str) -> str:
   280→        """Start a named timer. Returns the timer name."""
   281→        if not self._enabled:
   282→            return name
   283→        with self._lock:
   284→            self._active_timers[name] = time.perf_counter()
   285→        return name
   286→
   287→    def timer_stop(self, name: str) -> Optional[float]:
   288→        """Stop a named timer and record elapsed ms in histogram. Returns elapsed ms or None."""
   289→        if not self._enabled:
   290→            return None
   291→        with self._lock:
   292→            start = self._active_timers.pop(name, None)
   293→        if start is None:
   294→            return None
   295→        elapsed_ms = (time.perf_counter() - start) * 1000.0
   296→        self.histogram_observe(f"{name}_duration_ms", elapsed_ms)
   297→        return elapsed_ms
   298→
   299→    def timer_context(self, name: str) -> TimerContext:
   300→        """Return a TimerContext for use as a context manager."""
   301→        return TimerContext(self, name)
   302→
   303→    def get_snapshot(self) -> Dict[str, Any]:
   304→        """Return all current metrics as a dictionary."""
   305→        with self._lock:
   306→            snapshot: Dict[str, Any] = {
   307→                "counters": dict(self._counters),
   308→                "gauges": dict(self._gauges),
   309→            }
   310→            hist_summary: Dict[str, Dict[str, Any]] = {}
   311→            for name, values in self._histograms.items():
   312→                if values:
   313→                    sorted_vals = sorted(values)
   314→                    n = len(sorted_vals)
   315→                    hist_summary[name] = {
   316→                        "count": n,
   317→                        "min": sorted_vals[0],
   318→                        "max": sorted_vals[-1],
   319→                        "avg": sum(sorted_vals) / n,
   320→                        "p50": sorted_vals[n // 2],
   321→                        "p95": sorted_vals[int(n * 0.95)] if n >= 20 else sorted_vals[-1],
   322→                        "p99": sorted_vals[int(n * 0.99)] if n >= 100 else sorted_vals[-1],
   323→                    }
   324→                else:
   325→                    hist_summary[name] = {"count": 0}
   326→            snapshot["histograms"] = hist_summary
   327→            snapshot["active_timers"] = len(self._active_timers)
   328→        return snapshot
   329→
   330→    def reset(self) -> None:
   331→        """Clear all metrics."""
   332→        with self._lock:
   333→            self._counters.clear()
   334→            self._gauges.clear()
   335→            self._histograms.clear()
   336→            self._active_timers.clear()
   337→
   338→    def get_counter(self, name: str) -> float:
   339→        """Get current counter value."""
   340→        with self._lock:
   341→            return self._counters.get(name, 0)
   342→
   343→    def get_gauge(self, name: str) -> float:
   344→        """Get current gauge value."""
   345→        with self._lock:
   346→            return self._gauges.get(name, 0)
   347→

   352→@dataclass
   353→class ModuleTrace:
   354→    """Timing and result data for a single module within a scan."""
   355→    id: str
   356→    duration_ms: float = 0.0
   357→    findings: int = 0
   358→    status: str = "pending"  # pending | running | ok | error | skipped
   359→    error: Optional[str] = None
   360→    started_at: Optional[str] = None
   361→    completed_at: Optional[str] = None
   362→    details: Dict[str, Any] = field(default_factory=dict)
   363→
   364→    def to_dict(self) -> Dict[str, Any]:
   365→        return {
   366→            "id": self.id,
   367→            "duration_ms": round(self.duration_ms, 2),
   368→            "findings": self.findings,
   369→            "status": self.status,
   370→            "error": self.error,
   371→            "started_at": self.started_at,
   372→            "completed_at": self.completed_at,
   373→        }
   374→
   375→
   376→class ScanTracer:
   377→    """Trace a scan execution with detailed timing.
   378→
   379→    Produces an execution receipt:
   380→    {
   381→        "trace_id": "...",
   382→        "target": "example.com",
   383→        "started_at": "...",
   384→        "completed_at": "...",
   385→        "total_duration_ms": 12345,
   386→        "modules": [
   387→            {"id": "recon", "duration_ms": 1234, "findings": 5, "status": "ok"},
   388→            {"id": "auth", "duration_ms": 567, "findings": 2, "status": "ok"},
   389→        ],
   390→        "errors": [],
   391→        "score": 85,
   392→        "grade": "B"
   393→    }
   394→    """
   395→
   396→    def __init__(self, target: str, enabled: bool = True) -> None:
   397→        self._enabled = enabled
   398→        self.trace_id: str = _generate_trace_id()
   399→        self.target: str = target
   400→        self.started_at: Optional[str] = None
   401→        self.completed_at: Optional[str] = None
   402→        self._modules: OrderedDict[str, ModuleTrace] = OrderedDict()
   403→        self._errors: List[str] = []
   404→        self._score: Optional[int] = None
   405→        self._lock = threading.Lock()
   406→        self._start_perf: float = 0.0
   407→
   408→    @property
   409→    def enabled(self) -> bool:
   410→        return self._enabled
   411→
   412→    @enabled.setter
   413→    def enabled(self, value: bool) -> None:
   414→        self._enabled = value
   415→
   416→    def start_scan(self) -> None:
   417→        """Mark the scan as started."""
   418→        self.started_at = _utc_iso()
   419→        self._start_perf = time.perf_counter()
   420→
   421→    def end_scan(self, score: Optional[int] = None) -> Dict[str, Any]:
   422→        """Mark the scan as completed and return the execution receipt."""
   423→        self.completed_at = _utc_iso()
   424→        total_ms = (time.perf_counter() - self._start_perf) * 1000.0 if self._start_perf else 0.0
   425→        if score is not None:
   426→            self._score = max(MIN_SCORE, min(MAX_SCORE, score))
   427→
   428→        receipt: Dict[str, Any] = {
   429→            "trace_id": self.trace_id,
   430→            "target": self.target,
   431→            "started_at": self.started_at,
   432→            "completed_at": self.completed_at,
   433→            "total_duration_ms": round(total_ms, 2),
   434→            "modules": [m.to_dict() for m in self._modules.values()],
   435→            "errors": list(self._errors),
   436→        }
   437→        if self._score is not None:
   438→            receipt["score"] = self._score
   439→            receipt["grade"] = _score_to_grade(self._score)
   440→        return receipt
   441→
   442→    def start_module(self, module_id: str) -> None:
   443→        """Begin timing a module."""
   444→        if not self._enabled:
   445→            return
   446→        with self._lock:
   447→            mt = ModuleTrace(
   448→                id=module_id,
   449→                status="running",
   450→                started_at=_utc_iso(),
   451→            )
   452→            self._modules[module_id] = mt
   453→
   454→    def end_module(self, module_id: str, findings: int = 0, status: str = "ok", error: Optional[str] = None) -> None:
   455→        """End timing a module and record results."""
   456→        if not self._enabled:
   457→            return
   458→        with self._lock:
   459→            mt = self._modules.get(module_id)
   460→            if mt is None:
   461→                return
   462→            mt.completed_at = _utc_iso()
   463→            if mt.started_at:
   464→                try:
   465→                    started = datetime.fromisoformat(mt.started_at.replace("Z", "+00:00"))
   466→                    ended = datetime.fromisoformat(mt.completed_at.replace("Z", "+00:00"))
   467→                    mt.duration_ms = (ended - started).total_seconds() * 1000.0
   468→                except (ValueError, OSError):
   469→                    mt.duration_ms = 0.0
   470→            mt.findings = findings
   471→            mt.status = status
   472→            mt.error = error
   473→            if error:
   474→                self._errors.append(f"{module_id}: {error}")
   475→
   476→    def add_error(self, message: str) -> None:
   477→        """Add an error to the scan trace."""
   478→        self._errors.append(message)
   479→
   480→    def set_module_detail(self, module_id: str, key: str, value: Any) -> None:
   481→        """Set an arbitrary detail on a module trace."""
   482→        if not self._enabled:
   483→            return
   484→        with self._lock:
   485→            mt = self._modules.get(module_id)
   486→            if mt:
   487→                mt.details[key] = value
   488→
   489→    def get_receipt(self) -> Dict[str, Any]:
   490→        """Return the current execution receipt without ending the scan."""
   491→        total_ms = 0.0
   492→        if self._start_perf and self.started_at:
   493→            total_ms = (time.perf_counter() - self._start_perf) * 1000.0
   494→
   495→        receipt: Dict[str, Any] = {
   496→            "trace_id": self.trace_id,
   497→            "target": self.target,
   498→            "started_at": self.started_at,
   499→            "total_duration_ms": round(total_ms, 2),
   500→            "modules": [m.to_dict() for m in self._modules.values()],
   501→            "errors": list(self._errors),
   502→        }
   503→        if self._score is not None:
   504→            receipt["score"] = self._score
   505→            receipt["grade"] = _score_to_grade(self._score)
   506→        return receipt
   507→
   508→    def reset(self) -> None:
   509→        """Reset tracer for reuse."""
   510→        self.trace_id = _generate_trace_id()
   511→        self.started_at = None
   512→        self.completed_at = None
   513→        self._modules.clear()
   514→        self._errors.clear()
   515→        self._score = None
   516→        self._start_perf = 0.0

   523→class ModuleProfile:
   524→    """Performance profile for a single module."""
   525→    module_id: str
   526→    wall_time_ms: float = 0.0
   527→    http_requests: int = 0
   528→    bytes_sent: int = 0
   529→    bytes_received: int = 0
   530→    findings: int = 0
   531→    status: str = "ok"
   532→
   533→    def to_dict(self) -> Dict[str, Any]:

   535→            "module_id": self.module_id,

   538→            "bytes_sent": self.bytes_sent,

   540→            "findings": self.findings,

   544→
   545→class PerformanceProfiler:
   546→    """Per-module performance profiling.
   547→
   548→    Track per-module:
   549→    - Wall clock time
   550→    - Number of HTTP requests made
   551→    - Bytes sent/received
   552→    - Findings produced
   553→
   554→    Generates performance report comparing modules.
   555→    """
   556→
   557→    def __init__(self, enabled: bool = True) -> None:
   558→        self._enabled = enabled
   559→        self._profiles: OrderedDict[str, ModuleProfile] = OrderedDict()
   560→        self._start_times: Dict[str, float] = {}
   561→        self._lock = threading.Lock()
   562→
   563→    @property
   564→    def enabled(self) -> bool:
   565→        return self._enabled
   566→
   567→    @enabled.setter
   568→    def enabled(self, value: bool) -> None:
   569→        self._enabled = value
   570→
   571→    def start_module(self, module_id: str) -> None:
   572→        """Begin profiling a module."""
   573→        if not self._enabled:
   574→            return
   575→        with self._lock:
   576→            self._start_times[module_id] = time.perf_counter()
   577→            self._profiles[module_id] = ModuleProfile(module_id=module_id)
   578→
   579→    def end_module(self, module_id: str, status: str = "ok") -> None:
   580→        """End profiling a module."""
   581→        if not self._enabled:
   582→            return
   583→        with self._lock:
   584→            start = self._start_times.pop(module_id, None)
   585→            profile = self._profiles.get(module_id)
   586→            if profile and start is not None:
   587→                profile.wall_time_ms = (time.perf_counter() - start) * 1000.0
   588→                profile.status = status
   589→
   590→    def record_http_request(self, module_id: str, bytes_sent: int = 0, bytes_received: int = 0) -> None:
   591→        """Record an HTTP request for a module."""
   592→        if not self._enabled:
   593→            return
   594→        with self._lock:
   595→            profile = self._profiles.get(module_id)
   596→            if profile:
   597→                profile.http_requests += 1
   598→                profile.bytes_sent += bytes_sent
   599→                profile.bytes_received += bytes_received
   600→
   601→    def record_findings(self, module_id: str, count: int = 1) -> None:
   602→        """Record findings produced by a module."""
   603→        if not self._enabled:
   604→            return
   605→        with self._lock:
   606→            profile = self._profiles.get(module_id)
   607→            if profile:
   608→                profile.findings += count
   609→
   610→    def get_report(self) -> Dict[str, Any]:
   611→        """Generate a performance report comparing all profiled modules."""
   612→        with self._lock:
   613→            profiles = [p.to_dict() for p in self._profiles.values()]
   614→
   615→        total_wall_ms = sum(p["wall_time_ms"] for p in profiles)
   616→        total_requests = sum(p["http_requests"] for p in profiles)
   617→        total_bytes_sent = sum(p["bytes_sent"] for p in profiles)
   618→        total_bytes_recv = sum(p["bytes_received"] for p in profiles)
   619→        total_findings = sum(p["findings"] for p in profiles)
   620→
   621→        # Sort by wall time descending for ranking
   622→        ranked = sorted(profiles, key=lambda p: p["wall_time_ms"], reverse=True)
   623→
   624→        report: Dict[str, Any] = {
   625→            "total_wall_time_ms": round(total_wall_ms, 2),
   626→            "total_http_requests": total_requests,
   627→            "total_bytes_sent": total_bytes_sent,
   628→            "total_bytes_received": total_bytes_recv,
   629→            "total_findings": total_findings,
   630→            "modules": ranked,
   631→            "slowest_module": ranked[0]["module_id"] if ranked else None,
   632→            "fastest_module": ranked[-1]["module_id"] if ranked else None,
   633→        }
   634→        return report
   635→
   636→    def get_module_profile(self, module_id: str) -> Optional[Dict[str, Any]]:
   637→        """Get the profile for a specific module."""
   638→        with self._lock:
   639→            profile = self._profiles.get(module_id)
   640→            return profile.to_dict() if profile else None
   641→
   642→    def reset(self) -> None:
   643→        """Clear all profiles."""
   644→        with self._lock:
   645→            self._profiles.clear()
   646→            self._start_times.clear()

   649→# ── E. Health Monitor ──────────────────────────────────────────────────
   650→
   651→
   652→class HealthMonitor:
   653→    """System health monitoring.
   654→
   655→    Checks:
   656→    - Memory usage (via resource module or /proc/self/status)
   657→    - Disk usage for ~/.reconpro/
   658→    - Network connectivity
   659→    - SSL certificate expiry
   660→
   661→    Returns health report with status: healthy/degraded/critical.
   662→    """
   663→
   664→    def __init__(self, enabled: bool = True) -> None:
   665→        self._enabled = enabled
   666→
   667→    @property
   668→    def enabled(self) -> bool:
   669→        return self._enabled
   670→
   671→    @enabled.setter
   672→    def enabled(self, value: bool) -> None:
   673→        self._enabled = value
   674→
   675→    def _get_memory_usage(self) -> Dict[str, Any]:
   676→        """Get memory usage information."""
   677→        result: Dict[str, Any] = {"status": "unknown", "rss_mb": 0, "method": "none"}
   678→        try:
   679→            # Try /proc/self/status first (Linux)
   680→            if os.path.exists("/proc/self/status"):
   681→                with open("/proc/self/status", "r") as f:
   682→                    for line in f:
   683→                        if line.startswith("VmRSS:"):
   684→                            kb = int(line.split()[1])
   685→                            result["rss_mb"] = round(kb / 1024, 1)
   686→                            result["method"] = "proc"
   687→                            break
   688→                # Also get VmSize for context
   689→                with open("/proc/self/status", "r") as f:
   690→                    for line in f:
   691→                        if line.startswith("VmSize:"):
   692→                            kb = int(line.split()[1])
   693→                            result["vm_size_mb"] = round(kb / 1024, 1)
   694→                            break
   695→            else:
   696→                # Fallback: try resource module
   697→                import resource
   698→                rss = resource.getrusage(resource.RUSAGE_SELF).ru_maxrss
   699→                # On Linux, ru_maxrss is in KB; on macOS, in bytes
   700→                if os.uname().sysname == "Darwin":
   701→                    result["rss_mb"] = round(rss / (1024 * 1024), 1)
   702→                else:
   703→                    result["rss_mb"] = round(rss / 1024, 1)
   704→                result["method"] = "resource"
   705→        except Exception:
   706→            pass
   707→        return result
   708→
   709→    def _get_disk_usage(self) -> Dict[str, Any]:
   710→        """Get disk usage for ~/.reconpro/."""
   711→        result: Dict[str, Any] = {
   712→            "path": str(RECONPRO_HOME),
   713→            "total_mb": 0,
   714→            "used_mb": 0,
   715→            "free_mb": 0,
   716→            "usage_percent": 0,
   717→            "status": "unknown",
   718→        }
   719→        try:
   720→            usage = shutil.disk_usage(str(RECONPRO_HOME))
   721→            total_mb = round(usage.total / (1024 * 1024), 1)
   722→            used_mb = round(usage.used / (1024 * 1024), 1)
   723→            free_mb = round(usage.free / (1024 * 1024), 1)
   724→            percent = round(usage.used / usage.total * 100, 1) if usage.total else 0
   725→            result["total_mb"] = total_mb
   726→            result["used_mb"] = used_mb
   727→            result["free_mb"] = free_mb
   728→            result["usage_percent"] = percent
   729→            if percent > 90:
   730→                result["status"] = "critical"
   731→            elif percent > 75:
   732→                result["status"] = "degraded"
   733→            else:
   734→                result["status"] = "healthy"
   735→        except Exception:
   736→            pass
   737→        return result
   738→
   739→    def _get_network_connectivity(self) -> Dict[str, Any]:
   740→        """Check basic network connectivity via DNS resolution."""
   741→        result: Dict[str, Any] = {"status": "unknown", "dns_resolve_ms": 0}
   742→        try:
   743→            start = time.perf_counter()
   744→            socket.getaddrinfo("example.com", 443, socket.AF_INET, socket.SOCK_STREAM)
   745→            elapsed = (time.perf_counter() - start) * 1000.0
   746→            result["dns_resolve_ms"] = round(elapsed, 1)
   747→            result["status"] = "healthy" if elapsed < 2000 else "degraded"
   748→        except socket.gaierror:
   749→            result["status"] = "critical"
   750→        except Exception:
   751→            result["status"] = "unknown"
   752→        return result
   753→
   754→    def _get_ssl_check(self) -> Dict[str, Any]:
   755→        """Check SSL certificate for a known site."""
   756→        result: Dict[str, Any] = {
   757→            "host": "example.com",
   758→            "status": "unknown",
   759→            "days_until_expiry": None,
   760→        }
   761→        try:
   762→            context = ssl.create_default_context()
   763→            conn = context.wrap_socket(socket.socket(socket.AF_INET), server_hostname="example.com")
   764→            conn.settimeout(3)
   765→            conn.connect(("example.com", 443))
   766→            cert = conn.getpeercert()
   767→            conn.close()
   768→
   769→            # Parse notAfter
   770→            import email.utils
   771→            for field in cert.get("notAfter", ""):
   772→                pass  # no-op
   773→            not_after_str = cert.get("notAfter", "")
   774→            if not_after_str:
   775→                not_after = email.utils.parsedate_to_datetime(not_after_str)
   776→                now = datetime.now(timezone.utc)
   777→                days = (not_after - now).days
   778→                result["days_until_expiry"] = days
   779→                if days <= 0:
   780→                    result["status"] = "critical"
   781→                elif days <= 30:
   782→                    result["status"] = "degraded"
   783→                else:
   784→                    result["status"] = "healthy"
   785→        except Exception:
   786→            pass
   787→        return result
   788→
   789→    def check_health(self) -> Dict[str, Any]:
   790→        """Run all health checks and return a comprehensive report."""
   791→        if not self._enabled:
   792→            return {"status": "disabled", "checks": {}}
   793→
   794→        memory = self._get_memory_usage()
   795→        disk = self._get_disk_usage()
   796→        network = self._get_network_connectivity()
   797→        ssl = self._get_ssl_check()
   798→
   799→        checks = {
   800→            "memory": memory,
   801→            "disk": disk,
   802→            "network": network,
   803→            "ssl": ssl,
   804→        }
   805→
   806→        # Determine overall status
   807→        statuses = []
   808→        for check in checks.values():
   809→            s = check.get("status", "unknown")
   810→            if s == "critical":
   811→                statuses.append(0)
   812→            elif s == "degraded":
   813→                statuses.append(1)
   814→            elif s == "healthy":
   815→                statuses.append(2)
   816→            else:
   817→                statuses.append(1)  # unknown -> degraded
   818→
   819→        worst = min(statuses)
   820→        overall = ["critical", "degraded", "healthy"][worst]
   821→
   822→        return {
   823→            "status": overall,
   824→            "timestamp": _utc_iso(),
   825→            "checks": checks,
   826→        }
   827→
   828→    def check_memory(self) -> Dict[str, Any]:
   829→        """Run only the memory check."""
   830→        if not self._enabled:
   831→            return {"status": "disabled"}
   832→        return self._get_memory_usage()
   833→
   834→    def check_disk(self) -> Dict[str, Any]:
   835→        """Run only the disk check."""
   836→        if not self._enabled:
   837→            return {"status": "disabled"}
   838→        return self._get_disk_usage()
   839→
   840→    def check_network(self) -> Dict[str, Any]:
   841→        """Run only the network check."""
   842→        if not self._enabled:
   843→            return {"status": "disabled"}
   844→        return self._get_network_connectivity()

   849→
   850→class TelemetryManager:
   851→    """Central telemetry management.
   852→
   853→    Controls all observability features:
   854→    - Enable/disable structured logging
   855→    - Enable/disable metrics collection
   856→    - Enable/disable tracing
   857→    - Set output destination
   858→    - Get health report
   859→    - Get performance snapshot
   860→    """
   861→
   862→    _instance: Optional["TelemetryManager"] = None
   863→    _init_lock = threading.Lock()
   864→
   865→    def __init__(
   866→        self,
   867→        logging_enabled: bool = True,
   868→        metrics_enabled: bool = True,
   869→        tracing_enabled: bool = True,
   870→        profiling_enabled: bool = True,
   871→        health_enabled: bool = True,
   872→        output_file: Optional[str] = None,
   873→    ) -> None:
   874→        self._logging_enabled = logging_enabled
   875→        self._metrics_enabled = metrics_enabled
   876→        self._tracing_enabled = tracing_enabled
   877→        self._profiling_enabled = profiling_enabled
   878→        self._health_enabled = health_enabled
   879→        self._output_file = output_file
   880→
   881→        self._logger = StructuredLogger(
   882→            module="telemetry",
   883→            enabled=logging_enabled,
   884→            output_file=output_file,
   885→        )
   886→        self._metrics = MetricsCollector(enabled=metrics_enabled)
   887→        self._health = HealthMonitor(enabled=health_enabled)
   888→
   889→    @classmethod
   890→    def get_instance(cls) -> "TelemetryManager":
   891→        """Get or create the singleton TelemetryManager."""
   892→        if cls._instance is None:
   893→            with cls._init_lock:
   894→                if cls._instance is None:
   895→                    cls._instance = cls()
   896→        return cls._instance
   897→
   898→    @classmethod
   899→    def reset_instance(cls) -> None:
   900→        """Reset the singleton (for testing)."""
   901→        with cls._init_lock:
   902→            cls._instance = None
