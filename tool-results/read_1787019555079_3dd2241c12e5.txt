    26→def clean_mkdir(p: Path):
    27→    if p.exists():
    28→        shutil.rmtree(p)
    29→    p.mkdir(parents=True, exist_ok=True)
    30→    stats["folders"] += 1

    33→def copy(src: Path, dst: Path, required=True):
    34→    if not src.exists():
    35→        if required:
    36→            stats["missing"].append(str(src))
    37→        return False
    38→    dst.parent.mkdir(parents=True, exist_ok=True)
    39→    if not dst.parent.exists():
    40→        stats["folders"] += 1
    41→    shutil.copy2(src, dst)
    42→    stats["files"] += 1
    43→    return True
    44→
    45→
    46→def copy_tree(src: Path, dst: Path, exclude=None):
    47→    if not src.exists():
    48→        stats["missing"].append(str(src))
    49→        return 0
    50→    exclude = exclude or []
    51→    count = 0
    52→    dst.mkdir(parents=True, exist_ok=True)
    53→    stats["folders"] += 1
    54→    for item in sorted(src.rglob("*")):
    55→        # Skip excluded patterns and hidden files (except .env, .example)
    56→        skip = False
    57→        for pat in exclude:
    58→            if pat in str(item):
    59→                skip = True
    60→                break
    61→        if skip:
    62→            continue
    63→        # Skip hidden dirs/files except specific ones
    64→        parts = item.relative_to(src).parts
    65→        for p in parts:
    66→            if p.startswith(".") and p not in (".env", ".env.example", ".env.local", ".gitignore", ".dockerignore"):
    67→                skip = True
    68→                break
    69→        if skip:
    70→            continue
    71→        if item.is_file():
    72→            target = dst / item.relative_to(src)
    73→            target.parent.mkdir(parents=True, exist_ok=True)
    74→            shutil.copy2(item, target)
    75→            count += 1
    76→            stats["files"] += 1
    77→    return count

    88→def sha256(p: Path) -> str:
    89→    h = hashlib.sha256()
    90→    with open(p, "rb") as f:
    91→        for chunk in iter(lambda: f.read(65536), b""):
    92→            h.update(chunk)
    93→    return h.hexdigest()
