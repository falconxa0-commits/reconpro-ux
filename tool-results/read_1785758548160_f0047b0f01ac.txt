   281→  1278→    @work(exclusive=False, thread=True)
   282→  1279→    def _run_scan_worker(
   283→  1280→        self,
   284→  1281→        target: str,
   285→  1282→        modules: Optional[List[str]] = None,
   286→  1283→        is_local: bool = False,
   287→  1284→    ) -> None:
   288→  1285→        """Run a scan in a background thread."""
   289→  1286→        self.call_from_thread(self.is_scanning.set, True)
   290→  1287→
   291→  1288→        # Set modules to scanning
   292→  1289→        if modules:
   293→  1290→            for m in modules:
   294→  1291→                if m in self._module_cells:
   295→  1292→                    self.call_from_thread(self._module_cells[m].set_status, "scanning")
   296→  1293→        else:
   297→  1294→            all_mods = (
   298→  1295→                list(MODULE_NAMES.keys())
   299→  1296→                if is_local
   300→  1297→                else [k for k in MODULE_NAMES if k not in ("host", "dev", "doctor")]
   301→  1298→            )
   302→  1299→            for m in all_mods:
   303→  1300→                if m in self._module_cells:
   304→  1301→                    self.call_from_thread(self._module_cells[m].set_status, "scanning")
   305→  1302→
   306→  1303→        try:
   307→  1304→            from .scanner import scan, audit_scan
   308→  1305→            from .history import save_scan
   309→  1306→
   310→  1307→            self.call_from_thread(
   311→  1308→                self._chat, f"[{DIM_CYAN}]  ⏳ Scan in progress...[/]"
   312→  1309→            )
   313→  1310→
   314→  1311→            if is_local:
   315→  1312→                result = audit_scan(target=target, modules=modules)
   316→  1313→            else:
   317→  1314→                result = scan(target=target, modules=modules)
   318→  1315→
   319→  1316→            # Process findings
   320→  1317→            data = result.to_dict()
   321→  1318→            self.call_from_thread(self._on_scan_complete, data, modules, is_local)
   322→  1319→
   323→  1320→            # Save to history
   324→  1321→            try:
   325→  1322→                save_scan(data, label="nexus")
   326→  1323→            except Exception:
   327→  1324→                pass
   328→  1325→
   329→  1326→        except Exception as e:
   330→  1327→            self.call_from_thread(self._chat, f"[{RED}]  ✗ Scan error: {e}[/]")
   331→  1328→            if modules:
   332→  1329→                for m in modules:
   333→  1330→                    if m in self._module_cells:
   334→  1331→                        self.call_from_thread(self._module_cells[m].set_status, "error")
   335→  1332→            else:
   336→  1333→                self.call_from_thread(self._set_all_modules_status, "error")
   337→  1334→
   338→  1335→        finally:
   339→  1336→            self.call_from_thread(self.is_scanning.set, False)

   692→  1689→    def on_click(self, event: Click) -> None:
   693→  1690→        """Open finding detail modal for the clicked finding.
   694→  1691→
   695→  1692→        Uses the vertical click position to determine which finding index
   696→  1693→        the user clicked on, mapping it to _findings_list.
   697→  1694→        """
   698→  1695→        try:
   699→  1696→            if (
   700→  1697→                isinstance(event.widget, RichLog)
   701→  1698→                and event.widget.id == "findings-feed"
   702→  1699→                and self._findings_list
   703→  1700→            ):
   704→  1701→                # Get the line offset to determine which finding was clicked.
   705→  1702→                # Each finding entry is 1 line in the RichLog.
   706→  1703→                # We use the y coordinate relative to the widget to estimate the index.
   707→  1704→                region = event.widget.region
   708→  1705→                if region is not None:
   709→  1706→                    # Calculate which line was clicked relative to the scroll position
   710→  1707→                    scroll_offset = event.widget.scroll_y
   711→  1708→                    click_y = event.y - region.y
   712→  1709→                    # Each finding takes 1 line; estimate index from click position
   713→  1710→                    # Account for scroll position
   714→  1711→                    line_height = 1
   715→  1712→                    if hasattr(event.widget, "_line_height"):
   716→  1713→                        line_height = event.widget._line_height
   717→  1714→                    clicked_index = int((click_y + scroll_offset * line_height) / line_height)
   718→  1715→                    # Clamp to valid range
   719→  1716→                    clicked_index = max(0, min(clicked_index, len(self._findings_list) - 1))
   720→  1717→                    finding = self._findings_list[clicked_index]
   721→  1718→                else:
   722→  1719→                    finding = self._findings_list[-1]
   723→  1720→                self.push_screen(FindingDetailModal(finding))
   724→  1721→        except (NoMatches, Exception):
   725→  1722→            pass
