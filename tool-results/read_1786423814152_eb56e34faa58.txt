     1→"""ReconPro Age III — Auto Fix Proposal Engine.
     2→
     3→Analyzes detected issues and proposes automated fixes.  Every fix is a
     4→*proposal* that must be explicitly approved before it is applied.
     5→Dry-run mode is the default — nothing is ever written to disk without
     6→the caller's consent.
     7→
     8→Classes:
     9→    FixProposal      — Dataclass for a single fix proposal.
    10→    FixAnalyzer      — Generates fix proposals from issue patterns.
    11→    FixStore         — Persistent JSON-backed proposal storage.
    12→    AutoFixEngine    — Top-level orchestrator.
    13→
    14→Lifecycle of a proposal:
    15→    proposed → approved → applied → verified  (or  rejected / failed)
    16→"""

    71→# FixProposal dataclass
    72→# ═══════════════════════════════════════════════════════════════════════════
    73→
    74→
    75→@dataclass
    76→class FixProposal:
    77→    """A single proposed fix for a detected issue.
    78→
    79→    Attributes:
    80→        id:               Unique identifier (UUID4-based).
    81→        title:            Short human-readable title.
    82→        description:      Detailed explanation of the fix.
    83→        severity:         Severity of the *underlying issue* (critical/high/…).
    84→        category:         FixCategory enum value.
    85→        affected_file:    Absolute path to the file (or "").
    86→        affected_lines:   1-based line numbers impacted.
    87→        original_code:    Original source lines (may be empty for config fixes).
    88→        proposed_code:    Proposed replacement lines.
    89→        risk_level:       Risk of applying the fix.
    90→        confidence:       0.0–1.0 confidence that the fix is correct.
    91→        automated:        True if the fix can be applied without human review.
    92→        verification_steps: List of steps to confirm the fix worked.
    93→        rollback_plan:    How to undo the fix.
    94→        status:           Current ProposalStatus.
    95→        source_issue:     The original issue dict that triggered this proposal.
    96→        created_at:       ISO-8601 timestamp.
    97→        applied_at:       ISO-8601 timestamp (empty until applied).
    98→        verified_at:      ISO-8601 timestamp (empty until verified).
    99→        error_message:    If status is FAILED, the error text.
   100→    """
   101→    id: str = field(default_factory=lambda: uuid.uuid4().hex[:12])
   102→    title: str = ""
   103→    description: str = ""
   104→    severity: str = "medium"
   105→    category: str = FixCategory.CODE_QUALITY.value
   106→    affected_file: str = ""
   107→    affected_lines: List[int] = field(default_factory=list)
   108→    original_code: str = ""
   109→    proposed_code: str = ""
   110→    risk_level: str = RiskLevel.MEDIUM.value
   111→    confidence: float = 0.5
   112→    automated: bool = False
   113→    verification_steps: List[str] = field(default_factory=list)
   114→    rollback_plan: str = ""
   115→    status: str = ProposalStatus.PROPOSED.value
   116→    source_issue: Dict[str, Any] = field(default_factory=dict)
   117→    created_at: str = field(
   118→        default_factory=lambda: datetime.now(timezone.utc).isoformat(),
   119→    )
   120→    applied_at: str = ""
   121→    verified_at: str = ""
   122→    error_message: str = ""
   123→
   124→    # ── Serialisation ────────────────────────────────────────────────
   125→
   126→    def to_dict(self) -> Dict[str, Any]:
   127→        return {
   128→            "id": self.id,
   129→            "title": self.title,
   130→            "description": self.description,
   131→            "severity": self.severity,
   132→            "category": self.category,
   133→            "affected_file": self.affected_file,
   134→            "affected_lines": self.affected_lines,
   135→            "original_code": self.original_code,
   136→            "proposed_code": self.proposed_code,
   137→            "risk_level": self.risk_level,
   138→            "confidence": self.confidence,
   139→            "automated": self.automated,
   140→            "verification_steps": self.verification_steps,
   141→            "rollback_plan": self.rollback_plan,
   142→            "status": self.status,
   143→            "source_issue": self.source_issue,
   144→            "created_at": self.created_at,
   145→            "applied_at": self.applied_at,
   146→            "verified_at": self.verified_at,
   147→            "error_message": self.error_message,
   148→        }
   149→
   150→    @classmethod
   151→    def from_dict(cls, data: Dict[str, Any]) -> FixProposal:
   152→        return cls(
   153→            id=data.get("id", uuid.uuid4().hex[:12]),
   154→            title=data.get("title", ""),
   155→            description=data.get("description", ""),
   156→            severity=data.get("severity", "medium"),
   157→            category=data.get("category", FixCategory.CODE_QUALITY.value),
   158→            affected_file=data.get("affected_file", ""),
   159→            affected_lines=data.get("affected_lines", []),
   160→            original_code=data.get("original_code", ""),
   161→            proposed_code=data.get("proposed_code", ""),
   162→            risk_level=data.get("risk_level", RiskLevel.MEDIUM.value),
   163→            confidence=data.get("confidence", 0.5),
   164→            automated=data.get("automated", False),
   165→            verification_steps=data.get("verification_steps", []),
   166→            rollback_plan=data.get("rollback_plan", ""),
   167→            status=data.get("status", ProposalStatus.PROPOSED.value),
   168→            source_issue=data.get("source_issue", {}),
   169→            created_at=data.get("created_at", ""),
   170→            applied_at=data.get("applied_at", ""),
   171→            verified_at=data.get("verified_at", ""),
   172→            error_message=data.get("error_message", ""),
   173→        )

   564→        "match_category": "python",
   565→        "match_pattern": r"below minimum",
   566→        "title": "Upgrade Python to 3.9 or higher",
   567→        "description": (
   568→            "Python version is below the minimum supported (3.9). "
   569→            "Upgrade Python to the latest stable release."
   570→        ),
   571→        "category": FixCategory.DEPENDENCY.value,
   572→        "risk": RiskLevel.HIGH.value,
   573→        "confidence": 0.95,
   574→        "automated": False,
   575→    },
   576→]
   577→
   578→
   579→# Merge all templates for fast lookup
   580→_ALL_FIX_TEMPLATES: List[Dict[str, Any]] = (
   581→    _SECURITY_FIX_TEMPLATES
   582→    + _PERFORMANCE_FIX_TEMPLATES
   583→    + _CODE_QUALITY_TEMPLATES
   584→    + _CONFIG_FIX_TEMPLATES
   585→)

   588→# ═══════════════════════════════════════════════════════════════════════════
   589→# FixAnalyzer
   590→# ═══════════════════════════════════════════════════════════════════════════
   591→
   592→
   593→class FixAnalyzer:
   594→    """Analyses issues and generates FixProposal objects.
   595→
   596→    Uses pattern matching against a knowledge base of known issue → fix
   597→    mappings, plus AST-based code analysis for Python source files.
   598→    """
   599→
   600→    def __init__(self) -> None:
   601→        self._template_cache: Dict[str, List[Dict[str, Any]]] = {}
   602→        self._build_template_index()
   603→
   604→    def _build_template_index(self) -> None:
   605→        """Index templates by match_category for O(1) lookup."""
   606→        for tmpl in _ALL_FIX_TEMPLATES:
   607→            cat = tmpl["match_category"]
   608→            self._template_cache.setdefault(cat, []).append(tmpl)
   609→
   610→    # ── Public API ────────────────────────────────────────────────────
   611→
   612→    def analyze_issue(self, issue: Dict[str, Any]) -> Optional[FixProposal]:
   613→        """Analyse a single issue dict and return a FixProposal (or None)."""
   614→        # Determine which kind of issue this is
   615→        category = issue.get("category", "")
   616→        message = issue.get("message", "")
   617→        evidence = issue.get("evidence", "")
   618→        title = issue.get("title", "")
   619→        severity = issue.get("severity", "medium")
   620→        description = issue.get("description", "")
   621→        file_path = issue.get("asset", issue.get("file", ""))
   622→
   623→        # Combine searchable text
   624→        search_text = " ".join([title, message, evidence, description]).lower()
   625→
   626→        # 1. Try exact category match first
   627→        proposals = self._match_by_category(category, issue)
   628→        if proposals:
   629→            proposal = proposals[0]
   630→            proposal.severity = self._normalise_severity(severity)
   631→            proposal.source_issue = issue
   632→            proposal.affected_file = str(file_path) if file_path else ""
   633→            return proposal
   634→
   635→        # 2. Try pattern match on the text
   636→        proposals = self._match_by_pattern(search_text, issue)
   637→        if proposals:
   638→            proposal = proposals[0]
   639→            proposal.severity = self._normalise_severity(severity)
   640→            proposal.source_issue = issue
   641→            proposal.affected_file = str(file_path) if file_path else ""
   642→            return proposal
   643→
   644→        # 3. Generate a generic proposal if we have any issue at all
   645→        if message or title:
   646→            return self._generic_proposal(issue)
   647→
   648→        return None
   649→
   650→    def propose_code_fixes(
   651→        self,
   652→        file_path: str,
   653→        issues: List[Dict[str, Any]],
   654→    ) -> List[FixProposal]:
   655→        """Generate code-level fix proposals for issues in a file.
   656→
   657→        Uses AST analysis to propose concrete code changes when possible.
   658→        """
   659→        proposals: List[FixProposal] = []
   660→        path = Path(file_path)
   661→
   662→        if not path.is_file():
   663→            logger.warning("Cannot propose code fixes for non-existent file: %s", file_path)
   664→            return proposals
   665→
   666→        try:
   667→            source = path.read_text(encoding="utf-8", errors="replace")
   668→        except OSError as exc:
   669→            logger.error("Failed to read %s: %s", file_path, exc)
   670→            return proposals
   671→
   672→        lines = source.splitlines()
   673→
   674→        for issue in issues:
   675→            proposal = self._generate_code_proposal(file_path, source, lines, issue)
   676→            if proposal is not None:
   677→                proposals.append(proposal)
   678→
   679→        return proposals
   680→
   681→    def propose_config_fixes(
   682→        self,
   683→        config_issues: List[Dict[str, Any]],
   684→    ) -> List[FixProposal]:
   685→        """Generate configuration fix proposals."""
   686→        proposals: List[FixProposal] = []
   687→        for issue in config_issues:
   688→            proposal = self.analyze_issue(issue)
   689→            if proposal is not None:
   690→                proposals.append(proposal)
   691→        return proposals
   692→
   693→    # ── Internal matching ─────────────────────────────────────────────
   694→
   695→    def _match_by_category(
   696→        self,
   697→        category: str,
   698→        issue: Dict[str, Any],
   699→    ) -> List[FixProposal]:
   700→        """Find fix templates matching the issue category."""
   701→        results: List[FixProposal] = []
   702→        templates = self._template_cache.get(category, [])
   703→        for tmpl in templates:
   704→            proposal = self._template_to_proposal(tmpl, issue)
   705→            if proposal is not None:
   706→                results.append(proposal)
   707→        return results
   708→
   709→    def _match_by_pattern(
   710→        self,
   711→        text: str,
   712→        issue: Dict[str, Any],
   713→    ) -> List[FixProposal]:
   714→        """Find fix templates whose regex pattern matches the text."""
   715→        results: List[FixProposal] = []
   716→        seen_categories: Set[str] = set()
   717→        for tmpl in _ALL_FIX_TEMPLATES:
   718→            cat = tmpl["match_category"]
   719→            if cat in seen_categories:
   720→                continue
   721→            try:
   722→                pattern = re.compile(tmpl["match_pattern"], re.IGNORECASE)
   723→                if pattern.search(text):
   724→                    proposal = self._template_to_proposal(tmpl, issue)
   725→                    if proposal is not None:
   726→                        results.append(proposal)
   727→                        seen_categories.add(cat)
   728→            except re.error:
   729→                continue
   730→        return results
   731→
   732→    def _template_to_proposal(
   733→        self,
   734→        tmpl: Dict[str, Any],
   735→        issue: Dict[str, Any],
   736→    ) -> Optional[FixProposal]:
   737→        """Convert a fix template + issue into a FixProposal."""
   738→        title = tmpl.get("title", "")
   739→        if not title:
   740→            return None
   741→
   742→        evidence = issue.get("evidence", "")
   743→        file_path = issue.get("asset", issue.get("file", ""))
   744→
   745→        proposal = FixProposal(
   746→            title=title,
   747→            description=tmpl.get("description", ""),
   748→            severity=self._normalise_severity(issue.get("severity", "medium")),
   749→            category=tmpl.get("category", FixCategory.CODE_QUALITY.value),
   750→            affected_file=str(file_path) if file_path else "",
   751→            original_code=evidence,
   752→            risk_level=tmpl.get("risk", RiskLevel.MEDIUM.value),
   753→            confidence=tmpl.get("confidence", 0.5),
   754→            automated=tmpl.get("automated", False),
   755→            source_issue=issue,
   756→            rollback_plan=self._generate_rollback(tmpl, issue),
   757→            verification_steps=self._generate_verification(tmpl, issue),
   758→        )
   759→        return proposal
   760→
   761→    def _generic_proposal(self, issue: Dict[str, Any]) -> FixProposal:
   762→        """Create a generic fix proposal when no template matches."""
   763→        title = issue.get("title", "") or issue.get("message", "")
   764→        suggestion = issue.get("suggestion", "") or issue.get("remediation", "")
   765→        description = issue.get("description", "")
   766→        category = issue.get("category", "")
   767→
   768→        # Map issue category to fix category
   769→        fix_cat = FixCategory.CODE_QUALITY.value
   770→        if category in ("security", "headers", "tls", "ssl"):
   771→            fix_cat = FixCategory.SECURITY.value
   772→        elif category in ("performance", "speed"):
   773→            fix_cat = FixCategory.PERFORMANCE.value
   774→        elif category in ("paths", "config", "disk", "network", "python", "modules"):
   775→            fix_cat = FixCategory.CONFIG.value
   776→        elif category in ("deps", "dependencies"):
   777→            fix_cat = FixCategory.DEPENDENCY.value
   778→
   779→        return FixProposal(
   780→            title=f"Review and fix: {title}",
   781→            description=description or f"Address the issue: {title}",
   782→            severity=self._normalise_severity(issue.get("severity", "medium")),
   783→            category=fix_cat,
   784→            affected_file=issue.get("asset", ""),
   785→            proposed_code=suggestion,
   786→            risk_level=RiskLevel.MEDIUM.value,
   787→            confidence=0.3,
   788→            automated=False,
   789→            source_issue=issue,
   790→            verification_steps=[
   791→                f"Manually review the issue: {title}",
   792→                f"Apply the suggested fix: {suggestion}" if suggestion else "Determine and apply the appropriate fix",
   793→                "Re-run diagnostics or scan to confirm resolution",
   794→            ],
   795→            rollback_plan="Revert manual changes. No automated rollback available for generic proposals.",
   796→        )
   797→
   798→    # ── Code-level proposal generation ────────────────────────────────
   799→
   800→    def _generate_code_proposal(
   801→        self,
   802→        file_path: str,
   803→        source: str,
   804→        lines: List[str],
   805→        issue: Dict[str, Any],
   806→    ) -> Optional[FixProposal]:
   807→        """Generate a concrete code fix proposal using AST analysis."""
   808→        category = issue.get("category", "")
   809→        evidence = issue.get("evidence", "")
   810→
   811→        # Try AST-based transformations for Python files
   812→        if file_path.endswith(".py"):
   813→            proposal = self._ast_code_proposal(file_path, source, lines, issue)
   814→            if proposal is not None:
   815→                return proposal
   816→
   817→        # Fall back to text-based proposals
   818→        proposal = self.analyze_issue(issue)
   819→        if proposal is not None and not proposal.original_code and evidence:
   820→            proposal.original_code = evidence
   821→            proposal.affected_file = file_path
   822→        return proposal
   823→
   824→    def _ast_code_proposal(
   825→        self,
   826→        file_path: str,
   827→        source: str,
   828→        lines: List[str],
   829→        issue: Dict[str, Any],
   830→    ) -> Optional[FixProposal]:
   831→        """Use AST analysis to generate a concrete code fix."""
   832→        try:
   833→            tree = ast.parse(source, filename=file_path)
   834→        except SyntaxError:
   835→            return None
   836→
   837→        category = issue.get("category", "")
   838→        title = issue.get("title", "")
   839→        evidence = issue.get("evidence", "")
   840→
   841→        # Find the relevant node
   842→        finder = _IssueNodeFinder(category, evidence, title)
   843→        finder.visit(tree)
   844→
   845→        if not finder.matched_nodes:
   846→            return None
   847→
   848→        node = finder.matched_nodes[0]
   849→        start_line = getattr(node, "lineno", 0)
   850→        end_line = getattr(node, "end_lineno", start_line)
   851→
   852→        # Extract original code lines
   853→        orig_lines = lines[start_line - 1 : end_line]
   854→        original_code = "\n".join(orig_lines)
   855→
   856→        # Generate proposed code based on category
   857→        proposed_code = self._transform_node(category, node, lines, start_line, end_line)
   858→        if proposed_code is None:
   859→            return None
   860→
   861→        # Look up template for metadata
   862→        templates = self._template_cache.get(category, [])
   863→        tmpl = templates[0] if templates else {}
   864→
   865→        return FixProposal(
   866→            title=tmpl.get("title", f"Fix: {title}"),
   867→            description=tmpl.get("description", f"Address the {category} issue in {file_path}"),
   868→            severity=self._normalise_severity(issue.get("severity", "medium")),
   869→            category=tmpl.get("category", FixCategory.SECURITY.value),
   870→            affected_file=file_path,
   871→            affected_lines=list(range(start_line, end_line + 1)),
   872→            original_code=original_code,
   873→            proposed_code=proposed_code,
   874→            risk_level=tmpl.get("risk", RiskLevel.MEDIUM.value),
   875→            confidence=tmpl.get("confidence", 0.7),
   876→            automated=tmpl.get("automated", False),
   877→            source_issue=issue,
   878→            verification_steps=[
   879→                f"Verify the fix in {file_path} lines {start_line}-{end_line}",
   880→                "Run the AST analyzer again to confirm the issue is resolved",
   881→                "Run tests to ensure no regressions",
   882→            ],
   883→            rollback_plan=f"Restore original code at lines {start_line}-{end_line} in {file_path}",
   884→        )
   885→
   886→    def _transform_node(
   887→        self,
   888→        category: str,
   889→        node: ast.AST,
   890→        lines: List[str],
   891→        start_line: int,
   892→        end_line: int,
   893→    ) -> Optional[str]:
   894→        """Transform an AST node into fixed code.
   895→
   896→        Returns the proposed replacement code as a string, or None if no
   897→        transformation is possible.
   898→        """
   899→        indent = self._get_indent(lines, start_line - 1)
   900→
   901→        if category == "dangerous_eval" and isinstance(node, ast.Call):
   902→            return self._transform_eval(node, indent, lines, start_line)
   903→
   904→        if category == "dangerous_exec" and isinstance(node, ast.Call):
   905→            return self._transform_exec(node, indent, lines, start_line)
   906→
   907→        if category == "command_injection" and isinstance(node, ast.Call):
   908→            return self._transform_shell_true(node, indent, lines, start_line, end_line)
   909→
   910→        if category == "hardcoded_secret" and isinstance(node, ast.Assign):
   911→            return self._transform_hardcoded_secret(node, indent, lines, start_line)
   912→
   913→        if category == "bare_except":
   914→            return self._transform_bare_except(indent, lines, start_line)
   915→
   916→        if category == "debug_mode":
   917→            return self._transform_debug_mode(indent, lines, start_line)
   918→
   919→        if category == "cleartext_http":
   920→            return self._transform_http_to_https(indent, lines, start_line, end_line)
   921→
   922→        if category == "temp_file_race":
   923→            return self._transform_mktemp(indent, lines, start_line)
   924→
   925→        if category == "csrf_disabled":
   926→            return self._transform_csrf_disabled(indent, lines, start_line)
   927→
   928→        return None
   929→
   930→    # ── AST transform helpers ─────────────────────────────────────────
   931→
   932→    @staticmethod
   933→    def _get_indent(lines: List[str], index: int) -> str:
   934→        """Extract leading whitespace from a line."""
   935→        if 0 <= index < len(lines):
   936→            line = lines[index]
   937→            return line[: len(line) - len(line.lstrip())]
   938→        return ""
   939→
   940→    def _transform_eval(
   941→        self,
   942→        node: ast.Call,
   943→        indent: str,
   944→        lines: List[str],
   945→        start_line: int,
   946→    ) -> Optional[str]:
   947→        """Propose replacing eval() with ast.literal_eval()."""
   948→        func_name = _get_call_name(node)
   949→        if func_name != "eval":
   950→            return None
   951→
   952→        # Reconstruct the call argument
   953→        if node.args:
   954→            arg_src = self._extract_arg_source(node.args[0], lines)
   955→        else:
   956→            arg_src = "..."
   957→
   958→        return (
   959→            f"{indent}import ast as _ast  # noqa: added for safe evaluation\n"
   960→            f"{indent}_ast.literal_eval({arg_src})"
   961→        )
   962→
   963→    def _transform_exec(
   964→        self,
   965→        node: ast.Call,
   966→        indent: str,
   967→        lines: List[str],
   968→        start_line: int,
   969→    ) -> Optional[str]:
   970→        """Propose removing exec() or marking it as dangerous."""
   971→        func_name = _get_call_name(node)
   972→        if func_name != "exec":
   973→            return None
   974→
   975→        arg_src = ""
   976→        if node.args:
   977→            arg_src = self._extract_arg_source(node.args[0], lines)
   978→        else:
   979→            arg_src = "..."
   980→
   981→        return (
   982→            f"{indent}# WARNING: exec() removed — arbitrary code execution risk\n"
   983→            f"{indent}# If this is needed, use a restricted sandbox or whitelist approach\n"
   984→            f"{indent}# Original: exec({arg_src})\n"
   985→            f"{indent}raise NotImplementedError(\"exec() was removed for security reasons\")"
   986→        )
   987→
   988→    def _transform_shell_true(
   989→        self,
   990→        node: ast.Call,
   991→        indent: str,
   992→        lines: List[str],
   993→        start_line: int,
   994→        end_line: int,
   995→    ) -> Optional[str]:
   996→        """Remove shell=True from a subprocess call."""
   997→        func_name = _get_call_name(node)
   998→        if func_name not in ("run", "call", "Popen", "check_output", "check_call"):
   999→            return None
  1000→
  1001→        # Find and remove shell=True keyword
  1002→        new_lines: List[str] = []
  1003→        for i in range(start_line - 1, end_line):
  1004→            line = lines[i]
  1005→            if "shell=True" in line:
  1006→                # Remove the shell=True keyword argument
  1007→                line = re.sub(r",?\s*shell\s*=\s*True\s*,?", ",", line)
  1008→                line = re.sub(r",\s*,", ",", line)  # Clean double commas
  1009→                # If shell=True was the only keyword, clean up
  1010→                line = re.sub(r"\(\s*,", "(", line)
  1011→                line = re.sub(r",\s*\)", ")", line)
  1012→            new_lines.append(line)
  1013→
  1014→        return "\n".join(new_lines)

  1016→    def _transform_hardcoded_secret(
  1017→        self,
  1018→        node: ast.Assign,
  1019→        indent: str,
  1020→        lines: List[str],
  1021→        start_line: int,
  1022→    ) -> Optional[str]:
  1023→        """Replace hardcoded secret with os.environ.get()."""
  1024→        # Get variable name
  1025→        targets = node.targets
  1026→        if not targets or not isinstance(targets[0], ast.Name):
  1027→            return None
  1028→
  1029→        var_name = targets[0].id
  1030→        return (
  1031→            f"{indent}import os  # noqa: added for secret loading\n"
  1032→            f"{indent}{var_name} = os.environ.get(\"{var_name.upper()}\", \"\")  "
  1033→            f"# was hardcoded secret"
  1034→        )

  1036→    def _transform_bare_except(
  1037→        self,
  1038→        indent: str,
  1039→        lines: List[str],
  1040→        start_line: int,
  1041→    ) -> Optional[str]:
  1042→        """Replace bare 'except:' with 'except Exception:'."""
  1043→        if start_line - 1 < len(lines):
  1044→            line = lines[start_line - 1]
  1045→            if re.search(r"except\s*:\s*$", line):
  1046→                new_line = re.sub(r"except\s*:", "except Exception:", line)
  1047→                return new_line
  1048→        return None

  1050→    def _transform_debug_mode(
  1051→        self,
  1052→        indent: str,
  1053→        lines: List[str],
  1054→        start_line: int,
  1055→    ) -> Optional[str]:
  1056→        """Replace DEBUG=True with DEBUG=False."""
  1057→        if start_line - 1 < len(lines):
  1058→            line = lines[start_line - 1]
  1059→            new_line = re.sub(
  1060→                r"(DEBUG\s*=\s*)True",
  1061→                r"\1False  # disabled in production",
  1062→                line,
  1063→                flags=re.IGNORECASE,
  1064→            )
  1065→            if new_line != line:
  1066→                return new_line
  1067→        return None

  1069→    def _transform_http_to_https(
  1070→        self,
  1071→        indent: str,
  1072→        lines: List[str],
  1073→        start_line: int,
  1074→        end_line: int,
  1075→    ) -> Optional[str]:
  1076→        """Replace http:// with https:// in the matched lines."""
  1077→        new_lines: List[str] = []
  1078→        changed = False
  1079→        for i in range(start_line - 1, end_line):
  1080→            line = lines[i]
  1081→            new_line = re.sub(r'"http://', '"https://', line)
  1082→            new_line = re.sub(r"'http://", "'https://", new_line)
  1083→            if new_line != line:
  1084→                changed = True
  1085→            new_lines.append(new_line)
  1086→        return "\n".join(new_lines) if changed else None

  1088→    def _transform_mktemp(
  1089→        self,
  1090→        indent: str,
  1091→        lines: List[str],
  1092→        start_line: int,
  1093→    ) -> Optional[str]:
  1094→        """Replace mktemp() with mkstemp() or NamedTemporaryFile()."""
  1095→        if start_line - 1 < len(lines):
  1096→            line = lines[start_line - 1]
  1097→            if re.search(r"mktemp\s*\(", line):
  1098→                new_line = re.sub(
  1099→                    r"mktemp\s*\([^)]*\)",
  1100→                    "NamedTemporaryFile(delete=False).name",
  1101→                    line,
  1102→                )
  1103→                # Add import if needed
  1104→                return (
  1105→                    f"{indent}from tempfile import NamedTemporaryFile  # noqa: safe replacement\n"
  1106→                    f"{new_line}"
  1107→                )
  1108→        return None

  1113→        lines: List[str],

  1125→            if new_line != line:

  1141→            # Fallback: take from col_start to end of line

  1198→class _IssueNodeFinder(ast.NodeVisitor):
  1199→    """Walks the AST and collects nodes that match a given issue."""

  1201→    def __init__(self, category: str, evidence: str, title: str) -> None:
  1202→        self.category = category
  1203→        self.evidence = evidence.lower()
  1204→        self.title = title.lower()

  1206→        self._imported: Set[str] = set()
  1207→
  1208→    def visit_Import(self, node: ast.Import) -> None:
  1209→        for alias in node.names:
  1210→            self._imported.add(alias.name.split(".")[0])
  1211→        self.generic_visit(node)
  1212→
  1213→    def visit_ImportFrom(self, node: ast.ImportFrom) -> None:
  1214→        if node.module:
  1215→            self._imported.add(node.module.split(".")[0])
  1216→        self.generic_visit(node)
  1217→
  1218→    def visit_Call(self, node: ast.Call) -> None:
  1219→        fname = _get_call_name(node)
  1220→        func_full = _get_call_full_name(node)
  1221→
  1222→        if self.category == "dangerous_eval" and fname == "eval":
  1223→            self.matched_nodes.append(node)
  1224→        elif self.category == "dangerous_exec" and fname == "exec":
  1225→            self.matched_nodes.append(node)
  1226→        elif self.category == "command_injection":
  1227→            if fname in ("run", "call", "Popen", "check_output", "check_call"):
  1228→                for kw in node.keywords:
  1229→                    if kw.arg == "shell" and isinstance(kw.value, ast.Constant) and kw.value.value is True:
  1230→                        self.matched_nodes.append(node)
  1231→                        break
  1232→        elif self.category == "unsafe_deserialization":
  1233→            if func_full and ("pickle.loads" in func_full or "marshal.loads" in func_full):
  1234→                self.matched_nodes.append(node)
  1235→            elif func_full and "yaml.load" in func_full:
  1236→                self.matched_nodes.append(node)
  1237→        elif self.category == "jwt_no_verify":
  1238→            if func_full and "jwt.decode" in func_full:
  1239→                self.matched_nodes.append(node)
  1240→        elif self.category == "temp_file_race":
  1241→            if func_full and "mktemp" in func_full:
  1242→                self.matched_nodes.append(node)
  1243→
  1244→        self.generic_visit(node)
  1245→
  1246→    def visit_Assign(self, node: ast.Assign) -> None:
  1247→        if self.category == "hardcoded_secret":
  1248→            # Get the source text for the line
  1249→            start = getattr(node, "lineno", 0)
  1250→            end = getattr(node, "end_lineno", start)
  1251→            # We'll match by the parent logic
  1252→            for target in node.targets:
  1253→                if isinstance(target, ast.Name):
  1254→                    name = target.id.lower()
  1255→                    secret_names = ("password", "passwd", "secret", "token",
  1256→                                    "api_key", "apikey", "private_key")
  1257→                    if any(s in name for s in secret_names):
  1258→                        # Check if value is a string constant
  1259→                        if isinstance(node.value, (ast.Constant,)) and isinstance(node.value.value, str):
  1260→                            self.matched_nodes.append(node)
  1261→
  1262→        self.generic_visit(node)
  1263→
  1264→    def visit_ExceptHandler(self, node: ast.ExceptHandler) -> None:
  1265→        if self.category == "bare_except":
  1266→            if node.type is None:
  1267→                self.matched_nodes.append(node)
  1268→        self.generic_visit(node)
  1269→
  1270→    def visit_JoinedStr(self, node: ast.JoinedStr) -> None:
  1271→        if self.category == "sql_injection":
  1272→            parts: List[str] = []
  1273→            for val in node.values:
  1274→                if isinstance(val, ast.Constant):
  1275→                    parts.append(str(val.value))
  1276→                elif isinstance(val, ast.FormattedValue):
  1277→                    parts.append("{}")

  1280→                r"\b(?:SELECT|INSERT|UPDATE|DELETE|DROP|UNION|ALTER|CREATE|EXEC)\b", re.I,
  1281→            )
  1282→            if sql_kw.search(combined):
  1283→                self.matched_nodes.append(node)
  1284→        self.generic_visit(node)

  1287→        """Additional line-based matching for statement nodes."""
  1288→        # Handled in parent methods
  1289→        pass

  1304→def _get_call_name(node: ast.Call) -> str:
  1305→    """Get the simple function name from a Call node."""
  1306→    func = node.func
  1307→    if isinstance(func, ast.Name):
  1308→        return func.id
  1309→    if isinstance(func, ast.Attribute):
  1310→        return func.attr
  1311→    return ""
  1312→
  1313→
  1314→def _get_call_full_name(node: ast.Call) -> str:
  1315→    """Get the full dotted function name from a Call node."""
  1316→    func = node.func
  1317→    if isinstance(func, ast.Name):
  1318→        return func.id
  1319→    if isinstance(func, ast.Attribute):
  1320→        parts = [func.attr]
  1321→        current = func.value
  1322→        while isinstance(current, ast.Attribute):
  1323→            parts.append(current.attr)
  1324→            current = current.value
  1325→        if isinstance(current, ast.Name):
  1326→            parts.append(current.id)
  1327→        parts.reverse()
  1328→        return ".".join(parts)
  1329→    return ""

  1339→
  1340→    Stores proposals in ``RECONPRO_HOME/memory/fix_proposals.json``.
  1341→    Tracks the full lifecycle: proposed → approved → applied → verified.

  1343→
  1344→    def __init__(self, store_path: Optional[Path] = None) -> None:
  1345→        self._path = store_path or (MEMORY_DIR / "fix_proposals.json")
  1346→        self._proposals: Dict[str, FixProposal] = {}
  1347→        self._history: List[Dict[str, Any]] = []

  1351→
  1352→    def _ensure_dir(self) -> None:
  1353→        """Ensure the storage directory exists."""
  1354→        try:
  1355→            self._path.parent.mkdir(parents=True, exist_ok=True)
  1356→        except OSError as exc:
  1357→            logger.error("Cannot create fix store directory %s: %s", self._path.parent, exc)
  1358→
  1359→    def _load(self) -> None:
  1360→        """Load proposals from disk."""
  1361→        if not self._path.exists():
  1362→            self._proposals = {}
  1363→            self._history = []
  1364→            return
  1365→        try:
  1366→            raw = self._path.read_text(encoding="utf-8")
  1367→            data = json.loads(raw)
  1368→            proposals_data = data.get("proposals", {})
  1369→            self._proposals = {
  1370→                pid: FixProposal.from_dict(pdata)
  1371→                for pid, pdata in proposals_data.items()
  1372→            }
  1373→            self._history = data.get("history", [])
  1374→        except (json.JSONDecodeError, OSError, KeyError) as exc:
  1375→            logger.warning("Failed to load fix store from %s: %s", self._path, exc)
  1376→            self._proposals = {}
  1377→            self._history = []
  1378→
  1379→    def _save(self) -> None:
  1380→        """Persist proposals to disk."""
  1381→        self._ensure_dir()
  1382→        try:
  1383→            data = {
  1384→                "proposals": {
  1385→                    pid: p.to_dict() for pid, p in self._proposals.items()
  1386→                },
  1387→                "history": self._history[-500:],  # Keep last 500 history entries
  1388→                "updated_at": datetime.now(timezone.utc).isoformat(),
  1389→            }
  1390→            tmp_path = self._path.with_suffix(".tmp")
  1391→            tmp_path.write_text(json.dumps(data, indent=2, ensure_ascii=False), encoding="utf-8")
  1392→            tmp_path.replace(self._path)
  1393→        except OSError as exc:
  1394→            logger.error("Failed to save fix store to %s: %s", self._path, exc)

  1412→            self._proposals[proposal.id] = proposal

  1518→# AutoFixEngine — Top-level orchestrator

  1521→
  1522→class AutoFixEngine:
  1523→    """Top-level engine that analyses scan results and proposes automated fixes.
  1524→
  1525→    The engine is **dry-run by default** — it only *proposes* fixes.
  1526→    Applying a fix requires explicit approval via :meth:`apply_fix` with
  1527→    ``dry_run=False``.

  1529→    Usage::
  1530→
  1531→        engine = AutoFixEngine()
  1532→        proposals = engine.propose_fixes(scan_result)
  1533→        for p in proposals:
  1534→            print(f"[{p.severity}] {p.title} (confidence={p.confidence})")
  1535→

  1537→        result = engine.apply_fix(proposals[0], dry_run=False)
  1538→        print(result)
  1539→    """
  1540→
  1541→    def __init__(
  1542→        self,
  1543→        store: Optional[FixStore] = None,
  1544→        dry_run: bool = True,
  1545→    ) -> None:
  1546→        self._analyzer = FixAnalyzer()
  1547→        self._store = store or FixStore()
  1548→        self._dry_run = dry_run
  1549→        logger.info("AutoFixEngine initialised (dry_run=%s)", dry_run)
  1550→
  1551→    # ── Public API ────────────────────────────────────────────────────
  1552→
  1553→    def analyze_issue(self, issue: Dict[str, Any]) -> Optional[FixProposal]:
  1554→        """Analyse a single issue and return a fix proposal."""
  1555→        try:
  1556→            proposal = self._analyzer.analyze_issue(issue)
  1557→            if proposal is not None:
  1558→                self._store.add(proposal)
  1559→                logger.info(
  1560→                    "Proposed fix %s: %s (confidence=%.2f)",
  1561→                    proposal.id, proposal.title, proposal.confidence,
  1562→                )
  1563→            return proposal
  1564→        except Exception as exc:
  1565→            logger.error("Failed to analyze issue: %s", exc, exc_info=True)
  1566→            return None

  1568→    def propose_fixes(self, scan_result: Dict[str, Any]) -> List[FixProposal]:
  1569→        """Analyse a full scan result and propose fixes for all findings.
  1570→
  1571→        The scan_result dict can contain:
  1572→        - ``findings``: List of finding dicts (from http_layer.Finding.to_dict())
  1573→        - ``diagnostics``: Dict from diagnostics.run_diagnostics()
  1574→        - ``config_issues``: List from diagnostics.validate_config()
  1575→        - ``cross_validation``: Dict from cross_validator
  1576→
  1577→        Any combination of the above keys is supported.
  1578→        """
  1579→        proposals: List[FixProposal] = []
  1580→
  1581→        # 1. Process scan findings
  1582→        findings = scan_result.get("findings", [])
  1583→        if findings:
  1584→            proposals.extend(self._process_findings(findings))
  1585→
  1586→        # 2. Process diagnostics failures
  1587→        diagnostics = scan_result.get("diagnostics", {})
  1588→        if diagnostics:
  1589→            proposals.extend(self._process_diagnostics(diagnostics))
  1590→
  1591→        # 3. Process config issues
  1592→        config_issues = scan_result.get("config_issues", [])
  1593→        if config_issues:
  1594→            proposals.extend(self._process_config_issues(config_issues))
  1595→
  1596→        # 4. Process cross-validation mismatches
  1597→        cv = scan_result.get("cross_validation", {})
  1598→        if cv:
  1599→            proposals.extend(self._process_cross_validation(cv))
  1600→
  1601→        logger.info(
  1602→            "Generated %d fix proposals from scan result (%d findings, %d diagnostics, %d config, %d cv)",
  1603→            len(proposals),
  1604→            len(findings),
  1605→            len(diagnostics.get("checks", [])) if diagnostics else 0,
  1606→            len(config_issues),
  1607→            len(cv.get("mismatches", [])) if cv else 0,
  1608→        )
  1609→
  1610→        return proposals

  1617→        """Propose code-level fixes for issues in a specific file."""

  1627→    def propose_config_fixes(
  1628→        self,
  1629→        config_issues: List[Dict[str, Any]],
  1630→    ) -> List[FixProposal]:
