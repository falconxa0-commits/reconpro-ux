   182→   182→        rank = {"critical": 0, "degraded": 1, "healthy": 2, "unknown": 3, "disabled": 4}

   219→   219→class SimulationEngine:
   220→   220→    """Run what-if scenarios against a SystemModel without touching the real system.
   221→   221→
   222→   222→    Scenarios supported
   223→   223→    -------------------
   224→   224→    - "add_concurrent_scans:N"   — simulate adding N more concurrent scans
   225→   225→    - "fail_module:<module_id>"  — simulate a module failure
   226→   226→    - "increase_load:N"          — simulate N× load increase
   227→   227→    - "change_timeout:N"         — change default timeout to N seconds
   228→   228→    - "change_concurrency:N"     — change max concurrency to N
   229→   229→    - "disk_full"                — simulate disk full condition
   230→   230→    - "network_degraded"         — simulate degraded network
   231→   231→    - "memory_pressure"          — simulate high memory usage
   232→   232→    - "plugin_load:N"            — simulate loading N additional plugins
   233→   233→    - Custom scenario dict passed via ``what_if()``.
   234→   234→    """
   235→   235→
   236→   236→    # ── Scenario handlers ─────────────────────────────────────────────
   237→   237→
   238→   238→    def simulate(self, model: SystemModel, scenario: str) -> SimulationResult:
   239→   239→        """Run a named scenario against a snapshot *model*.
   240→   240→
   241→   241→        Returns a SimulationResult without mutating the input model.
   242→   242→        """
   243→   243→        sim_model = copy.deepcopy(model)
   244→   244→        affected: List[str] = ["engine", "scanner"]
   245→   245→        recommendations: List[str] = []
   246→   246→        risk = "low"
   247→   247→        perf_impact = "none"
   248→   248→        confidence = 0.7
   249→   249→
   250→   250→        try:
   251→   251→            if scenario.startswith("add_concurrent_scans:"):
   252→   252→                n = self._parse_int(scenario, 0)
   253→   253→                risk, perf_impact, recommendations = self._sim_add_concurrent(
   254→   254→                    sim_model, n
   255→   255→                )
   256→   256→                confidence = 0.8
   257→   257→
   258→   258→            elif scenario.startswith("fail_module:"):
   259→   259→                module_id = scenario.split(":", 1)[1].strip()
   260→   260→                risk, perf_impact, recommendations = self._sim_fail_module(
   261→   261→                    sim_model, module_id
   262→   262→                )
   263→   263→                confidence = 0.85
   264→   264→
   265→   265→            elif scenario.startswith("increase_load:"):
   266→   266→                factor = self._parse_int(scenario, 0)
   267→   267→                risk, perf_impact, recommendations = self._sim_increase_load(
   268→   268→                    sim_model, factor
   269→   269→                )
   270→   270→                confidence = 0.65
   271→   271→
   272→   272→            elif scenario.startswith("change_timeout:"):
   273→   273→                new_timeout = self._parse_int(scenario, 0)
   274→   274→                risk, perf_impact, recommendations = self._sim_change_timeout(
   275→   275→                    sim_model, new_timeout
   276→   276→                )
   277→   277→                confidence = 0.9
   278→   278→
   279→   279→            elif scenario.startswith("change_concurrency:"):
   280→   280→                new_conc = self._parse_int(scenario, 0)
   281→   281→                risk, perf_impact, recommendations = self._sim_change_concurrency(
   282→   282→                    sim_model, new_conc
   283→   283→                )
   284→   284→                confidence = 0.85
   285→   285→
   286→   286→            elif scenario == "disk_full":
   287→   287→                risk, perf_impact, recommendations = self._sim_disk_full(sim_model)
   288→   288→                confidence = 0.8
   289→   289→                affected.append("storage")
   290→   290→
   291→   291→            elif scenario == "network_degraded":
   292→   292→                risk, perf_impact, recommendations = self._sim_network_degraded(sim_model)
   293→   293→                confidence = 0.7
   294→   294→                affected.append("network")
   295→   295→
   296→   296→            elif scenario == "memory_pressure":
   297→   297→                risk, perf_impact, recommendations = self._sim_memory_pressure(sim_model)
   298→   298→                confidence = 0.75
   299→   299→                affected.append("storage")
   300→   300→
   301→   301→            elif scenario.startswith("plugin_load:"):
   302→   302→                n = self._parse_int(scenario, 0)
   303→   303→                risk, perf_impact, recommendations = self._sim_plugin_load(
   304→   304→                    sim_model, n
   305→   305→                )
   306→   306→                confidence = 0.7
   307→   307→                affected.append("plugins")
   308→   308→
   309→   309→            else:
   310→   310→                # Unknown scenario — generic estimate
   311→   311→                risk = "medium"
   312→   312→                perf_impact = "minimal"
   313→   313→                recommendations = [
   314→   314→                    f"Unknown scenario '{scenario}'. "
   315→   315→                    f"Provide a structured dict for custom simulations."
   316→   316→                ]
   317→   317→                confidence = 0.3
   318→   318→
   319→   319→        except Exception as exc:
   320→   320→            logger.debug("Simulation error: %s", exc, exc_info=True)
   321→   321→            risk = "medium"
   322→   322→            perf_impact = "unknown"
   323→   323→            recommendations = [f"Simulation failed: {exc}"]
   324→   324→            confidence = 0.1
   325→   325→
   326→   326→        return SimulationResult(
   327→   327→            scenario=scenario,
   328→   328→            projected_state=sim_model.to_dict(),
   329→   329→            risk_level=risk,
   330→   330→            performance_impact=perf_impact,
   331→   331→            affected_components=sorted(set(affected)),
   332→   332→            recommendations=recommendations,
   333→   333→            confidence=confidence,
   334→   334→        )

   336→   336→    def simulate_dict(self, model: SystemModel, params: Dict[str, Any]) -> SimulationResult:
   337→   337→        """Run a custom scenario described by a dictionary.
   338→   338→
   339→   339→        Keys:
   340→   340→            type        : str   — "fault", "load", "config", "resource"
   341→   341→            component   : str   — target component name
   342→   342→            parameter   : str   — what to change
   343→   343→            value       : Any   — new value
   344→   344→            description : str   — human-readable description
   345→   345→        """
   346→   346→        sim_model = copy.deepcopy(model)
   347→   347→        comp_name = params.get("component", "engine")
   348→   348→        sim_type = params.get("type", "config")
   349→   349→        desc = params.get("description", comp_name)
   350→   350→        value = params.get("value")
   351→   351→        param = params.get("parameter", "unknown")
   352→   352→
   353→   353→        affected = [comp_name]
   354→   354→        recommendations: List[str] = []
   355→   355→        risk = "low"
   356→   356→        perf_impact = "none"
   357→   357→
   358→   358→        # Apply the change to the simulated model
   359→   359→        comp = sim_model.all_components().get(comp_name)
   360→   360→        if comp is not None:
   361→   361→            comp.metrics[param] = value
   362→   362→            # Simple risk heuristic based on type
   363→   363→            if sim_type == "fault":
   364→   364→                comp.status = "critical"
   365→   365→                risk = "high"
   366→   366→                perf_impact = "significant"
   367→   367→                recommendations.append(
   368→   368→                    f"Component '{comp_name}' is now in CRITICAL state. "
   369→   369→                    f"Consider failover or restart."
   370→   370→                )
   371→   371→            elif sim_type == "load":
   372→   372→                risk = "medium"
   373→   373→                perf_impact = "moderate"
   374→   374→                recommendations.append(
   375→   375→                    f"Load change on '{comp_name}' ({param}={value}). "
   376→   376→                    f"Monitor for degradation."
   377→   377→                )
   378→   378→            elif sim_type == "resource":
   379→   379→                if comp_name == "storage":
   380→   380→                    risk = "medium"
   381→   381→                    perf_impact = "moderate"
   382→   382→                else:
   383→   383→                    risk = "low"
   384→   384→                    perf_impact = "minimal"
   385→   385→                recommendations.append(
   386→   386→                    f"Resource change on '{comp_name}' ({param}={value})."
   387→   387→                )
   388→   388→            else:
   389→   389→                # config change
   390→   390→                risk = "low"
   391→   391→                perf_impact = "minimal"
   392→   392→                recommendations.append(
   393→   393→                    f"Configuration change on '{comp_name}' ({param}={value})."
   394→   394→                )
   395→   395→
   396→   396→        return SimulationResult(
   397→   397→            scenario=desc,
   398→   398→            projected_state=sim_model.to_dict(),
   399→   399→            risk_level=risk,
   400→   400→            performance_impact=perf_impact,
   401→   401→            affected_components=affected,
   402→   402→            recommendations=recommendations,
   403→   403→            confidence=0.6,
   404→   404→        )

   406→   406→    # ── Internal scenario implementations ─────────────────────────────
   407→   407→
   408→   408→    @staticmethod
   409→   409→    def _parse_int(scenario: str, idx: int) -> int:
   410→   410→        """Extract an integer after the colon at position *idx* (0-based colon count)."""
   411→   411→        parts = scenario.split(":")
   412→   412→        if len(parts) > idx + 1:
   413→   413→            return int(parts[idx + 1].strip())
   414→   414→        raise ValueError(f"Cannot parse integer from scenario: {scenario}")
   415→   415→
   416→   416→    def _sim_add_concurrent(
   417→   417→        self, model: SystemModel, n: int
   418→   418→    ) -> Tuple[str, str, List[str]]:
   419→   419→        """Simulate adding *n* more concurrent scans."""
   420→   420→        current = model.engine.metrics.get("active_scans", 0)
   421→   421→        max_conc = model.engine.constraints.get("max_concurrency", DEFAULT_MAX_WORKERS)
   422→   422→        projected = current + n
   423→   423→
   424→   424→        model.engine.metrics["active_scans"] = projected
   425→   425→
   426→   426→        # Estimate latency impact (rough: linear scaling until thread-bound)
   427→   427→        utilisation = min(projected / max(max_conc, 1), 2.0)
   428→   428→        latency_factor = 1.0 + (utilisation - 0.5) * 0.8
   429→   429→        latency_factor = max(latency_factor, 1.0)
   430→   430→
   431→   431→        avg_latency = model.network.metrics.get("avg_latency_ms", _CONNECTION_LATENCY_BASELINE_MS)
   432→   432→        projected_latency = avg_latency * latency_factor
   433→   433→        model.network.metrics["projected_latency_ms"] = round(projected_latency, 1)
   434→   434→
   435→   435→        # Memory estimate: ~15 MB per concurrent scan (modules + findings)
   436→   436→        mem_per_scan = 15.0
   437→   437→        extra_mem = n * mem_per_scan
   438→   438→        current_mem = model.storage.metrics.get("rss_mb", 50.0)
   439→   439→        model.storage.metrics["projected_rss_mb"] = round(current_mem + extra_mem, 1)
   440→   440→
   441→   441→        # Risk assessment
   442→   442→        recommendations: List[str] = []
   443→   443→        if projected > max_conc:
   444→   444→            risk = "high"
   445→   445→            perf_impact = "significant"
   446→   446→            recommendations.append(
   447→   447→                f"Projected {projected} concurrent scans exceeds limit of {max_conc}. "
   448→   448→                f"Increase max_concurrency or reduce parallel targets."
   449→   449→            )
   450→   450→        elif utilisation > 0.8:
   451→   451→            risk = "medium"
   452→   452→            perf_impact = "moderate"
   453→   453→            recommendations.append(
   454→   454→                f"Utilisation at {utilisation:.0%}. "
   455→   455→                f"Latency may increase by {((latency_factor - 1) * 100):.0f}%. "
   456→   456→                f"Monitor response times."
   457→   457→            )
   458→   458→        else:
   459→   459→            risk = "low"
   460→   460→            perf_impact = "minimal"
   461→   461→            recommendations.append(
   462→   462→                f"Adding {n} scans is within capacity. "
   463→   463→                f"Projected utilisation: {utilisation:.0%}."
   464→   464→            )
   465→   465→
   466→   466→        _RISK_RANK = {"critical": 0, "high": 1, "medium": 2, "low": 3}
   467→   467→        if current_mem + extra_mem > _MEMORY_WARNING_MB:
   468→   468→            if _RISK_RANK.get(risk, 2) > _RISK_RANK.get("medium", 2):
   469→   469→                risk = "medium"
   470→   470→            recommendations.append(
   471→   471→                f"Memory may reach {current_mem + extra_mem:.0f} MB. "
   472→   472→                f"Consider reducing scan concurrency."
   473→   473→            )
   474→   474→
   475→   475→        return risk, perf_impact, recommendations

   477→   477→    def _sim_fail_module(
   478→   478→        self, model: SystemModel, module_id: str
   479→   479→    ) -> Tuple[str, str, List[str]]:
   480→   480→        """Simulate failure of a specific module."""
   481→   481→        # Check if module exists in registry
   482→   482→        module_list: List[str] = model.scanner.metrics.get("module_ids", [])
   483→   483→        broken: List[str] = list(model.scanner.metrics.get("broken_modules", []))
   484→   484→
   485→   485→        if module_id not in module_list and module_id not in broken:
   486→   486→            return (
   487→   487→                "low",
   488→   488→                "none",
   489→   489→                [f"Module '{module_id}' not found in registry. No impact."],
   490→   490→            )
   491→   491→
   492→   492→        # If already broken, no new impact
   493→   493→        if module_id in broken:
   494→   494→            return "low", "none", [f"Module '{module_id}' already broken. No new impact."]
   495→   495→
   496→   496→        # Mark module as broken
   497→   497→        broken.append(module_id)
   498→   498→        model.scanner.metrics["broken_modules"] = broken
   499→   499→
   500→   500→        total = len(module_list)
   501→   501→        broken_count = len(broken)
   502→   502→        integrity = "compromised" if broken_count > 0 else "ok"
   503→   503→        model.scanner.metrics["registry_integrity"] = integrity
   504→   504→
   505→   505→        # Assess impact
   506→   506→        recommendations: List[str] = []
   507→   507→
   508→   508→        broken_pct = broken_count / max(total, 1)
   509→   509→        if broken_pct > 0.5:
   510→   510→            risk = "critical"
   511→   511→            perf_impact = "significant"
   512→   512→            model.scanner.status = "critical"
   513→   513→            recommendations.append(
   514→   514→                f"Over 50% of modules are broken ({broken_count}/{total}). "
   515→   515→                f"System reliability is severely compromised."
   516→   516→            )
   517→   517→        elif broken_pct > 0.2:
   518→   518→            risk = "high"
   519→   519→            perf_impact = "moderate"
   520→   520→            model.scanner.status = "degraded"
   521→   521→            recommendations.append(
   522→   522→                f"{broken_count}/{total} modules broken. "
   523→   523→                f"Scan coverage is significantly reduced."
   524→   524→            )
   525→   525→        else:
   526→   526→            risk = "medium"
   527→   527→            perf_impact = "minimal"
   528→   528→            model.scanner.status = "degraded"
   529→   529→            recommendations.append(
   530→   530→                f"Module '{module_id}' failed. "
   531→   531→                f"{broken_count}/{total} modules broken. "
   532→   532→                f"Remaining modules can compensate."
   533→   533→            )
   534→   534→
   535→   535→        return risk, perf_impact, recommendations

   554→   554→        # Queue depth estimation

   764→   764→# ═══════════════════════════════════════════════════════════════════════════
   765→   765→#  TwinSync — keep twin in sync with the live system
   766→   766→# ═══════════════════════════════════════════════════════════════════════════
   767→   767→
   768→   768→
   769→   769→class TwinSync:
   770→   770→    """Periodically refresh the digital twin from the live system.
   771→   771→
   772→   772→    Detects drift between the twin model and reality by comparing
   773→   773→    successive snapshots.
   774→   774→    """
   775→   775→
   776→   776→    def __init__(
   777→   777→        self,
   778→   778→        twin: "DigitalTwin",
   779→   779→        refresh_interval: float = 30.0,
   780→   780→        auto_start: bool = False,
   781→   781→    ) -> None:
   782→   782→        self._twin = twin
   783→   783→        self._refresh_interval = refresh_interval
   784→   784→        self._lock = threading.Lock()
   785→   785→        self._running = False
   786→   786→        self._thread: Optional[threading.Thread] = None
   787→   787→        self._drift_history: List[Dict[str, Any]] = []
   788→   788→        self._last_model: Optional[SystemModel] = None
   789→   789→        self._sync_count = 0
   790→   790→
   791→   791→        if auto_start:
   792→   792→            self.start()
   793→   793→
   794→   794→    # ── Lifecycle ─────────────────────────────────────────────────────
   795→   795→
   796→   796→    def start(self) -> None:
   797→   797→        """Start background sync loop."""
   798→   798→        with self._lock:
   799→   799→            if self._running:
   800→   800→                return
   801→   801→            self._running = True
   802→   802→            self._thread = threading.Thread(
   803→   803→                target=self._sync_loop,
   804→   804→                name="twin-sync",
   805→   805→                daemon=True,
   806→   806→            )
   807→   807→            self._thread.start()
   808→   808→            logger.debug("TwinSync started (interval=%.1fs)", self._refresh_interval)
   809→   809→
   810→   810→    def stop(self) -> None:
   811→   811→        """Stop background sync loop."""
   812→   812→        with self._lock:
   813→   813→            self._running = False
   814→   814→        if self._thread is not None:
   815→   815→            self._thread.join(timeout=self._refresh_interval * 2)
   816→   816→            self._thread = None
   817→   817→            logger.debug("TwinSync stopped")
   818→   818→
   819→   819→    def sync_once(self) -> Dict[str, Any]:
   820→   820→        """Perform a single sync cycle. Returns drift report."""
   821→   821→        new_model = self._twin.capture_state()
   822→   822→        drift = self._compute_drift(self._last_model, new_model)
   823→   823→
   824→   824→        with self._lock:
   825→   825→            self._last_model = new_model
   826→   826→            self._sync_count += 1
   827→   827→            if drift["has_drift"]:
   828→   828→                self._drift_history.append(drift)
   829→   829→                # Keep bounded: trim to last 50 when exceeding 100
   830→   830→                while len(self._drift_history) > 100:
   831→   831→                    self._drift_history = self._drift_history[-50:]
   832→   832→
   833→   833→        return drift
   834→   834→
   835→   835→    # ── Drift detection ───────────────────────────────────────────────
   836→   836→
   837→   837→    def get_drift_history(self) -> List[Dict[str, Any]]:
   838→   838→        """Return accumulated drift reports."""
   839→   839→        with self._lock:
   840→   840→            return list(self._drift_history)
   841→   841→
   842→   842→    def get_sync_stats(self) -> Dict[str, Any]:
   843→   843→        """Return sync statistics."""
   844→   844→        with self._lock:
   845→   845→            return {
   846→   846→                "sync_count": self._sync_count,
   847→   847→                "running": self._running,
   848→   848→                "refresh_interval": self._refresh_interval,
   849→   849→                "drift_events": len(self._drift_history),
   850→   850→            }
   851→   851→
   852→   852→    def _compute_drift(
   853→   853→        self, old: Optional[SystemModel], new: SystemModel
   854→   854→    ) -> Dict[str, Any]:
   855→   855→        """Compare two system models and report differences."""
   856→   856→        if old is None:
   857→   857→            return {
   858→   858→                "has_drift": False,
   859→   859→                "message": "No previous model to compare against (first snapshot).",
   860→   860→                "changes": [],
   861→   861→                "timestamp": new.captured_at,
   862→   862→            }
   863→   863→
   864→   864→        changes: List[Dict[str, Any]] = []
   865→   865→        for comp_name, new_comp in new.all_components().items():
   866→   866→            old_comp = old.all_components().get(comp_name)
   867→   867→            if old_comp is None:
   868→   868→                changes.append({
   869→   869→                    "component": comp_name,
   870→   870→                    "type": "added",
   871→   871→                    "detail": f"Component '{comp_name}' appeared.",
   872→   872→                })
   873→   873→                continue
   874→   874→
   875→   875→            # Status change
   876→   876→            if old_comp.status != new_comp.status:
   877→   877→                changes.append({
   878→   878→                    "component": comp_name,
   879→   879→                    "type": "status_change",
   880→   880→                    "from": old_comp.status,
   881→   881→                    "to": new_comp.status,
   882→   882→                    "detail": f"{comp_name}: {old_comp.status} → {new_comp.status}",
   883→   883→                })
   884→   884→
   885→   885→            # Metric changes
   886→   886→            old_metrics = old_comp.metrics
   887→   887→            new_metrics = new_comp.metrics
   888→   888→            for key in set(list(old_metrics.keys()) + list(new_metrics.keys())):
   889→   889→                old_val = old_metrics.get(key)
   890→   890→                new_val = new_metrics.get(key)
   891→   891→                if old_val != new_val:
   892→   892→                    # Skip noisy counters (counters always change)
   893→   893→                    if key.endswith("_count") or key in ("rss_mb", "free_mb"):
   894→   894→                        # Only flag if change is significant
   895→   895→                        if isinstance(old_val, (int, float)) and isinstance(new_val, (int, float)):
   896→   896→                            pct = abs(new_val - old_val) / max(abs(old_val), 0.001)
   897→   897→                            if pct < 0.1:  # less than 10% change
   898→   898→                                continue
   899→   899→                    changes.append({
   900→   900→                        "component": comp_name,
   901→   901→                        "type": "metric_change",
   902→   902→                        "metric": key,
   903→   903→                        "from": old_val,
   904→   904→                        "to": new_val,
   905→   905→                        "detail": (
   906→   906→                            f"{comp_name}.{key}: {old_val} → {new_val}"
   907→   907→                        ),
   908→   908→                    })
   909→   909→
   910→   910→            # Constraint changes
   911→   911→            old_constraints = old_comp.constraints
   912→   912→            new_constraints = new_comp.constraints
   913→   913→            for key in set(list(old_constraints.keys()) + list(new_constraints.keys())):
   914→   914→                if old_constraints.get(key) != new_constraints.get(key):
   915→   915→                    changes.append({
   916→   916→                        "component": comp_name,
   917→   917→                        "type": "constraint_change",
   918→   918→                        "metric": key,
   919→   919→                        "from": old_constraints.get(key),
   920→   920→                        "to": new_constraints.get(key),
   921→   921→                        "detail": (
   922→   922→                            f"{comp_name} constraint {key}: "
   923→   923→                            f"{old_constraints.get(key)} → {new_constraints.get(key)}"
   924→   924→                        ),
   925→   925→                    })
   926→   926→
   927→   927→        # Filter to status changes and significant metric changes
   928→   928→        significant = [
   929→   929→            c for c in changes
   930→   930→            if c["type"] in ("status_change", "constraint_change")
   931→   931→            or c.get("type") == "metric_change"
   932→   932→        ]
   933→   933→
   934→   934→        return {
   935→   935→            "has_drift": len(significant) > 0,
   936→   936→            "message": (
   937→   937→                f"{len(significant)} change(s) detected."
   938→   938→                if significant
   939→   939→                else "No significant drift."
   940→   940→            ),
   941→   941→            "changes": changes,
   942→   942→            "significant_changes": significant,
   943→   943→            "timestamp": new.captured_at,
   944→   944→        }
   945→   945→
   946→   946→    # ── Background loop ───────────────────────────────────────────────
   947→   947→
   948→   948→    def _sync_loop(self) -> None:
   949→   949→        """Background sync loop (runs in daemon thread)."""
   950→   950→        while self._running:
   951→   951→            try:
   952→   952→                self.sync_once()
   953→   953→            except Exception as exc:
   954→   954→                logger.debug("TwinSync error: %s", exc, exc_info=True)
   955→   955→            # Sleep in small increments so we can stop promptly
   956→   956→            for _ in range(int(self._refresh_interval * 10)):
   957→   957→                if not self._running:
   958→   958→                    break
   959→   959→                time.sleep(0.1)
   960→   960→
   961→   961→
   962→   962→# ═══════════════════════════════════════════════════════════════════════════
   963→   963→#  DigitalTwin — main facade
   964→   964→# ═══════════════════════════════════════════════════════════════════════════

   966→   966→
   967→   967→class DigitalTwin:
   968→   968→    """Virtual model of the running ReconPro system.
   969→   969→
   970→   970→    Provides meta-level operations:
   971→   971→    - ``capture_state()`` — snapshot current system state
   972→   972→    - ``predict_impact(change)`` — simulate a change before applying
   973→   973→    - ``detect_anomaly()`` — compare current state vs expected model
   974→   974→    - ``what_if(scenario)`` — run hypothetical scenarios
   975→   975→    - ``get_system_topology()`` — return component graph with health
   976→   976→    - ``get_capacity_model()`` — current capacity vs limits
   977→   977→

   979→   979→    -----

   984→   984→        result = twin.what_if("add_concurrent_scans:5")

   986→   986→
   987→   987→        anomalies = twin.detect_anomaly()
   988→   988→        print(anomalies)
   989→   989→    """
   990→   990→
   991→   991→    def __init__(self, persist: bool = True) -> None:
   992→   992→        self._persist = persist
   993→   993→        self._model: Optional[SystemModel] = None
   994→   994→        self._lock = threading.Lock()

   997→   997→        self._baseline: Optional[SystemModel] = None
   998→   998→
   999→   999→    # ── State capture ─────────────────────────────────────────────────
  1000→  1000→
  1001→  1001→    def capture_state(self) -> SystemModel:
  1002→  1002→        """Snapshot the current system state from live diagnostics and observability.
  1003→  1003→
  1004→  1004→        Gathers data from:
  1005→  1005→        - diagnostics.py (health_check, module_status, validate_config)
  1006→  1006→        - observability.py (HealthMonitor, MetricsCollector via TelemetryManager)
  1007→  1007→        - connection_pool.py (stats — if pool instance is reachable)
  1008→  1008→        - registry.py (module counts, broken modules)
  1009→  1009→        - filesystem (disk, memory, directory existence)
  1010→  1010→
  1011→  1011→        Returns a SystemModel suitable for simulation and anomaly detection.
  1012→  1012→        """
  1013→  1013→        model = SystemModel(captured_at=_utc_iso())
  1014→  1014→
  1015→  1015→        self._capture_engine(model)
  1016→  1016→        self._capture_scanner(model)
  1017→  1017→        self._capture_modules(model)
  1018→  1018→        self._capture_plugins(model)
  1019→  1019→        self._capture_network(model)
  1020→  1020→        self._capture_storage(model)
  1021→  1021→        self._capture_configuration(model)
  1022→  1022→
  1023→  1023→        with self._lock:
  1024→  1024→            self._model = model
  1025→  1025→            if self._baseline is None:
  1026→  1026→                self._baseline = copy.deepcopy(model)
  1027→  1027→
  1028→  1028→        if self._persist:
  1029→  1029→            self._save(model)
  1030→  1030→
  1031→  1031→        return model

  1048→  1048→        return self._simulation.simulate(model, change)

  1056→  1056→    # ── Anomaly detection ──────────────────────────────────────────────
  1057→  1057→
  1058→  1058→    def detect_anomaly(self) -> Dict[str, Any]:
  1059→  1059→        """Compare current state vs the baseline (first captured) model.
  1060→  1060→
  1061→  1061→        Detects unexpected state deviations:
  1062→  1062→        - Component status changes (healthy → degraded/critical)
  1063→  1063→        - Significant metric deviations (> 2× baseline or new critical values)
  1064→  1064→        - Constraint violations
  1065→  1065→        - New broken modules
  1066→  1066→        - Resource exhaustion
  1067→  1067→
  1068→  1068→        Returns a dict with:
  1069→  1069→            anomalies: list of detected anomalies
  1070→  1070→            anomaly_count: int
  1071→  1071→            severity: str ("none" | "low" | "medium" | "high" | "critical")
  1072→  1072→        """
  1073→  1073→        current = self._get_or_capture()
  1074→  1074→        if self._baseline is None:
  1075→  1075→            self._baseline = copy.deepcopy(current)
  1076→  1076→            return {
  1077→  1077→                "anomalies": [],
  1078→  1078→                "anomaly_count": 0,
  1079→  1079→                "severity": "none",
  1080→  1080→                "message": "Baseline established from current state.",
  1081→  1081→            }
  1082→  1082→
  1083→  1083→        anomalies: List[Dict[str, Any]] = []
  1084→  1084→
  1085→  1085→        for comp_name, curr_comp in current.all_components().items():
  1086→  1086→            base_comp = self._baseline.all_components().get(comp_name)
  1087→  1087→            if base_comp is None:
  1088→  1088→                continue
  1089→  1089→
  1090→  1090→            # Status regression
  1091→  1091→            status_rank = {"critical": 0, "degraded": 1, "healthy": 2, "unknown": 3, "disabled": 4}
  1092→  1092→            curr_rank = status_rank.get(curr_comp.status, 3)
  1093→  1093→            base_rank = status_rank.get(base_comp.status, 3)
  1094→  1094→            if curr_rank < base_rank:
  1095→  1095→                anomalies.append({
  1096→  1096→                    "component": comp_name,
  1097→  1097→                    "type": "status_regression",
  1098→  1098→                    "severity": "high" if curr_rank == 0 else "medium",
  1099→  1099→                    "baseline": base_comp.status,
  1100→  1100→                    "current": curr_comp.status,
  1101→  1101→                    "description": (
  1102→  1102→                        f"{comp_name} regressed from {base_comp.status} to {curr_comp.status}"
  1103→  1103→                    ),
  1104→  1104→                })
  1105→  1105→
  1106→  1106→            # Broken modules increase
  1107→  1107→            base_broken = base_comp.metrics.get("broken_modules", [])
  1108→  1108→            curr_broken = curr_comp.metrics.get("broken_modules", [])
  1109→  1109→            new_broken = set(curr_broken) - set(base_broken)
  1110→  1110→            if new_broken:
  1111→  1111→                anomalies.append({
  1112→  1112→                    "component": comp_name,
  1113→  1113→                    "type": "new_broken_modules",
  1114→  1114→                    "severity": "high",
  1115→  1115→                    "new_broken": sorted(new_broken),
  1116→  1116→                    "description": (
  1117→  1117→                        f"New broken module(s) detected: {sorted(new_broken)}"
  1118→  1118→                    ),
  1119→  1119→                })
  1120→  1120→
  1121→  1121→            # Metric anomalies — compare numeric metrics
  1122→  1122→            for metric_key, curr_val in curr_comp.metrics.items():
  1123→  1123→                if not isinstance(curr_val, (int, float)):
  1124→  1124→                    continue
  1125→  1125→                base_val = base_comp.metrics.get(metric_key)
  1126→  1126→                if not isinstance(base_val, (int, float)):
  1127→  1127→                    continue
  1128→  1128→                if base_val == 0:
  1129→  1129→                    continue
  1130→  1130→
  1131→  1131→                ratio = curr_val / base_val if base_val != 0 else float("inf")
  1132→  1132→                # Flag if metric more than doubled or halved
  1133→  1133→                if ratio > 2.0 or ratio < 0.5:
  1134→  1134→                    # Skip known-volatile metrics
  1135→  1135→                    if metric_key in ("scan_count", "request_count", "total_requests"):
  1136→  1136→                        continue
  1137→  1137→                    anomalies.append({
  1138→  1138→                        "component": comp_name,
  1139→  1139→                        "type": "metric_anomaly",
  1140→  1140→                        "severity": "medium",
  1141→  1141→                        "metric": metric_key,
  1142→  1142→                        "baseline_value": base_val,
  1143→  1143→                        "current_value": curr_val,
  1144→  1144→                        "ratio": round(ratio, 2),
  1145→  1145→                        "description": (
  1146→  1146→                            f"{comp_name}.{metric_key}: {base_val} → {curr_val} "
  1147→  1147→                            f"(ratio: {ratio:.1f}×)"
  1148→  1148→                        ),
  1149→  1149→                    })
  1150→  1150→
  1151→  1151→            # Constraint violations
  1152→  1152→            for ckey, cval in curr_comp.constraints.items():
  1153→  1153→                if isinstance(cval, (int, float)) and cval > 0:
  1154→  1154→                    metric_val = curr_comp.metrics.get(ckey)
  1155→  1155→                    if isinstance(metric_val, (int, float)) and metric_val > cval:
  1156→  1156→                        anomalies.append({
  1157→  1157→                            "component": comp_name,
  1158→  1158→                            "type": "constraint_violation",
  1159→  1159→                            "severity": "high",
  1160→  1160→                            "constraint": ckey,
  1161→  1161→                            "limit": cval,
  1162→  1162→                            "actual": metric_val,
  1163→  1163→                            "description": (
  1164→  1164→                                f"{comp_name}.{ckey} = {metric_val} "
  1165→  1165→                                f"exceeds limit {cval}"
  1166→  1166→                            ),
  1167→  1167→                        })
  1168→  1168→
  1169→  1169→        # Determine overall severity
  1170→  1170→        severity = "none"
  1171→  1171→        if anomalies:
  1172→  1172→            sev_ranks = {"critical": 0, "high": 1, "medium": 2, "low": 3}
  1173→  1173→            worst = min(sev_ranks.get(a["severity"], 3) for a in anomalies)
  1174→  1174→            severity = ["critical", "high", "medium", "low"][worst]
  1175→  1175→
  1176→  1176→        return {
  1177→  1177→            "anomalies": anomalies,
  1178→  1178→            "anomaly_count": len(anomalies),
  1179→  1179→            "severity": severity,
  1180→  1180→            "timestamp": current.captured_at,
  1181→  1181→        }
  1182→  1182→
  1183→  1183→    # ── Topology ──────────────────────────────────────────────────────
  1184→  1184→
  1185→  1185→    def get_system_topology(self) -> Dict[str, Any]:
  1186→  1186→        """Return the component dependency graph with health status.
  1187→  1187→
  1188→  1188→        The topology is a static model of how ReconPro components relate:
  1189→  1189→        - engine depends on scanner, modules, network, plugins
  1190→  1190→        - scanner depends on modules, registry
  1191→  1191→        - modules depend on network, plugins
  1192→  1192→        - plugins depend on storage (for hook files)
  1193→  1193→        - storage is a leaf node
  1194→  1194→        - network is a leaf node
  1195→  1195→        - configuration is a leaf node (read at startup)
  1196→  1196→
  1197→  1197→        Returns:
  1198→  1198→            Dict with 'nodes' and 'edges' lists.
  1199→  1199→        """
  1200→  1200→        model = self._get_or_capture()
  1201→  1201→
  1202→  1202→        nodes: List[Dict[str, Any]] = []
  1203→  1203→        edges: List[Dict[str, str]] = []
  1204→  1204→
  1205→  1205→        for comp_name, comp in model.all_components().items():
  1206→  1206→            nodes.append({
  1207→  1207→                "id": comp_name,
  1208→  1208→                "status": comp.status,
  1209→  1209→                "metrics_count": len(comp.metrics),
  1210→  1210→            })
  1211→  1211→
  1212→  1212→        # Dependency edges
  1213→  1213→        _EDGES = [
  1214→  1214→            ("engine", "scanner"),
  1215→  1215→            ("engine", "modules"),
  1216→  1216→            ("engine", "network"),
  1217→  1217→            ("engine", "plugins"),
  1218→  1218→            ("scanner", "modules"),
  1219→  1219→            ("scanner", "configuration"),
  1220→  1220→            ("modules", "network"),
  1221→  1221→            ("modules", "plugins"),
  1222→  1222→            ("plugins", "storage"),
  1223→  1223→            ("plugins", "configuration"),
  1224→  1224→        ]
  1225→  1225→        for src, dst in _EDGES:
  1226→  1226→            edges.append({"from": src, "to": dst})
  1227→  1227→
  1228→  1228→        # Identify impacted paths (if any node is degraded/critical)
  1229→  1229→        impacted_paths: List[List[str]] = []
  1230→  1230→        for node in nodes:
  1231→  1231→            if node["status"] in ("degraded", "critical"):
  1232→  1232→                # Find all nodes that depend on this one
  1233→  1233→                dependents = [e["from"] for e in edges if e["to"] == node["id"]]
  1234→  1234→                for dep in dependents:
  1235→  1235→                    impacted_paths.append([node["id"], dep])
  1236→  1236→
  1237→  1237→        return {
  1238→  1238→            "nodes": nodes,
  1239→  1239→            "edges": edges,
  1240→  1240→            "overall_status": model.overall_status(),
  1241→  1241→            "impacted_paths": impacted_paths,
  1242→  1242→            "captured_at": model.captured_at,
  1243→  1243→        }
  1244→  1244→
  1245→  1245→    # ── Capacity model ─────────────────────────────────────────────────
  1246→  1246→
  1247→  1247→    def get_capacity_model(self) -> Dict[str, Any]:
  1248→  1248→        """Return current capacity usage vs known limits.
  1249→  1249→
  1250→  1250→        Dimensions:
  1251→  1251→        - concurrency: active / max
  1252→  1252→        - memory: current MB / warning / critical thresholds
  1253→  1253→        - disk: usage % / warning / critical thresholds
  1254→  1254→        - modules: healthy / total
  1255→  1255→        - network: latency vs baseline
  1256→  1256→        - plugins: loaded count
  1257→  1257→        """
  1258→  1258→        model = self._get_or_capture()
  1259→  1259→
  1260→  1260→        # Concurrency
  1261→  1261→        active_scans = model.engine.metrics.get("active_scans", 0)
  1262→  1262→        max_conc = model.engine.constraints.get("max_concurrency", DEFAULT_MAX_WORKERS)
  1263→  1263→        conc_pct = (active_scans / max(max_conc, 1)) * 100
  1264→  1264→
  1265→  1265→        # Memory
  1266→  1266→        rss_mb = model.storage.metrics.get("rss_mb", 0)
  1267→  1267→        mem_warning = model.storage.constraints.get("memory_warning_mb", _MEMORY_WARNING_MB)
  1268→  1268→        mem_critical = model.storage.constraints.get("memory_critical_mb", _MEMORY_CRITICAL_MB)
  1269→  1269→        mem_pct = (rss_mb / mem_critical * 100) if mem_critical > 0 else 0
  1270→  1270→
  1271→  1271→        # Disk
  1272→  1272→        disk_pct = model.storage.metrics.get("usage_percent", 0)
  1273→  1273→        disk_free = model.storage.metrics.get("free_mb", 0)
  1274→  1274→
  1275→  1275→        # Modules
  1276→  1276→        total_modules = model.scanner.metrics.get("total_modules", 0)
  1277→  1277→        broken_modules = len(model.scanner.metrics.get("broken_modules", []))
  1278→  1278→        healthy_modules = max(total_modules - broken_modules, 0)
  1279→  1279→
  1280→  1280→        # Network latency
  1281→  1281→        avg_latency = model.network.metrics.get("avg_latency_ms", 0)
  1282→  1282→        latency_baseline = model.network.constraints.get(
  1283→  1283→            "latency_baseline_ms", _CONNECTION_LATENCY_BASELINE_MS
  1284→  1284→        )
  1285→  1285→        latency_ratio = avg_latency / latency_baseline if latency_baseline > 0 else 0
  1286→  1286→
  1287→  1287→        # Plugins
  1288→  1288→        plugin_count = model.plugins.metrics.get("plugin_count", 0)
  1289→  1289→
  1290→  1290→        # Overall capacity score (0-100, 100 = no stress)
  1291→  1291→        scores: List[float] = []
  1292→  1292→        scores.append(max(0, 100 - conc_pct))  # concurrency headroom
  1293→  1293→        scores.append(max(0, 100 - mem_pct))    # memory headroom
  1294→  1294→        scores.append(max(0, 100 - disk_pct))   # disk headroom
  1295→  1295→        if total_modules > 0:
  1296→  1296→            scores.append((healthy_modules / total_modules) * 100)  # module health
  1297→  1297→        if latency_baseline > 0 and avg_latency > 0:
  1298→  1298→            scores.append(max(0, 100 - (latency_ratio - 1) * 50))  # latency health
  1299→  1299→
  1300→  1300→        capacity_score = min(sum(scores) / max(len(scores), 1), 100.0)
  1301→  1301→
  1302→  1302→        return {
  1303→  1303→            "capacity_score": round(capacity_score, 1),
  1304→  1304→            "concurrency": {
  1305→  1305→                "active": active_scans,
  1306→  1306→                "max": max_conc,
  1307→  1307→                "utilisation_pct": round(conc_pct, 1),
  1308→  1308→                "status": _capacity_status(conc_pct),
  1309→  1309→            },
  1310→  1310→            "memory": {
  1311→  1311→                "rss_mb": round(rss_mb, 1),
  1312→  1312→                "warning_mb": mem_warning,
  1313→  1313→                "critical_mb": mem_critical,
  1314→  1314→                "utilisation_pct": round(mem_pct, 1),
  1315→  1315→                "status": _capacity_status(mem_pct),
  1316→  1316→            },
  1317→  1317→            "disk": {
  1318→  1318→                "usage_pct": round(disk_pct, 1),
  1319→  1319→                "free_mb": round(disk_free, 1),
  1320→  1320→                "warning_pct": _DISK_WARNING_PCT,
  1321→  1321→                "critical_pct": _DISK_CRITICAL_PCT,
  1322→  1322→                "status": _capacity_status(disk_pct),
  1323→  1323→            },
  1324→  1324→            "modules": {
  1325→  1325→                "total": total_modules,
  1326→  1326→                "healthy": healthy_modules,
  1327→  1327→                "broken": broken_modules,
  1328→  1328→                "health_pct": round((healthy_modules / max(total_modules, 1)) * 100, 1),
  1329→  1329→                "status": "healthy" if broken_modules == 0 else ("degraded" if broken_modules <= 3 else "critical"),
  1330→  1330→            },
  1331→  1331→            "network": {
  1332→  1332→                "avg_latency_ms": round(avg_latency, 1),
  1333→  1333→                "baseline_ms": latency_baseline,
  1334→  1334→                "ratio": round(latency_ratio, 2),
  1335→  1335→                "status": "healthy" if latency_ratio < 1.5 else ("degraded" if latency_ratio < 3.0 else "critical"),
  1336→  1336→            },
  1337→  1337→            "plugins": {
  1338→  1338→                "count": plugin_count,
  1339→  1339→                "status": "healthy",
  1340→  1340→            },
  1341→  1341→            "overall_status": model.overall_status(),
  1342→  1342→            "captured_at": model.captured_at,
  1343→  1343→        }

  1363→  1363→    # ── Persistence ────────────────────────────────────────────────────
  1364→  1364→
  1365→  1365→    def load(self) -> Optional[SystemModel]:
  1366→  1366→        """Load the last persisted model from disk."""
  1367→  1367→        try:
  1368→  1368→            if not DIGITAL_TWIN_FILE.exists():
