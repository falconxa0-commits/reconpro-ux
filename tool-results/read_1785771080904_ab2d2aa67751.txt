   100→   100→class FindingDetailModal(ModalScreen):
   101→   101→    """Modal overlay showing full finding details."""
   102→   102→
   103→   103→    def __init__(self, finding: Dict[str, Any]) -> None:
   104→   104→        super().__init__()
   105→   105→        self.finding = finding
   106→   106→
   107→   107→    def compose(self) -> ComposeResult:
   108→   108→        f = self.finding
   109→   109→        sev = f.get("severity", "info").lower()
   110→   110→        sev_color = SEV_STYLES.get(sev, "white")
   111→   111→        title = f.get("title", "Unknown")
   112→   112→        category = f.get("category", "")
   113→   113→        module = f.get("module", "")
   114→   114→        pts = f.get("points_deducted", 0)
   115→   115→        evidence = f.get("evidence", "")
   116→   116→        remediation = f.get("remediation", "No remediation available.")
   117→   117→        description = f.get("description", "")
   118→   118→        asset = f.get("asset", "")
   119→   119→        related_cves = f.get("related_cves", [])
   120→   120→
   121→   121→        lines: List[str] = []
   122→   122→        lines.append(f"[{sev_color}][{sev.upper()}] {title}[/]")
   123→   123→        lines.append("")
   124→   124→        if asset:
   125→   125→            lines.append(f"[dim]Asset:[/] [cyan]{asset}[/]")
   126→   126→        if category:
   127→   127→            lines.append(f"[dim]Category:[/] {category}")
   128→   128→        if module:
   129→   129→            lines.append(f"[dim]Module:[/] {module}")
   130→   130→        lines.append(f"[dim]Points:[/] -{pts}")
   131→   131→        if related_cves:
   132→   132→            cve_ids = ", ".join(
   133→   133→                c.get("cve_id", "") for c in related_cves[:5] if c.get("cve_id")
   134→   134→            )
   135→   135→            if cve_ids:
   136→   136→                lines.append(f"[dim]CVEs:[/] [red]{cve_ids}[/]")
   137→   137→        lines.append("")
   138→   138→        if description:
   139→   139→            lines.append(f"[bold]Description:[/]")
   140→   140→            lines.append(f"[dim]{description}[/]")
   141→   141→            lines.append("")
   142→   142→        if evidence:
   143→   143→            lines.append(f"[bold]Evidence:[/]")
   144→   144→            lines.append(f"[dim]{evidence}[/]")
   145→   145→            lines.append("")
   146→   146→        lines.append(f"[bold]{GREEN}Remediation:[/]")
   147→   147→        lines.append(f"{GREEN}{remediation}[/]")
   148→   148→
   149→   149→        content = "\n".join(lines)
   150→   150→
   151→   151→        with Vertical(classes="modal-container"):
   152→   152→            yield Label(content, classes="modal-content")
   153→   153→            with Horizontal(classes="modal-actions"):
   154→   154→                yield Button("Close", variant="primary", id="close-modal")
   155→   155→
   156→   156→    def on_button_pressed(self, event: Button.Pressed) -> None:
   157→   157→        if event.button.id == "close-modal":
   158→   158→            self.dismiss()

   162→   162→# Boot Screen Widget
   163→   163→# ════════════════════════════════════════════════════════════════════════════════
   164→   164→
   165→   165→
   166→   166→class BootScreen(Widget):
   167→   167→    """Full-screen animated boot sequence."""
   168→   168→
   169→   169→    def __init__(self) -> None:
   170→   170→        super().__init__()
   171→   171→        self._step = 0
   172→   172→        self._spinner_idx = 0
   173→   173→        self._module_idx = 0
   174→   174→        self._module_names = list(MODULE_NAMES.values())
   175→   175→        self._done = False
   176→   176→        self._timer: Optional[Timer] = None
   177→   177→
   178→   178→    def compose(self) -> ComposeResult:
   179→   179→        with Vertical(classes="boot-container"):
   180→   180→            yield Static("", id="boot-line-1", classes="boot-title")
   181→   181→            yield Static("", id="boot-line-2", classes="boot-line")
   182→   182→            yield Static("", id="boot-line-3", classes="boot-line")
   183→   183→            yield Static("", id="boot-line-4", classes="boot-line")
   184→   184→
   185→   185→    def on_mount(self) -> None:
   186→   186→        self._timer = self.set_interval(0.04, self._tick)
   187→   187→
   188→   188→    def _tick(self) -> None:
   189→   189→        app = self.app
   190→   190→        if not isinstance(app, NexusApp):
   191→   191→            return
   192→   192→
   193→   193→        line1 = app.query_one("#boot-line-1", Static)
   194→   194→        line2 = app.query_one("#boot-line-2", Static)
   195→   195→        line3 = app.query_one("#boot-line-3", Static)
   196→   196→        line4 = app.query_one("#boot-line-4", Static)
   197→   197→
   198→   198→        if self._step == 0:
   199→   199→            # Letter-by-letter title
   200→   200→            title = "RECONPRO NEXUS"
   201→   201→            chars = title[: self._spinner_idx]
   202→   202→            self._spinner_idx += 1
   203→   203→            line1.update(f"[{CYAN} bold]{chars}[/]")
   204→   204→            if self._spinner_idx > len(title):
   205→   205→                self._spinner_idx = 0
   206→   206→                self._step = 1
   207→   207→                line1.update(f"[{CYAN} bold]{title}[/]  [{DIM_CYAN}]v{VERSION}[/]")
   208→   208→
   209→   209→        elif self._step == 1:
   210→   210→            # Spinner for "Initializing agent..."
   211→   211→            frame = SPINNER_FRAMES[self._spinner_idx % len(SPINNER_FRAMES)]
   212→   212→            self._spinner_idx += 1
   213→   213→            line2.update(f"  [{CYAN}]{frame}[/] [{DIM_CYAN}]Initializing agent...[/]")
   214→   214→            if self._spinner_idx > 30:
   215→   215→                self._spinner_idx = 0
   216→   216→                self._step = 2
   217→   217→                line2.update(f"  [{GREEN}]✓[/] [{DIM_CYAN}]Agent initialized.[/]")
   218→   218→
   219→   219→        elif self._step == 2:
   220→   220→            # Module names appearing one by one
   221→   221→            if self._module_idx < len(self._module_names):
   222→   222→                visible = self._module_names[: self._module_idx + 1]
   223→   223→                self._module_idx += 1
   224→   224→                modules_str = "  ".join(f"[{CYAN}]○[/] [{DIM_CYAN}]{m}[/]" for m in visible)
   225→   225→                remaining = len(self._module_names) - self._module_idx
   226→   226→                if remaining > 0:
   227→   227→                    modules_str += f"  [{DIM_CYAN}]+{remaining} more...[/]"
   228→   228→                frame = SPINNER_FRAMES[self._spinner_idx % len(SPINNER_FRAMES)]
   229→   229→                self._spinner_idx += 1
   230→   230→                line3.update(
   231→   231→                    f"  [{CYAN}]{frame}[/] [{DIM_CYAN}]Loading {len(self._module_names)} modules...[/]\n  {modules_str}"
   232→   232→                )
   233→   233→            else:
   234→   234→                all_mods = "  ".join(
   235→   235→                    f"[{GREEN}]●[/] [{DIM_CYAN}]{m}[/]" for m in self._module_names
   236→   236→                )
   237→   237→                line3.update(
   238→   238→                    f"  [{GREEN}]✓[/] [{DIM_CYAN}]All {len(self._module_names)} modules loaded.[/]\n  {all_mods}"
   239→   239→                )
   240→   240→                self._step = 3
   241→   241→                self._spinner_idx = 0
   242→   242→
   243→   243→        elif self._step == 3:
   244→   244→            # "Ready." with flash
   245→   245→            if self._spinner_idx == 0:
   246→   246→                line4.update(f"[{YELLOW} bold]  ▶  READY.[/]")
   247→   247→            elif self._spinner_idx == 6:
   248→   248→                line4.update(f"[{GREEN} bold]  ▶  READY.[/]")
   249→   249→            elif self._spinner_idx >= 10:
   250→   250→                self._done = True
   251→   251→                if self._timer:
   252→   252→                    self._timer.stop()
   253→   253→                self.app.call_from_thread(self.app._finish_boot)
   254→   254→            self._spinner_idx += 1
   255→   255→

   258→   258→# Module Status Cell Widget
   259→   259→# ════════════════════════════════════════════════════════════════════════════════
   260→   260→
   261→   261→
   262→   262→class ModuleCell(Static):
   263→   263→    """Single module status cell in the grid."""
   264→   264→
   265→   265→    def __init__(self, module_id: str, name: str) -> None:
   266→   266→        super().__init__()
   267→   267→        self.module_id = module_id
   268→   268→        self.module_name = name
   269→   269→        self.status = "idle"  # idle | scanning | done | error
   270→   270→        self.finding_count = 0
   271→   271→        self._spinner_idx = 0
   272→   272→        self._timer: Optional[Timer] = None
   273→   273→
   274→   274→    def on_mount(self) -> None:
   275→   275→        self._update_display()
   276→   276→        self._timer = self.set_interval(0.1, self._spin)
   277→   277→
   278→   278→    def _spin(self) -> None:
   279→   279→        if self.status == "scanning":
   280→   280→            self._spinner_idx = (self._spinner_idx + 1) % len(SPINNER_FRAMES)
   281→   281→            self._update_display()
   282→   282→
   283→   283→    def _update_display(self) -> None:
   284→   284→        name_short = self.module_name[:14]
   285→   285→        if self.status == "idle":
   286→   286→            icon = f"[{DIM_CYAN}]○[/]"
   287→   287→            status_text = f"[{DIM_CYAN}]idle[/]"
   288→   288→        elif self.status == "scanning":
   289→   289→            frame = SPINNER_FRAMES[self._spinner_idx]
   290→   290→            icon = f"[{CYAN}]{frame}[/]"
   291→   291→            status_text = f"[{CYAN}]scanning[/]"
   292→   292→        elif self.status == "done":
   293→   293→            icon = f"[{GREEN}]✓[/]"
   294→   294→            status_text = f"[{GREEN}]done ({self.finding_count})[/]"
   295→   295→        elif self.status == "error":
   296→   296→            icon = f"[{RED}]✗[/]"
   297→   297→            status_text = f"[{RED}]error[/]"
   298→   298→        else:
   299→   299→            icon = "○"
   300→   300→            status_text = "idle"
   301→   301→
   302→   302→        self.update(
   303→   303→            f"{icon} [{CYAN} bold]{name_short}[/]\n  {status_text}"
   304→   304→        )
   305→   305→
   306→   306→    def set_status(self, status: str, count: int = 0) -> None:
   307→   307→        self.status = status
   308→   308→        self.finding_count = count
   309→   309→        self._update_display()
   310→   310→
   311→   311→    def on_unmount(self) -> None:
   312→   312→        if self._timer:
   313→   313→            self._timer.stop()
   314→   314→

   317→   317→# Pulsing Status Dot Widget
   318→   318→# ════════════════════════════════════════════════════════════════════════════════
   319→   319→
   320→   320→
   321→   321→class StatusDot(Static):
   322→   322→    """Pulsing dot indicator for scan status."""
   323→   323→
   324→   324→    scanning: reactive[bool] = reactive(False)
   325→   325→
   326→   326→    def __init__(self) -> None:
   327→   327→        super().__init__("")
   328→   328→        self._phase = 0
   329→   329→
   330→   330→    def on_mount(self) -> None:
   331→   331→        self.set_interval(0.3, self._pulse)
   332→   332→
   333→   333→    def watch_scanning(self, old: bool, new: bool) -> None:
   334→   334→        # Update visual state based on scanning status
   335→   335→        if new:
   336→   336→            self.update(f"[{self._get_pulse_color()}]●[/]")
   337→   337→        else:
   338→   338→            self.update(f"[{GREEN}]●[/]")
   339→   339→
   340→   340→    def _get_pulse_color(self) -> str:
   341→   341→        return ["#003322", "#006644", CYAN, "#006644"][self._phase] if self.scanning else GREEN
   342→   342→
   343→   343→    def _pulse(self) -> None:
   344→   344→        self._phase = (self._phase + 1) % 4
   345→   345→        if self.scanning:
   346→   346→            brightness = self._get_pulse_color()
   347→   347→            self.update(f"[{brightness}]●[/]")
   348→   348→        else:
   349→   349→            self.update(f"[{GREEN}]●[/]")

   841→   841→    def on_key(self, event: Key) -> None:
   842→   842→        """Handle key events — boot skip + completer + history + vim-nav."""
   843→   843→        # Skip boot on ANY key press during boot sequence
   844→   844→        if not self._boot_done and not self._boot_skipped:
   845→   845→            self._boot_skipped = True
   846→   846→            try:
   847→   847→                boot = self.query_one("#boot-screen", BootScreen)
   848→   848→                if boot._timer:
   849→   849→                    boot._timer.stop()
   850→   850→            except NoMatches:
   851→   851→                pass
   852→   852→            self._finish_boot()
   853→   853→            return
   854→   854→
   855→   855→        # ── Number keys: quick jump to panels (0=input, 1=chat, 2=findings, 3=modules) ──
   856→   856→        if event.key in ("0", "1", "2", "3") and not self._input_focused():
   857→   857→            jump_map = {"0": "#command-input", "1": "#chat-log", "2": "#findings-feed", "3": "#module-grid"}
   858→   858→            # Skip panel jump to collapsed panels
   859→   859→            target_sel = jump_map[event.key]
   860→   860→            if event.key == "1" and self._left_collapsed:
   861→   861→                target_sel = "#findings-feed"
   862→   862→            if event.key == "3" and self._modules_collapsed:
   863→   863→                target_sel = "#findings-feed"
   864→   864→            self._focus_widget(target_sel)
   865→   865→            event.stop()
   866→   866→            return
   867→   867→
   868→   868→        # ── Phase D: [ and ] to toggle panel collapse ──
   869→   869→        if event.key == "[" and not self._input_focused():
   870→   870→            self._toggle_left_panel()
   871→   871→            event.stop()
   872→   872→            return
   873→   873→        if event.key == "]" and not self._input_focused():
   874→   874→            self._toggle_right_panel()
   875→   875→            event.stop()
   876→   876→            return
   877→   877→        if event.key == "=" and not self._input_focused():
   878→   878→            self._toggle_modules()
   879→   879→            event.stop()
   880→   880→            return
   881→   881→
   882→   882→        # ── Tab: completer accept/cycle / quick-pick on empty / cycle focus ──
   883→   883→        if event.key == "tab":
   884→   884→            try:
   885→   885→                inp = self.query_one("#command-input", Input)
   886→   886→                comp = self.query_one("#cmd-completer", CommandCompleter)
   887→   887→                if inp.has_focus:
   888→   888→                    # Completer visible → accept top suggestion
   889→   889→                    if comp.visible:
   890→   890→                        accepted = comp.accept_top()
   891→   891→                        if accepted:
   892→   892→                            self._apply_completion(inp, comp, accepted)
   893→   893→                        event.stop()
   894→   894→                        return
   895→   895→                    # Empty input → quick-pick menu
   896→   896→                    if not inp.value.strip():
   897→   897→                        comp.show_quick_pick()
   898→   898→                        event.stop()
   899→   899→                        return
   900→   900→            except NoMatches:
   901→   901→                pass
   902→   902→            # Fall through to default Tab behavior (cycle focus)
   903→   903→            return
   904→   904→
   905→   905→        # ── Escape: dismiss completer, close modal, or jump to input ──
   906→   906→        if event.key == "escape":
   907→   907→            try:
   908→   908→                comp = self.query_one("#cmd-completer", CommandCompleter)
   909→   909→                if comp.visible:
   910→   910→                    comp.hide()
   911→   911→                    event.stop()
   912→   912→                    return
   913→   913→            except NoMatches:
   914→   914→                pass
   915→   915→            # If any modal screen is open, Escape will dismiss it via Textual
   916→   916→            # Otherwise jump back to input
   917→   917→            if not self._input_focused():
   918→   918→                self._focus_input()
   919→   919→                event.stop()
   920→   920→                return
   921→   921→
   922→   922→        # ── Up/Down in completer: cycle suggestions ──
   923→   923→        if event.key in ("up", "down"):
   924→   924→            try:
   925→   925→                comp = self.query_one("#cmd-completer", CommandCompleter)
   926→   926→                inp = self.query_one("#command-input", Input)
   927→   927→                if comp.visible and inp.has_focus:
   928→   928→                    direction = -1 if event.key == "up" else 1
   929→   929→                    comp.cycle_selection(direction)
   930→   930→                    event.stop()
   931→   931→                    return
   932→   932→            except NoMatches:
   933→   933→                pass
   934→   934→
   935→   935→        # ── j/k vim-style navigation: findings feed + chat log ──
   936→   936→        if event.key in ("j", "k"):
   937→   937→            # Findings feed navigation
   938→   938→            try:
   939→   939→                feed = self.query_one("#findings-feed", RichLog)
   940→   940→                if feed.has_focus and self._findings_list:
   941→   941→                    self._vim_navigate_findings(event.key)
   942→   942→                    event.stop()
   943→   943→                    return
   944→   944→            except NoMatches:
   945→   945→                pass
   946→   946→            # Chat log scroll (j=down, k=up)
   947→   947→            try:
   948→   948→                chat = self.query_one("#chat-log", RichLog)
   949→   949→                if chat.has_focus:
   950→   950→                    if event.key == "j":
   951→   951→                        chat.scroll_relative(3)
   952→   952→                    else:
   953→   953→                        chat.scroll_relative(-3)
   954→   954→                    event.stop()
   955→   955→                    return
   956→   956→            except NoMatches:
   957→   957→                pass
   958→   958→
   959→   959→        # ── Arrow keys on module grid: navigate cells ──
   960→   960→        if event.key in ("up", "down", "left", "right"):
   961→   961→            try:
