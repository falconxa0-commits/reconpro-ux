  2416→    @work(exclusive=False, thread=True)
  2417→    def _run_scan_worker(
  2418→        self,
  2419→        target: str,
  2420→        modules: Optional[List[str]] = None,
  2421→        is_local: bool = False,
  2422→    ) -> None:
  2423→        """Run a scan in a background thread."""
  2424→        self.call_from_thread(self.is_scanning.set, True)
  2425→
  2426→        # Set modules to scanning + push module-specific hints
  2427→        if modules:
  2428→            for m in modules:
  2429→                if m in self._module_cells:
  2430→                    self.call_from_thread(self._module_cells[m].set_status, "scanning")
  2431→                    self.call_from_thread(self._push_module_scanning_hint, m)
  2432→        else:
  2433→            all_mods = (
  2434→                list(MODULE_NAMES.keys())
  2435→                if is_local
  2436→                else [k for k in MODULE_NAMES if k not in ("host", "dev", "doctor")]
  2437→            )
  2438→            for m in all_mods:
  2439→                if m in self._module_cells:
  2440→                    self.call_from_thread(self._module_cells[m].set_status, "scanning")
  2441→
  2442→        try:
  2443→            from .scanner import scan, audit_scan
  2444→            from .history import save_scan
  2445→
  2446→            self.call_from_thread(
  2447→                self._chat, f"[{DIM_CYAN}]  ⏳ Scan in progress...[/]"
  2448→            )
  2449→
  2450→            if is_local:
  2451→                result = audit_scan(target=target, modules=modules)
  2452→            else:
  2453→                result = scan(target=target, modules=modules)
  2454→
  2455→            # Process findings
  2456→            data = result.to_dict()
  2457→            self.call_from_thread(self._on_scan_complete, data, modules, is_local)
  2458→
  2459→            # Save to history
  2460→            try:
  2461→                save_scan(data, label="nexus")
  2462→            except Exception:
  2463→                pass
  2464→
  2465→        except Exception as e:
  2466→            self.call_from_thread(self._chat, f"[{RED}]  ✗ Scan error: {e}[/]")
  2467→            # Phase C: push error_state hints
  2468→            self.call_from_thread(self._push_error_hint)
  2469→            if modules:
  2470→                for m in modules:
  2471→                    if m in self._module_cells:
  2472→                        self.call_from_thread(self._module_cells[m].set_status, "error")
  2473→            else:
  2474→                self.call_from_thread(self._set_all_modules_status, "error")
  2475→
  2476→        finally:
  2477→            self.call_from_thread(self.is_scanning.set, False)
  2478→

  2510→    @work(exclusive=False, thread=True)
  2511→    def _run_generic_worker(
  2512→        self,
  2513→        label: str,
  2514→        modules: Optional[List[str]] = None,
  2515→        target: str = "",
  2516→        is_local: bool = False,
  2517→    ) -> None:
  2518→        """Generic background worker for Phase C extended commands.
  2519→
  2520→        Reuses the scan infrastructure for commands like passive, fuzzer,
  2521→        cve, profile, netmap, defense, compliance, delta, benchmark,
  2522→        iac, container, ast, cloud-recon.
  2523→        """
  2524→        self.call_from_thread(self.is_scanning.set, True)
  2525→
  2526→        # Set specified modules to scanning
  2527→        if modules:
  2528→            for m in modules:
  2529→                if m in self._module_cells:
  2530→                    self.call_from_thread(self._module_cells[m].set_status, "scanning")
  2531→
  2532→        try:
  2533→            self.call_from_thread(
  2534→                self._chat, f"[{DIM_CYAN}]  ⏳ {label} in progress...[/]"
  2535→            )
  2536→
  2537→            # Try to run as a scan if we have a target and modules
  2538→            if target and target not in ("defense", "compliance", "delta", "benchmark") and modules:
  2539→                from .scanner import scan, audit_scan
  2540→
  2541→                if is_local:
  2542→                    result = audit_scan(target=target, modules=modules)
  2543→                else:
  2544→                    result = scan(target=target, modules=modules)
  2545→
  2546→                data = result.to_dict()
  2547→                self.call_from_thread(self._process_scan_data, data, modules_list=modules)
  2548→            elif target and target not in ("defense", "compliance", "delta", "benchmark"):
  2549→                # Full scan with target
  2550→                from .scanner import scan, audit_scan
  2551→
  2552→                if is_local:
  2553→                    result = audit_scan(target=target)
  2554→                else:
  2555→                    result = scan(target=target)
  2556→
  2557→                data = result.to_dict()
  2558→                self.call_from_thread(self._process_scan_data, data, modules_list=None)
  2559→            else:
  2560→                # Non-scan operations (defense, compliance, delta, benchmark)
  2561→                self.call_from_thread(
  2562→                    self._chat,
  2563→                    f"[{GREEN}]  ✓ {label} operation complete.[/]",
  2564→                )
  2565→
  2566→        except ImportError:
  2567→            self.call_from_thread(
  2568→                self._chat,
  2569→                f"[{YELLOW}]  ⚠ {label} requires additional modules. Use [cyan]pip install reconpro[full][/][/]",
  2570→            )
  2571→        except Exception as e:
  2572→            self.call_from_thread(self._chat, f"[{RED}]  ✗ {label} error: {e}[/]")
  2573→            self.call_from_thread(self._push_error_hint)
  2574→            if modules:
  2575→                for m in modules:
  2576→                    if m in self._module_cells:
  2577→                        self.call_from_thread(self._module_cells[m].set_status, "error")
  2578→
  2579→        finally:
  2580→            self.call_from_thread(self.is_scanning.set, False)
  2581→            if modules:
  2582→                for m in modules:
  2583→                    if m in self._module_cells:
  2584→                        self.call_from_thread(self._module_cells[m].set_status, "done")
  2585→

  2909→    def on_click(self, event: Click) -> None:
  2910→        """Open finding detail modal for the clicked finding.
  2911→
  2912→        Uses the vertical click position to determine which finding index
  2913→        the user clicked on, mapping it to _findings_list.
  2914→        """
  2915→        try:
  2916→            if (
  2917→                isinstance(event.widget, RichLog)
  2918→                and event.widget.id == "findings-feed"
  2919→                and self._findings_list
  2920→            ):
  2921→                # Get the line offset to determine which finding was clicked.
  2922→                # Each finding entry is 1 line in the RichLog.
  2923→                # We use the y coordinate relative to the widget to estimate the index.
  2924→                region = event.widget.region
  2925→                if region is not None:
  2926→                    # Calculate which line was clicked relative to the scroll position
  2927→                    scroll_offset = event.widget.scroll_y
  2928→                    click_y = event.y - region.y
  2929→                    # Each finding takes 1 line; estimate index from click position
  2930→                    # Account for scroll position
  2931→                    line_height = 1
  2932→                    if hasattr(event.widget, "_line_height"):
  2933→                        line_height = event.widget._line_height
  2934→                    clicked_index = int((click_y + scroll_offset * line_height) / line_height)
  2935→                    # Clamp to valid range
  2936→                    clicked_index = max(0, min(clicked_index, len(self._findings_list) - 1))
  2937→                    finding = self._findings_list[clicked_index]
  2938→                else:
  2939→                    finding = self._findings_list[-1]
  2940→                self.push_screen(FindingDetailModal(finding))
  2941→        except (NoMatches, Exception):
  2942→            pass
