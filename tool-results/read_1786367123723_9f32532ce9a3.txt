  1323→def _findings_to_dicts(findings: Any) -> List[Dict[str, Any]]:
  1324→    """Normalize findings (list of dicts or objects) to list of dicts."""
  1325→    result: List[Dict[str, Any]] = []
  1326→    for f in findings:
  1327→        if isinstance(f, dict):
  1328→            result.append(f)
  1329→        elif hasattr(f, "to_dict"):
  1330→            result.append(f.to_dict())
  1331→        elif hasattr(f, "__dict__"):
  1332→            result.append(vars(f))
  1333→        else:
  1334→            result.append({"title": str(f)})
  1335→    return result
  1336→

  1343→    "severity", "module", "category", "title", "description",
  1344→    "evidence", "asset", "points_deducted", "remediation",
  1345→    "cvss_score", "cvss_vector",

  1352→def generate_csv_report(
  1353→    findings: Any,
  1354→    output_path: str,
  1355→    score: Optional[float] = None,
  1356→) -> str:
  1357→    """Export findings to CSV (pure stdlib).
  1358→
  1359→    Parameters
  1360→    ----------
  1361→    findings : list of dict-like
  1362→        Raw findings as produced by scanner modules.
  1363→    output_path : str
  1364→        Destination file path.
  1365→    score : float, optional
  1366→        Overall score (written as comment in header area — CSV has no
  1367→        native comment mechanism, so it is omitted from the CSV itself).
  1368→
  1369→    Returns
  1370→    -------
  1371→    str
  1372→        Absolute path of the written file.
  1373→    """
  1374→    rows = _findings_to_dicts(findings)
  1375→    path = Path(output_path)
  1376→    path.parent.mkdir(parents=True, exist_ok=True)
  1377→
  1378→    with open(str(path), "w", newline="", encoding="utf-8") as fh:
  1379→        writer = csv.DictWriter(fh, fieldnames=_CSV_COLUMNS, extrasaction="ignore")
  1380→        writer.writeheader()
  1381→        for row in rows:
  1382→            writer.writerow({col: row.get(col, "") for col in _CSV_COLUMNS})
  1383→
  1384→    return str(path.resolve())

  1398→def generate_sarif_report(
  1399→    findings: Any,
  1400→    output_path: str,
  1401→    target: str = "",
  1402→) -> str:
  1403→    """Export findings to SARIF v2.1.0 (GitHub Code Scanning compatible).
  1404→
  1405→    Parameters
  1406→    ----------
  1407→    findings : list of dict-like
  1408→    output_path : str
  1409→    target : str
  1410→        Target identifier included in run properties.
  1411→
  1412→    Returns
  1413→    -------
  1414→    str
  1415→        Absolute path of the written file.
  1416→    """
  1417→    rows = _findings_to_dicts(findings)
  1418→    path = Path(output_path)
  1419→    path.parent.mkdir(parents=True, exist_ok=True)
  1420→
  1421→    # Build unique rules from findings
  1422→    rules: List[Dict[str, Any]] = []
  1423→    seen_rules: Dict[str, int] = {}
  1424→    for f in rows:
  1425→        rule_id = f.get("category", f.get("title", "RP-0000"))
  1426→        rule_id = rule_id[:100] if len(rule_id) > 100 else rule_id
  1427→        if rule_id not in seen_rules:
  1428→            idx = len(rules)
  1429→            seen_rules[rule_id] = idx
  1430→            rules.append({
  1431→                "id": rule_id,
  1432→                "shortDescription": {
  1433→                    "text": f.get("title", rule_id),
  1434→                },
  1435→                "helpUri": f.get("remediation", ""),
  1436→                "properties": {
  1437→                    "security-severity": str(f.get("cvss_score", "")),
  1438→                },
  1439→            })
  1440→
  1441→    results: List[Dict[str, Any]] = []
  1442→    for f in rows:
  1443→        severity = _SARIF_SEVERITY_MAP.get(
  1444→            str(f.get("severity", "info")).lower(), "note"
  1445→        )
  1446→        rule_id = f.get("category", f.get("title", "RP-0000"))
  1447→        rule_id = rule_id[:100] if len(rule_id) > 100 else rule_id
  1448→        rule_index = seen_rules.get(rule_id, 0)
  1449→
  1450→        asset = f.get("asset", target)
  1451→        location: Dict[str, Any] = {
  1452→            "physicalLocation": {
  1453→                "artifactLocation": {
  1454→                    "uri": asset,
  1455→                },
  1456→            },
  1457→        }
  1458→
  1459→        results.append({
  1460→            "ruleId": rule_id,
  1461→            "ruleIndex": rule_index,
  1462→            "level": severity,
  1463→            "message": {
  1464→                "text": f.get("description", ""),
  1465→            },
  1466→            "locations": [location],
  1467→            "properties": {
  1468→                "points_deducted": f.get("points_deducted", 0),
  1469→                "module": f.get("module", ""),
  1470→                "evidence": f.get("evidence", "")[:500] if f.get("evidence") else "",
  1471→            },
  1472→        })
  1473→
  1474→    sarif: Dict[str, Any] = {
  1475→        "$schema": "https://raw.githubusercontent.com/oasis-tcs/sarif-spec/master/Schemata/sarif-schema-2.1.0.json",
  1476→        "version": "2.1.0",
  1477→        "runs": [
  1478→            {
  1479→                "tool": {
  1480→                    "driver": {
  1481→                        "name": "ReconPro",
  1482→                        "version": "9.0.0",
  1483→                        "rules": rules,
  1484→                    },
  1485→                },
  1486→                "results": results,
  1487→                "properties": {
  1488→                    "target": target,
  1489→                },
  1490→            }
  1491→        ],
  1492→    }
  1493→
  1494→    with open(str(path), "w", encoding="utf-8") as fh:
  1495→        json.dump(sarif, fh, indent=2, ensure_ascii=False)
  1496→        fh.write("\n")
  1497→
  1498→    return str(path.resolve())

  1668→                lines.append("")
  1669→            if f.get("remediation"):
  1670→                lines.append(f"**Remediation:** {f.get('remediation')}")
  1671→                lines.append("")
  1672→            lines.append("---")
  1673→            lines.append("")

  1675→    lines.append("---")
  1676→    lines.append("*Generated by ReconPro v9.0.0*  ")

  1685→# ── PDF (printable HTML) ───────────────────────────────────────────────

  1687→
  1688→def generate_pdf_report(
  1689→    findings: Any,
  1690→    output_path: str,
  1691→    score: Optional[float] = None,
  1692→    target: str = "",
  1693→) -> str:
  1694→    """Generate a printable-HTML report styled for PDF output.
  1695→
  1696→    Creates a self-contained HTML file with print-optimized CSS.
  1697→    The file uses .html extension internally so browsers can render it,
  1698→    but the output_path is respected as given by the caller.
  1699→
  1700→    Parameters
  1701→    ----------
  1702→    findings : list of dict-like
  1703→    output_path : str
  1704→    score : float, optional
  1705→    target : str, optional
  1706→
  1707→    Returns
  1708→    -------
  1709→    str
  1710→        Absolute path of the written file.
  1711→    """
  1712→    rows = _findings_to_dicts(findings)
  1713→    path = Path(output_path)
  1714→    path.parent.mkdir(parents=True, exist_ok=True)
  1715→
  1716→    # If caller passed a .pdf extension, use .html instead
  1717→    actual_path = path
  1718→    if path.suffix.lower() == ".pdf":
  1719→        actual_path = path.with_suffix(".html")
  1720→
  1721→    rows.sort(key=lambda f: _SEVERITY_ORDER.get(str(f.get("severity", "info")).lower(), 0), reverse=True)
  1722→
  1723→    # Severity counts
  1724→    counts: Dict[str, int] = {}
  1725→    for f in rows:
  1726→        s = str(f.get("severity", "info")).lower()
  1727→        counts[s] = counts.get(s, 0) + 1
  1728→
  1729→    # ASCII gauge
  1730→    gauge_width = 40
  1731→    filled = 0
  1732→    if score is not None:
  1733→        filled = int(round(score / 100 * gauge_width))
  1734→    gauge_bar = "█" * filled + "░" * (gauge_width - filled)
  1735→    score_display = f"{score:.1f}" if score is not None else "N/A"
  1736→
  1737→    # Build findings rows
  1738→    finding_rows_html = ""
  1739→    for idx, f in enumerate(rows, start=1):
  1740→        sev = str(f.get("severity", "info")).lower()
  1741→        title = f.get("title", "Untitled")
  1742→        desc = f.get("description", "")
  1743→        rem = f.get("remediation", "")
  1744→        pts = f.get("points_deducted", 0)
  1745→        mod = f.get("module", "")
  1746→        sev_color = {
  1747→            "critical": "#dc2626", "high": "#ea580c",
  1748→            "medium": "#d97706", "low": "#2563eb", "info": "#6b7280",
  1749→        }.get(sev, "#6b7280")
  1750→        finding_rows_html += f"""
  1751→        <tr>
  1752→          <td style="text-align:center">{idx}</td>
  1753→          <td style="color:{sev_color};font-weight:bold;text-transform:uppercase">{sev}</td>
  1754→          <td>{mod}</td>
  1755→          <td>{title}</td>
  1756→          <td style="text-align:center">{pts}</td>
  1757→        </tr>"""
  1758→
  1759→    # Remediation summary
  1760→    rem_html = ""
  1761→    for f in rows:
  1762→        if f.get("remediation"):
  1763→            rem_html += f"<li><strong>{f.get('title', 'Untitled')}</strong>: {f['remediation']}</li>\n"
  1764→
  1765→    html = f"""<!DOCTYPE html>
  1766→<html lang="en">
  1767→<head>
  1768→<meta charset="utf-8">
  1769→<title>ReconPro Report — {target or 'Scan'}</title>
  1770→<style>
  1771→  @page {{ size: A4; margin: 1.5cm; }}
  1772→  * {{ box-sizing: border-box; margin: 0; padding: 0; }}
  1773→  body {{ font-family: 'Segoe UI', Arial, sans-serif; font-size: 11pt; color: #1f2937; line-height: 1.6; padding: 2cm; max-width: 210mm; margin: auto; }}
  1774→  h1 {{ font-size: 20pt; margin-bottom: 4px; }}
  1775→  h2 {{ font-size: 14pt; margin: 20px 0 8px; border-bottom: 2px solid #e5e7eb; padding-bottom: 4px; }}
  1776→  .meta {{ color: #6b7280; font-size: 9pt; margin-bottom: 16px; }}
  1777→  .gauge {{ font-family: 'Courier New', monospace; font-size: 12pt; background: #f9fafb; padding: 12px 16px; border-radius: 6px; border: 1px solid #e5e7eb; margin: 12px 0 20px; }}
  1778→  .gauge-label {{ font-family: 'Segoe UI', Arial, sans-serif; font-size: 9pt; color: #6b7280; margin-bottom: 4px; }}
  1779→  table {{ width: 100%; border-collapse: collapse; margin: 12px 0; font-size: 9pt; }}
  1780→  th {{ background: #f3f4f6; text-align: left; padding: 6px 8px; border-bottom: 2px solid #d1d5db; }}
  1781→  td {{ padding: 5px 8px; border-bottom: 1px solid #e5e7eb; vertical-align: top; }}
  1782→  tr:nth-child(even) {{ background: #f9fafb; }}
  1783→  .summary-grid {{ display: grid; grid-template-columns: repeat(5, 1fr); gap: 8px; margin: 12px 0; }}
  1784→  .summary-box {{ text-align: center; padding: 10px; border-radius: 6px; border: 1px solid #e5e7eb; }}
  1785→  .summary-box .num {{ font-size: 18pt; font-weight: bold; }}
  1786→  .summary-box .label {{ font-size: 8pt; text-transform: uppercase; color: #6b7280; }}
  1787→  .rem-list {{ margin: 8px 0; padding-left: 20px; }}
  1788→  .rem-list li {{ margin-bottom: 4px; font-size: 9pt; }}
  1789→  .footer {{ margin-top: 30px; text-align: center; font-size: 8pt; color: #9ca3af; border-top: 1px solid #e5e7eb; padding-top: 8px; }}
  1790→  @media print {{ body {{ padding: 0; }} }}
  1791→</style>
  1792→</head>
  1793→<body>
  1794→<h1>ReconPro Security Report</h1>
  1795→<div class="meta">Generated: {datetime.utcnow().strftime('%Y-%m-%d %H:%M:%S UTC')} &nbsp;|&nbsp; Target: {target or 'N/A'}</div>
  1796→
  1797→<h2>Score</h2>
  1798→<div class="gauge-label">Security Score (out of 100)</div>
  1799→<div class="gauge">[{gauge_bar}] {score_display}/100</div>
  1800→
  1801→<h2>Severity Breakdown</h2>
  1802→<div class="summary-grid">
  1803→  <div class="summary-box" style="background:#fef2f2"><div class="num" style="color:#dc2626">{counts.get('critical', 0)}</div><div class="label">Critical</div></div>
  1804→  <div class="summary-box" style="background:#fff7ed"><div class="num" style="color:#ea580c">{counts.get('high', 0)}</div><div class="label">High</div></div>
  1805→  <div class="summary-box" style="background:#fffbeb"><div class="num" style="color:#d97706">{counts.get('medium', 0)}</div><div class="label">Medium</div></div>
  1806→  <div class="summary-box" style="background:#eff6ff"><div class="num" style="color:#2563eb">{counts.get('low', 0)}</div><div class="label">Low</div></div>
  1807→  <div class="summary-box" style="background:#f9fafb"><div class="num" style="color:#6b7280">{counts.get('info', 0)}</div><div class="label">Info</div></div>
  1808→</div>
  1809→
  1810→<h2>Findings ({len(rows)})</h2>
  1811→<table>
  1812→  <thead><tr><th>#</th><th>Severity</th><th>Module</th><th>Title</th><th style="text-align:center">Pts</th></tr></thead>

  1815→
  1816→{"<h2>Remediation Summary</h2><ul class=\"rem-list\">" + rem_html + "</ul>" if rem_html else ""}
  1817→
  1818→<div class="footer">Generated by <strong>ReconPro</strong> v9.0.0</div>
  1819→</body>
  1820→</html>"""
  1821→
  1822→    with open(str(actual_path), "w", encoding="utf-8") as fh:
  1823→        fh.write(html)
  1824→
  1825→    return str(actual_path.resolve())

  1898→def generate_report_auto(
  1899→    findings: Any,
  1900→    output_path: str,
  1901→    fmt: str = "html",
  1902→    score: Optional[float] = None,
  1903→    target: str = "",
  1904→) -> str:
  1905→    """Auto-detect report format from extension or explicit *fmt* parameter.
  1906→
  1907→    Parameters
  1908→    ----------
  1909→    findings : list of dict-like
  1910→    output_path : str
  1911→    fmt : str
  1912→        One of: html, csv, json, sarif, xml, md, markdown, pdf.
  1913→        If ``"auto"``, the format is inferred from *output_path*
  1914→        file extension.
  1915→    score : float, optional
  1916→    target : str, optional
  1917→
  1918→    Returns
  1919→    -------
  1920→    str
  1921→        Absolute path of the written file.
  1922→    """
  1923→    if fmt == "auto":
  1924→        ext = Path(output_path).suffix.lower()
  1925→        fmt = _EXT_MAP.get(ext, "html")
  1926→
  1927→    fmt = fmt.lower().strip()
  1928→    generator = _FORMAT_MAP.get(fmt)
  1929→    if generator is None:
  1930→        raise ValueError(
  1931→            f"Unknown report format '{fmt}'. "
  1932→            f"Supported: {', '.join(sorted(_FORMAT_MAP))}"
  1933→        )
  1934→
  1935→    # Build kwargs based on function signature
  1936→    import inspect
  1937→    sig = inspect.signature(generator)
  1938→    kwargs: Dict[str, Any] = {"findings": findings, "output_path": output_path}
  1939→    if "score" in sig.parameters:
  1940→        kwargs["score"] = score
  1941→    if "target" in sig.parameters:
  1942→        kwargs["target"] = target
  1943→
  1944→    return generator(**kwargs)
