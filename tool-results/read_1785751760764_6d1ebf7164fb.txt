    52→def _render_summary(result, title: str = "RECONPRO") -> None:
    53→    score = result.total_score
    54→    grade = result.grade
    55→    grade_color = GRADE_COLORS.get(grade, "bold bright_red")
    56→    sc = result.severity_counts
    57→    total = len(result.findings)
    58→    bar_width = 50
    59→    filled = int(score / 100 * bar_width)
    60→    bar = "█" * filled + "░" * (bar_width - filled)
    61→    lines = [
    62→        Text(f"\n  {bar} [bold {grade_color}]{score}/100 ({grade})[/{grade_color}]"),
    63→        Text(f"\n  Target: [cyan]{result.target}[/]"),
    64→        Text(f"  Modules: [dim]{', '.join(result.modules_run)}[/]"),
    65→        Text(""),
    66→        Text(
    67→            f"  Findings: [bold]{total}[/]  "
    68→            f"[bright_red]{sc.get('critical', 0)} critical[/], "
    69→            f"[red]{sc.get('high', 0)} high[/], "
    70→            f"[yellow]{sc.get('medium', 0)} medium[/], "
    71→            f"[green]{sc.get('low', 0)} low[/], "
    72→            f"[dim]{sc.get('info', 0)} info[/]",
    73→        ),
    74→    ]
    75→    if result.vibesec_score is not None:
    76→        lines.append(Text(
    77→            f"\n  VibeSec Score: [bold bright_green]{result.vibesec_score}/100 ({result.vibesec_grade})[/]"
    78→        ))
    79→    console.print(Panel(Group(*lines), border_style=grade_color,
    80→                         title=f"[bold]{title}[/bold]", title_align="left", padding=(1, 2)))

    83→def _render_findings_table(result, show_remediation: bool = False) -> None:
    84→    if not result.findings:
    85→        console.print("\n  [bright_green]No findings. Target is clean.[/]")
    86→        return
    87→    table = Table(title=f"Findings — {len(result.findings)} detected",
    88→                  border_style="bright_white", header_style="bold bright_white")
    89→    table.add_column("Severity", style="bold", width=10)
    90→    table.add_column("Module", style="cyan", width=12)
    91→    table.add_column("Category", style="magenta", width=18)
    92→    table.add_column("Finding", style="white")
    93→    table.add_column("Pts", style="yellow", width=5)
    94→    if show_remediation:
    95→        table.add_column("Fix", style="bright_green", width=50)
    96→    for f in result.findings[:80]:
    97→        sev = f.get("severity", "info")
    98→        color = SEV_COLORS.get(sev, "white")
    99→        row = [
   100→            f"[{color}]{sev.upper()}[/{color}]",
   101→            f.get("module", ""), f.get("category", ""),
   102→            f.get("title", "")[:70], str(f.get("points_deducted", 0)),
   103→        ]
   104→        if show_remediation:
   105→            row.append(f.get("remediation", "")[:50])
   106→        table.add_row(*row)
   107→    console.print(table)

   110→def _render_module_breakdown(result) -> None:
   111→    if not result.module_results:
   112→        return
   113→    t = Table(title="Module Breakdown", border_style="dim", header_style="bold dim")
   114→    t.add_column("Module", style="bold", width=16)
   115→    t.add_column("Findings", style="white", width=10)
   116→    t.add_column("Critical", style="bright_red", width=10)
   117→    t.add_column("High", style="red", width=8)
   118→    t.add_column("Medium", style="yellow", width=8)
   119→    for mod_id, mod_data in result.module_results.items():
   120→        findings = mod_data.get("findings", [])
   121→        t.add_row(mod_id.upper(), str(len(findings)),
   122→                   str(sum(1 for f in findings if f.get("severity") == "critical")),
   123→                   str(sum(1 for f in findings if f.get("severity") == "high")),
   124→                   str(sum(1 for f in findings if f.get("severity") == "medium")))
   125→    console.print(t)

   128→def _render_badge(result) -> None:
   129→    if result.badge_markdown:
   130→        console.print(Panel(Group(
   131→            Text("  GitHub README Badge (copy-paste):", style="bold dim"),
   132→            Text(f"  {result.badge_markdown}", style="cyan"),
   133→        ), border_style="dim", title="[dim]Badge[/dim]", title_align="left", padding=(0, 2)))

   136→def _render_remediations(result) -> None:
   137→    actionable = [f for f in result.findings
   138→                  if f.get("remediation") and f.get("severity") not in ("info",)
   139→                  and f.get("points_deducted", 0) > 0]
   140→    if not actionable:
   141→        console.print("\n  [bright_green]No actionable fixes needed.[/]")
   142→        return
   143→    console.print("\n[bold bright_white]  Fix Commands:[/bold bright_white]\n")
   144→    seen = set()
   145→    for f in actionable:
   146→        fix = f.get("remediation", "").strip()
   147→        sev = f.get("severity", "info")
   148→        color = SEV_COLORS.get(sev, "white")
   149→        if fix and fix not in seen:
   150→            seen.add(fix)
   151→            console.print(f"  [{color}]{sev.upper():8}[/{color}]  {fix}")
   152→    console.print()

   155→def _output_result(result, args, title: str = "RECONPRO", show_remediation: bool = False) -> None:
   156→    json_output = getattr(args, "json_output", False)
   157→    output_file = getattr(args, "output_file", None)
   158→    if json_output:
   159→        text = json.dumps(result.to_dict(), indent=2, default=str)
   160→        if output_file:
   161→            with open(output_file, "w") as fp:
   162→                fp.write(text)
   163→            console.print(f"  [green]JSON saved: [cyan]{output_file}[/][/]")
   164→        else:
   165→            console.print(text)
   166→    else:
   167→        _render_summary(result, title=title)
   168→        console.print()
   169→        _render_module_breakdown(result)
   170→        console.print()
   171→        _render_findings_table(result, show_remediation=show_remediation)
   172→        if show_remediation:
   173→            _render_remediations(result)
   174→        _render_badge(result)
   175→        if output_file:
   176→            with open(output_file, "w") as fp:
   177→                fp.write(json.dumps(result.to_dict(), indent=2, default=str))
   178→            console.print(f"\n  [green]JSON saved: [cyan]{output_file}[/][/]")
   179→        # Auto-save to history
   180→        save_scan(result.to_dict())
   181→    console.print()

   184→def _spinner_wrap(msg: str, fn, *a, **kw):
   185→    """Run a function with a spinner."""
   186→    with Progress(SpinnerColumn(), TextColumn("[bold]{task.description}[/]"),
   187→                   TimeElapsedColumn(), console=console) as progress:
   188→        task = progress.add_task(msg, total=None)
   189→        result = fn(*a, **kw)
   190→        progress.update(task, completed=True)
   191→    return result
