   464→# 3. TimingJitter
   465→# ---------------------------------------------------------------------------
   466→
   467→
   468→class TimingJitter:
   469→    """Add human-like timing jitter between requests.
   470→
   471→    Uses a Gaussian distribution with configurable stddev.  Supports
   472→    burst patterns that simulate real browsing (rapid clicks then pause).
   473→    """
   474→
   475→    MIN_DELAY: float = 0.05
   476→    MAX_DELAY: float = 5.0
   477→    DEFAULT_STDDEV: float = 0.3
   478→
   479→    def __init__(
   480→        self,
   481→        stddev: float = DEFAULT_STDDEV,
   482→        min_delay: float = MIN_DELAY,
   483→        max_delay: float = MAX_DELAY,
   484→        burst_enabled: bool = True,
   485→        seed: Optional[int] = None,
   486→    ) -> None:
   487→        self._rng = random.Random(seed)
   488→        self._stddev = stddev
   489→        self._min = min_delay
   490→        self._max = max_delay
   491→        self._burst_enabled = burst_enabled
   492→        self._burst_remaining = 0
   493→        self._in_burst = False
   494→
   495→    def add_jitter(self) -> float:
   496→        """Return a delay in seconds drawn from a Gaussian distribution,
   497→        clamped to [min_delay, max_delay].
   498→
   499→        If burst mode is enabled, some calls will return near-zero delay to
   500→        simulate rapid human clicks, followed by a longer pause.
   501→        """
   502→        if self._burst_enabled:
   503→            delay = self._burst_jitter()
   504→        else:
   505→            delay = self._gaussian_jitter()
   506→        return max(self._min, min(self._max, delay))
   507→
   508→    def _gaussian_jitter(self) -> float:
   509→        """Pure Gaussian jitter: mean=0, clamped so result is >= min."""
   510→        sample = self._rng.gauss(0, self._stddev)
   511→        # Shift so the result is a positive delay
   512→        return max(self._min, abs(sample) + self._min)
   513→
   514→    def _burst_jitter(self) -> float:
   515→        """Simulate a burst of rapid requests then a pause.
   516→
   517→        Each burst consists of 2-5 requests with minimal delay (~0.05-0.15s),
   518→        then a pause of 1.5-4.0s before the next burst.
   519→        """
   520→        if not self._in_burst:
   521→            # Decide whether to start a burst (40 % chance)
   522→            if self._rng.random() < 0.4:
   523→                self._in_burst = True
   524→                self._burst_remaining = self._rng.randint(2, 5)
   525→            else:
   526→                # Normal delay
   527→                return self._gaussian_jitter()
   528→
   529→        if self._in_burst:
   530→            if self._burst_remaining > 1:
   531→                self._burst_remaining -= 1
   532→                return self._rng.uniform(0.05, 0.15)
   533→            else:
   534→                # End of burst — return a longer pause
   535→                self._in_burst = False
   536→                self._burst_remaining = 0
   537→                return self._rng.uniform(1.5, 4.0)
   538→
   539→        return self._gaussian_jitter()

   543→# 4. PathCanonicalizer
   544→# ---------------------------------------------------------------------------

   546→
   547→class PathCanonicalizer:
   548→    """Generate variations of a URL path to test WAF / CDN normalisation.
   549→
   550→    Produces 8-12 variants using techniques like dot-segments, URL-encoding,
   551→    mixed case, trailing slashes, and overlong UTF-8 sequences.
   552→    """
   553→
   554→    def generate_variants(self, path: str) -> List[str]:
   555→        """Return a list of 8-12 path variations for WAF testing.
   556→
   557→        Parameters
   558→        ----------
   559→        path : str
   560→            The original URL path (e.g. ``/api/v1/users``).
   561→
   562→        Returns
   563→        -------
   564→        list[str]
   565→        """
   566→        if not path.startswith("/"):
   567→            path = "/" + path
   568→
   569→        variants: List[str] = []
   570→        # 1. Original
   571→        variants.append(path)
   572→        # 2. Dot-segment prefix
   573→        variants.append("/./" + path.lstrip("/"))
   574→        # 3. Double-encoding of first char after slash
   575→        variants.append(self._double_encode_char(path))
   576→        # 4. URL-encoded full first segment
   577→        variants.append(self._encode_segment(path))
   578→        # 5. Mixed case — title-case each segment
   579→        variants.append(self._mixed_case(path))
   580→        # 6. UPPERCASE
   581→        variants.append(path.upper())
   582→        # 7. Trailing slash
   583→        variants.append(path.rstrip("/") + "/")
   584→        # 8. Double trailing slash
   585→        variants.append(path.rstrip("/") + "//")
   586→        # 9. Trailing dot-segment
   587→        variants.append(path.rstrip("/") + "/.")
   588→        # 10. Overlong UTF-8 for first slash-separated segment
   589→        variants.append(self._overlong_utf8(path))
   590→        # 11. Null-byte injection (common WAF bypass)
   591→        variants.append(path.rstrip("/") + "%00")
   592→        # 12. Semicolon / path-param injection
   593→        variants.append(path.rstrip("/") + ";jsessionid=" + _rand_hex(8))
   594→
   595→        # De-duplicate while preserving order
   596→        seen: set = set()
   597→        unique: List[str] = []
   598→        for v in variants:
   599→            if v not in seen:
   600→                seen.add(v)
   601→                unique.append(v)
   602→
   603→        return unique
   604→
   605→    # -- variant generators --------------------------------------------------
   606→
   607→    @staticmethod
   608→    def _double_encode_char(path: str) -> str:
   609→        """Double-encode the first character of each path segment.
   610→
   611→        E.g. '/a/b' → '/%2561/%2562'
   612→        """
   613→        parts = path.strip("/").split("/")
   614→        encoded_parts: List[str] = []
   615→        for part in parts:
   616→            if not part:
   617→                continue
   618→            first = part[0]
   619→            double = "%25" + ("%02X" % ord(first))
   620→            encoded_parts.append(double + part[1:])
   621→        return "/" + "/".join(encoded_parts)
   622→
   623→    @staticmethod
   624→    def _encode_segment(path: str) -> str:
   625→        """Fully URL-encode the first path segment."""
   626→        parts = path.strip("/").split("/")
   627→        if not parts or not parts[0]:
   628→            return path
   629→        encoded_first = urllib.parse.quote(parts[0], safe="")
   630→        return "/" + encoded_first + ("/" + "/".join(parts[1:]) if len(parts) > 1 else "")
   631→
   632→    @staticmethod
   633→    def _mixed_case(path: str) -> str:
   634→        """Title-case each segment: /api/v1/users → /Api/V1/Users."""
   635→        parts = path.strip("/").split("/")
   636→        titled = [p[:1].upper() + p[1:].lower() if p else "" for p in parts]
   637→        return "/" + "/".join(titled)
   638→
   639→    @staticmethod
   640→    def _overlong_utf8(path: str) -> str:
   641→        """Replace the first character of the first segment with an overlong
   642→        UTF-8 2-byte sequence (e.g. 'a' → %C0%AF for '/').
   643→
   644→        Specifically this targets the forward-slash character to produce
   645→        /path%C0%AFrest style bypasses.
   646→        """
   647→        parts = path.strip("/").split("/")
   648→        if not parts:
   649→            return path
   650→        # Use a valid overlong encoding of 'a' (U+0061) as 2-byte: 0xC1 0xA1
   651→        # Note: many modern systems reject overlong UTF-8, but some WAFs still
   652→        # mishandle it. We encode the first char of the first segment.
   653→        first = parts[0][0] if parts[0] else "a"
   654→        cp = ord(first)
   655→        if cp < 0x80:
   656→            # Overlong 2-byte: 110xxxxx 10xxxxxx → 0xC0|cp>>6, 0x80|cp&0x3F
   657→            b1 = 0xC0 | (cp >> 6)
   658→            b2 = 0x80 | (cp & 0x3F)
   659→            overlong = f"%{b1:02X}%{b2:02X}"
   660→            parts[0] = overlong + parts[0][1:]
   661→        return "/" + "/".join(parts)
   662→
   663→
   664→def _rand_hex(n: int) -> str:
   665→    """Return a random hex string of length *n*."""
   666→    return "".join(random.choice("0123456789abcdef") for _ in range(n))

   745→        For custom profiles saved to disk, use :meth:`from_file`.

   747→        return cls(name=name)

   751→        """Load a custom profile from a JSON file."""
   752→        with open(path, "r", encoding="utf-8") as fh:

   754→        instance = cls(name=data.get("name", "CUSTOM"), seed=data.get("seed"))

   758→            stddev=instance._config.get("jitter_stddev", 0.3),

   763→    def save_profile(self, profile_path: Optional[str] = None) -> str:
   764→        """Save the current profile to ``~/.reconpro/profiles/<name>.json``.

   766→        Returns the absolute path of the saved file.
   767→        """
   768→        if profile_path is None:
   769→            _PROFILE_DIR.mkdir(parents=True, exist_ok=True)
   770→            profile_path = str(_PROFILE_DIR / f"{self._name.lower()}.json")
   771→        data = {
   772→            "name": self._name,
   773→            "config": self._config,
   774→        }
   775→        with open(profile_path, "w", encoding="utf-8") as fh:
   776→            json.dump(data, fh, indent=2)
   777→        return os.path.abspath(profile_path)

   779→    def apply_to_request(
   780→        self,
   781→        url: str,
   782→        method: str = "GET",
   783→        headers: Optional[Dict[str, str]] = None,
   784→    ) -> Dict[str, Any]:
   785→        """Build a fully evasive request configuration.
   786→
   787→        Returns
   788→        -------
   789→        dict with keys:
   790→            url     – possibly canonicalised URL
   791→            headers – rotated header dict
   792→            delay   – jitter delay in seconds (caller should ``time.sleep``)
   793→            method  – HTTP method
   794→        """
   795→        # Determine if we should rotate the path
   796→        final_url = url
   797→        if self._config.get("rotate_path") and self._config.get("path_variant_chance", 0) > 0:
   798→            if self._rng.random() < self._config["path_variant_chance"]:
   799→                parsed = urllib.parse.urlparse(url)
   800→                variants = self._path_canon.generate_variants(parsed.path)
   801→                chosen = self._rng.choice(variants)
   802→                final_url = urllib.parse.urlunparse(
   803→                    (parsed.scheme, parsed.netloc, chosen, parsed.params, parsed.query, parsed.fragment)
   804→                )
   805→
   806→        # Rotate headers
   807→        rotated = self._header_rotator.rotate(custom_headers=headers)
   808→
   809→        # Compute delay
   810→        delay = self._timing.add_jitter()
   811→
   812→        return {
   813→            "url": final_url,
   814→            "headers": rotated,
   815→            "delay": delay,
   816→            "method": method.upper(),
   817→        }
   818→

   820→    def config(self) -> Dict[str, Any]:
   821→        return dict(self._config)

   825→        return self._name

   829→        return list(cls._BUILTINS.keys())

   833→# 6. EvasionSession
   834→# ---------------------------------------------------------------------------

   836→
   837→class EvasionSession:
   838→    """Stateful evasion across multiple requests to the same target.
   839→
   840→    Maintains a consistent User-Agent, cookies, and timing pattern so that
   841→    a sequence of requests looks like one real human browsing a site.
   842→    """
   843→
   844→    def __init__(
   845→        self,
   846→        target_base: str,
   847→        profile_name: str = EvasionProfile.BALANCED,
   848→        seed: Optional[int] = None,
   849→    ) -> None:
   850→        self._target_base = target_base.rstrip("/")
   851→        self._profile = EvasionProfile(name=profile_name, seed=seed)
   852→        self._rng = random.Random(seed)
   853→
   854→        # Session state
   855→        self._request_count = 0
   856→        self._current_ua: Optional[str] = None
   857→        self._current_headers: Dict[str, str] = {}
   858→        self._cookies: Dict[str, str] = {}
   859→        self._visited_urls: List[str] = []
   860→        self._timing = self._profile._timing
   861→        self._tls = TlsFingerprint(seed=seed)
   862→        self._tls_profile_name = self._rng.choice(["CHROME_120", "FIREFOX_121", "SAFARI_17", "EDGE_120"])
   863→        self._tls_profile = self._tls.get_profile(self._tls_profile_name)
   864→
   865→        # Initialise with one consistent set of headers
   866→        self._current_headers = self._profile._header_rotator.rotate()
   867→        self._current_ua = self._current_headers.get("User-Agent", "")
   868→
   869→    def next_request(self, url: str) -> Dict[str, Any]:
   870→        """Return a complete request configuration for the next request.
   871→
   872→        The returned dict has keys: url, headers, delay, method, cookies,
   873→        tls_profile.
   874→
   875→        The session keeps the same User-Agent across all requests (to look
   876→        like one user) while still varying other headers slightly.
   877→        """
   878→        self._request_count += 1
   879→
   880→        # Keep the same UA, but lightly rotate other headers
   881→        headers = dict(self._current_headers)
   882→        headers["User-Agent"] = self._current_ua
   883→
   884→        # Set referer to the last visited URL (simulates navigation)
   885→        if self._visited_urls:
   886→            headers["Referer"] = self._visited_urls[-1]
   887→
   888→        # Vary Sec-Fetch-* headers based on navigation pattern
   889→        if self._request_count == 1:
   890→            headers["Sec-Fetch-Site"] = "none"
   891→            headers["Sec-Fetch-Mode"] = "navigate"
   892→        elif url.startswith(self._target_base):
   893→            headers["Sec-Fetch-Site"] = "same-origin"
   894→            # Every few requests simulate a sub-resource fetch
   895→            if self._rng.random() < 0.2:
   896→                headers["Sec-Fetch-Dest"] = "script"
   897→                headers["Sec-Fetch-Mode"] = "no-cors"
   898→            else:
   899→                headers["Sec-Fetch-Dest"] = "document"
   900→                headers["Sec-Fetch-Mode"] = "navigate"
   901→        else:
   902→            headers["Sec-Fetch-Site"] = "cross-site"
   903→            headers["Sec-Fetch-Mode"] = "navigate"
   904→
   905→        # Slightly vary Accept-Language on ~10 % of requests (user might
   906→        # have changed language setting, or we simulate a shared IP)
   907→        if self._rng.random() < 0.1:
   908→            lang_keys = list(_ACCEPT_LANGS.keys())
   909→            headers["Accept-Language"] = _ACCEPT_LANGS[self._rng.choice(lang_keys)]
   910→
   911→        # Compute delay
   912→        delay = self._timing.add_jitter()
   913→
   914→        # Track visited URL
   915→        self._visited_urls.append(url)
   916→
   917→        # Build cookie header if we have cookies
   918→        cookie_header = "; ".join(f"{k}={v}" for k, v in self._cookies.items())
   919→        if cookie_header:
   920→            headers["Cookie"] = cookie_header
   921→
   922→        return {
   923→            "url": url,
   924→            "headers": headers,
   925→            "delay": delay,
   926→            "method": "GET",
   927→            "cookies": dict(self._cookies),
   928→            "tls_profile": self._tls_profile,
   929→            "request_count": self._request_count,
   930→        }
   931→
   932→    def update_cookies(self, cookies: Dict[str, str]) -> None:
   933→        """Update the session cookie jar (e.g. from a Set-Cookie response)."""
   934→        self._cookies.update(cookies)
   935→
   936→    def clear_cookies(self) -> None:
   937→        """Clear all stored cookies."""
   938→        self._cookies.clear()
   939→
   940→    @property
   941→    def request_count(self) -> int:
   942→        return self._request_count
   943→
   944→    @property
   945→    def current_ua(self) -> str:
   946→        return self._current_ua or ""
   947→
   948→    @property
   949→    def tls_profile(self) -> Dict[str, Any]:
   950→        return dict(self._tls_profile)
   951→
   952→    @property
   953→    def visited_urls(self) -> List[str]:
   954→        return list(self._visited_urls)

   962→class WAFBypass:
   963→    """Advanced WAF bypass techniques using encoding, chunking, and unicode tricks."""
   964→
   965→    @staticmethod
   966→    def double_url_encode(payload: str) -> str:
   967→        """Double URL-encode a payload to bypass WAF pattern matching."""
   968→        import urllib.parse
   969→        first = urllib.parse.quote(payload)
   970→        second = urllib.parse.quote(first)
   971→        return second
   972→
   973→    @staticmethod
   974→    def unicode_normalize(payload: str) -> str:
   975→        """Apply Unicode normalization tricks to bypass WAF."""
   976→        replacements = {
   977→            "<": "\uff1c", ">": "\uff1e", "'": "\uff07", '"': "\uff02",
   978→            "(": "\uff08", ")": "\uff09", "/": "\uff0f", "%": "\uff05",
   979→            "+": "\uff0b", "=": "\uff1d", "&": "\uff06", "#": "\uff03",
   980→        }
   981→        result = []
   982→        for ch in payload:
   983→            result.append(replacements.get(ch, ch))
   984→        return "".join(result)
   985→
   986→    @staticmethod
   987→    def html_entity_encode(payload: str) -> str:
   988→        """Encode using HTML entities."""
   989→        result = []
   990→        for ch in payload:
   991→            if ch.isalpha() and random.random() < 0.5:
   992→                result.append(f"&#{ord(ch)};")
   993→            else:
   994→                result.append(ch)
   995→        return "".join(result)
   996→
   997→    @staticmethod
   998→    def chunked_encoding_bypass(payload: str) -> str:
   999→        """Break payload into chunks to bypass WAF content-length checks."""
  1000→        chunk_size = random.randint(4, 8)
  1001→        chunks = [payload[i:i + chunk_size] for i in range(0, len(payload), chunk_size)]
  1002→        return "".join(f"{len(c):x}\r\n{c}\r\n" for c in chunks) + "0\r\n\r\n"
  1003→
  1004→    @staticmethod
  1005→    def null_byte_inject(payload: str) -> str:
  1006→        """Inject NULL bytes to bypass filters."""
  1007→        return payload.replace("=", "%00=").replace("<", "%00<").replace(">", ">%00")
  1008→
  1009→    @staticmethod
  1010→    def comment_bypass(payload: str) -> str:
  1011→        """Insert inline comments to break WAF patterns (SQL specific)."""
  1012→        sql_keywords = [
  1013→            "SELECT", "UNION", "FROM", "WHERE", "AND", "OR",
  1014→            "INSERT", "UPDATE", "DELETE", "DROP",
  1015→        ]
  1016→        result = payload
  1017→        for kw in sql_keywords:
  1018→            if kw.lower() in result.lower():
  1019→                idx = result.lower().index(kw.lower())
  1020→                result = result[:idx] + kw[0] + "/**/" + kw[1:] + result[idx + len(kw):]
  1021→        return result
  1022→
  1023→    @staticmethod
  1024→    def case_alternation(payload: str) -> str:
  1025→        """Alternate case of each character to bypass case-sensitive WAF rules."""
  1026→        result = []
  1027→        upper = True
  1028→        for ch in payload:
  1029→            if ch.isalpha():
  1030→                result.append(ch.upper() if upper else ch.lower())
  1031→                upper = not upper
  1032→            else:
  1033→                result.append(ch)
  1034→        return "".join(result)
  1035→
  1036→    @staticmethod
  1037→    def tab_newline_replace(payload: str) -> str:
  1038→        """Replace spaces with tabs/newlines to break token matching."""
  1039→        return payload.replace(" ", "\t").replace(" ", "\n")

  1047→            "unicode": WAFBypass.unicode_normalize(payload),
  1048→            "html_entity": WAFBypass.html_entity_encode(payload),

  1050→            "comment_bypass": WAFBypass.comment_bypass(payload),

  1061→class PolymorphicPayload:
  1062→    """Generates polymorphic variants of payloads to evade signature-based detection."""
  1063→
  1064→    _ENCODERS = [
  1065→        WAFBypass.double_url_encode,
  1066→        WAFBypass.html_entity_encode,
  1067→        WAFBypass.unicode_normalize,
  1068→        WAFBypass.case_alternation,
  1069→    ]
  1070→
  1071→    def __init__(self, seed: Optional[int] = None):
  1072→        self._rng = random.Random(seed)
  1073→
  1074→    def generate(self, base_payload: str, count: int = 10) -> List[str]:
  1075→        """Generate N polymorphic variants."""
  1076→        variants = set()
  1077→        variants.add(base_payload)
  1078→        for _ in range(count):
  1079→            encoder = self._rng.choice(self._ENCODERS)
  1080→            variant = encoder(base_payload)
  1081→            variants.add(variant)
  1082→            # Chain two encoders
  1083→            encoder2 = self._rng.choice(self._ENCODERS)
  1084→            variants.add(encoder2(variant))
  1085→        return list(variants)

  1089→# 8. RateLimiter
  1090→# ---------------------------------------------------------------------------
  1091→
  1092→
  1093→class RateLimiter:
  1094→    """Token bucket rate limiter for scan requests."""
  1095→
  1096→    def __init__(self, max_requests: float = 10.0, period: float = 1.0):
  1097→        self._max = max_requests
  1098→        self._period = period
  1099→        self._tokens = max_requests
  1100→        self._last_refill = time.monotonic()
  1101→
  1102→    def acquire(self) -> bool:
  1103→        """Try to acquire a token. Returns True if request allowed."""
  1104→        now = time.monotonic()
  1105→        elapsed = now - self._last_refill
  1106→        self._tokens = min(self._max, self._tokens + elapsed * (self._max / self._period))
  1107→        self._last_refill = now
  1108→        if self._tokens >= 1.0:
  1109→            self._tokens -= 1.0
  1110→            return True
  1111→        return False
  1112→
  1113→    def wait(self) -> float:
  1114→        """How long to wait for the next token."""
  1115→        if self._tokens >= 1.0:
  1116→            return 0.0
  1117→        return (1.0 - self._tokens) * (self._period / self._max)
