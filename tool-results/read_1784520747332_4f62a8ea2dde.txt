    10→// ── Helper: Run shell command with timeout ─────────────────────────
    11→function run(cmd: string, timeout = 10000): string {
    12→  try {
    13→    return execSync(cmd, { timeout, encoding: 'utf-8', stdio: ['pipe', 'pipe', 'pipe'] }).trim();
    14→  } catch {
    15→    return '';
    16→  }
    17→}
    18→
    19→// ── Helper: DNS query via dig ─────────────────────────────────────
    20→function digQuery(domain: string, type: string): string[] {
    21→  const output = run(`dig +short ${domain} ${type}`, 8000);
    22→  if (!output) return [];
    23→  return output.split('\n').map(l => l.trim()).filter(Boolean);
    24→}
    25→
    26→function digFull(domain: string, type: string): string {
    27→  return run(`dig +noall +answer ${domain} ${type}`, 8000);
    28→}
    29→
    30→// ── Helper: Check if domain resolves ──────────────────────────────
    31→function resolves(domain: string): string | null {
    32→  const ip = run(`dig +short +time=2 +tries=1 ${domain} A`, 3000);
    33→  if (!ip) return null;
    34→  // Return first IP (skip CNAME lines)
    35→  for (const line of ip.split('\n')) {
    36→    const trimmed = line.trim();
    37→    if (/^\d+\.\d+\.\d+\.\d+$/.test(trimmed)) return trimmed;
    38→  }
    39→  return null;
    40→}

    43→// 1. REAL DNS ENUMERATION
    44→// ═══════════════════════════════════════════════════════════════════════
    45→function enumerateDNS(domain: string) {
    46→  const findings: Array<{
    47→    title: string; severity: string; category: string;
    48→    description: string; evidence: string; asset: string;
    49→  }> = [];
    50→
    51→  // ── A / AAAA records ────────────────────────────────────────────
    52→  const aRecords = digQuery(domain, 'A');
    53→  const aaaaRecords = digQuery(domain, 'AAAA');
    54→  const mainIp = aRecords[0] || 'unknown';
    55→
    56→  if (aRecords.length > 0) {
    57→    findings.push({
    58→      title: `DNS A Record — ${aRecords.length} IPv4 address(es)`,
    59→      severity: 'info', category: 'dns',
    60→      description: `Domain ${domain} resolves to ${aRecords.length} IPv4 address(es): ${aRecords.join(', ')}. Multiple A records may indicate round-robin DNS or CDN usage.`,
    61→      evidence: `A: ${aRecords.join(', ')}`,
    62→      asset: domain,
    63→    });
    64→  }
    65→  if (aaaaRecords.length > 0) {
    66→    findings.push({
    67→      title: `DNS AAAA Record — IPv6 enabled`,
    68→      severity: 'info', category: 'dns',
    69→      description: `Domain has ${aaaaRecords.length} IPv6 address(es): ${aaaaRecords.join(', ')}. IPv6 support is present.`,
    70→      evidence: `AAAA: ${aaaaRecords.join(', ')}`,
    71→      asset: domain,
    72→    });
    73→  }
    74→
    75→  // ── MX Records (email) ──────────────────────────────────────────
    76→  const mxRaw = digFull(domain, 'MX');
    77→  if (mxRaw) {
    78→    const mxRecords = mxRaw.split('\n').filter(l => l.includes('MX'));
    79→    findings.push({
    80→      title: `MX Records — ${mxRecords.length} mail server(s)`,
    81→      severity: 'info', category: 'dns',
    82→      description: `${mxRecords.length} mail server(s) configured for ${domain}:\n${mxRecords.map(m => '  ' + m.trim()).join('\n')}`,
    83→      evidence: mxRecords.join('; '),
    84→      asset: domain,
    85→    });
    86→  } else {
    87→    findings.push({
    88→      title: 'No MX Records — No Email Infrastructure',
    89→      severity: 'low', category: 'dns',
    90→      description: `No MX records found for ${domain}. This domain does not receive email directly, or uses a third-party email service with different MX configuration.`,
    91→      evidence: 'MX query returned empty',
    92→      asset: domain,
    93→    });
    94→  }
    95→
    96→  // ── NS Records ──────────────────────────────────────────────────
    97→  const nsRaw = digFull(domain, 'NS');
    98→  if (nsRaw) {
    99→    const nsRecords = nsRaw.split('\n').filter(l => l.includes('NS'));
   100→    findings.push({
   101→      title: `NS Records — ${nsRecords.length} nameserver(s)`,
   102→      severity: 'info', category: 'dns',
   103→      description: `Domain uses ${nsRecords.length} nameserver(s): ${nsRecords.map(n => n.trim().split(/\s+/).pop()).join(', ')}.`,
   104→      evidence: nsRecords.join('; '),
   105→      asset: domain,
   106→    });
   107→  }
   108→
   109→  // ── TXT Records ─────────────────────────────────────────────────
   110→  const txtRecords = digQuery(domain, 'TXT');
   111→  if (txtRecords.length > 0) {
   112→    findings.push({
   113→      title: `TXT Records — ${txtRecords.length} record(s)`,
   114→      severity: 'info', category: 'dns',
   115→      description: `Found ${txtRecords.length} TXT record(s) including verification records, SPF policies, or other domain-level configurations.`,
   116→      evidence: txtRecords.join(' | '),
   117→      asset: domain,
   118→    });
   119→  }
   120→
   121→  // ── SPF Check ───────────────────────────────────────────────────
   122→  const spfRecord = txtRecords.find(t => t.startsWith('"v=spf'));
   123→  if (!spfRecord) {
   124→    findings.push({
   125→      title: 'SPF Record Missing',
   126→      severity: 'high', category: 'dns',
   127→      description: `No SPF (Sender Policy Framework) record found for ${domain}. Without SPF, malicious actors can send spoofed emails appearing to come from this domain, leading to phishing attacks and brand damage. This is a critical email security failure.`,
   128→      evidence: `TXT query returned ${txtRecords.length} record(s), none containing "v=spf1"`,
   129→      asset: domain,
   130→    });
   131→  } else {
   132→    const hasAll = spfRecord.includes('+all') || spfRecord.includes('?all');
   133→    const hasNone = spfRecord.includes('-all') || spfRecord.includes('~all');
   134→    if (hasAll) {
   135→      findings.push({
   136→        title: 'SPF Record Uses +all or ?all (Open)',
   137→        severity: 'critical', category: 'dns',
   138→        description: `SPF record uses "${hasAll ? (spfRecord.includes('+all') ? '+all' : '?all') : ''}" which essentially allows ANY server to send email on behalf of ${domain}. This completely defeats the purpose of SPF and provides zero protection against email spoofing.`,
   139→        evidence: `SPF: ${spfRecord}`,
   140→        asset: domain,
   141→      });
   142→    } else if (hasNone) {
   143→      findings.push({
   144→        title: 'SPF Record Configured Properly',
   145→        severity: 'info', category: 'dns',
   146→        description: `SPF record found with proper hard/soft fail: ${spfRecord}. Email spoofing protection is in place.`,
   147→        evidence: `SPF: ${spfRecord}`,
   148→        asset: domain,
   149→      });
   150→    }
   151→  }
   152→
   153→  // ── DMARC Check ─────────────────────────────────────────────────
   154→  const dmarcRecord = digQuery(`_dmarc.${domain}`, 'TXT');
   155→  const dmarc = dmarcRecord.find(t => t.includes('v=DMARC'));
   156→  if (!dmarc) {
   157→    findings.push({
   158→      title: 'DMARC Record Not Found',
   159→      severity: 'high', category: 'dns',
   160→      description: `No DMARC record found for ${domain}. DMARC builds on SPF and DKIM to provide domain-level email authentication policy. Without DMARC, the domain cannot specify how receivers should handle unauthenticated emails, making phishing attacks much harder to detect and prevent.`,
   161→      evidence: `_dmarc.${domain} TXT query returned empty`,
   162→      asset: `_dmarc.${domain}`,
   163→    });
   164→  } else {
   165→    const pMatch = dmarc.match(/p=([a-z]+)/);
   166→    const policy = pMatch ? pMatch[1] : 'unknown';
   167→    if (policy === 'none') {
   168→      findings.push({
   169→        title: 'DMARC Policy Set to "none" (Monitor Only)',
   170→        severity: 'medium', category: 'dns',
   171→        description: `DMARC is configured but with p=none, meaning unauthenticated emails are not rejected — only monitored. While this provides visibility, it does not actively protect against phishing. Consider upgrading to p=quarantine or p=reject.`,
   172→        evidence: `DMARC: ${dmarc}`,
   173→        asset: `_dmarc.${domain}`,
   174→      });
   175→    } else {
   176→      findings.push({
   177→        title: `DMARC Policy Active (p=${policy})`,
   178→        severity: 'info', category: 'dns',
   179→        description: `DMARC record found with policy p=${policy}. Email authentication enforcement is configured.`,
   180→        evidence: `DMARC: ${dmarc}`,
   181→        asset: `_dmarc.${domain}`,
   182→      });
   183→    }
   184→  }
   185→
   186→  // ── DKIM Check ──────────────────────────────────────────────────
   187→  const dkimSelectors = ['selector1', 'selector2', 'google', 'k1', 'default', 's1', 'selector'];
   188→  let dkimFound = false;
   189→  for (const sel of dkimSelectors) {
   190→    const dkim = digQuery(`${sel}._domainkey.${domain}`, 'TXT');
   191→    if (dkim.length > 0 && dkim[0].includes('v=DKIM')) {
   192→      findings.push({
   193→        title: `DKIM Record Found (selector: ${sel})`,
   194→        severity: 'info', category: 'dns',
   195→        description: `DKIM signing is configured for ${domain} using selector "${sel}". This allows the domain to cryptographically sign outgoing emails, proving they were not tampered with in transit.`,
   196→        evidence: `DKIM selector: ${sel}, record length: ${dkim[0].length} chars`,
   197→        asset: `${sel}._domainkey.${domain}`,
   198→      });
   199→      dkimFound = true;
   200→      break;
   201→    }
   202→  }
   203→  if (!dkimFound) {
   204→    findings.push({
   205→      title: 'DKIM Record Not Found',
   206→      severity: 'medium', category: 'dns',
   207→      description: `No DKIM (DomainKeys Identified Mail) record found for common selectors. Without DKIM, outgoing emails cannot be cryptographically verified, reducing email deliverability and making it harder for recipients to distinguish legitimate from spoofed emails.`,
   208→      evidence: `Checked selectors: ${dkimSelectors.join(', ')}`,
   209→      asset: domain,
   210→    });
   211→  }
   212→
   213→  // ── DNSSEC Check ────────────────────────────────────────────────
   214→  const dnssecOutput = run(`dig +dnssec ${domain} A`, 8000);
   215→  const hasDnssec = dnssecOutput.includes('RRSIG') || dnssecOutput.includes('DO');
   216→  if (!hasDnssec) {
   217→    findings.push({
   218→      title: 'DNSSEC Not Enabled',
   219→      severity: 'medium', category: 'dns',
   220→      description: `DNSSEC is not configured for ${domain}. Without DNSSEC, DNS responses can be spoofed via cache poisoning attacks, potentially redirecting users to malicious sites. DNSSEC adds cryptographic signatures to DNS records to verify authenticity.`,
   221→      evidence: 'DNSSEC RRSIG not found in DNS response',
   222→      asset: domain,
   223→    });
   224→  } else {
   225→    findings.push({
   226→      title: 'DNSSEC Enabled',
   227→      severity: 'info', category: 'dns',
   228→      description: `DNSSEC is active for ${domain}, providing cryptographic authentication of DNS responses. This protects against DNS cache poisoning and spoofing attacks.`,
   229→      evidence: 'RRSIG records present in DNS response',
   230→      asset: domain,
   231→    });
   232→  }
   233→
   234→  // ── CNAME Records ───────────────────────────────────────────────
   235→  const cnameRecords = digQuery(domain, 'CNAME');
   236→  if (cnameRecords.length > 0) {
   237→    const cnames = cnameRecords.filter(c => !c.includes('.cdn.') && !c.includes('.cloudfront.') && !c.includes('.akamai.'));
   238→    if (cnames.length > 0) {
   239→      findings.push({
   240→        title: `CNAME Record — Points to ${cnames[0]}`,
   241→        severity: 'info', category: 'dns',
   242→        description: `Domain is aliased via CNAME to ${cnames[0]}. CNAME chains add DNS lookup latency and if the target becomes unavailable, this domain will also fail.`,
   243→        evidence: `CNAME: ${cnames[0]}`,
   244→        asset: domain,
   245→      });
   246→    }
   247→  }
   248→
   249→  return { findings, mainIp: mainIp === 'unknown' ? null : mainIp };
   250→}

   270→  'panel', 'cpanel', 'whm', 'plesk', 'webmin', 'server', 'node',

   273→
   274→async function enumerateSubdomains(domain: string): Promise<Array<{
   275→  title: string; severity: string; category: string;
   276→  description: string; evidence: string; asset: string;
   277→}>> {
   278→  const findings: Array<{
   279→    title: string; severity: string; category: string;
   280→    description: string; evidence: string; asset: string;
   281→  }> = [];
   282→
   283→  // Resolve in parallel batches
   284→  const batchSize = 30;
   285→  const results: Array<{ subdomain: string; ip: string | null }> = [];
   286→
   287→  for (let i = 0; i < SUBDOMAINS.length; i += batchSize) {
   288→    const batch = SUBDOMAINS.slice(i, i + batchSize);
   289→    const batchResults = await Promise.all(
   290→      batch.map(async (sub) => {
   291→        const fqdn = `${sub}.${domain}`;
   292→        const ip = resolves(fqdn);
   293→        return { subdomain: fqdn, ip };
   294→      })
   295→    );
   296→    results.push(...batchResults);
   297→  }
   298→
   299→  const discovered = results.filter(r => r.ip !== null);
   300→  const sensitiveSubs = ['admin', 'portal', 'dashboard', 'db', 'database', 'redis',
   301→    'jenkins', 'gitlab', 'git', 'internal', 'staging', 'dev', 'test',
   302→    'vpn', 'remote', 'proxy', 'elastic', 'kibana', 'grafana', 'prometheus',
   303→    'cpanel', 'plesk', 'webmin', 'phpmyadmin', 'server', 'smtp', 'webmail'];
   304→
   305→  for (const { subdomain, ip } of discovered) {
   306→    const subName = subdomain.replace(`.${domain}`, '');
   307→    const isSensitive = sensitiveSubs.some(s => subName.includes(s));
   308→
   309→    findings.push({
   310→      title: `Discovered Subdomain: ${subdomain}`,
   311→      severity: isSensitive ? 'high' : 'info',
   312→      category: 'subdomain',
   313→      description: `${subdomain} resolves to ${ip}.${isSensitive ? ` This appears to be a sensitive internal service (${subName}) that is publicly accessible. This is a significant security concern as it may expose administrative interfaces, development environments, or internal tools to the internet.` : ' This is a publicly accessible subdomain providing additional attack surface for reconnaissance and potential exploitation.'}`,
   314→      evidence: `A record: ${ip}`,
   315→      asset: subdomain,
   316→    });
   317→  }
   318→
   319→  return findings;
   320→}

   323→// 3. REAL HTTP HEADER ANALYSIS
   324→// ═══════════════════════════════════════════════════════════════════════
   325→function analyzeHTTPHeaders(domain: string): Array<{
   326→  title: string; severity: string; category: string;
   327→  description: string; evidence: string; asset: string;
   328→  technologies: string[];
   329→}> {
   330→  const findings: ReturnType<typeof analyzeHTTPHeaders> = [];
   331→  const technologies: string[] = [];
   332→
   333→  // Fetch headers from both HTTP and HTTPS
   334→  const httpsHeaders = run(`curl -sI --max-time 8 -L https://${domain} 2>/dev/null`, 12000);
   335→  const httpHeaders = run(`curl -sI --max-time 8 -L http://${domain} 2>/dev/null`, 12000);
   336→  const headers = httpsHeaders || httpHeaders;
   337→  const allHeaders = httpsHeaders + '\n' + httpHeaders;
   338→
   339→  if (!headers || !headers.includes('HTTP/')) {
   340→    findings.push({
   341→      title: 'HTTP Service Not Reachable',
   342→      severity: 'medium', category: 'header',
   343→      description: `Could not establish HTTP/HTTPS connection to ${domain}. The server may be down, blocking our IP, or using non-standard ports.`,
   344→      evidence: 'curl returned empty or non-HTTP response',
   345→      asset: domain,
   346→      technologies: [],
   347→    });
   348→    return findings;
   349→  }
   350→
   351→  // Parse headers into key-value map
   352→  const headerMap: Record<string, string> = {};
   353→  for (const line of headers.split('\n')) {
   354→    const match = line.match(/^([^:]+):\s*(.+)/);
   355→    if (match) headerMap[match[1].trim().toLowerCase()] = match[2].trim();
   356→  }
   357→
   358→  // ── HSTS ────────────────────────────────────────────────────────
   359→  if (!headerMap['strict-transport-security']) {
   360→    findings.push({
   361→      title: 'Missing Strict-Transport-Security (HSTS)',
   362→      severity: 'high', category: 'header',
   363→      description: `The HSTS header is not set on ${domain}. Without HSTS, browsers may attempt to connect via HTTP before redirecting to HTTPS, creating a window for man-in-the-middle attacks via SSL stripping. Setting HSTS with a long max-age (at least 1 year) and includeSubDomains ensures browsers always use HTTPS.`,
   364→      evidence: 'strict-transport-security header not found in response',
   365→      asset: domain,
   366→      technologies,
   367→    });
   368→  } else {
   369→    const hsts = headerMap['strict-transport-security'];
   370→    const maxAgeMatch = hsts.match(/max-age=(\d+)/);
   371→    const maxAge = maxAgeMatch ? parseInt(maxAgeMatch[1]) : 0;
   372→    const hasSubdomains = hsts.includes('includeSubDomains');
   373→    const hasPreload = hsts.includes('preload');
   374→    if (maxAge < 31536000) {
   375→      findings.push({
   376→        title: `HSTS max-age Too Short (${Math.round(maxAge / 86400)} days)`,
   377→        severity: 'medium', category: 'header',
   378→        description: `HSTS is set but max-age is only ${maxAge} seconds (${Math.round(maxAge / 86400)} days). Industry best practice recommends at least 1 year (31536000 seconds). Short max-age values leave users vulnerable during the period after certificate renewal or first visit.`,
   379→        evidence: `strict-transport-security: ${hsts}`,
   380→        asset: domain,
   381→        technologies,
   382→      });
   383→    } else if (!hasSubdomains) {
   384→      findings.push({
   385→        title: 'HSTS Missing includeSubDomains Directive',
   386→        severity: 'medium', category: 'header',
   387→        description: `HSTS is configured but without includeSubDomains. Subdomains remain vulnerable to SSL stripping attacks. Add includeSubDomains to extend HSTS protection across all subdomains.`,
   388→        evidence: `strict-transport-security: ${hsts}`,
   389→        asset: domain,
   390→        technologies,
   391→      });
   392→    } else {
   393→      findings.push({
   394→        title: 'HSTS Properly Configured',
   395→        severity: 'info', category: 'header',
   396→        description: `HSTS is active with max-age=${maxAge}${hasSubdomains ? ', includeSubDomains' : ''}${hasPreload ? ', preload' : ''}. HTTPS is enforced at the browser level.`,
   397→        evidence: `strict-transport-security: ${hsts}`,
   398→        asset: domain,
   399→        technologies,
   400→      });
   401→    }
   402→  }
   403→
   404→  // ── Content-Security-Policy ─────────────────────────────────────
   405→  if (!headerMap['content-security-policy'] && !headerMap['content-security-policy-report-only']) {
   406→    findings.push({
   407→      title: 'Missing Content-Security-Policy (CSP)',
   408→      severity: 'high', category: 'header',
   409→      description: `No Content-Security-Policy header found. CSP is the primary defense against Cross-Site Scripting (XSS) and data injection attacks. Without CSP, the browser will execute any inline scripts, load resources from any origin, and eval() arbitrary code. A strict CSP with nonce-based or hash-based script sources is strongly recommended.`,
   410→      evidence: 'content-security-policy header not found',
   411→      asset: domain,
   412→      technologies,
   413→    });
   414→  } else {
   415→    const csp = headerMap['content-security-policy'] || headerMap['content-security-policy-report-only'];
   416→    if (csp.includes("'unsafe-inline'") && csp.includes("'unsafe-eval'")) {
   417→      findings.push({
   418→        title: 'CSP Uses unsafe-inline and unsafe-eval',
   419→        severity: 'medium', category: 'header',
   420→        description: `Content-Security-Policy is present but includes both 'unsafe-inline' and 'unsafe-eval', which significantly weaken XSS protection. 'unsafe-inline' allows inline script execution and 'unsafe-eval' allows eval() calls, both of which are common XSS vectors. Consider using nonces or hashes instead.`,
   421→        evidence: `CSP: ${csp.substring(0, 200)}...`,
   422→        asset: domain,
   423→        technologies,
   424→      });
   425→    } else {
   426→      findings.push({
   427→        title: 'Content-Security-Policy Configured',
   428→        severity: 'info', category: 'header',
   429→        description: `CSP header is present, providing XSS and data injection protection.`,
   430→        evidence: `CSP: ${csp.substring(0, 150)}${csp.length > 150 ? '...' : ''}`,
   431→        asset: domain,
   432→        technologies,
   433→      });
   434→    }
   435→  }
   436→
   437→  // ── X-Frame-Options ─────────────────────────────────────────────
   438→  if (!headerMap['x-frame-options']) {
   439→    findings.push({
   440→      title: 'Missing X-Frame-Options Header',
   441→      severity: 'medium', category: 'header',
   442→      description: `No X-Frame-Options header detected. This makes the site vulnerable to clickjacking attacks where an attacker can embed this site in a transparent iframe and trick users into clicking hidden elements. Set to "DENY" or "SAMEORIGIN" to prevent framing.`,
   443→      evidence: 'x-frame-options header not found',
   444→      asset: domain,
   445→      technologies,
   446→    });
   447→  }
   448→
   449→  // ── X-Content-Type-Options ──────────────────────────────────────
   450→  if (!headerMap['x-content-type-options'] || headerMap['x-content-type-options'] !== 'nosniff') {
   451→    findings.push({
   452→      title: 'Missing X-Content-Type-Options: nosniff',
   453→      severity: 'low', category: 'header',
   454→      description: `X-Content-Type-Options header is not set to "nosniff". Without this header, browsers may perform MIME-type sniffing which can cause executable content to be treated as a different (more dangerous) type than intended by the server.`,
   455→      evidence: `x-content-type-options: ${headerMap['x-content-type-options'] || 'not set'}`,
   456→      asset: domain,
   457→      technologies,
   458→    });
   459→  }
   460→
   461→  // ── Referrer-Policy ─────────────────────────────────────────────
   462→  if (!headerMap['referrer-policy']) {
   463→    findings.push({
   464→      title: 'Missing Referrer-Policy Header',
   465→      severity: 'low', category: 'header',
   466→      description: `No Referrer-Policy header set. By default, browsers may send full URL referrers to third-party sites, potentially leaking sensitive path/query parameters. Set to "strict-origin-when-cross-origin" or "no-referrer" for better privacy.`,
   467→      evidence: 'referrer-policy header not found',
   468→      asset: domain,
   469→      technologies,
   470→    });
   471→  }
   472→
   473→  // ── Permissions-Policy ──────────────────────────────────────────
   474→  if (!headerMap['permissions-policy'] && !headerMap['feature-policy']) {
   475→    findings.push({
   476→      title: 'Missing Permissions-Policy Header',
   477→      severity: 'low', category: 'header',
   478→      description: `No Permissions-Policy (formerly Feature-Policy) header detected. This header controls which browser features (camera, microphone, geolocation, etc.) the site can use. Without it, the site may be vulnerable to malicious feature abuse through embedded iframes.`,
   479→      evidence: 'permissions-policy header not found',
   480→      asset: domain,
   481→      technologies,
   482→    });
   483→  }
   484→
   485→  // ── CORS ────────────────────────────────────────────────────────
   486→  if (headerMap['access-control-allow-origin']) {
   487→    const cors = headerMap['access-control-allow-origin'];
   488→    if (cors === '*') {
   489→      findings.push({
   490→        title: 'CORS Allows Any Origin (*)',
   491→        severity: 'high', category: 'header',
   492→        description: `Access-Control-Allow-Origin is set to "*" which means any website can make cross-origin requests to this domain and read the responses. If this endpoint serves sensitive data, it can be exfiltrated by any malicious website a user visits. Restrict to specific trusted origins.`,
   493→        evidence: `access-control-allow-origin: *`,
   494→        asset: domain,
   495→        technologies,
   496→      });
   497→    } else if (headerMap['access-control-allow-credentials'] === 'true' && cors !== '*') {
   498→      findings.push({
   499→        title: 'CORS with Credentials to Specific Origin',
   500→        severity: 'info', category: 'header',
   501→        description: `CORS is configured to allow credentials from specific origin: ${cors}. This is a secure configuration when the origin is trusted.`,
   502→        evidence: `access-control-allow-origin: ${cors}, access-control-allow-credentials: true`,
   503→        asset: domain,
   504→        technologies,
   505→      });
   506→    }
   507→  }
   508→
   509→  // ── Server Version Disclosure ───────────────────────────────────
   510→  if (headerMap['server']) {
   511→    const server = headerMap['server'];
   512→    technologies.push(server);
   513→    const hasVersion = /\d+\.\d+/.test(server);
   514→    if (hasVersion) {
   515→      findings.push({
   516→        title: 'Server Version Disclosure',
   517→        severity: 'medium', category: 'header',
   518→        description: `The Server header reveals "${server}" including version information. Attackers can use this to identify known vulnerabilities specific to this exact server version. Remove or obfuscate version numbers from the Server header to reduce information leakage.`,
   519→        evidence: `server: ${server}`,
   520→        asset: domain,
   521→        technologies,
   522→      });
   523→    }
   524→  }
   525→
   526→  // ── X-Powered-By Disclosure ─────────────────────────────────────
   527→  if (headerMap['x-powered-by']) {
   528→    const powered = headerMap['x-powered-by'];
   529→    technologies.push(powered);
   530→    findings.push({
   531→      title: `Technology Disclosure: X-Powered-By: ${powered}`,
   532→      severity: 'low', category: 'header',
   533→      description: `The X-Powered-By header reveals the backend technology stack: "${powered}". This information helps attackers narrow down their exploitation strategy to known vulnerabilities in this specific technology. Remove this header in production.`,
   534→      evidence: `x-powered-by: ${powered}`,
   535→      asset: domain,
   536→      technologies,
   537→    });
   538→  }
   539→
   540→  // ── Detect technologies from headers ────────────────────────────
   541→  if (headerMap['cf-ray'] || headerMap['cf-cache-status']) {
   542→    technologies.push('Cloudflare');
   543→  }
   544→  if (allHeaders.includes('X-Amz-Cf-Id') || allHeaders.includes('x-amzn-requestid')) {
   545→    technologies.push('AWS CloudFront');
   546→  }
   547→  if (headerMap['x-vercel-id']) {
   548→    technologies.push('Vercel');
   549→  }
   550→  if (headerMap['x-netlify-request-id']) {
   551→    technologies.push('Netlify');
   552→  }
   553→  if (headerMap['fly-request-id']) {
   554→    technologies.push('Fly.io');
   555→  }
   556→  if (headerMap['x-github-request-id']) {
   557→    technologies.push('GitHub Pages');
   558→  }
   559→  if (headerMap['alt-svc']?.includes('h3')) {
   560→    technologies.push('HTTP/3 (QUIC)');
   561→  }
   562→  if (headerMap['x-cache']?.includes('HIT')) {
   563→    technologies.push('CDN Cache Active');
   564→  }
   565→
   566→  return { findings, technologies };
   567→}
   568→
   569→// ═══════════════════════════════════════════════════════════════════════
   570→// 4. REAL SSL/TLS CERTIFICATE ANALYSIS
   571→// ═══════════════════════════════════════════════════════════════════════
   572→function analyzeSSL(domain: string): Array<{
   573→  title: string; severity: string; category: string;
   574→  description: string; evidence: string; asset: string;
   575→  technologies: string[];
   576→}> {
   577→  const findings: ReturnType<typeof analyzeSSL> = [];
   578→  const technologies: string[] = [];
   579→
   580→  // Get certificate details
   581→  const certInfo = run(
   582→    `echo | openssl s_client -connect ${domain}:443 -servername ${domain} 2>/dev/null | openssl x509 -noout -subject -issuer -dates -ext subjectAltName 2>/dev/null`,
   583→    10000
   584→  );
   585→
   586→  // Get full cert text for protocol/cipher analysis
   587→  const sslConnect = run(
   588→    `echo | openssl s_client -connect ${domain}:443 -servername ${domain} 2>&1`,
   589→    10000
   590→  );
   591→
   592→  if (!certInfo && !sslConnect.includes('SSL handshake')) {
   593→    // Try without SNI
   594→    const certInfoNoSni = run(
   595→      `echo | openssl s_client -connect ${domain}:443 2>/dev/null | openssl x509 -noout -subject -issuer -dates 2>/dev/null`,
   596→      10000
   597→    );
   598→    if (!certInfoNoSni) {
   599→      findings.push({
   600→        title: 'SSL/TLS Connection Failed',
   601→        severity: 'high', category: 'ssl',
   602→        description: `Could not establish SSL/TLS connection to ${domain}:443. The server may not support HTTPS, may be using a non-standard port, or may be rejecting the connection.`,
   603→        evidence: 'openssl s_client failed to connect',
   604→        asset: `${domain}:443`,
   605→        technologies,
   606→      });
   607→      return { findings, technologies };
   608→    }
   609→  }
   610→
   611→  const fullCert = certInfo || sslConnect;
   612→
   613→  // ── Certificate Subject ─────────────────────────────────────────
   614→  const subjectMatch = fullCert.match(/subject=([^\n]+)/);
   615→  const subject = subjectMatch ? subjectMatch[1].trim() : 'unknown';
   616→  findings.push({
   617→    title: `SSL Certificate Subject: ${subject}`,
   618→    severity: 'info', category: 'ssl',
   619→    description: `The SSL certificate is issued to: ${subject}. This identifies the entity the certificate was issued to.`,
   620→    evidence: `subject: ${subject}`,
   621→    asset: domain,
   622→    technologies,
   623→  });
   624→
   625→  // ── Issuer ──────────────────────────────────────────────────────
   626→  const issuerMatch = fullCert.match(/issuer=([^\n]+)/);
   627→  const issuer = issuerMatch ? issuerMatch[1].trim() : 'unknown';
   628→  if (issuer.includes("Let's Encrypt")) {
   629→    technologies.push("Let's Encrypt");
   630→  } else if (issuer.includes('DigiCert')) {
   631→    technologies.push('DigiCert');
   632→  } else if (issuer.includes('Sectigo') || issuer.includes('Comodo')) {
   633→    technologies.push('Sectigo');
   634→  } else if (issuer.includes('Cloudflare')) {
   635→    technologies.push('Cloudflare SSL');
   636→  } else if (issuer.includes('Amazon') || issuer.includes('AWS')) {
   637→    technologies.push('AWS Certificate Manager');
   638→  } else if (issuer.includes('Google')) {
   639→    technologies.push('Google Trust Services');
   640→  }
   641→
   642→  findings.push({
   643→    title: `SSL Certificate Issuer: ${issuer}`,
   644→    severity: 'info', category: 'ssl',
   645→    description: `Certificate issued by: ${issuer}. The CA (Certificate Authority) is responsible for verifying the identity of the certificate holder.`,
   646→    evidence: `issuer: ${issuer}`,
   647→    asset: domain,
   648→    technologies,
   649→  });
   650→
   651→  // ── Validity Dates ──────────────────────────────────────────────
   652→  const notBeforeMatch = fullCert.match(/notBefore=(.+)/);
   653→  const notAfterMatch = fullCert.match(/notAfter=(.+)/);
   654→  if (notBeforeMatch && notAfterMatch) {
   655→    const notBefore = new Date(notBeforeMatch[1].trim());
   656→    const notAfter = new Date(notAfterMatch[1].trim());
   657→    const now = new Date();
   658→    const daysUntilExpiry = Math.ceil((notAfter.getTime() - now.getTime()) / (1000 * 60 * 60 * 24));
   659→
   660→    findings.push({
   661→      title: `SSL Certificate Validity: ${notBefore.toISOString().split('T')[0]} — ${notAfter.toISOString().split('T')[0]}`,
   662→      severity: 'info', category: 'ssl',
   663→      description: `Certificate is valid from ${notBefore.toISOString().split('T')[0]} to ${notAfter.toISOString().split('T')[0]}. Certificate lifespan is ${Math.ceil((notAfter.getTime() - notBefore.getTime()) / (1000 * 60 * 60 * 24))} days.`,
   664→      evidence: `notBefore: ${notBeforeMatch[1].trim()}, notAfter: ${notAfterMatch[1].trim()}`,
   665→      asset: domain,
   666→      technologies,
   667→    });
   668→
   669→    if (daysUntilExpiry < 0) {
   670→      findings.push({
   671→        title: `SSL Certificate EXPIRED ${Math.abs(daysUntilExpiry)} days ago`,
   672→        severity: 'critical', category: 'ssl',
   673→        description: `The SSL certificate for ${domain} expired ${Math.abs(daysUntilExpiry)} days ago. All major browsers will display security warnings, users will be unable to access the site without manual override, and search engines will flag the site as insecure. This requires immediate remediation. Additionally, any automated systems relying on this certificate (APIs, webhooks, microservices) will fail with TLS errors.`,
   674→        evidence: `Expired on: ${notAfter.toISOString()}`,
   675→        asset: domain,
   676→        technologies,
   677→      });
   678→    } else if (daysUntilExpiry <= 7) {
   679→      findings.push({
   680→        title: `SSL Certificate Expiring in ${daysUntilExpiry} Days — CRITICAL`,
   681→        severity: 'critical', category: 'ssl',
   682→        description: `The SSL certificate expires in only ${daysUntilExpiry} days. Immediate action is required to avoid service disruption. Schedule certificate renewal and deployment as an emergency task.`,
   683→        evidence: `Expires: ${notAfter.toISOString()} (${daysUntilExpiry} days)`,
   684→        asset: domain,
   685→        technologies,
   686→      });
   687→    } else if (daysUntilExpiry <= 30) {
   688→      findings.push({
   689→        title: `SSL Certificate Expiring in ${daysUntilExpiry} Days`,
   690→        severity: 'high', category: 'ssl',
   691→        description: `The SSL certificate expires in ${daysUntilExpiry} days. While not immediately critical, this should be renewed promptly to avoid last-minute emergency deployments. Automated certificate renewal (e.g., certbot cron) is recommended.`,
   692→        evidence: `Expires: ${notAfter.toISOString()} (${daysUntilExpiry} days)`,
   693→        asset: domain,
   694→        technologies,
   695→      });
   696→    } else if (daysUntilExpiry <= 90) {
   697→      findings.push({
   698→        title: `SSL Certificate Expiring in ${daysUntilExpiry} Days`,
   699→        severity: 'medium', category: 'ssl',
   700→        description: `Certificate expires in ${daysUntilExpiry} days. Plan renewal within the next month to ensure continuity. Verify auto-renewal is configured correctly.`,
   701→        evidence: `Expires: ${notAfter.toISOString()} (${daysUntilExpiry} days)`,
   702→        asset: domain,
   703→        technologies,
   704→      });
   705→    }
   706→
   707→    // Check cert lifespan (>398 days violates Apple/Google CT policy)
   708→    const certLifespan = Math.ceil((notAfter.getTime() - notBefore.getTime()) / (1000 * 60 * 60 * 24));
   709→    if (certLifespan > 398) {
   710→      findings.push({
   711→        title: `Certificate Lifespan Exceeds 398 Days (${certLifespan} days)`,
   712→        severity: 'medium', category: 'ssl',
   713→        description: `Certificate lifespan is ${certLifespan} days, exceeding the 398-day maximum enforced by Apple (Safari) and Google (Chrome). New certificates issued after September 2020 must not exceed 398 days. This certificate will need to be replaced with a shorter-validity one upon renewal.`,
   714→        evidence: `Lifespan: ${certLifespan} days (max allowed: 398)`,
   715→        asset: domain,
   716→        technologies,
   717→      });
   718→    }
   719→  }
   720→
   721→  // ── SAN (Subject Alternative Names) ─────────────────────────────
   722→  const sanMatch = fullCert.match(/Subject Alternative Name[^\n]*\n([^\n]*)/);
   723→  if (sanMatch) {
   724→    const sans = sanMatch[1].match(/DNS:[^\s,]+/g);
   725→    if (sans) {
   726→      const sanList = sans.map(s => s.replace('DNS:', ''));
   727→      findings.push({
   728→        title: `Certificate Covers ${sanList.length} Domain(s) (SAN)`,
   729→        severity: 'info', category: 'ssl',
   730→        description: `The certificate covers ${sanList.length} domain(s): ${sanList.join(', ')}. Wildcard certificates (*.domain.com) cover all subdomains.`,
   731→        evidence: `SAN: ${sanList.join(', ')}`,
   732→        asset: domain,
   733→        technologies,
   734→      });
   735→    }
   736→  }
   737→
   738→  // ── TLS Protocol Version ────────────────────────────────────────
   739→  const protocolMatch = sslConnect.match(/Protocol\s*:\s*([^\n]+)/);
   740→  const protocol = protocolMatch ? protocolMatch[1].trim() : 'unknown';
   741→  if (protocol.includes('TLSv1 ') || protocol.includes('TLSv1.0')) {
   742→    findings.push({
   743→      title: 'Weak TLS Version: TLS 1.0 Detected',
   744→      severity: 'high', category: 'ssl',
   745→      description: `TLS 1.0 is supported on ${domain}:443. TLS 1.0 was deprecated by the IETF in 2020 (RFC 8996) due to known vulnerabilities including BEAST, POODLE, and RC4 weaknesses. PCI-DSS 3.2+ prohibits TLS 1.0. Disable TLS 1.0 on the server immediately.`,
   746→      evidence: `Protocol: ${protocol}`,
   747→      asset: `${domain}:443`,
   748→      technologies,
   749→    });
   750→  } else if (protocol.includes('TLSv1.1')) {
   751→    findings.push({
   752→      title: 'Deprecated TLS Version: TLS 1.1 Detected',
   753→      severity: 'high', category: 'ssl',
   754→      description: `TLS 1.1 is supported on ${domain}:443. TLS 1.1 was deprecated in 2020 (RFC 8996) and has known weaknesses. PCI-DSS 3.2+ requires disabling TLS 1.1. Support only TLS 1.2 and TLS 1.3.`,
   755→      evidence: `Protocol: ${protocol}`,
   756→      asset: `${domain}:443`,
   757→      technologies,
   758→    });
   759→  } else if (protocol.includes('TLSv1.3')) {
   760→    technologies.push('TLS 1.3');
   761→    findings.push({
   762→      title: 'Modern TLS 1.3 Negotiated',
   763→      severity: 'info', category: 'ssl',
   764→      description: `The connection successfully negotiated TLS 1.3, the latest and most secure TLS version. TLS 1.3 removes support for weak ciphers, adds 0-RTT for reduced latency, and mandates forward secrecy.`,
   765→      evidence: `Protocol: ${protocol}`,
   766→      asset: `${domain}:443`,
   767→      technologies,
   768→    });
   769→  } else if (protocol.includes('TLSv1.2')) {
   770→    technologies.push('TLS 1.2');
   771→    findings.push({
   772→      title: 'TLS 1.2 Negotiated (Supported)',
   773→      severity: 'info', category: 'ssl',
   774→      description: `TLS 1.2 is the negotiated protocol. TLS 1.2 is widely supported and secure when configured with strong cipher suites. Consider also enabling TLS 1.3 for improved performance and security.`,
   775→      evidence: `Protocol: ${protocol}`,
   776→      asset: `${domain}:443`,
   777→      technologies,
   778→    });
   779→  }
   780→
   781→  // ── Cipher Suite Analysis ───────────────────────────────────────
   782→  const cipherMatch = sslConnect.match(/Cipher\s*:\s*([^\n]+)/);
   783→  const cipher = cipherMatch ? cipherMatch[1].trim() : 'unknown';
   784→  const weakCiphers = ['RC4', 'DES', 'MD5', 'NULL', 'EXPORT', '3DES', 'AES128-SHA', 'AES256-SHA'];
   785→  const isWeak = weakCiphers.some(wc => cipher.toUpperCase().includes(wc));
   786→
   787→  if (isWeak) {
   788→    findings.push({
   789→      title: `Weak Cipher Suite: ${cipher}`,
   790→      severity: 'high', category: 'ssl',
   791→      description: `A weak cipher suite (${cipher}) is in use. Weak ciphers are vulnerable to known cryptographic attacks and provide reduced security. Modern servers should only use AEAD cipher suites (AES-GCM, ChaCha20-Poly1305) with ECDHE key exchange.`,
   792→      evidence: `Cipher: ${cipher}`,
   793→      asset: `${domain}:443`,
   794→      technologies,
   795→    });
   796→  } else {
   797→    findings.push({
   798→      title: `Cipher Suite: ${cipher}`,
   799→      severity: 'info', category: 'ssl',
   800→      description: `Negotiated cipher suite: ${cipher}.`,
   801→      evidence: `Cipher: ${cipher}`,
   802→      asset: `${domain}:443`,
   803→      technologies,
   804→    });
   805→  }
   806→
   807→  return { findings, technologies };
   808→}

   810→// ═══════════════════════════════════════════════════════════════════════
   811→// 5. REAL PORT PROBING (HTTP-based checks on common ports)
   812→// ═══════════════════════════════════════════════════════════════════════
   813→const WEB_PORTS = [
   814→  { port: 80, service: 'HTTP', risk: 'medium', desc: 'Standard HTTP — should redirect to HTTPS' },
   815→  { port: 443, service: 'HTTPS', risk: 'info', desc: 'Standard HTTPS — expected for web services' },
   816→  { port: 8080, service: 'HTTP-Alt', risk: 'high', desc: 'Common alternative HTTP port — dev servers, proxies, Tomcat' },
   817→  { port: 8443, service: 'HTTPS-Alt', risk: 'medium', desc: 'Common alternative HTTPS port — admin panels, Java apps' },
   818→  { port: 3000, service: 'Dev Server', risk: 'high', desc: 'Node.js/Express default — often development servers exposed' },
   819→  { port: 5000, service: 'Dev Server', risk: 'high', desc: 'Flask/React dev default — often development servers exposed' },
   820→  { port: 8000, service: 'HTTP-Dev', risk: 'high', desc: 'Common dev port — Django, Python HTTP server, Go dev' },
   821→  { port: 8888, service: 'Jupyter/Proxy', risk: 'critical', desc: 'Jupyter Notebook default or HTTP proxy — often admin access' },
   822→  { port: 9090, service: 'Prometheus/Proxy', risk: 'high', desc: 'Prometheus monitoring or alternative web server' },
   823→  { port: 9200, service: 'Elasticsearch', risk: 'critical', desc: 'Elasticsearch REST API — often unprotected, data exposure' },
   824→  { port: 5601, service: 'Kibana', risk: 'high', desc: 'Kibana dashboard — may expose log data and analytics' },
   825→  { port: 15672, service: 'RabbitMQ Mgmt', risk: 'high', desc: 'RabbitMQ management interface — default credentials common' },
   826→  { port: 27017, service: 'MongoDB', risk: 'critical', desc: 'MongoDB default port — data exposure if unauthenticated' },
   827→];
   828→
   829→async function probePorts(domain: string): Promise<Array<{
   830→  title: string; severity: string; category: string;
   831→  description: string; evidence: string; asset: string;
   832→}>> {
   833→  const findings: ReturnType<typeof probePorts> = [];
   834→
   835→  const portChecks = await Promise.all(
   836→    WEB_PORTS.map(async ({ port, service, risk, desc }) => {
   837→      // Skip port 443 for quick scans, skip 80
   838→      try {
   839→        const result = run(
   840→          `curl -sI --max-time 3 -k https://${domain}:${port} 2>/dev/null || curl -sI --max-time 3 http://${domain}:${port} 2>/dev/null`,
   841→          5000
   842→        );
   843→        if (result && result.includes('HTTP/')) {
   844→          const statusLine = result.split('\n')[0];
   845→          const serverMatch = result.match(/server:\s*(.+)/i);
   846→          const serverInfo = serverMatch ? ` [${serverMatch[1].trim()}]` : '';
   847→          return { port, service, risk, desc, open: true, statusLine, serverInfo };
   848→        }
   849→      } catch { /* port closed or filtered */ }
   850→      return { port, service, risk, desc, open: false, statusLine: '', serverInfo: '' };
   851→    })
   852→  );
   853→
   854→  const openPorts = portChecks.filter(p => p.open);
   855→
   856→  for (const { port, service, risk, desc, statusLine, serverInfo } of openPorts) {
   857→    // Skip informational findings for standard ports that are expected
   858→    if (port === 443) {
   859→      findings.push({
   860→        title: `Port ${port}/tcp OPEN — ${service}`,
   861→        severity: 'info', category: 'port',
   862→        description: `HTTPS (port 443) is open${serverInfo}. This is the standard HTTPS port and expected for web services.`,
   863→        evidence: statusLine + serverInfo,
   864→        asset: `${domain}:${port}`,
   865→      });
   866→    } else if (port === 80) {
   867→      findings.push({
   868→        title: `Port ${port}/tcp OPEN — ${service}`,
   869→        severity: 'low', category: 'port',
   870→        description: `HTTP (port 80) is open. This is expected but should redirect to HTTPS. If it serves content without redirect, the site may be accessible over unencrypted HTTP.`,
   871→        evidence: statusLine + serverInfo,
   872→        asset: `${domain}:${port}`,
   873→      });
   874→    } else {
   875→      findings.push({
   876→        title: `Port ${port}/tcp OPEN — ${service}`,
   877→        severity: risk, category: 'port',
   878→        description: `${service} is accessible on port ${port}. ${desc}. Exposing non-standard ports to the internet increases the attack surface. ${risk === 'critical' ? 'This port hosts a high-risk service that should NEVER be publicly accessible without strong authentication and network segmentation. Immediate remediation required.' : 'Review whether this service needs to be publicly accessible and implement proper access controls.'}`,
   879→        evidence: statusLine + serverInfo,
   880→        asset: `${domain}:${port}`,
   881→      });
   882→    }
   883→  }
   884→
   885→  return findings;
   886→}

   888→// ═══════════════════════════════════════════════════════════════════════
   889→// 6. REAL TRACEROUTE (Network Path Analysis)
   890→// ═══════════════════════════════════════════════════════════════════════
   891→function tracerouteAnalysis(domain: string): Array<{
   892→  title: string; severity: string; category: string;
   893→  description: string; evidence: string; asset: string;
   894→}> {
   895→  const findings: ReturnType<typeof tracerouteAnalysis> = [];
   896→
   897→  const tracerouteOutput = run(`traceroute -m 15 -w 2 ${domain} 2>&1`, 35000);
   898→  if (tracerouteOutput) {
   899→    const hops = tracerouteOutput.split('\n').filter(l => l.match(/^\s*\d/)).length;
   900→    const lastHop = tracerouteOutput.split('\n').filter(l => l.match(/^\s*\d/)).pop() || '';
   901→    const hasAsterisks = tracerouteOutput.includes('* * *');
   902→
   903→    findings.push({
   904→      title: `Network Path: ${hops} Hops to Destination`,
   905→      severity: 'info', category: 'dns',
   906→      description: `Traceroute to ${domain} completes in ${hops} hops. ${hasAsterisks ? 'Firewall or rate-limiting detected (filtered hops observed). ' : ''}Last visible hop: ${lastHop.trim()}. Fewer hops generally indicate better network proximity and lower latency.`,
   907→      evidence: `Hops: ${hops}, Firewall: ${hasAsterisks ? 'detected' : 'not detected'}`,
   908→      asset: domain,
   909→    });
   910→
   911→    if (hops <= 5) {
   912→      findings.push({
   913→        title: 'Low Hop Count — Likely CDN/Edge Hosted',
   914→        severity: 'info', category: 'dns',
   915→        description: `Only ${hops} hops to reach the destination. This suggests the target is hosted on or very close to a CDN edge node (e.g., Cloudflare, AWS CloudFront, Akamai). CDN-hosted targets benefit from DDoS protection and distributed availability.`,
   916→        evidence: `Hop count: ${hops}`,
   917→        asset: domain,
   918→      });
   919→    }
   920→  }
   921→
   922→  return findings;
   923→}
   924→
   925→// ═══════════════════════════════════════════════════════════════════════
   926→// MAIN SCAN ENDPOINT
   927→// ═══════════════════════════════════════════════════════════════════════
   928→export async function POST(request: NextRequest) {
   929→  try {
   930→    const body = await request.json();
   931→    const { domain, scanType } = body;
   932→
   933→    if (!domain || typeof domain !== 'string') {
   934→      return NextResponse.json({ error: 'Domain is required' }, { status: 400 });
   935→    }
   936→
   937→    const cleanDomain = domain
   938→      .replace(/^(https?:\/\/)?(www\.)?/, '')
   939→      .replace(/\/.*$/, '')
   940→      .replace(/:\d+$/, '')
   941→      .toLowerCase();
   942→
   943→    // Verify domain resolves before scanning
   944→    const preCheckIp = resolves(cleanDomain);
   945→    if (!preCheckIp) {
   946→      return NextResponse.json(
   947→        { error: `Domain "${cleanDomain}" does not resolve to any IP address. Check the domain name and try again.` },
   948→        { status: 400 }
   949→      );
   950→    }
   951→
   952→    // Create or find target with REAL IP
   953→    let target = await db.scanTarget.findFirst({ where: { domain: cleanDomain } });
   954→    if (!target) {
   955→      target = await db.scanTarget.create({
   956→        data: { domain: cleanDomain, ip: preCheckIp },
   957→      });
   958→    } else {
   959→      await db.scanTarget.update({ where: { id: target.id }, data: { ip: preCheckIp } });
   960→    }
   961→
   962→    // Create scan record
   963→    const scan = await db.scan.create({
   964→      data: {
   965→        targetId: target.id,
   966→        status: 'running',
   967→        scanType: scanType || 'full',
   968→      },
   969→    });
   970→
   971→    const isQuick = scanType === 'quick';
   972→    const allFindings: Array<{
   973→      title: string; severity: string; category: string;
   974→      description: string; evidence: string; asset: string;
   975→    }> = [];
   976→    const allTechnologies = new Set<string>();
   977→
   978→    // ── Run ALL real reconnaissance ────────────────────────────────
   979→
   980→    // 1. DNS Enumeration (always)
   981→    const dnsResult = enumerateDNS(cleanDomain);
   982→    allFindings.push(...dnsResult.findings);
   983→
   984→    // 2. Subdomain enumeration
   985→    const subdomainFindings = await enumerateSubdomains(cleanDomain);
   986→    allFindings.push(...subdomainFindings);
   987→
   988→    // 3. HTTP Header Analysis (always)
   989→    const httpResult = analyzeHTTPHeaders(cleanDomain);
   990→    allFindings.push(...httpResult.findings);
   991→    httpResult.technologies.forEach(t => allTechnologies.add(t));
   992→
   993→    // 4. SSL/TLS Certificate Analysis (always)
   994→    const sslResult = analyzeSSL(cleanDomain);
   995→    allFindings.push(...sslResult.findings);
   996→    sslResult.technologies.forEach(t => allTechnologies.add(t));
   997→
   998→    // 5. Port probing (full scan only)
   999→    if (!isQuick) {
  1000→      const portFindings = await probePorts(cleanDomain);
  1001→      allFindings.push(...portFindings);
  1002→
  1003→      // 6. Traceroute (full scan only)
  1004→      const traceFindings = tracerouteAnalysis(cleanDomain);
  1005→      allFindings.push(...traceFindings);
  1006→    }
  1007→
  1008→    // ── Add technology findings ────────────────────────────────────
  1009→    for (const tech of allTechnologies) {
  1010→      allFindings.push({
  1011→        title: `Technology Detected: ${tech}`,
  1012→        severity: 'info', category: 'technology',
  1013→        description: `${tech} was identified on ${cleanDomain} through HTTP header analysis, SSL certificate metadata, or network response characteristics. Technology fingerprinting helps build a complete infrastructure profile for targeted security assessment.`,
  1014→        evidence: `Detected via HTTP headers, SSL certificate, or response analysis`,
  1015→        asset: tech,
  1016→      });
  1017→    }
  1018→
  1019→    // ── Check for wildcard DNS ────────────────────────────────────
  1020→    const randomSub = `nonexistent-random-test-98765.${cleanDomain}`;
  1021→    const wildcardIp = resolves(randomSub);
  1022→    if (wildcardIp) {
  1023→      allFindings.push({
  1024→        title: 'Wildcard DNS Record Detected',
  1025→        severity: 'medium', category: 'dns',
  1026→        description: `A wildcard DNS record is configured for ${cleanDomain} — even nonexistent subdomains like ${randomSub} resolve to ${wildcardIp}. This means subdomain enumeration results may include false positives. Wildcard DNS can also be abused for subdomain takeover if the wildcard points to a service that can be claimed.`,
  1027→        evidence: `${randomSub} resolved to ${wildcardIp}`,
  1028→        asset: cleanDomain,
  1029→      });
  1030→    }
  1031→
  1032→    // ── Save all findings to DB ───────────────────────────────────
  1033→    for (const f of allFindings) {
  1034→      await db.finding.create({
  1035→        data: {
  1036→          scanId: scan.id,
  1037→          title: f.title,
  1038→          severity: f.severity,
  1039→          category: f.category,
  1040→          description: f.description,
  1041→          evidence: f.evidence,
  1042→          asset: f.asset,
  1043→        },
  1044→      });
  1045→    }
  1046→
  1047→    // ── Calculate risk score from real data ───────────────────────
  1048→    const critical = allFindings.filter(f => f.severity === 'critical').length;
  1049→    const high = allFindings.filter(f => f.severity === 'high').length;
  1050→    const medium = allFindings.filter(f => f.severity === 'medium').length;
  1051→    const low = allFindings.filter(f => f.severity === 'low').length;
  1052→    const info = allFindings.filter(f => f.severity === 'info').length;
  1053→    const riskScore = Math.min(100, Math.round(critical * 25 + high * 15 + medium * 8 + low * 3 + info * 1));
  1054→
  1055→    await db.scan.update({
  1056→      where: { id: scan.id },
  1057→      data: {
  1058→        status: 'completed',
  1059→        completedAt: new Date(),
  1060→        riskScore,
  1061→        totalVulns: allFindings.length,
  1062→        criticalCount: critical,
  1063→        highCount: high,
  1064→        mediumCount: medium,
  1065→        lowCount: low,
  1066→        infoCount: info,
  1067→      },
  1068→    });
  1069→
  1070→    return NextResponse.json({
  1071→      success: true,
  1072→      scan: {
  1073→        id: scan.id,
  1074→        domain: cleanDomain,
  1075→        status: 'completed',
  1076→        riskScore,
  1077→        totalVulns: allFindings.length,
  1078→        critical,
  1079→        high,
  1080→        medium,
  1081→        low,
  1082→        info,
  1083→        findings: allFindings.map(f => ({
  1084→          id: f.title.toLowerCase().replace(/\s+/g, '-').substring(0, 20),
  1085→          title: f.title,
  1086→          severity: f.severity,
  1087→          category: f.category,
  1088→          description: f.description,
  1089→          evidence: f.evidence,
  1090→          asset: f.asset,
  1091→        })),
  1092→      },
  1093→    });
  1094→  } catch (error) {
  1095→    console.error('Scan error:', error);
  1096→    return NextResponse.json({ error: 'Scan failed: ' + (error instanceof Error ? error.message : 'unknown') }, { status: 500 });
  1097→  }
  1098→}
