    55→    55→    def __init__(self):
    56→    56→        self._handlers = {}
    57→    57→
    58→    58→    def emit(self, event_type: str, **kwargs):
    59→    59→        results = []
    60→    60→        for cb in self._handlers.get(event_type, []):
    61→    61→            results.append(cb(**kwargs))
    62→    62→        return results
    63→    63→
    64→    64→    def on(self, event_type: str, callback):
    65→    65→        self._handlers.setdefault(event_type, []).append(callback)
    66→    66→
    67→    67→    def off(self, event_type: str, callback):
    68→    68→        if event_type in self._handlers:
    69→    69→            self._handlers[event_type] = [
    70→    70→                cb for cb in self._handlers[event_type] if cb != callback
    71→    71→            ]
