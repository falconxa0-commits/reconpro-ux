    75→function getInitials(name: string): string {
    76→  return name
    77→    .split(' ')
    78→    .map((n) => n[0])
    79→    .join('')
    80→    .toUpperCase();
    81→}
    82→
    83→function getRoleColor(role: string): string {
    84→  switch (role) {
    85→    case 'Admin':
    86→      return 'bg-[rgba(168,85,247,0.15)] text-[#888888] border-[rgba(168,85,247,0.3)]';
    87→    case 'Security Lead':
    88→      return 'bg-[rgba(52,211,153,0.15)] text-[#00ff88] border-[rgba(52,211,153,0.3)]';
    89→    case 'Analyst':
    90→      return 'bg-[rgba(88,166,255,0.15)] text-[#44aaff] border-[rgba(88,166,255,0.3)]';
    91→    case 'Viewer':
    92→      return 'bg-[rgba(139,148,158,0.15)] text-[#444444] border-[rgba(139,148,158,0.3)]';
    93→    default:
    94→      return 'bg-[rgba(139,148,158,0.15)] text-[#444444] border-[rgba(139,148,158,0.3)]';
    95→  }
    96→}
    97→
    98→function getStatusColor(status: string): string {
    99→  switch (status) {
   100→    case 'online':
   101→      return 'bg-[#00ff88]';
   102→    case 'away':
   103→      return 'bg-[#d29922]';
   104→    case 'offline':
   105→      return 'bg-[#333333]';
   106→    default:
   107→      return 'bg-[#333333]';
   108→  }
   109→}
   110→
   111→function getAvatarBg(name: string): string {
   112→  const colors = [
   113→    'rgba(52,211,153,0.15)',
   114→    'rgba(88,166,255,0.15)',
   115→    'rgba(168,85,247,0.15)',
   116→    'rgba(244,63,94,0.15)',
   117→    'rgba(210,153,34,0.15)',
   118→    'rgba(121,192,255,0.15)',
   119→    'rgba(255,123,114,0.15)',
   120→    'rgba(187,128,220,0.15)',
   121→  ];
   122→  const idx = name.charCodeAt(0) % colors.length;
   123→  return colors[idx];
   124→}
   125→
   126→function getAvatarTextColor(name: string): string {
   127→  const colors = ['#00ff88', '#44aaff', '#888888', '#ff3355', '#d29922', '#79c0ff', '#ff7b72', '#bb80d4'];
   128→  const idx = name.charCodeAt(0) % colors.length;
   129→  return colors[idx];
   130→}

   154→export function TeamManagement() {

   156→  const [teams, setTeams] = useState<Team[]>([]);
   157→  const [loading, setLoading] = useState(true);
   158→  const [error, setError] = useState('');
   159→  const [searchQuery, setSearchQuery] = useState('');
   160→  const [inviteOpen, setInviteOpen] = useState(false);

   165→

   170→

   173→  const [roleEditId, setRoleEditId] = useState('');

   175→

   189→    try {
   190→      const res = await fetch('/api/teams');

   195→      console.error('Failed to fetch teams:', err);

   208→      try {
   209→        await fetchAllRef();

   212→      }

   218→    (m) =>

   221→      m.role.toLowerCase().includes(searchQuery.toLowerCase())

   225→  const onlineCount = members.filter((m) => m.status === 'online').length;
   226→
   227→  const handleInvite = async () => {
   228→    if (!inviteName || !inviteEmail) return;
   229→    try {
   230→      await fetch('/api/members', {
   231→        method: 'POST',
   232→        headers: { 'Content-Type': 'application/json' },

   236→      setInviteEmail('');
   237→      setInviteName('');
   238→      setInviteRole('Viewer');
   239→      await fetchMembers();
   240→    } catch (err) {
   241→      console.error('Failed to invite member:', err);
   242→    }
   243→  };

   246→    if (!roleEditId || !roleEditValue) return;

   248→      await fetch('/api/members', {
   249→        method: 'PATCH',
   250→        headers: { 'Content-Type': 'application/json' },
   251→        body: JSON.stringify({ id: roleEditId, role: roleEditValue.toLowerCase().replace(' ', '_') }),
   252→      });
   253→      setRoleEditOpen(false);
   254→      await fetchMembers();
   255→    } catch (err) {
   256→      console.error('Failed to update role:', err);
   257→    }
   258→  };
   259→
   260→  const handleDeleteMember = async (id: string) => {
   261→    try {
   262→      await fetch(`/api/members?id=${id}`, { method: 'DELETE' });
   263→      await fetchMembers();
   264→    } catch (err) {
   265→      console.error('Failed to delete member:', err);
   266→    }
   267→  };

   270→    if (!teamName) return;
   271→    try {
   272→      await fetch('/api/teams', {
   273→        method: 'POST',
   274→        headers: { 'Content-Type': 'application/json' },
   275→        body: JSON.stringify({ name: teamName, description: teamDesc }),
   276→      });
   277→      setTeamOpen(false);
   278→      setTeamName('');
   279→      setTeamDesc('');
   280→      await fetchTeams();
   281→    } catch (err) {
   282→      console.error('Failed to create team:', err);
   283→    }
   284→  };
   285→
   286→  const handleDeleteTeam = async (id: string) => {
   287→    try {
   288→      await fetch(`/api/teams?id=${id}`, { method: 'DELETE' });
   289→      await fetchTeams();
   290→    } catch (err) {
   291→      console.error('Failed to delete team:', err);
   292→    }
   293→  };
   294→
   295→  const handleCopyId = (id: string) => {
   296→    navigator.clipboard.writeText(id);
   297→    setCopiedId(id);
   298→    setTimeout(() => setCopiedId(null), 2000);
   299→  };
   300→
   301→  const openRoleEdit = (member: Member) => {

   303→    setRoleEditValue(member.role);
   304→    setRoleEditOpen(true);
   305→  };
   306→
   307→  if (loading) {
   308→    return (
   309→      <div className="w-full flex items-center justify-center py-20">
   310→        <p className="text-[#444444]">Loading...</p>
   311→      </div>
   312→    );
   313→  }
   314→
   315→  if (error) {
   316→    return (
   317→      <div className="w-full flex flex-col items-center justify-center gap-3 py-20">
   318→        <AlertCircle className="h-8 w-8 text-red-400" />
   319→        <p className="text-sm text-red-400">{error}</p>
   320→        <Button variant="outline" size="sm" onClick={fetchAllRef} className="border-zinc-700 text-white hover:bg-zinc-800">
   321→          <RefreshCw className="mr-2 h-3.5 w-3.5" />
   322→          Retry
   323→        </Button>
   324→      </div>
   325→    );
   326→  }
   327→
   328→  return (
   329→    <motion.div
   330→      variants={containerVariants}
   331→      initial="hidden"
   332→      animate="visible"
   333→      className="w-full space-y-6"
   334→    >
   335→      {/* ── Header ──────────────────────────────────────────────────────── */}
   336→      <motion.div variants={itemVariants} className="flex flex-col sm:flex-row sm:items-center justify-between gap-4">
   337→        <div className="flex items-center gap-3">
   338→          <div className="p-2 rounded-lg bg-[rgba(52,211,153,0.1)] border border-[rgba(52,211,153,0.2)]">
   339→            <Users className="w-5 h-5 text-[#00ff88]" />
   340→          </div>
   341→          <div>
   342→            <h2 className="text-xl font-bold text-[#f0f0f0]">Team Management</h2>
   343→            <p className="text-sm text-[#444444]">Manage members, roles, and team assignments</p>
   344→          </div>
   345→        </div>
   346→        <Dialog open={inviteOpen} onOpenChange={setInviteOpen}>
   347→          <DialogTrigger asChild>
   348→            <Button className="bg-[#00ff88] hover:bg-[#00cc6a] text-[#0a0d14] font-semibold gap-2">
   349→              <UserPlus className="w-4 h-4" />
   350→              Invite Member
   351→            </Button>
   352→          </DialogTrigger>
   353→          <DialogContent className="bg-[#080b14] border-[#21262d] text-[#f0f0f0]">
   354→            <DialogHeader>
   355→              <DialogTitle className="text-[#f0f0f0]">Invite New Member</DialogTitle>
   356→              <DialogDescription className="text-[#444444]">
   357→                Send an invitation to add a new team member.
   358→              </DialogDescription>
   359→            </DialogHeader>
   360→            <div className="space-y-4 py-4">
   361→              <div className="space-y-2">
   362→                <label className="text-sm font-medium text-[#f0f0f0]">Name</label>
   363→                <Input
   364→                  placeholder="John Doe"
   365→                  value={inviteName}
   366→                  onChange={(e) => setInviteName(e.target.value)}
   367→                  className="bg-[#050710] border-[#21262d] text-[#f0f0f0] placeholder:text-[#333333] focus:border-[#00ff88]"
   368→                />
   369→              </div>
   370→              <div className="space-y-2">
   371→                <label className="text-sm font-medium text-[#f0f0f0]">Email Address</label>
   372→                <Input
   373→                  placeholder="colleague@company.com"
   374→                  value={inviteEmail}

   376→                  className="bg-[#050710] border-[#21262d] text-[#f0f0f0] placeholder:text-[#333333] focus:border-[#00ff88]"

   380→                <label className="text-sm font-medium text-[#f0f0f0]">Role</label>
   381→                <div className="grid grid-cols-2 gap-2">
   382→                  {['Admin', 'Security Lead', 'Analyst', 'Viewer'].map((r) => (
   383→                    <button
   384→                      key={r}
   385→                      onClick={() => setInviteRole(r)}
   386→                      className={`px-3 py-2 rounded-lg text-sm font-medium border transition-all ${
   387→                        inviteRole === r
   388→                          ? 'bg-[rgba(52,211,153,0.15)] border-[rgba(52,211,153,0.4)] text-[#00ff88]'
   389→                          : 'bg-[#050710] border-[#21262d] text-[#444444] hover:border-[#30363d]'

   391→                    >
   392→                      {r}
   393→                    </button>
   394→                  ))}
   395→                </div>
   396→              </div>
   397→            </div>
   398→            <DialogFooter>
   399→              <Button variant="outline" onClick={() => setInviteOpen(false)} className="border-[#21262d] text-[#444444] hover:bg-[#050710]">
   400→                Cancel
   401→              </Button>

   403→                Send Invitation
   404→              </Button>
   405→            </DialogFooter>
   406→          </DialogContent>
   407→        </Dialog>
   408→      </motion.div>
   409→
   410→      {/* ── Stats Row ───────────────────────────────────────────────────── */}
   411→      <motion.div variants={itemVariants} className="grid grid-cols-2 lg:grid-cols-4 gap-4">
   412→        {[
   413→          { label: 'Total Members', value: members.length, icon: Users, color: '#00ff88' },
   414→          { label: 'Active Teams', value: teams.length, icon: Shield, color: '#44aaff' },
   415→          { label: 'Pending Invites', value: 0, icon: AlertCircle, color: '#d29922' },
   416→          { label: 'Admins', value: adminCount, icon: Crown, color: '#888888' },

   422→            style={{ boxShadow: '0 0 0px transparent' }}

   426→              <stat.icon className="w-4 h-4" style={{ color: stat.color }} />
   427→            </div>
   428→            <p className="text-2xl font-bold text-[#f0f0f0]">{stat.value}</p>
   429→          </motion.div>

   432→
   433→      {/* ── Tabs ────────────────────────────────────────────────────────── */}
   434→      <motion.div variants={itemVariants}>
   435→        <Tabs defaultValue="members" className="w-full">

   437→            <TabsTrigger
   438→              value="members"
   439→              className="data-[state=active]:bg-[rgba(52,211,153,0.1)] data-[state=active]:text-[#00ff88] text-[#444444]"

   441→              <Users className="w-4 h-4 mr-1.5" />
   442→              Members
   443→            </TabsTrigger>
   444→            <TabsTrigger
   445→              value="teams"
   446→              className="data-[state=active]:bg-[rgba(52,211,153,0.1)] data-[state=active]:text-[#00ff88] text-[#444444]"

   449→              Teams

   452→
   453→          {/* ── Members Tab ─────────────────────────────────────────────── */}
   454→          <TabsContent value="members" className="mt-4">
   455→            <motion.div

   470→
   471→              {/* Table */}
   472→              <div className="max-h-[400px] overflow-y-auto">
   473→                {members.length === 0 ? (
   474→                  <div className="px-4 py-8 text-center">

   476→                    <p className="text-sm text-[#444444]">No members yet. Click &quot;Invite Member&quot; to add your first team member.</p>

   479→                <Table>
   480→                  <TableHeader>
   481→                    <TableRow className="border-b border-[#21262d] hover:bg-transparent">

   483→                      <TableHead className="text-[#444444] font-medium text-xs uppercase tracking-wider">Role</TableHead>
   484→                      <TableHead className="text-[#444444] font-medium text-xs uppercase tracking-wider hidden md:table-cell">Last Active</TableHead>

   486→                      <TableHead className="text-[#444444] font-medium text-xs uppercase tracking-wider text-right">Actions</TableHead>

   488→                  </TableHeader>
   489→                  <TableBody>

   506→                                    color: getAvatarTextColor(member.name),
