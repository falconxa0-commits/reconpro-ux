   648→      const target = parseFloat(el.textContent);
   649→      if (isNaN(target)) return;
   650→      const isFloat = el.textContent.includes('.');
   651→      const duration = 1200;
   652→      const start = performance.now();
   653→      el.textContent = '0';
   654→
   655→      function update(now) {
   656→        const elapsed = now - start;
   657→        const progress = Math.min(elapsed / duration, 1);
   658→        // easeOutQuart
   659→        const eased = 1 - Math.pow(1 - progress, 4);
   660→        const current = target * eased;
   661→        el.textContent = isFloat ? current.toFixed(1) : Math.round(current);
   662→        if (progress < 1) requestAnimationFrame(update);
   663→      }
   664→      requestAnimationFrame(update);
   665→    });
   666→
   667→    // Intersection observer for scroll animations

   669→      entries.forEach(function(entry) {
   670→        if (entry.isIntersecting) {

   672→        }
