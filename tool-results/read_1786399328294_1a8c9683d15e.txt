   459→def _stage_16_ai_endpoint_discovery(base_url: str, timeout: int = 8,
   460→                                      verify_tls: bool = True) -> List[Finding]:
   461→    """v9.1.0 Stage 16: AI-specific endpoint discovery and vendor fingerprinting."""
   462→    findings: List[Finding] = []
   463→    host = base_url.replace("https://", "").replace("http://", "").split("/")[0]
   464→
   465→    try:
   466→        from ..ai_red_team import AIEndpointDiscovery, AIVendorFingerprinter, SecretExtractor
   467→        from ..ai_cve_db import AICVEDatabase
   468→
   469→        # Discover AI endpoints
   470→        discovery = AIEndpointDiscovery()
   471→        endpoints = discovery.probe(host, base_url)
   472→        if endpoints:
   473→            ep_desc = ", ".join(e.get("path", "?") for e in endpoints[:8])
   474→            findings.append(Finding(
   475→                title="AI endpoints: {} discovered".format(len(endpoints)),
   476→                severity="medium", category="ai_endpoint_discovery",
   477→                module="gorgon",
   478→                description="AI-specific endpoints found: {}".format(ep_desc),
   479→                evidence="{} AI endpoints discovered".format(len(endpoints)),
   480→                asset=host, points_deducted=4,
   481→            ))
   482→
   483→        # Vendor fingerprinting
   484→        fp = AIVendorFingerprinter()
   485→        vendors = fp.fingerprint(base_url)
   486→        for v in vendors[:3]:
   487→            vendor_name = v.get("vendor", v.get("name", "Unknown"))
   488→            vendor_score = v.get("score", v.get("confidence", 0))
   489→            evidence_list = v.get("evidence", v.get("clues", []))
   490→            findings.append(Finding(
   491→                title="AI vendor detected: {}".format(vendor_name),
   492→                severity="medium", category="ai_vendor_fingerprint",
   493→                module="gorgon",
   494→                description="AI vendor {} detected (score: {})".format(vendor_name, vendor_score),
   495→                evidence="Evidence: {}".format("; ".join(str(e) for e in evidence_list[:5])),
   496→                asset=host, points_deducted=3,
   497→            ))
   498→
   499→        # AI CVE matching
   500→        db = AICVEDatabase()
   501→        if vendors:
   502→            all_cves = []
   503→            for v in vendors:
   504→                product = v.get("vendor", v.get("name", ""))
   505→                if product:
   506→                    cves = db.search(product=product)
   507→                    all_cves.extend(cves)
   508→            if all_cves:
   509→                for cve in all_cves[:3]:
   510→                    findings.append(Finding(
   511→                        title="AI CVE: {} (CVSS {})".format(cve["cve"], cve["cvss"]),
   512→                        severity="critical" if cve["cvss"] >= 9.0 else "high",
   513→                        category="ai_cve",
   514→                        module="gorgon",
   515→                        description="{}: {}".format(cve["cve"], cve["description"][:100]),
   516→                        evidence="Product: {}, Type: {}, CVSS: {}".format(cve["product"], cve["type"], cve["cvss"]),
   517→                        asset=host, points_deducted=12 if cve["cvss"] >= 9.0 else 8,
   518→                        remediation="Upgrade {} to {} or later.".format(cve["product"], cve.get("fix_version", "latest")),
   519→                    ))
   520→
   521→        # Secret extraction from root response
   522→        extractor = SecretExtractor()
   523→        try:
   524→            resp = http_probe(base_url, timeout=timeout, verify_tls=verify_tls)
   525→            body_text = resp.get("body", "") or ""
   526→            resp_headers = resp.get("headers", {}) or {}
   527→            secrets = extractor.extract(body_text, resp_headers)
   528→            for s in secrets[:5]:
   529→                sec_type = s.get("type", "unknown")
   530→                sec_desc = s.get("description", sec_type)
   531→                masked_val = s.get("value_masked", "***")
   532→                findings.append(Finding(
   533→                    title="Secret exposed: {}".format(sec_type),
   534→                    severity="critical", category="secret_exposure",
   535→                    module="gorgon",
   536→                    description="{} found in response: {} ({})".format(sec_type, sec_desc, masked_val),
   537→                    evidence="Source: {}, Pattern: {}".format(s.get("source", "?"), s.get("description", "?")),
   538→                    asset=host, points_deducted=12,
   539→                    remediation="Remove exposed secrets from responses. Rotate compromised credentials.",
   540→                ))
   541→        except Exception:
   542→            pass
   543→    except Exception:
   544→        pass
   545→
   546→    return findings

   579→def _stage_17_kill_chain(base_url: str, timeout: int = 8,
   580→                          verify_tls: bool = True) -> List[Finding]:
   581→    """v9.2.0 Stage 17: Kill Chain completion scoring + Shadow IT discovery."""
   582→    findings: List[Finding] = []
   583→    host = base_url.replace("https://", "").replace("http://", "").split("/")[0]
   584→
   585→    # Kill Chain analysis
   586→    try:
   587→        from ..kill_chain import KillChainEngine
   588→        kc = KillChainEngine()
   589→        kc_result = kc.run_kill_chain(host, base_url, timeout=min(timeout, 5), verify_tls=verify_tls)
   590→        completion = kc_result.get("kill_chain_completion", 0)
   591→        overall_risk = kc_result.get("overall_risk", {}).get("risk_score", 0)
   592→        risk_level = kc_result.get("overall_risk", {}).get("risk_level", "unknown")
   593→        findings.append(Finding(
   594→            title="Kill Chain: {:.0f}% complete (risk: {}/10)".format(completion * 100, overall_risk),
   595→            severity="critical" if overall_risk >= 7 else "high" if overall_risk >= 4 else "medium",
   596→            category="kill_chain",
   597→            module="gorgon",
   598→            description="Full Cyber Kill Chain analysis: {} completion, risk level {}, "
   599→                        "{} attack paths identified.".format(
   600→                "{:.0f}%".format(completion * 100), risk_level,
   601→                kc_result.get("attack_tree", {}).get("path_count", 0)),
   602→            evidence="Phases: {}".format(", ".join(
   603→                "{}={}".format(p.get("phase", "?"), p.get("status", "?"))
   604→                for p in kc_result.get("phases", [])[:7])),
   605→            asset=host, points_deducted=int(overall_risk),
   606→            remediation="Address each completed kill chain phase to increase defensive posture.",
   607→        ))
   608→    except Exception:
   609→        pass
   610→
   611→    # Shadow IT discovery
   612→    try:
   613→        from ..shadow_it import ShadowITScanner
   614→        sis = ShadowITScanner()
   615→        shadow = sis.scan(host, base_url, timeout=min(timeout, 5))
   616→        shadow_findings = shadow.get("findings", [])
   617→        high_decay = [f for f in shadow_findings if f.get("decay_score", 0) >= 60]
   618→        if high_decay:
   619→            findings.append(Finding(
   620→                title="Shadow IT: {} abandoned assets detected".format(len(high_decay)),
   621→                severity="high" if len(high_decay) >= 5 else "medium",
   622→                category="shadow_it",
   623→                module="gorgon",
   624→                description="Abandoned/forgotten infrastructure discovered: {}".format(
   625→                    ", ".join(f.get("url", "?")[:60] for f in high_decay[:5])),
   626→                evidence="Decay scores: {}".format(", ".join(
   627→                    str(f.get("decay_score", 0)) for f in high_decay[:5])),
   628→                asset=host, points_deducted=8,
   629→                remediation="Audit and decommission abandoned infrastructure. Shadow IT increases attack surface.",
   630→            ))
   631→    except Exception:
   632→        pass
   633→
   634→    return findings
