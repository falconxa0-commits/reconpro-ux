   124→function useCountdown(targetDate: string) {
   125→  const [timeLeft, setTimeLeft] = useState({ years: 0, months: 0, days: 0, hours: 0, minutes: 0, seconds: 0 });
   126→
   127→  useEffect(() => {
   128→    const tick = () => {
   129→      const target = new Date(targetDate);
   130→      const now = new Date();
   131→      const diff = target.getTime() - now.getTime();
   132→
   133→      if (diff <= 0) {
   134→        setTimeLeft({ years: 0, months: 0, days: 0, hours: 0, minutes: 0, seconds: 0 });
   135→        return;
   136→      }
   137→
   138→      // Approximate years/months
   139→      let years = target.getFullYear() - now.getFullYear();
   140→      let months = target.getMonth() - now.getMonth();
   141→      let days = target.getDate() - now.getDate();
   142→      if (days < 0) {
   143→        months--;
   144→        const prevMonth = new Date(target.getFullYear(), target.getMonth(), 0);
   145→        days += prevMonth.getDate();
   146→      }
   147→      if (months < 0) {
   148→        years--;
   149→        months += 12;
   150→      }
   151→
   152→      const hours = 23 - now.getHours();
   153→      const minutes = 59 - now.getMinutes();
   154→      const seconds = 59 - now.getSeconds();
   155→
   156→      setTimeLeft({ years, months, days, hours, minutes, seconds });
   157→    };
   158→
   159→    tick();
   160→    const interval = setInterval(tick, 1000);
   161→    return () => clearInterval(interval);
   162→  }, [targetDate]);
   163→
   164→  return timeLeft;
   165→}

   372→function AssetBreakdown({ assets, sortBy, onSortChange }: {

   379→      if (sortBy === 'doomDate') return a.estimatedBreakYear - b.estimatedBreakYear;

   382→        return (order[a.riskLevel] ?? 4) - (order[b.riskLevel] ?? 4);

   486→function MigrationRoadmap({ steps }: { steps: MigrationStep[] }) {
   487→  const totalCost = useMemo(() => {
   488→    // Parse cost estimates and sum ranges
   489→    let minTotal = 0;
   490→    let maxTotal = 0;
   491→    for (const step of steps) {
   492→      const match = step.costEstimate.match(/\$(\d+)K-\$(\d+)K/);
   493→      if (match) {
   494→        minTotal += parseInt(match[1]);
   495→        maxTotal += parseInt(match[2]);
   496→      }
   497→    }
   498→    return `$${minTotal}K-$${maxTotal}K`;
   499→  }, [steps]);
   500→
