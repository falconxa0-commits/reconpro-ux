    52→    52→#  EngineeringFact — the atomic unit of repository knowledge
    53→    53→# ======================================================================
    54→    54→
    55→    55→
    56→    56→@dataclass
    57→    57→class EngineeringFact:
    58→    58→    """A single piece of meta-knowledge about a repository.
    59→    59→
    60→    60→    Attributes:
    61→    61→        key:        Dot-path or slash-path identifier (e.g.
    62→    62→                    ``"arch/auth-decision"``, ``"perf/baseline/api-latency"``).
    63→    63→        value:      The stored value — any JSON-serialisable object.
    64→    64→        timestamp:  Unix epoch seconds when the fact was recorded.
    65→    65→        source:     Provenance tag (module name, agent id, "manual", etc.).
    66→    66→        tags:       Frozen set of classification tags for indexing.
    67→    67→        confidence: 0.0 – 1.0 certainty.  Facts below a query threshold are
    68→    68→                    excluded from results.
    69→    69→        expires:    Unix epoch seconds after which the fact is stale, or
    70→    70→                    ``None`` for permanent facts.
    71→    71→    """
    72→    72→
    73→    73→    key: str
    74→    74→    value: Any
    75→    75→    timestamp: float = field(default_factory=time.time)
    76→    76→    source: str = ""
    77→    77→    tags: FrozenSet[str] = field(default_factory=frozenset)
    78→    78→    confidence: float = DEFAULT_CONFIDENCE
    79→    79→    expires: Optional[float] = None
    80→    80→
    81→    81→    def is_expired(self, now: Optional[float] = None) -> bool:
    82→    82→        """Return ``True`` if this fact has passed its expiration."""
    83→    83→        if self.expires is None:
    84→    84→            return False
    85→    85→        check_time: float = now if now is not None else time.time()
    86→    86→        return check_time > self.expires
    87→    87→
    88→    88→    def to_dict(self) -> Dict[str, Any]:
    89→    89→        """Serialize to a JSON-safe dictionary."""
    90→    90→        d: Dict[str, Any] = {
    91→    91→            "key": self.key,
    92→    92→            "value": self.value,
    93→    93→            "timestamp": self.timestamp,
    94→    94→            "source": self.source,
    95→    95→            "tags": sorted(self.tags),
    96→    96→            "confidence": self.confidence,
    97→    97→            "expires": self.expires,
    98→    98→        }
    99→    99→        return d
   100→   100→
   101→   101→    @classmethod
   102→   102→    def from_dict(cls, data: Dict[str, Any]) -> EngineeringFact:
   103→   103→        """Deserialize from a dictionary (e.g. loaded from JSON)."""
   104→   104→        tags_raw = data.get("tags", [])
   105→   105→        if isinstance(tags_raw, (list, tuple, set)):
   106→   106→            tags = frozenset(str(t) for t in tags_raw)
   107→   107→        else:
   108→   108→            tags = frozenset()
   109→   109→        return cls(
   110→   110→            key=str(data["key"]),
   111→   111→            value=data["value"],
   112→   112→            timestamp=float(data.get("timestamp", time.time())),
   113→   113→            source=str(data.get("source", "")),
   114→   114→            tags=tags,
   115→   115→            confidence=float(data.get("confidence", DEFAULT_CONFIDENCE)),
   116→   116→            expires=data.get("expires"),  # keep None as-is
   117→   117→        )

   120→   120→# ======================================================================
   121→   121→#  MemoryIndex — fast multi-axes lookup
   122→   122→# ======================================================================
   123→   123→
   124→   124→
   125→   125→class MemoryIndex:
   126→   126→    """In-memory inverted index over :class:`EngineeringFact` entries.
   127→   127→
   128→   128→    Maintains three axes:
   129→   129→      * **tag index**  — fact keys by tag
   130→   130→      * **time index**  — fact keys by ISO-date bucket (``YYYY-MM-DD``)
   131→   131→      * **source index** — fact keys by source string
   132→   132→
   133→   133→    The index is rebuilt from the fact store on demand (lazy) or when
   134→   134→    :meth:`rebuild` is called explicitly.
   135→   135→    """
   136→   136→
   137→   137→    def __init__(self) -> None:
   138→   138→        self._tag_index: Dict[str, Set[str]] = defaultdict(set)
   139→   139→        self._time_index: Dict[str, Set[str]] = defaultdict(set)
   140→   140→        self._source_index: Dict[str, Set[str]] = defaultdict(set)
   141→   141→        self._key_set: Set[str] = set()
   142→   142→        self._dirty: bool = True
   143→   143→
   144→   144→    # ── Mutation ─────────────────────────────────────────────────────
   145→   145→
   146→   146→    def add(self, fact: EngineeringFact) -> None:
   147→   147→        """Register a single fact in all index axes."""
   148→   148→        self._key_set.add(fact.key)
   149→   149→        for tag in fact.tags:
   150→   150→            self._tag_index[tag].add(fact.key)
   151→   151→        date_bucket = self._date_bucket(fact.timestamp)
   152→   152→        self._time_index[date_bucket].add(fact.key)
   153→   153→        if fact.source:
   154→   154→            self._source_index[fact.source].add(fact.key)
   155→   155→
   156→   156→    def remove(self, fact: EngineeringFact) -> None:
   157→   157→        """Remove a single fact from all index axes."""
   158→   158→        self._key_set.discard(fact.key)
   159→   159→        for tag in fact.tags:
   160→   160→            bucket = self._tag_index.get(tag)
   161→   161→            if bucket is not None:
   162→   162→                bucket.discard(fact.key)
   163→   163→                if not bucket:
   164→   164→                    del self._tag_index[tag]
   165→   165→        date_bucket = self._date_bucket(fact.timestamp)
   166→   166→        bucket = self._time_index.get(date_bucket)
   167→   167→        if bucket is not None:
   168→   168→            bucket.discard(fact.key)
   169→   169→            if not bucket:
   170→   170→                del self._time_index[date_bucket]
   171→   171→        if fact.source:
   172→   172→            bucket = self._source_index.get(fact.source)
   173→   173→            if bucket is not None:
   174→   174→                bucket.discard(fact.key)
   175→   175→                if not bucket:
   176→   176→                    del self._source_index[fact.source]
   177→   177→
   178→   178→    def clear(self) -> None:
   179→   179→        """Wipe all index data."""
   180→   180→        self._tag_index.clear()
   181→   181→        self._time_index.clear()
   182→   182→        self._source_index.clear()
   183→   183→        self._key_set.clear()
   184→   184→        self._dirty = True
   185→   185→
   186→   186→    # ── Lookup ───────────────────────────────────────────────────────
   187→   187→
   188→   188→    def by_tag(self, tag: str) -> Set[str]:
   189→   189→        """Return fact keys indexed under *tag*."""
   190→   190→        return set(self._tag_index.get(tag, set()))
   191→   191→
   192→   192→    def by_tags(self, tags: Iterable[str]) -> Set[str]:
   193→   193→        """Return fact keys that have ALL of the given *tags* (intersection)."""
   194→   194→        tag_list = list(tags)
   195→   195→        if not tag_list:
   196→   196→            return set()
   197→   197→        result: Optional[Set[str]] = None
   198→   198→        for tag in tag_list:
   199→   199→            keys = self._tag_index.get(tag, set())
   200→   200→            if result is None:
   201→   201→                result = set(keys)
   202→   202→            else:
   203→   203→                result &= keys
   204→   204→            if not result:
   205→   205→                return set()
   206→   206→        return result or set()
   207→   207→
   208→   208→    def by_time_range(self, start: float, end: float) -> Set[str]:
   209→   209→        """Return fact keys whose timestamp falls in [start, end]."""
   210→   210→        start_bucket = self._date_bucket(start)
   211→   211→        end_bucket = self._date_bucket(end)
   212→   212→        result: Set[str] = set()
   213→   213→        for bucket, keys in self._time_index.items():
   214→   214→            if start_bucket <= bucket <= end_bucket:
   215→   215→                result |= keys
   216→   216→        return result
   217→   217→
   218→   218→    def by_source(self, source: str) -> Set[str]:
   219→   219→        """Return fact keys from a specific *source*."""
   220→   220→        return set(self._source_index.get(source, set()))
   221→   221→
   222→   222→    def all_keys(self) -> Set[str]:
   223→   223→        """Return every known fact key."""
   224→   224→        return set(self._key_set)
   225→   225→
   226→   226→    # ── Rebuild ──────────────────────────────────────────────────────
   227→   227→
   228→   228→    def rebuild(self, facts: Dict[str, EngineeringFact]) -> None:
   229→   229→        """Rebuild the entire index from a fact dictionary."""
   230→   230→        self.clear()
   231→   231→        for fact in facts.values():
   232→   232→            self.add(fact)
   233→   233→        self._dirty = False
   234→   234→
   235→   235→    @property
   236→   236→    def dirty(self) -> bool:
   237→   237→        """Whether the index is out of sync with the fact store."""
   238→   238→        return self._dirty
   239→   239→
   240→   240→    # ── Internals ────────────────────────────────────────────────────
   241→   241→
   242→   242→    @staticmethod
   243→   243→    def _date_bucket(timestamp: float) -> str:
   244→   244→        """Convert a unix timestamp to an ISO date string for bucketing."""
   245→   245→        try:
   246→   246→            dt = datetime.fromtimestamp(timestamp, tz=timezone.utc)
   247→   247→            return dt.strftime("%Y-%m-%d")
   248→   248→        except (OSError, ValueError, OverflowError):
   249→   249→            return "1970-01-01"
   250→   250→

   252→   252→# ======================================================================
   253→   253→#  MemoryQuery — composable query builder
   254→   254→# ======================================================================
   255→   255→
   256→   256→
   257→   257→class MemoryQuery:
   258→   258→    """Fluent query builder for :class:`RepositoryMemory` searches.
   259→   259→
   260→   260→    Usage::
   261→   261→
   262→   262→        results = (
   263→   263→            MemoryQuery()
   264→   264→            .with_tags({"auth", "decision"})
   265→   265→            .with_confidence(min_conf=0.7)
   266→   266→            .with_time_range(start_ts, end_ts)
   267→   267→            .build()
   268→   268→        )
   269→   269→        facts = memory.search(query=results)
   270→   270→
   271→   271→    Queries support AND / OR / NOT combinatorics via the class methods.
   272→   272→    """
   273→   273→
   274→   274→    def __init__(self) -> None:
   275→   275→        # Positive filters (all must match — AND semantics)
   276→   276→        self._tags: Set[str] = set()
   277→   277→        self._sources: Set[str] = set()
   278→   278→        self._key_pattern: Optional[str] = None
   279→   279→        self._min_confidence: float = 0.0
   280→   280→        self._max_confidence: float = 1.0
   281→   281→        self._time_start: Optional[float] = None
   282→   282→        self._time_end: Optional[float] = None
   283→   283→        self._text_query: Optional[str] = None
   284→   284→        self._exclude_tags: Set[str] = set()
   285→   285→        self._exclude_sources: Set[str] = set()
   286→   286→        self._exclude_keys: Set[str] = set()
   287→   287→
   288→   288→    # ── Fluent setters (return self for chaining) ───────────────────
   289→   289→
   290→   290→    def with_tags(self, tags: Iterable[str]) -> MemoryQuery:
   291→   291→        """Require ALL of *tags* to be present on matching facts."""
   292→   292→        self._tags.update(tags)
   293→   293→        return self
   294→   294→
   295→   295→    def with_any_tag(self, tags: Iterable[str]) -> MemoryQuery:
   296→   296→        """Require at least ONE of *tags* to be present (OR within AND).
   297→   297→
   298→   298→        Stored internally as a special marker; evaluated at query time.
   299→   299→        """
   300→   300→        # We use a private attribute to signal "any-tag" mode.
   301→   301→        # If _any_tags is set, the regular _tags AND semantics still apply,
   302→   302→        # but the any-tags subset is OR-checked separately.
   303→   303→        if not hasattr(self, "_any_tags"):
   304→   304→            self._any_tags: Set[str] = set()
   305→   305→        self._any_tags.update(tags)
   306→   306→        return self
   307→   307→
   308→   308→    def with_sources(self, sources: Iterable[str]) -> MemoryQuery:
   309→   309→        """Restrict to facts from any of the given *sources*."""
   310→   310→        self._sources.update(sources)
   311→   311→        return self
   312→   312→
   313→   313→    def with_key_pattern(self, pattern: str) -> MemoryQuery:
   314→   314→        """Shell-glob pattern matched against fact keys (fnmatch)."""
   315→   315→        self._key_pattern = pattern
   316→   316→        return self
   317→   317→
   318→   318→    def with_confidence(
   319→   319→        self,
   320→   320→        min_conf: float = 0.0,
   321→   321→        max_conf: float = 1.0,
   322→   322→    ) -> MemoryQuery:
   323→   323→        """Filter by confidence range [min_conf, max_conf]."""
   324→   324→        self._min_confidence = max(0.0, min(1.0, min_conf))
   325→   325→        self._max_confidence = max(0.0, min(1.0, max_conf))
   326→   326→        return self
   327→   327→
   328→   328→    def with_time_range(
   329→   329→        self, start: Optional[float] = None, end: Optional[float] = None
   330→   330→    ) -> MemoryQuery:
   331→   331→        """Restrict to facts recorded between *start* and *end* (unix epoch).
   332→   332→
   333→   333→        Either bound may be ``None`` to leave it open.
   334→   334→        """
   335→   335→        self._time_start = start
   336→   336→        self._time_end = end
   337→   337→        return self
   338→   338→
   339→   339→    def with_text(self, query_text: str) -> MemoryQuery:
   340→   340→        """Full-text search substring (case-insensitive) across values.
   341→   341→
   342→   342→        The text is matched against the JSON-serialised representation of
   343→   343→        each fact's ``value`` field.
   344→   344→        """
   345→   345→        self._text_query = query_text
   346→   346→        return self
   347→   347→
   348→   348→    def without_tags(self, tags: Iterable[str]) -> MemoryQuery:
   349→   349→        """Exclude facts that have ANY of the given *tags* (NOT)."""
   350→   350→        self._exclude_tags.update(tags)
   351→   351→        return self
   352→   352→
   353→   353→    def without_sources(self, sources: Iterable[str]) -> MemoryQuery:
   354→   354→        """Exclude facts from any of the given *sources* (NOT)."""
   355→   355→        self._exclude_sources.update(sources)
   356→   356→        return self
   357→   357→
   358→   358→    def without_keys(self, keys: Iterable[str]) -> MemoryQuery:
   359→   359→        """Exclude facts with any of the given exact *keys*."""
   360→   360→        self._exclude_keys.update(keys)
   361→   361→        return self
   362→   362→
   363→   363→    # ── Combinators ──────────────────────────────────────────────────
   364→   364→
   365→   365→    @classmethod
   366→   366→    def and_(cls, *queries: MemoryQuery) -> MemoryQuery:
   367→   367→        """Create a new query that ANDs the constraints of all *queries*.
   368→   368→
   369→   369→        Tags and sources from all queries are unioned (since each query's
   370→   370→        own filters are already AND within themselves).  The tightest
   371→   371→        confidence/time bounds win.
   372→   372→        """
   373→   373→        combined = cls()
   374→   374→        for q in queries:
   375→   375→            combined._tags |= q._tags
   376→   376→            combined._sources |= q._sources
   377→   377→            combined._exclude_tags |= q._exclude_tags
   378→   378→            combined._exclude_sources |= q._exclude_sources
   379→   379→            combined._exclude_keys |= q._exclude_keys
   380→   380→            combined._min_confidence = max(combined._min_confidence, q._min_confidence)
   381→   381→            combined._max_confidence = min(combined._max_confidence, q._max_confidence)
   382→   382→            if q._time_start is not None:
   383→   383→                if combined._time_start is None or q._time_start > combined._time_start:
   384→   384→                    combined._time_start = q._time_start
   385→   385→            if q._time_end is not None:
   386→   386→                if combined._time_end is None or q._time_end < combined._time_end:
   387→   387→                    combined._time_end = q._time_end
   388→   388→            if q._key_pattern and not combined._key_pattern:
   389→   389→                combined._key_pattern = q._key_pattern
   390→   390→            if q._text_query and not combined._text_query:
   391→   391→                combined._text_query = q._text_query
   392→   392→            # any_tags
   393→   393→            if hasattr(q, "_any_tags") and q._any_tags:
   394→   394→                if not hasattr(combined, "_any_tags"):
   395→   395→                    combined._any_tags = set()
   396→   396→                combined._any_tags |= q._any_tags
   397→   397→        return combined
   398→   398→
   399→   399→    @classmethod
   400→   400→    def or_(cls, *queries: MemoryQuery) -> MemoryQuery:
   401→   401→        """Create a new query whose filters are the LOOSEST of all *queries*.
   402→   402→
   403→   403→        This is useful for building an OR-style search: the caller runs
   404→   404→        each sub-query independently and unions the results.  This helper
   405→   405→        returns a query with relaxed constraints so that a single pass
   406→   406→        captures the superset.
   407→   407→        """
   408→   408→        combined = cls()
   409→   409→        for q in queries:
   410→   410→            # For OR, we keep the loosest bounds
   411→   411→            combined._min_confidence = min(combined._min_confidence, q._min_confidence)
   412→   412→            combined._max_confidence = max(combined._max_confidence, q._max_confidence)
   413→   413→            if q._time_start is not None:
   414→   414→                if combined._time_start is None or q._time_start < combined._time_start:
   415→   415→                    combined._time_start = q._time_start
   416→   416→            if q._time_end is not None:
   417→   417→                if combined._time_end is None or q._time_end > combined._time_end:
   418→   418→                    combined._time_end = q._time_end
   419→   419→        # OR semantics: collect all include-tags as any_tags
   420→   420→        for q in queries:
   421→   421→            if q._tags:
   422→   422→                if not hasattr(combined, "_any_tags"):
   423→   423→                    combined._any_tags = set()
   424→   424→                combined._any_tags |= q._tags
   425→   425→        # OR semantics: collect all exclude-tags as intersection (keep only
   426→   426→        # those excluded by ALL sub-queries)
   427→   427→        if queries:
   428→   428→            first_exclude = queries[0]._exclude_tags
   429→   429→            for q in queries[1:]:
   430→   430→                first_exclude = first_exclude & q._exclude_tags
   431→   431→            combined._exclude_tags = first_exclude
   432→   432→        return combined
   433→   433→
   434→   434→    @classmethod
   435→   435→    def not_(cls, query: MemoryQuery) -> MemoryQuery:
   436→   436→        """Invert a query: its positive tag/source constraints become exclusions.
   437→   437→
   438→   438→        Confidence and time ranges are left open (``0.0–1.0`` and
   439→   439→        ``None–None``) since inverting them would be semantically
   440→   440→        ambiguous.
   441→   441→        """
   442→   442→        inverted = cls()
   443→   443→        inverted._exclude_tags = query._tags.copy()
   444→   444→        inverted._exclude_sources = query._sources.copy()
   445→   445→        if query._key_pattern:
   446→   446→            inverted._key_pattern = None  # can't negate a glob
   447→   447→        return inverted
   448→   448→
   449→   449→    # ── Serialisation ────────────────────────────────────────────────
   450→   450→
   451→   451→    def to_dict(self) -> Dict[str, Any]:
   452→   452→        """Serialise the query to a JSON-safe dictionary."""
   453→   453→        d: Dict[str, Any] = {
   454→   454→            "tags": sorted(self._tags),
   455→   455→            "sources": sorted(self._sources),
   456→   456→            "key_pattern": self._key_pattern,
   457→   457→            "min_confidence": self._min_confidence,
   458→   458→            "max_confidence": self._max_confidence,
   459→   459→            "time_start": self._time_start,
   460→   460→            "time_end": self._time_end,
   461→   461→            "text_query": self._text_query,
   462→   462→            "exclude_tags": sorted(self._exclude_tags),
   463→   463→            "exclude_sources": sorted(self._exclude_sources),
   464→   464→            "exclude_keys": sorted(self._exclude_keys),
   465→   465→        }
   466→   466→        if hasattr(self, "_any_tags"):
   467→   467→            d["any_tags"] = sorted(self._any_tags)
   468→   468→        return d
   469→   469→
   470→   470→    # ── Matching ─────────────────────────────────────────────────────
   471→   471→
   472→   472→    def matches(self, fact: EngineeringFact) -> bool:
   473→   473→        """Return ``True`` if *fact* satisfies all query constraints."""
   474→   474→        # Tag AND filter
   475→   475→        if self._tags and not self._tags.issubset(fact.tags):
   476→   476→            return False
   477→   477→
   478→   478→        # Any-tag OR filter
   479→   479→        if hasattr(self, "_any_tags") and self._any_tags:
   480→   480→            if not self._any_tags & fact.tags:
   481→   481→                return False
   482→   482→
   483→   483→        # Source filter
   484→   484→        if self._sources and fact.source not in self._sources:
   485→   485→            return False
   486→   486→
   487→   487→        # Key pattern
   488→   488→        if self._key_pattern is not None:
   489→   489→            if not fnmatch.fnmatch(fact.key, self._key_pattern):
   490→   490→                return False
   491→   491→
   492→   492→        # Confidence range
   493→   493→        if not (self._min_confidence <= fact.confidence <= self._max_confidence):
   494→   494→            return False
   495→   495→
   496→   496→        # Time range
   497→   497→        if self._time_start is not None and fact.timestamp < self._time_start:
   498→   498→            return False
   499→   499→        if self._time_end is not None and fact.timestamp > self._time_end:
   500→   500→            return False
   501→   501→
   502→   502→        # Text query
   503→   503→        if self._text_query is not None:
   504→   504→            text_blob = json.dumps(fact.value, default=str).lower()
   505→   505→            if self._text_query.lower() not in text_blob:
   506→   506→                return False
   507→   507→
   508→   508→        # Exclusion filters (NOT)
   509→   509→        if self._exclude_tags and self._exclude_tags & fact.tags:
   510→   510→            return False
   511→   511→        if self._exclude_sources and fact.source in self._exclude_sources:
   512→   512→            return False
   513→   513→        if self._exclude_keys and fact.key in self._exclude_keys:
   514→   514→            return False
   515→   515→
   516→   516→        return True
   517→   517→
   518→   518→    def __repr__(self) -> str:
   519→   519→        parts: List[str] = []
   520→   520→        if self._tags:
   521→   521→            parts.append(f"tags={sorted(self._tags)!r}")
   522→   522→        if hasattr(self, "_any_tags") and self._any_tags:
   523→   523→            parts.append(f"any_tags={sorted(self._any_tags)!r}")
   524→   524→        if self._sources:
   525→   525→            parts.append(f"sources={sorted(self._sources)!r}")
   526→   526→        if self._key_pattern:
   527→   527→            parts.append(f"key_pattern={self._key_pattern!r}")
   528→   528→        if self._min_confidence > 0.0 or self._max_confidence < 1.0:
   529→   529→            parts.append(
   530→   530→                f"conf=[{self._min_confidence}, {self._max_confidence}]"
   531→   531→            )
   532→   532→        if self._time_start is not None or self._time_end is not None:
   533→   533→            parts.append(f"time=[{self._time_start}, {self._time_end}]")
   534→   534→        if self._text_query:
   535→   535→            parts.append(f"text={self._text_query!r}")
   536→   536→        if self._exclude_tags:
   537→   537→            parts.append(f"exclude_tags={sorted(self._exclude_tags)!r}")
   538→   538→        return f"MemoryQuery({', '.join(parts)})"
   539→   539→

   542→   542→#  Snapshot — point-in-time capture
   543→   543→# ======================================================================

   545→   545→
   546→   546→@dataclass
   547→   547→class MemorySnapshot:
   548→   548→    """An immutable point-in-time capture of repository memory.
   549→   549→
   550→   550→    Snapshots are plain dictionaries keyed by fact key, with values being
   551→   551→    the serialised :class:`EngineeringFact` dicts.  They can be compared
   552→   552→    with :meth:`RepositoryMemory.diff_snapshots`.
   553→   553→    """
   554→   554→
   555→   555→    id: str
   556→   556→    created_at: float
   557→   557→    facts: Dict[str, Dict[str, Any]]
   558→   558→    fact_count: int
   559→   559→    total_bytes: int
   560→   560→
   561→   561→
   562→   562→def _size_of_dict(d: Dict[str, Any]) -> int:
   563→   563→    """Approximate byte-size of a dict when JSON-serialised."""
   564→   564→    return len(json.dumps(d, default=str, separators=(",", ":")))
   565→   565→
   566→   566→
   567→   567→# ======================================================================
   568→   568→#  RepositoryMemory — the main entry point
   569→   569→# ======================================================================
   570→   570→
   571→   571→
   572→   572→class RepositoryMemory:
   573→   573→    """Persistent repository meta-knowledge store.
   574→   574→
   575→   575→    Stores engineering facts about scanned repositories and codebases:
   576→   576→    Architecture Decision Records, performance baselines, tech-stack
   577→   577→    observations, infrastructure notes, and similar meta-knowledge.
   578→   578→
   579→   579→    This is COMPLEMENTARY to ``memory.UnifiedMemoryStore`` which stores
   580→   580→    security scan data.  Repository memory stores knowledge ABOUT the
   581→   581→    target codebase itself.
   582→   582→
   583→   583→    Thread-safe via a reentrant lock.
   584→   584→    """
   585→   585→
   586→   586→    def __init__(self, persist_path: Optional[Path] = None) -> None:
   587→   587→        self._path: Path = persist_path or REPOSITORY_MEMORY_FILE
   588→   588→        self._lock = threading.RLock()
   589→   589→        self._facts: Dict[str, EngineeringFact] = {}
   590→   590→        self._index = MemoryIndex()
   591→   591→        self._loaded: bool = False
   592→   592→
   593→   593→    # ── Persistence ──────────────────────────────────────────────────
   594→   594→
   595→   595→    def _ensure_dir(self) -> None:
   596→   596→        """Create the parent directory if it doesn't exist."""
   597→   597→        try:
   598→   598→            self._path.parent.mkdir(parents=True, exist_ok=True)
   599→   599→        except OSError as exc:
   600→   600→            logger.error("Cannot create directory %s: %s", self._path.parent, exc)
   601→   601→            raise
   602→   602→
   603→   603→    def load(self) -> None:
   604→   604→        """Load facts from disk.  Idempotent — safe to call multiple times.
   605→   605→
   606→   606→        If the file doesn't exist, starts with an empty store.
   607→   607→        """
   608→   608→        with self._lock:
   609→   609→            if self._loaded:
   610→   610→                return
   611→   611→            self._ensure_dir()
   612→   612→            if not self._path.exists():
   613→   613→                logger.debug("No repository memory file at %s — starting fresh.", self._path)
   614→   614→                self._loaded = True
   615→   615→                return
   616→   616→            try:
   617→   617→                raw = self._path.read_text(encoding="utf-8")
   618→   618→                data = json.loads(raw) if raw.strip() else {}
   619→   619→            except (json.JSONDecodeError, OSError) as exc:
   620→   620→                logger.warning(
   621→   621→                    "Corrupt repository memory file %s: %s — starting fresh.",
   622→   622→                    self._path, exc,
   623→   623→                )
   624→   624→                data = {}
   625→   625→
   626→   626→            count = 0
   627→   627→            for key, fact_dict in data.items():
   628→   628→                if not isinstance(fact_dict, dict) or "key" not in fact_dict:
   629→   629→                    logger.debug("Skipping invalid fact entry: %s", key)
   630→   630→                    continue
   631→   631→                try:
   632→   632→                    fact = EngineeringFact.from_dict(fact_dict)
   633→   633→                    if fact.is_expired():
   634→   634→                        logger.debug("Skipping expired fact: %s", fact.key)
   635→   635→                        continue
   636→   636→                    self._facts[fact.key] = fact
   637→   637→                    count += 1
   638→   638→                except (KeyError, TypeError, ValueError) as exc:
   639→   639→                    logger.debug("Skipping malformed fact %s: %s", key, exc)
   640→   640→
   641→   641→            self._index.rebuild(self._facts)
   642→   642→            self._loaded = True
   643→   643→            logger.info(
   644→   644→                "Loaded %d repository facts from %s", count, self._path
   645→   645→            )
   646→   646→
   647→   647→    def save(self) -> None:
   648→   648→        """Flush all in-memory facts to disk.
   649→   649→
   650→   650→        Expired facts are pruned before writing.
   651→   651→        """
   652→   652→        with self._lock:
   653→   653→            self._ensure_dir()
   654→   654→            self._prune_expired()
   655→   655→            data: Dict[str, Dict[str, Any]] = {}
   656→   656→            for key, fact in self._facts.items():
   657→   657→                data[key] = fact.to_dict()
   658→   658→            try:
   659→   659→                tmp_path = self._path.with_suffix(".tmp")
   660→   660→                tmp_path.write_text(
   661→   661→                    json.dumps(data, indent=2, default=str, ensure_ascii=False),
   662→   662→                    encoding="utf-8",
   663→   663→                )
   664→   664→                # Atomic rename
   665→   665→                tmp_path.replace(self._path)
   666→   666→                logger.debug(
   667→   667→                    "Saved %d repository facts to %s", len(data), self._path
   668→   668→                )
   669→   669→            except OSError as exc:
   670→   670→                logger.error("Failed to save repository memory: %s", exc)
   671→   671→                raise
   672→   672→
   673→   673→    def _ensure_loaded(self) -> None:
   674→   674→        """Lazy-load if not already loaded."""
   675→   675→        if not self._loaded:
   676→   676→            self.load()
   677→   677→
   678→   678→    def _prune_expired(self) -> int:
   679→   679→        """Remove expired facts.  Returns count of pruned entries."""
   680→   680→        now = time.time()
   681→   681→        expired_keys = [
   682→   682→            k for k, f in self._facts.items() if f.is_expired(now)
   683→   683→        ]
   684→   684→        for key in expired_keys:
   685→   685→            self._index.remove(self._facts[key])
   686→   686→            del self._facts[key]
   687→   687→        if expired_keys:
   688→   688→            logger.debug("Pruned %d expired facts.", len(expired_keys))
   689→   689→        return len(expired_keys)
   690→   690→
   691→   691→    # ── Core CRUD ────────────────────────────────────────────────────
   692→   692→
   693→   693→    def remember(
   694→   694→        self,
   695→   695→        key: str,
   696→   696→        value: Any,
   697→   697→        metadata: Optional[Dict[str, Any]] = None,
   698→   698→    ) -> EngineeringFact:
   699→   699→        """Store a fact about the repository.
   700→   700→
   701→   701→        Args:
   702→   702→            key:      Unique identifier (dot/slash path recommended).
   703→   703→            value:    Any JSON-serialisable value.
   704→   704→            metadata: Optional dict with keys:
   705→   705→                      ``source`` (str), ``tags`` (list[str]),
   706→   706→                      ``confidence`` (float 0–1),
   707→   707→                      ``expires`` (float unix timestamp or None),
   708→   708→                      ``timestamp`` (float unix timestamp).
   709→   709→
   710→   710→        Returns:
   711→   711→            The created :class:`EngineeringFact`.
   712→   712→        """
   713→   713→        with self._lock:
   714→   714→            self._ensure_loaded()
   715→   715→            meta = metadata or {}
   716→   716→
   717→   717→            # If the key already exists, remove old index entries
   718→   718→            if key in self._facts:
   719→   719→                self._index.remove(self._facts[key])
   720→   720→
   721→   721→            tags_raw = meta.get("tags", [])
   722→   722→            if isinstance(tags_raw, str):
   723→   723→                tags_raw = [tags_raw]
   724→   724→            tags = frozenset(str(t) for t in tags_raw)
   725→   725→
   726→   726→            fact = EngineeringFact(
   727→   727→                key=key,
   728→   728→                value=value,
   729→   729→                timestamp=float(meta.get("timestamp", time.time())),
   730→   730→                source=str(meta.get("source", "")),
   731→   731→                tags=tags,
   732→   732→                confidence=float(meta.get("confidence", DEFAULT_CONFIDENCE)),
   733→   733→                expires=meta.get("expires"),
   734→   734→            )
   735→   735→
   736→   736→            self._facts[key] = fact
   737→   737→            self._index.add(fact)
   738→   738→            logger.debug("Remembered fact: %s", key)
   739→   739→            return fact
   740→   740→
   741→   741→    def recall(self, key: str) -> Optional[EngineeringFact]:
   742→   742→        """Retrieve a single fact by exact key.
   743→   743→
   744→   744→        Returns ``None`` if the key doesn't exist or the fact is expired.
   745→   745→        """
   746→   746→        with self._lock:
   747→   747→            self._ensure_loaded()
   748→   748→            fact = self._facts.get(key)
   749→   749→            if fact is None:
   750→   750→                return None
   751→   751→            if fact.is_expired():
   752→   752→                self._index.remove(fact)
   753→   753→                del self._facts[key]
   754→   754→                return None
   755→   755→            return fact
   756→   756→
   757→   757→    def recall_pattern(self, pattern_glob: str) -> List[EngineeringFact]:
   758→   758→        """Find facts whose keys match a shell glob *pattern_glob*.
   759→   759→
   760→   760→        Uses ``fnmatch`` for matching.  Returns facts sorted by key.
   761→   761→        Expired facts are excluded.
   762→   762→        """
   763→   763→        with self._lock:
   764→   764→            self._ensure_loaded()
   765→   765→            now = time.time()
   766→   766→            results: List[EngineeringFact] = []
   767→   767→            for key, fact in self._facts.items():
   768→   768→                if fnmatch.fnmatch(key, pattern_glob) and not fact.is_expired(now):
   769→   769→                    results.append(fact)
   770→   770→            results.sort(key=lambda f: f.key)
   771→   771→            return results
   772→   772→
   773→   773→    def forget(self, key: str) -> bool:
   774→   774→        """Remove a fact by exact key.
   775→   775→
   776→   776→        Returns ``True`` if the fact existed and was removed,
   777→   777→        ``False`` if it wasn't found.
   778→   778→        """
   779→   779→        with self._lock:
   780→   780→            self._ensure_loaded()
   781→   781→            fact = self._facts.pop(key, None)
   782→   782→            if fact is None:
   783→   783→                return False
   784→   784→            self._index.remove(fact)
   785→   785→            logger.debug("Forgot fact: %s", key)
   786→   786→            return True
   787→   787→
   788→   788→    # ── Search ───────────────────────────────────────────────────────
   789→   789→
   790→   790→    def search(
   791→   791→        self,
   792→   792→        query_text: Optional[str] = None,
   793→   793→        query: Optional[MemoryQuery] = None,
   794→   794→        limit: int = 100,
   795→   795→    ) -> List[EngineeringFact]:
   796→   796→        """Search facts.
   797→   797→
   798→   798→        Args:
   799→   799→            query_text: Free-text substring searched across fact keys and
   800→   800→                        JSON-serialised values (case-insensitive).  Used when
   801→   801→                        *query* is ``None``.
   802→   802→            query:      A :class:`MemoryQuery` object for structured search.
   803→   803→            limit:      Maximum results to return.
   804→   804→
   805→   805→        Returns:
   806→   806→            Matching facts sorted by confidence descending, then timestamp
   807→   807→            descending (most recent first).
   808→   808→        """
   809→   809→        with self._lock:
   810→   810→            self._ensure_loaded()
   811→   811→            now = time.time()
   812→   812→
   813→   813→            # Build a MemoryQuery from query_text if no query object given
   814→   814→            if query is None and query_text is not None:
   815→   815→                query = MemoryQuery().with_text(query_text)
   816→   816→
   817→   817→            if query is None:
   818→   818→                # No filters — return all non-expired facts
   819→   819→                results = [
   820→   820→                    f for f in self._facts.values() if not f.is_expired(now)
   821→   821→                ]
   822→   822→            else:
   823→   823→                results = [
   824→   824→                    f
   825→   825→                    for f in self._facts.values()
   826→   826→                    if not f.is_expired(now) and query.matches(f)
   827→   827→                ]
   828→   828→
   829→   829→            # Sort: confidence desc, then timestamp desc
   830→   830→            results.sort(key=lambda f: (f.confidence, f.timestamp), reverse=True)
   831→   831→            return results[:limit]

   833→   833→    # ── Snapshots ────────────────────────────────────────────────────
   834→   834→
   835→   835→    def snapshot(self, label: str = "") -> MemorySnapshot:
   836→   836→        """Create a point-in-time snapshot of all current (non-expired) facts.
   837→   837→
   838→   838→        Args:
   839→   839→            label: Optional human-readable label stored in the snapshot id.
   840→   840→
   841→   841→        Returns:
   842→   842→            A :class:`MemorySnapshot` containing copies of all facts.
   843→   843→        """
   844→   844→        with self._lock:
   845→   845→            self._ensure_loaded()
   846→   846→            now = time.time()
   847→   847→            facts_copy: Dict[str, Dict[str, Any]] = {}
   848→   848→            for key, fact in self._facts.items():
   849→   849→                if not fact.is_expired(now):
   850→   850→                    facts_copy[key] = fact.to_dict()
   851→   851→
   852→   852→            snap_id = f"snap_{int(now * 1000)}"
   853→   853→            if label:
   854→   854→                snap_id = f"{snap_id}_{label}"
   855→   855→
   856→   856→            total_bytes = _size_of_dict(facts_copy)
   857→   857→            return MemorySnapshot(
   858→   858→                id=snap_id,
   859→   859→                created_at=now,
   860→   860→                facts=facts_copy,
   861→   861→                fact_count=len(facts_copy),
   862→   862→                total_bytes=total_bytes,
   863→   863→            )
   864→   864→
   865→   865→    @staticmethod
   866→   866→    def diff_snapshots(
   867→   867→        snap_a: MemorySnapshot, snap_b: MemorySnapshot,
   868→   868→    ) -> Dict[str, Any]:
   869→   869→        """Compare two snapshots and return a structured diff.
   870→   870→
   871→   871→        Args:
   872→   872→            snap_a: The earlier snapshot.
   873→   873→            snap_b: The later snapshot.
   874→   874→
   875→   875→        Returns:
   876→   876→            A dict with keys:
   877→   877→              ``added``    — keys present in B but not A
   878→   878→              ``removed``  — keys present in A but not B
   879→   879→              ``modified`` — keys present in both but with different values
   880→   880→              ``unchanged``— keys present in both with identical values
   881→   881→              ``stats``    — summary counts
   882→   882→        """
   883→   883→        keys_a = set(snap_a.facts.keys())
   884→   884→        keys_b = set(snap_b.facts.keys())
   885→   885→
   886→   886→        added = sorted(keys_b - keys_a)
   887→   887→        removed = sorted(keys_a - keys_b)
   888→   888→        common = keys_a & keys_b
   889→   889→
   890→   890→        modified: List[str] = []
   891→   891→        unchanged: List[str] = []
   892→   892→        for key in sorted(common):
   893→   893→            val_a = json.dumps(
   894→   894→                snap_a.facts[key]["value"], sort_keys=True, default=str
   895→   895→            )
   896→   896→            val_b = json.dumps(
   897→   897→                snap_b.facts[key]["value"], sort_keys=True, default=str
   898→   898→            )
   899→   899→            if val_a != val_b:
   900→   900→                modified.append(key)
   901→   901→            else:
   902→   902→                unchanged.append(key)
   903→   903→
   904→   904→        return {
   905→   905→            "snapshot_a": {
   906→   906→                "id": snap_a.id,
   907→   907→                "created_at": snap_a.created_at,
   908→   908→                "fact_count": snap_a.fact_count,
   909→   909→            },
   910→   910→            "snapshot_b": {
   911→   911→                "id": snap_b.id,
   912→   912→                "created_at": snap_b.created_at,
   913→   913→                "fact_count": snap_b.fact_count,
   914→   914→            },
   915→   915→            "added": added,
   916→   916→            "removed": removed,
   917→   917→            "modified": modified,
   918→   918→            "unchanged": unchanged,
   919→   919→            "stats": {
   920→   920→                "added_count": len(added),
   921→   921→                "removed_count": len(removed),
   922→   922→                "modified_count": len(modified),
   923→   923→                "unchanged_count": len(unchanged),
   924→   924→                "net_change": len(added) - len(removed),
   925→   925→            },
   926→   926→        }
   927→   927→
   928→   928→    # ── Statistics ───────────────────────────────────────────────────
   929→   929→
   930→   930→    def get_stats(self) -> Dict[str, Any]:
   931→   931→        """Return summary statistics about the memory store.
   932→   932→
   933→   933→        Includes:
   934→   934→          - ``total_facts`` — count of non-expired facts
   935→   935→          - ``expired_facts`` — count of expired (but not yet pruned) facts
   936→   936→          - ``total_bytes`` — approximate JSON size of all facts
   937→   937→          - ``tag_distribution`` — tag → count mapping
   938→   938→          - ``source_distribution`` — source → count mapping
   939→   939→          - ``age_distribution`` — bucketed age ranges (in seconds)
   940→   940→          - ``confidence_stats`` — min, max, average confidence
   941→   941→          - ``oldest_fact`` — timestamp of the oldest fact
   942→   942→          - ``newest_fact`` — timestamp of the newest fact
   943→   943→        """
   944→   944→        with self._lock:
   945→   945→            self._ensure_loaded()
   946→   946→            now = time.time()
   947→   947→
   948→   948→            active_facts: List[EngineeringFact] = []
   949→   949→            expired_count = 0
   950→   950→            tag_dist: Dict[str, int] = defaultdict(int)
   951→   951→            source_dist: Dict[str, int] = defaultdict(int)
   952→   952→
   953→   953→            for fact in self._facts.values():
   954→   954→                if fact.is_expired(now):

   958→   958→                for tag in fact.tags:
   959→   959→                    tag_dist[tag] += 1
   960→   960→                if fact.source:
   961→   961→                    source_dist[fact.source] += 1
   962→   962→
   963→   963→            # Age distribution buckets (seconds)
   964→   964→            age_buckets: Dict[str, int] = {
   965→   965→                "<1h": 0,
   966→   966→                "1h-24h": 0,
   967→   967→                "1d-7d": 0,

   969→   969→                ">30d": 0,
   970→   970→            }
   971→   971→            for fact in active_facts:
   972→   972→                age = now - fact.timestamp
   973→   973→                if age < 3600:
   974→   974→                    age_buckets["<1h"] += 1
   975→   975→                elif age < 86400:
   976→   976→                    age_buckets["1h-24h"] += 1
   977→   977→                elif age < 604800:
   978→   978→                    age_buckets["1d-7d"] += 1
   979→   979→                elif age < 2592000:
   980→   980→                    age_buckets["7d-30d"] += 1
   981→   981→                else:
   982→   982→                    age_buckets[">30d"] += 1
   983→   983→
   984→   984→            # Confidence stats
   985→   985→            if active_facts:
   986→   986→                confs = [f.confidence for f in active_facts]
   987→   987→                conf_min = min(confs)
   988→   988→                conf_max = max(confs)
   989→   989→                conf_avg = sum(confs) / len(confs)
   990→   990→            else:
   991→   991→                conf_min = 0.0
   992→   992→                conf_max = 0.0
   993→   993→                conf_avg = 0.0
   994→   994→
   995→   995→            # Oldest / newest
   996→   996→            oldest: Optional[float] = None
   997→   997→            newest: Optional[float] = None
   998→   998→            for fact in active_facts:
   999→   999→                if oldest is None or fact.timestamp < oldest:
  1000→  1000→                    oldest = fact.timestamp
  1001→  1001→                if newest is None or fact.timestamp > newest:
  1002→  1002→                    newest = fact.timestamp
  1003→  1003→
  1004→  1004→            # Total bytes
  1005→  1005→            all_data = {k: f.to_dict() for k, f in self._facts.items()}
  1006→  1006→            total_bytes = _size_of_dict(all_data)
  1007→  1007→
  1008→  1008→            return {

  1012→  1012→                "tag_distribution": dict(sorted(tag_dist.items())),
  1013→  1013→                "source_distribution": dict(sorted(source_dist.items())),
  1014→  1014→                "age_distribution": age_buckets,
  1015→  1015→                "confidence_stats": {

  1018→  1018→                    "avg": round(conf_avg, 3),
  1019→  1019→                },
  1020→  1020→                "oldest_fact": oldest,
  1021→  1021→                "newest_fact": newest,
  1022→  1022→            }
