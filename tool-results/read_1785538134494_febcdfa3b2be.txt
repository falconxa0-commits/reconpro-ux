   211→function getScoreColor(score: number): string {
   212→  if (score >= 90) return '#00ff88';
   213→  if (score >= 75) return '#d29922';
   214→  return '#f85149';

   216→
   217→function getScoreGradient(score: number): string {
   218→  if (score >= 90) return 'from-[#00ff88]/20 to-[#00ff88]/5';
   219→  if (score >= 75) return 'from-[#d29922]/20 to-[#d29922]/5';
   220→  return 'from-[#f85149]/20 to-[#f85149]/5';
   221→}

   238→function ComplianceGauge({ score }: { score: number }) {
   239→  const radius = 70;

   241→  const offset = circumference - (score / 100) * circumference;

   247→        {/* Background circle */}
   248→        <circle cx="90" cy="90" r={radius} fill="none" stroke="#161b22" strokeWidth="10" />
   249→        {/* Progress arc */}
   250→        <motion.circle
   251→          cx="90"
   252→          cy="90"
   253→          r={radius}
   254→          fill="none"

   259→          initial={{ strokeDashoffset: circumference }}

   284→function TrendSparkline() {
   285→  const points = [62, 65, 68, 72, 70, 75, 78, 82, 85, 84, 87, 89];
   286→  const w = 200;
   287→  const h = 40;
   288→  const step = w / (points.length - 1);
   289→
   290→  const pathData = points
   291→    .map((p, i) => `${i === 0 ? 'M' : 'L'} ${i * step} ${h - (p / 100) * h}`)
   292→    .join(' ');
   293→
   294→  const areaData = `${pathData} L ${w} ${h} L 0 ${h} Z`;
   295→
   296→  return (
   297→    <svg width={w} height={h} viewBox={`0 0 ${w} ${h}`} className="overflow-visible">
   298→      <defs>
   299→        <linearGradient id="sparkGrad" x1="0" y1="0" x2="0" y2="1">
   300→          <stop offset="0%" stopColor="#00ff88" stopOpacity="0.3" />
   301→          <stop offset="100%" stopColor="#00ff88" stopOpacity="0" />
   302→        </linearGradient>
   303→      </defs>
   304→      <motion.path
   305→        d={areaData}
   306→        fill="url(#sparkGrad)"
   307→        initial={{ opacity: 0 }}
   308→        animate={{ opacity: 1 }}
   309→        transition={{ delay: 0.8, duration: 0.6 }}
   310→      />
   311→      <motion.path
   312→        d={pathData}
   313→        fill="none"
   314→        stroke="#00ff88"
   315→        strokeWidth="2"
   316→        strokeLinecap="round"
   317→        strokeLinejoin="round"
   318→        initial={{ pathLength: 0 }}
   319→        animate={{ pathLength: 1 }}
   320→        transition={{ delay: 0.5, duration: 1.2, ease: 'easeOut' }}
   321→      />
   322→      <motion.circle
   323→        cx={(points.length - 1) * step}
   324→        cy={h - (points[points.length - 1] / 100) * h}
   325→        r="3"
   326→        fill="#00ff88"
   327→        initial={{ opacity: 0, scale: 0 }}
   328→        animate={{ opacity: 1, scale: 1 }}
   329→        transition={{ delay: 1.5, duration: 0.3 }}
   330→      />
   331→    </svg>
   332→  );
   333→}
