   918→        ("GET", "/robots.txt", {}, None),
   919→        ("GET", "/sitemap.xml", {}, None),
   920→        ("GET", "/.well-known/security.txt", {}, None),
   921→        ("GET", "/nonexistent-random-abc123", {}, None),
   922→        ("GET", "/nonexistent-random-xyz789", {}, None),
   923→        ("POST", "/", {}, b"{}"),
   924→        ("POST", "/nonexistent", {}, b"{"key": "value"}"),
   925→        ("PUT", "/nonexistent", {}, b"test"),
   926→        ("DELETE", "/nonexistent", {}, None),
   927→        ("GET", "/api/v1/endpoint", {}, None),
   928→        ("GET", "/admin", {}, None),
   929→        ("GET", "/login", {}, None),
   930→        ("HEAD", "/", {}, None),
   931→    ]
   932→
   933→    results: List[Dict[str, Any]] = []
   934→    for method, path, extra_headers, body in unusual_requests:
   935→        url = _url_join(base_url, path)
   936→        resp = _safe_probe(
   937→            url, method=method, body=body,
   938→            headers=extra_headers if extra_headers else None,
   939→            timeout=timeout, verify_tls=verify_tls,
   940→        )
   941→        results.append({
   942→            "method": method,
   943→            "path": path,
   944→            "status": resp.get("status", 0),
   945→            "body_fp": _body_fingerprint(resp.get("body", "")),
   946→            "header_fp": _header_fingerprint(resp.get("headers", {})),
   947→            "body_len": len(resp.get("body", "")),
   948→        })
   949→
   950→    statuses = [r["status"] for r in results]
   951→    body_fps = [r["body_fp"] for r in results]
   952→    header_fps = [r["header_fp"] for r in results]
   953→    body_lens = [r["body_len"] for r in results]
   954→
   955→    unique_statuses = len(set(statuses))
   956→    unique_bodies = len(set(body_fps))
   957→    unique_headers = len(set(header_fps))
   958→
   959→    # Real servers have diverse responses
   960→    inconsistency_score = 0.0
   961→    evidence_parts: List[str] = []
   962→
   963→    if unique_statuses <= 2 and len(statuses) > 5:
   964→        inconsistency_score += 0.4
   965→        evidence_parts.append(
   966→            f"Only {unique_statuses} unique status codes across {len(statuses)} requests"
   967→        )
   968→
   969→    if unique_bodies <= 3 and len(body_fps) > 5:
   970→        inconsistency_score += 0.3
   971→        evidence_parts.append(
   972→            f"Only {unique_bodies} unique body fingerprints across {len(body_fps)} requests"
   973→        )
   974→
   975→    if unique_headers == 1 and len(header_fps) > 3:
   976→        inconsistency_score += 0.3
   977→        evidence_parts.append(
   978→            f"All {len(header_fps)} responses have identical header structure"
   979→        )
   980→
   981→    # Check body length variance
   982→    if len(body_lens) >= 3:
   983→        body_cv = _coefficient_of_variation([float(l) for l in body_lens if l > 0])
   984→        if body_cv is not None and body_cv < 0.05:
   985→            inconsistency_score += 0.3
   986→            evidence_parts.append(
   987→                f"Body length CV={body_cv:.4f} (near-zero variance in response sizes)"
   988→            )
   989→
   990→    is_honeypot_like = inconsistency_score >= 0.6
   991→
   992→    if is_honeypot_like:
   993→        findings.append(Finding(
   994→            title="Highly Consistent Behavior Detected (Honeypot-Like)",
   995→            severity="medium",
   996→            category="Honeypot Detection",
   997→            module="honeypot_dance",
   998→            description=(
   999→                f"Across {len(results)} varied requests (different methods, paths, content types), "
  1000→                f"the server responded with suspiciously uniform behavior. Real servers "
  1001→                f"exhibit natural inconsistency: different status codes for different errors, "
  1002→                f"varying body content, and evolving headers. Honeypots typically use a "
  1003→                f"single response template. Consistency score: {inconsistency_score:.2f}"
  1004→            ),
  1005→            evidence="; ".join(evidence_parts),
  1006→            asset=base_url,
  1007→            points_deducted=15,
  1008→            remediation="If this is a legitimate server, ensure different error paths return "
  1009→                        "appropriately varied responses (e.g., 404 for missing pages, 405 for "
  1010→                        "wrong methods, 415 for wrong content types).",
  1011→            dread_score=5.0,
  1012→        ))
  1013→    else:
  1014→        findings.append(Finding(
  1015→            title="Behavioral Inconsistency is Natural",
  1016→            severity="info",
  1017→            category="Honeypot Detection",
  1018→            module="honeypot_dance",
  1019→            description=(
  1020→                f"The server shows {unique_statuses} unique status codes, {unique_bodies} unique "
  1021→                f"body fingerprints, and {unique_headers} unique header structures across "
  1022→                f"{len(results)} requests. This natural variance is inconsistent with honeypot behavior."
  1023→            ),
  1024→            evidence=f"Statuses: {set(statuses)}; Consistency score: {inconsistency_score:.2f}",
  1025→            asset=base_url,
  1026→            points_deducted=0,
  1027→            remediation="",
  1028→            dread_score=0.5,
  1029→        ))
  1030→
  1031→    return {
  1032→        "findings": findings,
  1033→        "inconsistency_score": inconsistency_score,
  1034→        "unique_statuses": unique_statuses,
  1035→        "unique_bodies": unique_bodies,
  1036→        "unique_headers": unique_headers,
  1037→        "is_honeypot_like": is_honeypot_like,
  1038→    }

  1049→    """Detect if the server claims technologies that don't match observed behavior.
  1050→

  1055→    anomalies: List[str] = []

  1060→    body = main_resp.get("body", "")

  1066→    powered_by = headers.get("x-powered-by", "").lower()

  1068→    if "nginx" in server_hdr:
  1069→        claimed_techs.append("nginx")
  1070→    if "apache" in server_hdr:
  1071→        claimed_techs.append("apache")
  1072→    if "iis" in server_hdr:
  1073→        claimed_techs.append("iis")
  1074→    if "asp.net" in powered_by or "asp" in powered_by:
  1075→        claimed_techs.append("ASP.NET")
  1076→    if "php" in powered_by or "php" in server_hdr:
  1077→        claimed_techs.append("PHP")
  1078→    if "express" in powered_by:
  1079→        claimed_techs.append("Express")
  1080→    if "python" in powered_by or "werkzeug" in powered_by or "gunicorn" in server_hdr or "uvicorn" in server_hdr:
  1081→        claimed_techs.append("Python")
  1082→    if "java" in powered_by or "tomcat" in server_hdr or "jetty" in server_hdr:
  1083→        claimed_techs.append("Java")
  1084→    if "next" in powered_by:
  1085→        claimed_techs.append("Next.js")
  1086→    if "openresty" in server_hdr:
  1087→        claimed_techs.append("OpenResty")
  1088→

  1090→    observed_techs: List[str] = []
  1091→    if re.search(r'(?i)<meta\s+name=["\x27]generator["\x27]\s+content=["\x27]WordPress', body):
  1092→        observed_techs.append("wordpress")
  1093→    if re.search(r"(?i)wp-content|wp-includes", body):
  1094→        observed_techs.append("wordpress")

  1096→        observed_techs.append("next.js")
  1097→    if re.search(r"(?i)django|csrfmiddlewaretoken", body):
  1098→        observed_techs.append("django")
  1099→    if re.search(r"(?i)laravel|laravel_session", body):
  1100→        observed_techs.append("laravel")
  1101→    if re.search(r"(?i)rails|csrf-token|turbolinks", body):
  1102→        observed_techs.append("rails")
  1103→    if re.search(r"(?i)\.php", body):
  1104→        observed_techs.append("php")
  1105→    if re.search(r'(?i)\.asp[x]?', body):
  1106→        observed_techs.append("asp.net")
  1107→    if re.search(r'(?i)\.jsp|servlet', body):
  1108→        observed_techs.append("java")
  1109→    if re.search(r"(?i)node\.js|express|socket\.io", body):
  1110→        observed_techs.append("node")
  1111→
  1112→    # Check against TECH_STACK_INCONSISTENCIES

  1116→            continue

  1118→            for obs in observed_techs:
  1119→                if re.search(incompatible, obs, re.IGNORECASE):
  1120→                    anomalies.append(
  1121→                        f"Claims {claimed} (header) but body shows {obs} — {rule['description']}"
  1122→                    )
  1123→
  1124→    # Check for contradictory headers
  1125→    if "nginx" in server_hdr and "iis" in server_hdr:
  1126→        anomalies.append("Server header claims both nginx and IIS — impossible combination")
  1127→    if "apache" in server_hdr and "nginx" in server_hdr:
  1128→        anomalies.append("Server header claims both Apache and nginx — impossible combination")
  1129→
  1130→    # Check for PHP version in headers but no PHP indicators in body at all
  1131→    # (on a page that should have dynamic content)
  1132→    if "php" in powered_by and len(body) > 500:
  1133→        if not re.search(r"(?i)php|\$|->|::", body):
  1134→            anomalies.append(
  1135→                "Claims PHP via X-Powered-By but body shows no PHP artifacts "
  1136→                "(no PHP-generated content, variables, or patterns)"
  1137→            )
  1138→
  1139→    if anomalies:
  1140→        severity = "high" if len(anomalies) >= 3 else (
  1141→            "medium" if len(anomalies) >= 2 else "low"
  1142→        )
  1143→        findings.append(Finding(
  1144→            title="Technology Stack Anomalies Detected",
  1145→            severity=severity,
  1146→            category="Honeypot Detection",
  1147→            module="honeypot_dance",
  1148→            description=(
  1149→                f"The server claims to run {', '.join(claimed_techs) if claimed_techs else 'no identifiable technology'} "
  1150→                f"but exhibits behaviors inconsistent with that stack. Honeypots frequently "
  1151→                f"mismatch their claimed and actual technology. Found {len(anomalies)} anomalies."
  1152→            ),
  1153→            evidence="; ".join(anomalies[:8]),
  1154→            asset=base_url,
  1155→            points_deducted=min(len(anomalies) * 10, 30),
  1156→            remediation="Ensure Server and X-Powered-By headers accurately reflect the "
  1157→                        "actual technology stack. Inconsistent headers can flag the server as a honeypot.",
  1158→            dread_score=min(len(anomalies) * 2.5, 8.0),
  1159→        ))
  1160→    else:
  1161→        findings.append(Finding(
  1162→            title="Technology Stack is Consistent",
  1163→            severity="info",
  1164→            category="Honeypot Detection",
  1165→            module="honeypot_dance",
  1166→            description=(
  1167→                f"Claimed technologies ({', '.join(claimed_techs) if claimed_techs else 'none detected'}) "
  1168→                f"are consistent with observed body content and behavior."
  1169→            ),
  1170→            evidence=f"Claimed: {claimed_techs}; Observed: {observed_techs}",
  1171→            asset=base_url,
  1172→            points_deducted=0,
  1173→            remediation="",
  1174→            dread_score=0.5,
  1175→        ))
  1176→
  1177→    return {
  1178→        "findings": findings,
  1179→        "claimed_techs": claimed_techs,
  1180→        "observed_techs": observed_techs,
  1181→        "anomalies": anomalies,
  1182→    }

  1190→def _check_default_credentials(
  1191→    base_url: str, timeout: int, verify_tls: bool
  1192→) -> Dict[str, Any]:
  1193→    """Test common honeypot default credentials against login endpoints.
  1194→
  1195→    Many honeypots accept well-known default credentials (admin/admin, root/root, etc.)
  1196→    as a way to lure attackers and capture their activity.
  1197→    """

  1199→    accepted_creds: List[Dict[str, Any]] = []
  1200→

  1203→        "/login",
  1204→        "/admin/login",
  1205→        "/wp-login.php",
  1206→        "/auth/login",
  1207→        "/api/auth/login",
  1208→        "/signin",
  1209→        "/api/login",
  1210→        "/user/login",
  1211→        "/administrator/login",
  1212→    ]
  1213→
  1214→    # Find working login paths
  1215→    working_paths: List[str] = []

  1217→        url = _url_join(base_url, path)
  1218→        resp = _safe_probe(url, timeout=timeout, verify_tls=verify_tls)
  1219→        status = resp.get("status", 0)
  1220→        body = resp.get("body", "")
  1221→        if status == 200 and len(body) > 50:
  1222→            working_paths.append(path)
  1223→

  1225→    for path in working_paths:
  1226→        url = _url_join(base_url, path)
  1227→        for cred in HONEYPOT_DEFAULT_CREDENTIALS:
  1228→            # Try JSON body
  1229→            json_body = '{{"username": "{}", "password": "{}"}}'.format(
  1230→                cred["username"], cred["password"]
  1231→            ).encode("utf-8")
  1232→            resp = _safe_probe(
  1233→                url, method="POST", body=json_body,
  1234→                headers={"Content-Type": "application/json"},
  1235→                timeout=timeout, verify_tls=verify_tls,
  1236→            )
  1237→
  1238→            status = resp.get("status", 0)
  1239→            body = resp.get("body", "")
  1240→
  1241→            # Check if login succeeded (status 200 with a session/token/redirect indicator)
  1242→            is_accepted = False
  1243→            accept_reason = ""
  1244→
  1245→            if status == 200:
  1246→                # Check for success indicators in response
  1247→                if re.search(r"(?i)(token|session|welcome|dashboard|success|redirect)", body):
  1248→                    is_accepted = True
  1249→                    accept_reason = "200 with success indicators (token/session/welcome)"
  1250→                # Check for set-cookie with session
  1251→                resp_headers = resp.get("headers", {})
  1252→                cookie = resp_headers.get("set-cookie", "")
  1253→                if re.search(r"(?i)(session|token|auth|sid|jsessionid)", cookie):
  1254→                    is_accepted = True
  1255→                    accept_reason = "200 with auth session cookie"
  1256→            elif status == 302 or status == 301:
  1257→                location = resp.get("headers", {}).get("location", "")
  1258→                if re.search(r"(?i)(dashboard|admin|home|index|welcome)", location):
  1259→                    is_accepted = True
  1260→                    accept_reason = f"{status} redirect to {location}"
  1261→
  1262→            if is_accepted:
  1263→                accepted_creds.append({
  1264→                    "path": path,
  1265→                    "username": cred["username"],
  1266→                    "password": cred["password"],
  1267→                    "source": cred["source"],
  1268→                    "reason": accept_reason,
  1269→                })
  1270→
  1271→            # Also try form-encoded body
  1272→            form_body = "username={}&password={}".format(
  1273→                urllib.parse.quote(cred["username"], safe=""),
  1274→                urllib.parse.quote(cred["password"], safe=""),
  1275→            ).encode("utf-8")
  1276→            resp2 = _safe_probe(
  1277→                url, method="POST", body=form_body,
  1278→                headers={"Content-Type": "application/x-www-form-urlencoded"},
  1279→                timeout=timeout, verify_tls=verify_tls,
  1280→            )
  1281→
  1282→            status2 = resp2.get("status", 0)
  1283→            body2 = resp2.get("body", "")
  1284→            resp_headers2 = resp2.get("headers", {})
  1285→
  1286→            is_accepted2 = False
  1287→            accept_reason2 = ""
  1288→
  1289→            if status2 == 200:
  1290→                if re.search(r"(?i)(token|session|welcome|dashboard|success|redirect)", body2):
  1291→                    is_accepted2 = True
  1292→                    accept_reason2 = "200 with success indicators (form-encoded)"
  1293→                cookie2 = resp_headers2.get("set-cookie", "")
  1294→                if re.search(r"(?i)(session|token|auth|sid|jsessionid)", cookie2):
  1295→                    is_accepted2 = True
  1296→                    accept_reason2 = "200 with auth session cookie (form-encoded)"
  1297→            elif status2 in (301, 302):
  1298→                loc2 = resp_headers2.get("location", "")
  1299→                if re.search(r"(?i)(dashboard|admin|home|index|welcome)", loc2):
  1300→                    is_accepted2 = True
  1301→                    accept_reason2 = f"{status2} redirect to {loc2} (form-encoded)"
  1302→
  1303→            if is_accepted2:
  1304→                # Avoid duplicate if already found via JSON
  1305→                already_found = any(
  1306→                    c["path"] == path and c["username"] == cred["username"]
  1307→                    for c in accepted_creds
  1308→                )
  1309→                if not already_found:
  1310→                    accepted_creds.append({
  1311→                        "path": path,
  1312→                        "username": cred["username"],
  1313→                        "password": cred["password"],
  1314→                        "source": cred["source"],
  1315→                        "reason": accept_reason2,
  1316→                    })
  1317→
  1318→    if accepted_creds:
  1319→        cred_evidence = "; ".join(
  1320→            f"{c['path']} -> {c['username']}:{c['password']} ({c['source']}) [{c['reason']}]"

  1322→        )
  1323→        findings.append(Finding(
  1324→            title=f"Honeypot Default Credentials Accepted ({len(accepted_creds)} found)",
  1325→            severity="high",
  1326→            category="Honeypot Detection",
  1327→            module="honeypot_dance",
  1328→            description=(
  1329→                f"{len(accepted_creds)} common honeypot default credentials were accepted by the target. "
  1330→                f"Honeypots frequently use well-known default credentials to entice attackers. "
  1331→                f"This is a strong indicator of honeypot behavior, though it could also "
  1332→                f"indicate a misconfigured legitimate service."
  1333→            ),
  1334→            evidence=cred_evidence,
  1335→            asset=base_url,
  1336→            points_deducted=min(len(accepted_creds) * 8, 30),
  1337→            remediation="Change all default credentials immediately. If this is a honeypot, "
  1338→                        "consider using unique credentials that don't match known honeypot databases.",
  1339→            dread_score=7.5,
  1340→        ))
  1341→    else:
  1342→        findings.append(Finding(
  1343→            title="No Honeypot Default Credentials Accepted",
  1344→            severity="info",
  1345→            category="Honeypot Detection",
  1346→            module="honeypot_dance",
  1347→            description=(
  1348→                f"Tested {len(HONEYPOT_DEFAULT_CREDENTIALS)} default credential pairs against "
  1349→                f"{len(working_paths)} login endpoints. None were accepted."
  1350→            ),
  1351→            evidence=f"Probed {len(working_paths)} login paths, "
  1352→                    f"tested {len(HONEYPOT_DEFAULT_CREDENTIALS)} credential pairs",
  1353→            asset=base_url,
  1354→            points_deducted=0,
  1355→            remediation="",
  1356→            dread_score=0.5,
  1357→        ))
  1358→
  1359→    return {
  1360→        "findings": findings,
  1361→        "accepted_creds": accepted_creds,

  1363→        "working_paths": working_paths,

  1368→# Check 7: Interaction Pattern Analysis
  1369→# ═══════════════════════════════════════════════════════════════════════════
  1370→
  1371→
  1372→def _analyze_interaction_patterns(
  1373→    base_url: str, timeout: int, verify_tls: bool
  1374→) -> Dict[str, Any]:
  1375→    """Send known honeypot trigger commands and analyze response patterns.
  1376→
  1377→    Honeypots are designed to log specific attack patterns. When triggered, they
  1378→    often respond differently than real servers — either by logging everything
  1379→    perfectly (and thus having predictable responses) or by failing in telltale ways.
  1380→    """
  1381→    findings: List[Finding] = []
  1382→    trigger_results: List[Dict[str, Any]] = []
  1383→    honeypot_indicators: List[str] = []
  1384→
  1385→    # Get a baseline normal response
  1386→    baseline_resp = _safe_probe(base_url, timeout=timeout, verify_tls=verify_tls)
  1387→    baseline_status = baseline_resp.get("status", 0)
  1388→    baseline_body_len = len(baseline_resp.get("body", ""))
  1389→    baseline_headers = set(
  1390→        k.lower() for k in baseline_resp.get("headers", {}).keys()
  1391→    )
  1392→
  1393→    for payload in TRIGGER_PAYLOADS:
  1394→        url = _url_join(base_url, payload["path"])
  1395→        extra_headers = payload.get("headers")
  1396→        body = payload.get("body")
  1397→        method = payload.get("method", "GET")
  1398→
  1399→        resp = _safe_probe(
  1400→            url, method=method, body=body,
  1401→            headers=extra_headers,
  1402→            timeout=timeout, verify_tls=verify_tls,
  1403→        )
  1404→
  1405→        status = resp.get("status", 0)
  1406→        resp_body = resp.get("body", "")
  1407→        resp_body_len = len(resp_body)
  1408→        resp_headers = set(
  1409→            k.lower() for k in resp.get("headers", {}).keys()
  1410→        )
  1411→
  1412→        is_honeypot_like = False
  1413→        reason = ""
  1414→
  1415→        # Indicator 1: Response is identical to baseline (for obviously different requests)
  1416→        if (
  1417→            status == baseline_status
  1418→            and resp_body_len == baseline_body_len
  1419→            and _body_fingerprint(resp_body) == _body_fingerprint(baseline_resp.get("body", ""))
  1420→            and method != "GET"
  1421→        ):
  1422→            is_honeypot_like = True
  1423→            reason = f"{method} to {payload['path']} returns identical response to GET /"
  1424→
  1425→        # Indicator 2: Perfectly formatted error for malformed request
  1426→        if status in (400, 403, 404, 405, 500):
  1427→            if re.search(
  1428→                r"<(?i)(!doctype|html)", resp_body
  1429→            ) and re.search(
  1430→                r"<(?i)/\s*(html|body)>\s*$", resp_body, re.MULTILINE
  1431→            ):
  1432→                is_honeypot_like = True
  1433→                reason = f"Perfectly structured HTML error page for {method} {payload['path']}"
  1434→
  1435→        # Indicator 3: All trigger requests return the same error
  1436→        if trigger_results:
  1437→            prev = trigger_results[-1]
  1438→            if (
  1439→                status == prev["status"]
  1440→                and _body_fingerprint(resp_body) == prev["body_fp"]
  1441→                and payload["name"] != prev["name"]
  1442→            ):
  1443→                is_honeypot_like = True
  1444→                reason = (
  1445→                    f"Same response as previous trigger '{prev['name']}' "
  1446→                    f"(status={status}, body_fp={_body_fingerprint(resp_body)})"
  1447→                )
  1448→
  1449→        # Indicator 4: Headers are suspiciously consistent with baseline
  1450→        if resp_headers == baseline_headers and method in ("POST", "PUT", "DELETE", "TRACE", "OPTIONS"):
  1451→            is_honeypot_like = True
  1452→            reason = f"Headers identical to baseline for {method} request"
  1453→
  1454→        # Indicator 5: Response time is suspiciously consistent
  1455→        elapsed = resp.get("elapsed", 0)
  1456→        baseline_elapsed = baseline_resp.get("elapsed", 0)
  1457→        if elapsed > 0 and baseline_elapsed > 0:
  1458→            time_ratio = elapsed / baseline_elapsed
  1459→            if 0.95 <= time_ratio <= 1.05 and method != "GET":
  1460→                is_honeypot_like = True
  1461→                reason = f"Response time ratio={time_ratio:.3f} (nearly identical to baseline)"
  1462→
  1463→        trigger_results.append({
  1464→            "name": payload["name"],
  1465→            "method": method,
  1466→            "path": payload["path"],
  1467→            "status": status,
  1468→            "body_fp": _body_fingerprint(resp_body),
  1469→            "body_len": resp_body_len,
  1470→            "elapsed": elapsed,
  1471→            "is_honeypot_like": is_honeypot_like,
  1472→            "reason": reason,
  1473→        })
  1474→
  1475→        if is_honeypot_like:
  1476→            honeypot_indicators.append(f"[{payload['name']}] {reason}")
  1477→
  1478→    honeypot_like_count = sum(1 for r in trigger_results if r["is_honeypot_like"])
  1479→    honeypot_ratio = honeypot_like_count / len(trigger_results) if trigger_results else 0
  1480→
  1481→    if honeypot_ratio >= 0.5:
  1482→        severity = "high"
  1483→        points = 25
  1484→        dread = 7.0
  1485→    elif honeypot_ratio >= 0.3:
  1486→        severity = "medium"
  1487→        points = 15
  1488→        dread = 5.0
  1489→    elif honeypot_ratio >= 0.15:
  1490→        severity = "low"
  1491→        points = 8
  1492→        dread = 3.0
  1493→    else:
  1494→        severity = "info"
  1495→        points = 0
  1496→        dread = 0.5
  1497→
  1498→    if honeypot_indicators:
  1499→        findings.append(Finding(
  1500→            title=f"Honeypot Interaction Patterns Detected ({honeypot_like_count}/{len(trigger_results)})",
  1501→            severity=severity,
  1502→            category="Honeypot Detection",
  1503→            module="honeypot_dance",
  1504→            description=(
  1505→                f"{honeypot_like_count} out of {len(trigger_results)} trigger payloads produced responses "
  1506→                f"consistent with honeypot behavior (ratio: {honeypot_ratio:.0%}). Honeypots often "
  1507→                f"respond to attack payloads with template-driven, perfectly consistent responses "
  1508→                f"rather than the varied, sometimes messy responses of real servers."
  1509→            ),
  1510→            evidence="; ".join(honeypot_indicators[:10]),
  1511→            asset=base_url,
  1512→            points_deducted=points,
  1513→            remediation="Ensure the server handles malicious/unusual requests with natural "
  1514→                        "variation in responses. Avoid using a single error template for all "
  1515→                        "request types.",
  1516→            dread_score=dread,
  1517→        ))
  1518→    else:
  1519→        findings.append(Finding(
  1520→            title="Interaction Patterns Appear Natural",
  1521→            severity="info",
  1522→            category="Honeypot Detection",
  1523→            module="honeypot_dance",
  1524→            description=(
  1525→                f"None of the {len(trigger_results)} trigger payloads produced responses "
  1526→                f"consistent with honeypot behavior. The server handles unusual requests "
  1527→                f"with natural variation."
  1528→            ),
  1529→            evidence=f"Tested {len(trigger_results)} trigger payloads, 0 honeypot-like responses",
  1530→            asset=base_url,
  1531→            points_deducted=0,
  1532→            remediation="",
  1533→            dread_score=0.5,
  1534→        ))
  1535→
  1536→    return {
  1537→        "findings": findings,
  1538→        "trigger_results": trigger_results,
  1539→        "honeypot_like_count": honeypot_like_count,
  1540→        "honeypot_ratio": honeypot_ratio,
  1541→    }

  1545→# Check 8: Honeypot Effectiveness Scoring
  1546→# ═══════════════════════════════════════════════════════════════════════════

  1548→
  1549→def _score_honeypot_effectiveness(
  1550→    base_url: str,
  1551→    timing_result: Dict[str, Any],
  1552→    perfection_result: Dict[str, Any],
  1553→    fingerprint_result: Dict[str, Any],
  1554→    consistency_result: Dict[str, Any],
  1555→    tech_result: Dict[str, Any],
  1556→    credential_result: Dict[str, Any],
  1557→    interaction_result: Dict[str, Any],
  1558→) -> List[Finding]:
  1559→    """If the target appears to be a honeypot, score how convincing it is.
  1560→
  1561→    A high effectiveness score means the honeypot is very convincing and would
  1562→    fool most attackers. A low score means it has obvious tells.
  1563→    """
  1564→    findings: List[Finding] = []
  1565→
  1566→    # Aggregate signals
  1567→    signals: List[Dict[str, Any]] = []
  1568→
  1569→    # Timing: lower CV = more honeypot-like (less effective disguise)
  1570→    timing_cv = timing_result.get("cv")
  1571→    if timing_cv is not None:
  1572→        signals.append({
  1573→            "name": "Timing Naturalness",
  1574→            "score": min(timing_cv / 0.15, 1.0),  # Higher = more natural = more effective
  1575→            "weight": 0.15,
  1576→            "detail": f"CV={timing_cv:.4f}",
  1577→        })
  1578→
  1579→    # Error perfection: higher suspicion = less effective disguise
  1580→    perfection_suspicion = perfection_result.get("suspicion_score", 0)
  1581→    signals.append({
  1582→        "name": "Error Naturalness",
  1583→        "score": max(1.0 - perfection_suspicion / 3.0, 0.0),
  1584→        "weight": 0.15,
  1585→        "detail": f"Suspicion={perfection_suspicion:.2f}",
  1586→    })
  1587→
  1588→    # Fingerprint stealth: fewer detections = more effective
  1589→    detected_count = len(fingerprint_result.get("detected", []))
  1590→    signals.append({
  1591→        "name": "Fingerprint Stealth",
  1592→        "score": max(1.0 - detected_count * 0.3, 0.0),
  1593→        "weight": 0.25,
  1594→        "detail": f"Detected={detected_count} signatures",
  1595→    })
  1596→
  1597→    # Behavioral consistency: more inconsistency = more effective (more realistic)
  1598→    consistency = consistency_result.get("inconsistency_score", 0)
  1599→    signals.append({
  1600→        "name": "Behavioral Realism",
  1601→        "score": min(consistency / 0.8, 1.0),
  1602→        "weight": 0.15,
  1603→        "detail": f"Inconsistency={consistency:.2f}",
  1604→    })
  1605→
  1606→    # Tech stack: fewer anomalies = more effective
  1607→    anomaly_count = len(tech_result.get("anomalies", []))
  1608→    signals.append({
  1609→        "name": "Tech Stack Coherence",
  1610→        "score": max(1.0 - anomaly_count * 0.25, 0.0),
  1611→        "weight": 0.15,
  1612→        "detail": f"Anomalies={anomaly_count}",
  1613→    })
  1614→
  1615→    # Credentials: none accepted = more effective
  1616→    accepted_count = len(credential_result.get("accepted_creds", []))
  1617→    signals.append({
  1618→        "name": "Credential Realism",
  1619→        "score": max(1.0 - accepted_count * 0.2, 0.0),
  1620→        "weight": 0.10,
  1621→        "detail": f"Accepted={accepted_count} default creds",
  1622→    })
  1623→
  1624→    # Interaction patterns: lower ratio = more effective
  1625→    hp_ratio = interaction_result.get("honeypot_ratio", 0)
  1626→    signals.append({
  1627→        "name": "Interaction Realism",
  1628→        "score": max(1.0 - hp_ratio, 0.0),
  1629→        "weight": 0.05,
  1630→        "detail": f"Honeypot ratio={hp_ratio:.0%}",
  1631→    })
  1632→
  1633→    # Weighted effectiveness score
  1634→    total_effectiveness = 0.0
  1635→    weighted_details: List[str] = []
  1636→    for sig in signals:
  1637→        contribution = sig["score"] * sig["weight"]
  1638→        total_effectiveness += contribution
  1639→        weighted_details.append(
  1640→            f"{sig['name']}: {sig['score']:.0%} (w={sig['weight']:.2f}, {sig['detail']})"
  1641→        )
  1642→
  1643→    # Determine if this IS a honeypot
  1644→    is_honeypot = False
  1645→    honeypot_evidence: List[str] = []
  1646→
  1647→    if detected_count > 0:
  1648→        is_honeypot = True
  1649→        names = ", ".join(d["name"] for d in fingerprint_result["detected"])
  1650→        honeypot_evidence.append(f"Direct fingerprint match: {names}")
  1651→
  1652→    if timing_result.get("suspicious"):
  1653→        is_honeypot = True
  1654→        honeypot_evidence.append("Suspiciously precise timing")
  1655→
  1656→    if consistency_result.get("is_honeypot_like"):
  1657→        is_honeypot = True
  1658→        honeypot_evidence.append("Overly consistent behavioral responses")
  1659→
  1660→    if accepted_count > 2:
  1661→        is_honeypot = True
  1662→        honeypot_evidence.append(f"{accepted_count} default credentials accepted")
  1663→
  1664→    if hp_ratio >= 0.5:
  1665→        is_honeypot = True
  1666→        honeypot_evidence.append(f"{hp_ratio:.0%} of trigger payloads produced honeypot-like responses")
  1667→
  1668→    if is_honeypot:
  1669→        # Rate effectiveness
  1670→        if total_effectiveness >= 0.8:
  1671→            effectiveness_label = "HIGHLY CONVINCING"
  1672→            effectiveness_desc = (
  1673→                "This honeypot is very well-disguised and would likely fool "
  1674→                "most automated scanners and casual attackers."
  1675→            )
  1676→        elif total_effectiveness >= 0.6:
  1677→            effectiveness_label = "MODERATELY CONVINCING"
  1678→            effectiveness_desc = (
  1679→                "This honeypot has some realistic elements but exhibits "
  1680→                "noticeable tells under closer inspection."
  1681→            )
  1682→        elif total_effectiveness >= 0.4:
  1683→            effectiveness_label = "SOMEWHAT CONVINCING"
  1684→            effectiveness_desc = (
  1685→                "This honeypot has obvious flaws that would be detected by "
  1686→                "moderately thorough analysis."
  1687→            )
  1688→        else:
  1689→            effectiveness_label = "POORLY DISGUISED"
  1690→            effectiveness_desc = (
  1691→                "This honeypot has multiple obvious tells and would be easily "
  1692→                "identified by any scanner."
  1693→            )
  1694→
  1695→        severity = "high" if total_effectiveness >= 0.7 else (
  1696→            "medium" if total_effectiveness >= 0.4 else "low"
  1697→        )
  1698→
  1699→        findings.append(Finding(
  1700→            title=f"Honeypot Identified — Effectiveness: {total_effectiveness:.0%} ({effectiveness_label})",
  1701→            severity=severity,
  1702→            category="Honeypot Detection",
  1703→            module="honeypot_dance",
  1704→            description=(
  1705→                f"The target is assessed as a honeypot with {total_effectiveness:.0%} effectiveness. "
  1706→                f"{effectiveness_desc} "
  1707→                f"Evidence: {'; '.join(honeypot_evidence)}"
  1708→            ),
  1709→            evidence="Breakdown: " + " | ".join(weighted_details),
  1710→            asset=base_url,
  1711→            points_deducted=int(total_effectiveness * 40),
  1712→            remediation=(
  1713→                f"Honeypot effectiveness improvement: "
  1714→                f"1) Add timing jitter to responses (target CV > 0.10). "
  1715→                f"2) Remove identifying headers and honeypot-specific paths. "
  1716→                f"3) Vary error page content and structure. "
  1717→                f"4) Ensure technology stack claims match actual behavior. "
  1718→                f"5) Reject default credentials. "
  1719→                f"6) Introduce natural behavioral inconsistency."
  1720→            ),
  1721→            dread_score=total_effectiveness * 9.0,
  1722→        ))
  1723→    else:
  1724→        findings.append(Finding(
  1725→            title=f"Target Unlikely to be a Honeypot (Effectiveness Baseline: {total_effectiveness:.0%})",
  1726→            severity="info",
  1727→            category="Honeypot Detection",
  1728→            module="honeypot_dance",
  1729→            description=(
  1730→                f"No strong honeypot indicators were detected. The target shows {total_effectiveness:.0%} "
  1731→                f"baseline 'realism' which means it exhibits natural server behavior across all "
  1732→                f"analyzed dimensions. This does not guarantee it is not a honeypot, but no "
  1733→                f"conclusive indicators were found."
  1734→            ),
  1735→            evidence="Breakdown: " + " | ".join(weighted_details),
  1736→            asset=base_url,
  1737→            points_deducted=0,
  1738→            remediation="",
  1739→            dread_score=0.5,
  1740→        ))
  1741→
  1742→    return findings
  1743→

  1748→
  1749→
  1750→def run_honeypot_dance(
  1751→    target: str,
  1752→    base_url: str,
  1753→    timeout: int = 8,
  1754→    verify_tls: bool = True,
  1755→) -> List[Finding]:
  1756→    """Run the full honeypot detection and effectiveness analysis.
  1757→
  1758→    Args:
  1759→        target: The target identifier (hostname/IP) for evidence tagging.
  1760→        base_url: The base URL to probe (e.g., "https://example.com").
  1761→        timeout: Per-request timeout in seconds.
  1762→        verify_tls: Whether to verify TLS certificates.
  1763→
  1764→    Returns:
  1765→        A list of Finding objects detailing honeypot detection results.
  1766→    """
  1767→    all_findings: List[Finding] = []
  1768→
  1769→    # ── Check 1: Response Timing Analysis ──
  1770→    timing_result = _analyze_response_timing(base_url, timeout, verify_tls)
  1771→    all_findings.extend(timing_result["findings"])
  1772→
  1773→    # ── Check 2: Error Message Perfection ──
  1774→    perfection_result = _check_error_perfection(base_url, timeout, verify_tls)
  1775→    all_findings.extend(perfection_result["findings"])
  1776→
  1777→    # ── Check 3: Known Honeypot Fingerprinting ──
  1778→    fingerprint_result = _check_honeypot_fingerprints(base_url, timeout, verify_tls)
  1779→    all_findings.extend(fingerprint_result["findings"])
  1780→
  1781→    # ── Check 4: Behavioral Consistency ──
  1782→    consistency_result = _check_behavioral_consistency(base_url, timeout, verify_tls)
  1783→    all_findings.extend(consistency_result["findings"])
  1784→
  1785→    # ── Check 5: Technology Stack Anomaly Detection ──
  1786→    tech_result = _check_tech_stack_anomalies(base_url, timeout, verify_tls)
  1787→    all_findings.extend(tech_result["findings"])
  1788→
  1789→    # ── Check 6: Default Credential Detection ──
  1790→    credential_result = _check_default_credentials(base_url, timeout, verify_tls)
  1791→    all_findings.extend(credential_result["findings"])
  1792→
  1793→    # ── Check 7: Interaction Pattern Analysis ──
  1794→    interaction_result = _analyze_interaction_patterns(base_url, timeout, verify_tls)
  1795→    all_findings.extend(interaction_result["findings"])
  1796→
  1797→    # ── Check 8: Honeypot Effectiveness Scoring ──
  1798→    effectiveness_findings = _score_honeypot_effectiveness(
  1799→        base_url=base_url,
  1800→        timing_result=timing_result,
  1801→        perfection_result=perfection_result,
  1802→        fingerprint_result=fingerprint_result,
  1803→        consistency_result=consistency_result,
  1804→        tech_result=tech_result,
  1805→        credential_result=credential_result,
  1806→        interaction_result=interaction_result,
  1807→    )
  1808→    all_findings.extend(effectiveness_findings)
  1809→
  1810→    # Tag all findings with the target
  1811→    for f in all_findings:
  1812→        if not f.asset or f.asset == base_url:
  1813→            f.asset = target
  1814→
  1815→    return all_findings
