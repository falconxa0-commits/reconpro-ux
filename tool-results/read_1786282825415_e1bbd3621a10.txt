   101→   101→class SignatureBroadcaster:
   102→   102→    """Injects ReconPro signature into HTTP probes via X-R3c0nPr0 header.
   103→   103→
   104→   104→    Every outbound probe should pass through the broadcaster to ensure
   105→   105→    consistent identification and version tracking across all modules.
   106→   106→
   107→   107→    Attributes:
   108→   108→        version: The ReconPro version string embedded in the header.
   109→   109→        build:   The build/component identifier (e.g. 'wishes-engine').
   110→   110→        probes_sent: Running counter of how many probes were signed.
   111→   111→    """
   112→   112→
   113→   113→    def __init__(self, version: str = RECONPRO_VERSION, build: str = RECONPRO_BUILD) -> None:
   114→   114→        self.version = version
   115→   115→        self.build = build
   116→   116→        self._probes_sent: int = 0
   117→   117→        self._session_id = hashlib.sha256(
   118→   118→            f"{version}:{build}:{time.time()}".encode()
   119→   119→        ).hexdigest()[:12]
   120→   120→
   121→   121→    @property
   122→   122→    def probes_sent(self) -> int:
   123→   123→        """Number of probes that have been signed in this session."""
   124→   124→        return self._probes_sent
   125→   125→
   126→   126→    @property
   127→   127→    def session_id(self) -> str:
   128→   128→        """Unique session identifier for this broadcaster instance."""
   129→   129→        return self._session_id
   130→   130→
   131→   131→    def broadcast(self, headers: Optional[Dict[str, str]] = None) -> Dict[str, str]:
   132→   132→        """Add the ReconPro signature header to a headers dict.
   133→   133→
   134→   134→        Args:
   135→   135→            headers: Existing headers dict to augment, or None for a new dict.
   136→   136→
   137→   137→        Returns:
   138→   138→            The headers dict with the X-R3c0nPr0 header injected.
   139→   139→        """
   140→   140→        if headers is None:
   141→   141→            headers = {}
   142→   142→        headers[RECONPRO_SIGNATURE_HEADER] = (
   143→   143→            f"ReconPro/{self.version} ({self.build}); sid={self._session_id}"
   144→   144→        )
   145→   145→        self._probes_sent += 1
   146→   146→        return headers
   147→   147→
   148→   148→    def get_ua_with_signature(self) -> str:
   149→   149→        """Return a User-Agent string that includes the signature.
   150→   150→
   151→   151→        This UA can be used as a fallback when custom headers are not
   152→   152→        supported by the transport layer.
   153→   153→        """
   154→   154→        return (
   155→   155→            f"ReconPro/{self.version} (Wishes Framework v{self.version}; "
   156→   156→            f"build={self.build}; "
   157→   157→            f"+https://github.com/reconpro-security/reconpro)"
   158→   158→        )
   159→   159→
   160→   160→    def inject_into_request(self, method: str, url: str,
   161→   161→                            headers: Optional[Dict[str, str]] = None,
   162→   162→                            body: Optional[bytes] = None) -> Dict[str, Any]:
   163→   163→        """Build a complete request descriptor with signature injected.
   164→   164→
   165→   165→        Args:
   166→   166→            method:  HTTP method (GET, POST, etc.).
   167→   167→            url:     Target URL.
   168→   168→            headers: Optional existing headers.
   169→   169→            body:    Optional request body.
   170→   170→
   171→   171→        Returns:
   172→   172→            Dict with 'method', 'url', 'headers', 'body', 'signed_at'.
   173→   173→        """
   174→   174→        signed_headers = self.broadcast(headers)
   175→   175→        return {
   176→   176→            "method": method,
   177→   177→            "url": url,
   178→   178→            "headers": signed_headers,
   179→   179→            "body": body,
   180→   180→            "signed_at": _now_iso(),
   181→   181→            "session_id": self._session_id,
   182→   182→        }
   183→   183→
   184→   184→    def reset_counter(self) -> None:
   185→   185→        """Reset the probe counter (useful between scan targets)."""
   186→   186→        self._probes_sent = 0

   196→   196→# ══════════════════════════════════════════════════════════════════════
   197→   197→
   198→   198→class WitnessWriter:
   199→   199→    """Creates immutable audit witness records with cryptographic SHA-256 signature.
   200→   200→
   201→   201→    Each witness is a JSON file recording that a scan encounter occurred.
   202→   202→    The signature is computed over the canonical JSON of core fields,
   203→   203→    making the record tamper-evident. Witnesses are stored in
   204→   204→    ~/.reconpro/witness/ as individual JSON files.
   205→   205→
   206→   206→    Attributes:
   207→   207→        witness_dir: Directory where witness files are persisted.
   208→   208→    """
   209→   209→
   210→   210→    def __init__(self, witness_dir: Optional[Path] = None) -> None:
   211→   211→        self._dir = witness_dir or WITNESS_DIR
   212→   212→        self._dir.mkdir(parents=True, exist_ok=True)
   213→   213→
   214→   214→    @property
   215→   215→    def witness_dir(self) -> Path:
   216→   216→        return self._dir
   217→   217→
   218→   218→    def _build_core(self, encounter_id: str, target: str, verdict: str,
   219→   219→                    timestamp: str, score: float) -> Dict[str, Any]:
   220→   220→        """Construct the canonical core fields that are signed."""
   221→   221→        return {
   222→   222→            "encounter_id": encounter_id,
   223→   223→            "target": target,
   224→   224→            "verdict": verdict,
   225→   225→            "timestamp": timestamp,
   226→   226→            "score": score,
   227→   227→        }
   228→   228→
   229→   229→    def _sign(self, core: Dict[str, Any]) -> str:
   230→   230→        """Compute SHA-256 hex digest of the canonical core fields."""
   231→   231→        return hashlib.sha256(
   232→   232→            json.dumps(core, sort_keys=True).encode()
   233→   233→        ).hexdigest()
   234→   234→
   235→   235→    def write_witness(
   236→   236→        self,
   237→   237→        target: str,
   238→   238→        verdict: str,
   239→   239→        findings_summary: Dict[str, Any],
   240→   240→        score: float,
   241→   241→        modules_run: List[str],
   242→   242→        extra: Optional[Dict[str, Any]] = None,
   243→   243→    ) -> Dict[str, Any]:
   244→   244→        """Write an immutable witness record to disk.
   245→   245→
   246→   246→        Args:
   247→   247→            target:          The scan target (domain, IP, URL).
   248→   248→            verdict:         Unified verdict string (MUNDANE/NOTABLE/SUBSTANTIAL/CRITICAL).
   249→   249→            findings_summary: Dict with 'total' and optionally 'by_severity'.
   250→   250→            score:           Numeric unified score (0-100).
   251→   251→            modules_run:     List of module names that were executed.
   252→   252→            extra:           Optional additional fields to embed.
   253→   253→
   254→   254→        Returns:
   255→   255→            The complete witness dict as written to disk.
   256→   256→        """
   257→   257→        encounter_id = hashlib.sha256(
   258→   258→            f"{target}:{time.time()}".encode()
   259→   259→        ).hexdigest()[:16]
   260→   260→        timestamp = _now_iso()
   261→   261→
   262→   262→        core = self._build_core(encounter_id, target, verdict, timestamp, score)
   263→   263→        signature = self._sign(core)
   264→   264→
   265→   265→        witness: Dict[str, Any] = {
   266→   266→            "encounter_id": encounter_id,
   267→   267→            "timestamp": timestamp,
   268→   268→            "target": target,
   269→   269→            "verdict": verdict,
   270→   270→            "signature": signature,
   271→   271→            "findings_count": findings_summary.get("total", 0),
   272→   272→            "score": score,
   273→   273→            "modules_run": modules_run,
   274→   274→            "severity_breakdown": findings_summary.get("by_severity", {}),
   275→   275→            "manifest": "WITNESS MANIFEST \u2014 the encounter occurred and is recorded",
   276→   276→        }
   277→   277→        if extra:
   278→   278→            witness.update(extra)
   279→   279→
   280→   280→        path = self._dir / f"witness_{_safe_target(target)}_{encounter_id}.json"
   281→   281→        path.write_text(json.dumps(witness, indent=2), encoding="utf-8")
   282→   282→        logger.debug("Witness written: %s", path.name)
   283→   283→        return witness
   284→   284→
   285→   285→    def verify_witness(self, witness: Dict[str, Any]) -> bool:
   286→   286→        """Verify the cryptographic signature of a witness record.
   287→   287→
   288→   288→        Args:
   289→   289→            witness: A previously-written witness dict.
   290→   290→
   291→   291→        Returns:
   292→   292→            True if the signature is valid and untampered.
   293→   293→        """
   294→   294→        required = ("encounter_id", "target", "verdict", "timestamp", "score")
   295→   295→        if not all(k in witness for k in required):
   296→   296→            return False
   297→   297→        core = self._build_core(
   298→   298→            witness["encounter_id"],
   299→   299→            witness["target"],
   300→   300→            witness["verdict"],
   301→   301→            witness["timestamp"],
   302→   302→            witness["score"],
   303→   303→        )
   304→   304→        expected = self._sign(core)
   305→   305→        return witness.get("signature") == expected
   306→   306→
   307→   307→    def verify_all(self) -> Tuple[int, int]:
   308→   308→        """Verify every witness on disk.
   309→   309→
   310→   310→        Returns:
   311→   311→            Tuple of (valid_count, invalid_count).
   312→   312→        """
   313→   313→        valid, invalid = 0, 0
   314→   314→        for w in self.list_witnesses():
   315→   315→            if self.verify_witness(w):
   316→   316→                valid += 1
   317→   317→            else:
   318→   318→                invalid += 1
   319→   319→        return valid, invalid
   320→   320→
   321→   321→    def list_witnesses(self, target: Optional[str] = None) -> List[Dict[str, Any]]:
   322→   322→        """List all witness records, optionally filtered by target.
   323→   323→
   324→   324→        Args:
   325→   325→            target: If provided, only return witnesses for this target.
   326→   326→
   327→   327→        Returns:
   328→   328→            List of witness dicts, sorted by filename (chronological).
   329→   329→        """
   330→   330→        witnesses: List[Dict[str, Any]] = []
   331→   331→        if not self._dir.exists():
   332→   332→            return witnesses
   333→   333→        for f in sorted(self._dir.glob("witness_*.json")):
   334→   334→            try:
   335→   335→                w = json.loads(f.read_text(encoding="utf-8"))
   336→   336→                if target is None or w.get("target") == target:
   337→   337→                    witnesses.append(w)
   338→   338→            except Exception:
   339→   339→                pass
   340→   340→        return witnesses
   341→   341→
   342→   342→    def get_witness(self, encounter_id: str) -> Optional[Dict[str, Any]]:
   343→   343→        """Retrieve a single witness by its encounter_id.
   344→   344→
   345→   345→        Args:
   346→   346→            encounter_id: The 16-char hex encounter identifier.
   347→   347→
   348→   348→        Returns:
   349→   349→            The witness dict, or None if not found.
   350→   350→        """
   351→   351→        for w in self.list_witnesses():
   352→   352→            if w.get("encounter_id") == encounter_id:
   353→   353→                return w
   354→   354→        return None
   355→   355→
   356→   356→    def purge_witnesses(self, before_timestamp: Optional[str] = None,
   357→   357→                        target: Optional[str] = None) -> int:
   358→   358→        """Delete witness files, optionally filtered.
   359→   359→
   360→   360→        Args:
   361→   361→            before_timestamp: ISO timestamp; delete witnesses older than this.
   362→   362→            target:           Only delete witnesses for this target.
   363→   363→
   364→   364→        Returns:
   365→   365→            Number of witnesses deleted.
   366→   366→        """
   367→   367→        deleted = 0
   368→   368→        if not self._dir.exists():
   369→   369→            return deleted
   370→   370→        for f in self._dir.glob("witness_*.json"):
   371→   371→            try:
   372→   372→                w = json.loads(f.read_text(encoding="utf-8"))
   373→   373→                if target and w.get("target") != target:
   374→   374→                    continue
   375→   375→                if before_timestamp and w.get("timestamp", "") >= before_timestamp:
   376→   376→                    continue
   377→   377→                f.unlink()
   378→   378→                deleted += 1
   379→   379→            except Exception:
   380→   380→                pass
   381→   381→        return deleted
   382→   382→
   383→   383→    def get_stats(self) -> Dict[str, Any]:
   384→   384→        """Return aggregate statistics over all witnesses."""
   385→   385→        witnesses = self.list_witnesses()
   386→   386→        verdict_counts: Dict[str, int] = {}
   387→   387→        for w in witnesses:
   388→   388→            v = w.get("verdict", "UNKNOWN")
   389→   389→            verdict_counts[v] = verdict_counts.get(v, 0) + 1
   390→   390→        return {
   391→   391→            "total_witnesses": len(witnesses),
   392→   392→            "verdict_distribution": verdict_counts,
   393→   393→            "witness_dir": str(self._dir),
   394→   394→        }

   398→   398→# Hall of the Broken (GORGON encounters)
   399→   399→# ══════════════════════════════════════════════════════════════════════
   400→   400→
   401→   401→class HallOfTheBroken:
   402→   402→    """Persistent GORGON encounter registry — tracks Fear Index per target.
   403→   403→
   404→   404→    The Hall persists across scans in ~/.reconpro/hall/broken.json.
   405→   405→    It maintains running statistics: total encounters, average fear,
   406→   406→    most feared target, and highest single fear index.
   407→   407→
   408→   408→    Attributes:
   409→   409→        hall_path: Path to the broken.json persistence file.
   410→   410→    """
   411→   411→
   412→   412→    def __init__(self, hall_dir: Optional[Path] = None) -> None:
   413→   413→        self._dir = hall_dir or HALL_DIR
   414→   414→        self._dir.mkdir(parents=True, exist_ok=True)
   415→   415→        self._path = self._dir / "broken.json"
   416→   416→        self._encounters = self._load()
   417→   417→
   418→   418→    @property
   419→   419→    def hall_path(self) -> Path:
   420→   420→        return self._path
   421→   421→
   422→   422→    def _load(self) -> Dict[str, Any]:
   423→   423→        """Load the Hall from disk, or return empty structure."""
   424→   424→        if self._path.exists():
   425→   425→            try:
   426→   426→                return json.loads(self._path.read_text(encoding="utf-8"))
   427→   427→            except Exception:
   428→   428→                logger.warning("Failed to load Hall of the Broken, starting fresh")
   429→   429→        return {
   430→   430→            "encounters": [],
   431→   431→            "total_fear": 0.0,
   432→   432→            "average_fear": 0.0,
   433→   433→            "most_feared": None,
   434→   434→            "highest_fear": 0.0,
   435→   435→            "total_encounters": 0,
   436→   436→        }
   437→   437→
   438→   438→    def _save(self) -> None:
   439→   439→        """Persist the Hall to disk."""
   440→   440→        try:
   441→   441→            self._path.write_text(
   442→   442→                json.dumps(self._encounters, indent=2), encoding="utf-8"
   443→   443→            )
   444→   444→        except Exception as exc:
   445→   445→            logger.error("Failed to save Hall of the Broken: %s", exc)
   446→   446→
   447→   447→    def _recalculate(self) -> None:
   448→   448→        """Recalculate aggregate statistics from the encounter list."""
   449→   449→        encounters = self._encounters["encounters"]
   450→   450→        count = len(encounters)
   451→   451→        if count == 0:
   452→   452→            self._encounters["total_fear"] = 0.0
   453→   453→            self._encounters["average_fear"] = 0.0
   454→   454→            self._encounters["most_feared"] = None
   455→   455→            self._encounters["highest_fear"] = 0.0
   456→   456→            self._encounters["total_encounters"] = 0
   457→   457→            return
   458→   458→        total = sum(e.get("fear_index", 0) for e in encounters)
   459→   459→        self._encounters["total_fear"] = round(total, 1)
   460→   460→        self._encounters["average_fear"] = round(total / count, 1)
   461→   461→        self._encounters["total_encounters"] = count
   462→   462→        # Find most feared (highest fear_index)
   463→   463→        top = max(encounters, key=lambda e: e.get("fear_index", 0))
   464→   464→        self._encounters["most_feared"] = top.get("target")
   465→   465→        self._encounters["highest_fear"] = top.get("fear_index", 0)
   466→   466→
   467→   467→    def record_encounter(
   468→   468→        self,
   469→   469→        target: str,
   470→   470→        fear_index: float,
   471→   471→        endpoints_found: int = 0,
   472→   472→        vulnerable_count: int = 0,
   473→   473→        cves_matched: int = 0,
   474→   474→        bypasses_achieved: int = 0,
   475→   475→        secrets_extracted: int = 0,
   476→   476→    ) -> Dict[str, Any]:
   477→   477→        """Record a GORGON encounter in the Hall.
   478→   478→
   479→   479→        Args:
   480→   480→            target:            The scan target identifier.
   481→   481→            fear_index:        Computed Fear Index (0-100).
   482→   482→            endpoints_found:   Number of endpoints discovered.

   763→   763→# ══════════════════════════════════════════════════════════════════════
   764→   764→
   765→   765→class FearIndex:
   766→   766→    """Computes Fear Index (1-100) from GORGON scan engagement metrics.
   767→   767→
   768→   768→    The Fear Index is distinct from DREAD scoring — it measures the
   769→   769→    aggregate threat surface discovered during the GORGON engagement.
   770→   770→    Each input dimension is weighted to reflect real-world risk impact.
   771→   771→
   772→   772→    Weight rationale:
   773→   773→        - bypasses_achieved (15.0): WAF/auth bypasses are the most
   774→   774→          alarming because they bypass defense-in-depth.
   775→   775→        - vulnerable_count (12.0): Direct exploitable vulnerabilities.
   776→   776→        - secrets_extracted (8.0): Credential/secret exposure enables
   777→   777→          lateral movement and deeper compromise.
   778→   778→        - cves_matched (4.0): Known CVE matches indicate unpatched systems.
   779→   779→        - endpoints_found (1.5): Surface area discovery; lower individual
   780→   780→          risk but contributes to cumulative exposure.
   781→   781→
   782→   782→    The raw weighted sum is clamped to [0, 100] and mapped to a fear
   783→   783→    level (1-10) with a human-readable label.
   784→   784→    """
   785→   785→
   786→   786→    WEIGHTS: Dict[str, float] = {
   787→   787→        "endpoints": 1.5,
   788→   788→        "vulnerable": 12.0,
   789→   789→        "cves": 4.0,
   790→   790→        "bypasses": 15.0,
   791→   791→        "secrets": 8.0,
   792→   792→    }
   793→   793→
   794→   794→    def compute(
   795→   795→        self,
   796→   796→        endpoints_found: int = 0,
   797→   797→        vulnerable_count: int = 0,
   798→   798→        cves_matched: int = 0,
   799→   799→        bypasses_achieved: int = 0,
   800→   800→        secrets_extracted: int = 0,
   801→   801→    ) -> Dict[str, Any]:
   802→   802→        """Compute the Fear Index from GORGON engagement metrics.
   803→   803→
   804→   804→        Args:
   805→   805→            endpoints_found:   Number of unique endpoints discovered.
   806→   806→            vulnerable_count:  Number of endpoints with vulnerabilities.
   807→   807→            cves_matched:      Number of CVE IDs matched.
   808→   808→            bypasses_achieved: Number of WAF/auth bypass successes.
   809→   809→            secrets_extracted: Number of secrets/credentials extracted.
   810→   810→
   811→   811→        Returns:
   812→   812→            Dict with 'fear_index' (0-100), 'fear_level' (1-10),
   813→   813→            'fear_label', and 'components' breakdown.
   814→   814→        """
   815→   815→        weighted = {
   816→   816→            "endpoints": endpoints_found * self.WEIGHTS["endpoints"],
   817→   817→            "vulnerable": vulnerable_count * self.WEIGHTS["vulnerable"],
   818→   818→            "cves": cves_matched * self.WEIGHTS["cves"],
   819→   819→            "bypasses": bypasses_achieved * self.WEIGHTS["bypasses"],
   820→   820→            "secrets": secrets_extracted * self.WEIGHTS["secrets"],
   821→   821→        }
   822→   822→        raw = min(100.0, sum(weighted.values()))
   823→   823→        level = max(1, min(10, int(raw // 10) + (1 if raw > 0 else 0)))
   824→   824→
   825→   825→        return {
   826→   826→            "fear_index": round(raw, 1),
   827→   827→            "fear_level": level,
   828→   828→            "fear_label": FEAR_LEVELS[level],
   829→   829→            "components": {
   830→   830→                "endpoints_found": endpoints_found,
   831→   831→                "vulnerable_count": vulnerable_count,
   832→   832→                "cves_matched": cves_matched,
   833→   833→                "bypasses_achieved": bypasses_achieved,
   834→   834→                "secrets_extracted": secrets_extracted,
   835→   835→            },
   836→   836→            "weighted_breakdown": {k: round(v, 1) for k, v in weighted.items()},
   837→   837→        }
   838→   838→
   839→   839→    def from_findings(self, findings: List[Dict[str, Any]]) -> Dict[str, Any]:
   840→   840→        """Compute Fear Index from a list of Finding dicts.
   841→   841→
   842→   842→        Inspects each finding's 'category' and 'severity' fields to
   843→   843→        derive the five input metrics automatically.
   844→   844→
   845→   845→        Args:
   846→   846→            findings: List of finding dicts from scan modules.
   847→   847→
   848→   848→        Returns:
   849→   849→            Same structure as compute().
   850→   850→        """
   851→   851→        endpoints = sum(
   852→   852→            1 for f in findings
   853→   853→            if f.get("category") in ("surface_map", "api_discovery", "websocket")
   854→   854→        )
   855→   855→        vulnerable = sum(
   856→   856→            1 for f in findings
   857→   857→            if f.get("severity") in ("critical", "high")
   858→   858→        )
   859→   859→        cves = sum(
   860→   860→            1 for f in findings if "cve" in f.get("title", "").lower()
   861→   861→        )
   862→   862→        bypasses = sum(
   863→   863→            1 for f in findings
   864→   864→            if any(kw in f.get("title", "").lower()
   865→   865→                   for kw in ("bypass", "injection", "xss", "sqli"))
   866→   866→        )
   867→   867→        secrets = sum(
   868→   868→            1 for f in findings
   869→   869→            if any(kw in f.get("title", "").lower()
   870→   870→                   for kw in ("secret", "key", "credential", "token", "exposure"))
   871→   871→        )
   872→   872→        return self.compute(
   873→   873→            endpoints_found=endpoints,
   874→   874→            vulnerable_count=vulnerable,
   875→   875→            cves_matched=cves,
   876→   876→            bypasses_achieved=bypasses,
   877→   877→            secrets_extracted=secrets,
   878→   878→        )
   879→   879→

   882→   882→# Unified Verdict
   883→   883→# ══════════════════════════════════════════════════════════════════════
   884→   884→
   885→   885→class UnifiedVerdict:
   886→   886→    """Aggregates multi-module results into a unified verdict.
   887→   887→
   888→   888→    Verdict levels (ascending severity):
   889→   889→        MUNDANE     — Minimal or no significant findings.
   890→   890→        NOTABLE     — Some findings warranting attention.
   891→   891→        SUBSTANTIAL — Significant findings requiring remediation.
   892→   892→        CRITICAL    — Severe findings requiring immediate action.
   893→   893→
   894→   894→    The computation uses module-specific weights (higher weight for
   895→   895→    offensive modules like GORGON and OBLIVION) to produce a weighted
   896→   896→    average, then applies thresholds to select the verdict.
   897→   897→    """
   898→   898→
   899→   899→    def compute(self, module_scores: Dict[str, Dict[str, Any]]) -> Dict[str, Any]:
   900→   900→        """Compute unified verdict from per-module score dicts.
   901→   901→
   902→   902→        Args:
   903→   903→            module_scores: Dict mapping module name to a dict with at
   904→   904→                           least a 'score' key (0-100).
   905→   905→
   906→   906→        Returns:
   907→   907→            Dict with 'verdict', 'unified_score', 'modules_assessed',
   908→   908→            'critical_modules', 'max_score', 'min_score'.
   909→   909→        """
   910→   910→        if not module_scores:
   911→   911→            return {"verdict": "MUNDANE", "unified_score": 0.0,
   912→   912→                    "modules_assessed": 0, "critical_modules": 0,
   913→   913→                    "max_score": 0.0, "min_score": 0.0}
   914→   914→
   915→   915→        weighted_sum = 0.0
   916→   916→        total_weight = 0.0
   917→   917→        scores: List[float] = []
   918→   918→        critical_count = 0
   919→   919→
   920→   920→        for mod_name, mod_data in module_scores.items():
   921→   921→            score = mod_data.get("score", 0)
   922→   922→            if score is None:
   923→   923→                continue
   924→   924→            weight = VERDICT_WEIGHTS.get(mod_name, 1.0)
   925→   925→            weighted_sum += score * weight
   926→   926→            total_weight += weight
   927→   927→            scores.append(float(score))
   928→   928→            if score >= 80:
   929→   929→                critical_count += 1
   930→   930→
   931→   931→        if not scores:
   932→   932→            return {"verdict": "MUNDANE", "unified_score": 0.0,
   933→   933→                    "modules_assessed": 0, "critical_modules": 0,
   934→   934→                    "max_score": 0.0, "min_score": 0.0}
   935→   935→
   936→   936→        avg = weighted_sum / total_weight if total_weight > 0 else sum(scores) / len(scores)
   937→   937→        max_score = max(scores)
   938→   938→
   939→   939→        if avg >= 70 or critical_count >= 3:
   940→   940→            verdict = "CRITICAL"
