   150→class AgentMemory:
   151→    """Persistent memory for the Nexus agent across sessions."""
   152→
   153→    def __init__(self):
   154→        MEMORY_DIR.mkdir(parents=True, exist_ok=True)
   155→        self._data = self._load()
   156→
   157→    def _load(self) -> Dict:
   158→        if CONTEXT_FILE.exists():
   159→            try:
   160→                return json.loads(CONTEXT_FILE.read_text())
   161→            except Exception:
   162→                pass
   163→        return {
   164→            "sessions": [],
   165→            "target_contexts": {},
   166→            "learned_patterns": [],
   167→            "created": datetime.now(timezone.utc).isoformat(),
   168→        }
   169→
   170→    def _save(self) -> None:
   171→        # Keep only the last 50 sessions
   172→        if len(self._data.get("sessions", [])) > 50:
   173→            self._data["sessions"] = self._data["sessions"][-50:]
   174→        try:
   175→            CONTEXT_FILE.write_text(json.dumps(self._data, indent=2, default=str))
   176→        except OSError:
   177→            pass
   178→
   179→    def save_session(
   180→        self,
   181→        goal: str,
   182→        target: str,
   183→        plan: List[Dict],
   184→        tool_results: List[Dict],
   185→        summary: str,
   186→    ) -> None:
   187→        session = {
   188→            "goal": goal,
   189→            "target": target,
   190→            "plan": plan,
   191→            "tool_results": tool_results,
   192→            "summary": summary,
   193→            "timestamp": datetime.now(timezone.utc).isoformat(),
   194→        }
   195→        self._data.setdefault("sessions", []).append(session)
   196→        # Update target context
   197→        self._data.setdefault("target_contexts", {})[target] = {
   198→            "last_score": self._extract_score(tool_results),
   199→            "last_grade": self._extract_grade(tool_results),
   200→            "last_run": session["timestamp"],
   201→            "finding_count": self._extract_finding_count(tool_results),
   202→            "known_issues": self._extract_issues(tool_results),
   203→        }
   204→        self._save()
   205→
   206→    def get_target_context(self, target: str) -> Optional[Dict]:
   207→        return self._data.get("target_contexts", {}).get(target)
   208→
   209→    def get_recent_sessions(self, limit: int = 5) -> List[Dict]:
   210→        return self._data.get("sessions", [])[-limit:]
   211→
   212→    @staticmethod
   213→    def _extract_score(results: List[Dict]) -> Optional[int]:
   214→        for r in results:
   215→            d = r.get("data")
   216→            if isinstance(d, dict) and "total_score" in d:
   217→                return d["total_score"]
   218→        return None
   219→
   220→    @staticmethod
   221→    def _extract_grade(results: List[Dict]) -> Optional[str]:
   222→        for r in results:
   223→            d = r.get("data")
   224→            if isinstance(d, dict) and "grade" in d:
   225→                return d["grade"]
   226→        return None
   227→
   228→    @staticmethod
   229→    def _extract_finding_count(results: List[Dict]) -> int:
   230→        total = 0
   231→        for r in results:
   232→            d = r.get("data")
   233→            if isinstance(d, dict) and "findings" in d:
   234→                total += len(d["findings"])
   235→        return total
   236→
   237→    @staticmethod
   238→    def _extract_issues(results: List[Dict]) -> List[str]:
   239→        issues = []
   240→        for r in results:
   241→            d = r.get("data")
   242→            if isinstance(d, dict) and "findings" in d:
   243→                for f in d["findings"]:
   244→                    if isinstance(f, dict) and f.get("severity") in ("critical", "high"):
   245→                        issues.append(f.get("title", ""))
   246→        return issues[:20]

   561→def _tool_shell_command(command: str = "") -> ToolResult:
   562→    """Run an arbitrary shell command. DANGEROUS — use with caution."""
   563→    if not command:
   564→        return ToolResult("No command specified.", success=False)
   565→    dangerous_patterns = ["rm -rf /", "mkfs", "dd if=", ":(){ :|:& };:", "chmod -R 777 /"]
   566→    for dp in dangerous_patterns:
   567→        if dp in command:
   568→            return ToolResult(f"Blocked dangerous command pattern: {dp}", success=False)
   569→    try:
   570→        result = subprocess.run(
   571→            command, shell=True, capture_output=True, text=True, timeout=30,
   572→        )
   573→        output = (result.stdout + result.stderr)[:2000]
   574→        return ToolResult(
   575→            f"Exit code: {result.returncode}\n{output}",
   576→            success=result.returncode == 0,
   577→        )
   578→    except subprocess.TimeoutExpired:
   579→        return ToolResult("Command timed out (30s).", success=False)
   580→    except Exception as e:
   581→        return ToolResult(f"Command error: {e}", success=False)

   784→
   785→class RuleBasedPlanner:
   786→    """Smart rule-based planner that classifies intent, generates plans,
   787→    and adapts based on results — no LLM required.
   788→
   789→    The planner uses a weighted intent scoring system that considers:
   790→      - Direct keyword matches with configurable weights
   791→      - Modifier words that boost or reduce intent scores
   792→      - Target type classification (domain, IP, local, codebase)
   793→      - Proximity of intent words to target words
   794→      - Prior context from persistent memory
   795→    """
   796→
   797→    # Intent keyword weights: (word_pattern, base_weight)
   798→    _INTENT_KEYWORDS: Dict[str, List[Tuple[str, float]]] = {
   799→        "full_scan": [
   800→            ("scan", 0.4), ("audit", 0.3), ("assess", 0.3), ("pentest", 0.5),
   801→            ("security", 0.2), ("vuln", 0.4), ("test", 0.2),
   802→        ],
   803→        "subdomain_recon": [
   804→            ("subdomain", 0.8), ("sub-domain", 0.8), ("attack surface", 0.6),
   805→            ("enumerate", 0.5), ("discover", 0.3), ("surface map", 0.7),
   806→        ],
   807→        "local_audit": [
   808→            ("machine", 0.6), ("laptop", 0.6), ("computer", 0.5), ("local", 0.6),
   809→            ("my system", 0.7), ("this machine", 0.6), ("localhost", 0.7),
   810→        ],
   811→        "dev_security": [
   812→            ("project", 0.5), ("codebase", 0.5), ("repo", 0.4), ("source code", 0.5),
   813→            ("dependencies", 0.4), ("package", 0.3), ("code", 0.3),
   814→        ],
   815→        "secret_hunting": [
   816→            ("secrets", 0.7), ("secret", 0.7), ("credential", 0.7), ("api key", 0.6), ("token", 0.5),
   817→            ("password", 0.5), ("leak", 0.5), ("exposed", 0.3),
   818→        ],
   819→        "compliance": [
   820→            ("compliance", 0.7), ("health", 0.5), ("doctor", 0.5), ("diagnos", 0.5),
   821→            ("checkup", 0.5), ("hardening", 0.5), ("cis", 0.6), ("policy", 0.4),
   822→        ],
   823→        "attack_simulation": [
   824→            ("attack", 0.6), ("exploit", 0.6), ("hack", 0.5), ("swarm", 0.7),
   825→            ("adversarial", 0.7), ("red team", 0.6), ("offensive", 0.5),
   826→        ],
   827→        "code_analysis": [
   828→            (" ast ", 0.7), ("static analysis", 0.5), ("code review", 0.5), ("sast", 0.7),
   829→            ("analyze code", 0.6), ("code quality", 0.4), ("static code", 0.5),
   830→        ],
   831→        "threat_intel": [
   832→            ("cve", 0.7), ("threat", 0.6), ("intel", 0.6), ("enrich", 0.5),
   833→            ("knowledge graph", 0.7), ("graph", 0.4), ("chain", 0.3),
   834→        ],
   835→        "reporting": [
   836→            ("report", 0.7), ("html", 0.4), ("sarif", 0.6), ("pdf", 0.4),
   837→            ("markdown", 0.4), ("export", 0.5), ("document", 0.4), ("generate report", 0.7),
   838→        ],
   839→        "comparison": [
   840→            ("compare", 0.7), ("diff", 0.7), ("vs", 0.5), ("versus", 0.5),
   841→            ("trend", 0.5), ("progress", 0.3),
   842→        ],
   843→        "benchmarking": [
   844→            ("benchmark", 0.7), ("score", 0.4), ("grade", 0.4), ("rating", 0.5),
   845→            ("vibesec", 0.6), ("quick", 0.2),
   846→        ],
   847→    }
   848→
   849→    # Modifier words that amplify intent
   850→    _AMPLIFIERS = {
   851→        "full": 1.5, "deep": 1.5, "complete": 1.4, "everything": 1.5,
   852→        "all": 1.3, "thorough": 1.4, "comprehensive": 1.4, "maximum": 1.5,
   853→        "aggressive": 1.3, "exhaustive": 1.5, "entire": 1.3,
   854→    }
   855→
   856→    # Modifier words that reduce intent
   857→    _DAMPENERS = {
   858→        "quick": 0.7, "fast": 0.7, "brief": 0.6, "simple": 0.7,
   859→        "just": 0.7, "only": 0.6, "basic": 0.6, "lightweight": 0.6,
   860→    }
   861→
   862→    # Module-specific keywords
   863→    _MODULE_KEYWORDS: Dict[str, List[str]] = {
   864→        "recon": ["recon", "reconnaissance", "probe", "fingerprint"],
   865→        "auth": ["auth", "login", "bypass", "session", "cookie"],
   866→        "chain": ["ssrf", "chain", "redirect", "open redirect"],
   867→        "gorgon": ["sqli", "xss", "injection", "gorgon"],
   868→        "oblivion": ["oblivion", "dread", "deep scan", "ultimate"],
   869→        "bot": ["bot", "crawler", "c2", "automation"],
   870→        "nhi": ["nhi", "cloud", "identity", "oauth"],
   871→        "vibesec": ["vibesec", "vibe", "benchmark", "score"],
   872→    }
   873→
   874→    def __init__(self, registry: ToolRegistry, memory: AgentMemory):
   875→        self.registry = registry
   876→        self.memory = memory
   877→
   878→    def classify_intent(self, goal: str) -> Dict[str, float]:
   879→        """Score each intent based on the goal text.
   880→
   881→        Uses weighted keyword matching, proximity scoring, and modifiers.
   882→        Returns a dict mapping intent names to float scores in [0, 1+].
   883→        """
   884→        goal_lower = goal.lower()
   885→        # Pad for whole-word matching
   886→        padded = f' {goal_lower} '
   887→        scores: Dict[str, float] = {}
   888→        words = goal_lower.split()
   889→
   890→        # Calculate global modifier
   891→        modifier = 1.0
   892→        for w in words:
   893→            if w in self._AMPLIFIERS:
   894→                modifier = max(modifier, self._AMPLIFIERS[w])
   895→            if w in self._DAMPENERS:
   896→                modifier = min(modifier, self._DAMPENERS[w])
   897→
   898→        for intent, keywords in self._INTENT_KEYWORDS.items():
   899→            score = 0.0
   900→            for pattern, weight in keywords:
   901→                # Use whole-word matching to avoid false positives
   902→                # (e.g., "repo" matching inside "report")
   903→                if len(pattern.split()) > 1:
   904→                    # Multi-word pattern: check substring (already specific enough)
   905→                    matched = pattern in goal_lower
   906→                else:
   907→                    # Single-word pattern: check whole word with padding
   908→                    matched = f' {pattern} ' in padded
   909→                if matched:
   910→                    # Boost if near a target-like word
   911→                    proximity_boost = 1.0
   912→                    pattern_idx = goal_lower.find(pattern)
   913→                    for tw in (".com", ".io", ".net", ".org", "http", "localhost"):
   914→                        tw_idx = goal_lower.find(tw)
   915→                        if tw_idx >= 0 and abs(pattern_idx - tw_idx) < 40:
   916→                            proximity_boost = 1.3
   917→                            break
   918→                    score += weight * proximity_boost
   919→            scores[intent] = min(score * modifier, 2.0)
   920→
   921→        return scores
   922→
   923→    def extract_targets(self, goal: str) -> List[str]:
   924→        """Extract targets (domains, IPs, URLs, paths) from the goal."""
   925→        urls = re.findall(r'https?://[\w\-._~:/?#\[\]@!$&\'()*+,;=%]+', goal)
   926→        domains = re.findall(r'[\w][\w\-]*\.[\w]{2,}(?:\.[\w]{2,})?', goal)
   927→        skip = {
   928→            'the', 'and', 'for', 'with', 'all', 'of', 'to', 'from', 'that', 'this',
   929→            'scan', 'audit', 'check', 'find', 'get', 'show', 'me', 'my',
   930→        }
   931→        targets = list(set(
   932→            urls + [d for d in domains if d.lower() not in skip and len(d) > 4]
   933→        ))
   934→        return targets
   935→
   936→    def detect_target_type(self, goal: str, targets: List[str]) -> str:
   937→        """Classify the target type from context."""
   938→        goal_lower = goal.lower()
   939→        local_indicators = [
   940→            'machine', 'laptop', 'computer', 'local', 'my system', 'this machine',
   941→            'localhost', 'my mac', 'my pc', 'my server',
   942→        ]
   943→        for ind in local_indicators:
   944→            if ind in goal_lower:
   945→                return "local"
   946→        dev_indicators = ['project', 'codebase', 'repo', 'source code', 'code review']
   947→        padded = f' {goal_lower} '
   948→        for ind in dev_indicators:
   949→            if f' {ind} ' in padded and not targets:
   950→                return "codebase"
   951→        if targets:
   952→            for t in targets:
   953→                if t.startswith("http"):
   954→                    return "remote"
   955→                if re.match(r'^\d+\.\d+\.\d+\.\d+$', t):
   956→                    return "remote"
   957→            return "remote"
   958→        return "unknown"
   959→
   960→    def detect_modules(self, goal: str) -> Optional[List[str]]:
   961→        """Detect specific module requests from the goal text."""
   962→        goal_lower = goal.lower()
   963→        for mod_name, keywords in self._MODULE_KEYWORDS.items():
   964→            for kw in keywords:
   965→                if kw in goal_lower:
   966→                    return [mod_name]
   967→        return None

   969→    def generate_plan(
   970→        self, goal: str, scores: Dict[str, float],
   971→        targets: List[str], target_type: str,
   972→    ) -> List[Dict[str, Any]]:
   973→        """Generate an ordered execution plan based on classified intents.
   974→
   975→        Returns a list of step dicts: {"tool", "args", "reason"}
   976→        """
   977→        plan: List[Dict[str, Any]] = []
   978→        domain = ""
   979→        if targets:
   980→            domain = targets[0].replace("https://", "").replace("http://", "").split("/")[0]
   981→
   982→        # Sort intents by score, take significant ones
   983→        sorted_intents = sorted(scores.items(), key=lambda x: x[1], reverse=True)
   984→        significant = [(name, s) for name, s in sorted_intents if s > 0.2]
   985→
   986→        # Check prior context for this target
   987→        prior = None
   988→        if domain:
   989→            prior = self.memory.get_target_context(domain)
   990→
   991→        # ── Subdomain recon phase ──
   992→        if any(n == "subdomain_recon" and s > 0.3 for n, s in significant):
   993→            if domain:
   994→                plan.append({
   995→                    "tool": "subdomain_discover",
   996→                    "args": {"domain": domain},
   997→                    "reason": "User requested subdomain reconnaissance",
   998→                })
   999→
  1000→        # ── Benchmarking ──
  1001→        if any(n == "benchmarking" and s > 0.3 for n, s in significant):
  1002→            if domain:
  1003→                plan.append({
  1004→                    "tool": "vibesec_benchmark",
  1005→                    "args": {"target": domain},
  1006→                    "reason": "User wants a quick security benchmark",
  1007→                })
  1008→
  1009→        # ── Local audit ──
  1010→        if target_type == "local":
  1011→            if any(n == "compliance" and s > 0.3 for n, s in significant):
  1012→                plan.append({"tool": "doctor_check", "args": {}, "reason": "Health/compliance check requested"})
  1013→            if any(n == "local_audit" and s > 0.2 for n, s in significant):
  1014→                plan.append({"tool": "local_audit", "args": {"target_path": "."}, "reason": "Local machine audit requested"})
  1015→
  1016→        # ── Dev security ──
  1017→        if target_type == "codebase" or any(n == "dev_security" and s > 0.3 for n, s in significant):
  1018→            plan.append({"tool": "dev_scan", "args": {"path": "."}, "reason": "Developer project scan requested"})
  1019→
  1020→        # ── Secret hunting ──
  1021→        if any(n == "secret_hunting" and s > 0.3 for n, s in significant):
  1022→            plan.append({"tool": "secret_hunt", "args": {"path": "."}, "reason": "Secret hunting requested"})
  1023→
  1024→        # ── Code analysis ──
  1025→        if any(n == "code_analysis" and s > 0.3 for n, s in significant):
  1026→            plan.append({"tool": "ast_analyze", "args": {"path": "."}, "reason": "Static code analysis requested"})
  1027→
  1028→        # ── Full remote scan ──
  1029→        is_full = any(n == "full_scan" and s > 0.2 for n, s in significant)
  1030→        if targets and target_type == "remote":
  1031→            modules = self.detect_modules(goal)
  1032→            if len(targets) > 1:
  1033→                # Multiple targets → blitz (even if full_scan is set)
  1034→                plan.append({
  1035→                    "tool": "blitz_scan",
  1036→                    "args": {"targets_str": ",".join(targets)},
  1037→                    "reason": f"Multiple targets detected ({len(targets)}), using parallel blitz scan",
  1038→                })
  1039→            elif is_full or not plan:
  1040→                plan.append({
  1041→                    "tool": "remote_scan",
  1042→                    "args": {
  1043→                        "target": targets[0],
  1044→                        "modules": ",".join(modules) if modules else "",
  1045→                        "all_modules": is_full and not modules,
  1046→                    },
  1047→                    "reason": "Full remote security scan" +
  1048→                              (" with all modules" if is_full and not modules else ""),
  1049→                })
  1050→
  1051→        # ── Attack simulation ──
  1052→        if any(n == "attack_simulation" and s > 0.3 for n, s in significant):
  1053→            if domain:
  1054→                plan.append({
  1055→                    "tool": "adversarial_loop",
  1056→                    "args": {"target": domain, "rounds": 2},
  1057→                    "reason": "Attack simulation / adversarial remediation requested",
  1058→                })
  1059→            elif target_type == "local":
  1060→                plan.append({
  1061→                    "tool": "adversarial_loop",
  1062→                    "args": {"target": "localhost", "rounds": 2},
  1063→                    "reason": "Local adversarial remediation loop",
  1064→                })
  1065→
  1066→        # ── Swarm ──
  1067→        has_swarm_word = "swarm" in goal.lower()
  1068→        if has_swarm_word and any(n == "attack_simulation" and s > 0.3 for n, s in significant):
  1069→            if domain:
  1070→                # Replace adversarial_loop with swarm if swarm is explicitly requested
  1071→                plan = [s for s in plan if s["tool"] != "adversarial_loop"]
  1072→                plan.append({
  1073→                    "tool": "swarm_run",
  1074→                    "args": {"target": domain, "mode": "full"},
  1075→                    "reason": "Full swarm attack requested",
  1076→                })
  1077→
  1078→        # ── Threat intel / CVE enrichment ──
  1079→        # Note: CVE enrichment is added dynamically by adapt_plan() after scan results.
  1080→        # If user explicitly wants knowledge graph, add it now.
  1081→        if any(n == "threat_intel" and s > 0.3 for n, s in significant):
  1082→            if domain and ("graph" in goal.lower() or "chain" in goal.lower()):
  1083→                plan.append({
  1084→                    "tool": "knowledge_graph",
  1085→                    "args": {"action": "build", "target": domain},
  1086→                    "reason": "Knowledge graph analysis requested",
  1087→                })
  1088→
  1089→        # ── Comparison ──
  1090→        if any(n == "comparison" and s > 0.3 for n, s in significant):
  1091→            plan.append({"tool": "diff_scans", "args": {}, "reason": "Scan comparison requested"})
  1092→
  1093→        # ── Reporting ──
  1094→        wants_report = any(n == "reporting" and s > 0.3 for n, s in significant)
  1095→        has_scan_step = any(
  1096→            s["tool"] in ("remote_scan", "local_audit", "blitz_scan", "dev_scan",
  1097→                           "vibesec_benchmark", "adversarial_loop", "swarm_run")
  1098→            for s in plan
  1099→        )
  1100→        if wants_report:
  1101→            fmt = "sarif" if "sarif" in goal.lower() else "html"
  1102→            if has_scan_step:
  1103→                plan.append({
  1104→                    "tool": "generate_report",
  1105→                    "args": {"format": fmt},
  1106→                    "reason": f"Report generation requested ({fmt})",
  1107→                })
  1108→            else:
  1109→                # No scan was run — just generate from latest history
  1110→                plan.append({
  1111→                    "tool": "generate_report",
  1112→                    "args": {"format": fmt},
  1113→                    "reason": f"Generating {fmt} report from latest scan history",
  1114→                })
  1115→
  1116→        # ── Fallback: if no plan was generated ──
  1117→        if not plan:
  1118→            if targets:
  1119→                plan.append({
  1120→                    "tool": "remote_scan",
  1121→                    "args": {"target": targets[0]},
  1122→                    "reason": "Default: scanning the detected target",
  1123→                })
  1124→            elif target_type == "local":
  1125→                plan.append({"tool": "local_audit", "args": {}, "reason": "Default: auditing local machine"})
  1126→            elif target_type == "codebase":
  1127→                plan.append({"tool": "dev_scan", "args": {}, "reason": "Default: scanning developer project"})
  1128→            else:
  1129→                plan.append({"tool": "scan_history", "args": {}, "reason": "No clear target — showing scan history"})

  1131→        return plan

  1133→    def adapt_plan(
  1134→        self,
  1135→        plan: List[Dict[str, Any]],
  1136→        completed_results: List[Tuple[Dict, ToolResult]],
  1137→        current_idx: int,
  1138→    ) -> List[Dict[str, Any]]:
  1139→        """Adapt the remaining plan based on results seen so far.
  1140→
  1141→        This is the 'intelligence' of the rule engine — it inspects intermediate
  1142→        results and inserts new steps or modifies remaining ones.
  1143→        """
  1144→        additions: List[Dict[str, Any]] = []
  1145→        domain = ""
  1146→        last_score = None
  1147→        has_critical = False
  1148→        total_findings = 0
  1149→        all_findings: List[Dict] = []
  1150→        subdomains_found = False
  1151→
  1152→        for step, result in completed_results:
  1153→            d = result.data
  1154→            if isinstance(d, dict):
  1155→                # Extract score from scan results
  1156→                if "total_score" in d:
  1157→                    last_score = d["total_score"]
  1158→                if "vibesec_score" in d:
  1159→                    last_score = d["vibesec_score"]
  1160→                # Collect findings
  1161→                findings = d.get("findings", [])
  1162→                if isinstance(findings, list):
  1163→                    all_findings.extend(findings)
  1164→                    total_findings += len(findings)
  1165→                    for f in findings:
  1166→                        if isinstance(f, dict) and f.get("severity") in ("critical", "high"):
  1167→                            has_critical = True
  1168→                # Check for subdomains
  1169→                if "subdomains" in d and len(d["subdomains"]) > 3:
  1170→                    subdomains_found = True
  1171→                # Get target from data
  1172→                if not domain:
  1173→                    domain = d.get("target", "")
  1174→
  1175→        # If any findings found → add CVE enrichment (if not already planned)
  1176→        has_any_findings = total_findings > 0
  1177→        if has_any_findings and not any(s["tool"] == "cve_enrich" for s in plan[current_idx:]):
  1178→            if all_findings:
  1179→                try:
  1180→                    findings_json = json.dumps(all_findings[:20])
  1181→                except (TypeError, ValueError):
  1182→                    findings_json = "[]"
  1183→                reason = "Enriching findings with CVE data from NVD"
  1184→                if has_critical:
  1185→                    reason = "Critical/high findings detected — enriching with CVE data"
  1186→                additions.append({
  1187→                    "tool": "cve_enrich",
  1188→                    "args": {"findings_json": findings_json},
  1189→                    "reason": reason,
  1190→                })
  1191→
  1192→        # If many subdomains found and not already doing blitz → suggest blitz
  1193→        if subdomains_found and not any(s["tool"] == "blitz_scan" for s in plan[current_idx:]):
  1194→            # Don't add blitz automatically (too aggressive), but add knowledge graph
  1195→            if domain and not any(s["tool"] == "knowledge_graph" for s in plan[current_idx:]):
  1196→                additions.append({
  1197→                    "tool": "knowledge_graph",
  1198→                    "args": {"action": "build", "target": domain},
  1199→                    "reason": "Large attack surface detected — building knowledge graph",
  1200→                })
  1201→
  1202→        # If score is very low → add knowledge graph
  1203→        if last_score is not None and last_score < 40:
  1204→            if domain and not any(s["tool"] == "knowledge_graph" for s in plan[current_idx:]):
  1205→                additions.append({
  1206→                    "tool": "knowledge_graph",
  1207→                    "args": {"action": "build", "target": domain},
  1208→                    "reason": f"Critical score ({last_score}/100) — analyzing attack surface via knowledge graph",
  1209→                })
  1210→
  1211→        # If score is low and no adversarial loop → add one
  1212→        if last_score is not None and last_score < 50:
  1213→            if domain and not any(s["tool"] in ("adversarial_loop", "swarm_run") for s in plan + additions):
  1214→                additions.append({
  1215→                    "tool": "adversarial_loop",
  1216→                    "args": {"target": domain, "rounds": 2},
  1217→                    "reason": f"Low score ({last_score}/100) — running adversarial loop for remediation guidance",
  1218→                })
  1219→
  1220→        # Insert additions after current step
  1221→        remaining = plan[current_idx + 1:]
  1222→        plan = plan[:current_idx + 1] + additions + remaining
  1223→
  1224→        return plan

  1367→    def _execute_llm(self, goal: str, backend: str) -> Dict[str, Any]:
  1368→        """Execute using LLM tool-calling loop."""
  1369→        tool_descs = "\n".join(
  1370→            f"  - {t.name}: {t.description}" for t in self.registry.tools.values()

  1381→        self._print_header(goal, "LLM reasoning")

  1389→                else:

  1396→                self._emit("[yellow]LLM returned empty response — falling back to rules.[/]")
  1397→                return self._execute_rules(goal)
  1398→
  1399→            content = response.get("content", "")
  1400→            tool_calls = response.get("tool_calls", [])
  1401→
  1402→            # If no tool calls, the LLM is done
  1403→            if not tool_calls:
  1404→                if content:
  1405→                    self._emit(f"  [dim]{content[:500]}[/]")
  1406→                break
  1407→
  1408→            # Execute each tool call
  1409→            for tc in tool_calls:
  1410→                fn_name = tc.get("function", {}).get("name", "")
  1411→                fn_args_str = tc.get("function", {}).get("arguments", "{}")

  1429→                    messages.append({"role": "assistant", "content": None, "tool_calls": [tc]})
  1430→                    messages.append({
  1431→                        "role": "tool",
  1432→                        "tool_call_id": tc.get("id", f"call_{fn_name}"),
  1433→                        "content": result.output[:4000],
  1434→                    })
  1435→                else:
  1436→                    messages.append({
  1437→                        "role": "user",
  1438→                        "content": f"Tool {fn_name} result: {result.output[:3000]}",
  1439→                    })
  1440→
  1441→        summary = self._build_summary()
  1442→        self._print_footer(summary)
  1443→        return {

  1519→    def _execute_rules(self, goal: str) -> Dict[str, Any]:
  1520→        """Execute using the intelligent rule-based planner."""
  1521→        # THINK phase
  1522→        scores = self.planner.classify_intent(goal)
  1523→        targets = self.planner.extract_targets(goal)
  1524→        target_type = self.planner.detect_target_type(goal, targets)
  1525→
  1526→        domain = ""
  1527→        if targets:
  1528→            domain = targets[0].replace("https://", "").replace("http://", "").split("/")[0]
  1529→
  1530→        # Build intent display
  1531→        top_intents = sorted(scores.items(), key=lambda x: x[1], reverse=True)[:4]
  1532→        intent_str = " + ".join(f"{n} ({s:.1f})" for n, s in top_intents if s > 0.1)
  1533→
  1534→        self._print_header(goal, "rule-based planner")
  1535→        self._emit(f"  [THINK] Analyzing goal...")
  1536→        self._emit(f"    [dim]→ Target: {domain or target_type} ({target_type})[/]")
  1537→        if intent_str:
  1538→            self._emit(f"    [dim]→ Intent: {intent_str}[/]")
  1539→        self._emit(f"    [dim]→ Strategy: plan → execute → adapt[/]")
  1540→
  1541→        # PLAN phase
  1542→        plan = self.planner.generate_plan(goal, scores, targets, target_type)
  1543→        self._emit(f"")
  1544→        self._emit(f"  [PLAN] {len(plan)} step(s):")
  1545→        for i, step in enumerate(plan, 1):
  1546→            self._emit(f"    {i}. [bold]{step['tool']}[/] — {step['reason']}")
  1547→
  1548→        # EXECUTE phase with adaptation
  1549→        completed: List[Tuple[Dict, ToolResult]] = []
  1550→        step_idx = 0
  1551→        total_steps = len(plan)
  1552→
  1553→        while step_idx < total_steps and step_idx < MAX_ITERATIONS:
  1554→            step = plan[step_idx]
  1555→            tool_name = step["tool"]
  1556→            args = step["args"]
  1557→            current_display = step_idx + 1
  1558→
  1559→            self._emit(f"")
  1560→            self._emit(f"  [EXEC] Step {current_display}/{total_steps}: [bold]{tool_name}[/]...")
  1561→            self._emit_status(f"Executing step {current_display}/{total_steps}: {tool_name}")
  1562→
  1563→            result = self.registry.execute(tool_name, **args)
  1564→            success_mark = "[green]✓[/]" if result.success else "[red]✗[/]"
  1565→            self._emit(f"    {success_mark} {result.output[:300]}")
  1566→
  1567→            self._steps.append(tool_name)
  1568→            self._collect_result(tool_name, result)
  1569→            completed.append((step, result))
  1570→
  1571→            # Track targets
  1572→            if tool_name in ("remote_scan", "vibesec_benchmark", "subdomain_discover",
  1573→                             "adversarial_loop", "swarm_run"):
  1574→                if domain:
  1575→                    self._targets.append(domain)
  1576→            if tool_name == "blitz_scan" and result.data:
  1577→                for t in result.data.get("results", {}):
  1578→                    self._targets.append(t)
  1579→
  1580→            # VERIFY & ADAPT phase (after each step)
  1581→            old_len = len(plan)
  1582→            plan = self.planner.adapt_plan(plan, completed, step_idx)
  1583→            new_steps = len(plan) - old_len
  1584→            if new_steps > 0:
  1585→                total_steps = len(plan)
  1586→                added = plan[step_idx + 1: step_idx + 1 + new_steps]
  1587→                self._emit(f"")
  1588→                self._emit(f"  [ADAPT] Added {new_steps} step(s) based on results:")
  1589→                for a in added:
  1590→                    self._emit(f"    + [bold]{a['tool']}[/] — {a['reason']}")
  1591→
  1592→            step_idx += 1
  1593→
  1594→        summary = self._build_summary()
  1595→        self._print_footer(summary)
  1596→        return {
  1597→            "goal": goal,
  1598→            "steps": self._steps,
  1599→            "targets": self._targets,
  1600→            "total_findings": len(self._findings),
  1601→            "findings": self._findings,
  1602→            "summary": summary,
  1603→            "backend": "rules",
  1604→        }
