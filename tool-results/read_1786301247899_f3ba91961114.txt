  1460→                "Remove or justify all WebSocket connections. Ensure WebSocket traffic is encrypted (WSS) "
  1461→                "and serves a documented functional purpose."
  1462→            ),
  1463→            dread_score=_severity_dread(severity),
  1464→        ))
  1465→

  1490→                dread_score=0.5,
  1491→            ))
  1492→
  1493→    # ── DNS-based beaconing (encoded data in subdomains) ──
  1494→    all_urls = _extract_urls_from_text(body)

  1497→        hostname = parsed.hostname or ""
  1498→        # Check for hex-encoded subdomains (common in DNS tunneling)
  1499→        parts = hostname.split(".")

  1507→                    description=(
  1508→                        f"URL '{url}' contains a hex-encoded subdomain '{part}' ({len(part)} chars). "

  1510→                        f"exfiltrate data through DNS, bypassing most firewalls and proxies."
  1511→                    ),
  1512→                    evidence=f"URL: {url}; encoded subdomain: {part}",
  1513→                    asset=hostname or target,
  1514→                    points_deducted=25,
  1515→                    remediation=(
  1516→                        "Investigate the source of this URL. DNS tunneling indicates a compromised or "
  1517→                        "maliciously crafted document. Block the domain and analyze network DNS logs."
  1518→                    ),
  1519→                    dread_score=0.9,
  1520→                ))

  1710→    # ── HTML meta tag tracking fields ──

  1731→                    tracking_meta_fields.append(f"{tname}={content_match.group(1)}")

  1737→            category="metadata-tracking",

  1748→        ))

  1752→    suspicious_header_names = [
  1753→        "x-request-id", "x-correlation-id", "x-trace-id",
  1754→        "x-session-id", "x-visitor-id", "x-tracking-id",

  1759→        if hname.lower() in suspicious_header_names:
  1760→            tracking_headers.append(f"{hname}: {hval}")
  1761→
  1762→    if tracking_headers:
  1763→        findings.append(Finding(
  1764→            title="Document Metadata — Tracking IDs in HTTP Response Headers",
  1765→            severity="medium",
  1766→            category="metadata-tracking",
  1767→            module=MODULE_NAME,
  1768→            description=(

  1783→    )

  1788→            "mailchimp", "campaign monitor", "sendgrid", "hubspot",
  1789→            "marketo", "pardot", "activecampaign", "convertkit",
  1790→            "mailerlite", "brevo", "constant contact",
  1791→        ]

  1794→                findings.append(Finding(
  1795→                    title="Document Metadata — Email Marketing Platform Generator Detected",
  1796→                    severity="medium",
  1797→                    category="metadata-tracking",
  1798→                    module=MODULE_NAME,
  1799→                    description=(
  1800→                        f"Document generator is '{gen_val}' ({tg}). Email marketing platforms "
  1801→                        f"automatically embed tracking pixels and click tracking in all generated content."
  1802→                    ),
  1803→                    evidence=f"Generator: {gen_val}",
  1804→                    asset=target,
  1805→                    points_deducted=8,

  1807→                        f"Content from {tg} likely contains tracking. Export to clean HTML and "
  1808→                        f"strip all tracking elements before redistribution."
  1809→                    ),
  1810→                    dread_score=0.5,
  1811→                ))

  1815→    html_comments = re.findall(r"<!--(.*?)-->", body, re.DOTALL)

  1830→    if tracking_comments:

  1845→            dread_score=0.5,

  1853→            "x-powered-by", "x-frame-options", "x-content-type-options",

  1859→                len(hval) > 50):  # Long opaque value
  1860→                custom_tracking_headers.append(f"{hname}: {hval[:60]}")
  1861→

  1866→            category="metadata-tracking",

  1871→            ),

  1874→            points_deducted=3,
  1875→            remediation="Audit custom headers and remove any that serve only tracking purposes.",
  1876→            dread_score=0.3,
  1877→        ))
  1878→
  1879→    return findings

  1882→def _check_js_trackers(
  1883→    body: str, base_url: str, target: str
  1884→) -> List[Finding]:
  1885→    """Cat 6: Detect analytics, fingerprinting, and tracking scripts."""
  1886→    findings: List[Finding] = []
  1887→    if not body:
  1888→        return findings
  1889→
  1890→    # Extract all script content
  1891→    script_blocks = _RE_SCRIPT.findall(body)
  1892→    inline_js = "\n".join(script_blocks)
  1893→    all_js = body  # Also scan the full HTML for script src references
  1894→
  1895→    # ── Check against TRACKER_SIGNATURES database ──
  1896→    detected_trackers: List[Dict[str, Any]] = []
  1897→    for tracker in _compiled_trackers:
  1898→        matches = _count_matching_patterns(all_js, tracker["compiled"])
  1899→        if matches:
  1900→            detected_trackers.append({
  1901→                **tracker,
  1902→                "matched_evidence": matches[:3],
  1903→            })
  1904→
  1905→    # Deduplicate by category
  1906→    seen_categories: set = set()
  1907→    for dt in detected_trackers:
  1908→        cat = dt["category"]
  1909→        # Group analytics together, fingerprinting together, etc.
  1910→        if cat in seen_categories:
  1911→            continue
  1912→        # For analytics, allow up to 3 unique trackers before grouping
  1913→        if cat == "analytics" and len([d for d in detected_trackers if d["category"] == "analytics"]) > 3:
  1914→            if dt != [d for d in detected_trackers if d["category"] == "analytics"][0]:
  1915→                continue
  1916→
  1917→        seen_categories.add(cat)
  1918→
  1919→        # Count total trackers in this category
  1920→        category_count = len([d for d in detected_trackers if d["category"] == cat])
  1921→        other_names = [d["name"] for d in detected_trackers if d["category"] == cat and d["name"] != dt["name"]]
  1922→
  1923→        desc = dt["description"]
  1924→        if category_count > 1:
  1925→            desc += f" Also detected in same category: {', '.join(other_names[:3])}."
  1926→
  1927→        evidence_str = "; ".join(dt["matched_evidence"])
  1928→        if category_count > 1:
  1929→            evidence_str += f" (plus {category_count - 1} more in {cat} category)"
  1930→
  1931→        findings.append(Finding(
  1932→            title=f"JavaScript Tracker — {dt['name']} Detected",
  1933→            severity=dt["severity"],
  1934→            category=f"js-{cat}",
  1935→            module=MODULE_NAME,
  1936→            description=desc,
  1937→            evidence=_truncate_evidence(evidence_str),
  1938→            asset=target,
  1939→            points_deducted=_severity_points(dt["severity"]),
