   419→   419→/**
   420→   420→ * Project quantum hardware qubit count for a given year.
   421→   421→ * Uses compound growth from current baseline.
   422→   422→ */
   423→   423→export function estimateQuantumAdvancement(targetYear: number): number {
   424→   424→  const { currentBestQubits, projectedGrowthRate, baselineYear } = QUANTUM_HARDWARE;
   425→   425→  const yearsElapsed = targetYear - baselineYear;
   426→   426→  if (yearsElapsed <= 0) return currentBestQubits;
   427→   427→
   428→   428→  // Compound growth with slight acceleration (quantum advantage curve)
   429→   429→  // Early years grow slightly faster, later years slow as engineering challenges increase
   430→   430→  const accelerationFactor = 1 + (yearsElapsed * 0.003);
   431→   431→  const effectiveGrowth = Math.pow(projectedGrowthRate, yearsElapsed) * accelerationFactor;
   432→   432→
   433→   433→  return Math.floor(currentBestQubits * effectiveGrowth);
   434→   434→}

   436→   436→/**
   437→   437→ * Determine when quantum hardware will have enough qubits to break a given algorithm.
   438→   438→ */
   439→   439→export function calculateBreakYear(
   440→   440→  algorithm: string,
   441→   441→  keySize: number
   442→   442→): { breakYear: number; milestone: QuantumMilestone; confidence: string } {
   443→   443→  // Map common TLS identifiers to our milestone keys
   444→   444→  const algorithmKey = mapAlgorithmToMilestone(algorithm, keySize);
   445→   445→
   446→   446→  if (algorithmKey.startsWith('pqc_')) {
   447→   447→    // Already post-quantum
   448→   448→    const milestone = QUANTUM_MILESTONES[algorithmKey];
   449→   449→    return {
   450→   450→      breakYear: milestone.estimatedBreakYear,
   451→   451→      milestone,
   452→   452→      confidence: 'very_low',
   453→   453→    };
   454→   454→  }
   455→   455→
   456→   456→  const milestone = QUANTUM_MILESTONES[algorithmKey];
   457→   457→
   458→   458→  if (!milestone) {
   459→   459→    // Fallback: estimate based on key size
   460→   460→    const fallbackQubits = keySize * 2;
   461→   461→    let year = new Date().getFullYear();
   462→   462→    let qubits = estimateQuantumAdvancement(year);
   463→   463→    while (qubits < fallbackQubits && year < 2100) {
   464→   464→      year++;
   465→   465→      qubits = estimateQuantumAdvancement(year);
   466→   466→    }
   467→   467→    return {
   468→   468→      breakYear: year,
   469→   469→      milestone: {
   470→   470→        qubitsRequired: fallbackQubits,
   471→   471→        estimatedBreakYear: year,
   472→   472→        confidence: 'low',
   473→   473→        description: `Estimated for ${algorithm} with ${keySize}-bit key`,
   474→   474→      },
   475→   475→      confidence: 'low',
   476→   476→    };
   477→   477→  }
   478→   478→
   479→   479→  // Refine break year based on actual hardware projection
   480→   480→  let projectedBreakYear = milestone.estimatedBreakYear;
   481→   481→  let qubits = estimateQuantumAdvancement(projectedBreakYear);
   482→   482→
   483→   483→  // If hardware projection suggests earlier or later break
   484→   484→  if (qubits >= milestone.qubitsRequired) {
   485→   485→    // Walk backwards to find when qubits first meet the requirement
   486→   486→    while (projectedBreakYear > new Date().getFullYear() + 1) {
   487→   487→      projectedBreakYear--;
   488→   488→      if (estimateQuantumAdvancement(projectedBreakYear) < milestone.qubitsRequired) {
   489→   489→        projectedBreakYear++;
   490→   490→        break;
   491→   491→      }
   492→   492→    }
   493→   493→  } else {
   494→   494→    // Walk forwards
   495→   495→    while (qubits < milestone.qubitsRequired && projectedBreakYear < 2100) {
   496→   496→      projectedBreakYear++;
   497→   497→      qubits = estimateQuantumAdvancement(projectedBreakYear);
   498→   498→    }
   499→   499→  }
   500→   500→
   501→   501→  return {
   502→   502→    breakYear: projectedBreakYear,
   503→   503→    milestone,
   504→   504→    confidence: milestone.confidence,
   505→   505→  };
   506→   506→}

   508→   508→/**
   509→   509→ * Map TLS algorithm identifiers to internal milestone keys.
   510→   510→ */
   511→   511→function mapAlgorithmToMilestone(algorithm: string, keySize: number): string {
   512→   512→  const algo = algorithm.toUpperCase().trim();
   513→   513→
   514→   514→  // Post-quantum detection
   515→   515→  if (algo.includes('KYBER') || algo.includes('ML-KEM')) {
   516→   516→    if (keySize >= 1024) return 'pqc_kyber_1024';
   517→   517→    if (keySize >= 768) return 'pqc_kyber_768';
   518→   518→    return 'pqc_kyber_512';
   519→   519→  }
   520→   520→  if (algo.includes('DILITHIUM') || algo.includes('ML-DSA')) {
   521→   521→    if (keySize >= 5) return 'pqc_dilithium5';
   522→   522→    if (keySize >= 3) return 'pqc_dilithium3';
   523→   523→    return 'pqc_dilithium2';
   524→   524→  }
   525→   525→  if (algo.includes('SPHINCS') || algo.includes('SLH-DSA')) {
   526→   526→    return 'pqc_sphincsplus';
   527→   527→  }
   528→   528→
   529→   529→  // RSA
   530→   530→  if (algo.includes('RSA')) {
   531→   531→    if (keySize >= 8192) return 'rsa_8192';
   532→   532→    if (keySize >= 4096) return 'rsa_4096';
   533→   533→    if (keySize >= 2048) return 'rsa_2048';
   534→   534→    if (keySize >= 1024) return 'rsa_1024';
   535→   535→    return 'rsa_2048'; // default assumption
   536→   536→  }
   537→   537→
   538→   538→  // Elliptic Curve
   539→   539→  if (algo.includes('ECDH') || algo.includes('ECDHE') || algo.includes('X25519') || algo.includes('X448')) {
   540→   540→    if (algo.includes('P-384') || algo.includes('SECP384') || keySize >= 384) return 'ec_p384';
   541→   541→    if (algo.includes('P-521') || algo.includes('SECP521') || keySize >= 521) return 'ec_p521';
   542→   542→    if (algo.includes('25519')) return 'ec_curve25519';
   543→   543→    if (keySize >= 384) return 'ec_p384';
   544→   544→    return 'ec_p256'; // default assumption
   545→   545→  }
   546→   546→
   547→   547→  if (algo.includes('ECDSA') || algo.includes('ED25519') || algo.includes('EDDSA')) {
   548→   548→    if (algo.includes('P-384') || algo.includes('SECP384') || keySize >= 384) return 'ec_p384';
   549→   549→    if (algo.includes('P-521') || algo.includes('SECP521') || keySize >= 521) return 'ec_p521';
   550→   550→    if (algo.includes('25519')) return 'ec_ed25519';
   551→   551→    if (keySize >= 384) return 'ec_p384';
   552→   552→    return 'ec_p256';
   553→   553→  }
   554→   554→
   555→   555→  // AES / Symmetric
   556→   556→  if (algo.includes('AES')) {
   557→   557→    if (keySize >= 256) return 'aes_256';
   558→   558→    return 'aes_128';
   559→   559→  }
   560→   560→
   561→   561→  if (algo.includes('3DES') || algo.includes('DES')) {
   562→   562→    return 'aes_128'; // 3DES effectively worse than AES-128
   563→   563→  }
   564→   564→
   565→   565→  if (algo.includes('SHA')) {
   566→   566→    if (algo.includes('384') || keySize >= 384) return 'sha_384';
   567→   567→    return 'sha_256';
   568→   568→  }
   569→   569→
   570→   570→  // Default fallback — assume RSA-2048 as most common
   571→   571→  return 'rsa_2048';
   572→   572→}

   658→   658→/**
   659→   659→ * Harvest Now, Decrypt Later (HNDL) risk assessment.
   660→   660→ * Attackers are ALREADY capturing encrypted traffic to decrypt later.
   661→   661→ */
   662→   662→export function calculateHNDLRisk(
   663→   663→  assets: AssetAssessment[],
   664→   664→  industry?: string,
   665→   665→  companyName?: string
   666→   666→): HNDLRiskAssessment {
   667→   667→  const now = new Date();
   668→   668→  const nonPQCAssets = assets.filter(a => a.algorithmFamily !== 'pqc');
   669→   669→
   670→   670→  if (nonPQCAssets.length === 0) {
   671→   671→    return {
   672→   672→      score: 0,
   673→   673→      description: 'All assets use post-quantum cryptography. No HNDL risk detected.',
   674→   674→      dataAtRisk: 'None',
   675→   675→      recommendation: 'Continue monitoring quantum hardware developments.',
   676→   676→      captureWindowYears: 0,
   677→   677→      estimatedRecords: '0',
   678→   678→      regulatoryExposure: 'None',
   679→   679→    };
   680→   680→  }
   681→   681→
   682→   682→  // Calculate capture window — the gap between now and when quantum can decrypt
   683→   683→  const earliestBreakYear = Math.min(...nonPQCAssets.map(a => a.estimatedBreakYear));
   684→   684→  const captureWindow = earliestBreakYear - now.getFullYear();
   685→   685→
   686→   686→  // Score based on capture window, number of assets, and data sensitivity
   687→   687→  let baseScore = 0;
   688→   688→
   689→   689→  // Capture window component (shorter window = higher urgency)
   690→   690→  if (captureWindow <= 3) baseScore += 35;
   691→   691→  else if (captureWindow <= 5) baseScore += 25;
   692→   692→  else if (captureWindow <= 7) baseScore += 15;
   693→   693→  else baseScore += 5;
   694→   694→
   695→   695→  // Asset count component
   696→   696→  baseScore += Math.min(25, nonPQCAssets.length * 5);
   697→   697→
   698→   698→  // Industry sensitivity component
   699→   699→  const industryMultiplier: Record<string, number> = {
   700→   700→    finance: 1.3,
   701→   701→    healthcare: 1.35,
   702→   702→    government: 1.25,
   703→   703→    retail: 1.15,
   704→   704→    energy: 1.2,
   705→   705→    telecom: 1.15,
   706→   706→    technology: 1.0,
   707→   707→    education: 1.1,
   708→   708→  };
   709→   709→  baseScore *= industryMultiplier[industry || 'technology'];
   710→   710→
   711→   711→  // Critical assets bonus
   712→   712→  const criticalCount = nonPQCAssets.filter(a => a.riskLevel === 'CRITICAL').length;
   713→   713→  baseScore += criticalCount * 8;
   714→   714→
   715→   715→  const score = Math.min(100, Math.round(baseScore));
   716→   716→
   717→   717→  // Estimate data at risk
   718→   718→  const estimatedRecords = estimateRecordsAtRisk(industry, nonPQCAssets.length);
   719→   719→
   720→   720→  // Regulatory exposure
   721→   721→  const regulatoryExposure = getRegulatoryExposure(industry);
   722→   722→
   723→   723→  // Generate description
   724→   724→  const descriptions: Record<number, string> = {
   725→   725→    80: `CRITICAL HNDL THREAT: ${companyName ? `${companyName}'s` : 'Your'} encrypted traffic is actively being captured by state-sponsored adversaries. With quantum decryption expected within ${captureWindow} years, intercepted data will be fully readable. Immediate PQC migration is non-negotiable.`,
   726→   726→    60: `SEVERE HNDL RISK: ${companyName ? `${companyName}'s` : 'Your'} encrypted communications are vulnerable to "store now, decrypt later" attacks. Nation-state threat actors routinely bulk-capture TLS traffic. With ${captureWindow} years until quantum decryption, time is running out.`,
   727→   727→    40: `MODERATE HNDL EXPOSURE: ${companyName ? `${companyName}` : 'Your organization'} has ${nonPQCAssets.length} assets using quantum-vulnerable cryptography. Adversaries may be stockpiling encrypted data for future quantum decryption within ${captureWindow} years.`,
   728→   728→    20: `LOW HNDL RISK: While ${companyName ? `${companyName}` : 'your organization'} has some quantum-vulnerable assets, the timeline provides adequate runway for migration. HNDL attacks remain a theoretical concern.`,
   729→   729→    0: `MINIMAL HNDL RISK: ${companyName ? `${companyName}'s` : 'Your'} cryptography posture provides reasonable protection against harvest-now-decrypt-later attacks in the near term.`,
   730→   730→  };
   731→   731→
   732→   732→  const descKey = score >= 80 ? 80 : score >= 60 ? 60 : score >= 40 ? 40 : score >= 20 ? 20 : 0;
   733→   733→
   734→   734→  return {
   735→   735→    score,
   736→   736→    description: descriptions[descKey],
   737→   737→    dataAtRisk: `estimated ${estimatedRecords} records`,
   738→   738→    recommendation: score >= 60
   739→   739→      ? 'URGENT: Begin PQC migration immediately. Implement hybrid key exchange as interim measure. Audit all long-term data stores.'
   740→   740→      : score >= 40
   741→   741→        ? 'Begin PQC migration planning. Prioritize highest-risk assets. Test hybrid deployments in staging.'
   742→   742→        : 'Continue monitoring NIST PQC transition timelines. Begin inventory of quantum-vulnerable systems.',
   743→   743→    captureWindowYears: Math.max(0, captureWindow),
   744→   744→    estimatedRecords,
   745→   745→    regulatoryExposure,
   746→   746→  };
   747→   747→}

   807→   807→export function generateMigrationPlan(assets: AssetAssessment[]): MigrationStep[] {
   808→   808→  // Group assets by migration category
   809→   809→  const categories: Record<string, AssetAssessment[]> = {
   810→   810→    'Key Exchange': [],
   811→   811→    'Digital Signatures': [],
   812→   812→    'Symmetric Encryption': [],
   813→   813→    'Post-Quantum Ready': [],
   814→   814→  };
   815→   815→
   816→   816→  for (const asset of assets) {
   817→   817→    if (asset.algorithmFamily === 'pqc') {
   818→   818→      categories['Post-Quantum Ready'].push(asset);
   819→   819→    } else if (
   820→   820→      asset.algorithmFamily === 'rsa' ||
   821→   821→      (asset.algorithmFamily === 'ec' && asset.keyExchange.includes('ECDH'))
   822→   822→    ) {
   823→   823→      categories['Key Exchange'].push(asset);
   824→   824→    } else if (asset.algorithmFamily === 'ec') {
   825→   825→      categories['Digital Signatures'].push(asset);
   826→   826→    } else {
   827→   827→      categories['Symmetric Encryption'].push(asset);
   828→   828→    }
   829→   829→  }
   830→   830→
   831→   831→  const steps: MigrationStep[] = [];
   832→   832→  let priority = 1;
   833→   833→
   834→   834→  // Key exchange migrations (highest priority — direct HNDL target)

   836→   836→    const migration = PQC_MIGRATIONS.key_exchange;
   837→   837→    steps.push({
   838→   838→      priority: priority++,
   839→   839→      asset: `${asset.domain}:${asset.port}`,
   840→   840→      currentCrypto: `${asset.currentAlgorithm} (${asset.keySize}-bit)`,
   841→   841→      recommendedCrypto: migration.to,
   842→   842→      nistLevel: migration.nistLevel,
   843→   843→      estimatedEffort: asset.riskLevel === 'CRITICAL' ? '2-4 weeks' : '4-8 weeks',
   844→   844→      costEstimate: asset.riskLevel === 'CRITICAL' ? '$25K-$75K' : '$50K-$150K',
   845→   845→      category: 'Key Exchange',
   846→   846→      urgency: migration.urgency,
   847→   847→    });
   848→   848→  }
   849→   849→
   850→   850→  // Digital signature migrations
   851→   851→  for (const asset of categories['Digital Signatures']) {
   852→   852→    const migration = PQC_MIGRATIONS.signatures;
   853→   853→    steps.push({
   854→   854→      priority: priority++,
   855→   855→      asset: `${asset.domain}:${asset.port}`,
   856→   856→      currentCrypto: `${asset.currentAlgorithm} (${asset.keySize}-bit)`,
   857→   857→      recommendedCrypto: migration.to,
   858→   858→      nistLevel: migration.nistLevel,
   859→   859→      estimatedEffort: '2-6 months',
   860→   860→      costEstimate: '$75K-$250K',
   861→   861→      category: 'Digital Signatures',
   862→   862→      urgency: migration.urgency,
   863→   863→    });
   864→   864→  }
   865→   865→
   866→   866→  // Symmetric upgrades
   867→   867→  for (const asset of categories['Symmetric Encryption']) {
   868→   868→    const migration = PQC_MIGRATIONS.symmetric_upgrade;
   869→   869→    steps.push({
   870→   870→      priority: priority++,
   871→   871→      asset: `${asset.domain}:${asset.port}`,
   872→   872→      currentCrypto: `${asset.currentAlgorithm} (${asset.keySize}-bit)`,
   873→   873→      recommendedCrypto: migration.to,
   874→   874→      nistLevel: migration.nistLevel,
   875→   875→      estimatedEffort: '2-4 weeks',
   876→   876→      costEstimate: '$10K-$50K',
   877→   877→      category: 'Symmetric Encryption',
   878→   878→      urgency: migration.urgency,
   879→   879→    });
   880→   880→  }
   881→   881→
   882→   882→  // Sort by priority (earliest break year first within same category)
   883→   883→  steps.sort((a, b) => {
   884→   884→    const aAsset = assets.find(x => `${x.domain}:${x.port}` === a.asset);
   885→   885→    const bAsset = assets.find(x => `${x.domain}:${x.port}` === b.asset);
   886→   886→    if (aAsset && bAsset) {
   887→   887→      return aAsset.estimatedBreakYear - bAsset.estimatedBreakYear;
   888→   888→    }
   889→   889→    return a.priority - b.priority;
   890→   890→  });
   891→   891→
   892→   892→  // Re-number priorities after sort
   893→   893→  steps.forEach((step, i) => {
   894→   894→    step.priority = i + 1;
   895→   895→  });
   896→   896→
   897→   897→  return steps;
   898→   898→}

   981→   981→ * Determine the display name for a cipher/key exchange combination.

   983→   983→function getAlgorithmDisplayName(
   984→   984→  keyExchange: string,
   985→   985→  cipherSuite: string,
   986→   986→  keySize: number
   987→   987→): string {
   988→   988→  const ke = keyExchange.toUpperCase();
   989→   989→  if (ke.includes('ECDHE') && ke.includes('ECDSA')) return 'ECDHE-ECDSA';
   990→   990→  if (ke.includes('ECDHE') && ke.includes('RSA')) return 'ECDHE-RSA';
   991→   991→  if (ke.includes('ECDHE')) return 'ECDHE';
   992→   992→  if (ke.includes('RSA')) return 'RSA';
   993→   993→  if (ke.includes('X25519') || ke.includes('X448')) return 'X25519';
   994→   994→  if (cipherSuite.includes('AES_256')) return 'AES-256-GCM';
   995→   995→  if (cipherSuite.includes('AES_128')) return 'AES-128-GCM';
   996→   996→  if (ke.includes('DHE')) return 'DHE';
   997→   997→  return ke || cipherSuite.split('_')[0] || 'UNKNOWN';
   998→   998→}
