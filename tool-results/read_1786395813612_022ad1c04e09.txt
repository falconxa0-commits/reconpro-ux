   136→   136→
   137→   137→    Usage:
   138→   138→        disc = AIEndpointDiscovery()

   142→   142→    TIMEOUT = 8  # seconds
   143→   143→    USER_AGENT = "ReconPro-AI-RedTeam/9.1.0"
   144→   144→
   145→   145→    def __init__(self, extra_patterns: Optional[List[str]] = None) -> None:
   146→   146→        self._patterns = list(AI_ENDPOINT_PATTERNS)
   147→   147→        if extra_patterns:
   148→   148→            self._patterns.extend(extra_patterns)
   149→   149→
   150→   150→    def probe(self, target: str, base_url: str) -> List[Dict[str, Any]]:
   151→   151→        """Probe *base_url* for every known AI endpoint pattern.
   152→   152→
   153→   153→        Returns a list of dicts for endpoints that returned a response:
   154→   154→            {"path": ..., "status": ..., "api_type": ..., "vendor_clue": ...}
   155→   155→        """
   156→   156→        base_url = base_url.rstrip("/")
   157→   157→        results: List[Dict[str, Any]] = []
   158→   158→
   159→   159→        for path in self._patterns:
   160→   160→            url = f"{base_url}{path}"
   161→   161→            entry = self._probe_one(url, path)
   162→   162→            if entry is not None:
   163→   163→                results.append(entry)
   164→   164→
   165→   165→        return results
   166→   166→
   167→   167→    def _probe_one(self, url: str, path: str) -> Optional[Dict[str, Any]]:
   168→   168→        """Send a GET to *url*; return info dict or None on total failure."""
   169→   169→        try:
   170→   170→            req = urllib.request.Request(url, method="GET")
   171→   171→            req.add_header("User-Agent", self.USER_AGENT)
   172→   172→            req.add_header("Accept", "application/json, text/*, */*")
   173→   173→            with urllib.request.urlopen(req, timeout=self.TIMEOUT) as resp:
   174→   174→                body = resp.read(8192).decode("utf-8", errors="replace")
   175→   175→                headers = dict(resp.headers)
   176→   176→                return self._analyse(path, resp.status, headers, body)
   177→   177→        except urllib.error.HTTPError as exc:
   178→   178→            # 4xx/5xx still means the endpoint *exists*
   179→   179→            body = ""
   180→   180→            try:
   181→   181→                body = exc.read(8192).decode("utf-8", errors="replace")
   182→   182→            except Exception:
   183→   183→                pass
   184→   184→            headers = dict(exc.headers) if exc.headers else {}
   185→   185→            return self._analyse(path, exc.code, headers, body)
   186→   186→        except Exception:
   187→   187→            return None
   188→   188→
   189→   189→    def _analyse(self, path: str, status: int, headers: Dict[str, str],

   376→   376→class AIVendorFingerprinter:
   377→   377→    """Fingerprint the AI vendor behind a base_url.
   378→   378→
   379→   379→    Usage:
   380→   380→        fp = AIVendorFingerprinter()
   381→   381→        vendors = fp.fingerprint("https://api.example.com")
   382→   382→    """
   383→   383→
   384→   384→    TIMEOUT = 10
   385→   385→    USER_AGENT = "ReconPro-AI-RedTeam/9.1.0"
   386→   386→    PROBE_PATHS = ["/v1/models", "/", "/health", "/api/docs", "/v1/chat/completions"]
   387→   387→
   388→   388→    def fingerprint(self, base_url: str) -> List[Dict[str, Any]]:
   389→   389→        """Return list of {vendor, confidence, evidence} for detected vendors."""
   390→   390→        base_url = base_url.rstrip("/")
   391→   391→        all_headers: Dict[str, str] = {}
   392→   392→        all_bodies: List[str] = []
   393→   393→        found_paths: List[str] = []
   394→   394→
   395→   395→        for path in self.PROBE_PATHS:
   396→   396→            try:
   397→   397→                url = f"{base_url}{path}"
   398→   398→                req = urllib.request.Request(url, method="GET")
   399→   399→                req.add_header("User-Agent", self.USER_AGENT)
   400→   400→                req.add_header("Accept", "application/json")
   401→   401→                with urllib.request.urlopen(req, timeout=self.TIMEOUT) as resp:
   402→   402→                    all_headers.update(dict(resp.headers))
   403→   403→                    body = resp.read(16384).decode("utf-8", errors="replace")
   404→   404→                    all_bodies.append(body)
   405→   405→                    found_paths.append(path)
   406→   406→            except urllib.error.HTTPError as exc:
   407→   407→                if exc.headers:
   408→   408→                    all_headers.update(dict(exc.headers))
   409→   409→                try:
   410→   410→                    body = exc.read(16384).decode("utf-8", errors="replace")
   411→   411→                    all_bodies.append(body)
   412→   412→                    found_paths.append(path)
   413→   413→                except Exception:
   414→   414→                    pass
   415→   415→            except Exception:
   416→   416→                continue
   417→   417→
   418→   418→        combined_body = " ".join(all_bodies).lower()
   419→   419→        combined_headers = {k.lower(): v.lower() for k, v in all_headers.items()}
   420→   420→
   421→   421→        return self._match_vendors(combined_headers, combined_body, found_paths)
   422→   422→
   423→   423→    def _match_vendors(self, headers: Dict[str, str], body: str,
   424→   424→                       paths: List[str]) -> List[Dict[str, Any]]:
   425→   425→        """Score each vendor against collected evidence."""
   426→   426→        results: List[Dict[str, Any]] = []
   427→   427→
   428→   428→        for sig in _VENDOR_SIGNATURES:
   429→   429→            score = 0.0
   430→   430→            evidence: List[str] = []
   431→   431→
   432→   432→            # Header clues (high weight)
   433→   433→            for hc in sig["header_clues"]:
   434→   434→                if hc.lower() in headers:
   435→   435→                    score += 0.35
   436→   436→                    evidence.append(f"header:{hc}")
   437→   437→
   438→   438→            # Body clues (medium weight)
   439→   439→            for bc in sig["body_clues"]:
   440→   440→                if bc.lower() in body:
   441→   441→                    score += 0.15
   442→   442→                    evidence.append(f"body:{bc}")
   443→   443→
   444→   444→            # Endpoint clues (medium weight)
   445→   445→            for ec in sig["endpoint_clues"]:
   446→   446→                if ec in paths:
   447→   447→                    score += 0.20
   448→   448→                    evidence.append(f"endpoint:{ec}")
   449→   449→
   450→   450→            # Cap at 1.0
   451→   451→            score = min(round(score, 2), 1.0)
   452→   452→
   453→   453→            if score >= 0.15:
   454→   454→                results.append({
   455→   455→                    "vendor": sig["name"],
   456→   456→                    "confidence": score,
   457→   457→                    "evidence": evidence,
   458→   458→                })
   459→   459→
   460→   460→        results.sort(key=lambda r: r["confidence"], reverse=True)
   461→   461→        return results

   465→   465→# Watermark Analyzer
   466→   466→# ────────────────────────────────────────────────────────────────────────
   467→   467→
   468→   468→# Known watermark-related header keys and body patterns
   469→   469→_WATERMARK_HEADERS = {
   470→   470→    "x-watermark", "x-content-watermark", "x-ai-watermark",
   471→   471→    "x-signature-sha256", "x-hash-chain", "x-provenance",
   472→   472→}
   473→   473→
   474→   474→# Raw strings with \x22 for embedded double-quote in char classes
   475→   475→_WATERMARK_BODY_PATTERNS = [
   476→   476→    (r"watermark[\s_-]?id[:\s]+[\w-]+", "watermark-id"),
   477→   477→    (r"watermark[\s_-]?token[:\s]+[\w+/=]+", "watermark-token"),
   478→   478→    (r"[0-9a-f]{64}", "sha256-hex"),
   479→   479→    (r'x-watermark[^"]*', "x-watermark-header-leak"),
   480→   480→    (r'provenance[\x22:\s]+[\w./-]+', "provenance-chain"),
   481→   481→]
   482→   482→
   483→   483→# Token-frequency bias indicators: short, highly repeated n-grams
   484→   484→_BIAS_NGRAM_RE = re.compile(r"\b(\w{3,8})\b")
   485→   485→
   486→   486→
   487→   487→class WatermarkAnalyzer:
   488→   488→    """Detect cryptographic watermark indicators in LLM responses.
   489→   489→
   490→   490→    Usage:
   491→   491→        wa = WatermarkAnalyzer()
   492→   492→        result = wa.analyze(response_text, response_headers)
   493→   493→    """
   494→   494→
   495→   495→    def analyze(self, response_text: str,
   496→   496→                response_headers: Optional[Dict[str, str]] = None) -> Dict[str, Any]:
   497→   497→        """Return watermark analysis dict."""
   498→   498→        response_headers = response_headers or {}
   499→   499→        detected = False
   500→   500→        confidence = 0.0
   501→   501→        wm_type = "none"
   502→   502→        indicators: List[str] = []
   503→   503→
   504→   504→        # --- Header checks ---
   505→   505→        for hdr in _WATERMARK_HEADERS:
   506→   506→            val = response_headers.get(hdr, response_headers.get(hdr.lower(), ""))
   507→   507→            if val:
   508→   508→                detected = True
   509→   509→                confidence += 0.4
   510→   510→                indicators.append(f"header:{hdr}={val[:60]}")
   511→   511→                wm_type = "header-injected"
   512→   512→
   513→   513→        # --- Body pattern checks ---
   514→   514→        text_sample = response_text[:8192]
   515→   515→        for pattern, label in _WATERMARK_BODY_PATTERNS:
   516→   516→            try:
   517→   517→                matches = re.findall(pattern, text_sample, re.IGNORECASE)
   518→   518→            except re.error:
   519→   519→                matches = []
   520→   520→            if matches:
   521→   521→                detected = True
   522→   522→                confidence = min(confidence + 0.25, 1.0)
   523→   523→                indicators.append(f"body:{label} ({len(matches)} hit(s))")
   524→   524→                if wm_type == "none":
   525→   525→                    wm_type = label
   526→   526→
   527→   527→        # --- Bias analysis: check for unusually uniform token frequency ---
   528→   528→        bias_result = self._check_token_bias(text_sample)
   529→   529→        if bias_result["suspicious"]:
   530→   530→            detected = True
   531→   531→            confidence = min(confidence + bias_result["score"], 1.0)
   532→   532→            indicators.append(f"bias:{bias_result['reason']}")
   533→   533→            if wm_type == "none":
   534→   534→                wm_type = "token-bias"
   535→   535→
   536→   536→        return {
   537→   537→            "watermark_detected": detected,
   538→   538→            "confidence": round(min(confidence, 1.0), 3),
   539→   539→            "type": wm_type,
   540→   540→            "indicators": indicators,
   541→   541→        }
   542→   542→
   543→   543→    @staticmethod
   544→   544→    def _check_token_bias(text: str) -> Dict[str, Any]:
   545→   545→        """Look for suspiciously uniform short-token distributions.
   546→   546→
   547→   547→        Watermarking schemes like Kirchenbauer et al. introduce subtle
   548→   548→        token-frequency bias.  We approximate by checking if a short
   549→   549→        n-gram repeats far more than expected.
   550→   550→        """
   551→   551→        tokens = _BIAS_NGRAM_RE.findall(text.lower())
   552→   552→        if len(tokens) < 50:
   553→   553→            return {"suspicious": False, "score": 0.0, "reason": ""}
   554→   554→
   555→   555→        freq: Dict[str, int] = {}
   556→   556→        for t in tokens:
   557→   557→            freq[t] = freq.get(t, 0) + 1
   558→   558→
   559→   559→        total = len(tokens)
   560→   560→        unique = len(freq)
   561→   561→        if total == 0:
   562→   562→            return {"suspicious": False, "score": 0.0, "reason": ""}
   563→   563→
   564→   564→        # Type-token ratio: normal English ~0.4-0.6; watermarked may skew
   565→   565→        ttr = unique / total
   566→   566→
   567→   567→        # Top-token concentration
   568→   568→        top_count = max(freq.values())
   569→   569→        top_ratio = top_count / total
   570→   570→
   571→   571→        suspicious = False
   572→   572→        reason = ""
   573→   573→        score = 0.0
   574→   574→
   575→   575→        if ttr < 0.15 and total > 200:
   576→   576→            suspicious = True
   577→   577→            score += 0.3
   578→   578→            reason = f"low-ttr({ttr:.3f})"
   579→   579→
   580→   580→        if top_ratio > 0.08:
   581→   581→            suspicious = True
   582→   582→            score += 0.2
   583→   583→            reason = (reason + "; " if reason else "") + f"top-token-concentration({top_ratio:.3f})"
   584→   584→
   585→   585→        # Check for hex-block patterns (cryptographic residue)
   586→   586→        hex_blocks = re.findall(r"(?:^|\s)[0-9a-f]{32,}(?:\s|$)", text)
   587→   587→        if hex_blocks:
   588→   588→            suspicious = True
   589→   589→            score += 0.3
   590→   590→            reason = (reason + "; " if reason else "") + f"hex-blocks({len(hex_blocks)})"
   591→   591→
   592→   592→        return {"suspicious": suspicious, "score": score, "reason": reason}
   593→   593→
   594→   594→
   595→   595→# ────────────────────────────────────────────────────────────────────────
   596→   596→# Model Collapse Detector
   597→   597→# ────────────────────────────────────────────────────────────────────────
   598→   598→
   599→   599→_COLLAPSE_PATTERNS: List[Tuple[str, str, float]] = [
   600→   600→    # (regex, label, weight)
   601→   601→    (r"(?:(?:^|\n)\s*\S+\s*\n){8,}", "line-repetition", 0.3),
   602→   602→    (r"(.{20,60})\1{4,}", "substring-loop", 0.4),
   603→   603→    (r"(?:I (?:am|can|will|think|believe) ){5,}", "first-person-loop", 0.35),
   604→   604→    (r"(?:The (?:answer|result|output|response) is)\s+(?:yes|no)\b(.+\1){3,}", "answer-loop", 0.4),
   605→   605→    (r"\[REPEAT\]|\[LOOP\]|\[ERROR\]|\[RECURSION\]", "meta-error-token", 0.5),
   606→   606→    (r"(?:(?:sorry|apologies|regret),?\s+){3,}", "apology-loop", 0.3),
   607→   607→    (r"(?:(?:as an AI|as a language model|I'm (?:just |an? ))\s*(?:language model|AI|assistant).*){3,}", "identity-loop", 0.35),
   608→   608→    (r"N/A|NaN|null|undefined|<unk>|<pad>", "degenerate-token", 0.2),
   609→   609→    (r"(?:\\n|\\t){5,}", "escape-sequence-leak", 0.25),
   610→   610→    (r"<\|endoftext\|>|<\|im_start\|>|<\|im_end\|>|</s>|<eos>", "special-token-leak", 0.45),
   611→   611→]
   612→   612→
   613→   613→
   614→   614→class ModelCollapseDetector:
   615→   615→    """Detect model collapse / degenerate outputs in LLM responses.
   616→   616→
   617→   617→    Usage:
   618→   618→        mcd = ModelCollapseDetector()
   619→   619→        result = mcd.detect(response_text)
   620→   620→    """
   621→   621→
   622→   622→    def detect(self, response_text: str) -> Dict[str, Any]:
   623→   623→        """Analyze text for model-collapse indicators."""
   624→   624→        indicators: List[Dict[str, Any]] = []
   625→   625→        total_score = 0.0
   626→   626→
   627→   627→        for pattern, label, weight in _COLLAPSE_PATTERNS:
   628→   628→            try:
   629→   629→                matches = re.findall(pattern, response_text, re.IGNORECASE | re.MULTILINE)
   630→   630→            except re.error:
   631→   631→                matches = []
   632→   632→            if matches:
   633→   633→                total_score += weight
   634→   634→                indicators.append({
   635→   635→                    "type": label,
   636→   636→                    "matches": len(matches),
   637→   637→                    "weight": weight,
   638→   638→                    "sample": str(matches[0])[:120],
   639→   639→                })
   640→   640→
   641→   641→        # Additional heuristic: entropy of character distribution
   642→   642→        entropy_score = self._char_entropy_score(response_text)
   643→   643→        if entropy_score < 3.0 and len(response_text) > 200:
   644→   644→            total_score += 0.2
   645→   645→            indicators.append({
   646→   646→                "type": "low-entropy",
   647→   647→                "matches": 1,
   648→   648→                "weight": 0.2,
   649→   649→                "sample": f"char-entropy={entropy_score:.2f}",
   650→   650→            })
   651→   651→
   652→   652→        collapse_probability = min(round(total_score, 3), 1.0)
   653→   653→        return {
   654→   654→            "collapse_probability": collapse_probability,
   655→   655→            "indicators_found": indicators,
   656→   656→            "collapsed": collapse_probability >= 0.5,
   657→   657→        }
   658→   658→
   659→   659→    @staticmethod
   660→   660→    def _char_entropy_score(text: str) -> float:
   661→   661→        """Shannon entropy of character distribution in *text*."""
   662→   662→        import math
   663→   663→        if not text:
   664→   664→            return 0.0
   665→   665→        freq: Dict[str, int] = {}
   666→   666→        for ch in text:
   667→   667→            freq[ch] = freq.get(ch, 0) + 1
   668→   668→        length = len(text)
   669→   669→        entropy = 0.0
   670→   670→        for count in freq.values():
   671→   671→            p = count / length
   672→   672→            if p > 0:
   673→   673→                entropy -= p * math.log2(p)
   674→   674→        return entropy

   769→   769→    """Extract AI-specific secrets from response text and headers.
   770→   770→
   771→   771→    Usage:
   772→   772→        se = SecretExtractor()
   773→   773→        secrets = se.extract(response_text, response_headers)
   774→   774→    """
   775→   775→
   776→   776→    def extract(self, response_text: str,
   777→   777→                response_headers: Optional[Dict[str, str]] = None) -> List[Dict[str, Any]]:
   778→   778→        """Scan *response_text* and *response_headers* for secrets."""
   779→   779→        response_headers = response_headers or {}
   780→   780→        found: List[Dict[str, Any]] = []
   781→   781→        seen: set = set()
   782→   782→
   783→   783→        # Scan text body
   784→   784→        self._scan(response_text, "response_body", found, seen)
   785→   785→
   786→   786→        # Scan header values
   787→   787→        header_blob = json.dumps(response_headers, separators=("", ":"))
   788→   788→        self._scan(header_blob, "response_header", found, seen)
   789→   789→
   790→   790→        # Deduplicate by value hash
   791→   791→        deduped: List[Dict[str, Any]] = []
   792→   792→        seen_hashes: set = set()
   793→   793→        for s in found:
   794→   794→            h = hashlib.sha256(s["value_masked"].encode()).hexdigest()[:16]
   795→   795→            if h not in seen_hashes:
   796→   796→                seen_hashes.add(h)
   797→   797→                deduped.append(s)
   798→   798→
   799→   799→        return deduped
   800→   800→
   801→   801→    def _scan(self, text: str, source: str, results: List[Dict[str, Any]],
   802→   802→              seen: set) -> None:
   803→   803→        """Run all patterns against *text*, appending to *results*."""
   804→   804→        for pattern, label, description in _SECRET_PATTERNS:
   805→   805→            try:
   806→   806→                compiled = re.compile(pattern)
   807→   807→            except re.error:
   808→   808→                continue
   809→   809→            for m in compiled.finditer(text):
   810→   810→                value = m.group(0)
   811→   811→                # Mask middle of sensitive values for safe reporting
   812→   812→                masked = self._mask(value)
   813→   813→                key = (label, masked)
   814→   814→                if key not in seen:
   815→   815→                    seen.add(key)
   816→   816→                    results.append({
   817→   817→                        "type": label,
   818→   818→                        "description": description,
   819→   819→                        "value_masked": masked,
   820→   820→                        "source": source,
   821→   821→                        "start": m.start(),
   822→   822→                        "end": m.end(),
   823→   823→                    })
   824→   824→
   825→   825→    @staticmethod
   826→   826→    def _mask(value: str) -> str:
   827→   827→        """Mask the middle portion of a secret for safe display."""
   828→   828→        if len(value) <= 12:
   829→   829→            return value[:4] + "****" + value[-4:] if len(value) >= 8 else "****"
   830→   830→        return value[:6] + "****" + value[-4:]
