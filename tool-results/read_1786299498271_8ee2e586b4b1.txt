   874→                        severity=sev, category="error_disclosure",

   890→# CAPABILITY 3: VERSION-RESPONSE CORRELATION
   891→# ════════════════════════════════════════════════════════════════════════
   892→
   893→def _correlate_versions(
   894→    target: str, base_url: str, timeout: int, verify_tls: bool,
   895→    probe_responses: List[Dict[str, Any]],
   896→) -> List[Finding]:
   897→    """Cross-reference observed versions with known vulnerability databases."""
   898→    findings: List[Finding] = []
   899→    host = _host_from_url(base_url)
   900→
   901→    # Gather version strings from headers and body
   902→    version_sources: List[Tuple[str, str]] = []
   903→
   904→    for r in probe_responses:
   905→        # Check Server header
   906→        server = r.get("headers", {}).get("Server", "")
   907→        if server:
   908→            version_sources.append(("Server", server))
   909→
   910→        # Check X-Powered-By
   911→        powered = r.get("headers", {}).get("X-Powered-By", "")
   912→        if powered:
   913→            version_sources.append(("X-Powered-By", powered))
   914→
   915→        # Scan body for version patterns
   916→        body = r.get("body", "")
   917→        body_version_patterns = [
   918→            ("body", m.group(0))
   919→            for p in [r"nginx/[\d.]+", r"Apache/[\d.]+", r"OpenSSH_[\d.]+",
   920→                      r"PHP/[\d.]+", r"Tomcat/[\d.]+", r"OpenSSL\s*[\d.]+",
   921→                      r"Spring\s*Framework\s*[\d.]+", r"Node\.?js\s*v?[\d.]+",
   922→                      r"Express\s*[\d.]+", r"Django\s*[\d.]+", r"Rails\s*[\d.]+",
   923→                      r"Laravel\s*v?[\d.]+", r"Next\.?js\s*[\d.]+"]
   924→            for m in [re.search(p, body, re.IGNORECASE)] if m
   925→        ]
   926→        version_sources.extend(body_version_patterns)
   927→
   928→    # Deduplicate version sources
   929→    seen: set = set()
   930→    for source, value in version_sources:
   931→        key = f"{source}:{value}"
   932→        if key in seen:
   933→            continue
   934→        seen.add(key)
   935→
   936→        # Try each product in VERSION_VULN_DB
   937→        for product, entries in VERSION_VULN_DB.items():
   938→            for entry in entries:
   939→                pattern = entry["version_pattern"]
   940→                m = re.search(pattern, value, re.IGNORECASE)
   941→                if not m:
   942→                    continue
   943→                version = m.group(1)
   944→                for vuln_range in entry["vulnerable_ranges"]:
   945→                    if _version_in_range(version, vuln_range):
   946→                        cve = vuln_range["cve"]
   947→                        desc = vuln_range["desc"]
   948→                        findings.append(Finding(
   949→                            title=f"Version vulnerability: {product} {version} -> {cve}",
   950→                            severity="critical", category="version_vulnerability",
   951→                            module="zero_day_hunter",
   952→                            description=f"{product} version {version} detected in {source} header/body. This version is vulnerable to {cve}: {desc}.",
   953→                            evidence=f"{source}: {value} | Pattern: {pattern} | Version: {version} | CVE: {cve}",
   954→                            asset=host, points_deducted=15,
   955→                            remediation=f"Upgrade {product} to a patched version beyond the affected range. See {cve} for specific fix details.",
   956→                            dread_score=_dread_score("critical"),
   957→                        ))
   958→
   959→    return findings
   960→

   963→# CAPABILITY 4: BEHAVIORAL ANOMALY SCORING
   964→# ════════════════════════════════════════════════════════════════════════
   965→
   966→def _score_behavioral_anomalies(
   967→    target: str, base_url: str, timeout: int, verify_tls: bool,
   968→) -> List[Finding]:
   969→    """Score endpoints on inconsistent caching, timing, parameter sensitivity."""
   970→    findings: List[Finding] = []
   971→    host = _host_from_url(base_url)
   972→
   973→    # Test caching consistency: send same request twice, compare headers
   974→    cache_test_paths = ["/", "/api/health", "/api/v1"]
   975→    for path in cache_test_paths:
   976→        url = base_url.rstrip("/") + path
   977→        r1 = _safe_probe(url, timeout=timeout, verify_tls=verify_tls)
   978→        r2 = _safe_probe(url, timeout=timeout, verify_tls=verify_tls)
   979→
   980→        h1 = r1.get("headers", {})
   981→        h2 = r2.get("headers", {})
   982→
   983→        # Check Cache-Control consistency
   984→        cc1 = h1.get("Cache-Control", "")
   985→        cc2 = h2.get("Cache-Control", "")
   986→        if cc1 != cc2:
   987→            findings.append(Finding(
   988→                title=f"Inconsistent Cache-Control header on {path}",
   989→                severity="low", category="behavioral_anomaly",
   990→                module="zero_day_hunter",
   991→                description=f"Cache-Control header changed between identical requests: '{cc1}' -> '{cc2}'. May indicate load-balanced backends with inconsistent configurations.",
   992→                evidence=f"Request 1: Cache-Control={cc1} | Request 2: Cache-Control={cc2}",
   993→                asset=host, points_deducted=3,
   994→                remediation="Ensure all backend servers share identical cache configuration. Inconsistent caching can lead to cache poisoning.",
   995→                dread_score=_dread_score("low"),
   996→            ))
   997→
   998→        # Check ETag consistency
   999→        etag1 = h1.get("ETag", "")
  1000→        etag2 = h2.get("ETag", "")
  1001→        if etag1 and etag2 and etag1 != etag2 and r1["status"] == 200 and r2["status"] == 200:
  1002→            findings.append(Finding(
  1003→                title=f"ETag instability on {path}",
  1004→                severity="medium", category="behavioral_anomaly",
  1005→                module="zero_day_hunter",
  1006→                description=f"ETag changed between identical requests: '{etag1}' -> '{etag2}'. May reveal weak ETag generation or load balancer inconsistency.",
  1007→                evidence=f"Request 1: ETag={etag1} | Request 2: ETag={etag2}",
  1008→                asset=host, points_deducted=5,
  1009→                remediation="Use strong, deterministic ETags. ETag instability can facilitate cache-based attacks.",
  1010→                dread_score=_dread_score("medium"),
  1011→            ))
  1012→
  1013→    # Parameter sensitivity: test how different params affect response
  1014→    param_test_url = base_url.rstrip("/") + "/api/search"
  1015→    param_variants = [
  1016→        {},
  1017→        {"q": "test"},
  1018→        {"q": "' OR 1=1"},
  1019→        {"q": "<script>alert(1)</script>"},
  1020→        {"q": "../../../etc/passwd"},
  1021→        {"page": "1"},
  1022→        {"page": "-1"},
  1023→        {"page": "999999"},
  1024→        {"sort": "id"},
  1025→        {"sort": "id; DROP TABLE users"},
  1026→        {"_debug": "1"},
  1027→        {"admin": "true"},
  1028→    ]
  1029→
  1030→    baseline_sizes: List[int] = []
  1031→    baseline_statuses: List[int] = []
  1032→    param_responses: List[Dict[str, Any]] = []
  1033→
  1034→    for params in param_variants:
  1035→        qs = urllib.parse.urlencode(params) if params else ""
  1036→        url = f"{param_test_url}?{qs}" if qs else param_test_url
  1037→        t0 = time.monotonic()
  1038→        resp = _safe_probe(url, timeout=timeout, verify_tls=verify_tls)
  1039→        elapsed = (time.monotonic() - t0) * 1000
  1040→        resp["params"] = params
  1041→        resp["elapsed_ms"] = round(elapsed, 1)
  1042→        resp["body_length"] = len(resp.get("body", ""))
  1043→        param_responses.append(resp)
  1044→
  1045→        if not params:
  1046→            baseline_sizes.append(resp["body_length"])
  1047→            baseline_statuses.append(resp["status"])
  1048→
  1049→    if baseline_sizes:
  1050→        baseline_mean = statistics.mean(baseline_sizes)
  1051→        for r in param_responses:
  1052→            params = r["params"]
  1053→            if not params:
  1054→                continue
  1055→            bl = r["body_length"]
  1056→            status = r["status"]
  1057→            # Check for significant size deviation from baseline
  1058→            if baseline_mean > 0 and bl > 0:
  1059→                ratio = max(bl, baseline_mean) / max(min(bl, baseline_mean), 1)
  1060→                if ratio > ANOMALY_THRESHOLDS["consistency"]["max_size_ratio"]:
  1061→                    findings.append(Finding(
  1062→                        title=f"Parameter sensitivity anomaly: {params}",
  1063→                        severity="high", category="behavioral_anomaly",
  1064→                        module="zero_day_hunter",
  1065→                        description=f"Parameter set {params} caused a {ratio:.1f}x response size change ({baseline_mean:.0f} -> {bl} bytes). This magnitude of change may indicate injection vulnerability or unintended data exposure.",
  1066→                        evidence=f"Params: {params} | Baseline: {baseline_mean:.0f} bytes -> Response: {bl} bytes | Status: {status} | Time: {r['elapsed_ms']}ms",
  1067→                        asset=host, points_deducted=10,
  1068→                        remediation="Audit parameter handling for this endpoint. Large size changes from crafted inputs may indicate SQL injection, path traversal, or information disclosure.",
  1069→                        dread_score=_dread_score("high"),
  1070→                    ))
  1071→
  1072→            # Check for status code change from 200 baseline to error
  1073→            if baseline_statuses and baseline_statuses[0] == 200 and status >= 400:
  1074→                param_str = str(params)
  1075→                findings.append(Finding(
  1076→                    title=f"Parameter-induced error: {params}",
  1077→                    severity="medium", category="behavioral_anomaly",
  1078→                    module="zero_day_hunter",
  1079→                    description=f"Parameters {params} changed response from 200 to {status}. Error may reveal backend validation logic or input handling defects.",
  1080→                    evidence=f"Params: {params} | Status: {status} | Body: {r.get('body', '')[:200]}",
  1081→                    asset=host, points_deducted=5,
  1082→                    remediation="Implement robust input validation that returns consistent error responses without leaking implementation details.",
  1083→                    dread_score=_dread_score("medium"),
  1084→                ))
  1085→
  1086→    # Timing sensitivity: compare response times across parameter variants
  1087→    timings = [r["elapsed_ms"] for r in param_responses if r["elapsed_ms"] > 0]
  1088→    if len(timings) >= 3:
  1089→        mean_t = statistics.mean(timings)
  1090→        for r in param_responses:
  1091→            t = r["elapsed_ms"]
  1092→            if t > ANOMALY_THRESHOLDS["timing"]["time_based_blind_threshold_ms"]:
  1093→                findings.append(Finding(
  1094→                    title=f"Timing sensitivity: {r['params']} took {t:.0f}ms",
  1095→                    severity="high", category="behavioral_anomaly",
  1096→                    module="zero_day_hunter",
  1097→                    description=f"Parameter set {r['params']} caused {t:.0f}ms response (baseline ~{mean_t:.0f}ms). Timing delta of {t - mean_t:.0f}ms may indicate time-based blind injection vulnerability.",
  1098→                    evidence=f"Params: {r['params']} | Time: {t:.0f}ms | Baseline: ~{mean_t:.0f}ms | Status: {r['status']}",
  1099→                    asset=host, points_deducted=10,
  1100→                    remediation="Investigate timing differences for potential time-based blind SQL injection or command injection vulnerabilities.",
  1101→                    dread_score=_dread_score("high"),
  1102→                ))
  1103→
  1104→    return findings

  1108→# CAPABILITY 5: FUZZING RESULT ANALYSIS

  1110→
  1111→def _analyze_fuzzing_results(
  1112→    target: str, base_url: str, timeout: int, verify_tls: bool,
  1113→) -> List[Finding]:
  1114→    """Send crafted inputs and detect anomalous responses."""
  1115→    findings: List[Finding] = []
  1116→    host = _host_from_url(base_url)
  1117→
  1118→    # Establish baseline from clean request
  1119→    baseline_url = base_url.rstrip("/") + "/api/search"
  1120→    baseline_resp = _safe_probe(baseline_url, timeout=timeout, verify_tls=verify_tls)
  1121→    baseline_status = baseline_resp["status"]
  1122→    baseline_size = len(baseline_resp.get("body", ""))
  1123→    baseline_headers = set(baseline_resp.get("headers", {}).keys())
  1124→
  1125→    # Also establish baseline for the root endpoint
  1126→    root_resp = _safe_probe(base_url.rstrip("/") + "/", timeout=timeout, verify_tls=verify_tls)
  1127→    root_status = root_resp["status"]
  1128→    root_size = len(root_resp.get("body", ""))
  1129→
  1130→    fuzz_targets = [
  1131→        ("/api/search", "GET", None),
  1132→        ("/", "POST", None),
  1133→        ("/api/login", "POST", None),
  1134→        ("/api/user", "GET", None),
  1135→    ]
  1136→
  1137→    for target_path, method, _ in fuzz_targets:
  1138→        base_target = base_url.rstrip("/") + target_path
  1139→
  1140→        for payload in FUZZ_PAYLOADS:
  1141→            category = payload["category"]
  1142→            value = payload["value"]
  1143→            label = payload["label"]
  1144→
  1145→            if method == "GET":
  1146→                url = f"{base_target}?q={urllib.parse.quote(value)}"
  1147→                resp = _safe_probe(url, timeout=timeout, verify_tls=verify_tls)
  1148→            else:
  1149→                body = urllib.parse.urlencode({"input": value}).encode() if category != "xml" else value.encode("utf-8", errors="replace")
  1150→                extra_headers = {"Content-Type": "application/json"} if category in ("json", "ssti") else ("Content-Type": "application/xml" if category == "xml" else None)
  1151→                resp = _safe_probe(base_target, method=method, body=body, headers=extra_headers, timeout=timeout, verify_tls=verify_tls)
  1152→
  1153→            resp_body = resp.get("body", "")
  1154→            resp_status = resp["status"]
  1155→            resp_size = len(resp_body)
  1156→
  1157→            # Check for reflected payload (XSS/injection indication)
  1158→            if value in resp_body and len(value) > 5 and resp_status == 200:
  1159→                findings.append(Finding(
  1160→                    title=f"Payload reflected: {label}",
  1161→                    severity="high", category="fuzzing_anomaly",
  1162→                    module="zero_day_hunter",
  1163→                    description=f"Payload for {category} was reflected in the response body without sanitization. This strongly suggests the endpoint is vulnerable to {category}.",
  1164→                    evidence=f"{method} {target_path} with payload -> {resp_status} ({resp_size} bytes) | Payload found in response body",
  1165→                    asset=host, points_deducted=10,
  1166→                    remediation=f"Implement proper input validation and output encoding for {category} payloads. Use parameterized queries for SQL, template auto-escaping for XSS/SSTI.",
  1167→                    dread_score=_dread_score("high"),
  1168→                ))
  1169→                continue
  1170→
  1171→            # Check for SSTI arithmetic evaluation (e.g., {{7*7}} -> 49)
  1172→            if category == "ssti":
  1173→                if re.search(r"49", resp_body) and "{{7*7}}" not in resp_body:
  1174→                    findings.append(Finding(
  1175→                        title=f"SSTI confirmed: {label}",
  1176→                        severity="critical", category="fuzzing_anomaly",
  1177→                        module="zero_day_hunter",
  1178→                        description=f"Server-Side Template Injection confirmed — arithmetic expression {{7*7}} was evaluated to 49. This enables remote code execution.",
  1179→                        evidence=f"{method} {target_path} with SSTI payload -> {resp_status} | '49' found in response, original '{{7*7}}' not present",
  1180→                        asset=host, points_deducted=15,
  1181→                        remediation="Upgrade template engine, disable template rendering of user input, and use sandboxed template environments.",
  1182→                        dread_score=_dread_score("critical"),
  1183→                    ))
  1184→                elif re.search(r"49", resp_body) and "${7*7}" not in resp_body and "$" in value:
  1185→                    findings.append(Finding(
  1186→                        title=f"SSTI confirmed: {label}",
  1187→                        severity="critical", category="fuzzing_anomaly",
  1188→                        module="zero_day_hunter",
  1189→                        description=f"Server-Side Template Injection confirmed — ${7*7} was evaluated to 49. This enables remote code execution.",
  1190→                        evidence=f"{method} {target_path} -> {resp_status} | Arithmetic evaluation detected in response",
  1191→                        asset=host, points_deducted=15,
  1192→                        remediation="Disable template expression evaluation on user input. Use sandboxed rendering.",
  1193→                        dread_score=_dread_score("critical"),
  1194→                    ))
  1195→
  1196→            # Check for database error triggered by fuzzing
  1197→            db_error_found = False
  1198→            for sig in ERROR_SIGNATURES.get("database_errors", []):
  1199→                if re.search(sig["pattern"], resp_body, re.IGNORECASE):
  1200→                    db_error_found = True
  1201→                    findings.append(Finding(
  1202→                        title=f"Fuzz-triggered DB error: {label}",
  1203→                        severity="critical", category="fuzzing_anomaly",
  1204→                        module="zero_day_hunter",
  1205→                        description=f"{label} triggered a database error ({sig['name']}): {sig['description']}. This confirms SQL injection surface.",
  1206→                        evidence=f"{method} {target_path} -> {resp_status} | DB error: {sig['name']} in response",
  1207→                        asset=host, points_deducted=15,
  1208→                        remediation="Use parameterized queries/prepared statements. Never concatenate user input into SQL queries.",
  1209→                        dread_score=_dread_score("critical"),
  1210→                    ))
  1211→                    break
  1212→            if db_error_found:
  1213→                continue
  1214→
  1215→            # Check for crash-like behavior (0 status, very slow, or 500)
  1216→            if resp_status == 0 or resp_status >= 500:
  1217→                if resp_status == 500 and baseline_status != 500:
  1218→                    findings.append(Finding(
  1219→                        title=f"Fuzz-triggered 500 error: {label}",
  1220→                        severity="high", category="fuzzing_anomaly",
  1221→                        module="zero_day_hunter",
  1222→                        description=f"{label} caused an internal server error (500) where baseline returned {baseline_status}. May indicate unhandled exception, memory corruption, or incomplete input validation.",
  1223→                        evidence=f"{method} {target_path} with {category} payload -> 500 | Baseline: {baseline_status}",
  1224→                        asset=host, points_deducted=10,
  1225→                        remediation="Implement robust error handling and input validation. 500 errors from crafted inputs suggest the application does not properly sanitize or validate input.",
  1226→                        dread_score=_dread_score("high"),
  1227→                    ))
  1228→
  1229→            # Check for path traversal success indicators
  1230→            if category == "path_traversal":
  1231→                traversal_indicators = ["root:", "/bin/bash", "/bin/sh", "nobody:", "www-data:"]
  1232→                if any(ind in resp_body for ind in traversal_indicators):
  1233→                    findings.append(Finding(
  1234→                        title=f"Path traversal confirmed: {label}",
  1235→                        severity="critical", category="fuzzing_anomaly",
  1236→                        module="zero_day_hunter",
  1237→                        description=f"Path traversal payload successfully read file contents — file content markers detected in response. Arbitrary file read confirmed.",
  1238→                        evidence=f"{method} {target_path} -> {resp_status} | File content markers found in {resp_size} byte response",
  1239→                        asset=host, points_deducted=15,
  1240→                        remediation="Implement path canonicalization and whitelist allowed directories. Never pass user input directly to file system operations.",
  1241→                        dread_score=_dread_score("critical"),
  1242→                    ))
  1243→
  1244→            # Check for CRLF injection success (injected header in response)
  1245→            if category == "crlf" and "X-Injected" in str(resp.get("headers", {})):
  1246→                findings.append(Finding(
  1247→                    title="CRLF injection confirmed: header injection",
  1248→                    severity="high", category="fuzzing_anomaly",
  1249→                    module="zero_day_hunter",
  1250→                    description="CRLF payload successfully injected a custom HTTP response header. This can lead to response splitting, XSS, and cache poisoning.",
  1251→                    evidence=f"{method} {target_path} -> {resp_status} | X-Injected header found in response",
  1252→                    asset=host, points_deducted=10,
  1253→                    remediation="Sanitize newlines and carriage returns from user input before including in HTTP responses.",
  1254→                    dread_score=_dread_score("high"),
  1255→                ))
  1256→
  1257→    return findings

  1261→# CAPABILITY 6: HEADER ANOMALY DETECTION

  1264→def _detect_header_anomalies(

  1294→                asset=host, points_deducted=_severity_points(sev),

  1296→                dread_score=_dread_score(sev),
  1297→            ))
  1298→
  1299→    # Check for missing security headers
  1300→    for sec_header in ANOMALY_THRESHOLDS["headers"]["missing_security_headers"]:

  1302→            findings.append(Finding(
  1303→                title=f"Missing security header: {sec_header}",
  1304→                severity="medium", category="header_anomaly",
  1305→                module="zero_day_hunter",
  1306→                description=f"Security header '{sec_header}' is missing from all responses. This increases risk of clickjacking, MIME sniffing, and other client-side attacks.",
  1307→                evidence=f"Header '{sec_header}' not found in any of {len(probe_responses)} responses",
  1308→                asset=host, points_deducted=5,
  1309→                remediation=f"Add the {sec_header} header to all responses. Configure the web server or application framework to include it by default.",
  1310→                dread_score=_dread_score("medium"),
  1311→            ))
  1312→
  1313→    # Check for headers that reveal debug/dev mode
  1314→    debug_header_patterns = [
  1315→        (r"X-Debug", "Debug mode header detected"),
  1316→        (r"Debug", "Debug-related header present"),
  1317→        (r"X-Profiler", "Application profiler header exposed"),
  1318→        (r"X-Cake", "CakePHP debug header exposed"),
  1319→        (r"X-Laravel", "Laravel debug header exposed"),
  1320→        (r"X-RateLimit-Reset", "Rate limit header reveals API throttling config"),
  1321→        (r"X-Request-Id", "Request tracing ID reveals distributed architecture"),
  1322→    ]
  1323→    for pattern, desc in debug_header_patterns:
  1324→        for h in all_headers:
  1325→            if re.search(pattern, h, re.IGNORECASE):
  1326→                values = list(set(all_headers[h]))
  1327→                sev = "high" if "debug" in h.lower() or "profiler" in h.lower() else "low"
  1328→                findings.append(Finding(
  1329→                    title=f"Development artifact header: {h}",
  1330→                    severity=sev, category="header_anomaly",
  1331→                    module="zero_day_hunter",
  1332→                    description=f"{desc}. Value(s): {values}. Development headers in production reveal internal architecture and attack surface.",
  1333→                    evidence=f"Header: {h} | Values: {values}",
  1334→                    asset=host, points_deducted=_severity_points(sev),
  1335→                    remediation=f"Remove development/debug headers before deploying to production.",
  1336→                    dread_score=_dread_score(sev),
  1337→                ))
  1338→
  1339→    # Check for inconsistent headers across identical requests
  1340→    root_responses = [r for r in probe_responses if r.get("path") == "/"]
  1341→    if len(root_responses) >= 2:
  1342→        h1_keys = set(root_responses[0].get("headers", {}).keys())
  1343→        h2_keys = set(root_responses[1].get("headers", {}).keys())
  1344→        if h1_keys != h2_keys:
  1345→            only_in_first = h1_keys - h2_keys
  1346→            only_in_second = h2_keys - h1_keys
  1347→            findings.append(Finding(
  1348→                title="Inconsistent headers across identical requests",
  1349→                severity="medium", category="header_anomaly",
  1350→                module="zero_day_hunter",
  1351→                description=f"Headers differ between identical requests to /. Only in request 1: {only_in_first}. Only in request 2: {only_in_second}. May indicate load-balanced heterogeneous backends.",
  1352→                evidence=f"Headers only in R1: {only_in_first} | Only in R2: {only_in_second}",
  1353→                asset=host, points_deducted=5,
  1354→                remediation="Ensure all backend servers are configured identically. Header inconsistency can reveal infrastructure topology.",
  1355→                dread_score=_dread_score("medium"),
  1356→            ))
  1357→
  1358→    # Check for overly verbose Server header
  1359→    for h_name, h_values in all_headers.items():
  1360→        if h_name.lower() == "server":
  1361→            for v in h_values:
  1362→                if re.search(r"[\d]+\.[\d]+", v):
  1363→                    findings.append(Finding(
  1364→                        title=f"Version disclosure in Server header",
  1365→                        severity="medium", category="header_anomaly",
  1366→                        module="zero_day_hunter",
  1367→                        description=f"Server header reveals version information: '{v}'. Version disclosure enables targeted exploit selection from vulnerability databases.",
  1368→                        evidence=f"Server: {v}",
  1369→                        asset=host, points_deducted=5,
  1370→                        remediation="Configure the server to suppress version information in the Server header.",
  1371→                        dread_score=_dread_score("medium"),
  1372→                    ))
  1373→
  1374→    return findings

  1378→# CAPABILITY 7: ENDPOINT SENSITIVITY MAPPING
  1379→# ════════════════════════════════════════════════════════════════════════
  1380→
  1381→def _map_endpoint_sensitivity(
  1382→    target: str, base_url: str, timeout: int, verify_tls: bool,
  1383→) -> List[Finding]:
  1384→    """Test paths with/without authentication to find broken access control."""
  1385→    findings: List[Finding] = []
  1386→    host = _host_from_url(base_url)
  1387→
  1388→    # Test each sensitivity path without authentication
  1389→    for entry in SENSITIVITY_PATHS:
  1390→        path = entry["path"]
  1391→        expected = entry["expected_without_auth"]
  1392→        sensitivity = entry["sensitivity"]
  1393→        label = entry["label"]
  1394→
  1395→        url = base_url.rstrip("/") + path
  1396→        resp = _safe_probe(url, timeout=timeout, verify_tls=verify_tls)
  1397→        status = resp["status"]
  1398→        body = resp.get("body", "")
  1399→        body_len = len(body)
  1400→
  1401→        if status == 0:
  1402→            continue
  1403→
  1404→        # Check for broken access control: sensitive path accessible without auth
  1405→        if expected in (401, 403) and status == 200:
  1406→            findings.append(Finding(
  1407→                title=f"Broken access control: {label} ({path})",
  1408→                severity=sensitivity, category="access_control",
  1409→                module="zero_day_hunter",
  1410→                description=f"{label} at {path} returned HTTP 200 without authentication. Expected {expected}. This is a broken access control vulnerability — sensitive functionality is exposed to unauthenticated users.",
  1411→                evidence=f"GET {path} -> 200 ({body_len} bytes) | Expected: {expected} | Sensitivity: {sensitivity}",
  1412→                asset=host, points_deducted=_severity_points(sensitivity),
  1413→                remediation="Implement proper authentication and authorization checks on this endpoint. Deny access by default and require explicit authorization.",
  1414→                dread_score=_dread_score(sensitivity),
  1415→            ))
  1416→        elif expected in (401, 403) and status not in (401, 403, 0) and status < 500:
  1417→            # Unexpected non-auth status on protected path (e.g., 302 redirect without auth)
  1418→            if status in (301, 302, 303, 307, 308):
  1419→                location = resp.get("headers", {}).get("Location", "")
  1420→                findings.append(Finding(
  1421→                    title=f"Access control redirect: {label} ({path})",
  1422→                    severity="medium", category="access_control",
  1423→                    module="zero_day_hunter",
  1424→                    description=f"{label} at {path} returned {status} redirect to '{location}' instead of {expected}. Redirect-based access control may be bypassable by following redirects without authentication.",
  1425→                    evidence=f"GET {path} -> {status} Location: {location} | Expected: {expected}",
  1426→                    asset=host, points_deducted=5,
  1427→                    remediation="Use proper 401/403 status codes for unauthorized access instead of redirects. Validate authentication before redirecting.",
  1428→                    dread_score=_dread_score("medium"),
  1429→                ))
  1430→            else:
  1431→                findings.append(Finding(
  1432→                    title=f"Unexpected status on protected path: {label} ({path})",
  1433→                    severity="medium", category="access_control",
  1434→                    module="zero_day_hunter",
  1435→                    description=f"{label} at {path} returned {status} instead of expected {expected}. Protected endpoint returned unexpected status code.",
  1436→                    evidence=f"GET {path} -> {status} ({body_len} bytes) | Expected: {expected}",
  1437→                    asset=host, points_deducted=5,
  1438→                    remediation="Review access control configuration for this endpoint. Ensure consistent authorization behavior.",
  1439→                    dread_score=_dread_score("medium"),
  1440→                ))
  1441→
  1442→        # Check for information disclosure on error pages from protected paths
  1443→        if status >= 400 and body_len > 500:
  1444→            for sig_category, signatures in ERROR_SIGNATURES.items():
  1445→                for sig in signatures:
  1446→                    if re.search(sig["pattern"], body, re.IGNORECASE | re.DOTALL):
  1447→                        findings.append(Finding(
  1448→                            title=f"Info disclosure on protected path: {label}",
  1449→                            severity=sig["severity"], category="access_control",
  1450→                            module="zero_day_hunter",
  1451→                            description=f"{label} at {path} returned error ({status}) with {sig['name']} disclosure. Error pages on protected endpoints should not reveal internal details.",
  1452→                            evidence=f"GET {path} -> {status} | {sig['name']}: {sig['description']}",
  1453→                            asset=host, points_deducted=_severity_points(sig["severity"]),
  1454→                            remediation="Implement generic error pages for protected endpoints that do not reveal internal implementation details.",
  1455→                            dread_score=_dread_score(sig["severity"]),
  1456→                        ))
  1457→
  1458→    # Test HTTP method variations on sensitive paths
  1459→    method_test_paths = ["/admin", "/api/config", "/api/internal", "/api/users"]
  1460→    for path in method_test_paths:
  1461→        url = base_url.rstrip("/") + path
  1462→        for method in ["PUT", "DELETE", "PATCH", "OPTIONS"]:
  1463→            resp = _safe_probe(url, method=method, timeout=timeout, verify_tls=verify_tls)
  1464→            status = resp["status"]
  1465→            # If non-standard methods return 200 on protected paths, access control may be broken
  1466→            if status == 200 and method in ("PUT", "DELETE", "PATCH"):
  1467→                findings.append(Finding(
  1468→                    title=f"Method-based access control bypass: {method} {path}",
  1469→                    severity="critical", category="access_control",
  1470→                    module="zero_day_hunter",
  1471→                    description=f"{method} request to {path} returned HTTP 200 without authentication. Access control may only be applied to GET requests, allowing state-changing operations without authorization.",
  1472→                    evidence=f"{method} {path} -> 200 ({len(resp.get('body', ''))} bytes)",
  1473→                    asset=host, points_deducted=15,
  1474→                    remediation="Apply authentication and authorization to ALL HTTP methods, not just GET. Use framework-level middleware for consistent access control.",
  1475→                    dread_score=_dread_score("critical"),
  1476→                ))
  1477→            elif status == 200 and method == "OPTIONS":
  1478→                allow = resp.get("headers", {}).get("Allow", "")
  1479→                if allow:
  1480→                    findings.append(Finding(
  1481→                        title=f"OPTIONS method reveals allowed methods: {path}",
  1482→                        severity="low", category="access_control",
  1483→                        module="zero_day_hunter",
  1484→                        description=f"OPTIONS request to {path} reveals allowed methods: '{allow}'. This information helps an attacker target specific HTTP methods.",
  1485→                        evidence=f"OPTIONS {path} -> 200 | Allow: {allow}",
  1486→                        asset=host, points_deducted=3,
  1487→                        remediation="Restrict OPTIONS responses or remove the Allow header from unauthorized requests.",
  1488→                        dread_score=_dread_score("low"),
  1489→                    ))
  1490→
  1491→    return findings
  1492→

  1498→def run_zero_day_hunter(
  1499→    target: str,
  1500→    base_url: str,
  1501→    timeout: int = 8,
  1502→    verify_tls: bool = True,
  1503→) -> List[Finding]:
  1504→    """Run all zero-day hunter analysis capabilities against a target.
  1505→
  1506→    Performs seven-pass analysis:
  1507→      1. Response Anomaly Detection
  1508→      2. Error Message Analysis
  1509→      3. Version-Response Correlation
  1510→      4. Behavioral Anomaly Scoring
  1511→      5. Fuzzing Result Analysis
  1512→      6. Header Anomaly Detection
  1513→      7. Endpoint Sensitivity Mapping
  1514→
  1515→    Args:
  1516→        target: Target hostname or IP address.
  1517→        base_url: Base URL for the target (e.g., https://example.com).
  1518→        timeout: HTTP request timeout in seconds (default: 8).
  1519→        verify_tls: Whether to verify TLS certificates (default: True).
  1520→
  1521→    Returns:
  1522→        List of Finding objects documenting detected anomalies.
  1523→    """
  1524→    all_findings: List[Finding] = []
  1525→
  1526→    # ── Pass 1: Response Anomaly Detection ──
  1527→    anomaly_findings, probe_responses = _detect_response_anomalies(
  1528→        target, base_url, timeout, verify_tls,
  1529→    )
  1530→    all_findings.extend(anomaly_findings)
  1531→
  1532→    # ── Pass 2: Error Message Analysis ──
  1533→    error_findings = _analyze_error_messages(
  1534→        target, base_url, timeout, verify_tls,
  1535→    )
  1536→    all_findings.extend(error_findings)
  1537→
  1538→    # ── Pass 3: Version-Response Correlation ──
  1539→    version_findings = _correlate_versions(
  1540→        target, base_url, timeout, verify_tls, probe_responses,
  1541→    )
  1542→    all_findings.extend(version_findings)
  1543→
  1544→    # ── Pass 4: Behavioral Anomaly Scoring ──
  1545→    behavioral_findings = _score_behavioral_anomalies(
  1546→        target, base_url, timeout, verify_tls,
  1547→    )
  1548→    all_findings.extend(behavioral_findings)
  1549→
  1550→    # ── Pass 5: Fuzzing Result Analysis ──
  1551→    fuzzing_findings = _analyze_fuzzing_results(
  1552→        target, base_url, timeout, verify_tls,
  1553→    )
  1554→    all_findings.extend(fuzzing_findings)
  1555→
  1556→    # ── Pass 6: Header Anomaly Detection ──
  1557→    header_findings = _detect_header_anomalies(
  1558→        target, base_url, timeout, verify_tls, probe_responses,
  1559→    )
  1560→    all_findings.extend(header_findings)
  1561→
  1562→    # ── Pass 7: Endpoint Sensitivity Mapping ──
  1563→    sensitivity_findings = _map_endpoint_sensitivity(
  1564→        target, base_url, timeout, verify_tls,
  1565→    )
  1566→    all_findings.extend(sensitivity_findings)
  1567→
  1568→    return all_findings
