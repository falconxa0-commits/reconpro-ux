    53→class ToolResult:
    54→    """Result of a tool execution."""
    55→
    56→    def __init__(self, output: str, success: bool = True, data: Any = None):
    57→        self.output = output
    58→        self.success = success
    59→        self.data = data
    60→
    61→    def to_dict(self) -> Dict:
    62→        return {"output": self.output, "success": self.success, "data": self.data}

    64→
    65→class Tool:
    66→    """A callable tool the agent can use."""
    67→
    68→    def __init__(
    69→        self,
    70→        name: str,
    71→        description: str,
    72→        parameters: Dict[str, Any],
    73→        execute: Callable[..., ToolResult],
    74→        category: str = "scan",
    75→        dangerous: bool = False,
    76→    ):
    77→        self.name = name
    78→        self.description = description
    79→        self.parameters = parameters
    80→        self.execute = execute
    81→        self.category = category
    82→        self.dangerous = dangerous
    83→
    84→    def to_openai_tool(self) -> Dict:
    85→        return {
    86→            "type": "function",
    87→            "function": {
    88→                "name": self.name,
    89→                "description": self.description,
    90→                "parameters": {
    91→                    "type": "object",
    92→                    "properties": self.parameters,
    93→                    "required": list(self.parameters.keys()),
    94→                },
    95→            },
    96→        }
    97→
    98→    def to_anthropic_tool(self) -> Dict:
    99→        return {
   100→            "name": self.name,
   101→            "description": self.description,
   102→            "input_schema": {
   103→                "type": "object",
   104→                "properties": self.parameters,
   105→                "required": list(self.parameters.keys()),
   106→            },
   107→        }

   109→
   110→class ToolRegistry:
   111→    """Registry of all available tools."""
   112→
   113→    def __init__(self):
   114→        self.tools: Dict[str, Tool] = {}
   115→
   116→    def register(self, tool: Tool) -> None:
   117→        self.tools[tool.name] = tool
   118→
   119→    def get(self, name: str) -> Optional[Tool]:
   120→        return self.tools.get(name)
   121→
   122→    def execute(self, name: str, **kwargs) -> ToolResult:
   123→        tool = self.tools.get(name)
   124→        if not tool:
   125→            return ToolResult(f"Tool '{name}' not found.", success=False)
   126→        try:
   127→            return tool.execute(**kwargs)
   128→        except Exception as e:
   129→            return ToolResult(f"Tool error: {e}", success=False)
   130→
   131→    def list_tools(self) -> List[Dict]:
   132→        return [
   133→            {"name": t.name, "description": t.description, "category": t.category}
   134→            for t in self.tools.values()
   135→        ]
   136→
   137→    def to_openai_tools(self) -> List[Dict]:
   138→        return [t.to_openai_tool() for t in self.tools.values()]
   139→
   140→    def to_anthropic_tools(self) -> List[Dict]:
   141→        return [t.to_anthropic_tool() for t in self.tools.values()]

   148→
   149→class AgentMemory:
   150→    """Persistent memory for the Nexus agent across sessions."""
   151→
   152→    def __init__(self):
   153→        MEMORY_DIR.mkdir(parents=True, exist_ok=True)
   154→        self._data = self._load()
   155→
   156→    def _load(self) -> Dict:
   157→        if CONTEXT_FILE.exists():
   158→            try:
   159→                return json.loads(CONTEXT_FILE.read_text())
   160→            except Exception:
   161→                pass
   162→        return {
   163→            "sessions": [],
   164→            "target_contexts": {},
   165→            "learned_patterns": [],
   166→            "created": datetime.now(timezone.utc).isoformat(),
   167→        }
   168→
   169→    def _save(self) -> None:
   170→        # Keep only the last 50 sessions
   171→        if len(self._data.get("sessions", [])) > 50:
   172→            self._data["sessions"] = self._data["sessions"][-50:]
   173→        try:
   174→            CONTEXT_FILE.write_text(json.dumps(self._data, indent=2, default=str))
   175→        except OSError:
   176→            pass
   177→
   178→    def save_session(
   179→        self,
   180→        goal: str,
   181→        target: str,
   182→        plan: List[Dict],
   183→        tool_results: List[Dict],
   184→        summary: str,
   185→    ) -> None:
   186→        session = {
   187→            "goal": goal,
   188→            "target": target,
   189→            "plan": plan,
   190→            "tool_results": tool_results,
   191→            "summary": summary,
   192→            "timestamp": datetime.now(timezone.utc).isoformat(),
   193→        }
   194→        self._data.setdefault("sessions", []).append(session)
   195→        # Update target context
   196→        self._data.setdefault("target_contexts", {})[target] = {
   197→            "last_score": self._extract_score(tool_results),
   198→            "last_grade": self._extract_grade(tool_results),
   199→            "last_run": session["timestamp"],
   200→            "finding_count": self._extract_finding_count(tool_results),
   201→            "known_issues": self._extract_issues(tool_results),
   202→        }
   203→        self._save()
   204→
   205→    def get_target_context(self, target: str) -> Optional[Dict]:
   206→        return self._data.get("target_contexts", {}).get(target)
   207→
   208→    def get_recent_sessions(self, limit: int = 5) -> List[Dict]:
   209→        return self._data.get("sessions", [])[-limit:]
   210→
   211→    @staticmethod
   212→    def _extract_score(results: List[Dict]) -> Optional[int]:
   213→        for r in results:
   214→            d = r.get("data")
   215→            if isinstance(d, dict) and "total_score" in d:
   216→                return d["total_score"]
   217→        return None
   218→
   219→    @staticmethod
   220→    def _extract_grade(results: List[Dict]) -> Optional[str]:
   221→        for r in results:
   222→            d = r.get("data")
   223→            if isinstance(d, dict) and "grade" in d:
   224→                return d["grade"]
   225→        return None
   226→
   227→    @staticmethod
   228→    def _extract_finding_count(results: List[Dict]) -> int:
   229→        total = 0
   230→        for r in results:
   231→            d = r.get("data")
   232→            if isinstance(d, dict) and "findings" in d:
   233→                total += len(d["findings"])
   234→        return total
   235→
   236→    @staticmethod
   237→    def _extract_issues(results: List[Dict]) -> List[str]:
   238→        issues = []
   239→        for r in results:
   240→            d = r.get("data")
   241→            if isinstance(d, dict) and "findings" in d:
   242→                for f in d["findings"]:
   243→                    if isinstance(f, dict) and f.get("severity") in ("critical", "high"):
   244→                        issues.append(f.get("title", ""))
   245→        return issues[:20]

   430→    """Build or query the security knowledge graph."""
   431→    from .knowledge_graph import SecurityKnowledgeGraph
   432→    if not target:
   433→        return ToolResult("No target specified.", success=False)
   434→    kg = SecurityKnowledgeGraph()
   435→    if action == "build":
   436→        from .history import get_latest
   437→        latest = get_latest(target=target)
   438→        if not latest:
   439→            latest = get_latest()
   440→        if latest:
   441→            kg.add_scan_result(latest)
   442→        n_count = len(list(kg._g.nodes(data=False)))
   443→        e_count = len(list(kg._g.edges(data=False)))
   444→        return ToolResult(
   445→            f"Knowledge graph built: {n_count} nodes, {e_count} edges for {target}.",
   446→            success=True,
   447→            data={"node_count": n_count, "edge_count": e_count, "target": target},
   448→        )
   449→    elif action == "attack_chains":
   450→        chains = kg.find_attack_chains(max_depth=4)
   451→        return ToolResult(
   452→            f"Found {len(chains)} attack chain(s).", success=True,
   453→            data={"attack_chains": chains},
   454→        )
   455→    else:
   456→        return ToolResult(f"Unknown action: {action}. Use 'build' or 'attack_chains'.", success=False)

   859→
   860→class CVSSScorer:
   861→    """CVSS-inspired risk scoring for security findings.
   862→
   863→    Computes a base score (0.0–10.0) using the CVSS v3.1 formula
   864→    simplified for rule-based use:
   865→      - Attack Vector (AV): Network=0.85, Adjacent=0.62, Local=0.55, Physical=0.20
   866→      - Attack Complexity (AC): Low=0.77, High=0.44
   867→      - Privileges Required (PR): None=0.85, Low=0.62, High=0.27
   868→      - User Interaction (UI): None=0.85, Required=0.62
   869→      - Impact (C/I/A): None=0.00, Low=0.22, High=0.56
   870→
   871→    BaseScore = roundup(min(Impact + Exploitability, 10))
   872→    Impact = 1 - [(1-C)*(1-I)*(1-A)]
   873→    Exploitability = 8.22 * AV * AC * PR * UI
   874→    """
   875→
   876→    # Attack Vector metrics
   877→    _AV = {"N": 0.85, "A": 0.62, "L": 0.55, "P": 0.20}
   878→    # Attack Complexity metrics
   879→    _AC = {"L": 0.77, "H": 0.44}
   880→    # Privileges Required metrics
   881→    _PR = {"N": 0.85, "L": 0.62, "H": 0.27}
   882→    # User Interaction metrics
   883→    _UI = {"N": 0.85, "R": 0.62}
   884→    # CIA Impact metrics
   885→    _CIA = {"N": 0.00, "L": 0.22, "H": 0.56}
   886→
   887→    # Category-to-metric mappings. Keys are lowercase substrings matched
   888→    # against finding titles/descriptions.
   889→    _CATEGORY_PROFILES: List[Dict[str, Any]] = [
   890→        # TLS / transport security
   891→        {"patterns": ["hsts", "strict-transport", "tls", "ssl", "certificate", "https redirect",
   892→                        "mixed content", "tls downgrade"],
   893→         "av": "N", "ac": "L", "pr": "N", "ui": "N", "c": "L", "i": "H", "a": "N"},
   894→        # Security headers
   895→        {"patterns": ["x-frame", "csp", "content-security", "x-content-type", "x-xss",
   896→                        "referrer-policy", "permissions-policy", "missing header"],
   897→         "av": "N", "ac": "L", "pr": "N", "ui": "N", "c": "L", "i": "L", "a": "N"},
   898→        # Admin panel exposure
   899→        {"patterns": ["admin panel", "admin interface", "dashboard exposed", "wp-admin",
   900→                        "phpmyadmin", "control panel"],
   901→         "av": "N", "ac": "L", "pr": "N", "ui": "R", "c": "H", "i": "H", "a": "H"},
   902→        # Authentication weaknesses
   903→        {"patterns": ["weak password", "brute force", "no rate limit", "default credential",
   904→                        "auth bypass", "login", "session fix", "cookie"],
   905→         "av": "N", "ac": "L", "pr": "N", "ui": "N", "c": "H", "i": "H", "a": "L"},
   906→        # SQL injection / code injection
   907→        {"patterns": ["sqli", "sql injection", "xss", "cross-site scripting", "command injection",
   908→                        "code injection", "eval", "exec", "rce", "remote code"],
   909→         "av": "N", "ac": "L", "pr": "N", "ui": "R", "c": "H", "i": "H", "a": "H"},
   910→        # Sensitive data exposure
   911→        {"patterns": ["secret", "api key", "token", "credential", ".env", "hardcoded",
   912→                        "private key", "leaked", "exposed", "password in"],
   913→         "av": "N", "ac": "L", "pr": "N", "ui": "N", "c": "H", "i": "L", "a": "N"},
   914→        # Open ports / network
   915→        {"patterns": ["open port", "port ", "service exposed", "database exposed",
   916→                        "redis", "mongodb", "mysql", "ssh exposed"],
   917→         "av": "N", "ac": "L", "pr": "N", "ui": "N", "c": "H", "i": "H", "a": "H"},
   918→        # SSRF / redirect
   919→        {"patterns": ["ssrf", "open redirect", "redirect", "url redirect", "server-side request"],
   920→         "av": "N", "ac": "L", "pr": "N", "ui": "R", "c": "H", "i": "L", "a": "L"},

   922→        {"patterns": ["info leak", "version disclose", "server header", "directory listing",

   924→         "av": "N", "ac": "L", "pr": "N", "ui": "N", "c": "L", "i": "L", "a": "N"},

   936→        {"patterns": ["file permission", "chmod", "world-readable", "writable",

   938→         "av": "L", "ac": "L", "pr": "L", "ui": "N", "c": "H", "i": "H", "a": "H"},

   940→        {"patterns": ["bot", "crawler", "automation", "c2", "command and control"],
   941→         "av": "N", "ac": "L", "pr": "N", "ui": "N", "c": "L", "i": "L", "a": "L"},

   943→        {"patterns": ["cloud", "aws", "azure", "gcp", "s3 bucket", "iam", "oauth",
   944→                        "identity", "federation"],
   945→         "av": "N", "ac": "L", "pr": "N", "ui": "N", "c": "H", "i": "H", "a": "H"},
   946→        # Default fallback
   947→        {"patterns": [],
   948→         "av": "N", "ac": "L", "pr": "N", "ui": "N", "c": "L", "i": "L", "a": "N"},
   949→    ]
   950→
   951→    @classmethod
   952→    def score_finding(cls, finding: Dict) -> Dict[str, Any]:
   953→        """Compute CVSS-like score for a single finding.
   954→
   955→        Returns a dict with: base_score, severity, av, ac, pr, ui, c, i, a, vector_string.
   956→        """
   957→        title = (finding.get("title") or "").lower()
   958→        desc = (finding.get("description") or "").lower()
   959→        combined = f"{title} {desc}"
   960→        original_sev = (finding.get("severity") or "").lower()
   961→
   962→        # Find matching profile
   963→        profile = cls._CATEGORY_PROFILES[-1]  # default
   964→        for prof in cls._CATEGORY_PROFILES:
   965→            if prof["patterns"] and any(p in combined for p in prof["patterns"]):
   966→                profile = prof
   967→                break
   968→
   969→        # Override metrics based on original severity if it's explicitly critical
   970→        av = cls._AV[profile["av"]]
   971→        ac = cls._AC[profile["ac"]]
   972→        pr = cls._PR[profile["pr"]]
   973→        ui = cls._UI[profile["ui"]]
   974→        c = cls._CIA[profile["c"]]
   975→        i = cls._CIA[profile["i"]]
   976→        a = cls._CIA[profile["a"]]
   977→
   978→        # If the original severity is critical, boost impact to High
   979→        if original_sev == "critical":
   980→            c = max(c, cls._CIA["H"])
   981→            i = max(i, cls._CIA["H"])
   982→            a = max(a, cls._CIA["H"])
   983→        elif original_sev == "high":
   984→            c = max(c, cls._CIA["L"])
   985→            i = max(i, cls._CIA["L"])
   986→            a = max(a, cls._CIA["L"])
   987→
   988→        # CVSS base score calculation
   989→        iss = 1.0 - ((1.0 - c) * (1.0 - i) * (1.0 - a))
   990→        impact = 6.42 * iss
   991→        exploitability = 8.22 * av * ac * pr * ui
   992→
   993→        if impact <= 0:
   994→            base_score = 0.0
   995→        else:
   996→            base_score = min(impact + exploitability, 10.0)
   997→
   998→        # Round up to nearest 0.1
   999→        base_score = math.ceil(base_score * 10) / 10
  1000→
  1001→        # Derive severity from score
  1002→        if base_score >= 9.0:
  1003→            derived_sev = "critical"
  1004→        elif base_score >= 7.0:
  1005→            derived_sev = "high"
  1006→        elif base_score >= 4.0:
  1007→            derived_sev = "medium"
  1008→        elif base_score > 0.0:
  1009→            derived_sev = "low"
  1010→        else:
  1011→            derived_sev = "info"
  1012→
  1013→        # Use the more severe of original vs derived
  1014→        severity_order = {"critical": 4, "high": 3, "medium": 2, "low": 1, "info": 0}
  1015→        final_sev = original_sev if severity_order.get(original_sev, -1) >= severity_order.get(derived_sev, -1) else derived_sev
  1016→
  1017→        vector = f"CVSS:3.1/AV:{profile['av']}/AC:{profile['ac']}/PR:{profile['pr']}/UI:{profile['ui']}/C:{cls._cia_label(c)}/I:{cls._cia_label(i)}/A:{cls._cia_label(a)}"
  1018→
  1019→        return {
  1020→            "base_score": base_score,
  1021→            "severity": final_sev,
  1022→            "cvss_vector": vector,
  1023→            "metrics": {
  1024→                "av": profile["av"], "ac": profile["ac"], "pr": profile["pr"],
  1025→                "ui": profile["ui"], "c": cls._cia_label(c), "i": cls._cia_label(i),
  1026→                "a": cls._cia_label(a),
  1027→            },
  1028→        }
  1029→
  1030→    @classmethod
  1031→    def score_findings(cls, findings: List[Dict]) -> List[Dict]:

  1033→        enriched = []
  1034→        for f in findings:
  1035→            scored = dict(f)
  1036→            cvss = cls.score_finding(f)
  1037→            scored["cvss_score"] = cvss["base_score"]

  1040→            scored["cvss_metrics"] = cvss["metrics"]

  1361→    @classmethod
  1362→    def correlate(cls, findings: List[Dict]) -> List[Dict]:
  1363→        """Run all correlation rules against findings and return compound findings.
  1364→
  1365→        Each finding is checked as a lowercase string of title + description.
  1366→        A rule fires when ALL trigger groups have at least one pattern matched
  1367→        across the full finding set.
  1368→        """
  1369→        if not findings:
  1370→            return []
  1371→
  1372→        # Build a flat list of all finding text for matching
  1373→        finding_texts: List[str] = []
  1374→        for f in findings:
  1375→            title = (f.get("title") or "").lower()
  1376→            desc = (f.get("description") or "").lower()
  1377→            finding_texts.append(f"{title} {desc}")
  1378→
  1379→        compound_findings: List[Dict] = []
  1380→
  1381→        for rule in cls.CORRELATION_RULES:
  1382→            triggers = rule["triggers"]
  1383→            rule_fired = True
  1384→
  1385→            for trigger_group in triggers:
  1386→                # Each trigger group: at least one pattern must match across all findings
  1387→                group_matched = False
  1388→                for pattern in trigger_group:
  1389→                    for ft in finding_texts:
  1390→                        if pattern in ft:
  1391→                            group_matched = True
  1392→                            break
  1393→                    if group_matched:
  1394→                        break
  1395→                if not group_matched:
  1396→                    rule_fired = False
  1397→                    break
  1398→
  1399→            if rule_fired:
  1400→                compound = {
  1401→                    "title": rule["compound_title"],
  1402→                    "severity": rule["compound_severity"],
  1403→                    "description": rule["compound_description"],
  1404→                    "category": "compound",
  1405→                    "correlation_id": rule["id"],
  1406→                    "is_compound": True,
  1407→                }
  1408→                # Compute CVSS score for the compound finding
  1409→                if "cvss_override" in rule:
  1410→                    override = rule["cvss_override"]
  1411→                    av = CVSSScorer._AV[override["av"]]
  1412→                    ac = CVSSScorer._AC[override["ac"]]
  1413→                    pr = CVSSScorer._PR[override["pr"]]
  1414→                    ui = CVSSScorer._UI[override["ui"]]
  1415→                    c_val = CVSSScorer._CIA[override["c"]]
  1416→                    i_val = CVSSScorer._CIA[override["i"]]
  1417→                    a_val = CVSSScorer._CIA[override["a"]]
  1418→                    iss = 1.0 - ((1.0 - c_val) * (1.0 - i_val) * (1.0 - a_val))
  1419→                    impact = 6.42 * iss
  1420→                    exploitability = 8.22 * av * ac * pr * ui
  1421→                    base_score = math.ceil(min(impact + exploitability, 10.0) * 10) / 10
  1422→                    compound["cvss_score"] = base_score
  1423→                    compound["cvss_vector"] = (
  1424→                        f"CVSS:3.1/AV:{override['av']}/AC:{override['ac']}/"
  1425→                        f"PR:{override['pr']}/UI:{override['ui']}/"
  1426→                        f"C:{override['c']}/I:{override['i']}/A:{override['a']}"
  1427→                    )
  1428→
  1429→                compound_findings.append(compound)
  1430→
  1431→        return compound_findings

  1435→#  ATTACK SURFACE MAPPER

  1437→
  1438→
  1439→class AttackSurfaceMapper:
  1440→    """Computes an attack surface score (0–100, higher = more exposed).
  1441→
  1442→    Factors:
  1443→      1. Open ports (0–25 pts): 0 ports=0, 1–2=5, 3–5=10, 6–10=15, 11+=25
  1444→      2. Exposed paths/endpoints (0–20 pts): 0=0, 1–3=5, 4–7=10, 8–12=15, 13+=20
  1445→      3. Missing security headers (0–20 pts): 0=0, 1=3, 2=6, 3=10, 4=14, 5+=20
  1446→      4. Tech stack complexity (0–15 pts): 1–2 techs=2, 3–4=5, 5–6=8, 7–8=11, 9+=15
  1447→      5. TLS grade (0–20 pts): A=0, B=5, C=10, D=15, F=20
  1448→    """
  1449→
  1450→    @classmethod
  1451→    def compute(cls, findings: List[Dict], scan_data: Optional[Dict] = None) -> Dict[str, Any]:
  1452→        """Compute attack surface score and return breakdown."""
  1453→        findings_lower = []
  1454→        for f in findings:
  1455→            title = (f.get("title") or "").lower()
  1456→            desc = (f.get("description") or "").lower()

  1638→class RiskMatrixVisualizer:

  1664→    # Likelihood and Impact bands from CVSS scores

  1672→

  1684→    @classmethod
  1685→    def render(cls, findings: List[Dict], width: int = 70) -> str:
  1686→        """Render the risk matrix as a text string with Unicode box-drawing.
  1687→
  1688→        Args:
  1689→            findings: List of finding dicts with 'cvss_score' (or 'base_score') keys.

  1691→

  1695→        # Build a 5x5 grid of finding counts
  1696→        # Grid[y][x] where y=0 is top (Very High likelihood), x=0 is left (Negligible impact)
  1697→        grid: List[List[int]] = [[0] * 5 for _ in range(5)]

  1711→            impact_idx = cls._score_to_band_index(score)

  1713→

  1717→                    grid_labels[likelihood_idx][impact_idx].append(title)

  1721→        label_width = 12

  1748→                else:
  1749→                    row2 += f"{'':^{col_width}}│"
  1750→
  1751→            lines.append(row1)
  1752→            lines.append(row2)
  1753→

  1769→    @classmethod
  1770→    def _score_to_band_index(cls, score: float) -> int:
  1771→        """Map score to impact band index (0=Negligible, 4=Critical)."""
  1772→        for i, (name, lo, hi) in enumerate(cls._IMPACT_BANDS):
  1773→            if lo <= score < hi:
  1774→                return 4 - i  # Reverse so Critical is rightmost
  1775→        return 0

  1777→    @classmethod
  1778→    def _score_to_likelihood(cls, score: float, finding: Dict) -> int:
  1779→        """Estimate likelihood from score and finding metadata.
  1780→
  1781→        Higher CVSS exploitability metrics → higher likelihood.
  1782→        We approximate from score and severity.
  1783→        """
  1784→        severity = (finding.get("severity") or "").lower()
  1785→        is_compound = finding.get("is_compound", False)
  1786→        is_remote = True  # Most findings in recon are network-based
  1787→
  1788→        # Base likelihood from score
  1789→        base = score / 10.0
  1790→
  1791→        # Boost for compound findings (more likely to be exploitable)
  1792→        if is_compound:
  1793→            base = min(base + 0.15, 1.0)
  1794→
  1795→        # Adjust for severity
  1796→        sev_boost = {"critical": 0.1, "high": 0.05, "medium": 0, "low": -0.1, "info": -0.2}
  1797→        base += sev_boost.get(severity, 0)
  1798→        base = max(0.0, min(base, 1.0))
  1799→
  1800→        # Map to band index (0=Very Low, 4=Very High)
  1801→        if base >= 0.85:
  1802→            return 0
  1803→        elif base >= 0.65:
  1804→            return 1
  1805→        elif base >= 0.40:
  1806→            return 2
  1807→        elif base >= 0.15:
  1808→            return 3
  1809→        else:
  1810→            return 4
