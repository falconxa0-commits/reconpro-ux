    89→// ── Mapping helpers ──────────────────────────────────────────────────────────
    90→
    91→function mapFrameworkStatus(
    92→  apiStatus: ApiFramework['status']
    93→): ComplianceFramework['status'] {
    94→  switch (apiStatus) {
    95→    case 'pass':
    96→      return 'Compliant';
    97→    case 'warn':
    98→    case 'needs_review':
    99→      return 'Needs Attention';
   100→    case 'fail':
   101→      return 'Non-Compliant';
   102→    default:
   103→      return 'Needs Attention';
   104→  }
   105→}
   106→
   107→function mapControlStatus(
   108→  apiStatus: ApiControl['status']
   109→): FrameworkControl['status'] {
   110→  switch (apiStatus) {
   111→    case 'pass':
   112→      return 'pass';
   113→    case 'fail':
   114→      return 'fail';
   115→    case 'warn':
   116→      return 'pending';
   117→    default:
   118→      return 'pending';
   119→  }
   120→}
   121→
   122→function mapApiToFramework(api: ApiFramework): ComplianceFramework {
   123→  return {
   124→    id: api.id,
   125→    name: api.name,
   126→    icon: api.icon,
   127→    score: api.score,
   128→    status: mapFrameworkStatus(api.status),
   129→    controlsPassed: api.controlsPassed,
   130→    controlsTotal: api.controlsTotal,
   131→    lastAssessed: api.lastAssessed,
   132→    controls: api.controls.map((c) => ({
   133→      id: c.id,
   134→      description: c.name,
   135→      status: mapControlStatus(c.status),
   136→      category: c.category,
   137→      evidenceLink: c.evidence && c.evidence !== 'No relevant findings detected' && !c.evidence.startsWith('Not assessable') ? c.evidence : undefined,
   138→    })),
   139→  };
   140→}

   144→function getScoreColor(score: number): string {
   145→  if (score >= 90) return '#00ff88';

   150→function getScoreGradient(score: number): string {
   151→  if (score >= 90) return 'from-[#00ff88]/20 to-[#00ff88]/5';
   152→  if (score >= 75) return 'from-[#d29922]/20 to-[#d29922]/5';
   153→  return 'from-[#ff3355]/20 to-[#ff3355]/5';

   169→// ── Circular Gauge SVG ──────────────────────────────────────────────────────

   171→function ComplianceGauge({ score }: { score: number }) {
   172→  const radius = 70;
   173→  const circumference = 2 * Math.PI * radius;
   174→  const offset = circumference - (score / 100) * circumference;
   175→  const color = getScoreColor(score);
   176→
   177→  return (
   178→    <div className="relative inline-flex items-center justify-center">
   179→      <svg width="180" height="180" viewBox="0 0 180 180" className="transform -rotate-90">
   180→        {/* Background circle */}
   181→        <circle cx="90" cy="90" r={radius} fill="none" stroke="#161b22" strokeWidth="10" />
   182→        {/* Progress arc */}
   183→        <motion.circle
   184→          cx="90"
   185→          cy="90"
   186→          r={radius}
   187→          fill="none"
   188→          stroke={color}
   189→          strokeWidth="10"
   190→          strokeLinecap="round"
   191→          strokeDasharray={circumference}
   192→          initial={{ strokeDashoffset: circumference }}
   193→          animate={{ strokeDashoffset: offset }}
   194→          transition={{ duration: 1.5, ease: 'easeOut' as const, delay: 0.3 }}
   195→          style={{
   196→            filter: `drop-shadow(0 0 8px ${color}50)`,
   197→          }}
   198→        />
   199→      </svg>
   200→      <div className="absolute flex flex-col items-center justify-center">
   201→        <motion.span
   202→          className="text-3xl font-black text-[#f0f0f0]"
   203→          initial={{ opacity: 0, scale: 0.5 }}
   204→          animate={{ opacity: 1, scale: 1 }}
   205→          transition={{ delay: 0.5, duration: 0.5 }}
   206→        >
   207→          {score}%
   208→        </motion.span>
   209→        <span className="text-[10px] uppercase tracking-widest text-[#444444] font-medium">Overall</span>
   210→      </div>
   211→    </div>
   212→  );
   213→}

   217→function TrendSparkline() {
   218→  const points = [62, 65, 68, 72, 70, 75, 78, 82, 85, 84, 87, 89];
   219→  const w = 200;
   220→  const h = 40;
   221→  const step = w / (points.length - 1);
   222→
   223→  const pathData = points
   224→    .map((p, i) => `${i === 0 ? 'M' : 'L'} ${i * step} ${h - (p / 100) * h}`)
   225→    .join(' ');
   226→
   227→  const areaData = `${pathData} L ${w} ${h} L 0 ${h} Z`;
   228→
   229→  return (
   230→    <svg width={w} height={h} viewBox={`0 0 ${w} ${h}`} className="overflow-visible">
   231→      <defs>
   232→        <linearGradient id="sparkGrad" x1="0" y1="0" x2="0" y2="1">
   233→          <stop offset="0%" stopColor="#00ff88" stopOpacity="0.3" />
   234→          <stop offset="100%" stopColor="#00ff88" stopOpacity="0" />
   235→        </linearGradient>
   236→      </defs>
   237→      <motion.path
   238→        d={areaData}
   239→        fill="url(#sparkGrad)"
   240→        initial={{ opacity: 0 }}
   241→        animate={{ opacity: 1 }}
   242→        transition={{ delay: 0.8, duration: 0.6 }}
   243→      />
   244→      <motion.path
   245→        d={pathData}
   246→        fill="none"
   247→        stroke="#00ff88"
   248→        strokeWidth="2"
   249→        strokeLinecap="round"
   250→        strokeLinejoin="round"
   251→        initial={{ pathLength: 0 }}
   252→        animate={{ pathLength: 1 }}
   253→        transition={{ delay: 0.5, duration: 1.2, ease: 'easeOut' as const }}
   254→      />
   255→      <motion.circle
   256→        cx={(points.length - 1) * step}
   257→        cy={h - (points[points.length - 1] / 100) * h}
   258→        r="3"
   259→        fill="#00ff88"
   260→        initial={{ opacity: 0, scale: 0 }}
   261→        animate={{ opacity: 1, scale: 1 }}
   262→        transition={{ delay: 1.5, duration: 0.3 }}
   263→      />
   264→    </svg>
   265→  );
   266→}
