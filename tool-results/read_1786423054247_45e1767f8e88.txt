   213→   213→    def shortest_path(
   214→   214→        self, source: str, target: str
   215→   215→    ) -> Optional[List[Dict[str, Any]]]:
   216→   216→        """BFS shortest path between *source* and *target* nodes.
   217→   217→
   218→   218→        Returns a list of node-data dicts, or ``None`` if no path exists.
   219→   219→        """
   220→   220→        with self._lock:
   221→   221→            src_id = self._graph._resolve_node(source)
   222→   222→            tgt_id = self._graph._resolve_node(target)
   223→   223→            if src_id is None or tgt_id is None:
   224→   224→                return None
   225→   225→            if src_id == tgt_id:
   226→   226→                if src_id in (self._graph._g.nodes() if hasattr(self._graph._g, 'has_node') else []):
   227→   227→                    return []
   228→   228→                return None
   229→   229→
   230→   230→            # BFS
   231→   231→            from collections import deque
   232→   232→            visited: Set[str] = set()
   233→   233→            parent: Dict[str, Optional[str]] = {}
   234→   234→            queue: deque = deque()
   235→   235→            queue.append(src_id)
   236→   236→            visited.add(src_id)
   237→   237→            parent[src_id] = None
   238→   238→            found = False
   239→   239→            while queue and not found:
   240→   240→                current = queue.popleft()
   241→   241→                for neighbour in self._graph._g.successors(current):
   242→   242→                    if neighbour not in visited:
   243→   243→                        visited.add(neighbour)
   244→   244→                        parent[neighbour] = current
   245→   245→                        if neighbour == tgt_id:
   246→   246→                            found = True
   247→   247→                            break
   248→   248→                        queue.append(neighbour)
   249→   249→
   250→   250→            if not found:
   251→   251→                return None
   252→   252→
   253→   253→            # Reconstruct path
   254→   254→            path_ids: List[str] = []
   255→   255→            cur: Optional[str] = tgt_id
   256→   256→            while cur is not None:
   257→   257→                path_ids.append(cur)
   258→   258→                cur = parent.get(cur)
   259→   259→            path_ids.reverse()
   260→   260→
   261→   261→            result: List[Dict[str, Any]] = []
   262→   262→            for nid in path_ids:
   263→   263→                try:
   264→   264→                    data = dict(self._graph._g.nodes[nid])
   265→   265→                except Exception:
   266→   266→                    data = self._graph._g.get_node_data(nid)
   267→   267→                result.append(data)
   268→   268→
   269→   269→            return result

   292→   292→        """Lazily load all findings from disk on first access."""
   293→   293→        if self._findings_loaded:

   349→   349→        self,
   350→   350→        target: str,
   351→   351→        since: Optional[str] = None,
   352→   352→        severity: Optional[str] = None,
   353→   353→        module: Optional[str] = None,
   354→   354→    ) -> List[Dict[str, Any]]:
   355→   355→        """Retrieve findings for *target* with optional filters.
   356→   356→
   357→   357→        Parameters
   358→   358→        ----------
   359→   359→        target:
   360→   360→            The target hostname / IP to query.
   361→   361→        since:
   362→   362→            ISO-8601 timestamp; only findings after this time are returned.
   363→   363→        severity:
   364→   364→            If set, only findings matching this severity (case-insensitive).
   365→   365→        module:
   366→   366→            If set, only findings whose ``module`` field matches.
   367→   367→        """
   368→   368→        with self._lock:
   369→   369→            self._ensure_findings_loaded()
   370→   370→            entries = list(self._findings.get(target, []))
   371→   371→
   372→   372→            if since is not None:
   373→   373→                try:
   374→   374→                    since_dt = datetime.fromisoformat(since)
   375→   375→                except (ValueError, TypeError):
   376→   376→                    since_dt = None
   377→   377→                if since_dt is not None:
   378→   378→                    entries = [
   379→   379→                        e for e in entries
   380→   380→                        if "_ts" in e
   381→   381→                        and datetime.fromisoformat(e["_ts"]) >= since_dt
   382→   382→                    ]
   383→   383→
   384→   384→            if severity is not None:
   385→   385→                sev_lower = severity.strip().lower()
   386→   386→                entries = [
   387→   387→                    e for e in entries
   388→   388→                    if str(e.get("severity", "")).strip().lower() == sev_lower
   389→   389→                ]
   390→   390→
   391→   391→            if module is not None:
   392→   392→                mod_lower = module.strip().lower()
   393→   393→                entries = [
   394→   394→                    e for e in entries
   395→   395→                    if str(e.get("module", "")).strip().lower() == mod_lower
   396→   396→                ]
   397→   397→
   398→   398→            return entries

   423→   423→    def trend(self, target: str, days: int = 30) -> List[Dict[str, Any]]:
   424→   424→        """Daily severity breakdown for *target* over the last *days* days.
   425→   425→
   426→   426→        Returns a list of dicts, one per day that has findings::
   427→   427→
   428→   428→            [{"date": "2025-01-15", "critical": 0, "high": 1, "medium": 3, "low": 0, "score": 7.0}, ...]
   429→   429→        """
   430→   430→        with self._lock:
   431→   431→            self._ensure_findings_loaded()
   432→   432→            entries = self._findings.get(target, [])
   433→   433→
   434→   434→            now = datetime.now(timezone.utc)
   435→   435→            cutoff = now - timedelta(days=days)
   436→   436→
   437→   437→            daily: Dict[str, Dict[str, int]] = defaultdict(
   438→   438→                lambda: {"critical": 0, "high": 0, "medium": 0, "low": 0, "info": 0}
   439→   439→            )
   440→   440→
   441→   441→            for entry in entries:
   442→   442→                ts = entry.get("_ts", "")
   443→   443→                if not ts:
   444→   444→                    continue
   445→   445→                try:
   446→   446→                    dt = datetime.fromisoformat(ts)
   447→   447→                except (ValueError, TypeError):
   448→   448→                    continue
   449→   449→                if dt < cutoff:
   450→   450→                    continue
   451→   451→                day_str = dt.strftime("%Y-%m-%d")
   452→   452→                sev = str(entry.get("severity", "info")).strip().lower()
   453→   453→                if sev in daily[day_str]:
   454→   454→                    daily[day_str][sev] += 1
   455→   455→                else:
   456→   456→                    daily[day_str]["info"] += 1
   457→   457→
   458→   458→            # Build result sorted by date
   459→   459→            result: List[Dict[str, Any]] = []
   460→   460→            for day in sorted(daily.keys()):
   461→   461→                counts = daily[day]
   462→   462→                score = (
   463→   463→                    counts.get("critical", 0) * 10.0
   464→   464→                    + counts.get("high", 0) * 5.0
   465→   465→                    + counts.get("medium", 0) * 2.0
   466→   466→                    + counts.get("low", 0) * 0.5
   467→   467→                )
   468→   468→                result.append({
   469→   469→                    "date": day,
   470→   470→                    "critical": counts.get("critical", 0),
   471→   471→                    "high": counts.get("high", 0),
   472→   472→                    "medium": counts.get("medium", 0),
   473→   473→                    "low": counts.get("low", 0),
   474→   474→                    "score": round(score, 1),
   475→   475→                })
   476→   476→
   477→   477→            # Fill in missing days
   478→   478→            if result:
   479→   479→                start_date = datetime.strptime(result[0]["date"], "%Y-%m-%d").date()
   480→   480→                end_date = now.date()
   481→   481→                filled: List[Dict[str, Any]] = []
   482→   482→                idx = 0
   483→   483→                current = start_date
   484→   484→                while current <= end_date:
   485→   485→                    day_str = current.isoformat()
   486→   486→                    if idx < len(result) and result[idx]["date"] == day_str:
   487→   487→                        filled.append(result[idx])
   488→   488→                        idx += 1
   489→   489→                    else:
   490→   490→                        filled.append({
   491→   491→                            "date": day_str,
   492→   492→                            "critical": 0,
   493→   493→                            "high": 0,
   494→   494→                            "medium": 0,
   495→   495→                            "low": 0,
   496→   496→                            "score": 0.0,
   497→   497→                        })
   498→   498→                    current += timedelta(days=1)
   499→   499→                return filled
   500→   500→
   501→   501→            return result
   502→   502→

   507→   507→    def bb_set(
   508→   508→        self,
   509→   509→        key: str,
   510→   510→        value: Any,
   511→   511→        agent_id: str,
   512→   512→        ttl: int = 3600,
   513→   513→    ) -> None:
   514→   514→        """Store *value* on the blackboard with *agent_id* attribution and *ttl*.
   515→   515→
   516→   516→        The entry is automatically expired after *ttl* seconds (default 1 hour).
   517→   517→        """
   518→   518→        with self._lock:
   519→   519→            self._blackboard[key] = {
   520→   520→                "value": value,
   521→   521→                "agent_id": agent_id,
   522→   522→                "set_at": time.time(),
   523→   523→                "expires_at": time.time() + ttl,
   524→   524→            }
   525→   525→
   526→   526→    def bb_get(self, key: str) -> Optional[Any]:
   527→   527→        """Retrieve a value from the blackboard.
   528→   528→
   529→   529→        Returns ``None`` if the key does not exist or has expired.
   530→   530→        """
   531→   531→        with self._lock:
   532→   532→            entry = self._blackboard.get(key)
   533→   533→            if entry is None:
   534→   534→                return None
   535→   535→            if time.time() > entry.get("expires_at", 0):
   536→   536→                del self._blackboard[key]
   537→   537→                return None
   538→   538→            return entry["value"]
   539→   539→
   540→   540→    def bb_get_all(self, agent_id: Optional[str] = None) -> Dict[str, Any]:
   541→   541→        """Return all active (non-expired) blackboard entries.
   542→   542→
   543→   543→        If *agent_id* is given, only entries from that agent are returned.
   544→   544→
   545→   545→        Returns a dict of ``{key: value}``.
   546→   546→        """
   547→   547→        with self._lock:
   548→   548→            now = time.time()
   549→   549→            # Prune expired entries first
   550→   550→            expired_keys = [
   551→   551→                k for k, v in self._blackboard.items()
   552→   552→                if now > v.get("expires_at", 0)
   553→   553→            ]
   554→   554→            for k in expired_keys:
   555→   555→                del self._blackboard[k]
   556→   556→
   557→   557→            result: Dict[str, Any] = {}
   558→   558→            for key, entry in self._blackboard.items():
   559→   559→                if agent_id is not None and entry.get("agent_id") != agent_id:
   560→   560→                    continue
   561→   561→                result[key] = entry["value"]
   562→   562→            return result
   563→   563→
   564→   564→    def bb_clear(self) -> None:
   565→   565→        """Wipe all blackboard entries."""
   566→   566→        with self._lock:
   567→   567→            self._blackboard.clear()

   573→   573→    @staticmethod
   574→   574→    def _machine_key() -> bytes:
   575→   575→        """Derive a deterministic obfuscation key from machine identity."""
   576→   576→        raw = f"reconpro-vault:{platform.node()}:{platform.system()}"
   577→   577→        return hashlib.sha256(raw.encode("utf-8")).digest()
   578→   578→
   579→   579→    @staticmethod
   580→   580→    def _obfuscate(plaintext: str, key: bytes) -> str:
   581→   581→        """XOR + base64 obfuscation (NOT cryptographic encryption)."""
   582→   582→        data = plaintext.encode("utf-8")
   583→   583→        xored = bytes(b ^ key[i % len(key)] for i, b in enumerate(data))
   584→   584→        return base64.b64encode(xored).decode("ascii")
   585→   585→
   586→   586→    @staticmethod
   587→   587→    def _deobfuscate(ciphertext: str, key: bytes) -> str:
   588→   588→        """Reverse the XOR + base64 obfuscation."""
   589→   589→        try:
   590→   590→            xored = base64.b64decode(ciphertext.encode("ascii"))
   591→   591→            key_bytes = key
   592→   592→            data = bytes(b ^ key_bytes[i % len(key_bytes)] for i, b in enumerate(xored))
   593→   593→            return data.decode("utf-8")
   594→   594→        except Exception:
   595→   595→            return ""
   596→   596→
   597→   597→    def _ensure_vault_loaded(self) -> None:
   598→   598→        """Lazily load vault from disk on first access."""
   599→   599→        if self._vault_loaded:
   600→   600→            return
   601→   601→        with self._lock:
   602→   602→            if self._vault_loaded:
   603→   603→                return
   604→   604→            self._vault_loaded = True
   605→   605→            if not VAULT_PATH.is_file():
   606→   606→                return
   607→   607→            try:
   608→   608→                raw = json.loads(VAULT_PATH.read_text("utf-8"))
   609→   609→                self._vault = raw if isinstance(raw, list) else []
   610→   610→            except Exception:
   611→   611→                self._vault = []
   612→   612→
   613→   613→    def _flush_vault(self) -> None:
   614→   614→        """Persist vault to disk."""
   615→   615→        MEMORY_DIR.mkdir(parents=True, exist_ok=True)
   616→   616→        VAULT_PATH.write_text(
   617→   617→            json.dumps(self._vault, indent=2, default=str), "utf-8"
   618→   618→        )
   619→   619→
   620→   620→    def vault_store(
   621→   621→        self,
   622→   622→        source: str,
   623→   623→        username: str,
   624→   624→        password: str,
   625→   625→        url: Optional[str] = None,
   626→   626→        notes: str = "",
   627→   627→    ) -> None:
   628→   628→        """Store a credential in the vault.
   629→   629→
   630→   630→        Passwords are obfuscated with a machine-specific key before storage.
   631→   631→        """
   632→   632→        with self._lock:
   633→   633→            self._ensure_vault_loaded()
   634→   634→            key = self._machine_key()
   635→   635→            entry: Dict[str, str] = {
   636→   636→                "source": source,
   637→   637→                "username": username,
   638→   638→                "password": self._obfuscate(password, key),
   639→   639→                "_ts": datetime.now(timezone.utc).isoformat(),
   640→   640→            }
   641→   641→            if url is not None:
   642→   642→                entry["url"] = url
   643→   643→            if notes:
   644→   644→                entry["notes"] = notes
   645→   645→            self._vault.append(entry)
   646→   646→            self._flush_vault()
   647→   647→
   648→   648→    def vault_get_all(self, source: Optional[str] = None) -> List[Dict[str, str]]:
   649→   649→        """Retrieve all credentials, optionally filtered by *source*.
   650→   650→
   651→   651→        Passwords are deobfuscated before returning.
   652→   652→        """
   653→   653→        with self._lock:
   654→   654→            self._ensure_vault_loaded()
   655→   655→            key = self._machine_key()
   656→   656→            results: List[Dict[str, str]] = []
   657→   657→            for entry in self._vault:
   658→   658→                if source is not None and entry.get("source") != source:
   659→   659→                    continue
   660→   660→                decrypted = dict(entry)
   661→   661→                decrypted["password"] = self._deobfuscate(
   662→   662→                    entry.get("password", ""), key
   663→   663→                )
   664→   664→                decrypted.pop("_ts", None)
   665→   665→                results.append(decrypted)
   666→   666→            return results
   667→   667→
   668→   668→    def vault_search(self, query: str) -> List[Dict[str, str]]:
   669→   669→        """Search credentials by *query* (matches username, url, notes, source).
   670→   670→
   671→   671→        Case-insensitive substring match.
   672→   672→        """
   673→   673→        with self._lock:
   674→   674→            self._ensure_vault_loaded()
   675→   675→            key = self._machine_key()
   676→   676→            q_lower = query.lower()
   677→   677→            results: List[Dict[str, str]] = []
   678→   678→            for entry in self._vault:
   679→   679→                searchable = " ".join([
   680→   680→                    entry.get("source", ""),
   681→   681→                    entry.get("username", ""),
   682→   682→                    entry.get("url", ""),
   683→   683→                    entry.get("notes", ""),
   684→   684→                ]).lower()
   685→   685→                if q_lower in searchable:
   686→   686→                    decrypted = dict(entry)
   687→   687→                    decrypted["password"] = self._deobfuscate(
   688→   688→                        entry.get("password", ""), key
   689→   689→                    )
   690→   690→                    decrypted.pop("_ts", None)
   691→   691→                    results.append(decrypted)
   692→   692→            return results
   693→   693→
   694→   694→    def vault_purge(self) -> None:
   695→   695→        """Securely clear all credentials from memory and disk."""
   696→   696→        with self._lock:
   697→   697→            # Overwrite in-memory data
   698→   698→            for entry in self._vault:
   699→   699→                for k in list(entry.keys()):
   700→   700→                    entry[k] = "\x00" * len(str(entry[k]))
   701→   701→            self._vault.clear()
   702→   702→            self._vault_loaded = True
   703→   703→
   704→   704→            # Remove or overwrite file
   705→   705→            if VAULT_PATH.is_file():
   706→   706→                try:
   707→   707→                    size = VAULT_PATH.stat().st_size
   708→   708→                    VAULT_PATH.write_bytes(b"\x00" * size)
   709→   709→                    VAULT_PATH.unlink()
   710→   710→                except Exception:
   711→   711→                    pass

   717→   717→    def add_scan_result(self, scan_dict: Dict[str, Any]) -> None:
   718→   718→        """Ingest a full scan result dict.
   719→   719→
   720→   720→        Populates both the knowledge graph and the finding store.
   721→   721→
   722→   722→        Expected keys (all optional)::
   723→   723→            target, host, subdomains, ports, technologies,
   724→   724→            findings, certificates, endpoints, module, scan_time
   725→   725→        """
   726→   726→        with self._lock:
   727→   727→            # ── Feed the knowledge graph ──
   728→   728→            self._graph.add_scan_result(scan_dict)
   729→   729→
   730→   730→            # ── Feed the finding store ──
   731→   731→            target = scan_dict.get("target") or scan_dict.get("host", "unknown")
   732→   732→            module = scan_dict.get("module", "")
   733→   733→            scan_time = scan_dict.get("scan_time", "")
   734→   734→
   735→   735→            for finding in scan_dict.get("findings", []):
   736→   736→                entry = dict(finding)
   737→   737→                entry["_target"] = target
   738→   738→                if module:
   739→   739→                    entry["module"] = module
   740→   740→                if scan_time and "_ts" not in entry:
   741→   741→                    entry["_ts"] = scan_time
   742→   742→                self._findings[target].append(entry)
   743→   743→
   744→   744→            # Also record a summary entry for the scan itself
   745→   745→            summary: Dict[str, Any] = {
   746→   746→                "_target": target,
   747→   747→                "title": f"Scan completed: {module or 'general'}",
   748→   748→                "severity": "info",
   749→   749→                "module": module,
   750→   750→                "finding_count": len(scan_dict.get("findings", [])),
   751→   751→                "port_count": len(scan_dict.get("ports", [])),
   752→   752→                "tech_count": len(scan_dict.get("technologies", [])),
   753→   753→            }
   754→   754→            if scan_time:
   755→   755→                summary["_ts"] = scan_time
   756→   756→            self._findings[target].append(summary)
   757→   757→
   758→   758→            # Flush findings for this target
   759→   759→            self._flush_findings_target(target)

   761→   761→    def export_all(self) -> Dict[str, Any]:
   762→   762→        """Export everything for backup.
   763→   763→
   764→   764→        Returns a dict with keys ``graph``, ``findings``, ``vault``.
   765→   765→        """
   766→   766→        with self._lock:
   767→   767→            # Graph
   768→   768→            try:
   769→   769→                graph_data = self._graph.save()
   770→   770→                with open(graph_data, "r") as fh:
   771→   771→                    graph_json = json.load(fh)
   772→   772→            except Exception:
   773→   773→                graph_json = {}
   774→   774→
   775→   775→            # Findings
   776→   776→            self._ensure_findings_loaded()
   777→   777→            findings_data = {
   778→   778→                target: list(entries)
   779→   779→                for target, entries in self._findings.items()
   780→   780→            }
   781→   781→
   782→   782→            # Vault (export obfuscated — caller can deobfuscate if needed)
   783→   783→            self._ensure_vault_loaded()
   784→   784→            vault_data = list(self._vault)
   785→   785→
   786→   786→            return {
   787→   787→                "graph": graph_json,
   788→   788→                "findings": findings_data,
   789→   789→                "vault": vault_data,
   790→   790→                "exported_at": datetime.now(timezone.utc).isoformat(),
   791→   791→            }
