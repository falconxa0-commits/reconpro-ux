    12→function tcpProbe(host: string, port: number, timeoutMs: number): Promise<boolean> {
    13→  return new Promise((resolve) => {
    14→    const timer = setTimeout(() => { try { s.destroy(); } catch {} resolve(false); }, timeoutMs);
    15→    const s = net.createConnection({ host, port, timeout: timeoutMs });
    16→    s.on('connect', () => { clearTimeout(timer); try { s.destroy(); } catch {} resolve(true); });
    17→    s.on('timeout', () => { clearTimeout(timer); try { s.destroy(); } catch {} resolve(false); });
    18→    s.on('error', () => { clearTimeout(timer); resolve(false); });
    19→  });
    20→}

    22→function tcpBannerGrab(host: string, port: number, timeoutMs: number, maxBytes = 300): Promise<string> {
    23→  return new Promise((resolve) => {
    24→    let data = '';
    25→    const timer = setTimeout(() => { try { s.destroy(); } catch {} resolve(data); }, timeoutMs);
    26→    const s = net.createConnection({ host, port, timeout: timeoutMs });
    27→    s.on('data', (chunk: Buffer) => {
    28→      data += chunk.toString();
    29→      if (data.length >= maxBytes) { clearTimeout(timer); try { s.destroy(); } catch {} resolve(data.substring(0, maxBytes)); }
    30→    });
    31→    s.on('timeout', () => { clearTimeout(timer); try { s.destroy(); } catch {} resolve(data); });
    32→    s.on('error', () => { clearTimeout(timer); resolve(data); });
    33→    s.on('close', () => { clearTimeout(timer); resolve(data); });
    34→  });
    35→}

    37→async function handleCurl(cmd: string, timeout: number): Promise<string> {
    38→  const isHead = /\b-sI\b|\b-I\b/.test(cmd);
    39→  const timeoutMatch = cmd.match(/--max-time\s+(\d+)/);
    40→  const timeoutSec = timeoutMatch ? parseInt(timeoutMatch[1]) : 8;
    41→  const headers: Record<string, string> = { 'User-Agent': 'Mozilla/5.0' };
    42→  const headerRegex = /-H\s+["']([^"']+)["']/g;
    43→  let m: RegExpExecArray | null;
    44→  while ((m = headerRegex.exec(cmd)) !== null) {
    45→    const ci = m[1].indexOf(':');
    46→    if (ci > 0) headers[m[1].substring(0, ci).trim()] = m[1].substring(ci + 1).trim();
    47→  }
    48→  const urlMatch = cmd.match(/(?:https?:\/\/[^\s"'`]+|http:\/\/[^\s"'`]+)/);
    49→  const url = urlMatch ? urlMatch[0] : '';
    50→  if (!url) return '';
    51→  try {
    52→    const res = await safeFetch(url, { method: isHead ? 'HEAD' : 'GET', headers, followRedirects: false, timeout: timeoutSec * 1000 });
    53→    if (isHead) {
    54→      let out = `HTTP/1.1 ${res.status}\n`;
    55→      for (const [k, v] of Object.entries(res.headers)) { out += `${k}: ${v}\n`; }
    56→      return out;
    57→    }
    58→    return res.text;
    59→  } catch { return ''; }
    60→}

    62→async function handleDig(cmd: string, _timeout: number): Promise<string> {
    63→  const isPtr = cmd.includes('-x');
    64→  let cleaned = cmd.replace(/\bdig\b/, '').replace(/\+short\b/g, '').replace(/\+time=\d+/g, '').replace(/\+tries=\d+/g, '').replace(/@\S+/g, '').replace(/2>\/>\/dev\/null/g, '').replace(/2>&1/g, '').trim();
    65→  if (isPtr) {
    66→    const pm = cleaned.match(/-x\s+(\S+)/);
    67→    const ip = pm ? pm[1] : '';
    68→    if (!ip) return '';
    69→    try { return (await dns.resolvePtr(ip)).join('\n'); } catch { return ''; }
    70→  }
    71→  const tokens = cleaned.split(/\s+/).filter(Boolean);
    72→  if (tokens.length === 0) return '';
    73→  const recordTypes = ['A','AAAA','NS','CNAME','MX','TXT','PTR','SOA','SRV','ANY'];
    74→  let domain: string; let type = 'A';
    75→  const last = tokens[tokens.length - 1].toUpperCase();
    76→  if (tokens.length >= 2 && recordTypes.includes(last)) {
    77→    type = last; domain = tokens.slice(0, -1).join(' ');
    78→  } else { domain = tokens.join(' '); }
    79→  if (!domain) return '';
    80→  try {
    81→    let result: string[] = [];
    82→    switch (type) {
    83→      case 'A': result = await dns.resolve4(domain); break;
    84→      case 'NS': result = await dns.resolveNs(domain); break;
    85→      case 'CNAME': result = await dns.resolveCname(domain); break;
    86→      case 'TXT': { const r = await dns.resolveTxt(domain); result = r.map(x => x.join('')); break; }
    87→      case 'PTR': result = await dns.resolvePtr(domain); break;
    88→      default: result = await dns.resolve4(domain); break;
    89→    }
    90→    return result.join('\n');
    91→  } catch { return ''; }
    92→}

    94→async function handleNetcat(cmd: string, timeout: number): Promise<string> {
    95→  const bashMatch = cmd.match(/\/dev\/tcp\/([^/]+)\/(\d+)/);
    96→  if (bashMatch) {
    97→    const open = await tcpProbe(bashMatch[1], parseInt(bashMatch[2]), Math.min(timeout, 3000));
    98→    return open ? 'OPEN' : 'CLOSED';
    99→  }
   100→  const ncMatch = cmd.match(/nc\s+(?:-\w+\s+)*(\S+)\s+(\d+)/);
   101→  if (ncMatch) return tcpBannerGrab(ncMatch[1], parseInt(ncMatch[2]), timeout);
   102→  return '';
   103→}

   105→function applyPipe(input: string, pipeCmd: string): string {
   106→  const headBytes = pipeCmd.match(/head\s+-c\s+(\d+)/);
   107→  if (headBytes) return input.substring(0, parseInt(headBytes[1]));
   108→  const headLines = pipeCmd.match(/head\s+-(\d+)/);
   109→  if (headLines) return input.split('\n').slice(0, parseInt(headLines[1])).join('\n');
   110→  const grepMatch = pipeCmd.match(/^grep\s+(-\w+\s+)?(.+)$/);
   111→  if (grepMatch) {
   112→    const flags = grepMatch[1]?.trim() || '';
   113→    let pattern = grepMatch[2].trim();
   114→    if ((pattern.startsWith('"') && pattern.endsWith('"')) || (pattern.startsWith("'") && pattern.endsWith("'"))) pattern = pattern.slice(1, -1);
   115→    const isI = flags.includes('i');
   116→    try {
   117→      const regex = new RegExp(pattern, isI ? 'i' : '');
   118→      return input.split('\n').filter(l => regex.test(l)).join('\n');
   119→    } catch { return ''; }
   120→  }
   121→  return input;
   122→}

   124→async function run(cmd: string, timeout = 8000): Promise<string> {
   125→  try {
   126→    if (cmd.includes('/dev/tcp')) {
   127→      const m = cmd.match(/\/dev\/tcp\/([^/]+)\/(\d+)/);
   128→      if (m) { const open = await tcpProbe(m[1], parseInt(m[2]), Math.min(timeout, 3000)); return cmd.includes('OPEN') && cmd.includes('CLOSED') ? (open ? 'OPEN' : 'CLOSED') : ''; }
   129→      return '';
   130→    }
   131→    if (cmd.includes('whois')) return '';
   132→    let effectiveCmd = cmd.replace(/^echo\s*(""\s*)?\|\s*/, '').trim();
   133→    effectiveCmd = effectiveCmd.replace(/\|\|/g, '\x00\x00');
   134→    const segments: string[] = []; let cur = ''; let inQ = false; let qCh = '';
   135→    for (let i = 0; i < effectiveCmd.length; i++) {
   136→      const ch = effectiveCmd[i];
   137→      if (inQ) { cur += ch; if (ch === qCh) inQ = false; }
   138→      else if (ch === '"' || ch === "'") { inQ = true; qCh = ch; cur += ch; }
   139→      else if (ch === '|') { segments.push(cur.trim()); cur = ''; }
   140→      else { cur += ch; }
   141→    }
   142→    segments.push(cur.trim());
   143→    const restored = segments.map(s => s.replace(/\x00\x00/g, '||'));
   144→    const mainCmd = restored[0]; const pipes = restored.slice(1);
   145→    let output = '';
   146→    if (mainCmd.includes('curl')) output = await handleCurl(mainCmd, timeout);
   147→    else if (mainCmd.includes('dig')) output = await handleDig(mainCmd, timeout);
   148→    else if (mainCmd.includes('nc ') || mainCmd.includes('/dev/tcp')) output = await handleNetcat(mainCmd, timeout);
   149→    else return '';
   150→    for (const pipe of pipes) output = applyPipe(output, pipe);
   151→    return output.trim();
   152→  } catch { return ''; }
   153→}

   344→async function dnsBotDetection(domain: string): Promise<any[]> {
   345→  const indicators: any[] = [];
   346→
   347→  // Check for DGA (Domain Generation Algorithm) patterns
   348→  const dgaPatterns = /^[a-z]{12,20}\.(com|net|org|info|top|xyz)$/i;
   349→  const subdomainParts = domain.split('.');
   350→  for (const part of subdomainParts) {
   351→    if (dgaPatterns.test(part)) {
   352→      indicators.push({
   353→        type: 'DGA Domain Pattern',
   354→        severity: 'high',
   355→        description: `Subdomain "${part}" matches DGA pattern — 12-20 random lowercase characters, common in botnet C2 domains.`,
   356→        evidence: `Domain part "${part}" matches /^[a-z]{12,20}\\..*/`,
   357→        asset: domain,
   358→      });
   359→    }
   360→  }
   361→
   362→  // Check for fast-flux DNS (multiple IPs rotating)
   363→  const aRecords = await run(`dig +short +time=3 ${domain} A 2>/dev/null`, 5000);
   364→  const ips = aRecords.split('\n').filter(l => l.trim() && /^\d+\.\d+\.\d+\.\d+$/.test(l.trim()));
   365→  if (ips.length > 5) {
   366→    indicators.push({
   367→      type: 'Fast-Flux DNS Detected',
   368→      severity: 'critical',
   369→      description: `${ips.length} A records for single domain — possible fast-flux botnet technique. IP addresses rotate rapidly to evade blacklisting.`,
   370→      evidence: `${ips.length} IPs: ${ips.slice(0, 8).join(', ')}...`,
   371→      asset: domain,
   372→    });
   373→  }
   374→
   375→  // Check for known malicious record patterns
   376→  const txtRecords = await run(`dig +short +time=3 ${domain} TXT 2>/dev/null`, 5000);
   377→  if (txtRecords) {
   378→    const suspiciousPatterns = ['spf', 'dkim', 'v=spf'];
   379→    // Check for encoded strings (base64 C2 beacons)
   380→    for (const txt of txtRecords.split('\n').filter(Boolean)) {
   381→      if (/^[A-Za-z0-9+/]{20,}={0,2}$/.test(txt.trim())) {
   382→        indicators.push({
   383→          type: 'Suspicious Base64 TXT Record',
   384→          severity: 'high',
   385→          description: `TXT record contains base64-encoded data — possible C2 beacon or data exfiltration channel.`,
   386→          evidence: `TXT: ${txt.substring(0, 40)}...`,
   387→          asset: domain,
   388→        });
   389→      }
   390→    }
   391→  }
   392→
   393→  // DNS tunneling indicators
   394→  const cnames = await run(`dig +short +time=3 ${domain} CNAME 2>/dev/null`, 5000);
   395→  if (cnames && cnames.split('\n').length > 3) {
   396→    indicators.push({
   397→      type: 'Multiple CNAME Records — DNS Tunneling Indicator',
   398→      severity: 'high',
   399→      description: `${cnames.split('\n').filter(Boolean).length} CNAME records — could indicate DNS tunneling for C2 communication or data exfiltration.`,
   400→      evidence: `CNAMEs: ${cnames.substring(0, 150)}`,
   401→      asset: domain,
   402→    });
   403→  }
   404→
   405→  // Newly registered domain check
   406→  const whoisDate = await run(`whois ${domain} 2>/dev/null | grep -iE "creation date|created on|registered on|Registration Date" | head -1`, 8000);
   407→  if (whoisDate) {
   408→    const dateMatch = whoisDate.match(/(\d{4}[-/]\d{2}[-/]\d{2})/);
   409→    if (dateMatch) {
   410→      const regDate = new Date(dateMatch[1]);
   411→      const ageDays = Math.floor((Date.now() - regDate.getTime()) / 86400000);
   412→      if (ageDays < 30) {
   413→        indicators.push({
   414→          type: 'Newly Registered Domain — High Threat',
   415→          severity: 'high',
   416→          description: `Domain registered ${ageDays} days ago. Recently registered domains are 5x more likely to be used for phishing, malware, or C2.`,
   417→          evidence: `Registered: ${dateMatch[1]} (${ageDays} days ago)`,
   418→          asset: domain,
   419→        });
   420→      }
   421→    }
   422→  }
   423→
   424→  // Typosquatting detection against common brands
   425→  const brands = ['google', 'facebook', 'microsoft', 'apple', 'amazon', 'netflix', 'paypal', 'github', 'linkedin', 'twitter'];
   426→  const domainLower = domain.toLowerCase();
   427→  for (const brand of brands) {
   428→    if (domainLower.includes(brand) && !domainLower.startsWith('www.')) {
   429→      // Check if this is a typo or impersonation
   430→      const distance = levenshteinDistance(domainLower.split('.')[0], brand);
   431→      if (distance >= 1 && distance <= 3) {
   432→        indicators.push({
   433→          type: `Typosquatting — Impersonates ${brand}`,
   434→          severity: 'critical',
   435→          description: `Domain ${domain} resembles ${brand}.com (Levenshtein distance: ${distance}). Likely phishing or malware distribution.`,
   436→          evidence: `"${domain.split('.')[0]}" vs "${brand}" (distance: ${distance})`,
   437→          asset: domain,
   438→        });
   439→        break;
   440→      }
   441→    }
   442→  }
   443→
   444→  return indicators;
   445→}
   446→
   447→function levenshteinDistance(a: string, b: string): number {
   448→  const matrix: number[][] = [];
   449→  for (let i = 0; i <= b.length; i++) matrix[i] = [i];
   450→  for (let j = 0; j <= a.length; j++) matrix[0][j] = j;
   451→  for (let i = 1; i <= b.length; i++) {
   452→    for (let j = 1; j <= a.length; j++) {
   453→      matrix[i][j] = b[i - 1] === a[j - 1]
   454→        ? matrix[i - 1][j - 1]
   455→        : Math.min(matrix[i - 1][j - 1] + 1, matrix[i][j - 1] + 1, matrix[i - 1][j] + 1);
   456→    }
   457→  }
   458→  return matrix[b.length][a.length];
   459→}

   462→// 4. THREAT CLASSIFICATION ENGINE
   463→// ══════════════════════════════════════════════════════════════════════════════
   464→
   465→function classifyThreat(ipRep: any, c2Infra: any[], dnsIndicators: any[]): any {
   466→  let score = 0;
   467→  const vectors: string[] = [];
   468→
   469→  if (ipRep.reputation.blacklistHits > 0) {
   470→    score += ipRep.reputation.blacklistHits * 25;
   471→    vectors.push(`Blacklisted on ${ipRep.reputation.blacklistHits}/${ipRep.reputation.totalBlacklists} lists`);
   472→  }
   473→  if (ipRep.flags.tor) { score += 20; vectors.push('Tor exit node'); }
   474→  if (ipRep.flags.proxy) { score += 15; vectors.push('Proxy detected'); }
   475→  if (ipRep.flags.datacenter) { score += 5; vectors.push('Datacenter IP'); }
   476→  if (ipRep.flags.suspiciousASN) { score += 30; vectors.push(`Suspicious ASN: ${ipRep.asn}`); }
   477→
   478→  for (const c2 of c2Infra) {
   479→    score += c2.severity === 'critical' ? 30 : c2.severity === 'high' ? 20 : 10;
   480→    vectors.push(`C2 port ${c2.port} (${c2.service})`);
   481→    if (c2.malwareFamily) vectors.push(`Malware: ${c2.malwareFamily}`);
   482→  }
   483→
   484→  for (const ind of dnsIndicators) {
   485→    score += ind.severity === 'critical' ? 30 : ind.severity === 'high' ? 20 : 10;
   486→    vectors.push(ind.type);
   487→  }
   488→
   489→  score = Math.min(100, score);
   490→  const level = score >= 75 ? 'CRITICAL' : score >= 50 ? 'HIGH' : score >= 25 ? 'MEDIUM' : 'LOW';
   491→
   492→  return {
   493→    score,
   494→    level,
   495→    vectors,
   496→    recommendation: score >= 75
   497→      ? 'IMMEDIATE ACTION: Quarantine IP, block all traffic, investigate all connections'
   498→      : score >= 50
   499→      ? 'ESCALATE: Deep packet inspection required, monitor all connections, prepare containment'
   500→      : score >= 25
   501→      ? 'MONITOR: Flag for observation, increase logging, alert on new connections'
   502→      : 'WATCH: Low threat — standard monitoring sufficient',
   503→  };
   504→}

   507→// MAIN POST HANDLER
   508→// ══════════════════════════════════════════════════════════════════════════════
   509→
   510→export async function POST(request: NextRequest) {
   511→  const { allowed } = checkRateLimit(extractClientIP(request), 30, 60000);
   512→  if (!allowed) return NextResponse.json({ error: 'Rate limit exceeded' }, { status: 429 });
   513→
   514→  try {
   515→    const body = await request.json();
   516→    const rawTarget = body.target?.toString().trim().replace(/^https?:\/\//, '').replace(/\/.*$/, '');
   517→    const mode = body.mode || 'hunt';
   518→
   519→    const sanitized = sanitizeTarget(rawTarget);
   520→    if (!sanitized) {
   521→      return NextResponse.json({ error: 'Invalid target domain or IP' }, { status: 400 });
   522→    }
   523→    if (isBlockedDomain(sanitized)) {
   524→      return NextResponse.json({ error: 'Target domain is not permitted' }, { status: 403 });
   525→    }
   526→
   527→    // Resolve to IP if domain
   528→    const isIP = /^\d+\.\d+\.\d+\.\d+$/.test(sanitized);
   529→    const ip = isIP ? sanitized : (await run(`dig +short +time=3 +tries=1 ${sanitized} A`, 5000)).split('\n')[0]?.trim() || null;
   530→
   531→    if (!ip && !isIP) {
   532→      return NextResponse.json({ error: `Could not resolve ${sanitized}` }, { status: 400 });
   533→    }
   534→    if (ip && isPrivateIP(ip)) {
   535→      return NextResponse.json({ error: 'Target resolves to a private or reserved IP address' }, { status: 403 });
   536→    }
   537→
   538→    const resolvedIP = ip || sanitized;
   539→    const target = sanitized;
   540→
   541→    // Run all intelligence modules in parallel
   542→    const [
   543→      ipRep,
   544→      c2Infra,
   545→      dnsIndicators,
   546→    ] = await Promise.all([
   547→      ipReputationScan(resolvedIP, target),
   548→      scanC2Ports(resolvedIP, target),
   549→      dnsBotDetection(target),
   550→    ]);
   551→
   552→    // Classify threat
   553→    const threatClassification = classifyThreat(ipRep, c2Infra, dnsIndicators);
   554→
   555→    // Bot Cage data
   556→    const cageStatus = {
   557→      quarantined: threatClassification.score >= 75,
   558→      monitored: threatClassification.score >= 25,
   559→      threats: threatClassification.vectors,
   560→      quarantineReason: threatClassification.score >= 75
   561→        ? `${c2Infra.length} C2 ports + ${dnsIndicators.length} threat indicators + ${ipRep.reputation.blacklistHits} blacklist hits`
   562→        : null,
   563→      responsePlaybook: threatClassification.score >= 75
   564→        ? ['BLOCK: Add to firewall deny list', 'MONITOR: Enable full packet capture', 'INVESTIGATE: Trace all outbound connections', 'CONTAIN: Isolate from network', 'REPORT: Submit to threat intel sharing']
   565→        : threatClassification.score >= 50
   566→        ? ['MONITOR: Enable enhanced logging', 'ALERT: Set tripwires on new connections', 'INVESTIGATE: Analyze traffic patterns', 'PREPARE: Draft containment plan']
   567→        : ['LOG: Standard monitoring', 'WATCH: Alert on unusual patterns'],
   568→    };
   569→
   570→    return applySecurityHeaders(NextResponse.json({
   571→      success: true,
   572→      target,
   573→      resolvedIP,
   574→      mode,
   575→      botIntel: {
   576→        ipReputation: ipRep,
   577→        botnetIndicators: {
   578→          c2Ports: C2_PORTS.map(p => ({ port: p.port, service: p.service, severity: p.severity })),
   579→          malwareSignatures: C2_SIGNATURES.map(s => ({ family: s.family, severity: s.severity, desc: s.desc })),
   580→          detectedC2: c2Infra,
   581→        },
   582→        threatClassification,
   583→        dnsIntelligence: dnsIndicators,
   584→        c2Infrastructure: c2Infra,
   585→        attackVectors: threatClassification.vectors.map((v: string, i: number) => ({
   586→          id: `AV-${String(i + 1).padStart(3, '0')}`,
   587→          type: v,
   588→          severity: i < 2 ? 'critical' : i < 5 ? 'high' : 'medium',
   589→          description: v,
   590→          proof: i < c2Infra.length ? c2Infra[i]?.banner : 'DNS/Reputation analysis',
   591→        })),
   592→        cageStatus,
   593→      },
   594→    }));
   595→  } catch (error) {
   596→    return safeErrorResponse(error, 500, 'bot-hunter');
   597→  }
   598→}
