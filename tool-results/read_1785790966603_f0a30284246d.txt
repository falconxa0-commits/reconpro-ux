   100→
   101→class FindingDetailModal(ModalScreen):
   102→    """Modal overlay showing full finding details."""
   103→
   104→    def __init__(self, finding: Dict[str, Any]) -> None:
   105→        super().__init__()
   106→        self.finding = finding
   107→
   108→    def compose(self) -> ComposeResult:
   109→        f = self.finding
   110→        sev = f.get("severity", "info").lower()
   111→        sev_color = SEV_STYLES.get(sev, "white")
   112→        title = f.get("title", "Unknown")
   113→        category = f.get("category", "")
   114→        module = f.get("module", "")
   115→        pts = f.get("points_deducted", 0)
   116→        evidence = f.get("evidence", "")
   117→        remediation = f.get("remediation", "No remediation available.")
   118→        description = f.get("description", "")
   119→        asset = f.get("asset", "")
   120→        related_cves = f.get("related_cves", [])
   121→
   122→        lines: List[str] = []
   123→        lines.append(f"[{sev_color}][{sev.upper()}] {title}[/]")
   124→        lines.append("")
   125→        if asset:
   126→            lines.append(f"[dim]Asset:[/] [cyan]{asset}[/]")
   127→        if category:
   128→            lines.append(f"[dim]Category:[/] {category}")
   129→        if module:

   131→        lines.append(f"[dim]Points:[/] -{pts}")
   132→        if related_cves:
   133→            cve_ids = ", ".join(
   134→                c.get("cve_id", "") for c in related_cves[:5] if c.get("cve_id")
   135→            )
   136→            if cve_ids:
   137→                lines.append(f"[dim]CVEs:[/] [red]{cve_ids}[/]")
   138→        lines.append("")
   139→        if description:
   140→            lines.append(f"[bold]Description:[/]")
   141→            lines.append(f"[dim]{description}[/]")
   142→            lines.append("")
   143→        if evidence:
   144→            lines.append(f"[bold]Evidence:[/]")
   145→            lines.append(f"[dim]{evidence}[/]")
   146→            lines.append("")
   147→        lines.append(f"[bold]{GREEN}Remediation:[/]")
   148→        lines.append(f"{GREEN}{remediation}[/]")
   149→
   150→        content = "\n".join(lines)
   151→
   152→        with Vertical(classes="modal-container"):

   162→# ════════════════════════════════════════════════════════════════════════════════
   163→# Boot Screen Widget
   164→# ════════════════════════════════════════════════════════════════════════════════
   165→
   166→
   167→class BootScreen(Widget):
   168→    """Full-screen animated boot sequence."""
   169→
   170→    def __init__(self) -> None:
   171→        super().__init__()
   172→        self._step = 0
   173→        self._spinner_idx = 0
   174→        self._module_idx = 0
   175→        self._module_names = list(MODULE_NAMES.values())
   176→        self._done = False
   177→        self._timer: Optional[Timer] = None
   178→
   179→    def compose(self) -> ComposeResult:
   180→        with Vertical(classes="boot-container"):
   181→            yield Static("", id="boot-line-1", classes="boot-title")
   182→            yield Static("", id="boot-line-2", classes="boot-line")
   183→            yield Static("", id="boot-line-3", classes="boot-line")
   184→            yield Static("", id="boot-line-4", classes="boot-line")
   185→
   186→    def on_mount(self) -> None:
   187→        self._timer = self.set_interval(0.04, self._tick)
   188→
   189→    def _tick(self) -> None:
   190→        app = self.app
   191→        if not isinstance(app, NexusApp):
   192→            return
   193→
   194→        line1 = app.query_one("#boot-line-1", Static)
   195→        line2 = app.query_one("#boot-line-2", Static)
   196→        line3 = app.query_one("#boot-line-3", Static)
   197→        line4 = app.query_one("#boot-line-4", Static)
   198→
   199→        if self._step == 0:
   200→            # Letter-by-letter title
   201→            title = "RECONPRO NEXUS"
   202→            chars = title[: self._spinner_idx]
   203→            self._spinner_idx += 1
   204→            line1.update(f"[{CYAN} bold]{chars}[/]")
   205→            if self._spinner_idx > len(title):
   206→                self._spinner_idx = 0
   207→                self._step = 1
   208→                line1.update(f"[{CYAN} bold]{title}[/]  [{DIM_CYAN}]v{VERSION}[/]")
   209→
   210→        elif self._step == 1:
   211→            # Spinner for "Initializing agent..."
   212→            frame = SPINNER_FRAMES[self._spinner_idx % len(SPINNER_FRAMES)]
   213→            self._spinner_idx += 1
   214→            line2.update(f"  [{CYAN}]{frame}[/] [{DIM_CYAN}]Initializing agent...[/]")
   215→            if self._spinner_idx > 30:
   216→                self._spinner_idx = 0
   217→                self._step = 2
   218→                line2.update(f"  [{GREEN}]✓[/] [{DIM_CYAN}]Agent initialized.[/]")
   219→
   220→        elif self._step == 2:
   221→            # Module names appearing one by one
   222→            if self._module_idx < len(self._module_names):
   223→                visible = self._module_names[: self._module_idx + 1]
   224→                self._module_idx += 1
   225→                modules_str = "  ".join(f"[{CYAN}]○[/] [{DIM_CYAN}]{m}[/]" for m in visible)
   226→                remaining = len(self._module_names) - self._module_idx
   227→                if remaining > 0:
   228→                    modules_str += f"  [{DIM_CYAN}]+{remaining} more...[/]"
   229→                frame = SPINNER_FRAMES[self._spinner_idx % len(SPINNER_FRAMES)]
   230→                self._spinner_idx += 1
   231→                line3.update(
   232→                    f"  [{CYAN}]{frame}[/] [{DIM_CYAN}]Loading {len(self._module_names)} modules...[/]\n  {modules_str}"
   233→                )
   234→            else:
   235→                all_mods = "  ".join(
   236→                    f"[{GREEN}]●[/] [{DIM_CYAN}]{m}[/]" for m in self._module_names
   237→                )
   238→                line3.update(
   239→                    f"  [{GREEN}]✓[/] [{DIM_CYAN}]All {len(self._module_names)} modules loaded.[/]\n  {all_mods}"
   240→                )
   241→                self._step = 3
   242→                self._spinner_idx = 0
   243→
   244→        elif self._step == 3:
   245→            # "Ready." with flash
   246→            if self._spinner_idx == 0:
   247→                line4.update(f"[{YELLOW} bold]  ▶  READY.[/]")
   248→            elif self._spinner_idx == 6:
   249→                line4.update(f"[{GREEN} bold]  ▶  READY.[/]")
   250→            elif self._spinner_idx >= 10:
   251→                self._done = True
   252→                if self._timer:
   253→                    self._timer.stop()
   254→                self.app.call_from_thread(self.app._finish_boot)
   255→            self._spinner_idx += 1
   256→

   258→# ════════════════════════════════════════════════════════════════════════════════
   259→# Module Status Cell Widget
   260→# ════════════════════════════════════════════════════════════════════════════════
   261→
   262→
   263→class ModuleCell(Static):
   264→    """Single module status cell in the grid."""
   265→
   266→    def __init__(self, module_id: str, name: str) -> None:
   267→        super().__init__()
   268→        self.module_id = module_id
   269→        self.module_name = name
   270→        self.status = "idle"  # idle | scanning | done | error
   271→        self.finding_count = 0
   272→        self._spinner_idx = 0
   273→        self._timer: Optional[Timer] = None
   274→
   275→    def on_mount(self) -> None:
   276→        self._update_display()
   277→        self._timer = self.set_interval(0.1, self._spin)
   278→
   279→    def _spin(self) -> None:
   280→        if self.status == "scanning":
   281→            self._spinner_idx = (self._spinner_idx + 1) % len(SPINNER_FRAMES)
   282→            self._update_display()
   283→
   284→    def _update_display(self) -> None:
   285→        name_short = self.module_name[:14]
   286→        if self.status == "idle":
   287→            icon = f"[{DIM_CYAN}]○[/]"
   288→            status_text = f"[{DIM_CYAN}]idle[/]"
   289→        elif self.status == "scanning":
   290→            frame = SPINNER_FRAMES[self._spinner_idx]
   291→            icon = f"[{CYAN}]{frame}[/]"
   292→            status_text = f"[{CYAN}]scanning[/]"
   293→        elif self.status == "done":
   294→            icon = f"[{GREEN}]✓[/]"
   295→            status_text = f"[{GREEN}]done ({self.finding_count})[/]"
   296→        elif self.status == "error":
   297→            icon = f"[{RED}]✗[/]"
   298→            status_text = f"[{RED}]error[/]"
   299→        else:
   300→            icon = "○"
   301→            status_text = "idle"
   302→
   303→        self.update(
   304→            f"{icon} [{CYAN} bold]{name_short}[/]\n  {status_text}"
   305→        )
   306→
   307→    def set_status(self, status: str, count: int = 0) -> None:
   308→        self.status = status
   309→        self.finding_count = count
   310→        self._update_display()
   311→
   312→    def on_unmount(self) -> None:
   313→        if self._timer:
   314→            self._timer.stop()
   315→

   318→# Pulsing Status Dot Widget
   319→# ════════════════════════════════════════════════════════════════════════════════
   320→
   321→
   322→class StatusDot(Static):
   323→    """Pulsing dot indicator for scan status."""
   324→
   325→    scanning: reactive[bool] = reactive(False)
   326→
   327→    def __init__(self) -> None:
   328→        super().__init__("")
   329→        self._phase = 0
   330→
   331→    def on_mount(self) -> None:
   332→        self.set_interval(0.3, self._pulse)
   333→
   334→    def watch_scanning(self, old: bool, new: bool) -> None:
   335→        # Update visual state based on scanning status
   336→        if new:
   337→            self.update(f"[{self._get_pulse_color()}]●[/]")
   338→        else:
   339→            self.update(f"[{GREEN}]●[/]")
   340→
   341→    def _get_pulse_color(self) -> str:
   342→        return ["#003322", "#006644", CYAN, "#006644"][self._phase] if self.scanning else GREEN
   343→
   344→    def _pulse(self) -> None:
   345→        self._phase = (self._phase + 1) % 4
   346→        if self.scanning:
   347→            brightness = self._get_pulse_color()
   348→            self.update(f"[{brightness}]●[/]")
   349→        else:
   350→            self.update(f"[{GREEN}]●[/]")

   879→    def on_key(self, event: Key) -> None:
   880→        """Handle key events — boot skip + completer + history + vim-nav."""
   881→        # Skip boot on ANY key press during boot sequence
   882→        if not self._boot_done and not self._boot_skipped:
   883→            self._boot_skipped = True
   884→            try:
   885→                boot = self.query_one("#boot-screen", BootScreen)
   886→                if boot._timer:
   887→                    boot._timer.stop()
   888→            except NoMatches:
   889→                pass
   890→            self._finish_boot()
   891→            return
   892→
   893→        # ── Number keys: quick jump to panels (0=input, 1=chat, 2=findings, 3=modules) ──
   894→        if event.key in ("0", "1", "2", "3") and not self._input_focused():
   895→            jump_map = {"0": "#command-input", "1": "#chat-log", "2": "#findings-feed", "3": "#module-grid"}
   896→            # Skip panel jump to collapsed panels
   897→            target_sel = jump_map[event.key]
   898→            if event.key == "1" and self._left_collapsed:
   899→                target_sel = "#findings-feed"
   900→            if event.key == "3" and self._modules_collapsed:
   901→                target_sel = "#findings-feed"
   902→            self._focus_widget(target_sel)
   903→            event.stop()
   904→            return
   905→
   906→        # ── Phase D: [ and ] to toggle panel collapse ──
   907→        if event.key == "[" and not self._input_focused():
   908→            self._toggle_left_panel()
   909→            event.stop()
   910→            return
   911→        if event.key == "]" and not self._input_focused():
   912→            self._toggle_right_panel()
   913→            event.stop()
   914→            return
   915→        if event.key == "=" and not self._input_focused():
   916→            self._toggle_modules()
   917→            event.stop()
   918→            return
   919→        # ── Phase D: | resets split to auto, \\ resets to 50:50 ──
   920→        if event.key == "|" and not self._input_focused():
   921→            self._user_manually_resized = False
   922→            self._apply_responsive_layout(self.size.width, self.size.height)
   923→            event.stop()
   924→            return
   925→        if event.key == "\\" and not self._input_focused():
   926→            self._user_manually_resized = True
   927→            self._split_ratio = 50
   928→            self._apply_split_ratio()
   929→            event.stop()
   930→            return
   931→
   932→        # ── Phase E: ? opens help overlay ──
   933→        if event.key == "question_mark" and not self._input_focused():
   934→            self._show_help_overlay()
   935→            event.stop()
   936→            return
   937→
   938→        # ── Tab: completer accept/cycle / quick-pick on empty / cycle focus ──
   939→        if event.key == "tab":
   940→            try:
   941→                inp = self.query_one("#command-input", Input)
   942→                comp = self.query_one("#cmd-completer", CommandCompleter)
   943→                if inp.has_focus:
   944→                    # Completer visible → accept top suggestion
   945→                    if comp.visible:
   946→                        accepted = comp.accept_top()
   947→                        if accepted:
   948→                            self._apply_completion(inp, comp, accepted)
   949→                        event.stop()
   950→                        return
   951→                    # Empty input → quick-pick menu
   952→                    if not inp.value.strip():
   953→                        comp.show_quick_pick()
   954→                        event.stop()
   955→                        return
   956→            except NoMatches:
   957→                pass
   958→            # Fall through to default Tab behavior (cycle focus)
   959→            return
   960→
   961→        # ── Escape: dismiss completer, close modal, or jump to input ──
   962→        if event.key == "escape":
   963→            try:
   964→                comp = self.query_one("#cmd-completer", CommandCompleter)
   965→                if comp.visible:
   966→                    comp.hide()
   967→                    event.stop()
   968→                    return
   969→            except NoMatches:
   970→                pass
   971→            # If any modal screen is open, Escape will dismiss it via Textual
   972→            # Otherwise jump back to input
   973→            if not self._input_focused():
   974→                self._focus_input()
   975→                event.stop()
   976→                return
   977→
   978→        # ── Up/Down in completer: cycle suggestions ──
   979→        if event.key in ("up", "down"):
   980→            try:
   981→                comp = self.query_one("#cmd-completer", CommandCompleter)
   982→                inp = self.query_one("#command-input", Input)
   983→                if comp.visible and inp.has_focus:
   984→                    direction = -1 if event.key == "up" else 1
   985→                    comp.cycle_selection(direction)
   986→                    event.stop()
   987→                    return
   988→            except NoMatches:
   989→                pass
   990→
   991→        # ── j/k vim-style navigation: findings feed + chat log ──
   992→        if event.key in ("j", "k"):
   993→            # Findings feed navigation
   994→            try:
   995→                feed = self.query_one("#findings-feed", RichLog)
   996→                if feed.has_focus and self._findings_list:
   997→                    self._vim_navigate_findings(event.key)
   998→                    event.stop()
   999→                    return
  1000→            except NoMatches:
  1001→                pass
  1002→            # Chat log scroll (j=down, k=up)
  1003→            try:
  1004→                chat = self.query_one("#chat-log", RichLog)
  1005→                if chat.has_focus:
  1006→                    if event.key == "j":
  1007→                        chat.scroll_relative(3)
  1008→                    else:
  1009→                        chat.scroll_relative(-3)
  1010→                    event.stop()
  1011→                    return
  1012→            except NoMatches:
  1013→                pass
  1014→
  1015→        # ── Arrow keys on module grid: navigate cells ──
  1016→        if event.key in ("up", "down", "left", "right"):
  1017→            try:
  1018→                grid = self.query_one("#module-grid", Container)
  1019→                if grid.has_focus:
  1020→                    self._arrow_navigate_modules(event.key)
  1021→                    event.stop()
  1022→                    return
  1023→            except NoMatches:
  1024→                pass
  1025→
  1026→        # ── Enter on findings: open detail modal ──
  1027→        if event.key == "enter":
  1028→            try:
  1029→                feed = self.query_one("#findings-feed", RichLog)
  1030→                if feed.has_focus and self._findings_list:
  1031→                    idx = self._findings_cursor if 0 <= self._findings_cursor < len(self._findings_list) else len(self._findings_list) - 1
  1032→                    self.push_screen(FindingDetailModal(self._findings_list[idx]))
  1033→                    event.stop()
  1034→                    return
  1035→            except NoMatches:
  1036→                pass
  1037→
  1038→        # Command history navigation
  1039→        if event.key == "up" and self._history_index > 0:
  1040→            try:
  1041→                inp = self.query_one("#command-input", Input)
  1042→                if inp.has_focus:
  1043→                    self._history_index -= 1
  1044→                    inp.value = self._command_history[self._history_index]
  1045→            except NoMatches:
  1046→                pass
  1047→        elif event.key == "down":
  1048→            try:
  1049→                inp = self.query_one("#command-input", Input)
  1050→                if inp.has_focus:
  1051→                    if self._history_index < len(self._command_history) - 1:
  1052→                        self._history_index += 1
  1053→                        inp.value = self._command_history[self._history_index]
  1054→                    else:
  1055→                        self._history_index = len(self._command_history)
  1056→                        inp.value = ""
  1057→            except NoMatches:
  1058→                pass

  1125→    def _apply_completion(self, inp: Input, comp: CommandCompleter, accepted: str) -> None:
  1126→        """Apply a completion suggestion to the input field.
  1127→
  1128→        Phase C: handles both command-level and arg-level completions.
  1129→        For arg mode, appends/replaces the last word instead of the first.
  1130→        """
  1131→        parts = inp.value.split()
  1132→
  1133→        if comp._mode == "command":
  1134→            # Replace the first word with the accepted command
  1135→            rest = parts[1:] if len(parts) > 1 else []
  1136→            inp.value = accepted + (" " + " ".join(rest) if rest else "")
  1137→        else:
  1138→            # Arg-level: replace the last word
  1139→            if len(parts) >= 2:
  1140→                parts[-1] = accepted
  1141→                inp.value = " ".join(parts)
  1142→            else:
  1143→                inp.value = accepted
  1144→
  1145→        inp.cursor_position = len(inp.value)

  1147→    def _vim_navigate_findings(self, direction: str) -> None:

  1173→            sev = finding.get("severity", "info").lower()
  1174→            title = finding.get("title", "?")[:40]

  1184→    def _arrow_navigate_modules(self, direction: str) -> None:
  1185→        """Navigate module cells with arrow keys when module-grid has focus."""
  1186→        module_keys = list(self._module_cells.keys())
  1187→        if not module_keys:
  1188→            return
  1189→
  1190→        # Current focused module index
  1191→        current_idx = 0
  1192→        for i, key in enumerate(module_keys):
  1193→            try:
  1194→                if self._module_cells[key].has_focus:
  1195→                    current_idx = i
  1196→                    break
  1197→            except Exception:
  1198→                pass
  1199→
  1200→        new_idx = current_idx
  1201→        if direction == "right":
  1202→            new_idx = min(current_idx + 1, len(module_keys) - 1)
  1203→        elif direction == "left":
  1204→            new_idx = max(current_idx - 1, 0)
  1205→        elif direction == "down":
  1206→            new_idx = min(current_idx + self._GRID_COLS, len(module_keys) - 1)
  1207→        elif direction == "up":
  1208→            new_idx = max(current_idx - self._GRID_COLS, 0)
  1209→
  1210→        if new_idx != current_idx:
  1211→            key = module_keys[new_idx]
  1212→            self._module_cells[key].focus()
  1213→            # Show module hint
  1214→            mod_name = MODULE_NAMES.get(key, key)
  1215→            try:
  1216→                el = self.query_one("#bottom-info", Label)
  1217→                el.update(f"[{CYAN}]{mod_name}[/{CYAN}] [{MUTED}]{key}[/{MUTED}]")
  1218→            except NoMatches:
  1219→                pass

  1671→    def on_resize(self, event) -> None:
  1672→        """Adapt layout to terminal size changes.
  1673→
  1674→        Responsive breakpoints:
  1675→          - < 80 cols:  compact mode (stack panels vertically, 2-col grid)
  1676→          - 80–119 cols: standard (3-col grid, default 40:60 split)
  1677→          - 120–159 cols: wide (3-col grid, expanded stats)
  1678→          - >= 160 cols:  ultra-wide (4-col grid, 35:65 split)
  1679→          - >= 200 cols:  cinematic (4-col grid, 30:70 split)
  1680→        """
  1681→        # Debounce to avoid flicker during rapid resize
  1682→        if self._resize_timer is not None:
  1683→            self._resize_timer.stop()
  1684→        self._resize_timer = self.set_timer(0.15, lambda: self._apply_responsive_layout(event.width, event.height))

  1686→    def _apply_responsive_layout(self, width: int, height: int) -> None:
  1687→        """Apply layout changes based on terminal dimensions.
  1688→
  1689→        Dynamically syncs ``_GRID_COLS`` with the actual CSS grid columns
  1690→        so arrow-key navigation on the module grid stays correct.
  1691→        Adjusts split ratio defaults for wider terminals.
  1692→        """
  1693→        try:
  1694→            content = self.query_one("#content", Widget)
  1695→            grid = self.query_one("#module-grid", Container)
  1696→        except NoMatches:
  1697→            return
  1698→
  1699→        # ── Compact mode: < 80 cols ──
  1700→        if width < 80:
  1701→            if not self._compact_mode:
  1702→                content.set_class(True, "compact")
  1703→                grid.set_class(True, "grid-cols-2")
  1704→                grid.remove_class("grid-cols-4")
  1705→                self._compact_mode = True
  1706→                self._GRID_COLS = 2
  1707→                # Force-collapse left panel in very narrow terminals
  1708→                if width < 60 and not self._left_collapsed:
  1709→                    self._left_collapsed_before_compact = True
  1710→                    self._toggle_left_panel()
  1711→                self._flash_split_indicator(f"COMPACT {width}x{height}")
  1712→            return
  1713→
  1714→        # ── Ultra-wide: >= 160 cols ──
  1715→        if width >= 160:
  1716→            if self._compact_mode:
  1717→                content.remove_class("compact")
  1718→                self._compact_mode = False
  1719→                # Restore auto-collapsed left panel
  1720→                if getattr(self, '_left_collapsed_before_compact', False) and self._left_collapsed:
  1721→                    self._toggle_left_panel()
  1722→                    self._left_collapsed_before_compact = False
  1723→            grid.set_class(True, "grid-cols-4")
  1724→            grid.remove_class("grid-cols-2")
  1725→            self._GRID_COLS = 4
  1726→            # Auto-adjust split for cinematic widths
  1727→            if width >= 200:
  1728→                optimal_split = 30
  1729→            else:
  1730→                optimal_split = 35
  1731→            # Only auto-adjust if user hasn't manually set split
  1732→            if not self._user_manually_resized:
  1733→                self._split_ratio = optimal_split
  1734→                self._apply_split_ratio_silent()
  1735→            label = "CINEMATIC" if width >= 200 else "ULTRA"
  1736→            self._flash_split_indicator(f"{label} {width}x{height}")
  1737→            return
  1738→
  1739→        # ── Wide: 120–159 cols ──
  1740→        if width >= 120:
  1741→            if self._compact_mode:
  1742→                content.remove_class("compact")
  1743→                self._compact_mode = False
  1744→                if getattr(self, '_left_collapsed_before_compact', False) and self._left_collapsed:
  1745→                    self._toggle_left_panel()
  1746→                    self._left_collapsed_before_compact = False
  1747→            grid.remove_class("grid-cols-4")
  1748→            grid.remove_class("grid-cols-2")
  1749→            self._GRID_COLS = 3
  1750→            if not self._user_manually_resized:
  1751→                self._split_ratio = 38
  1752→                self._apply_split_ratio_silent()
  1753→            self._flash_split_indicator(f"WIDE {width}x{height}")
  1754→            return
  1755→
  1756→        # ── Standard: 80–119 cols ──
  1757→        if self._compact_mode:
  1758→            content.remove_class("compact")
  1759→            self._compact_mode = False
  1760→            if getattr(self, '_left_collapsed_before_compact', False) and self._left_collapsed:
  1761→                self._toggle_left_panel()
  1762→                self._left_collapsed_before_compact = False
  1763→        grid.remove_class("grid-cols-4")
  1764→        grid.remove_class("grid-cols-2")
  1765→        self._GRID_COLS = 3
  1766→        if not self._user_manually_resized:
  1767→            self._split_ratio = 40
  1768→            self._apply_split_ratio_silent()
