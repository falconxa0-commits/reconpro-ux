   442→        (r"information.*disclos|info.*leak", "info_disclosure", 0.80),
   443→        (r"injection", "injection", 0.80),
   444→    ]
   445→
   446→    _MODULE_RULES: Dict[str, str] = {
   447→        "chain": "ssrf",
   448→        "auth": "auth_bypass",
   449→        "quantum_fingerprint": "tech_fingerprint",
   450→        "dark_web_monitor": "dark_web",
   451→        "info_ops": "info_disclosure",
   452→        "steganography_detector": "steganography",
   453→        "covert_channel": "covert_channel",
   454→        "zero_day_hunter": "cve",
   455→        "infrastructure_ghost": "infrastructure",
   456→        "signal_intelligence": "beaconing",
   457→        "nation_state_attributor": "infrastructure",
   458→        "weaponized_report": "supply_chain",
   459→        "honeypot_dance": "honeypot",
   460→        "dead_drop": "dead_drop",
   461→        "pegasus": "credential_theft",
   462→        "cloud_recon": "cloud_infrastructure",
   463→        "container_sec": "container",
   464→        "iac_audit": "misconfiguration",
   465→        "ast_analyzer": "supply_chain",
   466→    }
   467→
   468→    def classify(self, finding: Dict[str, Any]) -> ClassificationResult:
   469→        """Classify a single finding into an attack category.
   470→
   471→        Returns ClassificationResult with attack_category, confidence, and
   472→        matching indicators.
   473→        """
   474→        title = finding.get("title", "").lower()
   475→        description = finding.get("description", "").lower()
   476→        evidence = finding.get("evidence", "").lower()
   477→        category = finding.get("category", "").lower()
   478→        module = finding.get("module", "").lower()
   479→        severity = finding.get("severity", "info").lower()
   480→
   481→        scores: Dict[str, float] = defaultdict(float)
   482→        indicators: Dict[str, List[str]] = defaultdict(list)
   483→
   484→        # Phase 1: Title pattern matching
   485→        combined = f"{title} {description}"
   486→        for pattern, cat, weight in self._TITLE_RULES:
   487→            if re.search(pattern, combined, re.IGNORECASE):
   488→                scores[cat] += weight
   489→                indicators[cat].append(f"title_match:{pattern[:20]}")
   490→
   491→        # Phase 2: Module-based classification
   492→        if module in self._MODULE_RULES:
   493→            cat = self._MODULE_RULES[module]
   494→            scores[cat] += 0.75
   495→            indicators[cat].append(f"module_source:{module}")
   496→
   497→        # Phase 3: Category hint
   498→        if category:
   499→            for pattern, cat, weight in self._TITLE_RULES:
   500→                if re.search(pattern, category, re.IGNORECASE):
   501→                    scores[cat] += 0.5
   502→                    indicators[cat].append(f"category_hint:{category}")
   503→
   504→        # Phase 4: Evidence enrichment
   505→        for pattern, cat, weight in self._TITLE_RULES:
   506→            if re.search(pattern, evidence, re.IGNORECASE):
   507→                scores[cat] += 0.3
   508→                indicators[cat].append(f"evidence_match:{pattern[:20]}")
   509→
   510→        if not scores:
   511→            return ClassificationResult(
   512→                attack_category="misc",
   513→                confidence=0.3,
   514→                indicators=["no_patterns_matched"],
   515→            )
   516→
   517→        # Select best category
   518→        best_cat = max(scores, key=scores.get)
   519→        best_score = scores[best_cat]
   520→
   521→        # Normalize confidence: weight by evidence strength
   522→        has_evidence = len(evidence) > 10
   523→        has_description = len(description) > 20
   524→        evidence_boost = 0.15 if has_evidence else 0.0
   525→        desc_boost = 0.1 if has_description else 0.0
   526→        severity_boost = _sev_val(severity) / 100.0
   527→
   528→        raw_conf = min(1.0, best_score / 2.0 + evidence_boost + desc_boost + severity_boost)
   529→
   530→        return ClassificationResult(
   531→            attack_category=best_cat,
   532→            confidence=round(raw_conf, 2),
   533→            indicators=indicators[best_cat][:5],
   534→        )
   535→
   536→    def classify_batch(self, findings: List[Dict[str, Any]]) -> List[ClassificationResult]:
   537→        """Classify a batch of findings."""
   538→        return [self.classify(f) for f in findings]

   567→
   568→class FindingCorrelator:
   569→    """Correlates findings across modules to detect relationships.
   570→
   571→    Groups findings by shared assets, detects duplicates, and identifies
   572→    attack chains.
   573→    """
   574→
   575→    def correlate(self, findings: List[Dict[str, Any]]) -> Tuple[List[CorrelationGroup], List[Dict[str, Any]]]:
   576→        """Correlate findings. Returns (correlation_groups, deduplicated_findings)."""
   577→        if not findings:
   578→            return [], []
   579→
   580→        # Phase 1: Deduplicate
   581→        unique, dedup_count = self._deduplicate(findings)
   582→
   583→        # Phase 2: Group by shared asset
   584→        asset_groups = self._group_by_asset(unique)
   585→
   586→        # Phase 3: Detect attack chains
   587→        chain_groups = self._detect_chains(unique)
   588→
   589→        # Merge asset groups with chain groups (avoid duplicates)
   590→        all_groups = self._merge_groups(asset_groups, chain_groups)
   591→
   592→        return all_groups, unique
   593→
   594→    def _deduplicate(self, findings: List[Dict[str, Any]]) -> Tuple[List[Dict[str, Any]], int]:
   595→        """Remove duplicate findings based on title + asset + category."""
   596→        seen: Dict[str, Dict[str, Any]] = {}
   597→        dup_count = 0
   598→        for f in findings:
   599→            key = self._finding_hash(f)
   600→            if key in seen:
   601→                dup_count += 1
   602→                # Keep the one with more evidence
   603→                existing = seen[key]
   604→                if len(f.get("evidence", "")) > len(existing.get("evidence", "")):
   605→                    seen[key] = f
   606→            else:
   607→                seen[key] = f
   608→        return list(seen.values()), dup_count
   609→
   610→    @staticmethod
   611→    def _finding_hash(f: Dict[str, Any]) -> str:
   612→        """Create a deterministic hash for dedup."""
   613→        raw = f"{f.get('title', '')}:{f.get('asset', '')}:{f.get('category', '')}".lower().strip()
   614→        return hashlib.sha256(raw.encode()).hexdigest()[:16]
   615→
   616→    def _group_by_asset(self, findings: List[Dict[str, Any]]) -> List[CorrelationGroup]:
   617→        """Group findings that share the same asset."""
   618→        groups: Dict[str, List[Dict[str, Any]]] = defaultdict(list)
   619→        for f in findings:
   620→            asset = f.get("asset", "unknown")
   621→            groups[asset].append(f)
   622→
   623→        result: List[CorrelationGroup] = []
   624→        for asset, items in groups.items():
   625→            if len(items) < 2:
   626→                continue
   627→            sevs = [f.get("severity", "info") for f in items]
   628→            combined_risk = self._compute_combined_risk(sevs)
   629→            result.append(CorrelationGroup(
   630→                findings=items,
   631→                correlation_type="shared_asset",
   632→                combined_risk=combined_risk,
   633→                combined_confidence=0.7,
   634→            ))
   635→        return result
   636→
   637→    def _detect_chains(self, findings: List[Dict[str, Any]]) -> List[CorrelationGroup]:
   638→        """Detect attack chains using templates and category analysis."""
   639→        MAX_PAIRS = 50000
   640→        # Classify all findings
   641→        classifier = FindingClassifier()
   642→        classified: List[Tuple[Dict[str, Any], ClassificationResult]] = []
   643→        for f in findings:
   644→            cr = classifier.classify(f)
   645→            classified.append((f, cr))
   646→
   647→        chains: List[CorrelationGroup] = []
   648→        # Check all pairs against chain templates (capped to prevent O(n²) exhaustion)
   649→        pair_count = 0
   650→        for i, j in combinations(range(len(classified)), 2):
   651→            if pair_count >= MAX_PAIRS:
   652→                break
   653→            f_a, cr_a = classified[i]
   654→            f_b, cr_b = classified[j]
   655→            pair_count += 1
   656→            for cat_a, cat_b, chain_name in _ATTACK_CHAIN_TEMPLATES:
   657→                if (cr_a.attack_category == cat_a and cr_b.attack_category == cat_b) or \
   658→                   (cr_b.attack_category == cat_a and cr_a.attack_category == cat_b):
   659→                    # Verify shared asset or related
   660→                    shared = (f_a.get("asset", "") == f_b.get("asset", "") or
   661→                              self._assets_related(f_a.get("asset", ""), f_b.get("asset", "")))
   662→                    if shared:
   663→                        conf = min(cr_a.confidence, cr_b.confidence) * 0.8
   664→                        chains.append(CorrelationGroup(
   665→                            findings=[f_a, f_b],
   666→                            correlation_type="attack_chain",
   667→                            attack_chain=[chain_name],
   668→                            combined_risk=self._max_severity([f_a, f_b]),
   669→                            combined_confidence=round(conf, 2),
   670→                        ))
   671→        return chains
   672→
   673→    @staticmethod
   674→    def _assets_related(a: str, b: str) -> bool:
   675→        """Check if two assets are related (same domain, subdomain)."""
   676→        a, b = a.lower(), b.lower()
   677→        if a == b:
   678→            return True
   679→        # Same base domain
   680→        parts_a = a.replace("www.", "").split(".")
   681→        parts_b = b.replace("www.", "").split(".")
   682→        if len(parts_a) >= 2 and len(parts_b) >= 2:
   683→            return parts_a[-2:] == parts_b[-2:]
   684→        return False
   685→
   686→    @staticmethod
   687→    def _compute_combined_risk(severities: List[str]) -> str:
   688→        """Compute combined risk from multiple severities."""
   689→        if "critical" in severities:
   690→            return "critical"
   691→        if "high" in severities:
   692→            return "high"
   693→        if any(s == "medium" for s in severities) and len(severities) >= 2:
   694→            return "high"
   695→        return max(severities, key=lambda s: _SEVERITY_WEIGHTS.get(s, 1.0)) if severities else "low"
   696→
   697→    @staticmethod
   698→    def _max_severity(findings: List[Dict[str, Any]]) -> str:
   699→        sevs = [f.get("severity", "info") for f in findings]
   700→        for s in ["critical", "high", "medium", "low", "info"]:
   701→            if s in sevs:
   702→                return s
   703→        return "info"
   704→
   705→    def _merge_groups(self, asset_groups: List[CorrelationGroup],
   706→                      chain_groups: List[CorrelationGroup]) -> List[CorrelationGroup]:
   707→        """Merge asset groups and chain groups, preferring chains."""
   708→        seen_sets: set = set()
   709→        result = list(chain_groups)
   710→        chain_keys = set()
   711→        for cg in chain_groups:
   712→            for f in cg.findings:
   713→                chain_keys.add(self._finding_hash(f))
   714→
   715→        for ag in asset_groups:
   716→            # Only add asset groups that don't overlap with existing chain findings
   717→            ag_keys = {self._finding_hash(f) for f in ag.findings}
   718→            if not ag_keys & chain_keys:
   719→                result.append(ag)
   720→
   721→        return result

   728→class ExploitabilityEstimator:
   729→    """Estimates exploit difficulty based on finding characteristics."""
   730→
   731→    # Category-based exploitability baselines
   732→    _CATEGORY_EXPLOITABILITY: Dict[str, float] = {
   733→        "sqli": 0.85, "xss": 0.80, "rce": 0.90, "ssrf": 0.75,
   734→        "auth_bypass": 0.70, "privilege_escalation": 0.75,
   735→        "credential_theft": 0.65, "data_exposure": 0.70,
   736→        "misconfiguration": 0.60, "crypto_failure": 0.55,
   737→        "dos": 0.80, "default_credentials": 0.90,
   738→        "open_redirect": 0.85, "jwt_exposure": 0.75,
   739→        "sensitive_api": 0.70, "injection": 0.80,
   740→        "xxe": 0.70, "supply_chain": 0.65,
   741→    }
   742→
   743→    def estimate(self, finding: Dict[str, Any]) -> Dict[str, Any]:
   744→        """Estimate exploitability metrics for a finding.
   745→
   746→        Returns dict with exploitability, attacker_effort, likelihood, confidence.
   747→        """
   748→        severity = finding.get("severity", "info").lower()
   749→        category = finding.get("category", "").lower()
   750→        evidence = finding.get("evidence", "")
   751→        title = finding.get("title", "").lower()
   752→        description = finding.get("description", "").lower()
   753→        dread = finding.get("dread_score", 0.0)
   754→        module = finding.get("module", "").lower()
   755→
   756→        # Classify for category-based baseline
   757→        classifier = FindingClassifier()
   758→        cls = classifier.classify(finding)
   759→        cat = cls.attack_category
   760→
   761→        # Base exploitability from severity + category
   762→        sev_exploit = _sev_val(severity) / 10.0
   763→        cat_exploit = self._CATEGORY_EXPLOITABILITY.get(cat, 0.5)
   764→        base_exploitability = (sev_exploit * 0.4 + cat_exploit * 0.4 + min(dread / 10.0, 1.0) * 0.2)
   765→
   766→        # Evidence strength boost
   767→        evidence_strength = min(len(evidence) / 200.0, 1.0)
   768→        description_strength = min(len(description) / 150.0, 1.0)
   769→        evidence_boost = (evidence_strength * 0.3 + description_strength * 0.2)
   770→
   771→        exploitability = min(1.0, base_exploitability + evidence_boost * 0.15)
   772→
   773→        # Attacker effort (inverse of exploitability)
   774→        if exploitability >= 0.8:
   775→            effort = "low"
   776→        elif exploitability >= 0.6:
   777→            effort = "medium"
   778→        elif exploitability >= 0.4:
   779→            effort = "high"
   780→        else:
   781→            effort = "very_high"
   782→
   783→        # Likelihood
   784→        likelihood = min(1.0, exploitability * 0.8 + evidence_boost * 0.2)
   785→
   786→        # Confidence
   787→        confidence = min(1.0, cls.confidence * 0.6 + evidence_strength * 0.3 + 0.1)
   788→
   789→        return {
   790→            "exploitability": round(exploitability, 2),
   791→            "attacker_effort": effort,
   792→            "likelihood": round(likelihood, 2),
   793→            "confidence": round(confidence, 2),
   794→        }

   895→class AttackPathDetector:
   896→    """Detects attack paths across findings.
   897→
   898→    Chains individual findings into multi-step attack sequences
   899→    and estimates path complexity, likelihood, and impact.
   900→    """
   901→
   902→    def detect(self, findings: List[Dict[str, Any]]) -> List[AttackPath]:
   903→        """Detect attack paths from a list of findings.
   904→
   905→        Returns a list of AttackPath objects sorted by confidence.
   906→        """
   907→        if len(findings) < 2:
   908→            return []
   909→
   910→        classifier = FindingClassifier()
   911→        classified: List[Tuple[Dict[str, Any], ClassificationResult]] = []
   912→        for f in findings:
   913→            cr = classifier.classify(f)
   914→            classified.append((f, cr))
   915→
   916→        paths: List[AttackPath] = []
   917→
   918→        # Check multi-step chain templates
   919→        for chain_cats, name, risk in _MULTI_STEP_CHAINS:
   920→            matching = self._find_matching_sequence(chain_cats, classified)
   921→            if matching and len(matching) >= 2:
   922→                path = self._build_path(matching, name, risk)
   923→                if path:
   924→                    paths.append(path)
   925→
   926→        # Detect two-step pairs
   927→        for cat_a, cat_b, chain_name in _ATTACK_CHAIN_TEMPLATES:
   928→            matching = self._find_pair(cat_a, cat_b, classified)
   929→            if matching:
   930→                path = self._build_two_step_path(matching, chain_name)
   931→                if path:
   932→                    paths.append(path)
   933→
   934→        # Sort by confidence
   935→        paths.sort(key=lambda p: p.confidence, reverse=True)
   936→        return paths
   937→
   938→    def _find_matching_sequence(
   939→        self, chain_cats: List[str],
   940→        classified: List[Tuple[Dict[str, Any], ClassificationResult]],
   941→    ) -> List[Dict[str, Any]]:
   942→        """Find findings that match a chain template sequence."""
   943→        matched: List[Dict[str, Any]] = []
   944→        for cat in chain_cats:
   945→            for f, cr in classified:
   946→                if cr.attack_category == cat and f not in matched:
   947→                    matched.append(f)
   948→                    break
   949→        return matched
   950→
   951→    def _find_pair(
   952→        self, cat_a: str, cat_b: str,
   953→        classified: List[Tuple[Dict[str, Any], ClassificationResult]],
   954→    ) -> Optional[Tuple[Dict[str, Any], Dict[str, Any]]]:
   955→        """Find a pair of findings matching a two-step chain."""
   956→        a_match = None
   957→        b_match = None
   958→        for f, cr in classified:
   959→            if cr.attack_category == cat_a and a_match is None:
   960→                a_match = f
   961→            elif cr.attack_category == cat_b and b_match is None:
   962→                b_match = f
   963→            if a_match and b_match:
   964→                return (a_match, b_match)
   965→        return None
   966→
   967→    def _build_path(self, findings: List[Dict[str, Any]], name: str, risk: str) -> Optional[AttackPath]:
   968→        """Build an AttackPath from a matched finding sequence."""
   969→        if not findings:
   970→            return None
   971→
   972→        steps = [
   973→            {
   974→                "title": f.get("title", ""),
   975→                "category": f.get("category", ""),
   976→                "severity": f.get("severity", "info"),
   977→                "asset": f.get("asset", ""),
   978→            }
   979→            for f in findings
   980→        ]
   981→
   982→        # Compute aggregate metrics
   983→        severities = [f.get("severity", "info") for f in findings]
   984→        avg_exploit = sum(_sev_val(s) for s in severities) / len(severities) / 10.0
   985→        likelihood = min(1.0, avg_exploit * 0.7 + 0.2)
   986→        impact = _sev_val(risk if risk in _SEVERITY_WEIGHTS else "high") / 10.0
   987→
   988→        # Complexity based on chain length
   989→        if len(findings) >= 4:
   990→            complexity = "very_high"
   991→        elif len(findings) >= 3:
   992→            complexity = "high"
   993→        elif len(findings) >= 2:
   994→            complexity = "medium"
   995→        else:
   996→            complexity = "low"
   997→
   998→        narrative = (
   999→            f"Attack path '{name}': An attacker could progress through {len(findings)} "
  1000→            f"stages — starting with '{steps[0]['title']}' against {steps[0].get('asset', 'target')}, "
  1001→            f"progressing through intermediate vulnerabilities, "
  1002→            f"and achieving '{steps[-1]['title']}' as the final objective. "
  1003→            f"Overall risk: {risk}. Complexity: {complexity}."
  1004→        )
  1005→
  1006→        return AttackPath(
  1007→            steps=steps,
  1008→            entry_point=steps[0].get("asset", ""),
  1009→            objective=name,
  1010→            complexity=complexity,
  1011→            likelihood=round(likelihood, 2),
  1012→            impact=round(impact, 2),
  1013→            confidence=round(min(1.0, likelihood * impact), 2),
  1014→            narrative=narrative,
  1015→        )
  1016→
  1017→    def _build_two_step_path(self, pair: Tuple[Dict[str, Any], Dict[str, Any]],
  1018→                             name: str) -> Optional[AttackPath]:
  1019→        """Build a two-step AttackPath."""
  1020→        return self._build_path(list(pair), name, "high")

  1028→class RemediationPrioritizer:
  1029→    """Prioritizes remediation based on multiple risk factors."""
  1030→
  1031→    def prioritize(self, extended_findings: List[Dict[str, Any]]) -> List[Dict[str, Any]]:
  1032→        """Sort findings by remediation priority.
  1033→
  1034→        Considers: severity, exploitability, business impact, blast radius, fix complexity.
  1035→        Returns ordered list with priority_rank and reasoning.
  1036→        """
  1037→        if not extended_findings:
  1038→            return []
  1039→
  1040→        scored: List[Tuple[float, int, Dict[str, Any]]] = []
  1041→        for i, ef in enumerate(extended_findings):
  1042→            priority = self._compute_priority(ef)
  1043→            scored.append((priority, i, ef))
  1044→
  1045→        scored.sort(key=lambda x: x[0], reverse=True)
  1046→
  1047→        result: List[Dict[str, Any]] = []
  1048→        for rank, (score, orig_idx, ef) in enumerate(scored, 1):
  1049→            entry = dict(ef)
  1050→            entry["priority_rank"] = rank
  1051→            entry["priority_score"] = round(score, 2)
  1052→            entry["priority_reasoning"] = self._reasoning(ef, score, rank)
  1053→            result.append(entry)
  1054→
  1055→        return result
  1056→
  1057→    def _compute_priority(self, ef: Dict[str, Any]) -> float:
  1058→        """Compute a priority score (higher = fix first)."""
  1059→        severity = ef.get("severity", "info").lower()
  1060→        exploitability = ef.get("exploitability", 0.5)
  1061→        impact = ef.get("business_impact_score", 0.5)
  1062→        confidence = ef.get("confidence_score", 0.5)
  1063→
  1064→        sev_weight = _sev_val(severity) / 10.0
  1065→        return (sev_weight * 0.35 + exploitability * 0.25 +
  1066→                impact * 0.25 + confidence * 0.15)
  1067→
  1068→    def _reasoning(self, ef: Dict[str, Any], score: float, rank: int) -> str:
  1069→        """Generate a human-readable reasoning for the priority."""
  1070→        severity = ef.get("severity", "info").upper()
  1071→        exploit = ef.get("exploitability", 0.5)
  1072→        impact = ef.get("business_impact_score", 0.5)
  1073→        title = ef.get("title", "Unknown")
  1074→
  1075→        parts = [f"#{rank} priority"]
  1076→        if severity == "CRITICAL":
  1077→            parts.append("critical severity")
  1078→        elif severity == "HIGH":
  1079→            parts.append("high severity")
  1080→        if exploit >= 0.7:
  1081→            parts.append("easily exploitable")
  1082→        if impact >= 0.7:
  1083→            parts.append("high business impact")
  1084→        parts.append(f"(score: {score:.2f})")
  1085→
  1086→        return f"{'; '.join(parts)} — {title}"

  1094→def resolve_mappings(attack_category: str) -> Tuple[Dict[str, str], List[str], List[str]]:
  1095→    """Resolve MITRE ATT&CK, CWE, and CAPEC mappings for a category.
  1096→
  1097→    Returns (mitre_primary, cwe_list, capec_list).
  1098→    """
  1099→    mitre_entries = MITRE_ATTACK_MAP.get(attack_category, [])
  1100→    mitre_primary: Dict[str, str] = {}
  1101→    if mitre_entries:
  1102→        m = mitre_entries[0]
  1103→        mitre_primary = {"tactic": m["tactic"], "technique": m["technique"], "name": m["name"]}
  1104→
  1105→    cwe_list = CWE_MAP.get(attack_category, [])
  1106→    capec_list = CAPEC_MAP.get(attack_category, [])
  1107→
  1108→    return mitre_primary, cwe_list, capec_list

  1111→def generate_cvss(severity: str, exploitability: float = 0.5, impact: float = 0.5) -> Dict[str, Any]:
  1112→    """Generate a CVSS v3 interpretation from severity and metrics.
  1113→
  1114→    Returns dict with base_score, severity, vector_string.
  1115→    """
  1116→    base = CVSS_BASE_SCORES.get(severity.lower(), 5.0)
  1117→
  1118→    # Adjust based on exploitability and impact
  1119→    adj = base + (exploitability - 0.5) * 1.5 + (impact - 0.5) * 2.0
  1120→    base_score = round(max(0.0, min(10.0, adj)), 1)
  1121→
  1122→    if base_score >= 9.0:
  1123→        sev = "CRITICAL"
  1124→    elif base_score >= 7.0:
  1125→        sev = "HIGH"
  1126→    elif base_score >= 4.0:
  1127→        sev = "MEDIUM"
  1128→    elif base_score > 0.0:
  1129→        sev = "LOW"
  1130→    else:
  1131→        sev = "NONE"
  1132→
  1133→    # AV:N/AC:L/PR:N/UI:N/S:U/C:H/I:H/A:H (typical remote vuln)
  1134→    if exploitability >= 0.7:
  1135→        av, ac, pr, ui = "N", "L", "N", "N"
  1136→    elif exploitability >= 0.4:
  1137→        av, ac, pr, ui = "N", "L", "L", "N"
  1138→    else:
  1139→        av, ac, pr, ui = "N", "H", "L", "R"
  1140→
  1141→    c = "H" if impact >= 0.7 else ("L" if impact >= 0.3 else "N")
  1142→    i = "H" if impact >= 0.7 else ("L" if impact >= 0.4 else "N")
  1143→    a = "H" if impact >= 0.8 else ("L" if impact >= 0.3 else "N")
  1144→
  1145→    vector = f"CVSS:3.1/AV:{av}/AC:{ac}/PR:{pr}/UI:{ui}/S:U/C:{c}/I:{i}/A:{a}"
  1146→
  1147→    return {
  1148→        "base_score": base_score,
  1149→        "severity": sev,
  1150→        "vector_string": vector,
  1151→    }

  1254→    def analyze_scan(self, findings: List[Dict[str, Any]], target: str = "") -> AnalysisReport:
  1255→        """Run full AI analysis on scan findings.
  1256→

  1266→        report = AnalysisReport(target=target, total_findings=len(findings))
  1267→
  1268→        if not findings:
  1269→            report.summary = "No findings to analyze."

  1308→        ]

  1311→        attack_paths = self.path_detector.detect(deduped)

  1346→    def _enrich_finding(self, finding: Dict[str, Any],

  1353→

  1362→            finding.get("severity", "info"),

  1364→            impact.get("impact_score", 0.5),

  1373→        # Confidence

  1375→        description = finding.get("description", "")
  1376→        evidence_strength = min(len(evidence) / 200.0, 1.0)
  1377→        desc_strength = min(len(description) / 150.0, 1.0)
  1378→        confidence = min(1.0, classification.confidence * 0.5 + evidence_strength * 0.3 + desc_strength * 0.2)
  1379→
  1380→        # Root cause inference
  1381→        root_cause = self._infer_root_cause(cat, finding)
  1382→
  1383→        # Executive summary
  1384→        severity = finding.get("severity", "info").upper()
  1385→        exec_summary = (
  1386→            f"A {severity}-severity {cat.replace('_', ' ')} vulnerability was identified "

  1389→            f"with a business impact score of {impact.get('impact_score', 0.5):.0%}."
  1390→        )
  1391→        if confidence < 0.3:
  1392→            exec_summary += " NOTE: Limited evidence available; assessment is probabilistic."
  1393→
  1394→        # Uncertainty note
  1395→        uncertainty = ""
  1396→        if confidence < 0.3:
  1397→            uncertainty = "Limited evidence; results are probabilistic and should be validated manually."
  1398→        elif confidence < 0.5:
  1399→            uncertainty = "Moderate evidence; some findings may require manual verification."
  1400→
  1401→        # Supporting evidence
  1402→        supporting = []
  1403→        if evidence:
  1404→            supporting.append(f"Raw evidence: {evidence[:200]}")
  1405→        if finding.get("module"):

  1407→        if classification.indicators:

  1409→
  1410→        return {

  1414→            "executive_summary": exec_summary,

  1416→            "attack_narrative": self._attack_narrative(finding, cat),
  1417→            "risk_justification": f"Severity: {severity}, Exploitability: {exploit.get('exploitability', 0):.0%}, "

  1421→            "mitre_attack_mapping": mitre,
  1422→            "cwe_mapping": cwe,
  1423→            "capec_mapping": capec,
  1424→            "cvss_interpretation": cvss,
  1425→            "remediation_plan": remediation_steps,
  1426→            "validation_steps": validation_steps,
  1427→            "confidence_score": round(confidence, 2),
  1428→            "supporting_evidence": supporting,
  1429→            "exploitability": exploit.get("exploitability", 0.5),
  1430→            "business_impact_score": impact.get("impact_score", 0.5),

  1540→    def _reduce_false_positives(self, findings: List[Dict]) -> List[Dict]:
  1541→        """Apply heuristic rules to identify likely false positives.
  1542→        
  1543→        Rules:
  1544→        1. Findings with very low severity and generic descriptions
  1545→        2. Duplicate findings with different titles but same asset+category
  1546→        3. Findings with empty or very short evidence
  1547→        4. Findings whose descriptions match common false-positive patterns
  1548→        """
  1549→        fp_indicators = [
  1550→            "example", "test", "sample", "demo", "placeholder",
  1551→            "localhost", "127.0.0.1", "0.0.0.0", "::1",
  1552→            "staging", "development", "dev-", "test-",
  1553→        ]
  1554→        
  1555→        result = []
  1556→        seen_signatures = set()
  1557→        
  1558→        for f in findings:
  1559→            fp_score = 0  # 0 = definitely real, higher = more likely FP
  1560→            
  1561→            # Rule 1: Check for FP indicator terms in evidence
  1562→            evidence = f.get("evidence", "").lower()
  1563→            asset = f.get("asset", "").lower()
  1564→            title = f.get("title", "").lower()
  1565→            
  1566→            for indicator in fp_indicators:
  1567→                if indicator in evidence or indicator in asset:
  1568→                    fp_score += 2
  1569→                    break
  1570→            
  1571→            # Rule 2: Deduplicate by asset+category signature
  1572→            sig = f"{f.get('asset', '')}:{f.get('category', '')}:{str(f.get('severity', ''))}"
  1573→            if sig in seen_signatures and f.get("severity", "").lower() in ("info", "low"):
  1574→                fp_score += 3
  1575→            seen_signatures.add(sig)
  1576→            
  1577→            # Rule 3: Very short or empty evidence
  1578→            if len(evidence.strip()) < 10:
  1579→                fp_score += 2
  1580→            
  1581→            # Rule 4: Generic description patterns
  1582→            generic_patterns = [
  1583→                r'^a .* (was|is) (found|detected|discovered)$',
  1584→                r'^possible .*$',
  1585→                r'^potential .*$',
  1586→            ]
  1587→            import re
  1588→            desc = f.get("description", "").lower().strip()
  1589→            for pattern in generic_patterns:
  1590→                if re.match(pattern, desc):
  1591→                    fp_score += 1
  1592→                    break
  1593→            
  1594→            # Mark the finding with FP probability
  1595→            f["false_positive_probability"] = min(1.0, fp_score / 10.0)
  1596→            
  1597→            # Only exclude if very high FP score AND low severity
  1598→            if fp_score >= 6 and f.get("severity", "").lower() in ("info",):
  1599→                continue  # Skip likely false positive
  1600→            
  1601→            result.append(f)
  1602→        
  1603→        return result

  1605→    def _analyze_asset_criticality(self, findings: List[Dict]) -> Dict[str, float]:
  1606→        """Analyze asset criticality based on finding density and severity.
  1607→        
  1608→        Returns dict mapping asset names to criticality scores (0-100).
  1609→        Assets with more critical/high findings are more critical.
  1610→        """
  1611→        asset_data: Dict[str, Dict] = {}
  1612→        
  1613→        sev_weights = {"critical": 10, "high": 8, "medium": 5, "low": 2, "info": 0.5}
  1614→        
  1615→        for f in findings:
  1616→            asset = f.get("asset", "unknown")
  1617→            if asset not in asset_data:
  1618→                asset_data[asset] = {"findings": 0, "weighted_score": 0.0, "max_sev": 0}
  1619→            
  1620→            asset_data[asset]["findings"] += 1
  1621→            sev = f.get("severity", "info").lower()
  1622→            asset_data[asset]["weighted_score"] += sev_weights.get(sev, 0.5)
  1623→            asset_data[asset]["max_sev"] = max(
  1624→                asset_data[asset]["max_sev"],
  1625→                sev_weights.get(sev, 0.5)
  1626→            )
  1627→        
  1628→        # Normalize to 0-100
  1629→        max_weighted = max((d["weighted_score"] for d in asset_data.values()), default=1)
  1630→        
  1631→        criticality = {}
  1632→        for asset, data in asset_data.items():
  1633→            raw = (data["weighted_score"] / max(max_weighted, 1)) * 100
  1634→            # Boost if has critical findings
  1635→            if data["max_sev"] >= 10:
  1636→                raw = min(100, raw + 10)
  1637→            criticality[asset] = round(raw, 1)
  1638→        
  1639→        return criticality
