     8→// ═══════════════════════════════════════════════════════════════════════
     9→// REAL RECONNAISSANCE ENGINE — Fully async, non-blocking
    10→// Every finding comes from actual dig/curl/openssl output
    11→// ═══════════════════════════════════════════════════════════════════════
    12→
    13→type Finding = {
    14→  title: string; severity: string; category: string;
    15→  description: string; evidence: string; asset: string;
    16→};
    17→
    18→// ── Async shell runner (never blocks event loop) ──────────────────
    19→async function run(cmd: string, timeout = 8000): Promise<string> {
    20→  try {
    21→    const { stdout } = await execAsync(cmd, { timeout, encoding: 'utf-8' });
    22→    return stdout.trim();
    23→  } catch { return ''; }
    24→}
    25→
    26→// ── Async DNS via dig ─────────────────────────────────────────────
    27→async function digShort(domain: string, type: string): Promise<string[]> {
    28→  const out = await run(`dig +short +time=2 +tries=1 ${domain} ${type}`, 5000);
    29→  return out ? out.split('\n').map(l => l.trim()).filter(Boolean) : [];
    30→}
    31→
    32→async function digAnswer(domain: string, type: string): Promise<string> {
    33→  return run(`dig +noall +answer +time=2 +tries=1 ${domain} ${type}`, 5000);
    34→}
    35→
    36→async function resolveIP(domain: string): Promise<string | null> {
    37→  const out = await run(`dig +short +time=2 +tries=1 ${domain} A`, 3000);
    38→  if (!out) return null;
    39→  for (const line of out.split('\n')) {
    40→    if (/^\d+\.\d+\.\d+\.\d+$/.test(line.trim())) return line.trim();
    41→  }
    42→  return null;
    43→}

    46→// 1. DNS ENUMERATION (all in parallel)
    47→// ═══════════════════════════════════════════════════════════════════════
    48→async function enumerateDNS(domain: string): Promise<{ findings: Finding[]; mainIp: string | null }> {
    49→  const findings: Finding[] = [];
    50→
    51→  // Fire all DNS queries in parallel
    52→  const [aRecords, aaaaRecords, mxRaw, nsRaw, txtRecords, spfDomain] = await Promise.all([
    53→    digShort(domain, 'A'),
    54→    digShort(domain, 'AAAA'),
    55→    digAnswer(domain, 'MX'),
    56→    digAnswer(domain, 'NS'),
    57→    digShort(domain, 'TXT'),
    58→    digShort(`_dmarc.${domain}`, 'TXT'),
    59→  ]);
    60→
    61→  const mainIp = aRecords[0] || null;
    62→
    63→  // A records
    64→  if (aRecords.length > 0) {
    65→    findings.push({
    66→      title: `DNS A Record — ${aRecords.length} IPv4 address(es)`,
    67→      severity: 'info', category: 'dns',
    68→      description: `Domain ${domain} resolves to ${aRecords.length} IPv4 address(es): ${aRecords.join(', ')}. Multiple A records indicate round-robin DNS or CDN usage.`,
    69→      evidence: `A: ${aRecords.join(', ')}`,
    70→      asset: domain,
    71→    });
    72→  }
    73→
    74→  // AAAA
    75→  if (aaaaRecords.length > 0) {
    76→    findings.push({
    77→      title: 'DNS AAAA Record — IPv6 enabled',
    78→      severity: 'info', category: 'dns',
    79→      description: `IPv6 address(es) found: ${aaaaRecords.join(', ')}. IPv6 support is present.`,
    80→      evidence: `AAAA: ${aaaaRecords.join(', ')}`,
    81→      asset: domain,
    82→    });
    83→  }
    84→
    85→  // MX
    86→  const mxLines = mxRaw ? mxRaw.split('\n').filter(l => l.includes('MX')) : [];
    87→  if (mxLines.length > 0) {
    88→    findings.push({
    89→      title: `MX Records — ${mxLines.length} mail server(s)`,
    90→      severity: 'info', category: 'dns',
    91→      description: `${mxLines.length} mail server(s) configured:\n${mxLines.map(m => '  ' + m.trim()).join('\n')}`,
    92→      evidence: mxLines.join('; '),
    93→      asset: domain,
    94→    });
    95→  } else {
    96→    findings.push({
    97→      title: 'No MX Records Found',
    98→      severity: 'low', category: 'dns',
    99→      description: `No MX records for ${domain}. Domain does not receive email directly or uses third-party email.`,
   100→      evidence: 'MX query returned empty',
   101→      asset: domain,
   102→    });
   103→  }
   104→
   105→  // NS
   106→  const nsLines = nsRaw ? nsRaw.split('\n').filter(l => l.includes('NS')) : [];
   107→  if (nsLines.length > 0) {
   108→    const nsNames = nsLines.map(n => n.trim().split(/\s+/).pop());
   109→    findings.push({
   110→      title: `NS Records — ${nsLines.length} nameserver(s)`,
   111→      severity: 'info', category: 'dns',
   112→      description: `Nameservers: ${nsNames.join(', ')}.`,
   113→      evidence: nsLines.join('; '),
   114→      asset: domain,
   115→    });
   116→  }
   117→
   118→  // TXT
   119→  if (txtRecords.length > 0) {
   120→    findings.push({
   121→      title: `TXT Records — ${txtRecords.length} record(s)`,
   122→      severity: 'info', category: 'dns',
   123→      description: `Found ${txtRecords.length} TXT record(s) including verification records, SPF, or other configurations.`,
   124→      evidence: txtRecords.join(' | '),
   125→      asset: domain,
   126→    });
   127→  }
   128→
   129→  // SPF
   130→  const spfRecord = txtRecords.find(t => t.startsWith('"v=spf'));
   131→  if (!spfRecord) {
   132→    findings.push({
   133→      title: 'SPF Record Missing',
   134→      severity: 'high', category: 'dns',
   135→      description: `No SPF record found for ${domain}. Without SPF, attackers can send spoofed emails appearing to come from this domain, enabling phishing attacks and brand damage.`,
   136→      evidence: `TXT query returned ${txtRecords.length} record(s), none with "v=spf1"`,
   137→      asset: domain,
   138→    });
   139→  } else {
   140→    if (spfRecord.includes('+all') || spfRecord.includes('?all')) {
   141→      findings.push({
   142→        title: 'SPF Record Uses +all or ?all (Open)',
   143→        severity: 'critical', category: 'dns',
   144→        description: `SPF record allows ANY server to send email for ${domain}, defeating the purpose of SPF entirely.`,
   145→        evidence: `SPF: ${spfRecord}`,
   146→        asset: domain,
   147→      });
   148→    } else {
   149→      findings.push({
   150→        title: 'SPF Record Configured Properly',
   151→        severity: 'info', category: 'dns',
   152→        description: `SPF record found with proper fail mechanism: ${spfRecord}.`,
   153→        evidence: `SPF: ${spfRecord}`,
   154→        asset: domain,
   155→      });
   156→    }
   157→  }
   158→
   159→  // DMARC
   160→  const dmarc = spfDomain.find(t => t.includes('v=DMARC'));
   161→  if (!dmarc) {
   162→    findings.push({
   163→      title: 'DMARC Record Not Found',
   164→      severity: 'high', category: 'dns',
   165→      description: `No DMARC record for ${domain}. DMARC builds on SPF/DKIM to provide email authentication policy. Without it, the domain cannot specify how receivers handle unauthenticated emails.`,
   166→      evidence: `_dmarc.${domain} TXT query empty`,
   167→      asset: `_dmarc.${domain}`,
   168→    });
   169→  } else {
   170→    const pMatch = dmarc.match(/p=([a-z]+)/);
   171→    const policy = pMatch ? pMatch[1] : 'unknown';
   172→    if (policy === 'none') {
   173→      findings.push({
   174→        title: 'DMARC Policy Set to "none" (Monitor Only)',
   175→        severity: 'medium', category: 'dns',
   176→        description: `DMARC p=none means unauthenticated emails are only monitored, not rejected. Upgrade to p=quarantine or p=reject.`,
   177→        evidence: `DMARC: ${dmarc}`,
   178→        asset: `_dmarc.${domain}`,
   179→      });
   180→    } else {
   181→      findings.push({
   182→        title: `DMARC Policy Active (p=${policy})`,
   183→        severity: 'info', category: 'dns',
   184→        description: `DMARC is enforcing with p=${policy}.`,
   185→        evidence: `DMARC: ${dmarc}`,
   186→        asset: `_dmarc.${domain}`,
   187→      });
   188→    }
   189→  }
   190→
   191→  // DKIM
   192→  const dkimSelectors = ['selector1', 'selector2', 'google', 'k1', 'default', 's1'];
   193→  const dkimChecks = await Promise.all(dkimSelectors.map(async sel => {
   194→    const recs = await digShort(`${sel}._domainkey.${domain}`, 'TXT');
   195→    return recs.find(t => t.includes('v=DKIM')) ? sel : null;
   196→  }));
   197→  const dkimSel = dkimChecks.find(Boolean);
   198→  if (dkimSel) {
   199→    findings.push({
   200→      title: `DKIM Record Found (selector: ${dkimSel})`,
   201→      severity: 'info', category: 'dns',
   202→      description: `DKIM signing configured with selector "${dkimSel}". Outgoing emails can be cryptographically verified.`,
   203→      evidence: `DKIM selector: ${dkimSel}`,
   204→      asset: `${dkimSel}._domainkey.${domain}`,
   205→    });
   206→  } else {
   207→    findings.push({
   208→      title: 'DKIM Record Not Found',
   209→      severity: 'medium', category: 'dns',
   210→      description: `No DKIM record found for common selectors (${dkimSelectors.join(', ')}). Outgoing emails cannot be cryptographically verified.`,
   211→      evidence: `Checked selectors: ${dkimSelectors.join(', ')}`,
   212→      asset: domain,
   213→    });
   214→  }
   215→
   216→  // DNSSEC
   217→  const dnssecOut = await run(`dig +dnssec +time=2 +tries=1 ${domain} A`, 5000);
   218→  if (!dnssecOut.includes('RRSIG')) {
   219→    findings.push({
   220→      title: 'DNSSEC Not Enabled',
   221→      severity: 'medium', category: 'dns',
   222→      description: `DNSSEC not configured for ${domain}. DNS responses can be spoofed via cache poisoning attacks.`,
   223→      evidence: 'RRSIG not found in DNS response',
   224→      asset: domain,
   225→    });
   226→  } else {
   227→    findings.push({
   228→      title: 'DNSSEC Enabled',
   229→      severity: 'info', category: 'dns',
   230→      description: `DNSSEC is active, providing cryptographic DNS response authentication.`,
   231→      evidence: 'RRSIG records present',
   232→      asset: domain,
   233→    });
   234→  }
   235→
   236→  return { findings, mainIp };
   237→}

   242→const QUICK_SUBS = [
   243→  'www','api','mail','admin','portal','dashboard','app','dev','staging',
   244→  'cdn','blog','shop','secure','auth','login','sso','vpn','gateway','proxy',
   245→  'ns1','ns2','dns','db','elastic','search','analytics','webhook','graphql',
   246→  'git','gitlab','github','docs','status','monitor','grafana','kibana','logs',
   247→  'jenkins','ci','internal','sandbox','demo','preview','m','mobile','crm',
   248→  'smtp','webmail','origin','edge','panel','cpanel','server','node','us','eu',
   249→];
   250→
   251→const FULL_SUBS = [
   252→  ...QUICK_SUBS,
   253→  'ftp','pop','imap','test','beta','static','assets','images','media','video',
   254→  'news','store','remote','mx1','mx2','redis','track','hooks','api-v1','api-v2',
   255→  'rest','repo','wiki','help','support','health','metrics','prometheus',
   256→  'hr','billing','token','keys','certs','v2','v3','staging1','staging2',
   257→  'phpmyadmin','amp','pwa','erp','payroll','oAuth','oauth2','pki',
   258→  'go','play','run','link','links','short','redirect','webmin','whm','plesk',
   259→  'web1','web2','app1','app2','uk','asia','cdn1','cdn2','static1',
   260→];
   261→
   262→const SENSITIVE_SUBS = ['admin','portal','dashboard','db','database','redis',
   263→  'jenkins','gitlab','git','internal','staging','dev','test','vpn','remote',
   264→  'proxy','elastic','kibana','grafana','prometheus','cpanel','plesk','webmin',
   265→  'phpmyadmin','server','smtp','webmail','crm','sandbox','ci','hooks','api-v1',
   266→  'staging1','staging2','test','dev'];
   267→
   268→async function enumerateSubdomains(domain: string, quick: boolean): Promise<Finding[]> {
   269→  const subs = quick ? QUICK_SUBS : FULL_SUBS;
   270→  const batchSize = 30;
   271→  const discovered: Array<{ fqdn: string; ip: string }> = [];
   272→
   273→  for (let i = 0; i < subs.length; i += batchSize) {
   274→    const batch = subs.slice(i, i + batchSize);
   275→    const results = await Promise.all(
   276→      batch.map(async (sub) => {
   277→        const ip = await resolveIP(`${sub}.${domain}`);
   278→        return ip ? { fqdn: `${sub}.${domain}`, ip } : null;
   279→      })
   280→    );
   281→    for (const r of results) { if (r) discovered.push(r); }
   282→  }
   283→
   284→  return discovered.map(({ fqdn, ip }) => {
   285→    const subName = fqdn.replace(`.${domain}`, '');
   286→    const isSensitive = SENSITIVE_SUBS.some(s => subName.includes(s));
   287→    return {
   288→      title: `Discovered Subdomain: ${fqdn}`,
   289→      severity: isSensitive ? 'high' : 'info',
   290→      category: 'subdomain',
   291→      description: `${fqdn} resolves to ${ip}.${isSensitive ? ` This is a sensitive internal service (${subName}) publicly accessible — significant security concern.` : ' Publicly accessible subdomain adding to attack surface.'}`,
   292→      evidence: `A record: ${ip}`,
   293→      asset: fqdn,
   294→    };
   295→  });
   296→}

   299→// 3. HTTP HEADER ANALYSIS (async)
   300→// ═══════════════════════════════════════════════════════════════════════
   301→async function analyzeHTTPHeaders(domain: string): Promise<{ findings: Finding[]; technologies: string[] }> {
   302→  const findings: Finding[] = [];
   303→  const technologies: string[] = [];
   304→
   305→  const [httpsH, httpH] = await Promise.all([
   306→    run(`curl -sI --max-time 8 -L https://${domain} 2>/dev/null`, 12000),
   307→    run(`curl -sI --max-time 5 -L http://${domain} 2>/dev/null`, 8000),
   308→  ]);
   309→  const headers = httpsH || httpH;
   310→
   311→  if (!headers || !headers.includes('HTTP/')) {
   312→    findings.push({
   313→      title: 'HTTP Service Not Reachable',
   314→      severity: 'medium', category: 'header',
   315→      description: `Could not connect to ${domain} via HTTP/HTTPS.`,
   316→      evidence: 'curl returned non-HTTP or empty response',
   317→      asset: domain,
   318→    });
   319→    return { findings, technologies };
   320→  }
   321→
   322→  const hmap: Record<string, string> = {};
   323→  for (const line of headers.split('\n')) {
   324→    const m = line.match(/^([^:]+):\s*(.+)/);
   325→    if (m) hmap[m[1].trim().toLowerCase()] = m[2].trim();
   326→  }
   327→
   328→  // HSTS
   329→  if (!hmap['strict-transport-security']) {
   330→    findings.push({
   331→      title: 'Missing Strict-Transport-Security (HSTS)',
   332→      severity: 'high', category: 'header',
   333→      description: `No HSTS header on ${domain}. Browsers may attempt HTTP before HTTPS, enabling SSL stripping MITM attacks. Set HSTS with max-age >= 31536000 and includeSubDomains.`,
   334→      evidence: 'strict-transport-security header not found',
   335→      asset: domain, technologies,
   336→    });
   337→  } else {
   338→    const hsts = hmap['strict-transport-security'];
   339→    const ma = hsts.match(/max-age=(\d+)/);
   340→    const age = ma ? parseInt(ma[1]) : 0;
   341→    if (age < 31536000) {
   342→      findings.push({ title: `HSTS max-age Too Short (${Math.round(age/86400)} days)`, severity: 'medium', category: 'header',
   343→        description: `HSTS max-age is ${age}s (${Math.round(age/86400)}d). Best practice: >= 1 year (31536000s).`, evidence: `HSTS: ${hsts}`, asset: domain, technologies });
   344→    } else if (!hsts.includes('includeSubDomains')) {
   345→      findings.push({ title: 'HSTS Missing includeSubDomains', severity: 'medium', category: 'header',
   346→        description: `HSTS active but without includeSubDomains — subdomains remain vulnerable.`, evidence: `HSTS: ${hsts}`, asset: domain, technologies });
   347→    } else {
   348→      findings.push({ title: 'HSTS Properly Configured', severity: 'info', category: 'header',
   349→        description: `HSTS active with proper configuration.`, evidence: `HSTS: ${hsts}`, asset: domain, technologies });
   350→    }
   351→  }
   352→
   353→  // CSP
   354→  if (!hmap['content-security-policy'] && !hmap['content-security-policy-report-only']) {
   355→    findings.push({ title: 'Missing Content-Security-Policy (CSP)', severity: 'high', category: 'header',
   356→      description: `No CSP header. CSP is the primary defense against XSS and data injection. Without it, browsers will execute any inline scripts and load resources from any origin.`, evidence: 'content-security-policy header not found', asset: domain, technologies });
   357→  } else {
   358→    const csp = hmap['content-security-policy'] || hmap['content-security-policy-report-only'];
   359→    if (csp.includes("'unsafe-inline'") && csp.includes("'unsafe-eval'")) {
   360→      findings.push({ title: 'CSP Uses unsafe-inline and unsafe-eval', severity: 'medium', category: 'header',
   361→        description: `CSP present but weakened by 'unsafe-inline' and 'unsafe-eval', significantly reducing XSS protection.`, evidence: `CSP: ${csp.substring(0,200)}`, asset: domain, technologies });
   362→    } else {
   363→      findings.push({ title: 'Content-Security-Policy Configured', severity: 'info', category: 'header',
   364→        description: `CSP header present, providing XSS protection.`, evidence: `CSP: ${csp.substring(0,150)}`, asset: domain, technologies });
   365→    }
   366→  }
   367→
   368→  // X-Frame-Options
   369→  if (!hmap['x-frame-options']) {
   370→    findings.push({ title: 'Missing X-Frame-Options Header', severity: 'medium', category: 'header',
   371→      description: `No X-Frame-Options — site is vulnerable to clickjacking. Set to "DENY" or "SAMEORIGIN".`, evidence: 'x-frame-options not found', asset: domain, technologies });
   372→  }
   373→
   374→  // X-Content-Type-Options
   375→  if (hmap['x-content-type-options'] !== 'nosniff') {
   376→    findings.push({ title: 'Missing X-Content-Type-Options: nosniff', severity: 'low', category: 'header',
   377→      description: `Without nosniff, browsers may MIME-sniff content types.`, evidence: `x-content-type-options: ${hmap['x-content-type-options'] || 'not set'}`, asset: domain, technologies });
   378→  }
   379→
   380→  // Referrer-Policy
   381→  if (!hmap['referrer-policy']) {
   382→    findings.push({ title: 'Missing Referrer-Policy Header', severity: 'low', category: 'header',
   383→      description: `No Referrer-Policy — full URLs may leak to third parties via Referer header.`, evidence: 'referrer-policy not found', asset: domain, technologies });
   384→  }
   385→
   386→  // Permissions-Policy
   387→  if (!hmap['permissions-policy'] && !hmap['feature-policy']) {
   388→    findings.push({ title: 'Missing Permissions-Policy Header', severity: 'low', category: 'header',
   389→      description: `No Permissions-Policy — browser features (camera, mic, etc.) not restricted.`, evidence: 'permissions-policy not found', asset: domain, technologies });
   390→  }
   391→
   392→  // CORS
   393→  if (hmap['access-control-allow-origin']) {
   394→    if (hmap['access-control-allow-origin'] === '*') {
   395→      findings.push({ title: 'CORS Allows Any Origin (*)', severity: 'high', category: 'header',
   396→        description: `Access-Control-Allow-Origin: * means any website can read responses. If sensitive data is served, it can be exfiltrated by any malicious site.`, evidence: 'access-control-allow-origin: *', asset: domain, technologies });
   397→    }
   398→  }
   399→
   400→  // Server disclosure
   401→  if (hmap['server']) {
   402→    technologies.push(hmap['server']);
   403→    if (/\d+\.\d+/.test(hmap['server'])) {
   404→      findings.push({ title: 'Server Version Disclosure', severity: 'medium', category: 'header',
   405→        description: `Server header reveals version: "${hmap['server']}". Remove version numbers to reduce info leakage.`, evidence: `server: ${hmap['server']}`, asset: domain, technologies });
   406→    }
   407→  }
   408→
   409→  // X-Powered-By
   410→  if (hmap['x-powered-by']) {
   411→    technologies.push(hmap['x-powered-by']);
   412→    findings.push({ title: `Technology Disclosure: X-Powered-By: ${hmap['x-powered-by']}`, severity: 'low', category: 'header',
   413→      description: `X-Powered-By reveals backend: "${hmap['x-powered-by']}". Remove in production.`, evidence: `x-powered-by: ${hmap['x-powered-by']}`, asset: domain, technologies });
   414→  }
   415→
   416→  // Detect tech from headers
   417→  const allH = httpsH + httpH;
   418→  if (hmap['cf-ray'] || hmap['cf-cache-status']) technologies.push('Cloudflare');
   419→  if (allH.includes('X-Amz-Cf-Id')) technologies.push('AWS CloudFront');
   420→  if (hmap['x-vercel-id']) technologies.push('Vercel');
   421→  if (hmap['x-netlify-request-id']) technologies.push('Netlify');
   422→  if (hmap['fly-request-id']) technologies.push('Fly.io');
   423→  if (hmap['alt-svc']?.includes('h3')) technologies.push('HTTP/3 (QUIC)');
   424→
   425→  return { findings, technologies };
   426→}

   429→// 4. SSL/TLS CERTIFICATE ANALYSIS (async)
   430→// ═══════════════════════════════════════════════════════════════════════
   431→async function analyzeSSL(domain: string): Promise<{ findings: Finding[]; technologies: string[] }> {
   432→  const findings: Finding[] = [];
   433→  const technologies: string[] = [];
   434→
   435→  const [certInfo, sslConnect] = await Promise.all([
   436→    run(`echo | openssl s_client -connect ${domain}:443 -servername ${domain} 2>/dev/null | openssl x509 -noout -subject -issuer -dates -ext subjectAltName 2>/dev/null`, 10000),
   437→    run(`echo | openssl s_client -connect ${domain}:443 -servername ${domain} 2>&1`, 10000),
   438→  ]);
   439→
   440→  if (!certInfo && !sslConnect.includes('SSL handshake')) {
   441→    findings.push({ title: 'SSL/TLS Connection Failed', severity: 'high', category: 'ssl',
   442→      description: `Could not establish SSL/TLS to ${domain}:443. Server may not support HTTPS.`, evidence: 'openssl s_client failed', asset: `${domain}:443`, technologies });
   443→    return { findings, technologies };
   444→  }
   445→
   446→  const full = certInfo || sslConnect;
   447→
   448→  // Subject
   449→  const subjM = full.match(/subject=([^\n]+)/);
   450→  if (subjM) findings.push({ title: `SSL Certificate Subject: ${subjM[1].trim()}`, severity: 'info', category: 'ssl',
   451→    description: `Certificate issued to: ${subjM[1].trim()}.`, evidence: `subject: ${subjM[1].trim()}`, asset: domain, technologies });
   452→
   453→  // Issuer + tech detection
   454→  const issM = full.match(/issuer=([^\n]+)/);
   455→  const issuer = issM ? issM[1].trim() : 'unknown';
   456→  if (issuer.includes("Let's Encrypt")) technologies.push("Let's Encrypt");
   457→  else if (issuer.includes('DigiCert')) technologies.push('DigiCert');
   458→  else if (issuer.includes('Sectigo') || issuer.includes('Comodo')) technologies.push('Sectigo');
   459→  else if (issuer.includes('Cloudflare')) technologies.push('Cloudflare SSL');
   460→  else if (issuer.includes('Amazon') || issuer.includes('AWS')) technologies.push('AWS ACM');
   461→  else if (issuer.includes('Google')) technologies.push('Google Trust Services');
   462→
   463→  findings.push({ title: `SSL Certificate Issuer: ${issuer}`, severity: 'info', category: 'ssl',
   464→    description: `Issued by: ${issuer}.`, evidence: `issuer: ${issuer}`, asset: domain, technologies });
   465→
   466→  // Dates + expiry
   467→  const nbM = full.match(/notBefore=(.+)/);
   468→  const naM = full.match(/notAfter=(.+)/);
   469→  if (nbM && naM) {
   470→    const nb = new Date(nbM[1].trim());
   471→    const na = new Date(naM[1].trim());
   472→    const days = Math.ceil((na.getTime() - Date.now()) / 86400000);
   473→    const lifespan = Math.ceil((na.getTime() - nb.getTime()) / 86400000);
   474→
   475→    findings.push({ title: `SSL Certificate Validity: ${nb.toISOString().split('T')[0]} — ${na.toISOString().split('T')[0]}`, severity: 'info', category: 'ssl',
   476→      description: `Certificate lifespan: ${lifespan} days. Valid from ${nb.toISOString().split('T')[0]} to ${na.toISOString().split('T')[0]}.`, evidence: `notBefore: ${nbM[1].trim()}, notAfter: ${naM[1].trim()}`, asset: domain, technologies });
   477→
   478→    if (days < 0) {
   479→      findings.push({ title: `SSL Certificate EXPIRED ${Math.abs(days)} days ago`, severity: 'critical', category: 'ssl',
   480→        description: `Certificate for ${domain} expired ${Math.abs(days)} days ago. Browsers show security warnings, automated integrations fail with TLS errors. Immediate renewal required.`, evidence: `Expired: ${na.toISOString()}`, asset: domain, technologies });
   481→    } else if (days <= 7) {
   482→      findings.push({ title: `SSL Certificate Expiring in ${days} Days — CRITICAL`, severity: 'critical', category: 'ssl',
   483→        description: `Certificate expires in ${days} days. Emergency renewal required.`, evidence: `Expires: ${na.toISOString()} (${days}d)`, asset: domain, technologies });
   484→    } else if (days <= 30) {
   485→      findings.push({ title: `SSL Certificate Expiring in ${days} Days`, severity: 'high', category: 'ssl',
   486→        description: `Certificate expires in ${days} days. Prompt renewal needed.`, evidence: `Expires: ${na.toISOString()} (${days}d)`, asset: domain, technologies });
   487→    } else if (days <= 90) {
   488→      findings.push({ title: `SSL Certificate Expiring in ${days} Days`, severity: 'medium', category: 'ssl',
   489→        description: `Certificate expires in ${days} days. Plan renewal.`, evidence: `Expires: ${na.toISOString()} (${days}d)`, asset: domain, technologies });
   490→    }
   491→
   492→    if (lifespan > 398) {
   493→      findings.push({ title: `Certificate Lifespan Exceeds 398 Days (${lifespan}d)`, severity: 'medium', category: 'ssl',
   494→        description: `Lifespan exceeds Apple/Google 398-day max. New certs must be shorter.`, evidence: `Lifespan: ${lifespan}d (max: 398)`, asset: domain, technologies });
   495→    }
   496→  }
   497→
   498→  // SAN
   499→  const sanM = full.match(/Subject Alternative Name[^\n]*\n([^\n]*)/);
   500→  if (sanM) {
   501→    const sans = sanM[1].match(/DNS:[^\s,]+/g);
   502→    if (sans) findings.push({ title: `Certificate Covers ${sans.length} Domain(s) (SAN)`, severity: 'info', category: 'ssl',
   503→      description: `SANs: ${sans.map(s=>s.replace('DNS:','')).join(', ')}.`, evidence: `SAN: ${sans.join(', ')}`, asset: domain, technologies });
   504→  }
   505→
   506→  // TLS version
   507→  const protoM = sslConnect.match(/Protocol\s*:\s*([^\n]+)/);
   508→  const proto = protoM ? protoM[1].trim() : '';
   509→  if (proto.includes('TLSv1 ') || proto.includes('TLSv1.0')) {
   510→    findings.push({ title: 'Weak TLS Version: TLS 1.0 Detected', severity: 'high', category: 'ssl',
   511→      description: `TLS 1.0 deprecated (RFC 8996, 2020). Vulnerable to BEAST, POODLE, RC4. PCI-DSS prohibits. Disable immediately.`, evidence: `Protocol: ${proto}`, asset: `${domain}:443`, technologies });
   512→  } else if (proto.includes('TLSv1.1')) {
   513→    findings.push({ title: 'Deprecated TLS 1.1 Detected', severity: 'high', category: 'ssl',
   514→      description: `TLS 1.1 deprecated in 2020. Known weaknesses. PCI-DSS requires disabling. Support TLS 1.2+ only.`, evidence: `Protocol: ${proto}`, asset: `${domain}:443`, technologies });
   515→  } else if (proto.includes('TLSv1.3')) {
   516→    technologies.push('TLS 1.3');
   517→    findings.push({ title: 'Modern TLS 1.3 Negotiated', severity: 'info', category: 'ssl',
   518→      description: `TLS 1.3 — latest version with 0-RTT, mandatory forward secrecy, no weak ciphers.`, evidence: `Protocol: ${proto}`, asset: `${domain}:443`, technologies });
   519→  } else if (proto.includes('TLSv1.2')) {
   520→    technologies.push('TLS 1.2');
   521→    findings.push({ title: 'TLS 1.2 Negotiated', severity: 'info', category: 'ssl',
   522→      description: `TLS 1.2 — secure when configured with strong cipher suites. Consider enabling TLS 1.3.`, evidence: `Protocol: ${proto}`, asset: `${domain}:443`, technologies });
   523→  }
   524→
   525→  // Cipher
   526→  const cipM = sslConnect.match(/Cipher\s*:\s*([^\n]+)/);
   527→  const cipher = cipM ? cipM[1].trim() : '';
   528→  const weakC = ['RC4','DES','MD5','NULL','EXPORT','3DES'];
   529→  if (weakC.some(w => cipher.toUpperCase().includes(w))) {
   530→    findings.push({ title: `Weak Cipher Suite: ${cipher}`, severity: 'high', category: 'ssl',
   531→      description: `Weak cipher (${cipher}) — vulnerable to known attacks. Use AEAD ciphers (AES-GCM, ChaCha20-Poly1305) with ECDHE.`, evidence: `Cipher: ${cipher}`, asset: `${domain}:443`, technologies });
   532→  } else if (cipher) {
   533→    findings.push({ title: `Cipher Suite: ${cipher}`, severity: 'info', category: 'ssl',
   534→      description: `Negotiated: ${cipher}.`, evidence: `Cipher: ${cipher}`, asset: `${domain}:443`, technologies });
   535→  }
   536→
   537→  return { findings, technologies };
   538→}
   539→
   540→// ═══════════════════════════════════════════════════════════════════════
   541→// 5. PORT PROBING (async parallel)
   542→// ═══════════════════════════════════════════════════════════════════════
   543→const WEB_PORTS = [
   544→  { port: 80, service: 'HTTP', risk: 'low', desc: 'Standard HTTP — should redirect to HTTPS' },
   545→  { port: 443, service: 'HTTPS', risk: 'info', desc: 'Standard HTTPS' },
   546→  { port: 8080, service: 'HTTP-Alt', risk: 'high', desc: 'Dev servers, proxies, Tomcat' },
   547→  { port: 8443, service: 'HTTPS-Alt', risk: 'medium', desc: 'Admin panels, Java apps' },
   548→  { port: 3000, service: 'Dev Server', risk: 'high', desc: 'Node.js/Express dev default' },
   549→  { port: 5000, service: 'Dev Server', risk: 'high', desc: 'Flask/React dev default' },
   550→  { port: 8000, service: 'HTTP-Dev', risk: 'high', desc: 'Django, Go dev' },
   551→  { port: 8888, service: 'Jupyter/Proxy', risk: 'critical', desc: 'Jupyter Notebook or proxy' },
   552→  { port: 9090, service: 'Prometheus', risk: 'high', desc: 'Prometheus monitoring' },
   553→  { port: 9200, service: 'Elasticsearch', risk: 'critical', desc: 'Elasticsearch REST API' },
   554→  { port: 5601, service: 'Kibana', risk: 'high', desc: 'Kibana dashboard' },
   555→  { port: 27017, service: 'MongoDB', risk: 'critical', desc: 'MongoDB default port' },
   556→];
   557→
   558→async function probePorts(domain: string): Promise<Finding[]> {
   559→  const results = await Promise.all(
   560→    WEB_PORTS.map(async ({ port, service, risk, desc }) => {
   561→      const r = await run(`curl -sI --max-time 3 -k https://${domain}:${port} 2>/dev/null || curl -sI --max-time 3 http://${domain}:${port} 2>/dev/null`, 5000);
   562→      if (r && r.includes('HTTP/')) {
   563→        const status = r.split('\n')[0];
   564→        const srvM = r.match(/server:\s*(.+)/i);
   565→        const srv = srvM ? ` [${srvM[1].trim()}]` : '';
   566→        return { port, service, risk, desc, status: status + srv };
   567→      }
   568→      return null;
   569→    })
   570→  );
   571→
   572→  return results.filter(Boolean).map(({ port, service, risk, desc, status }) => ({
   573→    title: `Port ${port}/tcp OPEN — ${service}`,
   574→    severity: risk === 'info' ? 'info' : risk, category: 'port',
   575→    description: `${service} accessible on port ${port}. ${desc}.${risk === 'critical' || risk === 'high' ? ' This should not be publicly accessible.' : ''}`,
   576→    evidence: status, asset: `${domain}:${port}`,
   577→  }));
   578→}

   581→// MAIN SCAN ENDPOINT
   582→// ═══════════════════════════════════════════════════════════════════════
   583→export async function POST(request: NextRequest) {
   584→  try {
   585→    const body = await request.json();
   586→    const { domain, scanType } = body;
   587→
   588→    if (!domain || typeof domain !== 'string') {
   589→      return NextResponse.json({ error: 'Domain is required' }, { status: 400 });
   590→    }
   591→
   592→    const cleanDomain = domain.replace(/^(https?:\/\/)?(www\.)?/, '').replace(/\/.*$/, '').replace(/:\d+$/, '').toLowerCase();
   593→
   594→    // Pre-check: domain must resolve
   595→    const preCheckIp = await resolveIP(cleanDomain);
   596→    if (!preCheckIp) {
   597→      return NextResponse.json({ error: `Domain "${cleanDomain}" does not resolve. Check the name and try again.` }, { status: 400 });
   598→    }
   599→
   600→    // DB: target + scan record
   601→    let target = await db.scanTarget.findFirst({ where: { domain: cleanDomain } });
   602→    if (!target) {
   603→      target = await db.scanTarget.create({ data: { domain: cleanDomain, ip: preCheckIp } });
   604→    } else {
   605→      await db.scanTarget.update({ where: { id: target.id }, data: { ip: preCheckIp } });
   606→    }
   607→
   608→    const scan = await db.scan.create({
   609→      data: { targetId: target.id, status: 'running', scanType: scanType || 'full' },
   610→    });
   611→
   612→    const isQuick = scanType === 'quick';
   613→    const allFindings: Finding[] = [];
   614→    const allTech = new Set<string>();
   615→
   616→    // ── 1. DNS (always, all queries parallel) ──────────────────────
   617→    const dnsResult = await enumerateDNS(cleanDomain);
   618→    allFindings.push(...dnsResult.findings);
   619→
   620→    // ── 2. Subdomains (parallel DNS resolution) ────────────────────
   621→    const subFindings = await enumerateSubdomains(cleanDomain, isQuick);
   622→    allFindings.push(...subFindings);
   623→
   624→    // ── 3. HTTP headers (async curl) ───────────────────────────────
   625→    const httpResult = await analyzeHTTPHeaders(cleanDomain);
   626→    allFindings.push(...httpResult.findings);
   627→    httpResult.technologies.forEach(t => allTech.add(t));
   628→
   629→    // ── 4. SSL/TLS (async openssl) ─────────────────────────────────
   630→    const sslResult = await analyzeSSL(cleanDomain);
   631→    allFindings.push(...sslResult.findings);
   632→    sslResult.technologies.forEach(t => allTech.add(t));
   633→
   634→    // ── 5. Port probing (full only, async parallel) ────────────────
   635→    if (!isQuick) {
   636→      const portFindings = await probePorts(cleanDomain);
   637→      allFindings.push(...portFindings);
   638→    }
   639→
   640→    // ── Technology findings ─────────────────────────────────────────
   641→    for (const tech of allTech) {
   642→      allFindings.push({
   643→        title: `Technology Detected: ${tech}`, severity: 'info', category: 'technology',
   644→        description: `${tech} identified on ${cleanDomain} via HTTP headers, SSL cert metadata, or response analysis.`,
   645→        evidence: 'Detected via HTTP/SSL/response fingerprinting', asset: tech,
   646→      });
   647→    }
   648→
   649→    // ── Wildcard DNS check ─────────────────────────────────────────
   650→    const wcIp = await resolveIP(`nonexistent-wildcard-test-12345.${cleanDomain}`);
   651→    if (wcIp) {
   652→      allFindings.push({
   653→        title: 'Wildcard DNS Record Detected', severity: 'medium', category: 'dns',
   654→        description: `Wildcard DNS active — even nonexistent subdomains resolve to ${wcIp}. Subdomain enumeration may have false positives.`,
   655→        evidence: `nonexistent-wildcard-test-12345.${cleanDomain} resolved to ${wcIp}`, asset: cleanDomain,
   656→      });
   657→    }
   658→
   659→    // ── Save to DB ─────────────────────────────────────────────────
   660→    for (const f of allFindings) {
   661→      await db.finding.create({ data: { scanId: scan.id, ...f } });
   662→    }
   663→
   664→    // ── Risk score ─────────────────────────────────────────────────
   665→    const c = allFindings.filter(f => f.severity === 'critical').length;
   666→    const h = allFindings.filter(f => f.severity === 'high').length;
   667→    const m = allFindings.filter(f => f.severity === 'medium').length;
   668→    const l = allFindings.filter(f => f.severity === 'low').length;
   669→    const i = allFindings.filter(f => f.severity === 'info').length;
   670→    const riskScore = Math.min(100, Math.round(c*25 + h*15 + m*8 + l*3 + i*1));
   671→
   672→    await db.scan.update({
   673→      where: { id: scan.id },
   674→      data: { status: 'completed', completedAt: new Date(), riskScore, totalVulns: allFindings.length, criticalCount: c, highCount: h, mediumCount: m, lowCount: l, infoCount: i },
   675→    });
   676→
   677→    return NextResponse.json({
   678→      success: true,
   679→      scan: {
   680→        id: scan.id, domain: cleanDomain, status: 'completed', riskScore,
   681→        totalVulns: allFindings.length, critical: c, high: h, medium: m, low: l, info: i,
   682→        findings: allFindings.map(f => ({
   683→          id: f.title.toLowerCase().replace(/\s+/g, '-').substring(0, 20),
   684→          ...f,
   685→        })),
   686→      },
   687→    });
   688→  } catch (error) {
   689→    console.error('Scan error:', error);
   690→    return NextResponse.json({ error: 'Scan failed: ' + (error instanceof Error ? error.message : 'unknown') }, { status: 500 });
   691→  }
   692→}
