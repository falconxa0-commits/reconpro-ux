    93→      return {
    94→        color: '#ff3355',
    95→        bg: 'rgba(244,63,94,0.15)',
    96→        border: 'rgba(244,63,94,0.3)',
    97→        icon: XCircle,
    98→        label: 'CRITICAL',
    99→      };
   100→    case 'high':
   101→      return {
   102→        color: '#ff8844',
   103→        bg: 'rgba(251,191,36,0.15)',
   104→        border: 'rgba(251,191,36,0.3)',
   105→        icon: AlertTriangle,
   106→        label: 'HIGH',
   107→      };
   108→    case 'medium':
   109→      return {
   110→        color: '#d29922',
   111→        bg: 'rgba(210,153,34,0.15)',
   112→        border: 'rgba(210,153,34,0.3)',
   113→        icon: AlertCircle,
   114→        label: 'MEDIUM',
   115→      };
   116→    case 'low':
   117→      return {
   118→        color: '#44aaff',
   119→        bg: 'rgba(88,166,255,0.15)',
   120→        border: 'rgba(88,166,255,0.3)',
   121→        icon: FileWarning,
   122→        label: 'LOW',
   123→      };
   124→    default:
   125→      return {
   126→        color: '#444444',
   127→        bg: 'rgba(139,148,158,0.15)',
   128→        border: 'rgba(139,148,158,0.3)',
   129→        icon: AlertCircle,
   130→        label: 'INFO',
   131→      };
   132→  }
   133→}
   134→
   135→function getScheduleColor(schedule: string): string {
   136→  switch (schedule) {
   137→    case 'hourly':
   138→      return 'bg-[rgba(52,211,153,0.15)] text-[#00ff88] border-[rgba(52,211,153,0.3)]';
   139→    case 'daily':
   140→      return 'bg-[rgba(88,166,255,0.15)] text-[#44aaff] border-[rgba(88,166,255,0.3)]';
   141→    case 'weekly':
   142→      return 'bg-[rgba(210,153,34,0.15)] text-[#d29922] border-[rgba(210,153,34,0.3)]';
   143→    case 'monthly':
   144→      return 'bg-[rgba(168,85,247,0.15)] text-[#888888] border-[rgba(168,85,247,0.3)]';
   145→    default:
   146→      return 'bg-[rgba(139,148,158,0.15)] text-[#444444] border-[rgba(139,148,158,0.3)]';
   147→  }
   148→}

   150→function getPolicyStatusConfig(status: string) {
   151→  switch (status) {
   152→    case 'active':
   153→      return { color: '#00ff88', label: 'Active' };
   154→    case 'warning':
   155→      return { color: '#d29922', label: 'Warning' };
   156→    case 'error':
   157→      return { color: '#ff3355', label: 'Error' };
   158→    case 'idle':
   159→      return { color: '#333333', label: 'Paused' };
   160→    default:
   161→      return { color: '#333333', label: 'Unknown' };
   162→  }
   163→}
   164→
   165→function getScanTypeIcon(type: string) {
   166→  switch (type) {
   167→    case 'Full Infrastructure':
   168→    case 'full':
   169→      return Server;
   170→    case 'API Surface':
   171→    case 'api':
   172→      return Globe;
   173→    case 'SSL/TLS':
   174→    case 'ssl':
   175→      return Shield;
   176→    case 'DNS Recon':
   177→    case 'dns':
   178→      return MonitorCheck;
   179→    default:
   180→      return Activity;
   181→  }
   182→}

   184→function getAlertStatusConfig(status: string) {
   185→  switch (status) {
   186→    case 'new':
   187→      return { color: '#ff3355', bg: 'rgba(244,63,94,0.15)', label: 'NEW' };
   188→    case 'acknowledged':
   189→      return { color: '#d29922', bg: 'rgba(210,153,34,0.15)', label: 'ACK' };
   190→    case 'resolved':
   191→      return { color: '#00ff88', bg: 'rgba(52,211,153,0.15)', label: 'FIXED' };
   192→    default:
   193→      return { color: '#333333', bg: 'rgba(72,79,88,0.15)', label: 'N/A' };
   194→  }

   215→
   216→export function MonitoringPanel() {
   217→  const authHeaders = useAuthHeaders();
   218→  const [data, setData] = useState<MonitoringData | null>(null);
   219→  const [loading, setLoading] = useState(true);
   220→  const [error, setError] = useState('');
   221→  const [addOpen, setAddOpen] = useState(false);
   222→  const [newName, setNewName] = useState('');
   223→  const [newDomain, setNewDomain] = useState('');
   224→  const [newSchedule, setNewSchedule] = useState<'hourly' | 'daily' | 'weekly' | 'monthly'>('daily');
   225→  const [newScanType, setNewScanType] = useState('full');
   226→
   227→  const fetchData = useCallback(async () => {
   228→    setLoading(true);
   229→    setError('');
   230→    try {
   231→      const res = await fetch('/api/monitoring', { headers: authHeaders });
   232→      if (!res.ok) throw new Error(`HTTP ${res.status}`);
   233→      const json = await res.json();
   234→      setData({
   235→        policies: json.policies || [],
   236→        scheduledRuns: json.scheduledRuns || [],
   237→        alerts: json.alerts || [],
   238→      });
   239→    } catch (err) {
   240→      console.error('Failed to fetch monitoring data:', err);
   241→      setError('Failed to load monitoring data. Please try again.');
   242→    } finally {
   243→      setLoading(false);
   244→    }
   245→  }, [authHeaders]);
   246→
   247→  useEffect(() => {
   248→    fetchData();
   249→  }, [fetchData]);
   250→
   251→  const policies = data?.policies || [];
   252→  const scheduledRuns = data?.scheduledRuns || [];
   253→  const alerts = data?.alerts || [];
   254→
   255→  const togglePolicy = async (id: string) => {
   256→    const policy = policies.find((p) => p.id === id);
   257→    if (!policy) return;
   258→    try {
   259→      await fetch('/api/monitoring', {
   260→        method: 'PATCH',
   261→        headers: { 'Content-Type': 'application/json', ...authHeaders },
   262→        body: JSON.stringify({ id, enabled: !policy.enabled }),
   263→      });
   264→      await fetchData();
   265→    } catch (err) {
   266→      console.error('Failed to toggle policy:', err);
   267→    }
   268→  };
   269→
   270→  const deletePolicy = async (id: string) => {
   271→    try {
   272→      await fetch(`/api/monitoring?id=${id}`, { method: 'DELETE' });
   273→      await fetchData();
   274→    } catch (err) {
   275→      console.error('Failed to delete policy:', err);
   276→    }
   277→  };
   278→
   279→  const handleNewPolicy = async () => {
   280→    if (!newName || !newDomain) return;
   281→    try {
   282→      await fetch('/api/monitoring', {
   283→        method: 'POST',
   284→        headers: { 'Content-Type': 'application/json', ...authHeaders },
   285→        body: JSON.stringify({
   286→          name: newName,
   287→          targetDomain: newDomain,
   288→          schedule: newSchedule,
   289→          scanType: newScanType,
   290→        }),
   291→      });
   292→      setAddOpen(false);
   293→      setNewName('');
   294→      setNewDomain('');
   295→      setNewSchedule('daily');
   296→      setNewScanType('full');
   297→      await fetchData();
   298→    } catch (err) {
   299→      console.error('Failed to create policy:', err);
   300→    }
   301→  };
   302→
   303→  const handleRunNow = (id: string) => {
   304→    console.log('Run policy now:', id);
   305→  };
   306→
   307→  const activePolicies = policies.filter((p) => p.enabled).length;
   308→  const totalFindings = policies.reduce((acc, p) => acc + p.findings, 0);
   309→
   310→  if (loading) {
   311→    return (
   312→      <div className="w-full flex items-center justify-center py-20">
   313→        <p className="text-[#444444]">Loading...</p>
   314→      </div>
   315→    );
   316→  }
   317→
   318→  if (error) {
   319→    return (
   320→      <div className="w-full flex flex-col items-center justify-center gap-3 py-20">
   321→        <AlertCircle className="h-8 w-8 text-red-400" />
   322→        <p className="text-sm text-red-400">{error}</p>
   323→        <Button variant="outline" size="sm" onClick={fetchData} className="border-zinc-700 text-white hover:bg-zinc-800">
   324→          <RefreshCw className="mr-2 h-3.5 w-3.5" />
   325→          Retry
   326→        </Button>
   327→      </div>
   328→    );
   329→  }
   330→
   331→  return (
   332→    <motion.div
   333→      variants={containerVariants}
   334→      initial="hidden"
   335→      animate="visible"
   336→      className="w-full space-y-6"
   337→    >
   338→      {/* ── Header ──────────────────────────────────────────────────────── */}
   339→      <motion.div variants={itemVariants} className="flex flex-col sm:flex-row sm:items-center justify-between gap-4">
   340→        <div className="flex items-center gap-3">
   341→          <div className="p-2 rounded-lg bg-[rgba(52,211,153,0.1)] border border-[rgba(52,211,153,0.2)]">
   342→            <Activity className="w-5 h-5 text-[#00ff88]" />
   343→          </div>
   344→          <div>
   345→            <h2 className="text-xl font-bold text-[#f0f0f0]">Continuous Monitoring</h2>
   346→            <p className="text-sm text-[#444444]">Automated attack surface surveillance and policy management</p>
   347→          </div>
   348→        </div>
   349→        <Dialog open={addOpen} onOpenChange={setAddOpen}>
   350→          <DialogTrigger asChild>
   351→            <Button className="bg-[#00ff88] hover:bg-[#00cc6a] text-[#0a0d14] font-semibold gap-2">
   352→              <Plus className="w-4 h-4" />
   353→              New Policy
   354→            </Button>
   355→          </DialogTrigger>
   356→          <DialogContent className="bg-[#080b14] border-[#21262d] text-[#f0f0f0]">
   357→            <DialogHeader>
   358→              <DialogTitle className="text-[#f0f0f0]">Create Monitoring Policy</DialogTitle>
   359→              <DialogDescription className="text-[#444444]">
   360→                Set up automated monitoring for a target domain.
   361→              </DialogDescription>
   362→            </DialogHeader>
   363→            <div className="space-y-4 py-4">
   364→              <div className="space-y-2">
   365→                <label className="text-sm font-medium text-[#f0f0f0]">Policy Name</label>
   366→                <Input
   367→                  placeholder="e.g. Production Infrastructure"
   368→                  value={newName}
   369→                  onChange={(e) => setNewName(e.target.value)}
   370→                  className="bg-[#050710] border-[#21262d] text-[#f0f0f0] placeholder:text-[#333333] focus:border-[#00ff88]"
   371→                />
   372→              </div>
   373→              <div className="space-y-2">
   374→                <label className="text-sm font-medium text-[#f0f0f0]">Target Domain</label>
   375→                <Input
   376→                  placeholder="e.g. example.com"
   377→                  value={newDomain}
   378→                  onChange={(e) => setNewDomain(e.target.value)}
   379→                  className="bg-[#050710] border-[#21262d] text-[#f0f0f0] placeholder:text-[#333333] focus:border-[#00ff88]"
   380→                />
   381→              </div>
   382→              <div className="space-y-2">
   383→                <label className="text-sm font-medium text-[#f0f0f0]">Schedule</label>
   384→                <div className="grid grid-cols-4 gap-2">
   385→                  {(['hourly', 'daily', 'weekly', 'monthly'] as const).map((s) => (
   386→                    <button
   387→                      key={s}
   388→                      onClick={() => setNewSchedule(s)}
   389→                      className={`px-3 py-2 rounded-lg text-sm font-medium border transition-all capitalize ${
   390→                        newSchedule === s
   391→                          ? 'bg-[rgba(52,211,153,0.15)] border-[rgba(52,211,153,0.4)] text-[#00ff88]'
   392→                          : 'bg-[#050710] border-[#21262d] text-[#444444] hover:border-[#30363d]'
   393→                      }`}
   394→                    >
   395→                      {s}
   396→                    </button>
   397→                  ))}
   398→                </div>
   399→              </div>
   400→              <div className="space-y-2">
   401→                <label className="text-sm font-medium text-[#f0f0f0]">Scan Type</label>
   402→                <div className="grid grid-cols-2 gap-2">
   403→                  {['full', 'quick', 'stealth'].map((t) => (
   404→                    <button
   405→                      key={t}
   406→                      onClick={() => setNewScanType(t)}
   407→                      className={`px-3 py-2 rounded-lg text-sm font-medium border transition-all capitalize ${
   408→                        newScanType === t
   409→                          ? 'bg-[rgba(52,211,153,0.15)] border-[rgba(52,211,153,0.4)] text-[#00ff88]'
   410→                          : 'bg-[#050710] border-[#21262d] text-[#444444] hover:border-[#30363d]'
   411→                      }`}
   412→                    >
   413→                      {t}
   414→                    </button>
   415→                  ))}
   416→                </div>
   417→              </div>
   418→            </div>
   419→            <DialogFooter>
   420→              <Button variant="outline" onClick={() => setAddOpen(false)} className="border-[#21262d] text-[#444444] hover:bg-[#050710]">
   421→                Cancel
   422→              </Button>
   423→              <Button onClick={handleNewPolicy} className="bg-[#00ff88] hover:bg-[#00cc6a] text-[#0a0d14] font-semibold">
   424→                Create Policy
   425→              </Button>
   426→            </DialogFooter>
   427→          </DialogContent>
   428→        </Dialog>
   429→      </motion.div>
   430→
   431→      {/* ── Stats Row ───────────────────────────────────────────────────── */}
   432→      <motion.div variants={itemVariants} className="grid grid-cols-2 lg:grid-cols-4 gap-4">
   433→        {[
   434→          { label: 'Active Policies', value: activePolicies, icon: Shield, color: '#00ff88', subtitle: `of ${policies.length} total` },
   435→          { label: 'Monitored Assets', value: policies.length, icon: Globe, color: '#44aaff', subtitle: 'domains & endpoints' },
   436→          { label: 'Alerts Today', value: alerts.filter((a) => a.status === 'new').length, icon: AlertTriangle, color: '#ff8844', subtitle: `${totalFindings} findings` },
   437→          { label: 'Uptime', value: policies.length > 0 ? '99.97%' : '—', icon: TrendingUp, color: '#00ff88', subtitle: 'last 30 days' },
   438→        ].map((stat) => (
   439→          <motion.div
   440→            key={stat.label}
   441→            whileHover={cardHover}
   442→            className="rounded-xl border border-[#21262d] bg-[#080b14] p-4"
   443→          >
   444→            <div className="flex items-center justify-between mb-3">
   445→              <span className="text-xs font-medium text-[#444444] uppercase tracking-wider">{stat.label}</span>
   446→              <stat.icon className="w-4 h-4" style={{ color: stat.color }} />
   447→            </div>
   448→            <p className="text-2xl font-bold text-[#f0f0f0]">{stat.value}</p>
   449→            <p className="text-[10px] text-[#333333] mt-1">{stat.subtitle}</p>
   450→          </motion.div>
   451→        ))}
   452→      </motion.div>
   453→
   454→      {/* ── Policy Cards ────────────────────────────────────────────────── */}
   455→      <motion.div variants={itemVariants}>
   456→        <div className="flex items-center justify-between mb-3">
   457→          <h3 className="text-sm font-semibold text-[#f0f0f0]">Monitoring Policies</h3>
   458→          <span className="text-[10px] text-[#333333] uppercase tracking-wider">{policies.length} policies</span>
   459→        </div>
   460→        {policies.length === 0 ? (
   461→          <div className="rounded-xl border border-[#21262d] bg-[#080b14] p-8 text-center">
   462→            <Shield className="w-8 h-8 text-[#333333] mx-auto mb-3" />
   463→            <p className="text-sm text-[#444444]">No policies yet. Create one to start monitoring.</p>
   464→          </div>
   465→        ) : (
   466→          <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
   467→            {policies.map((policy, idx) => {
   468→              const ScanIcon = getScanTypeIcon(policy.scanType);
   469→              const statusCfg = getPolicyStatusConfig(policy.status);
   470→              return (
   471→                <motion.div
   472→                  key={policy.id}
   473→                  variants={itemVariants}
   474→                  whileHover={cardHover}
   475→                  className="rounded-xl border border-[#21262d] bg-[#080b14] overflow-hidden relative"
   476→                >
   477→                  {/* Status indicator bar */}
   478→                  <div className="absolute top-0 left-0 right-0 h-0.5 opacity-60" style={{ backgroundColor: statusCfg.color }} />
   479→
   480→                  <div className="p-5 pt-6">
   481→                    {/* Top row: name + toggle */}
   482→                    <div className="flex items-start justify-between mb-4">
   483→                      <div className="flex items-center gap-3">
   484→                        <div
   485→                          className="w-10 h-10 rounded-lg flex items-center justify-center"
   486→                          style={{ backgroundColor: statusCfg.color + '15' }}
   487→                        >
   488→                          <ScanIcon className="w-5 h-5" style={{ color: statusCfg.color }} />
   489→                        </div>
   490→                        <div>
   491→                          <h4 className="text-sm font-bold text-[#f0f0f0]">{policy.name}</h4>
   492→                          <div className="flex items-center gap-2 mt-0.5">
   493→                            <span className={`inline-flex items-center px-2 py-0.5 rounded text-[10px] font-medium border ${getScheduleColor(policy.schedule)}`}>
   494→                              <Timer className="w-2.5 h-2.5 mr-1" />
   495→                              {policy.schedule}
   496→                            </span>
   497→                            <span
   498→                              className="w-1.5 h-1.5 rounded-full"
   499→                              style={{ backgroundColor: statusCfg.color }}
   500→                            />
   501→                            <span className="text-[10px] text-[#444444]">{statusCfg.label}</span>
   502→                          </div>
   503→                        </div>
   504→                      </div>
   505→                      <div className="flex items-center gap-2">
   506→                        <button
   507→                          onClick={() => handleRunNow(policy.id)}
   508→                          className="p-1.5 rounded-md border border-[#21262d] text-[#444444] hover:text-[#00ff88] hover:border-[rgba(52,211,153,0.3)] hover:bg-[rgba(52,211,153,0.1)] transition-all"
   509→                          title="Run now"
   510→                        >
   511→                          <Play className="w-3.5 h-3.5" />
   512→                        </button>
   513→                        <button
   514→                          onClick={() => deletePolicy(policy.id)}
   515→                          className="p-1.5 rounded-md border border-[#21262d] text-[#444444] hover:text-[#ff3355] hover:border-[rgba(244,63,94,0.3)] hover:bg-[rgba(244,63,94,0.1)] transition-all"
   516→                          title="Delete"
   517→                        >
   518→                          <Trash2 className="w-3.5 h-3.5" />
   519→                        </button>
   520→                        <Switch
   521→                          checked={policy.enabled}
   522→                          onCheckedChange={() => togglePolicy(policy.id)}

   526→                    </div>
   527→
   528→                    {/* Target & scan info */}
   529→                    <div className="grid grid-cols-2 gap-3 mb-4">
   530→                      <div className="space-y-1">
