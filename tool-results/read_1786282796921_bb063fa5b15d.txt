   105→
   106→class FindingDetailModal(ModalScreen):
   107→    """Modal overlay showing full finding details."""
   108→
   109→    def __init__(self, finding: Dict[str, Any]) -> None:
   110→        super().__init__()
   111→        self.finding = finding
   112→
   113→    def compose(self) -> ComposeResult:
   114→        f = self.finding
   115→        sev = f.get("severity", "info").lower()
   116→        sev_color = SEV_STYLES.get(sev, "white")
   117→        title = f.get("title", "Unknown")
   118→        category = f.get("category", "")
   119→        module = f.get("module", "")
   120→        pts = f.get("points_deducted", 0)
   121→        evidence = f.get("evidence", "")
   122→        remediation = f.get("remediation", "No remediation available.")
   123→        description = f.get("description", "")
   124→        asset = f.get("asset", "")
   125→        related_cves = f.get("related_cves", [])
   126→
   127→        lines: List[str] = []
   128→        lines.append(f"[{sev_color}][{sev.upper()}] {title}[/]")
   129→        lines.append("")
   130→        if asset:
   131→            lines.append(f"[dim]Asset:[/] [cyan]{asset}[/]")
   132→        if category:
   133→            lines.append(f"[dim]Category:[/] {category}")
   134→        if module:
   135→            lines.append(f"[dim]Module:[/] {module}")
   136→        lines.append(f"[dim]Points:[/] -{pts}")
   137→        if related_cves:
   138→            cve_ids = ", ".join(
   139→                c.get("cve_id", "") for c in related_cves[:5] if c.get("cve_id")
   140→            )
   141→            if cve_ids:
   142→                lines.append(f"[dim]CVEs:[/] [red]{cve_ids}[/]")
   143→        lines.append("")
   144→        if description:
   145→            lines.append(f"[bold]Description:[/]")
   146→            lines.append(f"[dim]{description}[/]")
   147→            lines.append("")
   148→        if evidence:
   149→            lines.append(f"[bold]Evidence:[/]")
   150→            lines.append(f"[dim]{evidence}[/]")
   151→            lines.append("")
   152→        lines.append(f"[bold]{GREEN}Remediation:[/]")
   153→        lines.append(f"{GREEN}{remediation}[/]")
   154→
   155→        content = "\n".join(lines)
   156→
   157→        with Vertical(classes="modal-container"):
   158→            yield Label(content, classes="modal-content")
   159→            with Horizontal(classes="modal-actions"):
   160→                yield Button("Close", variant="primary", id="close-modal")
   161→
   162→    def on_button_pressed(self, event: Button.Pressed) -> None:
   163→        if event.button.id == "close-modal":
   164→            self.dismiss()
   165→

   167→# ════════════════════════════════════════════════════════════════════════════════
   168→# Boot Screen Widget
   169→# ════════════════════════════════════════════════════════════════════════════════
   170→
   171→
   172→class BootScreen(Widget):
   173→    """Full-screen animated boot sequence."""
   174→
   175→    def __init__(self, **kwargs: Any) -> None:
   176→        super().__init__(**kwargs)
   177→        self._step = 0
   178→        self._spinner_idx = 0
   179→        self._module_idx = 0
   180→        self._module_names = list(MODULE_NAMES.values())
   181→        self._done = False
   182→        self._timer: Optional[Timer] = None
   183→
   184→    def compose(self) -> ComposeResult:
   185→        with Vertical(classes="boot-container"):
   186→            yield Static("", id="boot-line-1", classes="boot-title")
   187→            yield Static("", id="boot-line-2", classes="boot-line")
   188→            yield Static("", id="boot-line-3", classes="boot-line")
   189→            yield Static("", id="boot-line-4", classes="boot-line")
   190→
   191→    def on_mount(self) -> None:
   192→        self._timer = self.set_interval(0.04, self._tick)
   193→
   194→    def _tick(self) -> None:
   195→        app = self.app
   196→        if not isinstance(app, NexusApp):
   197→            return
   198→
   199→        line1 = app.query_one("#boot-line-1", Static)
   200→        line2 = app.query_one("#boot-line-2", Static)
   201→        line3 = app.query_one("#boot-line-3", Static)
   202→        line4 = app.query_one("#boot-line-4", Static)
   203→
   204→        if self._step == 0:
   205→            # Letter-by-letter title
   206→            title = "RECONPRO NEXUS"
   207→            chars = title[: self._spinner_idx]
   208→            self._spinner_idx += 1
   209→            line1.update(f"[{CYAN} bold]{chars}[/]")
   210→            if self._spinner_idx > len(title):
   211→                self._spinner_idx = 0
   212→                self._step = 1
   213→                line1.update(f"[{CYAN} bold]{title}[/]  [{DIM_CYAN}]v{VERSION}[/]")
   214→
   215→        elif self._step == 1:
   216→            # Spinner for "Initializing agent..."
   217→            frame = SPINNER_FRAMES[self._spinner_idx % len(SPINNER_FRAMES)]
   218→            self._spinner_idx += 1
   219→            line2.update(f"  [{CYAN}]{frame}[/] [{DIM_CYAN}]Initializing agent...[/]")
   220→            if self._spinner_idx > 30:
   221→                self._spinner_idx = 0
   222→                self._step = 2
   223→                line2.update(f"  [{GREEN}]✓[/] [{DIM_CYAN}]Agent initialized.[/]")
   224→
   225→        elif self._step == 2:
   226→            # Module names appearing one by one
   227→            if self._module_idx < len(self._module_names):
   228→                visible = self._module_names[: self._module_idx + 1]
   229→                self._module_idx += 1
   230→                modules_str = "  ".join(f"[{CYAN}]○[/] [{DIM_CYAN}]{m}[/]" for m in visible)
   231→                remaining = len(self._module_names) - self._module_idx
   232→                if remaining > 0:
   233→                    modules_str += f"  [{DIM_CYAN}]+{remaining} more...[/]"
   234→                frame = SPINNER_FRAMES[self._spinner_idx % len(SPINNER_FRAMES)]
   235→                self._spinner_idx += 1
   236→                line3.update(
   237→                    f"  [{CYAN}]{frame}[/] [{DIM_CYAN}]Loading {len(self._module_names)} modules...[/]\n  {modules_str}"
   238→                )
   239→            else:
   240→                all_mods = "  ".join(
   241→                    f"[{GREEN}]●[/] [{DIM_CYAN}]{m}[/]" for m in self._module_names
   242→                )
   243→                line3.update(
   244→                    f"  [{GREEN}]✓[/] [{DIM_CYAN}]All {len(self._module_names)} modules loaded.[/]\n  {all_mods}"
   245→                )
   246→                self._step = 3
   247→                self._spinner_idx = 0
   248→
   249→        elif self._step == 3:
   250→            # "Ready." with flash
   251→            if self._spinner_idx == 0:
   252→                line4.update(f"[{YELLOW} bold]  ▶  READY.[/]")
   253→            elif self._spinner_idx == 6:
   254→                line4.update(f"[{GREEN} bold]  ▶  READY.[/]")
   255→            elif self._spinner_idx >= 10:
   256→                self._done = True
   257→                if self._timer:
   258→                    self._timer.stop()
   259→                self.app.call_from_thread(self.app._finish_boot)
   260→            self._spinner_idx += 1
   261→

   263→# ════════════════════════════════════════════════════════════════════════════════
   264→# Module Status Cell Widget
   265→# ════════════════════════════════════════════════════════════════════════════════
   266→
   267→
   268→class ModuleCell(Static):
   269→    """Single module status cell in the grid."""
   270→
   271→    def __init__(self, module_id: str, name: str, **kwargs: Any) -> None:
   272→        super().__init__(**kwargs)
   273→        self.module_id = module_id
   274→        self.module_name = name
   275→        self.status = "idle"  # idle | scanning | done | error
   276→        self.finding_count = 0
   277→        self._spinner_idx = 0
   278→        self._timer: Optional[Timer] = None
   279→
   280→    def on_mount(self) -> None:
   281→        self._update_display()
   282→        self._timer = self.set_interval(0.1, self._spin)
   283→
   284→    def _spin(self) -> None:
   285→        if self.status == "scanning":
   286→            self._spinner_idx = (self._spinner_idx + 1) % len(SPINNER_FRAMES)
   287→            self._update_display()
   288→
   289→    def _update_display(self) -> None:
   290→        name_short = self.module_name[:14]
   291→        if self.status == "idle":
   292→            icon = f"[{DIM_CYAN}]○[/]"
   293→            status_text = f"[{DIM_CYAN}]idle[/]"
   294→        elif self.status == "scanning":
   295→            frame = SPINNER_FRAMES[self._spinner_idx]
   296→            icon = f"[{CYAN}]{frame}[/]"
   297→            status_text = f"[{CYAN}]scanning[/]"
   298→        elif self.status == "done":
   299→            icon = f"[{GREEN}]✓[/]"
   300→            status_text = f"[{GREEN}]done ({self.finding_count})[/]"
   301→        elif self.status == "error":
   302→            icon = f"[{RED}]✗[/]"
   303→            status_text = f"[{RED}]error[/]"
   304→        else:
   305→            icon = "○"
   306→            status_text = "idle"
   307→
   308→        self.update(
   309→            f"{icon} [{CYAN} bold]{name_short}[/]\n  {status_text}"
   310→        )
   311→
   312→    def set_status(self, status: str, count: int = 0) -> None:
   313→        self.status = status
   314→        self.finding_count = count
   315→        self._update_display()
   316→
   317→    def on_unmount(self) -> None:
   318→        if self._timer:
   319→            self._timer.stop()
   320→

   323→# Pulsing Status Dot Widget
   324→# ════════════════════════════════════════════════════════════════════════════════
   325→
   326→
   327→class StatusDot(Static):
   328→    """Pulsing dot indicator for scan status."""
   329→
   330→    scanning: reactive[bool] = reactive(False)
   331→
   332→    def __init__(self, **kwargs: Any) -> None:
   333→        super().__init__("", **kwargs)
   334→        self._phase = 0
   335→
   336→    def on_mount(self) -> None:
   337→        self.set_interval(0.3, self._pulse)
   338→
   339→    def watch_scanning(self, old: bool, new: bool) -> None:
   340→        # Update visual state based on scanning status
   341→        if new:
   342→            self.update(f"[{self._get_pulse_color()}]●[/]")
   343→        else:
   344→            self.update(f"[{GREEN}]●[/]")
   345→
   346→    def _get_pulse_color(self) -> str:
   347→        return ["#003322", "#006644", CYAN, "#006644"][self._phase] if self.scanning else GREEN
   348→
   349→    def _pulse(self) -> None:
   350→        self._phase = (self._phase + 1) % 4
   351→        if self.scanning:
   352→            brightness = self._get_pulse_color()
   353→            self.update(f"[{brightness}]●[/]")
   354→        else:
   355→            self.update(f"[{GREEN}]●[/]")

   904→    def on_key(self, event: Key) -> None:
   905→        """Handle key events — boot skip + completer + history + vim-nav."""
   906→        # Skip boot on ANY key press during boot sequence
   907→        if not self._boot_done and not self._boot_skipped:
   908→            self._boot_skipped = True
   909→            try:
   910→                boot = self.query_one("#boot-screen", BootScreen)
   911→                if boot._timer:
   912→                    boot._timer.stop()
   913→            except NoMatches:
   914→                pass
   915→            self._finish_boot()
   916→            return
   917→
   918→        # ── Number keys: quick jump to panels (0=input, 1=chat, 2=findings, 3=modules) ──
   919→        if event.key in ("0", "1", "2", "3") and not self._input_focused():
   920→            jump_map = {"0": "#command-input", "1": "#chat-log", "2": "#findings-feed", "3": "#module-grid"}
   921→            # Skip panel jump to collapsed panels
   922→            target_sel = jump_map[event.key]
   923→            if event.key == "1" and self._left_collapsed:
   924→                target_sel = "#findings-feed"
   925→            if event.key == "3" and self._modules_collapsed:
   926→                target_sel = "#findings-feed"
   927→            self._focus_widget(target_sel)
   928→            event.stop()
   929→            return
   930→
   931→        # ── Phase D: [ and ] to toggle panel collapse ──
   932→        if event.key == "[" and not self._input_focused():
   933→            self._toggle_left_panel()
   934→            event.stop()
   935→            return
   936→        if event.key == "]" and not self._input_focused():
   937→            self._toggle_right_panel()
   938→            event.stop()
   939→            return
   940→        if event.key == "=" and not self._input_focused():
   941→            self._toggle_modules()
   942→            event.stop()
   943→            return
   944→        # ── Phase D: | resets split to auto, \\ resets to 50:50 ──
   945→        if event.key == "|" and not self._input_focused():
   946→            self._user_manually_resized = False
   947→            self._apply_responsive_layout(self.size.width, self.size.height)
   948→            event.stop()
   949→            return
   950→        if event.key == "\\" and not self._input_focused():
   951→            self._user_manually_resized = True
   952→            self._split_ratio = 50
   953→            self._apply_split_ratio()
   954→            event.stop()
   955→            return
   956→
   957→        # ── Phase E: ? opens help overlay ──
   958→        if event.key == "question_mark" and not self._input_focused():
   959→            self._show_help_overlay()
   960→            event.stop()
   961→            return
   962→
   963→        # ── Tab: completer accept/cycle / quick-pick on empty / cycle focus ──
   964→        if event.key == "tab":
   965→            try:
   966→                inp = self.query_one("#command-input", Input)
   967→                comp = self.query_one("#cmd-completer", CommandCompleter)
   968→                if inp.has_focus:
   969→                    # Completer visible → accept top suggestion
   970→                    if comp.visible:
   971→                        accepted = comp.accept_top()
   972→                        if accepted:
   973→                            self._apply_completion(inp, comp, accepted)
   974→                        event.stop()
   975→                        return
   976→                    # Empty input → quick-pick menu
   977→                    if not inp.value.strip():
   978→                        comp.show_quick_pick()
   979→                        event.stop()
   980→                        return
   981→            except NoMatches:
   982→                pass
   983→            # Fall through to default Tab behavior (cycle focus)
   984→            return
   985→
   986→        # ── Escape: dismiss completer, close modal, or jump to input ──
   987→        if event.key == "escape":
   988→            try:
   989→                comp = self.query_one("#cmd-completer", CommandCompleter)
   990→                if comp.visible:
   991→                    comp.hide()
   992→                    event.stop()
   993→                    return
   994→            except NoMatches:
   995→                pass
   996→            # If any modal screen is open, Escape will dismiss it via Textual
   997→            # Otherwise jump back to input
   998→            if not self._input_focused():
   999→                self._focus_input()
  1000→                event.stop()
  1001→                return
  1002→
  1003→        # ── Up/Down in completer: cycle suggestions ──
  1004→        if event.key in ("up", "down"):
  1005→            try:
  1006→                comp = self.query_one("#cmd-completer", CommandCompleter)
  1007→                inp = self.query_one("#command-input", Input)
  1008→                if comp.visible and inp.has_focus:
  1009→                    direction = -1 if event.key == "up" else 1
  1010→                    comp.cycle_selection(direction)
  1011→                    event.stop()
  1012→                    return
  1013→            except NoMatches:
  1014→                pass
  1015→
  1016→        # ── j/k vim-style navigation: findings feed + chat log ──
  1017→        if event.key in ("j", "k"):
  1018→            # Findings feed navigation
  1019→            try:
  1020→                feed = self.query_one("#findings-feed", RichLog)
  1021→                if feed.has_focus and self._findings_list:
  1022→                    self._vim_navigate_findings(event.key)
  1023→                    event.stop()
  1024→                    return
  1025→            except NoMatches:
  1026→                pass
  1027→            # Chat log scroll (j=down, k=up)
  1028→            try:
  1029→                chat = self.query_one("#chat-log", RichLog)
  1030→                if chat.has_focus:
  1031→                    if event.key == "j":
  1032→                        chat.scroll_relative(3)
  1033→                    else:
  1034→                        chat.scroll_relative(-3)
  1035→                    event.stop()
  1036→                    return
  1037→            except NoMatches:
  1038→                pass
  1039→
  1040→        # ── Arrow keys on module grid: navigate cells ──
  1041→        if event.key in ("up", "down", "left", "right"):
  1042→            try:
  1043→                grid = self.query_one("#module-grid", Container)
  1044→                if grid.has_focus:
  1045→                    self._arrow_navigate_modules(event.key)
  1046→                    event.stop()
  1047→                    return
  1048→            except NoMatches:
  1049→                pass
  1050→
  1051→        # ── Enter on findings: open detail modal ──
  1052→        if event.key == "enter":
  1053→            try:
  1054→                feed = self.query_one("#findings-feed", RichLog)
  1055→                if feed.has_focus and self._findings_list:
  1056→                    idx = self._findings_cursor if 0 <= self._findings_cursor < len(self._findings_list) else len(self._findings_list) - 1
  1057→                    self.push_screen(FindingDetailModal(self._findings_list[idx]))
  1058→                    event.stop()
  1059→                    return
  1060→            except NoMatches:
  1061→                pass
  1062→
  1063→        # Command history navigation
  1064→        if event.key == "up" and self._history_index > 0:
  1065→            try:
  1066→                inp = self.query_one("#command-input", Input)
  1067→                if inp.has_focus:
  1068→                    self._history_index -= 1
  1069→                    inp.value = self._command_history[self._history_index]
  1070→            except NoMatches:
  1071→                pass
  1072→        elif event.key == "down":
  1073→            try:
  1074→                inp = self.query_one("#command-input", Input)
  1075→                if inp.has_focus:
  1076→                    if self._history_index < len(self._command_history) - 1:
  1077→                        self._history_index += 1
  1078→                        inp.value = self._command_history[self._history_index]
  1079→                    else:
  1080→                        self._history_index = len(self._command_history)
  1081→                        inp.value = ""
  1082→            except NoMatches:
  1083→                pass

  1144→            try:

  1150→    def _apply_completion(self, inp: Input, comp: CommandCompleter, accepted: str) -> None:
  1151→        """Apply a completion suggestion to the input field.
  1152→
  1153→        Phase C: handles both command-level and arg-level completions.
  1154→        For arg mode, appends/replaces the last word instead of the first.
  1155→        """
  1156→        parts = inp.value.split()
  1157→
  1158→        if comp._mode == "command":
  1159→            # Replace the first word with the accepted command
  1160→            rest = parts[1:] if len(parts) > 1 else []
  1161→            inp.value = accepted + (" " + " ".join(rest) if rest else "")
  1162→        else:
  1163→            # Arg-level: replace the last word
  1164→            if len(parts) >= 2:
