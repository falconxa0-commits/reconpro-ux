  1028→def parse_cf_file(path: str) -> List[Dict[str, Any]]:
  1029→    """Parse a CloudFormation YAML/JSON template and return a list of issue dicts."""
  1030→    issues: List[Dict[str, Any]] = []
  1031→    fname = os.path.basename(path)
  1032→
  1033→    try:
  1034→        text = Path(path).read_text(encoding="utf-8", errors="replace")
  1035→    except OSError:
  1036→        return issues
  1037→
  1038→    template = _simple_yaml_load(text)
  1039→    if not isinstance(template, dict):
  1040→        return issues
  1041→
  1042→    resources = template.get("Resources", {})
  1043→    if not isinstance(resources, dict):
  1044→        return issues
  1045→
  1046→    for res_name, resource in resources.items():
  1047→        if not isinstance(resource, dict):
  1048→            continue
  1049→        resource["_cf_name"] = res_name
  1050→        res_type = str(resource.get("Type", ""))
  1051→
  1052→        if "SecurityGroup" in res_type:
  1053→            issues.extend(_cf_check_security_group(resource, fname))
  1054→        if "BucketPolicy" in res_type:
  1055→            issues.extend(_cf_check_bucket_policy(resource, fname))
  1056→        if "DBInstance" in res_type or "RDS" in res_type:
  1057→            issues.extend(_cf_check_rds(resource, fname))
  1058→        if res_type == "AWS::S3::Bucket":
  1059→            issues.extend(_cf_check_s3_bucket(resource, fname))
  1060→        if "IAM" in res_type and ("Role" in res_type or "Policy" in res_type or "User" in res_type):
  1061→            issues.extend(_cf_check_iam(resource, fname))
  1062→
  1063→        # Generic: Lambda without VPC
  1064→        if "Lambda" in res_type and "Function" in res_type:
  1065→            props = resource.get("Properties", {})
  1066→            if isinstance(props, dict) and not props.get("VpcConfig"):
  1067→                issues.append({
  1068→                    "title": f"CloudFormation: Lambda '{res_name}' Not in VPC",
  1069→                    "severity": "low", "category": "network_exposure",
  1070→                    "description": f"Lambda function '{res_name}' has no VpcConfig, running with direct internet access.",
  1071→                    "evidence": f"{fname}: {res_name} missing VpcConfig",
  1072→                    "remediation": "Add VpcConfig with SubnetIds and SecurityGroupIds.",
  1073→                    "dread_score": _dread("low"), "points_deducted": _pts("low"),
  1074→                })
  1075→
  1076→        # EC2 Instance without IAM instance profile
  1077→        if res_type == "AWS::EC2::Instance":
  1078→            props = resource.get("Properties", {})
  1079→            if isinstance(props, dict) and not props.get("IamInstanceProfile"):
  1080→                issues.append({
  1081→                    "title": f"CloudFormation: EC2 '{res_name}' Without IAM Profile",
  1082→                    "severity": "medium", "category": "iam_misconfiguration",
  1083→                    "description": f"EC2 instance '{res_name}' has no IamInstanceProfile for fine-grained AWS access.",
  1084→                    "evidence": f"{fname}: {res_name} missing IamInstanceProfile",
  1085→                    "remediation": "Add an IamInstanceProfile property referencing an IAM instance profile.",
  1086→                    "dread_score": _dread("medium"), "points_deducted": _pts("medium"),
  1087→                })
  1088→
  1089→        # SecurityGroup missing egress
  1090→        if "SecurityGroup" in res_type and "Ingress" not in res_type:
  1091→            props = resource.get("Properties", {})
  1092→            if isinstance(props, dict):
  1093→                has_egress = props.get("SecurityGroupEgress", [])
  1094→                if not has_egress:
  1095→                    issues.append({
  1096→                        "title": f"CloudFormation: SecurityGroup '{res_name}' Missing Egress Rules",
  1097→                        "severity": "medium", "category": "network_exposure",
  1098→                        "description": f"SecurityGroup '{res_name}' has no egress rules, defaulting to allow all outbound.",
  1099→                        "evidence": f"{fname}: {res_name} missing SecurityGroupEgress",
  1100→                        "remediation": "Add explicit SecurityGroupEgress rules to restrict outbound traffic.",
  1101→                        "dread_score": _dread("medium"), "points_deducted": _pts("medium"),
  1102→                    })
  1103→
  1104→    return issues

  1146→def parse_dockerfile(path: str) -> List[Dict[str, Any]]:
  1147→    """Parse a Dockerfile and return a list of issue dicts."""
  1148→    issues: List[Dict[str, Any]] = []
  1149→    fname = os.path.basename(path)
  1150→
  1151→    try:
  1152→        text = Path(path).read_text(encoding="utf-8", errors="replace")
  1153→    except OSError:
  1154→        return issues
  1155→
  1156→    lines = text.splitlines()
  1157→    has_user = False
  1158→    has_healthcheck = False
  1159→    has_from = False
  1160→    from_line = 0
  1161→    user_line_num = 0
  1162→
  1163→    for idx, line in enumerate(lines):
  1164→        stripped = line.strip()
  1165→        if not stripped or stripped.startswith("#"):
  1166→            continue
  1167→
  1168→        ln = idx + 1
  1169→
  1170→        # ── FROM checks ───────────────────────────────────────────────
  1171→        if stripped.upper().startswith("FROM "):
  1172→            has_from = True
  1173→            from_line = ln
  1174→            from_val = stripped[5:].strip()
  1175→
  1176→            # Check: FROM scratch
  1177→            if from_val.lower() == "scratch":
  1178→                issues.append({
  1179→                    "title": "Dockerfile Uses FROM scratch",
  1180→                    "severity": "medium", "category": "container_image",
  1181→                    "description": "FROM scratch produces a minimal image but provides no package management or shell for debugging.",
  1182→                    "evidence": _ev(fname, ln),
  1183→                    "remediation": "Use a minimal base like alpine or distroless if possible.",
  1184→                    "dread_score": _dread("medium"), "points_deducted": _pts("medium"),
  1185→                })
  1186→            # Check: FROM without tag (mutable image)
  1187→            elif ":" not in from_val.split("@")[0]:
  1188→                issues.append({
  1189→                    "title": "Dockerfile FROM Without Specific Tag",
  1190→                    "severity": "medium", "category": "container_image",
  1191→                    "description": f"Base image '{from_val}' has no specific tag, using 'latest' which is mutable.",
  1192→                    "evidence": _ev(fname, ln),
  1193→                    "remediation": f"Pin the image: {from_val}:<specific-tag> or use a digest @{from_val}@sha256:...",
  1194→                    "dread_score": _dread("medium"), "points_deducted": _pts("medium"),
  1195→                })
  1196→            # Check: FROM using :latest tag
  1197→            elif from_val.endswith(":latest"):
  1198→                issues.append({
  1199→                    "title": "Dockerfile FROM Uses :latest Tag",
  1200→                    "severity": "medium", "category": "container_image",
  1201→                    "description": f"Base image uses ':latest' tag which is mutable and non-reproducible.",
  1202→                    "evidence": _ev(fname, ln),
  1203→                    "remediation": f"Pin to a specific version tag instead of :latest.",
  1204→                    "dread_score": _dread("medium"), "points_deducted": _pts("medium"),
  1205→                })
  1206→
  1207→        # ── USER checks ───────────────────────────────────────────────
  1208→        if stripped.upper().startswith("USER "):
  1209→            user_val = stripped[5:].strip()
  1210→            has_user = True
  1211→            user_line_num = ln
  1212→
  1213→            if user_val in ("root", "0"):
  1214→                issues.append({
  1215→                    "title": "Dockerfile Explicitly Runs as Root",
  1216→                    "severity": "high", "category": "container_privilege",
  1217→                    "description": "USER root/0 is explicitly set, running the container with full root privileges.",
  1218→                    "evidence": _ev(fname, ln),
  1219→                    "remediation": "Create a non-root user and set USER <username>.",
  1220→                    "dread_score": _dread("high"), "points_deducted": _pts("high"),
  1221→                })
  1222→
  1223→        # ── COPY / ADD checks ─────────────────────────────────────────
  1224→        if stripped.upper().startswith(("COPY ", "ADD ")):
  1225→            # Check for sensitive files
  1226→            for pat, desc in _SENSITIVE_FILE_PATTERNS:
  1227→                if pat.search(stripped):
  1228→                    issues.append({
  1229→                        "title": f"Dockerfile Copies Sensitive File ({desc})",
  1230→                        "severity": "critical", "category": "secrets_leak",
  1231→                        "description": f"COPY or ADD instruction includes {desc} in the image.",
  1232→                        "evidence": _ev(fname, ln),
  1233→                        "remediation": f"Do not copy {desc} into the image. Use runtime secrets or multi-stage builds.",
  1234→                        "dread_score": _dread("critical"), "points_deducted": _pts("critical"),
  1235→                    })
  1236→                    break
  1237→
  1238→            # Check: ADD with remote URL (supply chain risk)
  1239→            if re.search(r'ADD\s+https?://', stripped, re.I):
  1240→                issues.append({
  1241→                    "title": "Dockerfile ADD from Remote URL",
  1242→                    "severity": "high", "category": "supply_chain",
  1243→                    "description": "ADD instruction fetches from a remote URL, creating a supply chain risk.",
  1244→                    "evidence": _ev(fname, ln),
  1245→                    "remediation": "Use RUN curl/wget with checksum verification instead of ADD for remote URLs.",
  1246→                    "dread_score": _dread("high"), "points_deducted": _pts("high"),
  1247→                })
  1248→
  1249→        # ── ENV checks ────────────────────────────────────────────────
  1250→        if stripped.upper().startswith("ENV "):
  1251→            env_val = stripped[4:].strip()
  1252→            if _SECRET_ENV_PATTERNS.search(env_val):
  1253→                issues.append({
  1254→                    "title": "Dockerfile ENV Contains Secret",
  1255→                    "severity": "high", "category": "hardcoded_secret",
  1256→                    "description": "An ENV variable appears to contain a hardcoded secret/password/token.",
  1257→                    "evidence": _ev(fname, ln),
  1258→                    "remediation": "Use Docker secrets, Kubernetes secrets, or runtime environment injection.",

  1402→def _k8s_get_pods(doc: Dict[str, Any]) -> List[Dict[str, Any]]:
  1403→    """Extract pod specs from a K8s manifest (handles Deployments, StatefulSets, etc.)."""
  1404→    specs: List[Dict[str, Any]] = []
  1405→    spec = doc.get("spec", {})
  1406→    if not isinstance(spec, dict):
  1407→        return specs
  1408→
  1409→    # Direct Pod spec
  1410→    if spec.get("containers") or spec.get("initContainers"):
  1411→        specs.append(spec)
  1412→    # Template spec (Deployment, StatefulSet, DaemonSet, Job, CronJob)
  1413→    template = spec.get("template", {})
  1414→    if isinstance(template, dict):
  1415→        pod_spec = template.get("spec", {})
  1416→        if isinstance(pod_spec, dict) and (pod_spec.get("containers") or pod_spec.get("initContainers")):
  1417→            specs.append(pod_spec)
  1418→        # CronJob nested template
  1419→        job_spec = template.get("spec", {})
  1420→        if isinstance(job_spec, dict):
  1421→            inner_template = job_spec.get("template", {})
  1422→            if isinstance(inner_template, dict):
  1423→                inner_spec = inner_template.get("spec", {})
  1424→                if isinstance(inner_spec, dict) and (inner_spec.get("containers") or inner_spec.get("initContainers")):
  1425→                    specs.append(inner_spec)
  1426→
  1427→    return specs

  1668→            "dread_score": _dread("medium"), "points_deducted": _pts("medium"),
  1669→        })
  1670→
  1671→    return issues

  1674→def _k8s_check_volumes(pod_spec: Dict[str, Any], doc_name: str, fname: str) -> List[Dict[str, Any]]:
  1675→    """Check volume mounts for dangerous host paths."""
  1676→    issues: List[Dict[str, Any]] = []
  1677→
  1678→    for container in _k8s_get_containers_from_spec(pod_spec):
  1679→        cname = container.get("name", "<unnamed>")
  1680→        mounts = container.get("volumeMounts", [])
  1681→        if not isinstance(mounts, list):
  1682→            continue
  1683→
  1684→        for mount in mounts:
  1685→            if not isinstance(mount, dict):
  1686→                continue
  1687→            mpath = mount.get("mountPath", "")
  1688→
  1689→            # Docker socket
  1690→            if "/var/run/docker.sock" in mpath:
  1691→                issues.append({
  1692→                    "title": f"K8s: '{doc_name}/{cname}' Mounts Docker Socket",
  1693→                    "severity": "critical", "category": "container_privilege",
  1694→                    "description": f"Container '{cname}' mounts /var/run/docker.sock — full host container escape.",
  1695→                    "evidence": f"{fname}: {doc_name}/{cname} mountPath: {mpath}",
  1696→                    "remediation": "Never mount docker.sock into workloads. Use Kubernetes APIs directly.",
  1697→                    "dread_score": _dread("critical"), "points_deducted": _pts("critical"),
  1698→                })
  1699→
  1700→            # Root filesystem mounts
  1701→            dangerous_roots = ["/", "/etc", "/root", "/var", "/usr", "/bin", "/sbin", "/lib"]
  1702→            if mpath in dangerous_roots:
  1703→                issues.append({
  1704→                    "title": f"K8s: '{doc_name}/{cname}' Mounts Host Path {mpath}",
  1705→                    "severity": "high", "category": "container_privilege",
  1706→                    "description": f"Container '{cname}' mounts host path '{mpath}' — potential host file access/tampering.",
  1707→                    "evidence": f"{fname}: {doc_name}/{cname} mountPath: {mpath}",
  1708→                    "remediation": f"Avoid mounting host path '{mpath}'. Use ConfigMaps, Secrets, or emptyDir instead.",
  1709→                    "dread_score": _dread("high"), "points_deducted": _pts("high"),
  1710→                })
  1711→
  1712→    return issues

  1733→def _k8s_check_tolerations(pod_spec: Dict[str, Any], doc_name: str, fname: str) -> List[Dict[str, Any]]:
  1734→    """Check for overly permissive tolerations."""
  1735→    issues: List[Dict[str, Any]] = []
  1736→
  1737→    tolerations = pod_spec.get("tolerations", [])
  1738→    if not isinstance(tolerations, list):
  1739→        return issues
  1740→
  1741→    for tol in tolerations:
  1742→        if not isinstance(tol, dict):
  1743→            continue
  1744→        operator = tol.get("operator", "")
  1745→        effect = tol.get("effect", "")
  1746→        key = tol.get("key", "")
  1747→        if operator == "Exists" and not key:
  1748→            sev = "medium" if effect else "high"
  1749→            effect_desc = f" with effect '{effect}'" if effect else ""
  1750→            issues.append({
  1751→                "title": f"K8s: '{doc_name}' Toleration With operator=Exists (No Key)",
  1752→                "severity": sev, "category": "scheduling",
  1753→                "description": f"Pod '{doc_name}' tolerates all taints{effect_desc} — may schedule on dedicated/master nodes.",
  1754→                "evidence": f"{fname}: {doc_name} toleration operator: Exists, key: (empty)",
  1755→                "remediation": "Specify a specific taint key or restrict tolerations to known taints.",
  1756→                "dread_score": _dread(sev), "points_deducted": _pts(sev),
  1757→            })
  1758→
  1759→    return issues

  1762→def parse_k8s_yaml(path: str) -> List[Dict[str, Any]]:
  1763→    """Parse a Kubernetes YAML manifest and return a list of issue dicts."""
  1764→    issues: List[Dict[str, Any]] = []
  1765→    fname = os.path.basename(path)
  1766→
  1767→    try:
  1768→        text = Path(path).read_text(encoding="utf-8", errors="replace")
  1769→    except OSError:
  1770→        return issues
  1771→
  1772→    docs = _yaml_load_all(text)
  1773→
  1774→    for doc in docs:
  1775→        if not isinstance(doc, dict):
  1776→            continue
  1777→
  1778→        kind = str(doc.get("kind", ""))
  1779→        metadata = doc.get("metadata", {})
  1780→        doc_name = metadata.get("name", "<unnamed>") if isinstance(metadata, dict) else "<unnamed>"
  1781→
  1782→        pod_specs = _k8s_get_pods(doc)
  1783→        if not pod_specs:
  1784→            continue
  1785→
  1786→        for pod_spec in pod_specs:
  1787→            # Security context checks
  1788→            issues.extend(_k8s_check_security_context(pod_spec, doc_name, fname))
  1789→            # Host namespace checks
  1790→            issues.extend(_k8s_check_host_settings(pod_spec, doc_name, fname))
  1791→            # Volume mount checks
  1792→            issues.extend(_k8s_check_volumes(pod_spec, doc_name, fname))
  1793→            # Service account checks
  1794→            issues.extend(_k8s_check_service_account(pod_spec, doc_name, fname))
  1795→            # Toleration checks
  1796→            issues.extend(_k8s_check_tolerations(pod_spec, doc_name, fname))
  1797→
  1798→    return issues

  1806→# File extensions and names we scan
  1807→_TF_EXTS = {".tf", ".tfstate", ".tfvars"}
  1808→_CF_EXTS = {".yaml", ".yml", ".json"}
  1809→_DOCKERFILE_NAMES = {"Dockerfile", "dockerfile", "DockerFile"}
  1810→_DOCKER_COMPOSE_NAMES = {"docker-compose.yml", "docker-compose.yaml", "docker-compose"}

  1813→def _is_likely_cloudformation(path: Path, text: str) -> bool:
  1814→    """Heuristic to determine if a YAML/JSON file is a CloudFormation template."""
  1815→    if re.search(r'AWSTemplateFormatVersion|AWS::', text):
  1816→        return True
  1817→    if re.search(r'Resources:\s*$', text, re.M):
  1818→        # Could be K8s or CF — check for CF-specific keys
  1819→        if re.search(r'Type:\s*AWS::|Description:\s*', text):
  1820→            return True
  1821→    return False
  1822→
  1823→
  1824→def _is_likely_k8s(path: Path, text: str) -> bool:
  1825→    """Heuristic to determine if a YAML file is a Kubernetes manifest."""
  1826→    return bool(re.search(r'apiVersion:\s*(?:v1|apps/|extensions/|batch/|networking\.k8s\.io|rbac\.authorization)', text))
  1827→
  1828→
  1829→def _is_likely_docker_compose(path: Path, text: str) -> bool:
  1830→    """Heuristic to determine if a YAML file is a docker-compose file."""
  1831→    return bool(re.search(r'version:\s*["\']?3|services:\s*$', text, re.M))

  1834→def run(target: str, base_url: str, **kwargs) -> Tuple[List[Finding], int, str, str]:
  1835→    """Scan a directory for IaC security issues.
  1836→
  1837→    Recursively discovers Terraform, CloudFormation, Dockerfile, and
  1838→    Kubernetes manifest files, runs the appropriate parser, and
  1839→    converts all issues to Finding objects.
  1840→
  1841→    Returns:
  1842→        (findings, max_points, module_name, category)
  1843→    """
  1844→    findings: List[Finding] = []
  1845→    target_path = Path(target)
  1846→
  1847→    max_points = 100
  1848→    module_name = "iac_audit"
  1849→    category = "infrastructure_as_code"
  1850→
  1851→    if not target_path.is_dir():
  1852→        # Single file mode
  1853→        _scan_file(target_path, findings)
  1854→    else:
  1855→        # Recursive directory scan
  1856→        for p in sorted(target_path.rglob("*")):
  1857→            if not p.is_file():
  1858→                continue
  1859→            # Skip hidden dirs and common non-IaC dirs
  1860→            parts = p.parts
  1861→            if any(part.startswith(".") for part in parts if part != "."):
  1862→                continue
  1863→            if any(part in ("node_modules", "vendor", "__pycache__", ".git", "venv") for part in parts):
  1864→                continue
  1865→            _scan_file(p, findings)
  1866→
  1867→    # Deduplicate by title + evidence
  1868→    seen: set = set()
  1869→    deduped: List[Finding] = []
  1870→    for f in findings:
  1871→        key = (f.title, f.evidence)
  1872→        if key not in seen:
  1873→            seen.add(key)
  1874→            deduped.append(f)
  1875→
  1876→    return (deduped, max_points, module_name, category)

  1879→def _scan_file(p: Path, findings: List[Finding]) -> None:
  1880→    """Route a file to the appropriate parser."""
  1881→    name = p.name
  1882→    ext = p.suffix.lower()
  1883→
  1884→    try:
  1885→        text = p.read_text(encoding="utf-8", errors="replace")
  1886→    except OSError:
  1887→        return
  1888→
  1889→    # ── Terraform ─────────────────────────────────────────────────────
  1890→    if ext in _TF_EXTS:
  1891→        for issue in parse_tf_file(str(p)):
  1892→            findings.append(Finding(
  1893→                title=issue["title"],
  1894→                severity=issue["severity"],
  1895→                category=issue["category"],
  1896→                module="iac_audit",
  1897→                description=issue["description"],
  1898→                evidence=issue["evidence"],
  1899→                asset=str(p),
  1900→                points_deducted=issue["points_deducted"],
  1901→                remediation=issue["remediation"],
  1902→                dread_score=issue["dread_score"],
  1903→            ))
  1904→        return
  1905→
  1906→    # ── Dockerfile ─────────────────────────────────────────────────────
  1907→    if name in _DOCKERFILE_NAMES:
  1908→        for issue in parse_dockerfile(str(p)):
  1909→            findings.append(Finding(
  1910→                title=issue["title"],
  1911→                severity=issue["severity"],
  1912→                category=issue["category"],
  1913→                module="iac_audit",
  1914→                description=issue["description"],
  1915→                evidence=issue["evidence"],
  1916→                asset=str(p),
  1917→                points_deducted=issue["points_deducted"],
  1918→                remediation=issue["remediation"],
  1919→                dread_score=issue["dread_score"],
  1920→            ))
  1921→        return
  1922→
  1923→    # ── Docker Compose ────────────────────────────────────────────────
  1924→    if name in _DOCKER_COMPOSE_NAMES:
  1925→        # Analyze docker-compose files for security issues
  1926→        for issue in _parse_docker_compose(str(p), text):
  1927→            findings.append(Finding(
  1928→                title=issue["title"],
  1929→                severity=issue["severity"],
  1930→                category=issue["category"],
  1931→                module="iac_audit",
  1932→                description=issue["description"],
  1933→                evidence=issue["evidence"],
  1934→                asset=str(p),
  1935→                points_deducted=issue["points_deducted"],
  1936→                remediation=issue["remediation"],
  1937→                dread_score=issue["dread_score"],
  1938→            ))
  1939→        return
  1940→
  1941→    # ── YAML / JSON: CloudFormation or Kubernetes ──────────────────────
  1942→    if ext in _CF_EXTS:
  1943→        if _is_likely_cloudformation(p, text):
  1944→            for issue in parse_cf_file(str(p)):
  1945→                findings.append(Finding(
  1946→                    title=issue["title"],
  1947→                    severity=issue["severity"],
  1948→                    category=issue["category"],
  1949→                    module="iac_audit",
  1950→                    description=issue["description"],
  1951→                    evidence=issue["evidence"],
  1952→                    asset=str(p),
  1953→                    points_deducted=issue["points_deducted"],
  1954→                    remediation=issue["remediation"],
  1955→                    dread_score=issue["dread_score"],
  1956→                ))
  1957→        elif _is_likely_k8s(p, text):
  1958→            for issue in parse_k8s_yaml(str(p)):
  1959→                findings.append(Finding(
  1960→                    title=issue["title"],
  1961→                    severity=issue["severity"],
  1962→                    category=issue["category"],
  1963→                    module="iac_audit",
  1964→                    description=issue["description"],
  1965→                    evidence=issue["evidence"],
  1966→                    asset=str(p),
  1967→                    points_deducted=issue["points_deducted"],
  1968→                    remediation=issue["remediation"],
  1969→                    dread_score=issue["dread_score"],
  1970→                ))
  1971→        # If neither CF nor K8s, skip silently (not an IaC file)
  1972→
  1973→

  1977→def _parse_docker_compose(path: str, text: str) -> List[Dict[str, Any]]:
  1978→    """Analyze docker-compose files for security issues."""
  1979→    issues: List[Dict[str, Any]] = []
  1980→    fname = os.path.basename(path)
  1981→    doc = _simple_yaml_load(text)
  1982→    if not isinstance(doc, dict):
  1983→        return issues
  1984→
  1985→    services = doc.get("services", {})
  1986→    if not isinstance(services, dict):
  1987→        return issues
  1988→
  1989→    for svc_name, svc_config in services.items():
  1990→        if not isinstance(svc_config, dict):
  1991→            continue
  1992→
  1993→        # Check: privileged
  1994→        if svc_config.get("privileged") is True:
  1995→            issues.append({
  1996→                "title": f"Docker Compose: '{svc_name}' Runs Privileged",
  1997→                "severity": "critical", "category": "container_privilege",
  1998→                "description": f"Service '{svc_name}' runs with --privileged flag.",
  1999→                "evidence": f"{fname}: services.{svc_name}.privileged: true",
  2000→                "remediation": "Remove privileged: true from the service.",
  2001→                "dread_score": _dread("critical"), "points_deducted": _pts("critical"),
  2002→            })
  2003→
  2004→        # Check: security_opt with no-new-privileges missing
  2005→        sec_opts = svc_config.get("security_opt", [])
  2006→        if isinstance(sec_opts, list) and "no-new-privileges:true" not in sec_opts and "no-new-privileges" not in sec_opts:
  2007→            issues.append({
  2008→                "title": f"Docker Compose: '{svc_name}' Missing no-new-privileges",
  2009→                "severity": "medium", "category": "container_privilege",
  2010→                "description": f"Service '{svc_name}' does not set security_opt: no-new-privileges:true.",
  2011→                "evidence": f"{fname}: services.{svc_name} missing security_opt",
  2012→                "remediation": "Add 'security_opt: [no-new-privileges:true]' to the service.",
  2013→                "dread_score": _dread("medium"), "points_deducted": _pts("medium"),
  2014→            })
  2015→
  2016→        # Check: sensitive ports exposed to host
  2017→        ports = svc_config.get("ports", [])
  2018→        if isinstance(ports, list):
  2019→            for port_mapping in ports:
  2020→                port_str = str(port_mapping)
  2021→                for port_num, svc_desc in _DANGEROUS_PORTS.items():
  2022→                    if port_num in port_str and (":" in port_str or port_str == port_num):
  2023→                        issues.append({
  2024→                            "title": f"Docker Compose: '{svc_name}' Exposes {svc_desc} (Port {port_num})",
  2025→                            "severity": "medium", "category": "network_exposure",
  2026→                            "description": f"Service '{svc_name}' exposes port {port_num} ({svc_desc}) to the host.",
  2027→                            "evidence": f"{fname}: services.{svc_name}.ports: {port_str}",
  2028→                            "remediation": f"Do not bind port {port_num} to the host. Use Docker internal networking.",
  2029→                            "dread_score": _dread("medium"), "points_deducted": _pts("medium"),
  2030→                        })
  2031→
  2032→        # Check: environment secrets
  2033→        env_vars = svc_config.get("environment", {})
  2034→        if isinstance(env_vars, dict):
  2035→            for k, v in env_vars.items():
  2036→                if v and re.search(r'(?:PASSWORD|SECRET|TOKEN|KEY|CREDENTIAL)', k, re.I) and not str(v).startswith("$"):
  2037→                    issues.append({
  2038→                        "title": f"Docker Compose: '{svc_name}' Hardcoded Secret in Environment",
  2039→                        "severity": "high", "category": "hardcoded_secret",
  2040→                        "description": f"Service '{svc_name}' has a hardcoded secret in env variable '{k}'.",
  2041→                        "evidence": f"{fname}: services.{svc_name}.environment.{k}",
  2042→                        "remediation": "Use env_file or Docker secrets instead of hardcoded values.",
  2043→                        "dread_score": _dread("high"), "points_deducted": _pts("high"),
  2044→                    })
  2045→                    break  # One per service
  2046→
  2047→        # Check: no resource limits
  2048→        deploy = svc_config.get("deploy", {})
  2049→        if isinstance(deploy, dict):
  2050→            resources = deploy.get("resources", {})
  2051→            if isinstance(resources, dict):
  2052→                limits = resources.get("limits", {})
  2053→                if not limits:
  2054→                    issues.append({
  2055→                        "title": f"Docker Compose: '{svc_name}' No Resource Limits",
  2056→                        "severity": "medium", "category": "resource_limit",
  2057→                        "description": f"Service '{svc_name}' has no deploy.resources.limits.",
  2058→                        "evidence": f"{fname}: services.{svc_name}.deploy.resources.limits missing",
  2059→                        "remediation": "Add deploy.resources.limits with cpus and memory.",
  2060→                        "dread_score": _dread("medium"), "points_deducted": _pts("medium"),
  2061→                    })
  2062→        else:
  2063→            # No deploy block at all
  2064→            issues.append({
  2065→                "title": f"Docker Compose: '{svc_name}' No Deploy Resource Limits",
  2066→                "severity": "low", "category": "resource_limit",
  2067→                "description": f"Service '{svc_name}' has no deploy block — no resource limits defined.",
  2068→                "evidence": f"{fname}: services.{svc_name} missing deploy.resources.limits",
  2069→                "remediation": "Add deploy.resources.limits with cpus and memory.",
  2070→                "dread_score": _dread("low"), "points_deducted": _pts("low"),
  2071→            })
  2072→
  2073→        # Check: /var/run/docker.sock mount
  2074→        volumes = svc_config.get("volumes", [])
  2075→        if isinstance(volumes, list):
  2076→            for vol in volumes:
  2077→                vol_str = str(vol)
  2078→                if "/var/run/docker.sock" in vol_str:
  2079→                    issues.append({
  2080→                        "title": f"Docker Compose: '{svc_name}' Mounts Docker Socket",
  2081→                        "severity": "critical", "category": "container_privilege",
  2082→                        "description": f"Service '{svc_name}' mounts docker.sock — full host container escape.",
  2083→                        "evidence": f"{fname}: services.{svc_name}.volumes: {vol_str}",
  2084→                        "remediation": "Never mount docker.sock. Use Docker APIs directly.",
  2085→                        "dread_score": _dread("critical"), "points_deducted": _pts("critical"),
  2086→                    })
  2087→
  2088→        # Check: read_only not set
  2089→        if not svc_config.get("read_only"):
  2090→            issues.append({
  2091→                "title": f"Docker Compose: '{svc_name}' Writable Root Filesystem",
  2092→                "severity": "low", "category": "container_hardening",
  2093→                "description": f"Service '{svc_name}' does not set read_only: true.",
  2094→                "evidence": f"{fname}: services.{svc_name} missing read_only: true",
  2095→                "remediation": "Set read_only: true and use tmpfs/volumes for writable directories.",
  2096→                "dread_score": _dread("low"), "points_deducted": _pts("low"),
  2097→            })
  2098→
  2099→        # Check: user: root
  2100→        if str(svc_config.get("user", "")).strip() in ("root", "0"):
  2101→            issues.append({
  2102→                "title": f"Docker Compose: '{svc_name}' Runs as Root",
  2103→                "severity": "high", "category": "container_privilege",
  2104→                "description": f"Service '{svc_name}' explicitly runs as root (user: root/0).",
  2105→                "evidence": f"{fname}: services.{svc_name}.user: root",
  2106→                "remediation": "Set user to a non-root user or add a USER directive in the Dockerfile.",
  2107→                "dread_score": _dread("high"), "points_deducted": _pts("high"),
  2108→            })
  2109→
  2110→        # Check: no healthcheck
  2111→        if not svc_config.get("healthcheck"):
  2112→            issues.append({
  2113→                "title": f"Docker Compose: '{svc_name}' No Healthcheck",
  2114→                "severity": "low", "category": "monitoring",
  2115→                "description": f"Service '{svc_name}' has no healthcheck defined.",
  2116→                "evidence": f"{fname}: services.{svc_name} missing healthcheck",
  2117→                "remediation": "Add a healthcheck with test, interval, and retries.",
  2118→                "dread_score": _dread("low"), "points_deducted": _pts("low"),
  2119→            })
  2120→
  2121→    return issues
