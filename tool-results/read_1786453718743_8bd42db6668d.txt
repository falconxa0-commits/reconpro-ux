    51→class PolicyRule:
    52→    """A single rule within a security policy.
    53→
    54→    Attributes:
    55→        name: Human-readable rule name.
    56→        description: What the rule enforces.
    57→        condition: Callable(context) -> bool. Returns True if rule applies.
    58→        action: Callable(context) -> str. Returns "allow", "deny", or "warn".
    59→        severity: Rule severity level.
    60→        enabled: Whether the rule is active.
    61→    """
    62→
    63→    def __init__(
    64→        self,
    65→        name: str,
    66→        description: str,
    67→        condition: Callable[[Dict[str, Any]], bool],
    68→        action: Callable[[Dict[str, Any]], str],
    69→        severity: str = "medium",
    70→        enabled: bool = True,
    71→    ) -> None:
    72→        self.name = name
    73→        self.description = description
    74→        self.condition = condition
    75→        self.action = action
    76→        self.severity = severity
    77→        self.enabled = enabled
    78→
    79→    def evaluate(self, context: Dict[str, Any]) -> Tuple[str, str]:
    80→        """Evaluate the rule against a context.
    81→
    82→        Returns:
    83→            Tuple of (action, rule_name). Action is "allow", "deny", or "skip"
    84→            (if rule is disabled or condition not met).
    85→        """
    86→        if not self.enabled:
    87→            return ("skip", self.name)
    88→        try:
    89→            if self.condition(context):
    90→                result = self.action(context)
    91→                return (result, self.name)
    92→        except Exception as exc:
    93→            logger.warning("Policy rule '%s' evaluation error: %s", self.name, exc)
    94→        return ("skip", self.name)
    95→
    96→    def to_dict(self) -> Dict[str, Any]:
    97→        return {
    98→            "name": self.name,
    99→            "description": self.description,
   100→            "severity": self.severity,
   101→            "enabled": self.enabled,
   102→        }
   103→
   104→
   105→class SecurityPolicy:
   106→    """A named collection of policy rules.
   107→
   108→    Attributes:
   109→        name: Policy name.
   110→        description: Human-readable description.
   111→        rules: Ordered list of PolicyRule objects.
   112→        created_at: ISO timestamp of creation.
   113→    """
   114→
   115→    def __init__(self, name: str, description: str = "") -> None:
   116→        self.name = name
   117→        self.description = description
   118→        self.rules: List[PolicyRule] = []
   119→        self.created_at = datetime.now(timezone.utc).isoformat()
   120→
   121→    def add_rule(self, rule: PolicyRule) -> None:
   122→        """Append a rule to this policy."""
   123→        self.rules.append(rule)
   124→
   125→    def evaluate(self, context: Dict[str, Any]) -> List[Dict[str, Any]]:
   126→        """Evaluate all rules against the context.
   127→
   128→        Returns:
   129→            List of dicts with keys: rule_name, action, severity.
   130→        """
   131→        results: List[Dict[str, Any]] = []
   132→        for rule in self.rules:
   133→            action, rule_name = rule.evaluate(context)
   134→            if action != "skip":
   135→                results.append({
   136→                    "rule_name": rule_name,
   137→                    "action": action,
   138→                    "severity": rule.severity,
   139→                })
   140→        return results
   141→
   142→    def to_dict(self) -> Dict[str, Any]:
   143→        return {
   144→            "name": self.name,
   145→            "description": self.description,
   146→            "created_at": self.created_at,
   147→            "rule_count": len(self.rules),
   148→            "rules": [r.to_dict() for r in self.rules],
   149→        }

   151→
   152→class SecurityPolicyEngine:
   153→    """Central policy engine for defining, enforcing, and auditing
   154→    security policies across all ReconPro components.
   155→
   156→    Provides:
   157→    - define_policy(name, rules): Create a named policy.
   158→    - enforce_policy(policy_name, context): Enforce a policy, return results.
   159→    - evaluate_compliance(): Check all components against all policies.
   160→    - get_policy_report(): Full compliance report.
   161→    """
   162→
   163→    def __init__(self, audit_logger: Optional[SecurityAuditLogger] = None) -> None:
   164→        self._policies: Dict[str, SecurityPolicy] = {}
   165→        self._audit = audit_logger
   166→        self._compliance_cache: Optional[Dict[str, Any]] = None
   167→        self._compliance_lock = threading.Lock()
   168→        self._setup_builtin_policies()
   169→
   170→    # ── Policy Management ───────────────────────────────────────────────
   171→
   172→    def define_policy(
   173→        self,
   174→        name: str,
   175→        rules: Optional[List[PolicyRule]] = None,
   176→        description: str = "",
   177→    ) -> SecurityPolicy:
   178→        """Define (or replace) a security policy.
   179→
   180→        Args:
   181→            name: Unique policy name.
   182→            rules: List of PolicyRule objects. If None, creates empty policy.
   183→            description: Human-readable description.
   184→
   185→        Returns:
   186→            The created SecurityPolicy object.
   187→        """
   188→        policy = SecurityPolicy(name=name, description=description)
   189→        if rules:
   190→            for rule in rules:
   191→                policy.add_rule(rule)
   192→        self._policies[name] = policy
   193→        self._compliance_cache = None  # Invalidate cache
   194→        if self._audit:
   195→            self._audit.info(
   196→                "POLICY_DEFINED",
   197→                target=name,
   198→                details={
   199→                    "rule_count": len(rules) if rules else 0,
   200→                    "description": description,
   201→                },
   202→            )
   203→        return policy
   204→
   205→    def remove_policy(self, name: str) -> bool:
   206→        """Remove a policy by name."""
   207→        if name in self._policies:
   208→            del self._policies[name]
   209→            self._compliance_cache = None
   210→            return True
   211→        return False
   212→
   213→    def get_policy(self, name: str) -> Optional[SecurityPolicy]:
   214→        """Retrieve a policy by name."""
   215→        return self._policies.get(name)
   216→
   217→    def list_policies(self) -> List[str]:
   218→        """Return names of all defined policies."""
   219→        return list(self._policies.keys())
   220→
   221→    # ── Enforcement ─────────────────────────────────────────────────────
   222→
   223→    def enforce_policy(
   224→        self,
   225→        policy_name: str,
   226→        context: Dict[str, Any],
   227→    ) -> Dict[str, Any]:
   228→        """Enforce a named policy against the given context.
   229→
   230→        Args:
   231→            policy_name: Name of the policy to enforce.
   232→            context: Dict describing the operation to evaluate.
   233→
   234→        Returns:
   235→            Dict with keys:
   236→            - policy: policy name
   237→            - allowed: bool (True only if no rule returned "deny")
   238→            - warnings: list of rule names that returned "warn"
   239→            - denied_by: list of rule names that returned "deny"
   240→            - results: full rule evaluation results
   241→        """
   242→        policy = self._policies.get(policy_name)
   243→        if policy is None:
   244→            return {
   245→                "policy": policy_name,
   246→                "allowed": False,
   247→                "error": f"Policy '{policy_name}' not found",
   248→                "warnings": [],
   249→                "denied_by": [],
   250→                "results": [],
   251→            }
   252→
   253→        results = policy.evaluate(context)
   254→        denied_by = [r["rule_name"] for r in results if r["action"] == "deny"]
   255→        warnings = [r["rule_name"] for r in results if r["action"] == "warn"]
   256→        allowed = len(denied_by) == 0
   257→
   258→        if self._audit:
   259→            level = "ALERT" if not allowed else ("WARN" if warnings else "INFO")
   260→            event_type = "POLICY_DENIED" if not allowed else "POLICY_PASSED"
   261→            self._audit._emit(
   262→                level,
   263→                event_type,
   264→                target=policy_name,
   265→                details={
   266→                    "context_keys": list(context.keys()),
   267→                    "denied_by": denied_by,
   268→                    "warnings": warnings,
   269→                },
   270→            )
   271→
   272→        return {
   273→            "policy": policy_name,
   274→            "allowed": allowed,
   275→            "warnings": warnings,
   276→            "denied_by": denied_by,
   277→            "results": results,
   278→        }
   279→
   280→    def enforce_all(
   281→        self,
   282→        context: Dict[str, Any],
   283→    ) -> Dict[str, Any]:
   284→        """Enforce ALL policies against the given context.
   285→
   286→        Returns:
   287→            Dict with keys:
   288→            - allowed: bool (True only if ALL policies allow)
   289→            - policy_results: dict mapping policy name -> enforce_policy result
   290→        """
   291→        all_results: Dict[str, Dict[str, Any]] = {}
   292→        overall_allowed = True
   293→        for name in self._policies:
   294→            result = self.enforce_policy(name, context)
   295→            all_results[name] = result
   296→            if not result["allowed"]:
   297→                overall_allowed = False
   298→        return {
   299→            "allowed": overall_allowed,
   300→            "policy_results": all_results,
   301→        }
   302→
   303→    # ── Compliance ──────────────────────────────────────────────────────
   304→
   305→    def evaluate_compliance(
   306→        self,
   307→        components: Optional[Dict[str, Dict[str, Any]]] = None,
   308→    ) -> Dict[str, Any]:
   309→        """Check all components against all policies.
   310→
   311→        Args:
   312→            components: Dict mapping component name -> component context.
   313→                       If None, uses a built-in default set of components.
   314→
   315→        Returns:
   316→            Dict with keys:
   317→            - compliant: bool (True if all components pass all policies)
   318→            - total_checks: int
   319→            - passed: int
   320→            - failed: int
   321→            - warnings: int
   322→            - details: list of per-component, per-policy results
   323→        """
   324→        if components is None:
   325→            components = self._default_components()
   326→
   327→        total_checks = 0
   328→        passed = 0
   329→        failed = 0
   330→        warnings = 0
   331→        details: List[Dict[str, Any]] = []
   332→
   333→        for comp_name, comp_ctx in components.items():
   334→            for policy_name in self._policies:
   335→                total_checks += 1
   336→                result = self.enforce_policy(policy_name, comp_ctx)
   337→                entry = {
   338→                    "component": comp_name,
   339→                    "policy": policy_name,
   340→                    "allowed": result["allowed"],
   341→                    "denied_by": result["denied_by"],
   342→                    "warnings": result["warnings"],
   343→                }
   344→                details.append(entry)
   345→                if result["allowed"]:
   346→                    if result["warnings"]:
   347→                        warnings += 1
   348→                    else:
   349→                        passed += 1
   350→                else:
   351→                    failed += 1
   352→
   353→        compliant = failed == 0
   354→        report = {
   355→            "compliant": compliant,
   356→            "total_checks": total_checks,
   357→            "passed": passed,
   358→            "failed": failed,
   359→            "warnings": warnings,
   360→            "details": details,
   361→            "evaluated_at": datetime.now(timezone.utc).isoformat(),
   362→        }
   363→
   364→        with self._compliance_lock:
   365→            self._compliance_cache = report
   366→
   367→        return report
   368→
   369→    def get_policy_report(self) -> Dict[str, Any]:
   370→        """Generate a comprehensive policy compliance report.
   371→
   372→        Returns:
   373→            Dict with keys:
   374→            - policies: list of policy summaries
   375→            - compliance: latest compliance evaluation
   376→            - summary: human-readable summary string
   377→        """
   378→        if self._compliance_cache is None:
   379→            self.evaluate_compliance()
   380→
   381→        policies_summary = []
   382→        for name, policy in self._policies.items():
   383→            policies_summary.append(policy.to_dict())
   384→
   385→        compliance = self._compliance_cache or {}
   386→        total = compliance.get("total_checks", 0)
   387→        failed = compliance.get("failed", 0)
   388→        warn = compliance.get("warnings", 0)
   389→        passed = compliance.get("passed", 0)
   390→
   391→        if total == 0:
   392→            summary = "No compliance checks performed."
   393→        elif failed > 0:
   394→            summary = (
   395→                f"NON-COMPLIANT: {failed}/{total} checks failed, "
   396→                f"{warn} warnings, {passed} passed."
   397→            )
   398→        elif warn > 0:
   399→            summary = (
   400→                f"COMPLIANT WITH WARNINGS: {passed}/{total} passed, "
   401→                f"{warn} warnings."
   402→            )
   403→        else:
   404→            summary = f"FULLY COMPLIANT: all {total} checks passed."
   405→
   406→        return {
   407→            "policies": policies_summary,
   408→            "compliance": compliance,
   409→            "summary": summary,
   410→            "generated_at": datetime.now(timezone.utc).isoformat(),
   411→        }

   968→    def _log_operation(
   969→        self,
   970→        operation: str,

   972→    ) -> None:

   974→        entry = {
   975→            "timestamp": datetime.now(timezone.utc).isoformat(),
   976→            "operation": operation,
   977→            "details": details or {},
   978→        }
   979→        with self._log_lock:
   980→            self._operation_log.append(entry)
   981→        if self._audit:

   988→    def _check_memory_usage(self) -> int:
   989→        """Return approximate current memory usage in bytes.
   990→
   991→        Uses resource module if available (Unix), otherwise falls back
   992→        to a basic tracking mechanism.
   993→        """
   994→        try:
   995→            import resource as _resource  # type: ignore[import-untyped]
   996→            return _resource.getrusage(_resource.RUSAGE_SELF).ru_maxrss * 1024
   997→        except (ImportError, AttributeError):
   998→            # Fallback: track based on sys.getsizeof of known objects
   999→            return 0
  1000→
  1001→    def _enforce_cpu_limit(self) -> None:
  1002→        """Set CPU time limit on platforms that support it (Unix)."""
  1003→        try:
  1004→            import resource as _resource  # type: ignore[import-untyped]
  1005→            _resource.setrlimit(
  1006→                _resource.RLIMIT_CPU,
  1007→                (int(self.cpu_time_seconds), int(self.cpu_time_seconds * 1.5)),
  1008→            )
  1009→        except (ImportError, AttributeError, OSError, ValueError):
  1010→            pass  # Not available on this platform
  1011→
  1012→    def _restore_cpu_limit(self) -> None:
  1013→        """Restore CPU time limit after sandboxed execution."""
  1014→        try:
  1015→            import resource as _resource  # type: ignore[import-untyped]
  1016→            # Set to a very high value (effectively unlimited)
  1017→            _resource.setrlimit(_resource.RLIMIT_CPU, (0x7FFFFFFF, 0x7FFFFFFF))
  1018→        except (ImportError, AttributeError, OSError, ValueError):
  1019→            pass

  1021→    def _validate_output(self, result: Any) -> List[Dict[str, Any]]:
  1022→        """Validate plugin output is safe and within limits.
  1023→
  1024→        Args:
  1025→            result: The return value from the plugin.
  1026→
  1027→        Returns:
  1028→            List of finding dicts, or raises on invalid output.
  1029→        """
  1030→        if not isinstance(result, list):
  1031→            raise SandboxViolation(
  1032→                "output_validation",
  1033→                f"Plugin must return a list, got {type(result).__name__}",
  1034→            )
  1035→
  1036→        output_str = json.dumps(result, default=str)
  1037→        if len(output_str) > self.max_output_size:
  1038→            raise ResourceLimitExceeded(
  1039→                "output_size",
  1040→                self.max_output_size,
  1041→                len(output_str),
  1042→            )
  1043→
  1044→        validated: List[Dict[str, Any]] = []
  1045→        required_keys = {"title", "severity", "category"}
  1046→        for i, item in enumerate(result):
  1047→            if not isinstance(item, dict):
  1048→                raise SandboxViolation(
  1049→                    f"output_item_{i}",
  1050→                    f"Each finding must be a dict, got {type(item).__name__}",
  1051→                )
  1052→            if not required_keys.issubset(item.keys()):
  1053→                missing = required_keys - item.keys()
  1054→                raise SandboxViolation(
  1055→                    f"output_item_{i}",
  1056→                    f"Finding missing required keys: {missing}",
  1057→                )
  1058→            validated.append(item)
  1059→
  1060→        return validated

  1062→    def execute(
  1063→        self,
  1064→        plugin_func: Callable,
  1065→        plugin_name: str = "unknown",
  1066→        **kwargs: Any,
  1067→    ) -> Dict[str, Any]:
  1068→        """Execute a plugin function inside the sandbox.
  1069→
  1070→        The plugin function is called with **kwargs as arguments.
  1071→        Its execution is wrapped with:
  1072→        - Restricted builtins
  1073→        - CPU time limits (where supported)
  1074→        - Memory usage checks
  1075→        - Output validation
  1076→        - Full audit logging
  1077→
  1078→        Args:
  1079→            plugin_func: The plugin's callable (typically its ``run`` function).
  1080→            plugin_name: Human-readable plugin name for logging.
  1081→            **kwargs: Arguments forwarded to the plugin function.
  1082→
  1083→        Returns:
  1084→            Dict with keys:
  1085→            - success: bool
  1086→            - findings: list of validated finding dicts (on success)
  1087→            - error: str (on failure)
  1088→            - execution_time: float (seconds)
  1089→            - memory_used: int (bytes, 0 if unmeasurable)
  1090→            - operations: count of sandbox-tracked operations
  1091→        """
  1092→        self._log_operation("sandbox_enter", {"plugin": plugin_name})
  1093→        start_time = time.monotonic()
  1094→        memory_before = self._check_memory_usage()
  1095→
  1096→        try:
  1097→            # Set CPU time limit
  1098→            self._enforce_cpu_limit()
  1099→            self._log_operation("cpu_limit_set", {
  1100→                "limit_seconds": self.cpu_time_seconds,
  1101→            })
  1102→
  1103→            # Execute the plugin function
  1104→            # Note: Python does not provide true sandboxing. The restricted
  1105→            # builtins are applied when the plugin code uses exec/eval.
  1106→            # For direct function calls, we wrap and monitor.
  1107→            self._log_operation("plugin_execute_start", {
  1108→                "plugin": plugin_name,
  1109→                "args": list(kwargs.keys()),
  1110→            })
  1111→
  1112→            try:
  1113→                result = plugin_func(**kwargs)
  1114→            except TypeError as e:
  1115→                raise SandboxViolation(
  1116→                    "plugin_call",
  1117→                    f"Plugin call failed: {e}",
  1118→                )
  1119→
  1120→            self._log_operation("plugin_execute_end", {
  1121→                "plugin": plugin_name,
  1122→                "result_type": type(result).__name__,
  1123→            })
  1124→
  1125→            # Validate output
  1126→            findings = self._validate_output(result)
  1127→            self._log_operation("output_validated", {
  1128→                "plugin": plugin_name,
  1129→                "finding_count": len(findings),
  1130→            })
  1131→
  1132→            memory_after = self._check_memory_usage()
  1133→            if (memory_after > 0
  1134→                    and memory_before > 0):
  1135→                memory_delta = memory_after - memory_before
  1136→                if memory_delta > self.memory_limit_bytes:
  1137→                    raise ResourceLimitExceeded(
  1138→                        "memory",
  1139→                        self.memory_limit_bytes,
  1140→                        memory_delta,
  1141→                    )
  1142→
  1143→            return {
  1144→                "success": True,
  1145→                "findings": findings,
  1146→                "execution_time": time.monotonic() - start_time,
  1147→                "memory_used": max(0, memory_after - memory_before),
  1148→                "operations": len(self._operation_log),
  1149→            }
  1150→
  1151→        except SandboxViolation as exc:
  1152→            self._log_operation("sandbox_violation", {
  1153→                "plugin": plugin_name,
  1154→                "operation": exc.operation,
  1155→                "reason": exc.reason,
  1156→            })
  1157→            return {
  1158→                "success": False,
  1159→                "error": str(exc),
  1160→                "execution_time": time.monotonic() - start_time,
  1161→                "memory_used": 0,
  1162→                "operations": len(self._operation_log),
  1163→            }
  1164→
  1165→        except ResourceLimitExceeded as exc:
  1166→            self._log_operation("resource_limit_exceeded", {
  1167→                "plugin": plugin_name,
  1168→                "resource": exc.resource,
  1169→                "limit": exc.limit,
  1170→                "actual": exc.actual,
  1171→            })
  1172→            return {
  1173→                "success": False,
  1174→                "error": str(exc),
  1175→                "execution_time": time.monotonic() - start_time,
  1176→                "memory_used": 0,
  1177→                "operations": len(self._operation_log),
  1178→            }
  1179→
  1180→        except Exception as exc:
  1181→            self._log_operation("plugin_error", {
  1182→                "plugin": plugin_name,
  1183→                "error_type": type(exc).__name__,
  1184→                "error": str(exc)[:500],
  1185→            })
  1186→            return {
  1187→                "success": False,
  1188→                "error": f"Plugin execution error: {type(exc).__name__}: {exc}",
  1189→                "execution_time": time.monotonic() - start_time,
  1190→                "memory_used": 0,
  1191→                "operations": len(self._operation_log),
  1192→            }
  1193→
  1194→        finally:
  1195→            # Restore CPU limits
  1196→            self._restore_cpu_limit()
  1197→            self._log_operation("sandbox_exit", {"plugin": plugin_name})
  1198→

  1209→    @staticmethod
  1210→    def check_code_for_violations(code: str) -> List[Dict[str, Any]]:
  1211→        """Statically analyze code for sandbox violations.
  1212→
  1213→        Checks for:
  1214→        - Use of forbidden builtins (eval, exec, compile, open, __import__)
  1215→        - Import of forbidden modules
  1216→        - Use of dangerous attributes (os.system, subprocess.call, etc.)
  1217→
  1218→        Args:
  1219→            code: Python source code to analyze.
  1220→
  1221→        Returns:
  1222→            List of violation dicts with keys:
  1223→            - type: violation type
  1224→            - match: the matching text
  1225→            - line: line number
  1226→            - severity: "critical" or "high"
  1227→        """
  1228→        violations: List[Dict[str, Any]] = []
  1229→        lines = code.split("\n")
  1230→
  1231→        # Check for forbidden builtins
  1232→        forbidden_builtins_pattern = re.compile(
  1233→            r'\b(' + '|'.join(re.escape(b) for b in _FORBIDDEN_BUILTINS) + r')\b'
  1234→        )
  1235→        for line_num, line in enumerate(lines, start=1):
  1236→            # Skip comments
  1237→            stripped = line.strip()
  1238→            if stripped.startswith("#"):
  1239→                continue
  1240→            # Check for string contexts (simple heuristic)
  1241→            # Remove string literals to avoid false positives
  1242→            clean_line = re.sub(r'"[^"]*"', '""', line)
  1243→            clean_line = re.sub(r"'[^']*'", "''", clean_line)
  1244→
  1245→            for match in forbidden_builtins_pattern.finditer(clean_line):
  1246→                builtin = match.group(1)
  1247→                # Avoid flagging __import__ when it's just a string reference
  1248→                if builtin == "__import__" and "__import__" not in clean_line.split("=")[-1]:
  1249→                    violations.append({
  1250→                        "type": "forbidden_builtin",
  1251→                        "match": builtin,
  1252→                        "line": line_num,
  1253→                        "severity": "critical",
  1254→                    })
  1255→                elif builtin != "__import__":
  1256→                    violations.append({
  1257→                        "type": "forbidden_builtin",
  1258→                        "match": builtin,
  1259→                        "line": line_num,
  1260→                        "severity": "critical",
  1261→                    })
  1262→
  1263→        # Check for forbidden imports
  1264→        import_pattern = re.compile(
  1265→            r'\b(?:import|from)\s+([a-zA-Z_][a-zA-Z0-9_]*)'
  1266→        )
  1267→        for line_num, line in enumerate(lines, start=1):
  1268→            stripped = line.strip()
  1269→            if stripped.startswith("#"):
  1270→                continue
  1271→            for match in import_pattern.finditer(line):
  1272→                module = match.group(1)
  1273→                if module in _FORBIDDEN_MODULES:
  1274→                    violations.append({
  1275→                        "type": "forbidden_module",
  1276→                        "match": module,
  1277→                        "line": line_num,
  1278→                        "severity": "high",
  1279→                    })
  1280→
  1281→        # Check for dangerous attribute access patterns
  1282→        dangerous_patterns = [
  1283→            (r'\bos\.system\b', "os.system", "critical"),
  1284→            (r'\bos\.popen\b', "os.popen", "critical"),
  1285→            (r'\bsubprocess\.', "subprocess", "critical"),
  1286→            (r'\bshutil\.', "shutil", "high"),
  1287→            (r'\bctypes\.', "ctypes", "critical"),
  1288→            (r'\b__builtins__\b', "__builtins__", "high"),
  1289→            (r'\bglobals\(\)', "globals()", "high"),
  1290→            (r'\blocals\(\)', "locals()", "medium"),
  1291→            (r'\bgetattr\s*\(\s*__builtins__', "getattr(__builtins__", "critical"),
  1292→        ]
  1293→        for line_num, line in enumerate(lines, start=1):
  1294→            stripped = line.strip()
  1295→            if stripped.startswith("#"):
  1296→                continue
  1297→            for pattern, name, severity in dangerous_patterns:
  1298→                if re.search(pattern, line):
  1299→                    violations.append({
  1300→                        "type": "dangerous_pattern",
  1301→                        "match": name,
  1302→                        "line": line_num,
  1303→                        "severity": severity,
  1304→                    })
  1305→
  1306→        return violations

  1388→@dataclass
  1389→class SecretFinding:
  1390→    """A single secret detection finding.
  1391→
  1392→    Attributes:
  1393→        secret_type: Category of the secret (e.g., "aws_key", "private_key_rsa").
  1394→        match: The matched text (redacted for high-severity findings).
  1395→        line: Line number where found (1-based), or 0 if not line-based.
  1396→        offset: Character offset within the line.
  1397→        severity: "critical", "high", "medium", or "low".
  1398→        description: Human-readable description of what was found.
  1399→        source: Where the content came from (e.g., filename, "string_input").
  1400→    """
  1401→    secret_type: str
  1402→    match: str
  1403→    line: int
  1404→    offset: int
  1405→    severity: str
  1406→    description: str
  1407→    source: str = "unknown"
  1408→
  1409→
  1410→class SecretsManager:
  1411→    """Enhanced secret detection and management.
  1412→
  1413→    Builds on top of security.detect_secrets_in_text() with:
  1414→    - Additional patterns (Slack, Google, Stripe, Twilio, SSH, PGP, etc.)
  1415→    - File scanning capability
  1416→    - Structured severity reporting
  1417→    - Summary statistics
  1418→
  1419→    Usage::
  1420→
  1421→        sm = SecretsManager()
  1422→        findings = sm.scan_content("some text with AKIAIOSFODNN7EXAMPLE key")
  1423→        report = sm.get_report()
  1424→    """
  1425→
  1426→    SEVERITY_ORDER: Dict[str, int] = {
  1427→        "critical": 0,
  1428→        "high": 1,
  1429→        "medium": 2,
  1430→        "low": 3,
  1431→    }
  1432→
  1433→    def __init__(self, audit_logger: Optional[SecurityAuditLogger] = None) -> None:
  1434→        self._findings: List[SecretFinding] = []
  1435→        self._audit = audit_logger
  1436→        self._scan_count = 0
  1437→        self._files_scanned = 0
  1438→        self._lock = threading.Lock()
  1439→
  1440→    def scan_for_secrets(
  1441→        self,
  1442→        content: str,
  1443→        source: str = "string_input",
  1444→    ) -> List[Dict[str, Any]]:
  1445→        """Scan content for secrets using both built-in and extended patterns.
  1446→
  1447→        This method first uses the existing detect_secrets_in_text() from
  1448→        security.py, then applies additional patterns from this module.
  1449→
  1450→        Args:
  1451→            content: Text content to scan.
  1452→            source: Identifier for where the content came from.
  1453→
  1454→        Returns:
  1455→            List of dicts with keys:
  1456→            - secret_type: str
  1457→            - match: str (redacted for critical/high)
  1458→            - line: int
  1459→            - offset: int
  1460→            - severity: str
  1461→            - description: str
  1462→            - source: str
  1463→            - detector: str ("core" or "extended")
  1464→        """
  1465→        if not content:
  1466→            return []
  1467→
  1468→        self._scan_count += 1
  1469→        results: List[Dict[str, Any]] = []
  1470→        lines = content.split("\n")
  1471→
  1472→        # Phase 1: Use existing core detection from security.py
  1473→        core_findings = detect_secrets_in_text(content)
  1474→        for finding in core_findings:
  1475→            redacted = self._redact_match(finding["match"], finding["confidence"])
  1476→            results.append({
  1477→                "secret_type": finding["type"],
  1478→                "match": redacted,
  1479→                "line": finding["line"],
  1480→                "offset": finding["offset"],
  1481→                "severity": finding["confidence"],
  1482→                "description": self._describe_finding(finding["type"]),
  1483→                "source": source,
  1484→                "detector": "core",
  1485→            })
  1486→
  1487→        # Phase 2: Extended patterns
  1488→        for secret_type, pattern, severity, description in _EXTENDED_SECRET_PATTERNS:
  1489→            for line_num, line in enumerate(lines, start=1):
  1490→                for match in pattern.finditer(line):
  1491→                    matched_text = match.group(0)
  1492→                    redacted = self._redact_match(matched_text, severity)
  1493→                    results.append({
  1494→                        "secret_type": secret_type,
  1495→                        "match": redacted,
  1496→                        "line": line_num,
  1497→                        "offset": match.start(),
  1498→                        "severity": severity,
  1499→                        "description": description,
  1500→                        "source": source,
  1501→                        "detector": "extended",
  1502→                    })
  1503→
  1504→        # Store findings
  1505→        with self._lock:
  1506→            for r in results:
  1507→                self._findings.append(SecretFinding(
  1508→                    secret_type=r["secret_type"],
  1509→                    match=r["match"],
  1510→                    line=r["line"],
  1511→                    offset=r["offset"],
  1512→                    severity=r["severity"],
  1513→                    description=r["description"],
  1514→                    source=source,
  1515→                ))
  1516→
  1517→        # Audit log
  1518→        if results and self._audit:
  1519→            severity_counts: Dict[str, int] = {}
  1520→            for r in results:
  1521→                s = r["severity"]
  1522→                severity_counts[s] = severity_counts.get(s, 0) + 1
  1523→            self._audit.warn(
  1524→                "SECRETS_DETECTED",
  1525→                target=source,
  1526→                details={
  1527→                    "total": len(results),
  1528→                    "severity_counts": severity_counts,
  1529→                    "types": list(set(r["secret_type"] for r in results)),
  1530→                },
  1531→            )
  1532→
  1533→        return results
  1534→
  1535→    def scan_file(self, filepath: str) -> List[Dict[str, Any]]:
  1536→        """Scan a file for secrets.
  1537→
  1538→        Args:
  1539→            filepath: Path to the file to scan.
  1540→
  1541→        Returns:
  1542→            Same format as scan_for_secrets().
  1543→        """
  1544→        if not os.path.isfile(filepath):
  1545→            return []
  1546→
  1547→        try:
  1548→            with open(filepath, "r", encoding="utf-8", errors="ignore") as f:
  1549→                content = f.read()
  1550→        except OSError as exc:
  1551→            logger.warning("Cannot read file for secret scanning: %s: %s", filepath, exc)
  1552→            return []
  1553→
  1554→        self._files_scanned += 1
  1555→        return self.scan_for_secrets(content, source=filepath)
  1556→
  1557→    def scan_directory(
  1558→        self,
  1559→        dirpath: str,
  1560→        extensions: Optional[Set[str]] = None,
  1561→        max_files: int = 1000,
  1562→    ) -> List[Dict[str, Any]]:
  1563→        """Recursively scan a directory for secrets.
  1564→
  1565→        Args:
  1566→            dirpath: Root directory to scan.
  1567→            extensions: Set of file extensions to scan (e.g., {".py", ".js"}).
  1568→                      If None, scans all files.
  1569→            max_files: Maximum number of files to scan.
  1570→
  1571→        Returns:
  1572→            Combined list of all findings.
  1573→        """
  1574→        all_results: List[Dict[str, Any]] = []
  1575→        files_scanned = 0
  1576→
  1577→        if not os.path.isdir(dirpath):
  1578→            return []
  1579→
  1580→        for root, _dirs, files in os.walk(dirpath):
  1581→            if files_scanned >= max_files:
  1582→                break
  1583→            for fname in sorted(files):
  1584→                if files_scanned >= max_files:
  1585→                    break
  1586→                # Skip hidden files and common non-text files
  1587→                if fname.startswith("."):
  1588→                    continue
  1589→                if fname.endswith((".pyc", ".pyo", ".so", ".dll", ".exe",
  1590→                                  ".png", ".jpg", ".jpeg", ".gif", ".ico",
  1591→                                  ".zip", ".tar", ".gz", ".bz2", ".xz")):
  1592→                    continue
  1593→
  1594→                filepath = os.path.join(root, fname)
  1595→                if extensions is not None:
  1596→                    _, ext = os.path.splitext(fname)
  1597→                    if ext.lower() not in extensions:
  1598→                        continue
  1599→
  1600→                results = self.scan_file(filepath)
  1601→                all_results.extend(results)
  1602→                files_scanned += 1
  1603→
  1604→        return all_results

  1606→    def get_report(self) -> Dict[str, Any]:
  1607→        """Generate a comprehensive secrets scanning report.
  1608→
  1609→        Returns:
  1610→            Dict with keys:
  1611→            - total_findings: int
  1612→            - by_severity: dict mapping severity -> count
  1613→            - by_type: dict mapping secret_type -> count
  1614→            - by_source: dict mapping source -> count
  1615→            - by_detector: dict mapping detector -> count
  1616→            - critical_findings: list of critical/high severity findings
  1617→            - scan_stats: scanning statistics
  1618→        """
  1619→        with self._lock:
  1620→            findings = list(self._findings)
  1621→
  1622→        by_severity: Dict[str, int] = {}
  1623→        by_type: Dict[str, int] = {}
  1624→        by_source: Dict[str, int] = {}
  1625→        by_detector: Dict[str, int] = {}
  1626→        critical_findings: List[Dict[str, Any]] = []
  1627→
  1628→        for f in findings:
  1629→            by_severity[f.severity] = by_severity.get(f.severity, 0) + 1
  1630→            by_type[f.secret_type] = by_type.get(f.secret_type, 0) + 1
  1631→            by_source[f.source] = by_source.get(f.source, 0) + 1
  1632→            _CORE_TYPES = {
  1633→                "aws_key", "github_token", "github_token_fine_grained",
  1634→                "generic_api_key", "private_key_rsa", "private_key_ec",
  1635→                "private_key_generic", "password_in_url", "db_connection_string",
  1636→                "jwt_token",
  1637→            }
  1638→            detector = "core" if f.secret_type in _CORE_TYPES else "extended"
  1639→            by_detector[detector] = by_detector.get(detector, 0) + 1
  1640→
  1641→            if f.severity in ("critical", "high"):
  1642→                critical_findings.append({
  1643→                    "type": f.secret_type,
  1644→                    "line": f.line,
  1645→                    "severity": f.severity,
  1646→                    "description": f.description,
  1647→                    "source": f.source,
  1648→                })
  1649→
  1650→        return {
  1651→            "total_findings": len(findings),
  1652→            "by_severity": by_severity,
  1653→            "by_type": by_type,
  1654→            "by_source": by_source,
  1655→            "by_detector": by_detector,
  1656→            "critical_findings": critical_findings,
  1657→            "scan_stats": {
  1658→                "content_scans": self._scan_count,
  1659→                "files_scanned": self._files_scanned,
  1660→            },
  1661→            "generated_at": datetime.now(timezone.utc).isoformat(),
  1662→        }

  1664→    def clear_findings(self) -> None:

  1666→        with self._lock:
  1667→            self._findings.clear()
  1668→            self._scan_count = 0
  1669→            self._files_scanned = 0

  1859→    def verify_chain(self) -> Dict[str, Any]:
  1860→        """Verify the integrity of the entire log chain.
  1861→
  1862→        Checks that:
  1863→        1. Sequence numbers are monotonically increasing.
  1864→        2. Each entry's hash matches its content.
  1865→        3. Each entry's previous_hash matches the prior entry's hash.
