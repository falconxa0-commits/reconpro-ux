    63→function TrendArrow({ trend, change, size = 16 }: { trend: TrendDirection; change: number; size?: number }) {
    64→  const color = trend === 'rising' ? '#ff3355' : trend === 'falling' ? '#22c55e' : '#6b7280';

    70→// ── Fear Gauge (SVG Arc) ───────────────────────────────────────────────
    71→
    72→function FearGauge({ score, level }: { score: number; level: FearLevel }) {
    73→  const color = LEVEL_COLORS[level];
    74→  const isCritical = level === 'CRITICAL' || level === 'SEVERE';
    75→  const size = 280;
    76→  const strokeWidth = 24;
    77→  const radius = (size - strokeWidth) / 2;
    78→  const circumference = Math.PI * radius; // 270 degrees = 3/4 of full circle
    79→  const cx = size / 2;
    80→  const cy = size / 2 + 20;
    81→
    82→  // 270-degree arc: from 135° to 405° (in standard math angles)
    83→  const startAngle = 135;
    84→  const endAngle = 405;
    85→  const totalAngle = endAngle - startAngle;
    86→  const scoreAngle = startAngle + (score / 100) * totalAngle;
    87→
    88→  function polarToCartesian(angle: number) {
    89→    const rad = (angle * Math.PI) / 180;
    90→    return {
    91→      x: cx + radius * Math.cos(rad),
    92→      y: cy + radius * Math.sin(rad),
    93→    };
    94→  }
    95→
    96→  function describeArc(start: number, end: number) {
    97→    const s = polarToCartesian(start);
    98→    const e = polarToCartesian(end);
    99→    const largeArc = end - start > 180 ? 1 : 0;
   100→    return `M ${s.x} ${s.y} A ${radius} ${radius} 0 ${largeArc} 1 ${e.x} ${e.y}`;
   101→  }
   102→
   103→  // Background arc path
   104→  const bgPath = describeArc(startAngle, endAngle);
   105→  // Score arc path
   106→  const scorePath = score > 0 ? describeArc(startAngle, scoreAngle) : '';
   107→  // Needle position
   108→  const needlePos = polarToCartesian(scoreAngle);
   109→
   110→  // Tick marks
   111→  const ticks = [0, 20, 40, 60, 80, 100];
   112→  const tickMarks = ticks.map(t => {
   113→    const angle = startAngle + (t / 100) * totalAngle;
   114→    const inner = polarToCartesian(angle);
   115→    const outer = polarToCartesian(angle);
   116→    const ix = cx + (radius - 15) * Math.cos((angle * Math.PI) / 180);
   117→    const iy = cy + (radius - 15) * Math.sin((angle * Math.PI) / 180);
   118→    const ox = cx + (radius + 8) * Math.cos((angle * Math.PI) / 180);
   119→    const oy = cy + (radius + 8) * Math.sin((angle * Math.PI) / 180);
   120→    return { label: t, ix, iy, ox, oy };
   121→  });
   122→
   123→  return (
   124→    <div className="relative flex flex-col items-center">
   125→      {isCritical && (
   126→        <div className="absolute inset-0 rounded-full animate-pulse"
   127→          style={{
   128→            background: `radial-gradient(circle, ${color}33 0%, transparent 70%)`,
   129→            filter: 'blur(20px)',
   130→          }}
   131→        />
   132→      )}
   133→      <svg width={size} height={size} viewBox={`0 0 ${size} ${size}`} className="drop-shadow-lg">
   134→        {/* Background arc */}
   135→        <path d={bgPath} fill="none" stroke="#1f2937" strokeWidth={strokeWidth} strokeLinecap="round" />
   136→        {/* Score arc */}
   137→        <motion.path
   138→          d={scorePath}
   139→          fill="none"
   140→          stroke={color}
   141→          strokeWidth={strokeWidth}
   142→          strokeLinecap="round"
   143→          initial={{ pathLength: 0 }}
   144→          animate={{ pathLength: 1 }}
   145→          transition={{ duration: 1.5, ease: 'easeOut' as const }}
   146→          style={{ filter: `drop-shadow(0 0 8px ${color}88)` }}
   147→        />
   148→        {/* Tick marks */}
   149→        {tickMarks.map((tick) => (
   150→          <g key={tick.label}>
   151→            <line x1={tick.ix} y1={tick.iy} x2={tick.ox} y2={tick.oy} stroke="#4b5563" strokeWidth={1.5} />
   152→            <text
   153→              x={cx + (radius + 22) * Math.cos(((startAngle + (tick.label / 100) * totalAngle) * Math.PI) / 180)}
   154→              y={cy + (radius + 22) * Math.sin(((startAngle + (tick.label / 100) * totalAngle) * Math.PI) / 180)}
   155→              textAnchor="middle" dominantBaseline="middle"
   156→              fill="#6b7280" fontSize={11}
   157→            >
   158→              {tick.label}
   159→            </text>
   160→          </g>
   161→        ))}
   162→        {/* Needle dot */}
   163→        <motion.circle
   164→          cx={needlePos.x} cy={needlePos.y} r={6}
   165→          fill={color}
   166→          initial={{ opacity: 0, scale: 0 }}
   167→          animate={{ opacity: 1, scale: 1 }}
   168→          transition={{ delay: 1.2, duration: 0.3 }}
   169→          style={{ filter: `drop-shadow(0 0 6px ${color})` }}
   170→        />
   171→      </svg>
   172→      {/* Center score */}
   173→      <div className="absolute inset-0 flex flex-col items-center justify-center" style={{ paddingTop: 30 }}>
   174→        <motion.span
   175→          className="font-mono font-black tabular-nums"
   176→          style={{ color, fontSize: 72, lineHeight: 1, textShadow: `0 0 40px ${color}44` }}
   177→          initial={{ opacity: 0, scale: 0.5 }}
   178→          animate={{ opacity: 1, scale: 1 }}
   179→          transition={{ delay: 0.3, duration: 0.8, ease: 'easeOut' as const }}
   180→        >
   181→          {score.toFixed(1)}
   182→        </motion.span>
   183→        <span className="text-xs text-gray-500 mt-1 uppercase tracking-[0.3em]">out of 100</span>
   184→      </div>
   185→    </div>
   186→  );
   187→}

   189→// ── Sparkline (pure SVG) ──────────────────────────────────────────────
   190→
   191→function Sparkline({ data, color = '#60a5fa', width = 120, height = 32 }: {
   192→  data: number[]; color?: string; width?: number; height?: number;
   193→}) {
   194→  if (data.length < 2) return null;
   195→  const min = Math.min(...data);
   196→  const max = Math.max(...data);
   197→  const range = max - min || 1;
   198→  const points = data.map((v, i) => {
   199→    const x = (i / (data.length - 1)) * width;
   200→    const y = height - ((v - min) / range) * (height - 4) - 2;
   201→    return `${x},${y}`;
   202→  }).join(' ');
   203→
   204→  return (
   205→    <svg width={width} height={height} className="overflow-visible">
   206→      <polyline points={points} fill="none" stroke={color} strokeWidth={1.5} strokeLinecap="round" strokeLinejoin="round" />
   207→    </svg>
   208→  );
   209→}

   211→// ── 90-Day Trend Chart (pure SVG) ─────────────────────────────────────
   212→
   213→function TrendChart({ data }: { data: HistoricalDataPoint[] }) {
   214→  const width = 900;
   215→  const height = 280;
   216→  const padding = { top: 20, right: 20, bottom: 40, left: 45 };
   217→  const chartW = width - padding.left - padding.right;
   218→  const chartH = height - padding.top - padding.bottom;
   219→
   220→  const scores = data.map(d => d.score);
   221→  const movingAvg = calculateMovingAverage(scores, 7);
   222→  const minScore = 0;
   223→  const maxScore = 100;
   224→
   225→  function xScale(i: number) { return padding.left + (i / (data.length - 1)) * chartW; }
   226→  function yScale(v: number) { return padding.top + chartH - ((v - minScore) / (maxScore - minScore)) * chartH; }
   227→
   228→  // Level zone backgrounds
   229→  const zones = [
   230→    { from: 0, to: 20, color: '#22c55e08' },
   231→    { from: 20, to: 40, color: '#ffaa0008' },
   232→    { from: 40, to: 60, color: '#ff884408' },
   233→    { from: 60, to: 80, color: '#ff335508' },
   234→    { from: 80, to: 100, color: '#7f1d1d08' },
   235→  ];
   236→
   237→  // Line path
   238→  const linePoints = data.map((d, i) => `${xScale(i)},${yScale(d.score)}`).join(' ');
   239→  // Area path
   240→  const areaPath = `M ${xScale(0)},${yScale(minScore)} ` +
   241→    data.map((d, i) => `L ${xScale(i)},${yScale(d.score)}`).join(' ') +
   242→    ` L ${xScale(data.length - 1)},${yScale(minScore)} Z`;
   243→
   244→  // Moving average path
   245→  const maPoints = movingAvg.map((v, i) => `${xScale(i)},${yScale(v)}`).join(' ');
   246→
   247→  // Find spike points (score > 15 above 7-day MA)
   248→  const spikes: number[] = [];
   249→  for (let i = 7; i < data.length; i++) {
   250→    if (data[i].score - movingAvg[i] > 15) spikes.push(i);
   251→  }
   252→
   253→  // Date labels (every ~15 days)
   254→  const dateLabels: { x: number; label: string }[] = [];
   255→  const step = Math.max(1, Math.floor(data.length / 6));
   256→  for (let i = 0; i < data.length; i += step) {
   257→    const d = new Date(data[i].date);
   258→    dateLabels.push({ x: xScale(i), label: `${d.getMonth() + 1}/${d.getDate()}` });
   259→  }
   260→
   261→  const [hoveredSpike, setHoveredSpike] = useState<number | null>(null);
   262→
   263→  return (
   264→    <div className="w-full overflow-x-auto">
   265→      <svg viewBox={`0 0 ${width} ${height}`} className="w-full h-auto" style={{ minWidth: 600 }}>
   266→        {/* Zone backgrounds */}
   267→        {zones.map(z => (
   268→          <rect
   269→            key={z.to}
   270→            x={padding.left} y={yScale(z.to)} width={chartW} height={yScale(z.from) - yScale(z.to)}
   271→            fill={z.color}
   272→          />
   273→        ))}
   274→        {/* Grid lines */}
   275→        {[0, 20, 40, 60, 80, 100].map(v => (
   276→          <line key={v} x1={padding.left} y1={yScale(v)} x2={width - padding.right} y2={yScale(v)} stroke="#374151" strokeWidth={0.5} strokeDasharray="4 4" />
   277→        ))}
   278→        {/* Y-axis labels */}
   279→        {[0, 20, 40, 60, 80, 100].map(v => (
   280→          <text key={v} x={padding.left - 8} y={yScale(v) + 4} textAnchor="end" fill="#6b7280" fontSize={11}>{v}</text>
   281→        ))}
   282→        {/* Date labels */}
   283→        {dateLabels.map((d, i) => (
   284→          <text key={i} x={d.x} y={height - 8} textAnchor="middle" fill="#6b7280" fontSize={10}>{d.label}</text>
   285→        ))}
   286→        {/* Area fill */}
   287→        <path d={areaPath} fill="url(#fearGradient)" opacity={0.3} />
   288→        <defs>
   289→          <linearGradient id="fearGradient" x1="0" y1="0" x2="0" y2="1">
   290→            <stop offset="0%" stopColor="#ff3355" stopOpacity={0.4} />
   291→            <stop offset="100%" stopColor="#ff3355" stopOpacity={0} />
   292→          </linearGradient>
   293→        </defs>
   294→        {/* Score line */}
   295→        <motion.polyline
   296→          points={linePoints} fill="none" stroke="#60a5fa" strokeWidth={2} strokeLinejoin="round"
   297→          initial={{ pathLength: 0 }} animate={{ pathLength: 1 }} transition={{ duration: 2, ease: 'easeOut' as const }}
   298→        />
   299→        {/* Moving average */}
   300→        <motion.polyline
   301→          points={maPoints} fill="none" stroke="#f59e0b" strokeWidth={1.5} strokeDasharray="6 3" strokeLinejoin="round" opacity={0.7}
   302→          initial={{ pathLength: 0 }} animate={{ pathLength: 1 }} transition={{ duration: 2, delay: 0.5, ease: 'easeOut' as const }}
   303→        />
   304→        {/* Spike markers */}
   305→        {spikes.map(i => (
   306→          <g key={i}
   307→            onMouseEnter={() => setHoveredSpike(i)} onMouseLeave={() => setHoveredSpike(null)}
   308→            className="cursor-pointer"
   309→          >
   310→            <circle cx={xScale(i)} cy={yScale(data[i].score)} r={5} fill="#ff3355" stroke="#1f2937" strokeWidth={2} />
   311→            <line x1={xScale(i)} y1={yScale(data[i].score)} x2={xScale(i)} y2={yScale(0)} stroke="#ff335544" strokeWidth={1} strokeDasharray="3 3" />
   312→          </g>
   313→        ))}
   314→        {/* Spike tooltip */}
   315→        {hoveredSpike !== null && (
   316→          <g>
   317→            <rect
   318→              x={Math.min(xScale(hoveredSpike) - 60, width - 170)}
   319→              y={yScale(data[hoveredSpike].score) - 50}
   320→              width={160} height={36} rx={6} fill="#1f2937" stroke="#374151" strokeWidth={1}
   321→            />
   322→            <text
   323→              x={Math.min(xScale(hoveredSpike) - 52, width - 162)}
   324→              y={yScale(data[hoveredSpike].score) - 28}
   325→              fill="#f8fafc" fontSize={11}
   326→            >
   327→              {data[hoveredSpike].date}: {data[hoveredSpike].score.toFixed(1)} — {data[hoveredSpike].level}
   328→            </text>
   329→          </g>
   330→        )}
   331→        {/* Legend */}
   332→        <line x1={width - 180} y1={12} x2={width - 160} y2={12} stroke="#60a5fa" strokeWidth={2} />
   333→        <text x={width - 155} y={16} fill="#9ca3af" fontSize={10}>Daily Score</text>
   334→        <line x1={width - 90} y1={12} x2={width - 70} y2={12} stroke="#f59e0b" strokeWidth={1.5} strokeDasharray="6 3" />
   335→        <text x={width - 65} y={16} fill="#9ca3af" fontSize={10}>7-day MA</text>
   336→      </svg>
   337→    </div>
   338→  );
   339→}

   437→// ── Threat Feed ────────────────────────────────────────────────────────
   438→
   439→function ThreatFeed({ threats }: { threats: string[] }) {
   440→  const [activeIndex, setActiveIndex] = useState(0);
   441→  const intervalRef = useRef<NodeJS.Timeout | null>(null);
   442→
   443→  useEffect(() => {
   444→    intervalRef.current = setInterval(() => {
   445→      setActiveIndex(prev => (prev + 1) % threats.length);
   446→    }, 5000);
   447→    return () => { if (intervalRef.current) clearInterval(intervalRef.current); };
   448→  }, [threats.length]);
   449→
   450→  const severityIcons = [AlertTriangle, Zap, Shield, Activity, Radio];
   451→  const severityColors = ['#ff3355', '#ff8844', '#ffaa00', '#60a5fa', '#a78bfa'];
   452→
   453→  return (
   454→    <div className="space-y-2">
   455→      <AnimatePresence mode="wait">
   456→        {threats.map((threat, i) => (
   457→          <motion.div
   458→            key={`${i}-${activeIndex === i ? 'active' : 'inactive'}`}
   459→            initial={{ opacity: 0, x: 20 }}
   460→            animate={{ opacity: activeIndex === i ? 1 : 0.4, x: 0 }}
   461→            exit={{ opacity: 0, x: -20 }}
   462→            transition={{ duration: 0.4 }}
   463→            className={`flex items-start gap-3 p-3 rounded-lg border transition-all ${
   464→              activeIndex === i ? 'bg-gray-800/50 border-gray-700' : 'border-transparent'
   465→            }`}
   466→            onMouseEnter={() => { if (intervalRef.current) clearInterval(intervalRef.current); }}
   467→            onMouseLeave={() => {
   468→              intervalRef.current = setInterval(() => {
   469→                setActiveIndex(prev => (prev + 1) % threats.length);
   470→              }, 5000);
   471→            }}
   472→          >
   473→            <div className="mt-0.5 shrink-0">
   474→              {(() => { const C = severityIcons[i % severityIcons.length]; return <C size={16} style={{ color: severityColors[i % severityColors.length] }} />; })()}

   481→        ))}
   482→      </AnimatePresence>

   490→    </div>
   491→  );
