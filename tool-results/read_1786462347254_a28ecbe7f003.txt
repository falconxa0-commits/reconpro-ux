   165→
   166→class QualityTrend:
   167→    """Track quality over time, detect regressions, project trajectory.
   168→
   169→    Snapshots are persisted at RECONPRO_HOME/memory/quality_snapshots.json.
   170→    """
   171→
   172→    MAX_SNAPSHOTS: int = 500
   173→
   174→    def __init__(self) -> None:
   175→        self._snapshots: List[QualitySnapshot] = []
   176→        self._load()
   177→
   178→    # ── Persistence ──────────────────────────────────────────────────────
   179→
   180→    def _load(self) -> None:
   181→        """Load snapshots from disk."""
   182→        try:
   183→            if SNAPSHOTS_FILE.exists():
   184→                raw = json.loads(SNAPSHOTS_FILE.read_text(encoding="utf-8"))
   185→                for entry in raw:
   186→                    self._snapshots.append(QualitySnapshot(
   187→                        timestamp=entry["timestamp"],
   188→                        composite_score=entry["composite_score"],
   189→                        dimensions=entry.get("dimensions", {}),
   190→                        file_count=entry.get("file_count", 0),
   191→                        total_loc=entry.get("total_loc", 0),
   192→                        repository_path=entry.get("repository_path", ""),
   193→                    ))
   194→                logger.debug("Loaded %d quality snapshots", len(self._snapshots))
   195→        except Exception as exc:
   196→            logger.warning("Failed to load quality snapshots: %s", exc)
   197→            self._snapshots = []
   198→
   199→    def _save(self) -> None:
   200→        """Persist snapshots to disk."""
   201→        try:
   202→            MEMORY_DIR.mkdir(parents=True, exist_ok=True)
   203→            data = [s.to_dict() for s in self._snapshots]
   204→            SNAPSHOTS_FILE.write_text(
   205→                json.dumps(data, indent=2), encoding="utf-8"
   206→            )
   207→            logger.debug("Saved %d quality snapshots", len(self._snapshots))
   208→        except Exception as exc:
   209→            logger.error("Failed to save quality snapshots: %s", exc)
   210→
   211→    # ── Public API ───────────────────────────────────────────────────────
   212→
   213→    def record_snapshot(
   214→        self,
   215→        composite_score: float,
   216→        dimensions: Dict[str, float],
   217→        file_count: int = 0,
   218→        total_loc: int = 0,
   219→        repository_path: str = "",
   220→    ) -> QualitySnapshot:
   221→        """Record a new quality snapshot and persist it.
   222→
   223→        Returns the created snapshot.
   224→        """
   225→        snapshot = QualitySnapshot(
   226→            timestamp=datetime.now(timezone.utc).isoformat(),
   227→            composite_score=composite_score,
   228→            dimensions=dict(dimensions),
   229→            file_count=file_count,
   230→            total_loc=total_loc,
   231→            repository_path=repository_path,
   232→        )
   233→        self._snapshots.append(snapshot)
   234→
   235→        # Trim to MAX_SNAPSHOTS
   236→        if len(self._snapshots) > self.MAX_SNAPSHOTS:
   237→            self._snapshots = self._snapshots[-self.MAX_SNAPSHOTS:]
   238→
   239→        self._save()
   240→        return snapshot
   241→
   242→    def get_snapshots(self) -> List[QualitySnapshot]:
   243→        """Return all recorded snapshots (newest last)."""
   244→        return list(self._snapshots)
   245→
   246→    def get_latest(self) -> Optional[QualitySnapshot]:
   247→        """Return the most recent snapshot, or None."""
   248→        return self._snapshots[-1] if self._snapshots else None

   250→    def compare_with_previous(self) -> Optional[Dict[str, Any]]:
   251→        """Compare the latest snapshot against the previous one.
   252→
   253→        Returns a dict with delta information, or None if fewer than 2
   254→        snapshots exist.
   255→        """
   256→        if len(self._snapshots) < 2:
   257→            return None
   258→
   259→        current = self._snapshots[-1]
   260→        previous = self._snapshots[-2]
   261→
   262→        score_delta = current.composite_score - previous.composite_score
   263→
   264→        # Per-dimension deltas
   265→        dim_deltas: Dict[str, float] = {}
   266→        all_dims = set(previous.dimensions) | set(current.dimensions)
   267→        for dim in sorted(all_dims):
   268→            prev_val = previous.dimensions.get(dim, 0.0)
   269→            curr_val = current.dimensions.get(dim, 0.0)
   270→            dim_deltas[dim] = round(curr_val - prev_val, 1)
   271→
   272→        # Identify regressions (any dimension dropped by > 5 points)
   273→        regressions = [
   274→            dim for dim, delta in dim_deltas.items() if delta < -5.0
   275→        ]
   276→        improvements = [
   277→            dim for dim, delta in dim_deltas.items() if delta > 5.0
   278→        ]
   279→
   280→        direction = "improving" if score_delta > 2 else (
   281→            "regressing" if score_delta < -2 else "stable"
   282→        )
   283→
   284→        return {
   285→            "current_timestamp": current.timestamp,
   286→            "previous_timestamp": previous.timestamp,
   287→            "score_delta": round(score_delta, 1),
   288→            "direction": direction,
   289→            "dimension_deltas": dim_deltas,
   290→            "regressions": regressions,
   291→            "improvements": improvements,
   292→        }
   293→
   294→    def detect_regressions(self, threshold: float = 5.0) -> List[Dict[str, Any]]:
   295→        """Scan all snapshot pairs for regressions exceeding *threshold*.
   296→
   297→        Returns a list of regression events.
   298→        """
   299→        regressions: List[Dict[str, Any]] = []
   300→        for i in range(1, len(self._snapshots)):
   301→            prev = self._snapshots[i - 1]
   302→            curr = self._snapshots[i]
   303→            delta = curr.composite_score - prev.composite_score
   304→            if delta < -threshold:
   305→                # Find which dimensions regressed
   306→                dim_regs = []
   307→                all_dims = set(prev.dimensions) | set(curr.dimensions)
   308→                for dim in sorted(all_dims):
   309→                    d = curr.dimensions.get(dim, 0.0) - prev.dimensions.get(dim, 0.0)
   310→                    if d < -threshold:
   311→                        dim_regs.append({
   312→                            "dimension": dim,
   313→                            "before": round(prev.dimensions.get(dim, 0.0), 1),
   314→                            "after": round(curr.dimensions.get(dim, 0.0), 1),
   315→                            "delta": round(d, 1),
   316→                        })
   317→                regressions.append({
   318→                    "from_timestamp": prev.timestamp,
   319→                    "to_timestamp": curr.timestamp,
   320→                    "score_delta": round(delta, 1),
   321→                    "regressed_dimensions": dim_regs,
   322→                })
   323→        return regressions
   324→
   325→    def project_trajectory(self, snapshots_ahead: int = 5) -> Optional[Dict[str, Any]]:
   326→        """Simple linear regression to project quality trajectory.
   327→
   328→        Returns projection data or None if fewer than 3 snapshots exist.
   329→        """
   330→        if len(self._snapshots) < 3:
   331→            return None
   332→
   333→        scores = [s.composite_score for s in self._snapshots]
   334→        n = len(scores)
   335→
   336→        # Linear regression: y = mx + b
   337→        x_vals = list(range(n))
   338→        x_mean = sum(x_vals) / n
   339→        y_mean = sum(scores) / n
   340→
   341→        numerator = sum((x - x_mean) * (y - y_mean) for x, y in zip(x_vals, scores))
   342→        denominator = sum((x - x_mean) ** 2 for x in x_vals)
   343→
   344→        if denominator == 0:
   345→            return None
   346→
   347→        slope = numerator / denominator
   348→        intercept = y_mean - slope * x_mean
   349→
   350→        # Project forward
   351→        projected: List[float] = []
   352→        for i in range(n, n + snapshots_ahead):
   353→            proj = intercept + slope * i
   354→            projected.append(round(max(0.0, min(100.0, proj)), 1))
   355→
   356→        # Trend description
   357→        if slope > 0.5:
   358→            trend = "improving"
   359→        elif slope < -0.5:
   360→            trend = "declining"
   361→        else:
   362→            trend = "stable"
   363→
   364→        return {
   365→            "slope": round(slope, 3),
   366→            "intercept": round(intercept, 1),
   367→            "trend": trend,
   368→            "projected_scores": projected,
   369→            "current_score": round(scores[-1], 1),
   370→            "data_points": n,
   371→        }

   381→    """Result of evaluating a quality gate."""

   385→    timestamp: str
   386→    score: float
   387→    threshold: float

   389→    actions_taken: List[str] = field(default_factory=list)
   390→
   391→    def to_dict(self) -> Dict[str, Any]:
   392→        return {
   393→            "gate_name": self.gate_name,
   394→            "passed": self.passed,
   395→            "timestamp": self.timestamp,
   396→            "score": self.score,
   397→            "threshold": self.threshold,
   398→            "failures": self.failures,

   402→
   403→class QualityGate:
   404→    """Gatekeeper for quality thresholds.
   405→
   406→    Gates are defined with per-dimension minimums and optional actions
   407→    (callables) triggered on failure.
   408→    """
   409→
   410→    def __init__(self) -> None:
   411→        self._gates: Dict[str, Dict[str, Any]] = {}
   412→        self._history: List[GateResult] = []
   413→        self._load_history()
   414→
   415→    # ── Persistence ──────────────────────────────────────────────────────
   416→
   417→    def _load_history(self) -> None:
   418→        """Load gate history from disk."""
   419→        try:
   420→            if GATE_HISTORY_FILE.exists():
   421→                raw = json.loads(GATE_HISTORY_FILE.read_text(encoding="utf-8"))
   422→                for entry in raw:
   423→                    self._history.append(GateResult(
   424→                        gate_name=entry["gate_name"],
   425→                        passed=entry["passed"],
   426→                        timestamp=entry["timestamp"],
   427→                        score=entry.get("score", 0.0),
   428→                        threshold=entry.get("threshold", 0.0),
   429→                        failures=entry.get("failures", []),
   430→                        actions_taken=entry.get("actions_taken", []),
   431→                    ))
   432→        except Exception as exc:
   433→            logger.warning("Failed to load gate history: %s", exc)
   434→            self._history = []
   435→
   436→    def _save_history(self) -> None:
   437→        """Persist gate history to disk."""
   438→        try:
   439→            MEMORY_DIR.mkdir(parents=True, exist_ok=True)
   440→            data = [r.to_dict() for r in self._history]
   441→            GATE_HISTORY_FILE.write_text(
   442→                json.dumps(data, indent=2), encoding="utf-8"
   443→            )
   444→        except Exception as exc:
   445→            logger.error("Failed to save gate history: %s", exc)
   446→
   447→    # ── Public API ───────────────────────────────────────────────────────
   448→
   449→    def define_gate(
   450→        self,
   451→        name: str,
   452→        thresholds: Dict[str, float],
   453→        actions: Optional[List[Callable[[], None]]] = None,
   454→    ) -> None:
   455→        """Define a quality gate.
   456→
   457→        Args:
   458→            name:        Unique gate identifier.
   459→            thresholds:  Dict mapping dimension name to minimum score.
   460→                         Use "_composite" for the overall score threshold.
   461→            actions:      Optional list of callables invoked on gate failure.
   462→        """
   463→        self._gates[name] = {
   464→            "thresholds": dict(thresholds),
   465→            "actions": actions or [],
   466→        }
   467→        logger.info("Defined quality gate '%s' with %d thresholds", name, len(thresholds))
   468→
   469→    def evaluate(
   470→        self,
   471→        gate_name: str,
   472→        quality_data: Dict[str, Any],
   473→    ) -> GateResult:
   474→        """Evaluate a named gate against quality data.
   475→
   476→        Args:
   477→            gate_name:    Name of a previously defined gate.
   478→            quality_data: Dict with "composite_score" (float) and
   479→                          "dimensions" (dict of dimension_name -> score).
   480→
   481→        Returns:
   482→            GateResult with pass/fail status and failure details.
   483→        """
   484→        if gate_name not in self._gates:
   485→            logger.warning("Gate '%s' not defined, auto-passing", gate_name)
   486→            result = GateResult(
   487→                gate_name=gate_name,
   488→                passed=True,
   489→                timestamp=datetime.now(timezone.utc).isoformat(),
   490→                score=quality_data.get("composite_score", 0.0),
   491→                threshold=0.0,
   492→                failures=[{"dimension": "_gate_not_found", "message": f"Gate '{gate_name}' is not defined"}],
   493→            )
   494→            self._history.append(result)
   495→            self._save_history()
   496→            return result
   497→
   498→        gate_def = self._gates[gate_name]
   499→        thresholds = gate_def["thresholds"]
   500→        actions = gate_def["actions"]
   501→
   502→        composite_score = quality_data.get("composite_score", 0.0)
   503→        dimensions = quality_data.get("dimensions", {})
   504→
   505→        failures: List[Dict[str, Any]] = []
   506→        composite_threshold = thresholds.get("_composite", 0.0)
   507→
   508→        # Check composite threshold
   509→        if composite_threshold > 0 and composite_score < composite_threshold:
   510→            failures.append({
   511→                "dimension": "_composite",
   512→                "message": f"Composite score {composite_score:.1f} below threshold {composite_threshold:.1f}",
   513→                "actual": round(composite_score, 1),
   514→                "required": composite_threshold,
   515→            })
   516→
   517→        # Check per-dimension thresholds
   518→        for dim_name, min_score in thresholds.items():
   519→            if dim_name == "_composite":
   520→                continue
   521→            actual = dimensions.get(dim_name, 0.0)
   522→            if actual < min_score:
   523→                failures.append({
   524→                    "dimension": dim_name,
   525→                    "message": f"{dim_name} score {actual:.1f} below threshold {min_score:.1f}",
   526→                    "actual": round(actual, 1),
   527→                    "required": min_score,
   528→                })
   529→
   530→        passed = len(failures) == 0
   531→
   532→        # Execute failure actions
   533→        actions_taken: List[str] = []
   534→        if not passed and actions:
   535→            for action in actions:
   536→                try:
   537→                    action()
   538→                    actions_taken.append(action.__name__)
   539→                except Exception as exc:
   540→                    actions_taken.append(f"{action.__name__} (failed: {exc})")
   541→                    logger.error("Gate action %s failed: %s", action.__name__, exc)
   542→
   543→        result = GateResult(
   544→            gate_name=gate_name,
   545→            passed=passed,
   546→            timestamp=datetime.now(timezone.utc).isoformat(),
   547→            score=composite_score,
   548→            threshold=composite_threshold,
   549→            failures=failures,
   550→            actions_taken=actions_taken,
   551→        )
   552→
   553→        self._history.append(result)
   554→        self._save_history()
   555→
   556→        status = "PASSED" if passed else "FAILED"
   557→        logger.info(
   558→            "Gate '%s' %s (score=%.1f, threshold=%.1f)",
   559→            gate_name, status, composite_score, composite_threshold,
   560→        )
   561→        return result
   562→
   563→    def get_gate_history(self, gate_name: Optional[str] = None) -> List[Dict[str, Any]]:
   564→        """Return gate evaluation history, optionally filtered by name."""
   565→        results = self._history
   566→        if gate_name is not None:
   567→            results = [r for r in results if r.gate_name == gate_name]
   568→        return [r.to_dict() for r in results]
   569→
   570→    def get_gate_pass_rate(self, gate_name: Optional[str] = None) -> float:
   571→        """Return pass rate as a fraction (0.0-1.0)."""
   572→        results = self._history
   573→        if gate_name is not None:
   574→            results = [r for r in results if r.gate_name == gate_name]
   575→        if not results:
   576→            return 0.0
   577→        return sum(1 for r in results if r.passed) / len(results)

   585→def _parse_file_safe(file_path: str) -> Optional[ast.AST]:
   586→    """Parse a Python file, returning None on any error."""

   596→class _ASTAnalyzer(ast.NodeVisitor):
   597→    """Collects code metrics from an AST tree."""
   598→
   599→    def __init__(self, source_lines: List[str]) -> None:
   600→        self.source_lines = source_lines
   601→        self.total_loc: int = len(source_lines)
   602→        self.blank_lines: int = 0
   603→        self.comment_lines: int = 0
   604→        self.functions: List[Dict[str, Any]] = []
   605→        self.classes: List[Dict[str, Any]] = []
   606→        self.imports: List[str] = []
   607→        self.max_nesting_depth: int = 0
   608→        self.total_branches: int = 0
   609→        self.has_type_hints: int = 0
   610→        self.total_params: int = 0
   611→        self.docstring_count: int = 0
   612→        self.docstring_targets: int = 0
   613→        self.except_handlers: int = 0
   614→        self.bare_except_handlers: int = 0
   615→        self.try_blocks: int = 0
   616→        self.security_issues: List[Dict[str, Any]] = []
   617→        self.naming_violations: List[Dict[str, Any]] = []
   618→        self._current_depth: int = 0
   619→
   620→        # Count blanks and comments
   621→        for line in source_lines:
   622→            stripped = line.strip()
   623→            if not stripped:
   624→                self.blank_lines += 1
   625→            elif stripped.startswith("#"):
   626→                self.comment_lines += 1
   627→
   628→    # ── Visitors ─────────────────────────────────────────────────────────
   629→
   630→    def visit_FunctionDef(self, node: ast.AST) -> None:  # type: ignore[override]
   631→        self._analyse_function(node)
   632→        self.generic_visit(node)
   633→
   634→    def visit_AsyncFunctionDef(self, node: ast.AST) -> None:  # type: ignore[override]
   635→        self._analyse_function(node)
   636→        self.generic_visit(node)
   637→
   638→    def visit_ClassDef(self, node: ast.AST) -> None:  # type: ignore[override]
   639→        name = getattr(node, "name", "")
   640→        bases = [getattr(b, "id", "") if isinstance(b, ast.Name) else "" for b in getattr(node, "bases", [])]
   641→        docstring = ast.get_docstring(node)
   642→        has_doc = docstring is not None
   643→
   644→        # Check class naming
   645→        if name and not _NAMING_PATTERNS["class"].match(name) and not name.startswith("_"):
   646→            self.naming_violations.append({
   647→                "name": name, "kind": "class", "line": getattr(node, "lineno", 0),
   648→            })
   649→
   650→        self.classes.append({
   651→            "name": name,
   652→            "bases": bases,
   653→            "has_docstring": has_doc,
   654→            "line": getattr(node, "lineno", 0),
   655→            "end_line": getattr(node, "end_lineno", 0),
   656→        })
   657→        if has_doc:
   658→            self.docstring_count += 1
   659→        self.docstring_targets += 1
   660→
   661→        self.generic_visit(node)
   662→
   663→    def visit_Import(self, node: ast.AST) -> None:  # type: ignore[override]
   664→        for alias in getattr(node, "names", []):
   665→            self.imports.append(getattr(alias, "name", ""))
   666→        self.generic_visit(node)
   667→
   668→    def visit_ImportFrom(self, node: ast.AST) -> None:  # type: ignore[override]
   669→        module = getattr(node, "module", "") or ""
   670→        for alias in getattr(node, "names", []):
   671→            self.imports.append(f"{module}.{getattr(alias, 'name', '')}")
   672→        self.generic_visit(node)
   673→
   674→    def visit_If(self, node: ast.AST) -> None:  # type: ignore[override]
   675→        self.total_branches += 1
   676→        self._current_depth += 1
   677→        self.max_nesting_depth = max(self.max_nesting_depth, self._current_depth)
   678→        self.generic_visit(node)
   679→        self._current_depth -= 1
   680→
   681→    def visit_For(self, node: ast.AST) -> None:  # type: ignore[override]
   682→        self.total_branches += 1
   683→        self._current_depth += 1
   684→        self.max_nesting_depth = max(self.max_nesting_depth, self._current_depth)
   685→        self.generic_visit(node)
   686→        self._current_depth -= 1
   687→
   688→    def visit_While(self, node: ast.AST) -> None:  # type: ignore[override]
   689→        self.total_branches += 1
   690→        self._current_depth += 1
   691→        self.max_nesting_depth = max(self.max_nesting_depth, self._current_depth)
   692→        self.generic_visit(node)
   693→        self._current_depth -= 1
   694→
   695→    def visit_With(self, node: ast.AST) -> None:  # type: ignore[override]
   696→        self.total_branches += 1
   697→        self._current_depth += 1
   698→        self.max_nesting_depth = max(self.max_nesting_depth, self._current_depth)
   699→        self.generic_visit(node)
   700→        self._current_depth -= 1
   701→
   702→    def visit_Try(self, node: ast.AST) -> None:  # type: ignore[override]
   703→        self.try_blocks += 1
   704→        for handler in getattr(node, "handlers", []):
   705→            self.except_handlers += 1
   706→            exc_type = getattr(handler, "type", None)
   707→            if exc_type is None:
   708→                self.bare_except_handlers += 1
   709→        self.generic_visit(node)
   710→
   711→    # ── Helpers ──────────────────────────────────────────────────────────
   712→
   713→    def _analyse_function(self, node: ast.AST) -> None:
   714→        """Analyse a function/method node."""
   715→        name = getattr(node, "name", "")
   716→        args = getattr(node, "args", None)
   717→        lineno = getattr(node, "lineno", 0)
   718→        end_lineno = getattr(node, "end_lineno", 0)
   719→        func_loc = max(1, (end_lineno or lineno) - lineno + 1)
   720→        docstring = ast.get_docstring(node)
   721→        has_doc = docstring is not None
   722→
   723→        # Parameter analysis
   724→        param_count = 0
   725→        hinted_params = 0
   726→        returns_hinted = False
   727→
   728→        if args:
   729→            all_args = (
   730→                list(getattr(args, "posonlyargs", [])) +
   731→                list(getattr(args, "args", [])) +
   732→                list(getattr(args, "kwonlyargs", []))
   733→            )
   734→            param_count = len(all_args)
   735→            for arg in all_args:
   736→                if getattr(arg, "annotation", None) is not None:
   737→                    hinted_params += 1
   738→            if getattr(args, "vararg", None) and getattr(args.vararg, "annotation", None):
   739→                hinted_params += 1
   740→                param_count += 1
   741→            if getattr(args, "kwarg", None) and getattr(args.kwarg, "annotation", None):
   742→                hinted_params += 1
   743→                param_count += 1
   744→
   745→        self.total_params += param_count
   746→        self.has_type_hints += hinted_params
   747→
   748→        if getattr(node, "returns", None) is not None:
   749→            returns_hinted = True
   750→
   751→        # Count nested branches
   752→        branch_count = 0
   753→        for child in ast.walk(node):
   754→            if isinstance(child, (ast.If, ast.For, ast.While, ast.With, ast.Try)):
   755→                branch_count += 1
   756→
   757→        # Check naming (skip dunder methods and private)
   758→        if (
   759→            name
   760→            and not name.startswith("__")
   761→            and not name.startswith("_")
   762→            and not _NAMING_PATTERNS["function"].match(name)
   763→        ):
   764→            self.naming_violations.append({
   765→                "name": name, "kind": "function", "line": lineno,
   766→            })
   767→
   768→        self.functions.append({
   769→            "name": name,
   770→            "loc": func_loc,
   771→            "params": param_count,
   772→            "hinted_params": hinted_params,
   773→            "returns_hinted": returns_hinted,
   774→            "has_docstring": has_doc,
   775→            "branches": branch_count,
   776→            "line": lineno,
   777→        })
   778→
   779→        if has_doc:
   780→            self.docstring_count += 1
   781→        self.docstring_targets += 1
   782→

   784→def _scan_security_patterns(source: str) -> List[Dict[str, Any]]:
   785→    """Scan source text for security anti-patterns."""
   786→    issues: List[Dict[str, Any]] = []
   787→    for line_no, line in enumerate(source.splitlines(), 1):
   788→        for pattern, description in _SECURITY_ANTI_PATTERNS:
   789→            if re.search(pattern, line):
   790→                issues.append({
   791→                    "line": line_no,
   792→                    "pattern": description,
   793→                    "snippet": line.strip()[:100],
   794→                })
   795→    return issues

   798→def _estimate_test_coverage(source_lines: List[str]) -> Dict[str, Any]:
   799→    """Estimate test coverage heuristically from test file content.
   800→
   801→    This is NOT an actual coverage measurement — it estimates the
   802→    *quality* of test files (assertions per function, etc.).
   803→    """
   804→    total_lines = len(source_lines)
   805→    if total_lines == 0:
   806→        return {"assert_count": 0, "test_function_count": 0, "mock_count": 0, "asserts_per_test": 0.0, "quality_estimate": 0.0}
   807→
   808→    assert_count = sum(1 for line in source_lines if re.search(r"\bassert\b", line))
   809→    test_funcs = sum(1 for line in source_lines if re.search(r"\bdef\s+test_", line))
   810→    mock_count = sum(1 for line in source_lines if re.search(r"\b(mock|patch|MagicMock)\b", line))
   811→
   812→    # Heuristic: quality based on asserts-per-test-function ratio
   813→    if test_funcs > 0:
   814→        asserts_per_test = assert_count / test_funcs
   815→    else:
   816→        asserts_per_test = 0.0
   817→
   818→    # Score: more asserts per test, more mocks = better
   819→    quality = min(100.0, asserts_per_test * 15 + mock_count * 2)
   820→    quality = max(0.0, quality)
   821→
   822→    return {
   823→        "assert_count": assert_count,
   824→        "test_function_count": test_funcs,
   825→        "mock_count": mock_count,
   826→        "asserts_per_test": round(asserts_per_test, 2),
   827→        "quality_estimate": round(quality, 1),
   828→    }

   831→def _compute_maintainability_index(
   832→    total_loc: int,
   833→    comment_lines: int,
   834→    blank_lines: int,
   835→    max_nesting: int,
   836→    avg_func_loc: float,
   837→) -> float:
   838→    """Compute a maintainability index (0-100).
   839→
   840→    Inspired by the original Maintainability Index formula but
   841→    adapted for simplicity and stdlib-only operation.
   842→    """
   843→    if total_loc == 0:
   844→        return 100.0
   845→
   846→    code_loc = max(1, total_loc - comment_lines - blank_lines)
   847→    comment_ratio = comment_lines / total_loc
   848→
   849→    # Penalise long functions
   850→    func_length_penalty = max(0.0, (avg_func_loc - 30) / 70) * 30  # 30 LOC is ideal
   851→
   852→    # Penalise deep nesting
   853→    nesting_penalty = max(0.0, (max_nesting - 3) / 7) * 25  # 3 levels is ideal
   854→
   855→    # Reward comments
   856→    comment_bonus = comment_ratio * 20
   857→
   858→    raw = 100.0 - func_length_penalty - nesting_penalty + comment_bonus
   859→    return max(0.0, min(100.0, round(raw, 1)))

   862→def _assign_severity(score: float) -> str:
   863→    """Map a quality score (0-100) to a ReconPro severity."""
   864→    if score >= 80:
   865→        return "low"
   866→    elif score >= 60:
   867→        return "medium"
   868→    elif score >= 40:
   869→        return "high"
   870→    return "critical"

   873→def _assign_grade(score: float) -> str:
   874→    """Map a quality score (0-100) to a ReconPro grade."""
   875→    for threshold, grade in GRADE_THRESHOLDS:
   876→        if score >= threshold:
   877→            return grade
   878→    return "F"

   886→        return "A"
   887→    elif score >= 70:
   888→        return "B"
   889→    elif score >= 55:

   891→    elif score >= 40:
   892→        return "D"

   900→
   901→class QualityIntelligence:
   902→    """Comprehensive quality analysis for Python source code.
   903→
   904→    Usage:
   905→        qi = QualityIntelligence()
   906→
   907→        # Analyse a single file
   908→        result = qi.analyze_code_quality("/path/to/file.py")
   909→
   910→        # Analyse entire repository
   911→        repo_result = qi.analyze_repository_quality("/path/to/repo")
   912→
   913→        # Get composite score
   914→        print(qi.get_quality_score())
   915→
   916→        # Get formatted report
   917→        print(qi.get_quality_report())
   918→    """
   919→
   920→    def __init__(
   921→        self,
   922→        repository_path: Optional[str] = None,
   923→        weights: Optional[Dict[str, float]] = None,
   924→    ) -> None:
   925→        self._repo_path = repository_path
   926→        self._weights = dict(weights or DEFAULT_DIMENSION_WEIGHTS)
   927→        self._dimensions: Dict[str, QualityDimension] = {}
   928→        self._file_analyses: Dict[str, Dict[str, Any]] = {}
   929→        self._composite_score: float = 0.0
   930→        self._thresholds: Dict[str, float] = {
   931→            "_composite": 60.0,  # Default: reject below 60
   932→        }
   933→        self._trend = QualityTrend()
   934→        self._gate = QualityGate()
   935→        self._total_files: int = 0
   936→        self._total_loc: int = 0
   937→
   938→        # Define a default "release" gate
   939→        self._gate.define_gate(
   940→            "release_ready",
   941→            {"_composite": 60.0, "security": 50.0, "reliability": 40.0},
   942→        )
   943→
   944→    # ── Public API ───────────────────────────────────────────────────────
   945→
   946→    def analyze_code_quality(self, file_path: str) -> Dict[str, Any]:
   947→        """Analyse a single file's quality.
   948→
   949→        Returns a dict with per-dimension scores and file-level metrics.
   950→        """
   951→        file_path = os.path.abspath(file_path)
   952→        logger.info("Analysing code quality: %s", file_path)
   953→
   954→        # Read source
   955→        try:
   956→            with open(file_path, "r", encoding="utf-8", errors="ignore") as f:
   957→                source = f.read()
   958→        except OSError as exc:
   959→            logger.error("Cannot read %s: %s", file_path, exc)
   960→            return {
   961→                "file": file_path,
   962→                "error": f"Cannot read file: {exc}",
   963→                "dimensions": {},
   964→            }
   965→
   966→        source_lines = source.splitlines()
   967→        tree = _parse_file_safe(file_path)
   968→
   969→        if tree is None:
   970→            # Can't parse — give minimal analysis
   971→            return {
   972→                "file": file_path,
   973→                "loc": len(source_lines),
   974→                "error": "Failed to parse file (syntax error)",
   975→                "dimensions": {},
   976→                "overall_score": 0.0,
   977→            }
   978→
   979→        # Run AST analyser
   980→        analyser = _ASTAnalyzer(source_lines)
   981→        analyser.visit(tree)
   982→
   983→        # Security pattern scan
   984→        security_issues = _scan_security_patterns(source)
   985→
   986→        # Is this a test file? (check filename and immediate parent dir only)
   987→        path_obj = Path(file_path)
   988→        is_test = (
   989→            path_obj.name.startswith("test_")
   990→            or path_obj.name.endswith("_test.py")
   991→            or path_obj.parent.name == "tests"
   992→            or path_obj.parent.name == "test"
   993→        )
   994→
   995→        # Compute dimension scores
   996→        dimensions = self._compute_dimensions(
   997→            analyser, security_issues, is_test, source
   998→        )
   999→
  1000→        # Store
  1001→        self._file_analyses[file_path] = {
  1002→            "dimensions": {k: v.to_dict() for k, v in dimensions.items()},
  1003→            "analyzer": analyser,
  1004→            "security_issues": security_issues,
  1005→            "loc": len(source_lines),
  1006→        }
  1007→
  1008→        return {
  1009→            "file": file_path,
  1010→            "loc": len(source_lines),
  1011→            "overall_score": round(self._weighted_score(dimensions), 1),
  1012→            "dimensions": {k: v.to_dict() for k, v in dimensions.items()},
  1013→            "security_issues": security_issues,
  1014→            "functions": len(analyser.functions),
  1015→            "classes": len(analyser.classes),
  1016→        }
  1017→
  1018→    def analyze_repository_quality(
  1019→        self,
  1020→        repository_path: Optional[str] = None,
  1021→    ) -> Dict[str, Any]:
  1022→        """Analyse all Python files in a repository.
  1023→
  1024→        Args:
  1025→            repository_path: Root directory. Defaults to the path
  1026→                             provided at construction time.
  1027→
  1028→        Returns:
  1029→            Repository-level quality analysis dict.
  1030→        """
  1031→        repo = repository_path or self._repo_path
  1032→        if not repo:
  1033→            logger.error("No repository path specified")
  1034→            return {"error": "No repository path specified", "dimensions": {}}
  1035→
  1036→        repo = os.path.abspath(repo)
  1037→        logger.info("Analysing repository quality: %s", repo)
  1038→
  1039→        # Collect all Python files
  1040→        py_files: List[str] = []
  1041→        for root, _dirs, files in os.walk(repo):
  1042→            # Skip hidden directories, __pycache__, .git, etc.
  1043→            skip_parts = {"__pycache__", ".git", ".tox", ".eggs", "node_modules", ".venv", "venv"}
  1044→            if any(part in skip_parts for part in Path(root).parts):
  1045→                continue
  1046→            for fname in sorted(files):
  1047→                if fname.endswith(".py"):
  1048→                    py_files.append(os.path.join(root, fname))
  1049→
  1050→        if not py_files:
  1051→            return {"error": "No Python files found", "dimensions": {}}
  1052→
  1053→        # Aggregate metrics across all files
  1054→        aggregate = _ASTAnalyzer([])
  1055→        total_security_issues: List[Dict[str, Any]] = []
  1056→        all_dimensions: List[Dict[str, QualityDimension]] = []
  1057→        total_loc = 0
  1058→        file_count = 0
  1059→
  1060→        for fpath in py_files:
  1061→            result = self.analyze_code_quality(fpath)
  1062→            if "error" in result and "dimensions" not in result:
  1063→                continue
  1064→            if not result.get("dimensions"):
  1065→                continue
  1066→
  1067→            file_count += 1
  1068→            total_loc += result.get("loc", 0)
  1069→            total_security_issues.extend(result.get("security_issues", []))
  1070→
  1071→            # Reconstruct dimensions for aggregation
  1072→            file_dims: Dict[str, QualityDimension] = {}
  1073→            for dim_name, dim_data in result.get("dimensions", {}).items():
  1074→                file_dims[dim_name] = QualityDimension(
  1075→                    name=dim_name,
  1076→                    score=dim_data["score"],
  1077→                    weight=dim_data["weight"],
  1078→                    metrics=dim_data.get("metrics", {}),
  1079→                )
  1080→            all_dimensions.append(file_dims)
  1081→
  1082→        # Compute aggregate dimensions
  1083→        agg_dimensions = self._aggregate_dimensions(all_dimensions)
  1084→        self._dimensions = agg_dimensions
  1085→        self._total_files = file_count
  1086→        self._total_loc = total_loc
  1087→
  1088→        # Compute composite
  1089→        self._composite_score = self._weighted_score(agg_dimensions)
  1090→
  1091→        # Record snapshot
  1092→        dim_scores = {k: v.score for k, v in agg_dimensions.items()}
  1093→        self._trend.record_snapshot(
  1094→            composite_score=self._composite_score,
  1095→            dimensions=dim_scores,
  1096→            file_count=file_count,
  1097→            total_loc=total_loc,
  1098→            repository_path=repo,
  1099→        )
  1100→
  1101→        # Evaluate default gate
  1102→        gate_result = self._gate.evaluate(
  1103→            "release_ready",
  1104→            {"composite_score": self._composite_score, "dimensions": dim_scores},
  1105→        )
  1106→
  1107→        # Trend analysis
  1108→        comparison = self._trend.compare_with_previous()
  1109→        trajectory = self._trend.project_trajectory()
  1110→
  1111→        return {
  1112→            "repository": repo,
  1113→            "file_count": file_count,
  1114→            "total_loc": total_loc,

  1258→    def _compute_dimensions(
  1259→        self,
  1260→        analyzer: _ASTAnalyzer,
  1261→        security_issues: List[Dict[str, Any]],
  1262→        is_test: bool,

  1267→

  1269→        dimensions["coverage"] = self._dim_coverage(analyzer, is_test, source)
  1270→        dimensions["type_safety"] = self._dim_type_safety(analyzer)
  1271→        dimensions["documentation"] = self._dim_documentation(analyzer)
  1272→        dimensions["security"] = self._dim_security(analyzer, security_issues)
  1273→        dimensions["maintainability"] = self._dim_maintainability(analyzer)

  1279→    def _dim_complexity(self, a: _ASTAnalyzer) -> QualityDimension:
  1280→        """Score code complexity (LOC per function, nesting depth)."""
  1281→        if not a.functions:
  1282→            return QualityDimension(
  1283→                name="complexity", score=100.0, weight=self._weights["complexity"],
  1284→                severity="info", grade="A+", details="No functions to analyse.",
  1285→            )
  1286→
  1287→        avg_loc = sum(f["loc"] for f in a.functions) / len(a.functions)
  1288→        max_loc = max(f["loc"] for f in a.functions)
  1289→        max_nesting = a.max_nesting_depth
  1290→        total_branches = a.total_branches
  1291→
  1292→        # Scoring: penalise large functions and deep nesting
  1293→        loc_score = max(0, 100 - (avg_loc - 20) * 2)  # 20 LOC = perfect
  1294→        loc_score = max(0, min(100, loc_score))
  1295→
  1296→        nesting_score = max(0, 100 - (max_nesting - 2) * 12)  # 2 levels = perfect
  1297→        nesting_score = max(0, min(100, nesting_score))
  1298→
  1299→        # Long function penalty
  1300→        long_funcs = sum(1 for f in a.functions if f["loc"] > 50)
  1301→        long_penalty = long_funcs * 5
  1302→
  1303→        score = max(0, min(100, (loc_score * 0.4 + nesting_score * 0.4 + 20) - long_penalty))
  1304→
  1305→        return QualityDimension(
  1306→            name="complexity",
  1307→            score=round(score, 1),
  1308→            weight=self._weights["complexity"],
  1309→            severity=_assign_severity(score),
  1310→            grade=_assign_dimension_grade(score),
  1311→            details=f"avg_func_loc={avg_loc:.0f}, max_func_loc={max_loc}, max_nesting={max_nesting}, branches={total_branches}",
  1312→            metrics={
  1313→                "avg_func_loc": round(avg_loc, 1),
  1314→                "max_func_loc": max_loc,
  1315→                "max_nesting_depth": max_nesting,
  1316→                "total_branches": total_branches,
  1317→                "long_functions": long_funcs,
  1318→                "function_count": len(a.functions),
  1319→            },
  1320→        )

  1338→
  1339→        # For non-test files: estimate based on whether test file likely exists

  1355→    def _dim_type_safety(self, a: _ASTAnalyzer) -> QualityDimension:
  1356→        """Score type hint coverage."""
  1357→        if a.total_params == 0 and not a.functions:
  1358→            return QualityDimension(
  1359→                name="type_safety", score=100.0, weight=self._weights["type_safety"],
  1360→                severity="info", grade="A+", details="No functions to analyse.",
  1361→            )
  1362→
  1363→        if a.total_params == 0:
  1364→            return QualityDimension(
  1365→                name="type_safety", score=80.0, weight=self._weights["type_safety"],
  1366→                severity="low", grade="A-", details="Functions exist but have no parameters.",
  1367→            )
  1368→
  1369→        param_ratio = a.has_type_hints / a.total_params
  1370→
  1371→        # Check return type hints
  1372→        funcs_with_return = 0
  1373→        funcs_total = len(a.functions)
  1374→        for f in a.functions:
  1375→            if f["returns_hinted"]:
  1376→                funcs_with_return += 1
  1377→        return_ratio = funcs_with_return / funcs_total if funcs_total > 0 else 1.0
  1378→
  1379→        score = (param_ratio * 60 + return_ratio * 40)
  1380→        score = max(0, min(100, score))
  1381→
  1382→        return QualityDimension(
  1383→            name="type_safety",
  1384→            score=round(score, 1),
  1385→            weight=self._weights["type_safety"],
  1386→            severity=_assign_severity(score),
  1387→            grade=_assign_dimension_grade(score),
  1388→            details=f"param_hints={a.has_type_hints}/{a.total_params} ({param_ratio:.0%}), return_hints={funcs_with_return}/{funcs_total} ({return_ratio:.0%})",
  1389→            metrics={
  1390→                "params_hinted": a.has_type_hints,
  1391→                "total_params": a.total_params,
  1392→                "param_hint_ratio": round(param_ratio, 3),
  1393→                "returns_hinted": funcs_with_return,
  1394→                "total_functions": funcs_total,
  1395→                "return_hint_ratio": round(return_ratio, 3),
  1396→            },
  1397→        )

  1436→    def _dim_security(self, a: _ASTAnalyzer, issues: List[Dict[str, Any]]) -> QualityDimension:
  1437→        """Score security (lower anti-pattern count = higher score)."""
  1438→        issue_count = len(issues)
  1439→
  1440→        # Start at 100, deduct per issue
  1441→        deductions = {
  1442→            "use of eval()": 20,
  1443→            "use of exec()": 20,
  1444→            "pickle deserialization (unsafe)": 15,
  1445→            "marshal deserialization (unsafe)": 15,
  1446→            "os.system() command injection risk": 15,
  1447→            "shell=True in subprocess": 12,
  1448→            "SSL verification disabled": 10,
  1449→            "certificate verification disabled": 8,
  1450→            "hostname verification disabled": 8,
  1451→            "yaml.load without Loader (unsafe)": 12,
  1452→            "compile with exec mode": 15,
  1453→            "weak hash algorithm (MD5)": 8,
  1454→            "weak hash algorithm (SHA1)": 5,
  1455→            "mktemp is insecure (race condition)": 10,
  1456→            "dynamic import via __import__": 8,
  1457→            "attribute name introspection": 3,
  1458→            "dunder attribute access": 3,
  1459→            "noqa comment (suppressed linting)": 2,

  1464→
  1465→        total_deduction = 0
  1466→        for issue in issues:
  1467→            pattern_desc = issue.get("pattern", "")
  1468→            total_deduction += deductions.get(pattern_desc, 3)

  1470→        score = max(0, 100 - total_deduction)
  1471→
  1472→        # Also penalise bare except
  1473→        if a.bare_except_handlers > 0:
  1474→            score = max(0, score - a.bare_except_handlers * 8)
  1475→
  1476→        return QualityDimension(
  1477→            name="security",
  1478→            score=round(score, 1),
  1479→            weight=self._weights["security"],
  1480→            severity=_assign_severity(score),
  1481→            grade=_assign_dimension_grade(score),
  1482→            details=f"anti_patterns={issue_count}, bare_except={a.bare_except_handlers}, security_deduction={total_deduction}",
  1483→            metrics={
  1484→                "anti_pattern_count": issue_count,
  1485→                "bare_except_handlers": a.bare_except_handlers,
  1486→                "total_deduction": total_deduction,
  1487→                "issues_sample": issues[:10],
  1488→            },
  1489→        )

  1560→    def _dim_consistency(self, a: _ASTAnalyzer) -> QualityDimension:
  1561→        """Score naming/style consistency."""
  1562→        violations = a.naming_violations
  1563→        violation_count = len(violations)
  1564→
  1565→        # Score: start at 100, deduct per violation
  1566→        score = max(0, 100 - violation_count * 8)
  1567→
  1568→        # Check for TODO/FIXME/HACK comments
  1569→        todo_count = sum(
  1570→            1 for line in a.source_lines
  1571→            if re.search(r"\b(TODO|FIXME|HACK|XXX)\b", line, re.IGNORECASE)
  1572→        )
  1573→        todo_penalty = min(20, todo_count * 3)
  1574→        score = max(0, score - todo_penalty)
  1575→
  1576→        # Check for mixed indentation (tabs vs spaces)
  1577→        tab_lines = sum(1 for line in a.source_lines if "\t" in line)
  1578→        tab_penalty = min(15, tab_lines * 0.5)
  1579→        score = max(0, score - tab_penalty)
  1580→
  1581→        return QualityDimension(
  1582→            name="consistency",
  1583→            score=round(score, 1),
  1584→            weight=self._weights["consistency"],
  1585→            severity=_assign_severity(score),
  1586→            grade=_assign_dimension_grade(score),
  1587→            details=f"naming_violations={violation_count}, todos={todo_count}, tab_lines={tab_lines}",
  1588→            metrics={
  1589→                "naming_violations": violation_count,
  1590→                "naming_violation_details": violations[:10],
  1591→                "todo_count": todo_count,
  1592→                "tab_lines": tab_lines,
  1593→            },

  1602→        """Aggregate per-file dimensions into repository-level dimensions.
  1603→
  1604→        Uses weighted averages across all files.
  1605→        """
  1606→        if not file_dimensions:
  1607→            return {}
  1608→
  1609→        dim_names = DEFAULT_DIMENSION_WEIGHTS.keys()
  1610→        aggregated: Dict[str, QualityDimension] = {}
  1611→
  1612→        for dim_name in dim_names:
  1613→            scores = []
  1614→            all_metrics: List[Dict[str, Any]] = []
  1615→            weight = self._weights.get(dim_name, 0.1)
  1616→
  1617→            for file_dims in file_dimensions:
  1618→                dim = file_dims.get(dim_name)
  1619→                if dim is not None:
  1620→                    scores.append(dim.score)
  1621→                    if dim.metrics:
  1622→                        all_metrics.append(dim.metrics)
  1623→
  1624→            if not scores:
  1625→                aggregated[dim_name] = QualityDimension(
  1626→                    name=dim_name, score=0.0, weight=weight,
  1627→                    severity="critical", grade="F",
  1628→                    details="No data available.",
  1629→                )
  1630→                continue
  1631→
  1632→            avg_score = sum(scores) / len(scores)
  1633→
  1634→            # Merge metrics (sum numeric values)
  1635→            merged_metrics: Dict[str, Any] = {}
  1636→            for m in all_metrics:
  1637→                for k, v in m.items():
  1638→                    if isinstance(v, (int, float)) and not isinstance(v, bool):
  1639→                        merged_metrics[k] = merged_metrics.get(k, 0) + v
  1640→
  1641→            aggregated[dim_name] = QualityDimension(
  1642→                name=dim_name,
  1643→                score=round(avg_score, 1),
  1644→                weight=weight,
  1645→                severity=_assign_severity(avg_score),
  1646→                grade=_assign_dimension_grade(avg_score),
  1647→                details=f"average across {len(scores)} files",
  1648→                metrics=merged_metrics,
  1649→            )
  1650→
  1651→        return aggregated
  1652→
  1653→    def _weighted_score(self, dimensions: Dict[str, QualityDimension]) -> float:
  1654→        """Compute weighted composite score from dimensions."""
  1655→        if not dimensions:
  1656→            return 0.0
  1657→
  1658→        total_weight = 0.0
  1659→        weighted_sum = 0.0
  1660→
  1661→        for dim_name, dim in dimensions.items():
  1662→            w = self._weights.get(dim_name, 0.1)
  1663→            weighted_sum += dim.score * w
  1664→            total_weight += w
  1665→
  1666→        if total_weight == 0:
  1667→            return 0.0
  1668→
  1669→        return max(MIN_SCORE, min(MAX_SCORE, weighted_sum / total_weight))
