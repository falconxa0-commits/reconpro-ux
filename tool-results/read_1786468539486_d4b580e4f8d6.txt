     1→"""Async scan orchestration engine for ReconPro v10.0.
     2→
     3→Runs modules concurrently with an event-driven architecture.  The public
     4→``scan()`` function is a drop-in replacement for ``scanner.scan()`` — it
     5→accepts the same arguments and returns an identical ``ReconProResult`` so
     6→that every existing consumer (CLI, TUI, reports, tests) works unchanged.
     7→
     8→Components
     9→──────────
    10→1. **ScanEvent** — lightweight dataclass emitted at every scan milestone.
    11→2. **ScanEngine** — async orchestrator that resolves modules, runs them
    12→   with bounded concurrency, and aggregates results.
    13→3. **EventCollector** — convenience helper that buffers events for later
    14→   inspection (useful in tests and TUI dashboards).
    15→4. **concurrent_scan()** — scan many targets in parallel.
    16→"""
    17→
    18→from __future__ import annotations
    19→
    20→import asyncio
    21→import json
    22→import logging
    23→import time

    25→from typing import Any, Callable, Dict, List, Optional
    26→
    27→logger = logging.getLogger(__name__)

    41→# ── Scan Event ──────────────────────────────────────────────────────────
    42→
    43→
    44→@dataclass
    45→class ScanEvent:
    46→    """Event emitted during scan execution.
    47→
    48→    Attributes
    49→    ----------
    50→    type : str
    51→        One of ``SCAN_START``, ``MODULE_START``, ``FINDING``,
    52→        ``MODULE_COMPLETE``, ``SCAN_COMPLETE``.
    53→    module_id : str | None
    54→        The module that produced the event (``None`` for scan-level events).
    55→    target : str | None
    56→        The scan target (``None`` for module-level events).
    57→    finding : Finding | None
    58→        Present only for ``FINDING`` events.
    59→    findings_count : int
    60→        Present only for ``MODULE_COMPLETE`` events.
    61→    duration : float
    62→        Elapsed seconds for the module (``MODULE_COMPLETE``) or the
    63→        whole scan (``SCAN_COMPLETE``).
    64→    result : ReconProResult | None
    65→        Present only for ``SCAN_COMPLETE``.
    66→    modules : List[str]
    67→        Present only for ``SCAN_START`` — the resolved module list.
    68→    timestamp : float
    69→        ``time.monotonic()`` when the event was created.
    70→    """
    71→
    72→    type: str
    73→    module_id: Optional[str] = None
    74→    target: Optional[str] = None
    75→    finding: Optional[Finding] = None
    76→    findings_count: int = 0
    77→    duration: float = 0.0
    78→    result: Optional[ReconProResult] = None
    79→    modules: List[str] = field(default_factory=list)
    80→    timestamp: float = field(default_factory=time.monotonic)
    81→
    82→
    83→# ── Event Collector ─────────────────────────────────────────────────────
    84→
    85→
    86→class EventCollector:
    87→    """Buffer that stores every ``ScanEvent`` for later inspection.
    88→
    89→    Usage::
    90→
    91→        collector = EventCollector()
    92→        engine = ScanEngine(event_callback=collector)
    93→        await engine.run("example.com")
    94→        print(collector.timeline())
    95→        for f in collector.findings:
    96→            print(f.title)
    97→    """
    98→
    99→    def __init__(self) -> None:
   100→        self.events: List[ScanEvent] = []
   101→        self._start: float = 0.0
   102→
   103→    def __call__(self, event: ScanEvent) -> None:
   104→        """Append *event*; record scan start time for timeline."""
   105→        if event.type == "SCAN_START" and not self._start:
   106→            self._start = event.timestamp
   107→        self.events.append(event)
   108→
   109→    # -- derived views -----------------------------------------------------
   110→
   111→    @property
   112→    def findings(self) -> List[Finding]:
   113→        """All findings extracted from ``FINDING`` events, in order."""
   114→        return [e.finding for e in self.events if e.type == "FINDING" and e.finding is not None]
   115→
   116→    def by_module(self) -> Dict[str, List[ScanEvent]]:
   117→        """Events grouped by ``module_id``.
   118→
   119→        Scan-level events (``SCAN_START``, ``SCAN_COMPLETE``) are stored
   120→        under the key ``"_scan"``.
   121→        """
   122→        out: Dict[str, List[ScanEvent]] = {}
   123→        for e in self.events:
   124→            key = e.module_id if e.module_id else "_scan"
   125→            out.setdefault(key, []).append(e)
   126→        return out
   127→
   128→    def timeline(self) -> List[str]:
   129→        """Human-readable timeline of events relative to scan start.
   130→
   131→        Returns a list of formatted strings suitable for printing.
   132→        """
   133→        lines: List[str] = []
   134→        anchor = self._start or (self.events[0].timestamp if self.events else 0.0)
   135→        for e in self.events:
   136→            dt = e.timestamp - anchor
   137→            tag = f"[{dt:6.2f}s] {e.type}"
   138→            if e.module_id:
   139→                tag += f" ({e.module_id})"
   140→            if e.target and e.type in ("SCAN_START", "SCAN_COMPLETE"):
   141→                tag += f" target={e.target}"
   142→            if e.type == "FINDING" and e.finding:
   143→                tag += f" [{e.finding.severity}] {e.finding.title}"
   144→            if e.type == "MODULE_COMPLETE":
   145→                tag += f" {e.findings_count} findings in {e.duration:.2f}s"
   146→            if e.type == "SCAN_COMPLETE" and e.result:
   147→                tag += f" score={e.result.total_score} grade={e.result.grade}"
   148→            lines.append(tag)
   149→        return lines
   150→

   152→# ── Scan Engine ─────────────────────────────────────────────────────────
   153→
   154→
   155→class ScanEngine:
   156→    """Async scan orchestrator with bounded concurrency and event emission.
   157→
   158→    Parameters
   159→    ----------
   160→    event_callback : Callable[[ScanEvent], None] | None
   161→        If provided, called for every ``ScanEvent`` produced during the scan.
   162→    concurrency : int
   163→        Maximum number of modules that run simultaneously.
   164→    rate_limit : float
   165→        Default requests-per-second cap (overridden per-scan).
   166→    use_async : bool
   167→        When ``True`` the engine calls ``async_run()`` on modules that
   168→        expose it; otherwise every module is wrapped in
   169→        ``asyncio.to_thread``.
   170→    run_engineering : bool
   171→        When ``True``, the engineering pipeline runs automatically after
   172→        each scan completes.  Default ``False`` to preserve existing
   173→        behaviour.
   174→    engineering_repo_path : str
   175→        Repository path passed to the
   176→        ``ContinuousEngineeringOrchestrator`` when engineering is enabled.
   177→    """
   178→
   179→    def __init__(
   180→        self,
   181→        event_callback: Optional[Callable[[ScanEvent], None]] = None,
   182→        concurrency: int = 5,
   183→        rate_limit: float = 50.0,
   184→        use_async: bool = True,
   185→        run_engineering: bool = False,
   186→        engineering_repo_path: str = ".",
   187→        run_intelligence: bool = True,
   188→        run_quality: bool = False,
   189→        run_defense: bool = False,
   190→    ) -> None:
   191→        self._callback = event_callback
   192→        self._concurrency = concurrency
   193→        self._default_rate_limit = rate_limit
   194→        self._use_async = use_async
   195→        self._run_engineering = run_engineering
   196→        self._engineering_repo_path = engineering_repo_path
   197→        self._rate_limit = rate_limit
   198→        self._run_intelligence = run_intelligence
   199→        self._run_quality = run_quality
   200→        self._run_defense = run_defense
   201→
   202→        # Agent 5: graceful shutdown + per-module adaptive timeouts
   203→        self._shutting_down: bool = False
   204→        self._module_timings: Dict[str, List[float]] = {}  # mod_id -> [durations]
   205→        self._GRACE_PERIOD: float = 5.0  # seconds to let in-flight modules finish
   206→
   207→    # -- internal helpers --------------------------------------------------
   208→
   209→    def _emit(self, event: ScanEvent) -> None:
   210→        """Fire the event callback (if any)."""
   211→        if self._callback is not None:
   212→            try:
   213→                self._callback(event)
   214→            except Exception:
   215→                logger.debug("Event callback error", exc_info=True)
   216→
   217→    def request_shutdown(self) -> None:
   218→        """Signal the engine to stop accepting new module executions."""
   219→        self._shutting_down = True
   220→        logger.info("ScanEngine shutdown requested — no new modules will start")
   221→
   222→    def _get_adaptive_timeout(self, mod_id: str, default_timeout: int) -> int:
   223→        """Compute per-module adaptive timeout.
   224→
   225→        Returns ``max(default_timeout, avg_time * 2.5)`` based on a rolling
   226→        window of the last 5 execution times for *mod_id*.
   227→        """
   228→        timings = self._module_timings.get(mod_id, [])
   229→        if not timings:
   230→            return default_timeout
   231→        avg = sum(timings[-5:]) / len(timings[-5:])
   232→        return int(max(default_timeout, avg * 2.5))
   233→
   234→    def _record_module_timing(self, mod_id: str, duration: float) -> None:
   235→        """Append an execution duration to the rolling window."""
   236→        self._module_timings.setdefault(mod_id, []).append(duration)
   237→        if len(self._module_timings[mod_id]) > 5:
   238→            self._module_timings[mod_id] = self._module_timings[mod_id][-5:]
   239→
   240→    @staticmethod
   241→    def _resolve_remote_modules(
   242→        modules: Optional[List[str]],
   243→        all_modules: bool,
   244→    ) -> List[str]:
   245→        """Determine which remote modules to run (mirrors scanner.py logic)."""
   246→        if all_modules:
   247→            return [m for m in ALL_MODULES if m in MODULE_REGISTRY]
   248→        if modules:
   249→            return [m.strip().lower() for m in modules if m.strip().lower() in MODULE_REGISTRY]
   250→        return list(DEFAULT_MODULES)
   251→
   252→    @staticmethod
   253→    def _resolve_local_modules(
   254→        modules: Optional[List[str]],
   255→        all_modules: bool,
   256→    ) -> List[str]:
   257→        """Determine which local modules to run (mirrors scanner.py logic)."""
   258→        if all_modules:
   259→            return list(DEFAULT_LOCAL_MODULES)
   260→        if modules:
   261→            return [m.strip().lower() for m in modules if m.strip().lower() in LOCAL_MODULES]
   262→        return list(DEFAULT_LOCAL_MODULES)
   263→
   264→    def _build_result(
   265→        self,
   266→        target: str,
   267→        mods: List[str],
   268→        all_findings: List[Finding],
   269→        module_results: Dict[str, Dict[str, Any]],
   270→        vibesec_score: Optional[int],
   271→        vibesec_grade: Optional[str],
   272→    ) -> ReconProResult:
   273→        """Aggregate findings into a ``ReconProResult`` identical to scanner.py."""
   274→        total_score = compute_score(all_findings)
   275→        grade = compute_grade(total_score)
   276→        host = target.replace("https://", "").replace("http://", "").split("/")[0]
   277→        badge = badge_markdown(host, grade)
   278→
   279→        sev_counts: Dict[str, int] = {}
   280→        for f in all_findings:
   281→            sev_counts[f.severity] = sev_counts.get(f.severity, 0) + 1
   282→
   283→        return ReconProResult(
   284→            target=host,
   285→            modules_run=mods,
   286→            findings=[f.to_dict() for f in all_findings],
   287→            severity_counts=sev_counts,
   288→            total_score=total_score,
   289→            grade=grade,
   290→            badge_markdown=badge,
   291→            vibesec_score=vibesec_score,
   292→            vibesec_grade=vibesec_grade,
   293→            module_results=module_results,
   294→        )
   295→
   296→    # -- single module execution -------------------------------------------
   297→
   298→    @staticmethod
   299→    def _is_plugin_module(mod_id: str, runner: Any, entry: Any) -> bool:
   300→        """Determine if a module is a third-party plugin (vs built-in)."""
   301→        if entry and isinstance(entry, dict) and entry.get("is_plugin"):
   302→            return True
   303→        if hasattr(runner, "__module__") and "plugins" in (runner.__module__ or ""):
   304→            return True
   305→        return False
   306→
   307→    async def _run_module(
   308→        self,
   309→        mod_id: str,
   310→        runner: Any,
   311→        target: str,
   312→        base_url: str,
   313→        timeout: int,
   314→        verify_tls: bool,
   315→        limiter: Any,
   316→        semaphore: asyncio.Semaphore,
   317→        findings_lock: asyncio.Lock,
   318→        all_findings: List[Finding],
   319→        module_results: Dict[str, Dict[str, Any]],
   320→        vibesec_state: Dict[str, Any],
   321→        _entry: Any = None,
   322→    ) -> None:
   323→        """Execute one module under the concurrency semaphore."""
   324→        # Security hardening: validate plugin modules through PluginSandbox
   325→        if self._is_plugin_module(mod_id, runner, _entry):
   326→            try:
   327→                from .security_hardening import PluginSandbox
   328→                sandbox = PluginSandbox(cpu_time_seconds=float(timeout))
   329→                sandbox_result = sandbox.execute(
   330→                    runner,
   331→                    plugin_name=mod_id,
   332→                    target=target,
   333→                    base_url=base_url,
   334→                    timeout=timeout,
   335→                    verify_tls=verify_tls,
   336→                    limiter=limiter,
   337→                )
   338→                if not sandbox_result.get("success"):
   339→                    _module_health.record_failure(mod_id)
   340→                    logger.error(
   341→                        "Plugin module '%s' blocked by sandbox: %s",
   342→                        mod_id, sandbox_result.get("error", "unknown"),
   343→                    )
   344→                    return
   345→            except Exception:
   346→                logger.debug("PluginSandbox check for '%s' failed, running unsandboxed", mod_id, exc_info=True)
   347→
   348→        async with semaphore:
   349→            self._emit(ScanEvent(type="MODULE_START", module_id=mod_id))
   350→            t0 = time.monotonic()
   351→            try:
   352→                if self._use_async and hasattr(runner, "async_run"):
   353→                    result = await runner.async_run(
   354→                        target=target,
   355→                        base_url=base_url,
   356→                        timeout=timeout,
   357→                        verify_tls=verify_tls,
   358→                        limiter=limiter,
   359→                    )
   360→                else:
   361→                    result = await asyncio.to_thread(
   362→                        runner,
   363→                        target=target,
   364→                        base_url=base_url,
   365→                        timeout=timeout,
   366→                        verify_tls=verify_tls,
   367→                        limiter=limiter,
   368→                    )
   369→
   370→                duration = time.monotonic() - t0
   371→
   372→                # Dispatch based on return type.
   373→                # vibesec returns (findings, score, grade, badge_md).
   374→                # All other modules return list[Finding].
   375→                if isinstance(result, tuple) and len(result) == 4:
   376→                    mod_findings, score, grade, badge_md = result
   377→                    vibesec_state["score"] = score
   378→                    vibesec_state["grade"] = grade
   379→                    async with findings_lock:
   380→                        all_findings.extend(mod_findings)
   381→                    module_results[mod_id] = {
   382→                        "findings": [f.to_dict() for f in mod_findings],
   383→                        "score": score,
   384→                        "grade": grade,
   385→                        "badge": badge_md,
   386→                    }
   387→                else:
   388→                    mod_findings = result
   389→                    async with findings_lock:
   390→                        all_findings.extend(mod_findings)
   391→                        for f in mod_findings:
   392→                            self._emit(
   393→                                ScanEvent(type="FINDING", module_id=mod_id, finding=f)
   394→                            )
   395→                            try:
   396→                                from .plugins import HookManager
   397→                                HookManager.fire("post_finding", finding=f)
   398→                            except Exception:
   399→                                logger.debug("Hook 'post_finding' error", exc_info=True)
   400→                    module_results[mod_id] = {
   401→                        "findings": [f.to_dict() for f in mod_findings],
   402→                        "count": len(mod_findings),
   403→                    }
   404→
   405→                _module_health.record_success(mod_id)
   406→
   407→                self._emit(
   408→                    ScanEvent(
   409→                        type="MODULE_COMPLETE",
   410→                        module_id=mod_id,
   411→                        findings_count=len(mod_findings),
   412→                        duration=duration,
   413→                    )
   414→                )
   415→            except Exception as exc:
   416→                duration = time.monotonic() - t0
   417→                _module_health.record_failure(mod_id)
   418→                logger.error("Module '%s' failed: %s", mod_id, exc, exc_info=True)
   419→                # Emit a finding-like error so the scan doesn't silently
   420→                # swallow module failures.
   421→                err_finding = Finding(
   422→                    title=f"Module {mod_id} error: {exc}",
   423→                    severity="info",
   424→                    category="engine_error",
   425→                    module=mod_id,
   426→                    description=f"Module {mod_id} raised an exception during execution.",
   427→                    evidence=str(exc),
   428→                    asset=target,
   429→                )
   430→                async with findings_lock:
   431→                    all_findings.append(err_finding)
   432→                module_results[mod_id] = {
   433→                    "findings": [err_finding.to_dict()],
   434→                    "count": 1,
   435→                    "error": str(exc),
   436→                }
   437→                self._emit(
   438→                    ScanEvent(
   439→                        type="MODULE_COMPLETE",
   440→                        module_id=mod_id,
   441→                        findings_count=1,
   442→                        duration=duration,
   443→                    )
   444→                )
   445→
   446→    # -- main async entry point --------------------------------------------
   447→
   448→    async def run(
   449→        self,
   450→        target: str,
   451→        modules: Optional[List[str]] = None,
   452→        all_modules: bool = False,
   453→        timeout: int = 10,
   454→        verify_tls: bool = True,
   455→        rate_limit: Optional[float] = None,
   456→        is_local: bool = False,
   457→    ) -> ReconProResult:
   458→        """Run a scan against *target* with concurrent module execution.
   459→
   460→        This is the async counterpart of :func:`scanner.scan` and produces
   461→        an identical ``ReconProResult``.  The only difference is that modules
   462→        are executed concurrently (up to ``self._concurrency`` at a time)
   463→        instead of sequentially.
   464→
   465→        Parameters
   466→        ----------
   467→        target : str
   468→            Domain, URL, or directory path (for local scans).
   469→        modules : list[str] | None
   470→            Explicit module IDs.  ``None`` → defaults.
   471→        all_modules : bool
   472→            Run every registered module for the scan type.
   473→        timeout : int
   474→            Per-request timeout in seconds.
   475→        verify_tls : bool
   476→            Whether to verify TLS certificates.
   477→        rate_limit : float | None
   478→            Requests-per-second cap.  Falls back to the engine default.
   479→        is_local : bool
   480→            If ``True``, use ``LOCAL_MODULES`` instead of
   481→            ``MODULE_REGISTRY`` (same as ``audit_scan``).
   482→
   483→        Returns
   484→        -------
   485→        ReconProResult
   486→            Identical shape to what ``scanner.scan`` returns.
   487→        """
   488→        effective_rate = rate_limit if rate_limit is not None else self._default_rate_limit
   489→
   490→        # Resolve modules list.
   491→        if is_local:
   492→            mods = self._resolve_local_modules(modules, all_modules)
   493→        else:
   494→            mods = self._resolve_remote_modules(modules, all_modules)
   495→
   496→        base_url = target if target.startswith("http") else f"https://{target}"
   497→        host = target.replace("https://", "").replace("http://", "").split("/")[0]
   498→
   499→        self._emit(ScanEvent(type="SCAN_START", target=target, modules=list(mods)))
   500→        try:
   501→            from .plugins import HookManager
   502→            HookManager.fire("pre_scan", target=target, modules=mods)
   503→        except Exception:
   504→            logger.debug("Hook 'pre_scan' error", exc_info=True)
   505→        scan_t0 = time.monotonic()
   506→
   507→        semaphore = asyncio.Semaphore(self._concurrency)
   508→        findings_lock = asyncio.Lock()
   509→        all_findings: List[Finding] = []
   510→        module_results: Dict[str, Dict[str, Any]] = {}
   511→        vibesec_state: Dict[str, Any] = {}
   512→
   513→        # Create rate limiter for modules that accept it.
   514→        from .http_layer import RateLimiter
   515→        limiter = RateLimiter(effective_rate)
   516→
   517→        # Build coroutine tasks for each module.
   518→        tasks: List[asyncio.Task[None]] = []
   519→        for mod_id in mods:
   520→            if is_local:
   521→                entry = LOCAL_MODULES.get(mod_id)
   522→            else:
   523→                entry = MODULE_REGISTRY.get(mod_id)
   524→
   525→            if entry is None:
   526→                continue
   527→
   528→            if not _module_health.is_healthy(mod_id):
   529→                if _module_health.is_quarantined(mod_id):
   530→                    logger.warning("Module '%s' is quarantined, skipping", mod_id)
   531→                else:
   532→                    logger.warning("Module '%s' is unhealthy (circuit breaker open), skipping", mod_id)
   533→                continue
   534→
   535→            # Auto-recovery probe: quarantine expired, try one execution
   536→            _is_probe = _module_health.is_probe(mod_id)
   537→            if _is_probe:
   538→                logger.info("Module '%s' running as probe (auto-recovery)", mod_id)
   539→
   540→            runner = entry.get("runner")
   541→            if runner is None:
   542→                # Special case: vibesec — lazy import.
   543→                if mod_id == "vibesec" and not is_local:
   544→                    from .modules.vibesec import run_vibesec  # type: ignore[no-redef]
   545→                    runner = run_vibesec
   546→                else:
   547→                    continue
   548→
   549→            tasks.append(
   550→                asyncio.create_task(
   551→                    self._run_module(
   552→                        mod_id=mod_id,
   553→                        runner=runner,
   554→                        target=target,
   555→                        base_url=base_url,
   556→                        timeout=timeout,
   557→                        verify_tls=verify_tls,
   558→                        limiter=limiter,
   559→                        semaphore=semaphore,
   560→                        findings_lock=findings_lock,
   561→                        all_findings=all_findings,
   562→                        module_results=module_results,
   563→                        vibesec_state=vibesec_state,
   564→                        _entry=entry,
   565→                    )
   566→                )
   567→            )
   568→
   569→        # Run all module tasks concurrently (bounded by semaphore).
   570→        if tasks:
   571→            await asyncio.gather(*tasks, return_exceptions=True)
   572→
   573→        scan_duration = time.monotonic() - scan_t0
   574→
   575→        vibesec_score = vibesec_state.get("score")
   576→        vibesec_grade = vibesec_state.get("grade")
   577→
   578→        result = self._build_result(
   579→            target=host if not is_local else (target if target != "." else "local-audit"),
   580→            mods=mods,
   581→            all_findings=all_findings,
   582→            module_results=module_results,
   583→            vibesec_score=vibesec_score,
   584→            vibesec_grade=vibesec_grade,
   585→        )
   586→
   587→        # ── Post-scan Secret Detection & Tamper-Evident Logging ──
   588→        secret_findings: List[Dict[str, Any]] = []
   589→        if all_findings:
   590→            try:
   591→                from .security_hardening import SecretsManager, TamperEvidenceLogger
   592→                secrets_mgr = SecretsManager()
   593→                for f in all_findings:
   594→                    finding_text = json.dumps(f.to_dict(), default=str)
   595→                    sf = secrets_mgr.scan_for_secrets(finding_text, source=f"finding:{f.module}")
   596→                    secret_findings.extend(sf)
   597→                if secret_findings:
   598→                    logger.warning(
   599→                        "Secret detection found %d potential secret(s) in scan findings",
   600→                        len(secret_findings),
   601→                    )
   602→            except Exception as sec_err:
   603→                logger.debug("Post-scan secret detection error: %s", sec_err, exc_info=True)
   604→
   605→            # Tamper-evident logging of scan result
   606→            try:
   607→                from .security_hardening import TamperEvidenceLogger
   608→                tamper_logger = TamperEvidenceLogger(module_name="engine")
   609→                tamper_logger.log(
   610→                    level="INFO",
   611→                    event_type="SCAN_COMPLETE",
   612→                    target=target,
   613→                    details={
   614→                        "findings_count": len(all_findings),
   615→                        "score": result.total_score,
   616→                        "grade": result.grade,
   617→                        "modules_run": mods,
   618→                        "secrets_found": len(secret_findings),
   619→                    },
   620→                )
   621→            except Exception as te_err:
   622→                logger.debug("Tamper-evident logging failed: %s", te_err)
   623→
   624→        # ── Post-scan Intelligence Pipeline ──────────────────────
   625→        intelligence_data = None
   626→        if all_findings and self._run_intelligence:
   627→            try:
   628→                # Apply prompt defense to finding text before AI analyst processes it
   629→                # Always runs when intelligence pipeline is active
   630→                try:
   631→                    from .prompt_defense import PromptDefense
   632→                    _pd = PromptDefense(enable_logging=False)
   633→                    _sanitized_findings = []
   634→                    for _f in all_findings:
   635→                        _title = _f.title or ""
   636→                        _desc = _f.description or ""
   637→                        _evidence = _f.evidence or ""
   638→                        _pd_result = _pd.scan(f"{_title} {_desc} {_evidence}", source="engine-intel")
   639→                        if _pd_result.is_safe:
   640→                            _sanitized_findings.append(_f)
   641→                        else:
   642→                            # Replace description with sanitized version to prevent prompt injection
   643→                            logger.warning(
   644→                                "Prompt defense sanitized finding '%s': %s",
   645→                                _f.title, _pd_result.max_severity.value,
   646→                            )
   647→                            _safe_f = Finding(
   648→                                title=_f.title,
   649→                                severity=_f.severity,
   650→                                category=_f.category,
   651→                                module=_f.module,
   652→                                description=_pd_result.sanitized_input,
   653→                                evidence=_f.evidence,
   654→                                asset=_f.asset,
   655→                                points_deducted=_f.points_deducted,
   656→                                remediation=_f.remediation,
   657→                                references=_f.references,
   658→                            )
   659→                            _sanitized_findings.append(_safe_f)
   660→                except Exception:
   661→                    logger.debug("Prompt defense check skipped", exc_info=True)
   662→                    _sanitized_findings = all_findings
   663→
   664→                from .intelligence_pipeline import IntelligencePipeline
   665→                intel_pipeline = IntelligencePipeline(
   666→                    enable_ai_analyst=True,
   667→                    enable_attack_graph=True,
   668→                    enable_threat_intel=True,
   669→                )
   670→                intel_result = intel_pipeline.analyze(_sanitized_findings, result.to_dict())
   671→                intelligence_data = intel_result.to_dict()
   672→
   673→                # Scan AI-generated intelligence data for secrets
   674→                try:
   675→                    _intel_text = json.dumps(intelligence_data, default=str)
   676→                    from .security_hardening import SecretsManager as _SM2
   677→                    _intel_sm = _SM2()
   678→                    _intel_sf = _intel_sm.scan_for_secrets(_intel_text, source="intelligence_pipeline")
   679→                    if _intel_sf:
   680→                        logger.warning(
   681→                            "Secret detection found %d potential secret(s) in AI-generated content",
   682→                            len(_intel_sf),
   683→                        )
   684→                        secret_findings.extend(_intel_sf)
   685→                except Exception:
   686→                    pass
   687→
   688→                # Inject intelligence into result
   689→                result.intelligence = intelligence_data
   690→                logger.info(
   691→                    "Intelligence pipeline completed: risk=%.1f exposure=%.1f chains=%d cves=%d",
   692→                    intel_result.executive_risk_score,
   693→                    intel_result.exposure_score,
   694→                    len(intel_result.attack_chains),
   695→                    len(intel_result.cve_matches),
   696→                )
   697→            except Exception as e:
   698→                logger.warning("Intelligence pipeline error: %s", e, exc_info=True)
   699→
   700→        self._emit(
   701→            ScanEvent(
   702→                type="SCAN_COMPLETE",
   703→                target=target,
   704→                result=result,
   705→                duration=scan_duration,
   706→            )
   707→        )
   708→        try:
   709→            from .plugins import HookManager
   710→            HookManager.fire("post_scan", target=target, result=result, duration=scan_duration)
   711→        except Exception:
   712→            logger.debug("Hook 'post_scan' error", exc_info=True)
   713→
   714→        # ── Post-scan Engineering Pipeline (if enabled) ──────────────
   715→        if self._run_engineering and all_findings:
   716→            try:
   717→                from .engineering_workflow import ContinuousEngineeringOrchestrator
   718→                eng_orchestrator = ContinuousEngineeringOrchestrator(self._engineering_repo_path)
   719→                eng_result = eng_orchestrator.run_full_cycle(self._engineering_repo_path)
   720→                # Store engineering result metadata in result for later retrieval
   721→                result.engineering = eng_result.to_dict()
   722→                logger.info(
   723→                    "Engineering pipeline completed: status=%s stages=%d duration=%.2fs",
   724→                    eng_result.overall_status,
   725→                    len(eng_result.stages),
   726→                    eng_result.total_duration_s,
   727→                )
   728→            except Exception as e:
   729→                logger.warning("Engineering pipeline error: %s", e, exc_info=True)
   730→
   731→        # ── Post-engineering Quality Intelligence ──
   732→        if self._run_quality or self._run_engineering:
   733→            try:
   734→                from .quality_intelligence import QualityIntelligence
   735→                qi = QualityIntelligence(repository_path=self._engineering_repo_path)
   736→                qi.analyze_repository_quality(self._engineering_repo_path)
   737→                quality_score = qi.get_quality_score()
   738→                quality_dims = qi.get_quality_dimensions()
   739→                result.quality = {
   740→                    "composite_score": quality_score,
   741→                    "dimensions": quality_dims,
   742→                }
   743→                logger.info(
   744→                    "Quality intelligence completed: score=%.1f", quality_score,
   745→                )
   746→            except Exception as e:
   747→                logger.debug("Quality intelligence skipped: %s", e)
   748→
   749→        return result

   753→    def scan_one(
   754→        self,
   755→        target: str,
   756→        modules: Optional[List[str]] = None,
   757→        timeout: int = 10,
   758→        verify_tls: bool = True,
   759→        rate_limit: Optional[float] = None,
   760→        is_local: bool = False,
   761→        run_engineering: bool = False,
   762→        engineering_repo_path: str = ".",
   763→        run_intelligence: bool = True,
   764→        run_quality: bool = False,
   765→        run_defense: bool = False,
   766→    ) -> ReconProResult:
   767→        """Synchronous wrapper around :meth:`run`.
   768→
   769→        Creates a fresh event loop (or reuses the running one) so callers
   770→        don't need to care about asyncio.
   771→
   772→        Parameters are identical to :meth:`run` except ``all_modules`` is
   773→        not exposed — pass an explicit ``modules`` list instead.
   774→        """
   775→        # Override instance-level settings if caller provides them.
   776→        if run_engineering:
   777→            self._run_engineering = True
   778→        if engineering_repo_path != ".":
   779→            self._engineering_repo_path = engineering_repo_path
   780→        if not run_intelligence:
   781→            self._run_intelligence = False
   782→        if run_quality:
   783→            self._run_quality = True
   784→        if run_defense:
   785→            self._run_defense = True
   786→        try:
   787→            loop = asyncio.get_running_loop()
   788→        except RuntimeError:
   789→            loop = None
   790→
   791→        if loop is not None and loop.is_running():
   792→            # Already inside an async context — use nest_asyncio-safe
   793→            # approach or fallback to a new thread.
   794→            import concurrent.futures
   795→
   796→            with concurrent.futures.ThreadPoolExecutor(max_workers=1) as pool:
   797→                future = pool.submit(
   798→                    asyncio.run,
   799→                    self.run(
   800→                        target=target,
   801→                        modules=modules,
   802→                        timeout=timeout,
   803→                        verify_tls=verify_tls,
   804→                        rate_limit=rate_limit,
   805→                        is_local=is_local,
   806→                    ),
   807→                )
   808→                return future.result()
   809→
   810→        return asyncio.run(
   811→            self.run(
   812→                target=target,
   813→                modules=modules,
   814→                timeout=timeout,
   815→                verify_tls=verify_tls,
   816→                rate_limit=rate_limit,
   817→                is_local=is_local,
   818→            )
   819→        )

   822→# ── Module Health Tracking ────────────────────────────────────────────
   823→
   824→
   825→class ModuleHealthState:
   826→    """Tracks health of scanner modules with circuit-breaker, quarantine, and scoring."""
   827→
   828→    QUARANTINE_CONSECUTIVE_FAILURES: int = 5
   829→    QUARANTINE_DURATION: float = 600.0  # 10 minutes
   830→    HEALTH_WINDOW: int = 20  # last N executions for scoring
   831→
   832→    def __init__(self, failure_threshold: int = 3, cooldown_seconds: float = 60.0):
   833→        self._failure_counts: Dict[str, int] = {}
   834→        self._consecutive_failures: Dict[str, int] = {}
   835→        self._last_failure_time: Dict[str, float] = {}
   836→        self._quarantine_until: Dict[str, float] = {}  # module_id -> monotonic timestamp
   837→        self._quarantine_probe: Dict[str, bool] = {}  # module_id -> whether this is a probe run
   838→        self._history: Dict[str, List[bool]] = {}  # module_id -> [True=success, False=failure]
   839→        self._failure_threshold = failure_threshold
   840→        self._cooldown_seconds = cooldown_seconds
   841→
   842→    def record_success(self, module_id: str) -> None:
   843→        self._failure_counts.pop(module_id, None)
   844→        self._last_failure_time.pop(module_id, None)
   845→        self._consecutive_failures[module_id] = 0
   846→        self._quarantine_until.pop(module_id, None)
   847→        self._quarantine_probe.pop(module_id, None)
   848→        self._history.setdefault(module_id, []).append(True)
   849→        if len(self._history[module_id]) > self.HEALTH_WINDOW:
   850→            self._history[module_id] = self._history[module_id][-self.HEALTH_WINDOW:]
   851→
   852→    def record_failure(self, module_id: str) -> None:
   853→        self._failure_counts[module_id] = self._failure_counts.get(module_id, 0) + 1
   854→        self._last_failure_time[module_id] = time.monotonic()
   855→        self._consecutive_failures[module_id] = self._consecutive_failures.get(module_id, 0) + 1
   856→        self._history.setdefault(module_id, []).append(False)
   857→        if len(self._history[module_id]) > self.HEALTH_WINDOW:
   858→            self._history[module_id] = self._history[module_id][-self.HEALTH_WINDOW:]
   859→
   860→        # Check if should quarantine
   861→        if self._consecutive_failures.get(module_id, 0) >= self.QUARANTINE_CONSECUTIVE_FAILURES:
   862→            self._quarantine_until[module_id] = time.monotonic() + self.QUARANTINE_DURATION
   863→            logger.warning(
   864→                "Module '%s' quarantined for %.0fs after %d consecutive failures",
   865→                module_id, self.QUARANTINE_DURATION, self._consecutive_failures[module_id],
   866→            )
   867→
   868→    def is_quarantined(self, module_id: str) -> bool:
   869→        """Check if a module is currently in quarantine."""
   870→        until = self._quarantine_until.get(module_id)
   871→        if until is None:
   872→            return False
   873→        if time.monotonic() >= until:
   874→            # Quarantine expired — allow one probe execution
   875→            self._quarantine_until.pop(module_id, None)
   876→            self._quarantine_probe[module_id] = True
   877→            logger.info("Module '%s' quarantine expired, allowing probe execution", module_id)
   878→            return False
   879→        return True
   880→
   881→    def is_probe(self, module_id: str) -> bool:
   882→        """Check if the next execution of this module is a probe (auto-recovery test)."""
   883→        return self._quarantine_probe.pop(module_id, False)
   884→
   885→    def health_score(self, module_id: str) -> float:
   886→        """Return health score 0.0-1.0 based on last 20 executions."""
   887→        hist = self._history.get(module_id, [])
   888→        if not hist:
   889→            return 1.0  # no data = assumed healthy
   890→        return sum(1 for x in hist if x) / len(hist)
   891→
   892→    def is_healthy(self, module_id: str) -> bool:
   893→        """Check if a module is healthy enough to run.
   894→
   895→        Returns False if quarantined or if circuit breaker is open.
   896→        """
   897→        # Check quarantine first
   898→        if self.is_quarantined(module_id):
   899→            return False
   900→
   901→        count = self._failure_counts.get(module_id, 0)
   902→        if count < self._failure_threshold:
   903→            return True
   904→        # In cooldown?
   905→        last_fail = self._last_failure_time.get(module_id, 0)
   906→        if time.monotonic() - last_fail > self._cooldown_seconds:
   907→            # Cooldown expired, reset
   908→            self.record_success(module_id)
   909→            return True
   910→        return False
   911→
   912→    def get_all_module_health(self) -> Dict[str, Dict[str, Any]]:
   913→        """Return comprehensive health dict for every tracked module.
   914→
   915→        Returns
   916→        -------
   917→        dict[str, dict]
   918→            Keys are module IDs.  Values contain ``health_score``,
   919→            ``is_healthy``, ``is_quarantined``, ``failure_count``,
   920→            ``consecutive_failures``, ``quarantine_remaining``.
   921→        """
   922→        result: Dict[str, Dict[str, Any]] = {}
   923→        all_known = set(self._failure_counts) | set(self._history) | set(self._quarantine_until)
   924→        for mid in all_known:
   925→            until = self._quarantine_until.get(mid)
   926→            quarantine_remaining = max(0.0, until - time.monotonic()) if until else 0.0
   927→            result[mid] = {
   928→                "health_score": round(self.health_score(mid), 3),
   929→                "is_healthy": self.is_healthy(mid),
   930→                "is_quarantined": until is not None and time.monotonic() < until,
   931→                "failure_count": self._failure_counts.get(mid, 0),
   932→                "consecutive_failures": self._consecutive_failures.get(mid, 0),
   933→                "quarantine_remaining_s": round(quarantine_remaining, 1),
   934→                "is_probe": mid in self._quarantine_probe,
   935→            }
   936→        return result
   937→
   938→    def get_status(self) -> Dict[str, Dict[str, Any]]:
   939→        """Return health status for all tracked modules (legacy compat)."""
   940→        return {
   941→            mid: {
   942→                "failure_count": count,
   943→                "is_healthy": self.is_healthy(mid),
   944→                "in_cooldown": not self.is_healthy(mid),
   945→            }
   946→            for mid, count in self._failure_counts.items()
   947→        }
   948→
   949→
   950→# Global module health tracker
   951→_module_health = ModuleHealthState()

  1076→async def concurrent_scan(
  1077→    targets: List[str],
  1078→    modules: Optional[List[str]] = None,
  1079→    all_modules: bool = False,
  1080→    timeout: int = 10,
  1081→    verify_tls: bool = True,
  1082→    rate_limit: float = 50.0,
  1083→    max_concurrency: int = 10,
  1084→    is_local: bool = False,
  1085→    event_callback: Optional[Callable[[ScanEvent], None]] = None,
  1086→) -> Dict[str, ReconProResult]:
  1087→    """Scan multiple targets concurrently with bounded total parallelism.
  1088→
  1089→    Spawns one ``ScanEngine`` per target but shares a single
  1090→    ``asyncio.Semaphore`` so the total number of in-flight modules
  1091→    across *all* targets never exceeds ``max_concurrency``.
  1092→
  1093→    Parameters
  1094→    ----------
  1095→    targets : list[str]
  1096→        Domains / URLs to scan.
  1097→    modules, all_modules, timeout, verify_tls, rate_limit
  1098→        Forwarded to each per-target scan.
  1099→    max_concurrency : int
  1100→        Upper bound on simultaneous module executions across all targets.
  1101→    is_local : bool
  1102→        If ``True``, run local audit modules instead of remote ones.
  1103→    event_callback : callable | None
  1104→        Receives ``ScanEvent`` from every target scan.
  1105→
  1106→    Returns
  1107→    -------
  1108→    dict[str, ReconProResult]
  1109→        Mapping of target → result for every target.
  1110→    """
  1111→    results: Dict[str, ReconProResult] = {}
  1112→    lock = asyncio.Lock()
  1113→
  1114→    async def _scan_one(t: str) -> None:
  1115→        engine = ScanEngine(
  1116→            event_callback=event_callback,
  1117→            concurrency=max_concurrency,
  1118→            rate_limit=rate_limit,
  1119→            use_async=True,
