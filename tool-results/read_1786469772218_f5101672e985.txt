   234→    @classmethod
   235→    def load_latest(cls, directory: Optional[Path] = None) -> Optional["EngineeringCycleResult"]:
   236→        """Load the most recent cycle result from JSON files.
   237→
   238→        Args:
   239→            directory: Override directory.
   240→
   241→        Returns:
   242→            EngineeringCycleResult or None if no files found.
   243→        """
   244→        search_dir = directory or ENGINEERING_CYCLES_DIR
   245→        if not search_dir.exists():
   246→            return None
   247→        try:
   248→            files = sorted(search_dir.glob("cycle_*.json"), key=os.path.getmtime)
   249→            if not files:
   250→                return None
   251→            with open(files[-1], "r", encoding="utf-8") as f:
   252→                data = json.load(f)
   253→            result = cls()
   254→            result.cycle_id = data.get("cycle_id", "")
   255→            result.repository_path = data.get("repository_path", "")
   256→            result.mode = data.get("mode", "full")
   257→            result.overall_status = data.get("overall_status", "pass")
   258→            result.started_at = data.get("started_at", "")
   259→            result.finished_at = data.get("finished_at", "")
   260→            result.total_duration_s = data.get("total_duration_s", 0.0)
   261→            result.metrics = data.get("metrics", {})
   262→            result.recommendations_count = data.get("recommendations_count", 0)
   263→            result.fix_proposals_count = data.get("fix_proposals_count", 0)
   264→            result.anomaly_count = data.get("anomaly_count", 0)
   265→            result.regression_count = data.get("regression_count", 0)
   266→            for sdata in data.get("stages", []):
   267→                sr = StageResult(
   268→                    stage=EngineeringStage(sdata.get("stage", "unknown")),
   269→                    status=sdata.get("status", "pass"),
   270→                    duration_s=sdata.get("duration_s", 0.0),
   271→                    data=sdata.get("data", {}),
   272→                    error=sdata.get("error", ""),
   273→                )
   274→                result.stages.append(sr)
   275→            return result
   276→        except (OSError, json.JSONDecodeError, KeyError, ValueError):
   277→            logger.warning("Failed to load latest cycle result", exc_info=True)
   278→            return None

   409→    # ── Stage execution helpers ────────────────────────────────────────
   410→
   411→    def _execute_stage(
   412→        self,
   413→        stage: EngineeringStage,
   414→        stage_fn: Any,
   415→        result: EngineeringCycleResult,
   416→    ) -> StageResult:
   417→        """Execute a single stage with timing and error handling.
   418→
   419→        Args:
   420→            stage: The stage enum value.
   421→            stage_fn: Callable that performs the stage work.
   422→            result: The cycle result to append this stage result to.
   423→
   424→        Returns:
   425→            StageResult with outcome data.
   426→        """
   427→        stage_result = StageResult(stage=stage)
   428→        t0 = time.monotonic()
   429→        try:
   430→            data = stage_fn()
   431→            if isinstance(data, dict):
   432→                stage_result.data = data
   433→            elif data is not None:
   434→                stage_result.data = {"result": data}
   435→            stage_result.status = "pass"
   436→            logger.info("Stage %s completed successfully", stage.value)
   437→        except Exception as exc:
   438→            stage_result.status = "fail"
   439→            stage_result.error = f"{type(exc).__name__}: {exc}"
   440→            logger.warning(
   441→                "Stage %s failed (continuing): %s", stage.value, exc, exc_info=True
   442→            )
   443→        finally:
   444→            stage_result.duration_s = time.monotonic() - t0
   445→        result.stages.append(stage_result)
   446→        return stage_result
   447→
   448→    def _compute_overall_status(self, result: EngineeringCycleResult) -> str:
   449→        """Compute overall pass/fail/warn from stage statuses.
   450→
   451→        Rules:
   452→            - Any 'fail' → overall 'fail'
   453→            - Any 'warn' (and no 'fail') → overall 'warn'
   454→            - All 'pass' or 'skipped' → overall 'pass'
   455→        """
   456→        has_fail = any(s.status == "fail" for s in result.stages)
   457→        has_warn = any(s.status == "warn" for s in result.stages)
   458→        if has_fail:
   459→            return "fail"
   460→        if has_warn:
   461→            return "warn"
   462→        return "pass"

   787→    def run_full_cycle(
   788→        self,
   789→        repository_path: Optional[str] = None,
   790→    ) -> EngineeringCycleResult:
   791→        """Execute the complete engineering cycle (all 14 stages, a–m).
   792→
   793→        Steps:
   794→            a. Digital Twin: capture state, detect anomalies
   795→            b. Repository Memory: recall previous knowledge
   796→            c. Auto Validation: run full validation pipeline
   797→            d. Diagnostics: health checks
   798→            e. Drift Detection: detect code drift
   799→            f. Benchmark: quick benchmarks, performance regressions
   800→            g. Regression Intelligence: detect regressions
   801→            h. Learning: learn from collected data
   802→            i. Recommendations: generate from findings
   803→            j. Auto Fix: propose fixes
   804→            k. Quality Gate: enforce thresholds
   805→            l. Reporting: comprehensive report
   806→            m. Repository Memory: persist results
   807→
   808→        Args:
   809→            repository_path: Override repository path for this cycle.
   810→
   811→        Returns:
   812→            EngineeringCycleResult with all stage results.
   813→        """
   814→        if repository_path:
   815→            self._repo_path = str(repository_path)
   816→            self._repo_resolved = Path(repository_path).resolve()
   817→
   818→        result = EngineeringCycleResult(
   819→            cycle_id=datetime.now(timezone.utc).strftime("%Y%m%d%H%M%S%f"),
   820→            repository_path=self._repo_path,
   821→            mode="full",
   822→            started_at=datetime.now(timezone.utc).isoformat(),
   823→        )
   824→
   825→        self._last_result = result
   826→        cycle_start = time.monotonic()
   827→        logger.info("Starting full engineering cycle %s for %s", result.cycle_id, self._repo_path)
   828→
   829→        for stage in _FULL_CYCLE_STAGES:
   830→            method_name = self._STAGE_DISPATCH.get(stage)
   831→            if method_name is None:
   832→                logger.warning("No dispatch for stage %s — skipping", stage.value)
   833→                result.stages.append(StageResult(stage=stage, status="skipped"))
   834→                continue
   835→            stage_fn = getattr(self, method_name, None)
   836→            if stage_fn is None:
   837→                result.stages.append(StageResult(stage=stage, status="skipped"))
   838→                continue
   839→            sr = self._execute_stage(stage, stage_fn, result)
   840→            # Track aggregate counts
   841→            if sr.status == "pass" and sr.data:
   842→                if sr.stage == EngineeringStage.DIGITAL_TWIN_ANOMALY:
   843→                    result.anomaly_count = sr.data.get("anomaly_count", 0)
   844→                if sr.stage == EngineeringStage.REGRESSION_INTELLIGENCE:
   845→                    result.regression_count = sr.data.get("regression_count", 0)
   846→                if sr.stage == EngineeringStage.RECOMMENDATIONS:
   847→                    result.recommendations_count = sr.data.get("recommendations_count", 0)
   848→                if sr.stage == EngineeringStage.AUTO_FIX:
   849→                    result.fix_proposals_count = sr.data.get("fix_proposals_count", 0)
   850→                if sr.stage == EngineeringStage.QUALITY_INTELLIGENCE and sr.data:
   851→                    result.quality_score = sr.data.get("composite_score", 0.0)
   852→                    result.quality_grade = sr.data.get("grade", "")
   853→
   854→        result.finished_at = datetime.now(timezone.utc).isoformat()
   855→        result.total_duration_s = time.monotonic() - cycle_start
   856→        result.overall_status = self._compute_overall_status(result)
   857→
   858→        # Calculate engineering score for the cycle
   859→        result.engineering_score = self._compute_engineering_score(result)
   860→
   861→        result.metrics = {
   862→            "total_stages": len(result.stages),
   863→            "passed": sum(1 for s in result.stages if s.status == "pass"),
   864→            "failed": sum(1 for s in result.stages if s.status == "fail"),
   865→            "warned": sum(1 for s in result.stages if s.status == "warn"),
   866→            "skipped": sum(1 for s in result.stages if s.status == "skipped"),
   867→        }
   868→
   869→        # Persist results
   870→        result.save()
   871→
   872→        self._cycle_count += 1
   873→        logger.info(
   874→            "Engineering cycle %s complete: status=%s duration=%.2fs",
   875→            result.cycle_id,
   876→            result.overall_status,
   877→            result.total_duration_s,
   878→        )
   879→        return result
   880→
   881→    def _compute_engineering_score(self, result: EngineeringCycleResult) -> float:
   882→        """Calculate a composite engineering score (0-100) for the cycle.
   883→
   884→        Formula: 40% module_health + 30% quality_history + 30% scan_reliability
   885→        Falls back to module health if no quality history exists.
   886→        """
   887→        total = len(result.stages)
   888→        if total == 0:
   889→            return 0.0
   890→
   891→        passed = sum(1 for s in result.stages if s.status == "pass")
   892→        failed = sum(1 for s in result.stages if s.status == "fail")
   893→        module_health = (passed / total) * 100.0
   894→
   895→        # Quality history: use quality intelligence score if available
   896→        quality_history = 0.0
   897→        has_quality = result.quality_score > 0
   898→        if has_quality:
   899→            quality_history = result.quality_score
   900→        else:
   901→            # Try to recall from memory
   902→            try:
   903→                mem = self._get_repository_memory()
   904→                qi_facts = mem.search(query_text="quality_score")
   905→                if qi_facts:
   906→                    quality_history = float(qi_facts[0].value) if isinstance(qi_facts[0].value, (int, float)) else 50.0
   907→                else:
   908→                    quality_history = module_health  # default to current scan
   909→            except Exception:
   910→                quality_history = module_health
   911→
   912→        # Scan reliability: inverse of failure rate, weighted by stage count
   913→        reliability = 100.0 if failed == 0 else max(0.0, (1.0 - failed / total) * 100.0)
   914→
   915→        score = 0.40 * module_health + 0.30 * quality_history + 0.30 * reliability
   916→        return round(max(0.0, min(100.0, score)), 1)
