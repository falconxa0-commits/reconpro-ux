   209→   209→// ── Helper: Map protocol crypto strings to ClassicalAlgorithm keys ─────

   211→   211→function matchClassicalAlgorithm(cryptoName: string): ClassicalAlgorithm | null {
   212→   212→  const name = cryptoName.toLowerCase();
   213→   213→  if (name.includes('rsa-1024') || name === 'rsa_1024') return CLASSICAL_ALGORITHMS.rsa_1024;
   214→   214→  if (name.includes('rsa-2048') || name === 'rsa_2048') return CLASSICAL_ALGORITHMS.rsa_2048;
   215→   215→  if (name.includes('rsa-4096') || name === 'rsa_4096') return CLASSICAL_ALGORITHMS.rsa_4096;
   216→   216→  if (name.includes('ecdsa') && (name.includes('p-256') || name.includes('p256'))) return CLASSICAL_ALGORITHMS.ecdsa_p256;
   217→   217→  if (name.includes('ecdsa') && (name.includes('p-384') || name.includes('p384'))) return CLASSICAL_ALGORITHMS.ecdsa_p384;
   218→   218→  if (name.includes('ecdh') && (name.includes('p-256') || name.includes('p256'))) return CLASSICAL_ALGORITHMS.ecdh_p256;
   219→   219→  if (name.includes('aes-128') || name.includes('aes128')) return CLASSICAL_ALGORITHMS.aes_128;
   220→   220→  if (name.includes('aes-256') || name.includes('aes256')) return CLASSICAL_ALGORITHMS.aes_256;
   221→   221→  if (name.includes('sha-256') || name.includes('sha256')) return CLASSICAL_ALGORITHMS.sha_256;
   222→   222→  if (name.includes('sha-384') || name.includes('sha384')) return CLASSICAL_ALGORITHMS.sha_384;
   223→   223→  if (name.includes('3des') || name.includes('3-des') || name.includes('tripledes')) {
   224→   224→    // 3DES is quantum-breakable via Grover's — treat as weak symmetric
   225→   225→    return { name: '3DES', type: 'symmetric', securityBits: 80, quantumBreakable: true, qubitsRequired: 4096, nistLevel: 0 };
   226→   226→  }
   227→   227→  if (name.includes('sha-1') || name.includes('sha1')) {
   228→   228→    return { name: 'SHA-1', type: 'hash', securityBits: 40, quantumBreakable: true, qubitsRequired: 1024, nistLevel: 0 };
   229→   229→  }
   230→   230→  if (name.includes('chacha20') || name.includes('chacha')) {
   231→   231→    return { name: 'ChaCha20-Poly1305', type: 'symmetric', securityBits: 256, quantumBreakable: false, groverReduction: 128, nistLevel: 2 };
   232→   232→  }
   233→   233→  if (name.includes('tls 1.2')) {
   234→   234→    return { name: 'TLS 1.2', type: 'key_exchange', securityBits: 112, quantumBreakable: true, qubitsRequired: 4096, nistLevel: 0 };
   235→   235→  }
   236→   236→  // Unknown — assume vulnerable for safety
   237→   237→  return null;
   238→   238→}

   242→   242→function generateRecommendations(cryptoName: string, orgType: OrganizationType): PQCRecommendation[] {
   243→   243→  const name = cryptoName.toLowerCase();
   244→   244→  const recs: PQCRecommendation[] = [];
   245→   245→  const isHighAssurance = orgType === 'central_bank' || orgType === 'clearing_house' || orgType === 'government';
   246→   246→
   247→   247→  // RSA key exchange → Kyber KEM
   248→   248→  if (name.includes('rsa') || name.includes('tls 1.2') || name.includes('3des')) {
   249→   249→    const target = isHighAssurance ? 'CRYSTALS-Kyber-1024' : 'CRYSTALS-Kyber-768';
   250→   250→    const nistLevel = isHighAssurance ? 5 : 3;
   251→   251→    recs.push({
   252→   252→      from: cryptoName,
   253→   253→      to: target,
   254→   254→      nistLevel,
   255→   255→      fips: 'FIPS 203',
   256→   256→      priority: name.includes('3des') || name.includes('sha-1') ? 'CRITICAL' : 'HIGH',
   257→   257→      effort: '6-12 months',
   258→   258→    });
   259→   259→  }
   260→   260→
   261→   261→  // ECDSA / ECDH → Kyber KEM + Dilithium signatures
   262→   262→  if (name.includes('ecdsa') || name.includes('ecdh')) {
   263→   263→    const kemTarget = isHighAssurance ? 'CRYSTALS-Kyber-1024' : 'CRYSTALS-Kyber-768';
   264→   264→    const sigTarget = isHighAssurance ? 'CRYSTALS-Dilithium-5' : 'CRYSTALS-Dilithium-3';
   265→   265→    const nistLevel = isHighAssurance ? 5 : 3;
   266→   266→
   267→   267→    if (name.includes('ecdh') || name.includes('key_exchange')) {
   268→   268→      recs.push({
   269→   269→        from: cryptoName,
   270→   270→        to: kemTarget,
   271→   271→        nistLevel,
   272→   272→        fips: 'FIPS 203',
   273→   273→        priority: 'HIGH',
   274→   274→        effort: '4-9 months',
   275→   275→      });
   276→   276→    } else {
   277→   277→      recs.push({
   278→   278→        from: cryptoName,
   279→   279→        to: sigTarget,
   280→   280→        nistLevel,
   281→   281→        fips: 'FIPS 204',
   282→   282→        priority: 'HIGH',
   283→   283→        effort: '6-12 months',
   284→   284→      });
   285→   285→    }
   286→   286→  }
   287→   287→
   288→   288→  // AES-128 → AES-256 (Grover mitigation)
   289→   289→  if (name.includes('aes-128') || name.includes('aes128')) {
   290→   290→    recs.push({
   291→   291→      from: cryptoName,
   292→   292→      to: 'AES-256-GCM',
   293→   293→      nistLevel: 3,
   294→   294→      fips: 'FIPS 140-3',
   295→   295→      priority: 'MODERATE',
   296→   296→      effort: '1-3 months',
   297→   297→    });
   298→   298→  }
   299→   299→
   300→   300→  // SHA-1 → SHA-384
   301→   301→  if (name.includes('sha-1') || name.includes('sha1')) {
   302→   302→    recs.push({
   303→   303→      from: cryptoName,
   304→   304→      to: 'SHA-384',
   305→   305→      nistLevel: 3,
   306→   306→      fips: 'FIPS 180-4',
   307→   307→      priority: 'CRITICAL',
   308→   308→      effort: '1-3 months',
   309→   309→    });
   310→   310→  }
   311→   311→
   312→   312→  return recs;
   313→   313→}

   317→   317→function buildProtocolComplianceGaps(protocolId: string, entry: ProtocolEntry): string[] {

   324→   324→    gaps.push(`Deprecated hash algorithm SHA-1 in use on ${entry.name}`);
   325→   325→  }

   327→   327→    gaps.push(`3DES block cipher deprecated and quantum-vulnerable on ${entry.name}`);
   328→   328→  }
   329→   329→  if (entry.currentCrypto.some(c => c.toLowerCase().includes('tls 1.2'))) {
   330→   330→    gaps.push(`TLS 1.2 lacks mandatory PQC key exchange on ${entry.name}`);
   331→   331→  }
   332→   332→  return gaps;

   379→   379→      'Require PQC compliance certificates from all critical third-party providers',

   382→   382→  };
   383→   383→
   384→   384→  // ── DORA (Digital Operational Resilience Act) ──
   385→   385→  const doraScore = hasVulnerableProtocols
   386→   386→    ? Math.round(Math.max(15, 55 - vulnerableRatio * 40))

   388→   388→  const dora: ComplianceFramework = {

   392→   392→      ? [
   393→   393→          'Digital operational resilience testing does not cover quantum attack scenarios',

   395→   395→          'Third-party ICT risk management lacks PQC requirements',
   396→   396→        ]
   397→   397→      : [],
   398→   398→    recommendations: [
   399→   399→      'Update DORA Article 25 threat-led penetration testing to include quantum scenarios',
   400→   400→      'Add PQC migration milestones to ICT business continuity plans',

   584→   584→    p.vulnerabilities.some(v => v.quantumBreakable && v.qubitsRequired && v.qubitsRequired < 4096),
   585→   585→  );

   592→   592→  } else if (hasCriticalVulns) {
   593→   593→    currentRisk = 'CRITICAL — Quantum adversary capable of decrypting transaction payloads and forging signatures on payment instructions.';
   594→   594→  } else if (hasVulnerable) {
   595→   595→    currentRisk = 'HIGH — Key exchange vulnerabilities enable "harvest now, decrypt later" attacks on archived transactions.';
   596→   596→  } else {
   597→   597→    currentRisk = 'LOW — All transaction paths protected with quantum-resistant cryptography.';
   598→   598→  }
   599→   599→
   600→   600→  const recommendations: string[] = [];
   601→   601→  if (hasCriticalVulns) {
   602→   602→    recommendations.push('Immediately deploy hybrid classical/PQC key exchange for all payment channels');
   603→   603→    recommendations.push('Implement transaction signing with ML-DSA (FIPS 204) for non-repudiation');
   604→   604→    recommendations.push('Establish quantum-safe key ceremony procedures for HSM migration');
   605→   605→  }
   606→   606→  if (hasVulnerable) {
   607→   607→    recommendations.push('Mandate PQC cipher suites for all inter-institutional communication');
   608→   608→    recommendations.push('Deploy PQC-protected message authentication codes (MAC) for transaction integrity');
   609→   609→  }
   610→   610→  recommendations.push('Implement crypto-agility to enable rapid algorithm rotation as NIST standards evolve');
   611→   611→  if (isHighAssurance) {
   612→   612→    recommendations.push('Establish bilateral PQC key agreement with all correspondent banking partners');
   613→   613→  }
   614→   614→
   615→   615→  return {
   616→   616→    currentRisk,
   617→   617→    pqcProtected: !hasVulnerable,
   618→   618→    recommendations,
   619→   619→  };
   620→   620→}

   642→   642→export function analyzePQCVault(params: PQCVaultParams): PQCVaultResult {
   643→   643→  const { organizationType, protocols: protocolIds, tlsData, complianceFrameworks } = params;
   644→   644→
   645→   645→  // ── 1. Analyze each protocol ──
   646→   646→  const protocolResults: ProtocolAnalysis[] = [];
   647→   647→  let totalVulns = 0;
   648→   648→  let criticalVulns = 0;
   649→   649→
   650→   650→  for (const pid of protocolIds) {
   651→   651→    const entry = PROTOCOL_ANALYSIS[pid];
   652→   652→    if (!entry) continue;
   653→   653→
   654→   654→    // Build vulnerabilities from current crypto
   655→   655→    const vulnerabilities: Vulnerability[] = [];
   656→   656→    for (const crypto of entry.currentCrypto) {
   657→   657→      const matched = matchClassicalAlgorithm(crypto);
   658→   658→      if (matched) {
   659→   659→        vulnerabilities.push({
   660→   660→          algorithm: matched.name,
   661→   661→          type: matched.type,
   662→   662→          quantumBreakable: matched.quantumBreakable,
   663→   663→          qubitsRequired: matched.qubitsRequired,
   664→   664→          nistLevel: matched.nistLevel,
   665→   665→        });
   666→   666→        totalVulns++;
   667→   667→        if (matched.quantumBreakable) criticalVulns++;
   668→   668→      } else {
   669→   669→        // Unknown algorithm — assume vulnerable
   670→   670→        vulnerabilities.push({
   671→   671→          algorithm: crypto,
   672→   672→          type: 'unknown',
   673→   673→          quantumBreakable: true,
   674→   674→          nistLevel: 0,
   675→   675→        });
   676→   676→        totalVulns++;
   677→   677→        criticalVulns++;
   678→   678→      }
   679→   679→    }
   680→   680→
   681→   681→    // Build PQC recommendations
   682→   682→    const pqcRecommendations: PQCRecommendation[] = [];
   683→   683→    const seenTargets = new Set<string>();
   684→   684→    for (const crypto of entry.currentCrypto) {
   685→   685→      const recs = generateRecommendations(crypto, organizationType);
   686→   686→      for (const rec of recs) {
   687→   687→        const key = `${rec.from}→${rec.to}`;
   688→   688→        if (!seenTargets.has(key)) {
   689→   689→          seenTargets.add(key);
   690→   690→          pqcRecommendations.push(rec);
   691→   691→        }
   692→   692→      }
   693→   693→    }
   694→   694→
   695→   695→    // Compliance gaps
   696→   696→    const complianceGaps = buildProtocolComplianceGaps(pid, entry);
   697→   697→
   698→   698→    protocolResults.push({
   699→   699→      id: pid,
   700→   700→      name: entry.name,
   701→   701→      quantumVulnerable: entry.quantumVulnerable,
   702→   702→      riskAssessment: entry.riskAssessment,
   703→   703→      currentCrypto: entry.currentCrypto,
   704→   704→      vulnerabilities,
   705→   705→      pqcRecommendations,
   706→   706→      complianceGaps,
   707→   707→    });
   708→   708→  }
   709→   709→
   710→   710→  // ── 2. Compute readiness score ──
   711→   711→  const isHighAssurance = organizationType === 'central_bank' || organizationType === 'clearing_house' || organizationType === 'government';
   712→   712→  const vulnerableProtocols = protocolResults.filter(p => p.quantumVulnerable).length;
   713→   713→  const totalProtocols = protocolResults.length || 1;
   714→   714→  const vulnRatio = vulnerableProtocols / totalProtocols;
   715→   715→
   716→   716→  // Base score deduction for vulnerable protocols
   717→   717→  let score = 100;
   718→   718→  score -= Math.round(vulnRatio * 60);
   719→   719→
   720→   720→  // Extra deduction for critical algorithms (SHA-1, 3DES)
   721→   721→  for (const proto of protocolResults) {
   722→   722→    if (proto.currentCrypto.some(c => c.toLowerCase().includes('sha-1'))) score -= 5;
   723→   723→    if (proto.currentCrypto.some(c => c.toLowerCase().includes('3des'))) score -= 8;
   724→   724→  }
   725→   725→
   726→   726→  // High-assurance organizations need higher bar
   727→   727→  if (isHighAssurance && vulnerableProtocols > 0) {
   728→   728→    score = Math.min(score, 35);
   729→   729→  }
   730→   730→
   731→   731→  score = Math.max(0, Math.min(100, score));
   732→   732→
   733→   733→  let readinessLevel: ReadinessLevel;
   734→   734→  if (score >= 85) readinessLevel = 'EXCELLENT';
   735→   735→  else if (score >= 70) readinessLevel = 'GOOD';
   736→   736→  else if (score >= 45) readinessLevel = 'MODERATE';
   737→   737→  else if (score >= 25) readinessLevel = 'HIGH';
   738→   738→  else readinessLevel = 'CRITICAL';
   739→   739→
   740→   740→  // ── 3. Compliance mapping ──
   741→   741→  const compliance = getComplianceMapping(protocolResults, organizationType, complianceFrameworks);
   742→   742→
   743→   743→  // ── 4. Migration roadmap ──
   744→   744→  const migrationRoadmap = buildMigrationRoadmap(protocolResults, organizationType);
   745→   745→
   746→   746→  // ── 5. Transaction integrity ──
   747→   747→  const transactionIntegrity = buildTransactionIntegrity(protocolResults, organizationType);
   748→   748→
   749→   749→  // ── 6. Cost estimation ──
   750→   750→  const multiplier = getOrgMultiplier(organizationType);
   751→   751→  const baseMigrationCost = criticalVulns * 500000 + (totalVulns - criticalVulns) * 250000;
   752→   752→  const totalCost = Math.round(baseMigrationCost * multiplier) + Math.round(2000000 * multiplier);
   753→   753→  const estimatedMigrationCost = `$${totalCost.toLocaleString()}`;
   754→   754→
   755→   755→  let recommendedTimeline: string;
   756→   756→  if (criticalVulns > 0) recommendedTimeline = '24-36 months (immediate action required)';
   757→   757→  else if (vulnerableProtocols > 0) recommendedTimeline = '30-42 months';
   758→   758→  else recommendedTimeline = '12-18 months (validation phase)';
   759→   759→
   760→   760→  return {
   761→   761→    pqcReadinessScore: score,
   762→   762→    readinessLevel,
   763→   763→    protocols: protocolResults,
   764→   764→    compliance,
   765→   765→    migrationRoadmap,
   766→   766→    transactionIntegrity,
   767→   767→    totalVulnerabilities: totalVulns,
   768→   768→    criticalVulnerabilities: criticalVulns,
   769→   769→    estimatedMigrationCost,
   770→   770→    recommendedTimeline,
   771→   771→  };
   772→   772→}
