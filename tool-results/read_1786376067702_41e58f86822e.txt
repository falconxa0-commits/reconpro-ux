   107→
   108→class FindingDetailModal(ModalScreen):
   109→    """Modal overlay showing full finding details."""
   110→
   111→    def __init__(self, finding: Dict[str, Any]) -> None:
   112→        super().__init__()
   113→        self.finding = finding
   114→
   115→    def compose(self) -> ComposeResult:
   116→        f = self.finding
   117→        sev = f.get("severity", "info").lower()
   118→        sev_color = SEV_STYLES.get(sev, "white")
   119→        title = f.get("title", "Unknown")
   120→        category = f.get("category", "")
   121→        module = f.get("module", "")
   122→        pts = f.get("points_deducted", 0)
   123→        evidence = f.get("evidence", "")
   124→        remediation = f.get("remediation", "No remediation available.")
   125→        description = f.get("description", "")
   126→        asset = f.get("asset", "")
   127→        related_cves = f.get("related_cves", [])
   128→
   129→        lines: List[str] = []
   130→        lines.append(f"[{sev_color}][{sev.upper()}] {title}[/]")
   131→        lines.append("")
   132→        if asset:
   133→            lines.append(f"[dim]Asset:[/] [cyan]{asset}[/]")
   134→        if category:
   135→            lines.append(f"[dim]Category:[/] {category}")
   136→        if module:
   137→            lines.append(f"[dim]Module:[/] {module}")
   138→        lines.append(f"[dim]Points:[/] -{pts}")
   139→        if related_cves:
   140→            cve_ids = ", ".join(
   141→                c.get("cve_id", "") for c in related_cves[:5] if c.get("cve_id")
   142→            )
   143→            if cve_ids:
   144→                lines.append(f"[dim]CVEs:[/] [red]{cve_ids}[/]")
   145→        lines.append("")
   146→        if description:
   147→            lines.append(f"[bold]Description:[/]")
   148→            lines.append(f"[dim]{description}[/]")
   149→            lines.append("")
   150→        if evidence:
   151→            lines.append(f"[bold]Evidence:[/]")
   152→            lines.append(f"[dim]{evidence}[/]")
   153→            lines.append("")
   154→        lines.append(f"[bold]{GREEN}Remediation:[/]")
   155→        lines.append(f"{GREEN}{remediation}[/]")
   156→
   157→        content = "\n".join(lines)
   158→
   159→        with Vertical(classes="modal-container"):
   160→            yield Label(content, classes="modal-content")
   161→            with Horizontal(classes="modal-actions"):
   162→                yield Button("Close", variant="primary", id="close-modal")
   163→
   164→    def on_button_pressed(self, event: Button.Pressed) -> None:
   165→        if event.button.id == "close-modal":
   166→            self.dismiss()
   167→

   169→# ════════════════════════════════════════════════════════════════════════════════
   170→# Boot Screen Widget
   171→# ════════════════════════════════════════════════════════════════════════════════
   172→
   173→
   174→class BootScreen(Widget):
   175→    """Full-screen animated boot sequence."""
   176→
   177→    def __init__(self, **kwargs: Any) -> None:
   178→        super().__init__(**kwargs)
   179→        self._step = 0
   180→        self._spinner_idx = 0
   181→        self._module_idx = 0
   182→        self._module_names = list(MODULE_NAMES.values())
   183→        self._done = False
   184→        self._timer: Optional[Timer] = None
   185→
   186→    def compose(self) -> ComposeResult:
   187→        with Vertical(classes="boot-container"):
   188→            yield Static("", id="boot-line-1", classes="boot-title")
   189→            yield Static("", id="boot-line-2", classes="boot-line")
   190→            yield Static("", id="boot-line-3", classes="boot-line")
   191→            yield Static("", id="boot-line-4", classes="boot-line")
   192→
   193→    def on_mount(self) -> None:
   194→        self._timer = self.set_interval(0.04, self._tick)
   195→
   196→    def _tick(self) -> None:
   197→        app = self.app
   198→        if not isinstance(app, NexusApp):
   199→            return
   200→
   201→        line1 = app.query_one("#boot-line-1", Static)
   202→        line2 = app.query_one("#boot-line-2", Static)
   203→        line3 = app.query_one("#boot-line-3", Static)
   204→        line4 = app.query_one("#boot-line-4", Static)
   205→
   206→        if self._step == 0:
   207→            # Letter-by-letter title
   208→            title = "RECONPRO NEXUS"
   209→            chars = title[: self._spinner_idx]
   210→            self._spinner_idx += 1
   211→            line1.update(f"[{CYAN} bold]{chars}[/]")
   212→            if self._spinner_idx > len(title):
   213→                self._spinner_idx = 0
   214→                self._step = 1
   215→                line1.update(f"[{CYAN} bold]{title}[/]  [{DIM_CYAN}]v{VERSION}[/]")
   216→
   217→        elif self._step == 1:
   218→            # Spinner for "Initializing agent..."
   219→            frame = SPINNER_FRAMES[self._spinner_idx % len(SPINNER_FRAMES)]
   220→            self._spinner_idx += 1
   221→            line2.update(f"  [{CYAN}]{frame}[/] [{DIM_CYAN}]Initializing agent...[/]")
   222→            if self._spinner_idx > 30:
   223→                self._spinner_idx = 0
   224→                self._step = 2
   225→                line2.update(f"  [{GREEN}]✓[/] [{DIM_CYAN}]Agent initialized.[/]")
   226→
   227→        elif self._step == 2:
   228→            # Module names appearing one by one
   229→            if self._module_idx < len(self._module_names):
   230→                visible = self._module_names[: self._module_idx + 1]
   231→                self._module_idx += 1
   232→                modules_str = "  ".join(f"[{CYAN}]○[/] [{DIM_CYAN}]{m}[/]" for m in visible)
   233→                remaining = len(self._module_names) - self._module_idx
   234→                if remaining > 0:
   235→                    modules_str += f"  [{DIM_CYAN}]+{remaining} more...[/]"
   236→                frame = SPINNER_FRAMES[self._spinner_idx % len(SPINNER_FRAMES)]
   237→                self._spinner_idx += 1
   238→                line3.update(
   239→                    f"  [{CYAN}]{frame}[/] [{DIM_CYAN}]Loading {len(self._module_names)} modules...[/]\n  {modules_str}"
   240→                )
   241→            else:
   242→                all_mods = "  ".join(
   243→                    f"[{GREEN}]●[/] [{DIM_CYAN}]{m}[/]" for m in self._module_names
   244→                )
   245→                line3.update(
   246→                    f"  [{GREEN}]✓[/] [{DIM_CYAN}]All {len(self._module_names)} modules loaded.[/]\n  {all_mods}"
   247→                )
   248→                self._step = 3
   249→                self._spinner_idx = 0
   250→
   251→        elif self._step == 3:
   252→            # "Ready." with flash
   253→            if self._spinner_idx == 0:
   254→                line4.update(f"[{YELLOW} bold]  ▶  READY.[/]")
   255→            elif self._spinner_idx == 6:
   256→                line4.update(f"[{GREEN} bold]  ▶  READY.[/]")
   257→            elif self._spinner_idx >= 10:
   258→                self._done = True
   259→                if self._timer:
   260→                    self._timer.stop()
   261→                self.app.call_from_thread(self.app._finish_boot)
   262→            self._spinner_idx += 1
   263→

   265→# ════════════════════════════════════════════════════════════════════════════════
   266→# Module Status Cell Widget
   267→# ════════════════════════════════════════════════════════════════════════════════
   268→
   269→
   270→class ModuleCell(Static):
   271→    """Single module status cell in the grid."""
   272→
   273→    def __init__(self, module_id: str, name: str, **kwargs: Any) -> None:
   274→        super().__init__(**kwargs)
   275→        self.module_id = module_id
   276→        self.module_name = name
   277→        self.status = "idle"  # idle | scanning | done | error
   278→        self.finding_count = 0
   279→        self._spinner_idx = 0
   280→        self._timer: Optional[Timer] = None
   281→
   282→    def on_mount(self) -> None:
   283→        self._update_display()
   284→        self._timer = self.set_interval(0.1, self._spin)
   285→
   286→    def _spin(self) -> None:
   287→        if self.status == "scanning":
   288→            self._spinner_idx = (self._spinner_idx + 1) % len(SPINNER_FRAMES)
   289→            self._update_display()
   290→
   291→    def _update_display(self) -> None:
   292→        name_short = self.module_name[:14]
   293→        if self.status == "idle":
   294→            icon = f"[{DIM_CYAN}]○[/]"
   295→            status_text = f"[{DIM_CYAN}]idle[/]"
   296→        elif self.status == "scanning":
   297→            frame = SPINNER_FRAMES[self._spinner_idx]
   298→            icon = f"[{CYAN}]{frame}[/]"
   299→            status_text = f"[{CYAN}]scanning[/]"
   300→        elif self.status == "done":
   301→            icon = f"[{GREEN}]✓[/]"
   302→            status_text = f"[{GREEN}]done ({self.finding_count})[/]"
   303→        elif self.status == "error":
   304→            icon = f"[{RED}]✗[/]"
   305→            status_text = f"[{RED}]error[/]"
   306→        else:
   307→            icon = "○"
   308→            status_text = "idle"
   309→
   310→        self.update(
   311→            f"{icon} [{CYAN} bold]{name_short}[/]\n  {status_text}"
   312→        )
   313→
   314→    def set_status(self, status: str, count: int = 0) -> None:
   315→        self.status = status
   316→        self.finding_count = count
   317→        self._update_display()
   318→
   319→    def on_unmount(self) -> None:
   320→        if self._timer:
   321→            self._timer.stop()
   322→

   325→# Pulsing Status Dot Widget
   326→# ════════════════════════════════════════════════════════════════════════════════
   327→
   328→
   329→class StatusDot(Static):
   330→    """Pulsing dot indicator for scan status."""
   331→
   332→    scanning: reactive[bool] = reactive(False)
   333→
   334→    def __init__(self, **kwargs: Any) -> None:
   335→        super().__init__("", **kwargs)
   336→        self._phase = 0
   337→
   338→    def on_mount(self) -> None:
   339→        self.set_interval(0.3, self._pulse)
   340→
   341→    def watch_scanning(self, old: bool, new: bool) -> None:
   342→        # Update visual state based on scanning status
   343→        if new:
   344→            self.update(f"[{self._get_pulse_color()}]●[/]")
   345→        else:
   346→            self.update(f"[{GREEN}]●[/]")
   347→
   348→    def _get_pulse_color(self) -> str:
   349→        return ["#003322", "#006644", CYAN, "#006644"][self._phase] if self.scanning else GREEN
   350→
   351→    def _pulse(self) -> None:
   352→        self._phase = (self._phase + 1) % 4
   353→        if self.scanning:
   354→            brightness = self._get_pulse_color()
   355→            self.update(f"[{brightness}]●[/]")
   356→        else:
   357→            self.update(f"[{GREEN}]●[/]")

   906→    def on_key(self, event: Key) -> None:
   907→        """Handle key events — boot skip + completer + history + vim-nav."""
   908→        # Skip boot on ANY key press during boot sequence
   909→        if not self._boot_done and not self._boot_skipped:
   910→            self._boot_skipped = True
   911→            try:
   912→                boot = self.query_one("#boot-screen", BootScreen)
   913→                if boot._timer:
   914→                    boot._timer.stop()
   915→            except NoMatches:
   916→                pass
   917→            self._finish_boot()
   918→            return
   919→
   920→        # ── Number keys: quick jump to panels (0=input, 1=chat, 2=findings, 3=modules) ──
   921→        if event.key in ("0", "1", "2", "3") and not self._input_focused():
   922→            jump_map = {"0": "#command-input", "1": "#chat-log", "2": "#findings-feed", "3": "#module-grid"}
   923→            # Skip panel jump to collapsed panels
   924→            target_sel = jump_map[event.key]
   925→            if event.key == "1" and self._left_collapsed:
   926→                target_sel = "#findings-feed"
   927→            if event.key == "3" and self._modules_collapsed:
   928→                target_sel = "#findings-feed"
   929→            self._focus_widget(target_sel)
   930→            event.stop()
   931→            return
   932→
   933→        # ── Phase D: [ and ] to toggle panel collapse ──
   934→        if event.key == "[" and not self._input_focused():
   935→            self._toggle_left_panel()
   936→            event.stop()
   937→            return
   938→        if event.key == "]" and not self._input_focused():
   939→            self._toggle_right_panel()
   940→            event.stop()
   941→            return
   942→        if event.key == "=" and not self._input_focused():
   943→            self._toggle_modules()
   944→            event.stop()
   945→            return
   946→        # ── Phase D: | resets split to auto, \\ resets to 50:50 ──
   947→        if event.key == "|" and not self._input_focused():
   948→            self._user_manually_resized = False
   949→            self._apply_responsive_layout(self.size.width, self.size.height)
   950→            event.stop()
   951→            return
   952→        if event.key == "\\" and not self._input_focused():
   953→            self._user_manually_resized = True
   954→            self._split_ratio = 50
   955→            self._apply_split_ratio()
   956→            event.stop()
   957→            return
   958→
   959→        # ── Phase E: ? opens help overlay ──
   960→        if event.key == "question_mark" and not self._input_focused():
   961→            self._show_help_overlay()
   962→            event.stop()
   963→            return
   964→
   965→        # ── Tab: completer accept/cycle / quick-pick on empty / cycle focus ──
   966→        if event.key == "tab":
   967→            try:
   968→                inp = self.query_one("#command-input", Input)
   969→                comp = self.query_one("#cmd-completer", CommandCompleter)
   970→                if inp.has_focus:
   971→                    # Completer visible → accept top suggestion
   972→                    if comp.visible:
   973→                        accepted = comp.accept_top()
   974→                        if accepted:
   975→                            self._apply_completion(inp, comp, accepted)
   976→                        event.stop()
   977→                        return
   978→                    # Empty input → quick-pick menu
   979→                    if not inp.value.strip():
   980→                        comp.show_quick_pick()
   981→                        event.stop()
   982→                        return
   983→            except NoMatches:
   984→                pass
   985→            # Fall through to default Tab behavior (cycle focus)
   986→            return
   987→
   988→        # ── Escape: dismiss completer, close modal, or jump to input ──
   989→        if event.key == "escape":
   990→            try:
   991→                comp = self.query_one("#cmd-completer", CommandCompleter)
   992→                if comp.visible:
   993→                    comp.hide()
   994→                    event.stop()
   995→                    return
   996→            except NoMatches:
   997→                pass
   998→            # If any modal screen is open, Escape will dismiss it via Textual
   999→            # Otherwise jump back to input
  1000→            if not self._input_focused():
  1001→                self._focus_input()
  1002→                event.stop()
  1003→                return
  1004→
  1005→        # ── Up/Down in completer: cycle suggestions ──
  1006→        if event.key in ("up", "down"):
  1007→            try:
  1008→                comp = self.query_one("#cmd-completer", CommandCompleter)
  1009→                inp = self.query_one("#command-input", Input)
  1010→                if comp.visible and inp.has_focus:
  1011→                    direction = -1 if event.key == "up" else 1
  1012→                    comp.cycle_selection(direction)
  1013→                    event.stop()
  1014→                    return
  1015→            except NoMatches:
  1016→                pass
  1017→
  1018→        # ── j/k vim-style navigation: findings feed + chat log ──
  1019→        if event.key in ("j", "k"):
  1020→            # Findings feed navigation
  1021→            try:
  1022→                feed = self.query_one("#findings-feed", RichLog)
  1023→                if feed.has_focus and self._findings_list:
  1024→                    self._vim_navigate_findings(event.key)
  1025→                    event.stop()
  1026→                    return
  1027→            except NoMatches:
  1028→                pass
  1029→            # Chat log scroll (j=down, k=up)
  1030→            try:
  1031→                chat = self.query_one("#chat-log", RichLog)
  1032→                if chat.has_focus:
  1033→                    if event.key == "j":
  1034→                        chat.scroll_relative(3)
  1035→                    else:
  1036→                        chat.scroll_relative(-3)
  1037→                    event.stop()
  1038→                    return
  1039→            except NoMatches:
  1040→                pass
  1041→
  1042→        # ── Arrow keys on module grid: navigate cells ──
  1043→        if event.key in ("up", "down", "left", "right"):
  1044→            try:
  1045→                grid = self.query_one("#module-grid", Container)
  1046→                if grid.has_focus:
  1047→                    self._arrow_navigate_modules(event.key)
  1048→                    event.stop()
  1049→                    return
  1050→            except NoMatches:
  1051→                pass
  1052→
  1053→        # ── Enter on findings: open detail modal ──
  1054→        if event.key == "enter":
  1055→            try:
  1056→                feed = self.query_one("#findings-feed", RichLog)
  1057→                if feed.has_focus and self._findings_list:
  1058→                    idx = self._findings_cursor if 0 <= self._findings_cursor < len(self._findings_list) else len(self._findings_list) - 1
  1059→                    self.push_screen(FindingDetailModal(self._findings_list[idx]))
  1060→                    event.stop()
  1061→                    return
  1062→            except NoMatches:
  1063→                pass
  1064→
  1065→        # Command history navigation
  1066→        if event.key == "up" and self._history_index > 0:
  1067→            try:
  1068→                inp = self.query_one("#command-input", Input)
  1069→                if inp.has_focus:
  1070→                    self._history_index -= 1
  1071→                    inp.value = self._command_history[self._history_index]
  1072→            except NoMatches:
  1073→                pass
  1074→        elif event.key == "down":
  1075→            try:
  1076→                inp = self.query_one("#command-input", Input)
  1077→                if inp.has_focus:
  1078→                    if self._history_index < len(self._command_history) - 1:
  1079→                        self._history_index += 1
  1080→                        inp.value = self._command_history[self._history_index]
  1081→                    else:
  1082→                        self._history_index = len(self._command_history)
  1083→                        inp.value = ""
  1084→            except NoMatches:
  1085→                pass

  1120→    def _update_panel_focus(self, focused_selector: str) -> None:
  1121→        """Update the panel-focused CSS class on panels.

  1145→        if panel_id:
  1146→            try:
  1147→                self.query_one(panel_id, Widget).set_class(True, "panel-focused")
  1148→                self.query_one(panel_id, Widget).set_class(True, "panel-zoom")
  1149→            except NoMatches:
  1150→                pass

  1152→    def _apply_completion(self, inp: Input, comp: CommandCompleter, accepted: str) -> None:
  1153→        """Apply a completion suggestion to the input field.
  1154→
  1155→        Phase C: handles both command-level and arg-level completions.
  1156→        For arg mode, appends/replaces the last word instead of the first.
  1157→        """
  1158→        parts = inp.value.split()
  1159→
  1160→        if comp._mode == "command":
  1161→            # Replace the first word with the accepted command
  1162→            rest = parts[1:] if len(parts) > 1 else []
  1163→            inp.value = accepted + (" " + " ".join(rest) if rest else "")
