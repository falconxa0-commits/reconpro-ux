   691→def _shannon_entropy(data: str) -> float:
   692→    """Calculate Shannon entropy of a string in bits."""
   693→    if not data:
   694→        return 0.0
   695→    freq: Dict[str, int] = {}
   696→    for ch in data:
   697→        freq[ch] = freq.get(ch, 0) + 1
   698→    length = len(data)
   699→    entropy = 0.0
   700→    for count in freq.values():
   701→        p = count / length
   702→        if p > 0:
   703→            entropy -= p * math.log2(p)
   704→    return round(entropy, 2)

   707→def _version_cmp(v1: str, v2: str) -> int:
   708→    """Compare two version strings. Returns -1, 0, or 1."""
   709→    def _parts(v: str) -> List[int]:
   710→        parts = []
   711→        for p in re.split(r"\.|-", v):
   712→            try:
   713→                parts.append(int(p))
   714→            except ValueError:
   715→                parts.append(0)
   716→        return parts
   717→    p1, p2 = _parts(v1), _parts(v2)
   718→    max_len = max(len(p1), len(p2))
   719→    p1.extend([0] * (max_len - len(p1)))
   720→    p2.extend([0] * (max_len - len(p2)))
   721→    for a, b in zip(p1, p2):
   722→        if a < b:
   723→            return -1
   724→        if a > b:
   725→            return 1
   726→    return 0

   897→        mean_time = statistics.mean(timings)
   898→        try:
   899→            stdev_time = statistics.stdev(timings)
   900→        except statistics.StatisticsError:
   901→            stdev_time = 0.0
   902→
   903→        time_sigma = ANOMALY_THRESHOLDS["timing"]["timing_variance_sigma"]
   904→        for r in responses:
   905→            t = r["elapsed_ms"]
   906→            if stdev_time > 0 and t > 0:
   907→                z_time = abs(t - mean_time) / stdev_time
   908→                if z_time > time_sigma:
   909→                    sev = "high" if t > ANOMALY_THRESHOLDS["timing"]["suspicious_slow_ms"] else "medium"
   910→                    findings.append(Finding(
   911→                        title=f"Timing anomaly at {r['path']} ({t:.0f}ms)",
   912→                        severity=sev,
   913→                        category="response_anomaly",
   914→                        module="zero_day_hunter",
   915→                        description=f"Response time ({t:.0f}ms) is {z_time:.1f}σ from baseline ({mean_time:.0f}ms). May indicate time-based blind injection vector, backend resource exhaustion, or uncached computation.",
   916→                        evidence=f"GET {r['path']} -> {r['status']} ({t:.0f}ms, mean={mean_time:.0f}ms, σ={stdev_time:.0f}ms)",
   917→                        asset=host,
   918→                        points_deducted=_severity_points(sev),
   919→                        remediation="Investigate slow endpoints for missing caching, N+1 queries, or time-based side channels. Slow responses on simple paths may indicate injection vulnerabilities.",
   920→                        dread_score=_dread_score(sev),
   921→                    ))
   922→
   923→    # Body entropy analysis
   924→    for r in responses:
   925→        body = r.get("body", "")
   926→        if len(body) > 50:
   927→            entropy = _shannon_entropy(body)
   928→            if entropy < ANOMALY_THRESHOLDS["body_entropy"]["low_entropy_threshold"]:
   929→                findings.append(Finding(
   930→                    title=f"Low-entropy response body at {r['path']} (H={entropy})",
   931→                    severity="low",
   932→                    category="response_anomaly",
   933→                    module="zero_day_hunter",
   934→                    description=f"Response body has very low Shannon entropy ({entropy} bits), suggesting template content, placeholder text, or repeatable patterns that may indicate a default/vulnerable installation.",
   935→                    evidence=f"GET {r['path']} -> {r['status']} ({r['body_length']} bytes, entropy={entropy})",
   936→                    asset=host,
   937→                    points_deducted=3,
   938→                    remediation="Low entropy may indicate default configurations; ensure custom content and security hardening are applied.",
   939→                    dread_score=_dread_score("low"),
   940→                ))
   941→            elif entropy > ANOMALY_THRESHOLDS["body_entropy"]["high_entropy_threshold"]:
   942→                if r["status"] == 200 and not any(r["path"].lower().endswith(ext) for ext in [".png", ".jpg", ".ico", ".woff", ".ttf", ".svg"]):
   943→                    findings.append(Finding(
   944→                        title=f"High-entropy response at {r['path']} (H={entropy})",
   945→                        severity="medium",
   946→                        category="response_anomaly",
   947→                        module="zero_day_hunter",
   948→                        description=f"Response body has high Shannon entropy ({entropy} bits), suggesting encrypted/compressed content, binary data leak, or minified source maps on a non-static endpoint.",
   949→                        evidence=f"GET {r['path']} -> {r['status']} ({r['body_length']} bytes, entropy={entropy})",
   950→                        asset=host,
   951→                        points_deducted=5,
   952→                        remediation="High-entropy text responses on API endpoints may indicate source map leakage or serialized binary data exposure.",
   953→                        dread_score=_dread_score("medium"),
   954→                    ))
   955→
   956→    return findings, responses
