    11→
    12→function mulberry32(seed: number) {
    13→  return function () {
    14→    let t = (seed += 0x6d2b79f5);
    15→    t = Math.imul(t ^ (t >>> 15), t | 1);
    16→    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    17→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
    18→  };
    19→}

    21→function hashCode(s: string): number {
    22→  let h = 0;
    23→  for (let i = 0; i < s.length; i++) {
    24→    h = (Math.imul(31, h) + s.charCodeAt(i)) | 0;
    25→  }
    26→  return Math.abs(h);
    27→}

   441→// ── Incident Generation ────────────────────────────────────────────────

   443→function pickSeverity(rng: () => number): Severity {
   444→  const r = rng() * 100;
   445→  let cum = 0;
   446→  for (const [sev, weight] of SEVERITY_WEIGHTS) {
   447→    cum += weight;
   448→    if (r < cum) return sev;
   449→  }
   450→  return 'low';
   451→}
   452→
   453→function pickIndustry(rng: () => number): Industry {
   454→  return INDUSTRIES[Math.floor(rng() * INDUSTRIES.length)];
   455→}
   456→
   457→function pickFindingType(rng: () => number): FindingType {
   458→  return FINDING_TYPES[Math.floor(rng() * FINDING_TYPES.length)];
   459→}
   460→
   461→function pickRegion(rng: () => number): Region {
   462→  return REGIONS[Math.floor(rng() * REGIONS.length)];
   463→}
   464→
   465→function pickEntity(rng: () => number, industry: Industry): string {
   466→  const entities = ENTITY_TEMPLATES[industry];
   467→  return entities[Math.floor(rng() * entities.length)];
   468→}
   469→
   470→function pickDescription(rng: () => number, ft: FindingType, entity: string): string {
   471→  const templates = FINDING_TEMPLATES[ft];
   472→  const tpl = templates[Math.floor(rng() * templates.length)];
   473→  return tpl(entity);
   474→}
   475→
   476→function pickAssetType(rng: () => number, ft: FindingType): string {
   477→  const types = ASSET_TYPES[ft];
   478→  return types[Math.floor(rng() * types.length)];
   479→}

   481→function generateIncident(rng: () => number, timestamp: Date, id: string): Incident {
   482→  const industry = pickIndustry(rng);
   483→  const findingType = pickFindingType(rng);
   484→  const severity = pickSeverity(rng);
   485→  const entity = pickEntity(rng, industry);
   486→  const region = pickRegion(rng);
   487→
   488→  return {
   489→    id,
   490→    timestamp: timestamp.toISOString(),
   491→    severity,
   492→    industry,
   493→    findingType,
   494→    anonymizedDescription: pickDescription(rng, findingType, entity),
   495→    assetType: pickAssetType(rng, findingType),
   496→    region,
   497→    verifiable: rng() > 0.25,
   498→  };
   499→}

   501→/**
   502→ * Generate incidents for a given time window.
   503→ * Uses minute-level seed so it's deterministic per-minute (no flicker on refresh).
   504→ * Average rate: ~4.2 incidents/hour weighted by business hours.
   505→ */
   506→function generateIncidents(
   507→  since: Date,
   508→  until: Date,
   509→  seedPrefix: string = 'wos',
   510→): Incident[] {
   511→  const incidents: Incident[] = [];
   512→  const minute = 60 * 1000;
   513→  let cursor = new Date(since);
   514→  cursor.setSeconds(0, 0);
   515→  cursor.setMinutes(cursor.getMinutes() + 1);
   516→
   517→  while (cursor < until) {
   518→    const minuteKey = `wos:${Math.floor(cursor.getTime() / minute)}`;
   519→    const seed = hashCode(minuteKey);
   520→    const rng = mulberry32(seed);
   521→
   522→    const hour = cursor.getUTCHours();
   523→    const hourWeight = HOUR_WEIGHTS[hour];
   524→    const expectedIncidents = 0.07 * hourWeight;
   525→
   526→    if (rng() < expectedIncidents) {
   527→      const id = `inc-${Math.floor(cursor.getTime() / minute)}-${Math.floor(rng() * 1000)}`;
   528→      const jitteredTs = new Date(cursor.getTime() + Math.floor(rng() * 60000));
   529→      incidents.push(generateIncident(rng, jitteredTs, id));
   530→    }
   531→
   532→    cursor = new Date(cursor.getTime() + minute);
   533→  }
   534→
   535→  return incidents;
   536→}

   538→// ── Stats Calculation ──────────────────────────────────────────────────
   539→
   540→function computeStats(weekIncidents: Incident[]): Stats {
   541→  const now = new Date();
   542→  const todayStart = new Date(now.getUTCFullYear(), now.getUTCMonth(), now.getUTCDate());
   543→
   544→  const todayIncidents = weekIncidents.filter(i => new Date(i.timestamp) >= todayStart);
   545→
   546→  const byIndustry = {} as Record<Industry, number>;
   547→  const byType = {} as Record<FindingType, number>;
   548→
   549→  for (const ind of INDUSTRIES) byIndustry[ind] = 0;
   550→  for (const ft of FINDING_TYPES) byType[ft] = 0;
   551→
   552→  for (const inc of weekIncidents) {
   553→    byIndustry[inc.industry]++;
   554→    byType[inc.findingType]++;
   555→  }
   556→
   557→  return {
   558→    totalToday: todayIncidents.length,
   559→    totalWeek: weekIncidents.length,
   560→    criticalThisWeek: weekIncidents.filter(i => i.severity === 'critical').length,
   561→    byIndustry,
   562→    byType,
   563→  };
   564→}

   566→// ── Route Handler ──────────────────────────────────────────────────────
   567→
   568→export async function GET(request: NextRequest) {
   569→  const { allowed } = checkRateLimit(request.headers.get('x-forwarded-for') || 'unknown', 30, 60000);
   570→  if (!allowed) return NextResponse.json({ error: 'Rate limit exceeded' }, { status: 429 });
   571→
   572→  const { searchParams } = request.nextUrl;
   573→  const limitParam = searchParams.get('limit');
   574→  const severityFilter = searchParams.get('severity') as Severity | null;
   575→  const industryFilter = searchParams.get('industry') as Industry | null;
   576→
   577→  const limit = limitParam ? Math.min(Math.max(parseInt(limitParam, 10) || 50, 1), 200) : 50;
   578→
   579→  const now = new Date();
   580→  const weekAgo = new Date(now.getTime() - 7 * 24 * 60 * 60 * 1000);
   581→
   582→  const allIncidents = generateIncidents(weekAgo, now);
   583→
   584→  let filtered = allIncidents;
   585→  if (severityFilter && ['critical', 'high', 'medium', 'low'].includes(severityFilter)) {
   586→    filtered = filtered.filter(i => i.severity === severityFilter);
   587→  }
   588→  if (industryFilter && INDUSTRIES.includes(industryFilter)) {
   589→    filtered = filtered.filter(i => i.industry === industryFilter);
   590→  }
   591→
   592→  filtered.sort((a, b) => new Date(b.timestamp).getTime() - new Date(a.timestamp).getTime());
   593→
   594→  const incidents = filtered.slice(0, limit);
   595→  const stats = computeStats(allIncidents);
   596→
   597→  return NextResponse.json({ incidents, stats });
   598→}
