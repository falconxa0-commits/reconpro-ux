    51→def _read_file(path: str, max_size: int = 1048576) -> str:
    52→    """Read file contents safely."""
    53→    try:
    54→        size = os.path.getsize(path)
    55→        if size > max_size:
    56→            return "[file too large]"
    57→        with open(path, errors="replace") as f:
    58→            return f.read()
    59→    except Exception:
    60→        return ""

    62→
    63→def _simple_yaml_parse(text: str) -> Any:
    64→    """Fallback YAML parser when PyYAML is not installed."""
    65→    try:
    66→        import yaml  # type: ignore[import-untyped]
    67→        return yaml.safe_load(text)
    68→    except Exception:
    69→        pass
    70→    try:
    71→        return json.loads(text)
    72→    except Exception:
    73→        pass
    74→    result: Dict[str, Any] = {}
    75→    current_list_key: Optional[str] = None
    76→    list_items: List[Dict[str, Any]] = []
    77→    for line in text.splitlines():
    78→        stripped = line.strip()
    79→        if not stripped or stripped.startswith("#"):
    80→            continue
    81→        list_match = re.match(r"^\s*-\s+(\w+):\s*(.*)$", line)
    82→        if list_match:
    83→            key, val = list_match.group(1), list_match.group(2).strip()
    84→            if key != current_list_key:
    85→                if current_list_key is not None and list_items:
    86→                    result[current_list_key] = list_items
    87→                current_list_key = key
    88→                list_items = []
    89→            item: Dict[str, Any] = {key: val}
    90→            list_items.append(item)
    91→            continue
    92→        kv_match = re.match(r"^([\w.]+):\s*(.*)$", stripped)
    93→        if kv_match:
    94→            key, val = kv_match.group(1), kv_match.group(2).strip()
    95→            if current_list_key is not None:
    96→                pass
    97→            if val.lower() in ("true", "false", "null"):
    98→                result[key] = {"true": True, "false": False, "null": None}[val.lower()]
    99→            else:
   100→                try:
   101→                    result[key] = json.loads(val)
   102→                except Exception:
   103→                    result[key] = val
   104→    if current_list_key is not None and list_items:
   105→        result[current_list_key] = list_items
   106→    return result
   107→
   108→
   109→def _yaml_load_all(text: str) -> List[Any]:
   110→    """Load all YAML documents from text."""
   111→    try:
   112→        import yaml  # type: ignore[import-untyped]
   113→        return list(yaml.safe_load_all(text)) or []
   114→    except Exception:
   115→        pass
   116→    docs: List[Any] = [_simple_yaml_parse(text)]
   117→    return docs

   120→def _dict_get(data: Any, *keys: str, default: Any = None) -> Any:
   121→    """Deep get from nested dicts, safe against missing keys."""
   122→    current = data
   123→    for k in keys:
   124→        if isinstance(current, dict):
   125→            current = current.get(k, default)
   126→        else:
   127→            return default
   128→    return current
   129→
   130→
   131→def _search_dicts(data: Any, key: str) -> List[Any]:
   132→    """Recursively find all values for a given key in nested structures."""
   133→    results: List[Any] = []
   134→    if isinstance(data, dict):
   135→        if key in data:
   136→            results.append(data[key])
   137→        for v in data.values():
   138→            results.extend(_search_dicts(v, key))
   139→    elif isinstance(data, list):
   140→        for item in data:
   141→            results.extend(_search_dicts(item, key))
   142→    return results
   143→
   144→
   145→def _has_value_in_structure(data: Any, key: str, target_value: Any) -> bool:
   146→    """Check if any occurrence of key has the target value."""
   147→    values = _search_dicts(data, key)
   148→    return target_value in values
   149→
   150→
   151→def _get_key_values(data: Any, key: str) -> List[Any]:
   152→    """Get all values for a key, flattening lists."""
   153→    flat: List[Any] = []
   154→    for v in _search_dicts(data, key):
   155→        if isinstance(v, list):
   156→            flat.extend(v)
   157→        else:
   158→            flat.append(v)
   159→    return flat
   160→
   161→
   162→def _find_container_specs(data: Any) -> List[Dict[str, Any]]:
   163→    """Find all container spec dicts in a K8s manifest."""
   164→    specs: List[Dict[str, Any]] = []
   165→    if isinstance(data, dict):
   166→        spec = data.get("spec")
   167→        if isinstance(spec, dict):
   168→            for ck in ("containers", "initContainers"):
   169→                containers = spec.get(ck)
   170→                if isinstance(containers, list):
   171→                    specs.extend(containers)
   172→            template = spec.get("template")
   173→            if isinstance(template, dict):
   174→                tmpl_spec = template.get("spec")
   175→                if isinstance(tmpl_spec, dict):
   176→                    for ck in ("containers", "initContainers"):
   177→                        containers = tmpl_spec.get(ck)
   178→                        if isinstance(containers, list):
   179→                            specs.extend(containers)
   180→    return specs
   181→
   182→
   183→def _find_pod_or_template_spec(data: Any) -> Optional[Dict[str, Any]]:
   184→    """Find the pod-level spec (or template spec for Deployments)."""
   185→    if isinstance(data, dict):
   186→        spec = data.get("spec")
   187→        if isinstance(spec, dict):
   188→            template = spec.get("template")
   189→            if isinstance(template, dict):
   190→                tmpl_spec = template.get("spec")
   191→                if isinstance(tmpl_spec, dict):
   192→                    return tmpl_spec
   193→            return spec
   194→    return None

   201→class DockerfileAnalyzer:
   202→    """Analyzes Dockerfiles for security misconfigurations and escape vectors."""
   203→
   204→    SENSITIVE_PATHS = re.compile(
   205→        r"\.(env|aws|ssh)|id_rsa|.*\.pem|credentials",
   206→        re.IGNORECASE,
   207→    )
   208→    SECRET_ENV_PATTERN = re.compile(
   209→        r"(?i)ENV\s+.*(PASSWORD|SECRET|TOKEN|KEY|CREDENTIAL)",
   210→    )
   211→    DANGEROUS_PORTS = {22, 2375, 2376, 6379, 27017, 5432, 3306}
   212→
   213→    def analyze(self, path: str) -> List[Dict[str, Any]]:
   214→        """Analyze a Dockerfile and return a list of finding dicts."""
   215→        findings: List[Dict[str, Any]] = []
   216→        content = _read_file(path)
   217→        if not content or content.startswith("[file too large]"):
   218→            return findings
   219→
   220→        lines = content.splitlines()
   221→        fname = os.path.basename(path)
   222→        has_user = False
   223→        had_nonroot_user = False
   224→        has_healthcheck = False
   225→        has_from = False
   226→        from_lines: List[int] = []
   227→        secret_env_in_builder = False
   228→
   229→        # Check 1: No .dockerignore found
   230→        dockerignore_path = os.path.join(os.path.dirname(path), ".dockerignore")
   231→        if not os.path.exists(dockerignore_path):
   232→            findings.append({
   233→                "title": "No .dockerignore Found",
   234→                "severity": "low",
   235→                "category": "docker_build",
   236→                "description": "No .dockerignore file found alongside Dockerfile. Unnecessary files may be included in the build context.",
   237→                "evidence": f"{fname}: missing .dockerignore",
   238→                "remediation": "Create a .dockerignore file to exclude .git, .env, node_modules, and other unnecessary files from the build context.",
   239→                "line_number": 0,
   240→            })
   241→
   242→        for idx, line in enumerate(lines, start=1):
   243→            stripped = line.strip()
   244→            directive_match = re.match(r"^(?:FROM|RUN|COPY|ADD|ENV|EXPOSE|USER|HEALTHCHECK|ARG|ENTRYPOINT|CMD)\b", stripped, re.IGNORECASE)
   245→
   246→            # Check 2: FROM without tag or :latest
   247→            if re.match(r"^FROM\b", stripped, re.IGNORECASE):
   248→                has_from = True
   249→                from_lines.append(idx)
   250→                from_part = stripped[4:].strip()
   251→                if from_part.startswith("--platform="):
   252→                    from_part = re.sub(r"^--platform=\S+\s+", "", from_part)
   253→                if re.match(r"^AS\b", from_part, re.IGNORECASE):
   254→                    from_part = re.sub(r"^AS\s+\S+\s*", "", from_part, flags=re.IGNORECASE)
   255→                if ":" not in from_part:
   256→                    findings.append({
   257→                        "title": f"FROM Without Tag: {from_part}",
   258→                        "severity": "medium",
   259→                        "category": "docker_build",
   260→                        "description": f"Base image '{from_part}' has no explicit tag. Defaults to ':latest' which is non-reproducible.",
   261→                        "evidence": _ev(fname, idx),
   262→                        "remediation": f"Pin the base image to a specific digest: FROM {from_part}:<tag>@<digest>",
   263→                        "line_number": idx,
   264→                    })
   265→                elif from_part.endswith(":latest"):
   266→                    findings.append({
   267→                        "title": f"FROM Uses :latest Tag: {from_part}",
   268→                        "severity": "medium",
   269→                        "category": "docker_build",
   270→                        "description": f"Base image '{from_part}' uses the ':latest' tag. Builds are non-reproducible and may break unexpectedly.",
   271→                        "evidence": _ev(fname, idx),
   272→                        "remediation": f"Pin to a specific version and digest: FROM {from_part.split(':')[0]}:<version>@<digest>",
   273→                        "line_number": idx,
   274→                    })
   275→
   276→            # Check 3: COPY/ADD of sensitive paths
   277→            copy_match = re.match(r"^(?:COPY|ADD)\s+(.+?)(?:\s+(?:/|\S+))?$", stripped, re.IGNORECASE)
   278→            if copy_match:
   279→                src_part = copy_match.group(1).strip()
   280→                # Remove flags like --chown=, --from=
   281→                src_part = re.sub(r"^--\w+[=\s]+\S+\s+", "", src_part)
   282→                # Could be multi-source: "src1 src2  dest"
   283→                src_parts = src_part.split()
   284→                for sp in src_parts:
   285→                    if self.SENSITIVE_PATHS.search(sp):
   286→                        findings.append({
   287→                            "title": f"Sensitive Path in COPY/ADD: {sp}",
   288→                            "severity": "critical",
   289→                            "category": "docker_secret_leak",
   290→                            "description": f"Sensitive file '{sp}' is being copied into the container image. This may expose secrets, SSH keys, or cloud credentials.",
   291→                            "evidence": _ev(fname, idx),
   292→                            "remediation": f"Avoid copying '{sp}' into the image. Use multi-stage builds, Docker secrets, or runtime volume mounts instead.",
   293→                            "line_number": idx,
   294→                        })
   295→
   296→            # Check 4: ADD of remote URL (supply chain risk)
   297→            add_match = re.match(r"^ADD\s+(https?://\S+)", stripped, re.IGNORECASE)
   298→            if add_match:
   299→                url = add_match.group(1).split()[0]
   300→                findings.append({
   301→                    "title": f"ADD Remote URL (Supply Chain Risk): {url[:60]}",
   302→                    "severity": "medium",
   303→                    "category": "docker_supply_chain",
   304→                    "description": f"ADD fetches from a remote URL. The remote content could change, introducing malicious files.",
   305→                    "evidence": _ev(fname, idx),
   306→                    "remediation": "Use COPY with a locally-verified file, or use RUN curl/wget with integrity verification (checksums, GPG).",
   307→                    "line_number": idx,
   308→                })
   309→
   310→            # Check 5: ENV with secret patterns
   311→            if self.SECRET_ENV_PATTERN.search(stripped):
   312→                env_match = re.match(r"^ENV\s+(\S+?)=(.*)$", stripped, re.IGNORECASE)
   313→                if env_match:
   314→                    var_name = env_match.group(1)
   315→                    var_val = env_match.group(2).strip().strip('"').strip("'")
   316→                    if var_val and not var_val.startswith("$"):
   317→                        findings.append({
   318→                            "title": f"Secret in ENV Variable: {var_name}",
   319→                            "severity": "high",
   320→                            "category": "docker_secret_leak",
   321→                            "description": f"Environment variable '{var_name}' appears to contain a hardcoded secret. This is baked into the image and visible in 'docker inspect'.",
   322→                            "evidence": _ev(fname, idx),
   323→                            "remediation": f"Use Docker secrets, a secret management system (Vault, AWS Secrets Manager), or pass secrets at runtime via -e or --env-file.",
   324→                            "line_number": idx,
   325→                        })
   326→                        if len(from_lines) > 1:
   327→                            secret_env_in_builder = True
   328→
   329→            # Check 6: EXPOSE dangerous ports
   330→            expose_match = re.match(r"^EXPOSE\s+(.+)$", stripped, re.IGNORECASE)
   331→            if expose_match:
   332→                ports_str = expose_match.group(1)
   333→                for port_tok in re.findall(r"(\d+)", ports_str):
   334→                    port = int(port_tok)
   335→                    if port in self.DANGEROUS_PORTS:
   336→                        svc_map = {22: "SSH", 2375: "Docker API (unencrypted)", 2376: "Docker API (TLS)",
   337→                                   6379: "Redis", 27017: "MongoDB", 5432: "PostgreSQL", 3306: "MySQL"}
   338→                        findings.append({
   339→                            "title": f"Dangerous Port Exposed: {port} ({svc_map.get(port, 'unknown')})",
   340→                            "severity": "medium",
   341→                            "category": "docker_network",
   342→                            "description": f"Port {port} ({svc_map.get(port, 'unknown')}) is exposed. Exposing database or management ports increases the attack surface.",
   343→                            "evidence": _ev(fname, idx),
   344→                            "remediation": f"Remove the EXPOSE directive for port {port} or ensure it is only accessible within an internal Docker network.",
   345→                            "line_number": idx,
   346→                        })
   347→
   348→            # Check 7: --privileged in RUN
   349→            if re.search(r"--privileged", stripped, re.IGNORECASE):
   350→                findings.append({
   351→                    "title": "--privileged Flag in RUN",
   352→                    "severity": "critical",
   353→                    "category": "docker_privilege",
   354→                    "description": "The --privileged flag in RUN gives the build step full host access. This is almost never needed and is extremely dangerous.",
   355→                    "evidence": _ev(fname, idx),
   356→                    "remediation": "Remove --privileged. If specific capabilities are needed, use --cap-add for individual capabilities.",
   357→                    "line_number": idx,
   358→                })
   359→
   360→            # Check 8: apt-get without no-install-recommends or without rm -rf /var/lib/apt
   361→            if "apt-get" in stripped and "install" in stripped:
   362→                if "no-install-recommends" not in stripped:
   363→                    findings.append({
   364→                        "title": "apt-get install Without --no-install-recommends",
   365→                        "severity": "low",
   366→                        "category": "docker_image_size",
   367→                        "description": "apt-get install without --no-install-recommends installs unnecessary recommended packages, increasing image size and attack surface.",
   368→                        "evidence": _ev(fname, idx),
   369→                        "remediation": "Use 'apt-get install --no-install-recommends' to minimize installed packages.",
   370→                        "line_number": idx,
   371→                    })
   372→
   373→            # Track USER directive
   374→            if re.match(r"^USER\b", stripped, re.IGNORECASE):
   375→                has_user = True
   376→                user_val = stripped[4:].strip()
   377→                if user_val.lower() in ("root", "0"):
   378→                    if had_nonroot_user:
   379→                        # Check 10: USER changed back to root after being non-root
   380→                        findings.append({
   381→                            "title": "USER Reverted to Root After Non-Root User",
   382→                            "severity": "high",
   383→                            "category": "docker_privilege",
   384→                            "description": "A non-root USER was set earlier, but has been changed back to root. This defeats the purpose of dropping privileges.",
   385→                            "evidence": _ev(fname, idx),
   386→                            "remediation": "Ensure the final USER directive is a non-root user. If root is needed for specific steps, use multi-stage builds.",
   387→                            "line_number": idx,
   388→                        })
   389→                else:
   390→                    had_nonroot_user = True
   391→
   392→            # Check 9: HEALTHCHECK
   393→            if re.match(r"^HEALTHCHECK\b", stripped, re.IGNORECASE):
   394→                has_healthcheck = True
   395→
   396→        # Check 9 (post-loop): No HEALTHCHECK
   397→        if not has_healthcheck:
   398→            findings.append({
   399→                "title": "No HEALTHCHECK Instruction",
   400→                "severity": "low",
   401→                "category": "docker_monitoring",
   402→                "description": "No HEALTHCHECK instruction found. The container orchestrator cannot determine if the application is healthy.",
   403→                "evidence": f"{fname}: no HEALTHCHECK",
   404→                "remediation": "Add a HEALTHCHECK instruction: HEALTHCHECK --interval=30s --timeout=5s --retries=3 CMD <health-command>",
   405→                "line_number": 0,
   406→            })
   407→
   408→        # Check 1 (post-loop): Runs as root (no USER directive or USER root)
   409→        if not has_user:
   410→            findings.append({
   411→                "title": "Container Runs as Root",
   412→                "severity": "high",
   413→                "category": "docker_privilege",
   414→                "description": "No USER directive found. The container will run as root by default, giving full control to any process inside the container.",
   415→                "evidence": f"{fname}: no USER directive",
   416→                "remediation": "Add 'USER <non-root-user>' near the end of the Dockerfile. Create a dedicated user with 'RUN useradd -m appuser'.",
   417→                "line_number": 0,
   418→            })
   419→        elif not had_nonroot_user:
   420→            # USER was set but only to root
   421→            findings.append({
   422→                "title": "Container Explicitly Runs as Root",
   423→                "severity": "high",
   424→                "category": "docker_privilege",
   425→                "description": "USER directive is set to root. The container runs with full root privileges inside the container.",
   426→                "evidence": f"{fname}: USER root",
   427→                "remediation": "Change USER to a non-root account. Create a dedicated user and switch to it before ENTRYPOINT/CMD.",
   428→                "line_number": 0,
   429→            })
   430→
   431→        # Check apt-get cleanup (post-loop, look for rm -rf /var/lib/apt in any RUN)
   432→        has_apt_install = False
   433→        has_apt_cleanup = False
   434→        for line in lines:
   435→            stripped = line.strip()
   436→            if "apt-get" in stripped and "install" in stripped:
   437→                has_apt_install = True
   438→            if "rm" in stripped and ("/var/lib/apt" in stripped or "/var/cache/apt" in stripped):
   439→                has_apt_cleanup = True
   440→        if has_apt_install and not has_apt_cleanup:
   441→            findings.append({
   442→                "title": "apt-get Cache Not Cleaned",
   443→                "severity": "low",
   444→                "category": "docker_image_size",
   445→                "description": "apt-get install is used but the package cache is not cleaned. This significantly increases image size.",
   446→                "evidence": f"{fname}: missing 'rm -rf /var/lib/apt/lists/*'",
   447→                "remediation": "Combine apt-get install and cleanup in one RUN layer: 'apt-get update && apt-get install --no-install-recommends -y pkg && rm -rf /var/lib/apt/lists/*'",
   448→                "line_number": 0,
   449→            })
   450→
   451→        # Check 12: Multi-stage build with secrets in builder stage
   452→        if len(from_lines) > 1 and secret_env_in_builder:
   453→            findings.append({
   454→                "title": "Multi-Stage Build with Secrets in Builder Stage",
   455→                "severity": "high",
   456→                "category": "docker_secret_leak",
   457→                "description": "Multi-stage build detected with ENV secrets in builder stages. Builder stage secrets may persist in build cache or intermediate layers.",
   458→                "evidence": f"{fname}: {len(from_lines)} FROM directives, secrets in ENV",
   459→                "remediation": "Use --secret flag (BuildKit) or pass secrets via ARG with --build-arg. Never use ENV for secrets in builder stages.",
   460→                "line_number": from_lines[1] if len(from_lines) > 1 else 0,
   461→            })
   462→
   463→        return findings
   464→

   467→#  K8sAnalyzer
   468→# ══════════════════════════════════════════════════════════════════════
   469→
   470→class K8sAnalyzer:
   471→    """Analyzes Kubernetes YAML manifests for security misconfigurations."""
   472→
   473→    DANGEROUS_CAPS = {"ALL", "NET_RAW", "SYS_ADMIN", "SYS_PTRACE", "SYS_MODULE", "NET_ADMIN"}
   474→    HOST_MOUNT_PATHS = {"/", "/etc", "/root", "/var/run", "/sys", "/proc"}
   475→
   476→    def analyze_file(self, path: str) -> List[Dict[str, Any]]:
   477→        """Analyze a K8s YAML manifest and return a list of finding dicts."""
   478→        findings: List[Dict[str, Any]] = []
   479→        content = _read_file(path)
   480→        if not content or content.startswith("[file too large]"):
   481→            return findings
   482→
   483→        fname = os.path.basename(path)
   484→        docs = _yaml_load_all(content)
   485→        kind_counts: Dict[str, int] = {}
   486→
   487→        for doc in docs:
   488→            if not isinstance(doc, dict):
   489→                continue
   490→            kind = doc.get("kind", "")
   491→            if kind:
   492→                kind_counts[kind] = kind_counts.get(kind, 0) + 1
   493→            findings.extend(self._analyze_doc(doc, fname, path))
   494→
   495→        # Check 17: No NetworkPolicy in namespace
   496→        if "NetworkPolicy" not in kind_counts:
   497→            if any(k in kind_counts for k in ("Pod", "Deployment", "DaemonSet", "StatefulSet", "ReplicaSet", "Job")):

   499→                    "title": "No NetworkPolicy Defined",

   505→                    "line_number": 0,
   506→                })
   507→
   508→        # Check 18: Service type LoadBalancer without annotation

   514→                if svc_type == "LoadBalancer":
   515→                    annotations = _dict_get(doc, "metadata", "annotations", default={})
   516→                    if not isinstance(annotations, dict):
   517→                        annotations = {}
   518→                    has_safety_annotation = any(
   519→                        k.lower() for k in annotations
   520→                        if "whitelist" in k.lower() or "allowlist" in k.lower()
   521→                        or "security" in k.lower() or "firewall" in k.lower()

   529→                            "evidence": f"{fname}: Service type=LoadBalancer, no security annotation",
   530→                            "remediation": "Add network restrictions via annotations or use an Ingress with TLS and auth. Consider internal load balancers for non-public services.",
   531→                            "line_number": 0,
   532→                        })
   533→
   534→        # Check 19: Ingress without TLS
   535→        for doc in docs:
   536→            if not isinstance(doc, dict):
   537→                continue
   538→            if doc.get("kind") == "Ingress":
   539→                tls_config = _dict_get(doc, "spec", "tls")
   540→                if not tls_config:
   541→                    findings.append({
   542→                        "title": "Ingress Without TLS",
   543→                        "severity": "medium",
   544→                        "category": "k8s_network",
   545→                        "description": "Ingress resource has no TLS configuration. Traffic is served over unencrypted HTTP.",
   546→                        "evidence": f"{fname}: Ingress missing spec.tls",
   547→                        "remediation": "Add a TLS section to the Ingress spec with a secretRef to a TLS certificate secret.",
   548→                        "line_number": 0,
   549→                    })
   550→
   551→        return findings

   908→        compound: List[Dict[str, Any]] = []
   909→
   910→        for pattern in self._PATTERNS:
   911→            matched_conditions: List[str] = []
   912→            matched_evidence: List[str] = []
   913→            all_match = True
   914→
   915→            for condition in pattern["any_of"]:
   916→                found = False
   917→                for f in findings:
   918→                    cat = f.get("category", "")
   919→                    title = f.get("title", "")
   920→                    if cat == condition["category"] and condition["title_sub"] in title:
   921→                        matched_conditions.append(title)
   922→                        matched_evidence.append(f.get("evidence", ""))
   923→                        found = True
   924→                        break
   925→                if not found:
   926→                    all_match = False
   927→                    break
   928→
   929→            if all_match:
   930→                compound.append({
   931→                    "title": f"Escape Vector: {pattern['name']}",
   932→                    "severity": pattern["severity"],
   933→                    "category": "container_escape_vector",
   934→                    "description": pattern["description"],
   935→                    "evidence": "; ".join(matched_evidence),
   936→                    "remediation": pattern["remediation"],
   937→                    "line_number": 0,
   938→                    "component_findings": matched_conditions,
   939→                })
   940→
   941→        return compound
