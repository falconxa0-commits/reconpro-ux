    98→# Signature Broadcaster
    99→# ══════════════════════════════════════════════════════════════════════
   100→
   101→class SignatureBroadcaster:
   102→    """Injects ReconPro signature into HTTP probes via X-R3c0nPr0 header.
   103→
   104→    Every outbound probe should pass through the broadcaster to ensure
   105→    consistent identification and version tracking across all modules.
   106→
   107→    Attributes:
   108→        version: The ReconPro version string embedded in the header.
   109→        build:   The build/component identifier (e.g. 'wishes-engine').
   110→        probes_sent: Running counter of how many probes were signed.
   111→    """
   112→
   113→    def __init__(self, version: str = RECONPRO_VERSION, build: str = RECONPRO_BUILD) -> None:
   114→        self.version = version
   115→        self.build = build
   116→        self._probes_sent: int = 0
   117→        self._session_id = hashlib.sha256(
   118→            f"{version}:{build}:{time.time()}".encode()
   119→        ).hexdigest()[:12]
   120→
   121→    @property
   122→    def probes_sent(self) -> int:
   123→        """Number of probes that have been signed in this session."""
   124→        return self._probes_sent
   125→
   126→    @property
   127→    def session_id(self) -> str:
   128→        """Unique session identifier for this broadcaster instance."""
   129→        return self._session_id
   130→
   131→    def broadcast(self, headers: Optional[Dict[str, str]] = None) -> Dict[str, str]:
   132→        """Add the ReconPro signature header to a headers dict.
   133→
   134→        Args:
   135→            headers: Existing headers dict to augment, or None for a new dict.
   136→
   137→        Returns:
   138→            The headers dict with the X-R3c0nPr0 header injected.
   139→        """
   140→        if headers is None:
   141→            headers = {}
   142→        headers[RECONPRO_SIGNATURE_HEADER] = (
   143→            f"ReconPro/{self.version} ({self.build}); sid={self._session_id}"
   144→        )
   145→        self._probes_sent += 1
   146→        return headers
   147→
   148→    def get_ua_with_signature(self) -> str:
   149→        """Return a User-Agent string that includes the signature.
   150→
   151→        This UA can be used as a fallback when custom headers are not
   152→        supported by the transport layer.
   153→        """
   154→        return (
   155→            f"ReconPro/{self.version} (Wishes Framework v{self.version}; "
   156→            f"build={self.build}; "
   157→            f"+https://github.com/reconpro-security/reconpro)"
   158→        )
   159→
   160→    def inject_into_request(self, method: str, url: str,
   161→                            headers: Optional[Dict[str, str]] = None,
   162→                            body: Optional[bytes] = None) -> Dict[str, Any]:
   163→        """Build a complete request descriptor with signature injected.
   164→
   165→        Args:
   166→            method:  HTTP method (GET, POST, etc.).
   167→            url:     Target URL.
   168→            headers: Optional existing headers.
   169→            body:    Optional request body.
   170→
   171→        Returns:
   172→            Dict with 'method', 'url', 'headers', 'body', 'signed_at'.
   173→        """
   174→        signed_headers = self.broadcast(headers)
   175→        return {
   176→            "method": method,
   177→            "url": url,
   178→            "headers": signed_headers,
   179→            "body": body,
   180→            "signed_at": _now_iso(),
   181→            "session_id": self._session_id,
   182→        }
   183→
   184→    def reset_counter(self) -> None:
   185→        """Reset the probe counter (useful between scan targets)."""
   186→        self._probes_sent = 0
   187→
   188→
   189→def _default_broadcaster() -> SignatureBroadcaster:
   190→    """Factory for a default SignatureBroadcaster instance."""
   191→    return SignatureBroadcaster()

   195→# Witness Writer
   196→# ══════════════════════════════════════════════════════════════════════
   197→
   198→class WitnessWriter:
   199→    """Creates immutable audit witness records with cryptographic SHA-256 signature.
   200→
   201→    Each witness is a JSON file recording that a scan encounter occurred.
   202→    The signature is computed over the canonical JSON of core fields,
   203→    making the record tamper-evident. Witnesses are stored in
   204→    ~/.reconpro/witness/ as individual JSON files.
   205→
   206→    Attributes:
   207→        witness_dir: Directory where witness files are persisted.
   208→    """
   209→
   210→    def __init__(self, witness_dir: Optional[Path] = None) -> None:
   211→        self._dir = witness_dir or WITNESS_DIR
   212→        self._dir.mkdir(parents=True, exist_ok=True)
   213→
   214→    @property
   215→    def witness_dir(self) -> Path:
   216→        return self._dir
   217→
   218→    def _build_core(self, encounter_id: str, target: str, verdict: str,
   219→                    timestamp: str, score: float) -> Dict[str, Any]:
   220→        """Construct the canonical core fields that are signed."""
   221→        return {
   222→            "encounter_id": encounter_id,
   223→            "target": target,
   224→            "verdict": verdict,
   225→            "timestamp": timestamp,
   226→            "score": score,
   227→        }
   228→
   229→    def _sign(self, core: Dict[str, Any]) -> str:
   230→        """Compute SHA-256 hex digest of the canonical core fields."""
   231→        return hashlib.sha256(
   232→            json.dumps(core, sort_keys=True).encode()
   233→        ).hexdigest()
   234→
   235→    def write_witness(
   236→        self,
   237→        target: str,
   238→        verdict: str,
   239→        findings_summary: Dict[str, Any],
   240→        score: float,
   241→        modules_run: List[str],
   242→        extra: Optional[Dict[str, Any]] = None,
   243→    ) -> Dict[str, Any]:
   244→        """Write an immutable witness record to disk.
   245→
   246→        Args:
   247→            target:          The scan target (domain, IP, URL).
   248→            verdict:         Unified verdict string (MUNDANE/NOTABLE/SUBSTANTIAL/CRITICAL).
   249→            findings_summary: Dict with 'total' and optionally 'by_severity'.
   250→            score:           Numeric unified score (0-100).
   251→            modules_run:     List of module names that were executed.
   252→            extra:           Optional additional fields to embed.
   253→
   254→        Returns:
   255→            The complete witness dict as written to disk.
   256→        """
   257→        encounter_id = hashlib.sha256(
   258→            f"{target}:{time.time()}".encode()
   259→        ).hexdigest()[:16]
   260→        timestamp = _now_iso()
   261→
   262→        core = self._build_core(encounter_id, target, verdict, timestamp, score)
   263→        signature = self._sign(core)
   264→
   265→        witness: Dict[str, Any] = {
   266→            "encounter_id": encounter_id,
   267→            "timestamp": timestamp,
   268→            "target": target,
   269→            "verdict": verdict,
   270→            "signature": signature,
   271→            "findings_count": findings_summary.get("total", 0),
   272→            "score": score,
   273→            "modules_run": modules_run,
   274→            "severity_breakdown": findings_summary.get("by_severity", {}),
   275→            "manifest": "WITNESS MANIFEST \u2014 the encounter occurred and is recorded",
   276→        }
   277→        if extra:
   278→            witness.update(extra)
   279→
   280→        path = self._dir / f"witness_{_safe_target(target)}_{encounter_id}.json"
   281→        path.write_text(json.dumps(witness, indent=2), encoding="utf-8")
   282→        logger.debug("Witness written: %s", path.name)
   283→        return witness
   284→
   285→    def verify_witness(self, witness: Dict[str, Any]) -> bool:
   286→        """Verify the cryptographic signature of a witness record.
   287→
   288→        Args:
   289→            witness: A previously-written witness dict.
   290→
   291→        Returns:
   292→            True if the signature is valid and untampered.
   293→        """
   294→        required = ("encounter_id", "target", "verdict", "timestamp", "score")
   295→        if not all(k in witness for k in required):
   296→            return False
   297→        core = self._build_core(
   298→            witness["encounter_id"],
   299→            witness["target"],
   300→            witness["verdict"],
   301→            witness["timestamp"],
   302→            witness["score"],
   303→        )
   304→        expected = self._sign(core)
   305→        return witness.get("signature") == expected
   306→
   307→    def verify_all(self) -> Tuple[int, int]:
   308→        """Verify every witness on disk.
   309→
   310→        Returns:
   311→            Tuple of (valid_count, invalid_count).
   312→        """
   313→        valid, invalid = 0, 0
   314→        for w in self.list_witnesses():
   315→            if self.verify_witness(w):
   316→                valid += 1
   317→            else:
   318→                invalid += 1
   319→        return valid, invalid
   320→
   321→    def list_witnesses(self, target: Optional[str] = None) -> List[Dict[str, Any]]:
   322→        """List all witness records, optionally filtered by target.
   323→
   324→        Args:
   325→            target: If provided, only return witnesses for this target.
   326→
   327→        Returns:
   328→            List of witness dicts, sorted by filename (chronological).
   329→        """
   330→        witnesses: List[Dict[str, Any]] = []
   331→        if not self._dir.exists():
   332→            return witnesses
   333→        for f in sorted(self._dir.glob("witness_*.json")):
   334→            try:
   335→                w = json.loads(f.read_text(encoding="utf-8"))
   336→                if target is None or w.get("target") == target:
   337→                    witnesses.append(w)
   338→            except Exception:
   339→                pass
   340→        return witnesses
   341→
   342→    def get_witness(self, encounter_id: str) -> Optional[Dict[str, Any]]:
   343→        """Retrieve a single witness by its encounter_id.
   344→
   345→        Args:
   346→            encounter_id: The 16-char hex encounter identifier.
   347→
   348→        Returns:
   349→            The witness dict, or None if not found.
   350→        """
   351→        for w in self.list_witnesses():
   352→            if w.get("encounter_id") == encounter_id:
   353→                return w
   354→        return None
   355→
   356→    def purge_witnesses(self, before_timestamp: Optional[str] = None,
   357→                        target: Optional[str] = None) -> int:
   358→        """Delete witness files, optionally filtered.
   359→
   360→        Args:
   361→            before_timestamp: ISO timestamp; delete witnesses older than this.
   362→            target:           Only delete witnesses for this target.
   363→
   364→        Returns:
   365→            Number of witnesses deleted.
   366→        """
   367→        deleted = 0
   368→        if not self._dir.exists():
   369→            return deleted
   370→        for f in self._dir.glob("witness_*.json"):
   371→            try:
   372→                w = json.loads(f.read_text(encoding="utf-8"))
   373→                if target and w.get("target") != target:
   374→                    continue
   375→                if before_timestamp and w.get("timestamp", "") >= before_timestamp:
   376→                    continue
   377→                f.unlink()
   378→                deleted += 1
   379→            except Exception:
   380→                pass
   381→        return deleted
   382→
   383→    def get_stats(self) -> Dict[str, Any]:
   384→        """Return aggregate statistics over all witnesses."""
   385→        witnesses = self.list_witnesses()
   386→        verdict_counts: Dict[str, int] = {}
   387→        for w in witnesses:
   388→            v = w.get("verdict", "UNKNOWN")
   389→            verdict_counts[v] = verdict_counts.get(v, 0) + 1
   390→        return {
   391→            "total_witnesses": len(witnesses),
   392→            "verdict_distribution": verdict_counts,
   393→            "witness_dir": str(self._dir),
   394→        }

   397→# ══════════════════════════════════════════════════════════════════════
   398→# Hall of the Broken (GORGON encounters)
   399→# ══════════════════════════════════════════════════════════════════════
   400→
   401→class HallOfTheBroken:
   402→    """Persistent GORGON encounter registry — tracks Fear Index per target.
   403→
   404→    The Hall persists across scans in ~/.reconpro/hall/broken.json.
   405→    It maintains running statistics: total encounters, average fear,
   406→    most feared target, and highest single fear index.
   407→
   408→    Attributes:
   409→        hall_path: Path to the broken.json persistence file.
   410→    """
   411→
   412→    def __init__(self, hall_dir: Optional[Path] = None) -> None:
   413→        self._dir = hall_dir or HALL_DIR
   414→        self._dir.mkdir(parents=True, exist_ok=True)
   415→        self._path = self._dir / "broken.json"
   416→        self._encounters = self._load()
   417→
   418→    @property
   419→    def hall_path(self) -> Path:
   420→        return self._path
   421→
   422→    def _load(self) -> Dict[str, Any]:
   423→        """Load the Hall from disk, or return empty structure."""
   424→        if self._path.exists():
   425→            try:
   426→                return json.loads(self._path.read_text(encoding="utf-8"))
   427→            except Exception:
   428→                logger.warning("Failed to load Hall of the Broken, starting fresh")
   429→        return {
   430→            "encounters": [],
   431→            "total_fear": 0.0,
   432→            "average_fear": 0.0,
   433→            "most_feared": None,
   434→            "highest_fear": 0.0,
   435→            "total_encounters": 0,
   436→        }
   437→
   438→    def _save(self) -> None:
   439→        """Persist the Hall to disk."""
   440→        try:
   441→            self._path.write_text(
   442→                json.dumps(self._encounters, indent=2), encoding="utf-8"
   443→            )
   444→        except Exception as exc:
   445→            logger.error("Failed to save Hall of the Broken: %s", exc)
   446→
   447→    def _recalculate(self) -> None:
   448→        """Recalculate aggregate statistics from the encounter list."""
   449→        encounters = self._encounters["encounters"]
   450→        count = len(encounters)
   451→        if count == 0:
   452→            self._encounters["total_fear"] = 0.0
   453→            self._encounters["average_fear"] = 0.0
   454→            self._encounters["most_feared"] = None
   455→            self._encounters["highest_fear"] = 0.0
   456→            self._encounters["total_encounters"] = 0
   457→            return
   458→        total = sum(e.get("fear_index", 0) for e in encounters)
   459→        self._encounters["total_fear"] = round(total, 1)
   460→        self._encounters["average_fear"] = round(total / count, 1)
   461→        self._encounters["total_encounters"] = count
   462→        # Find most feared (highest fear_index)
   463→        top = max(encounters, key=lambda e: e.get("fear_index", 0))
   464→        self._encounters["most_feared"] = top.get("target")
   465→        self._encounters["highest_fear"] = top.get("fear_index", 0)
   466→
   467→    def record_encounter(
   468→        self,
   469→        target: str,
   470→        fear_index: float,
   471→        endpoints_found: int = 0,
   472→        vulnerable_count: int = 0,
   473→        cves_matched: int = 0,
   474→        bypasses_achieved: int = 0,
   475→        secrets_extracted: int = 0,
   476→    ) -> Dict[str, Any]:
   477→        """Record a GORGON encounter in the Hall.
   478→
   479→        Args:
   480→            target:            The scan target identifier.
   481→            fear_index:        Computed Fear Index (0-100).
   482→            endpoints_found:   Number of endpoints discovered.
   483→            vulnerable_count: Number of vulnerable endpoints.
   484→            cves_matched:      Number of CVEs matched.
   485→            bypasses_achieved: Number of WAF/auth bypasses.
   486→            secrets_extracted: Number of secrets/credentials found.
   487→
   488→        Returns:
   489→            The encounter entry dict that was recorded.
   490→        """
   491→        entry: Dict[str, Any] = {
   492→            "target": target,
   493→            "fear_index": fear_index,
   494→            "fear_level": _fear_label(fear_index),
   495→            "endpoints_found": endpoints_found,
   496→            "vulnerable_count": vulnerable_count,
   497→            "cves_matched": cves_matched,
   498→            "bypasses_achieved": bypasses_achieved,
   499→            "secrets_extracted": secrets_extracted,
   500→            "timestamp": _now_iso(),
   501→        }
   502→        self._encounters["encounters"].append(entry)
   503→        self._recalculate()
   504→        self._save()
   505→        logger.debug("Hall of the Broken: recorded %s (fear=%.1f)", target, fear_index)
   506→        return entry
   507→
   508→    def get_stats(self) -> Dict[str, Any]:
   509→        """Return aggregated Hall statistics."""
   510→        return dict(self._encounters)
   511→
   512→    def get_top_feared(self, n: int = 5) -> List[Dict[str, Any]]:
   513→        """Return the top N most feared targets by fear_index.
   514→
   515→        Args:
   516→            n: Number of top entries to return.
   517→
   518→        Returns:
   519→            List of encounter dicts, sorted descending by fear_index.
   520→        """
   521→        sorted_encounters = sorted(
   522→            self._encounters["encounters"],
   523→            key=lambda e: e.get("fear_index", 0),
   524→            reverse=True,
   525→        )
   526→        return sorted_encounters[:n]
   527→
   528→    def get_target_history(self, target: str) -> List[Dict[str, Any]]:
   529→        """Return all encounters for a specific target.
   530→
   531→        Args:
   532→            target: The target to filter by.
   533→
   534→        Returns:
   535→            List of encounter dicts for the given target.
   536→        """
   537→        return [
   538→            e for e in self._encounters["encounters"]
   539→            if e.get("target") == target
   540→        ]
   541→
   542→    def get_average_fear(self) -> float:
   543→        """Return the average fear index across all encounters."""
   544→        return self._encounters.get("average_fear", 0.0)
   545→
   546→    def get_most_feared(self) -> Optional[str]:
   547→        """Return the target with the highest single fear index."""
   548→        return self._encounters.get("most_feared")
   549→
   550→    def get_highest_fear(self) -> float:
   551→        """Return the highest single fear index ever recorded."""
   552→        return self._encounters.get("highest_fear", 0.0)
   553→
   554→    def purge(self, before_timestamp: Optional[str] = None) -> int:
   555→        """Remove encounters, optionally filtered by timestamp.
   556→
   557→        Args:
   558→            before_timestamp: Remove encounters older than this ISO timestamp.
   559→
   560→        Returns:
   561→            Number of encounters removed.
   562→        """
   563→        before = len(self._encounters["encounters"])
   564→        if before_timestamp:
   565→            self._encounters["encounters"] = [
   566→                e for e in self._encounters["encounters"]
   567→                if e.get("timestamp", "") >= before_timestamp
   568→            ]
   569→        else:
   570→            self._encounters["encounters"] = []
   571→        after = len(self._encounters["encounters"])
   572→        self._recalculate()
   573→        self._save()
   574→        return before - after
   575→
   576→    def export_json(self) -> str:
   577→        """Export the full Hall as a JSON string."""
   578→        return json.dumps(self._encounters, indent=2)

   631→    def _recalculate(self) -> None:
   632→        """Recalculate aggregate statistics from the encounter list."""
   633→        encounters = self._encounters["encounters"]
   634→        count = len(encounters)
   635→        if count == 0:
   636→            self._encounters["total_dread"] = 0.0
   637→            self._encounters["average_dread"] = 0.0
   638→            self._encounters["most_dreaded"] = None
   639→            self._encounters["highest_dread"] = 0.0
   640→            self._encounters["total_encounters"] = 0
   641→            return
   642→        total = sum(e.get("dread_index", 0) for e in encounters)
   643→        self._encounters["total_dread"] = round(total, 1)
   644→        self._encounters["average_dread"] = round(total / count, 1)
   645→        self._encounters["total_encounters"] = count
   646→        top = max(encounters, key=lambda e: e.get("dread_index", 0))
   647→        self._encounters["most_dreaded"] = top.get("target")
   648→        self._encounters["highest_dread"] = top.get("dread_index", 0)

   765→class FearIndex:
   766→    """Computes Fear Index (1-100) from GORGON scan engagement metrics.
   767→
   768→    The Fear Index is distinct from DREAD scoring — it measures the
   769→    aggregate threat surface discovered during the GORGON engagement.
   770→    Each input dimension is weighted to reflect real-world risk impact.
   771→
   772→    Weight rationale:
   773→        - bypasses_achieved (15.0): WAF/auth bypasses are the most
   774→          alarming because they bypass defense-in-depth.
   775→        - vulnerable_count (12.0): Direct exploitable vulnerabilities.
   776→        - secrets_extracted (8.0): Credential/secret exposure enables
   777→          lateral movement and deeper compromise.
   778→        - cves_matched (4.0): Known CVE matches indicate unpatched systems.
   779→        - endpoints_found (1.5): Surface area discovery; lower individual
   780→          risk but contributes to cumulative exposure.
   781→
   782→    The raw weighted sum is clamped to [0, 100] and mapped to a fear
   783→    level (1-10) with a human-readable label.
   784→    """
   785→
   786→    WEIGHTS: Dict[str, float] = {
   787→        "endpoints": 1.5,
   788→        "vulnerable": 12.0,
   789→        "cves": 4.0,
   790→        "bypasses": 15.0,
   791→        "secrets": 8.0,
   792→    }
   793→
   794→    def compute(
   795→        self,
   796→        endpoints_found: int = 0,
   797→        vulnerable_count: int = 0,
   798→        cves_matched: int = 0,
   799→        bypasses_achieved: int = 0,
   800→        secrets_extracted: int = 0,
   801→    ) -> Dict[str, Any]:
   802→        """Compute the Fear Index from GORGON engagement metrics.
   803→
   804→        Args:
   805→            endpoints_found:   Number of unique endpoints discovered.
   806→            vulnerable_count:  Number of endpoints with vulnerabilities.
   807→            cves_matched:      Number of CVE IDs matched.
   808→            bypasses_achieved: Number of WAF/auth bypass successes.
   809→            secrets_extracted: Number of secrets/credentials extracted.
   810→
   811→        Returns:
   812→            Dict with 'fear_index' (0-100), 'fear_level' (1-10),
   813→            'fear_label', and 'components' breakdown.
   814→        """
   815→        weighted = {
   816→            "endpoints": endpoints_found * self.WEIGHTS["endpoints"],
   817→            "vulnerable": vulnerable_count * self.WEIGHTS["vulnerable"],
   818→            "cves": cves_matched * self.WEIGHTS["cves"],
   819→            "bypasses": bypasses_achieved * self.WEIGHTS["bypasses"],
   820→            "secrets": secrets_extracted * self.WEIGHTS["secrets"],
   821→        }
   822→        raw = min(100.0, sum(weighted.values()))
   823→        level = max(1, min(10, int(raw // 10) + (1 if raw > 0 else 0)))
   824→
   825→        return {
   826→            "fear_index": round(raw, 1),
   827→            "fear_level": level,
   828→            "fear_label": FEAR_LEVELS[level],
   829→            "components": {
   830→                "endpoints_found": endpoints_found,
   831→                "vulnerable_count": vulnerable_count,
   832→                "cves_matched": cves_matched,
   833→                "bypasses_achieved": bypasses_achieved,
   834→                "secrets_extracted": secrets_extracted,
   835→            },
   836→            "weighted_breakdown": {k: round(v, 1) for k, v in weighted.items()},
   837→        }
   838→
   839→    def from_findings(self, findings: List[Dict[str, Any]]) -> Dict[str, Any]:
   840→        """Compute Fear Index from a list of Finding dicts.
   841→
   842→        Inspects each finding's 'category' and 'severity' fields to
   843→        derive the five input metrics automatically.
   844→
   845→        Args:
   846→            findings: List of finding dicts from scan modules.
   847→
   848→        Returns:
   849→            Same structure as compute().
   850→        """
   851→        endpoints = sum(
   852→            1 for f in findings
   853→            if f.get("category") in ("surface_map", "api_discovery", "websocket")
   854→        )
   855→        vulnerable = sum(
   856→            1 for f in findings
   857→            if f.get("severity") in ("critical", "high")
   858→        )
   859→        cves = sum(
   860→            1 for f in findings if "cve" in f.get("title", "").lower()
   861→        )
   862→        bypasses = sum(
   863→            1 for f in findings
   864→            if any(kw in f.get("title", "").lower()
   865→                   for kw in ("bypass", "injection", "xss", "sqli"))
   866→        )
   867→        secrets = sum(
   868→            1 for f in findings
   869→            if any(kw in f.get("title", "").lower()
   870→                   for kw in ("secret", "key", "credential", "token", "exposure"))
   871→        )
   872→        return self.compute(
   873→            endpoints_found=endpoints,
   874→            vulnerable_count=vulnerable,
   875→            cves_matched=cves,
   876→            bypasses_achieved=bypasses,
   877→            secrets_extracted=secrets,
   878→        )
   879→

   881→# ══════════════════════════════════════════════════════════════════════
   882→# Unified Verdict
   883→# ══════════════════════════════════════════════════════════════════════
   884→
   885→class UnifiedVerdict:
   886→    """Aggregates multi-module results into a unified verdict.
   887→
   888→    Verdict levels (ascending severity):
   889→        MUNDANE     — Minimal or no significant findings.
   890→        NOTABLE     — Some findings warranting attention.
   891→        SUBSTANTIAL — Significant findings requiring remediation.
   892→        CRITICAL    — Severe findings requiring immediate action.
   893→
   894→    The computation uses module-specific weights (higher weight for
   895→    offensive modules like GORGON and OBLIVION) to produce a weighted
   896→    average, then applies thresholds to select the verdict.
   897→    """
   898→
   899→    def compute(self, module_scores: Dict[str, Dict[str, Any]]) -> Dict[str, Any]:
   900→        """Compute unified verdict from per-module score dicts.
   901→
   902→        Args:
   903→            module_scores: Dict mapping module name to a dict with at
   904→                           least a 'score' key (0-100).
   905→
   906→        Returns:
   907→            Dict with 'verdict', 'unified_score', 'modules_assessed',
   908→            'critical_modules', 'max_score', 'min_score'.
   909→        """
   910→        if not module_scores:
   911→            return {"verdict": "MUNDANE", "unified_score": 0.0,
   912→                    "modules_assessed": 0, "critical_modules": 0,
   913→                    "max_score": 0.0, "min_score": 0.0}
   914→
   915→        weighted_sum = 0.0
   916→        total_weight = 0.0
   917→        scores: List[float] = []
   918→        critical_count = 0
   919→
   920→        for mod_name, mod_data in module_scores.items():
   921→            score = mod_data.get("score", 0)
   922→            if score is None:
   923→                continue
   924→            weight = VERDICT_WEIGHTS.get(mod_name, 1.0)
   925→            weighted_sum += score * weight
   926→            total_weight += weight
   927→            scores.append(float(score))
   928→            if score >= 80:
   929→                critical_count += 1
   930→
   931→        if not scores:
   932→            return {"verdict": "MUNDANE", "unified_score": 0.0,
   933→                    "modules_assessed": 0, "critical_modules": 0,
   934→                    "max_score": 0.0, "min_score": 0.0}
   935→
   936→        avg = weighted_sum / total_weight if total_weight > 0 else sum(scores) / len(scores)
   937→        max_score = max(scores)
   938→
   939→        if avg >= 70 or critical_count >= 3:
   940→            verdict = "CRITICAL"
   941→        elif avg >= 50 or critical_count >= 2:
   942→            verdict = "SUBSTANTIAL"
   943→        elif avg >= 25 or max_score >= 60:
   944→            verdict = "NOTABLE"
   945→        else:
   946→            verdict = "MUNDANE"
   947→
   948→        return {
   949→            "verdict": verdict,
   950→            "unified_score": round(avg, 1),
   951→            "modules_assessed": len(scores),
   952→            "critical_modules": critical_count,
   953→            "max_score": max_score,
   954→            "min_score": min(scores),
   955→        }
   956→
   957→    def from_findings(self, findings: List[Dict[str, Any]]) -> Dict[str, Any]:
   958→        """Compute verdict from a flat list of findings.
   959→
   960→        Args:
   961→            findings: List of finding dicts with 'severity' and
   962→                      optionally 'points_deducted'.
   963→
   964→        Returns:
   965→            Dict with 'verdict', 'unified_score', and severity counts.
   966→        """
   967→        total_pts = sum(f.get("points_deducted", 0) for f in findings)
   968→        critical = sum(1 for f in findings if f.get("severity") == "critical")
   969→        high = sum(1 for f in findings if f.get("severity") == "high")
   970→        medium = sum(1 for f in findings if f.get("severity") == "medium")
   971→        low = sum(1 for f in findings if f.get("severity") == "low")
   972→        score = min(100.0, total_pts * 2.0)
   973→
   974→        if score >= 70 or critical >= 5:
   975→            verdict = "CRITICAL"
   976→        elif score >= 40 or critical >= 2:
   977→            verdict = "SUBSTANTIAL"
   978→        elif score >= 15 or high >= 3:
   979→            verdict = "NOTABLE"
   980→        else:
   981→            verdict = "MUNDANE"
   982→
   983→        return {
   984→            "verdict": verdict,
   985→            "unified_score": score,
   986→            "total_findings": len(findings),
   987→            "critical_count": critical,
   988→            "high_count": high,
   989→            "medium_count": medium,
   990→            "low_count": low,
   991→        }
