    95→
    96→class FindingDetailModal(ModalScreen):
    97→    """Modal overlay showing full finding details."""
    98→
    99→    def __init__(self, finding: Dict[str, Any]) -> None:
   100→        super().__init__()
   101→        self.finding = finding
   102→
   103→    def compose(self) -> ComposeResult:
   104→        f = self.finding
   105→        sev = f.get("severity", "info").lower()
   106→        sev_color = SEV_STYLES.get(sev, "white")
   107→        title = f.get("title", "Unknown")
   108→        category = f.get("category", "")
   109→        module = f.get("module", "")
   110→        pts = f.get("points_deducted", 0)
   111→        evidence = f.get("evidence", "")
   112→        remediation = f.get("remediation", "No remediation available.")
   113→        description = f.get("description", "")
   114→        asset = f.get("asset", "")
   115→        related_cves = f.get("related_cves", [])
   116→
   117→        lines: List[str] = []
   118→        lines.append(f"[{sev_color}][{sev.upper()}] {title}[/]")
   119→        lines.append("")
   120→        if asset:
   121→            lines.append(f"[dim]Asset:[/] [cyan]{asset}[/]")
   122→        if category:
   123→            lines.append(f"[dim]Category:[/] {category}")
   124→        if module:
   125→            lines.append(f"[dim]Module:[/] {module}")
   126→        lines.append(f"[dim]Points:[/] -{pts}")
   127→        if related_cves:
   128→            cve_ids = ", ".join(
   129→                c.get("cve_id", "") for c in related_cves[:5] if c.get("cve_id")
   130→            )
   131→            if cve_ids:
   132→                lines.append(f"[dim]CVEs:[/] [red]{cve_ids}[/]")
   133→        lines.append("")
   134→        if description:
   135→            lines.append(f"[bold]Description:[/]")
   136→            lines.append(f"[dim]{description}[/]")
   137→            lines.append("")
   138→        if evidence:
   139→            lines.append(f"[bold]Evidence:[/]")
   140→            lines.append(f"[dim]{evidence}[/]")
   141→            lines.append("")
   142→        lines.append(f"[bold]{GREEN}Remediation:[/]")
   143→        lines.append(f"{GREEN}{remediation}[/]")
   144→
   145→        content = "\n".join(lines)
   146→
   147→        with Vertical(classes="modal-container"):
   148→            yield Label(content, classes="modal-content")
   149→            with Horizontal(classes="modal-actions"):
   150→                yield Button("Close", variant="primary", id="close-modal")
   151→
   152→    def on_button_pressed(self, event: Button.Pressed) -> None:
   153→        if event.button.id == "close-modal":
   154→            self.dismiss()
   155→

   157→# ════════════════════════════════════════════════════════════════════════════════
   158→# Boot Screen Widget
   159→# ════════════════════════════════════════════════════════════════════════════════
   160→
   161→
   162→class BootScreen(Widget):
   163→    """Full-screen animated boot sequence."""
   164→
   165→    def __init__(self) -> None:
   166→        super().__init__()
   167→        self._step = 0
   168→        self._spinner_idx = 0
   169→        self._module_idx = 0
   170→        self._module_names = list(MODULE_NAMES.values())
   171→        self._done = False
   172→        self._timer: Optional[Timer] = None
   173→
   174→    def compose(self) -> ComposeResult:
   175→        with Vertical(classes="boot-container"):
   176→            yield Static("", id="boot-line-1", classes="boot-title")
   177→            yield Static("", id="boot-line-2", classes="boot-line")
   178→            yield Static("", id="boot-line-3", classes="boot-line")
   179→            yield Static("", id="boot-line-4", classes="boot-line")
   180→
   181→    def on_mount(self) -> None:
   182→        self._timer = self.set_interval(0.04, self._tick)
   183→
   184→    def _tick(self) -> None:
   185→        app = self.app
   186→        if not isinstance(app, NexusApp):
   187→            return
   188→
   189→        line1 = app.query_one("#boot-line-1", Static)
   190→        line2 = app.query_one("#boot-line-2", Static)
   191→        line3 = app.query_one("#boot-line-3", Static)
   192→        line4 = app.query_one("#boot-line-4", Static)
   193→
   194→        if self._step == 0:
   195→            # Letter-by-letter title
   196→            title = "RECONPRO NEXUS"
   197→            chars = title[: self._spinner_idx]
   198→            self._spinner_idx += 1
   199→            line1.update(f"[{CYAN} bold]{chars}[/]")
   200→            if self._spinner_idx > len(title):
   201→                self._spinner_idx = 0
   202→                self._step = 1
   203→                line1.update(f"[{CYAN} bold]{title}[/]  [{DIM_CYAN}]v{VERSION}[/]")
   204→
   205→        elif self._step == 1:
   206→            # Spinner for "Initializing agent..."
   207→            frame = SPINNER_FRAMES[self._spinner_idx % len(SPINNER_FRAMES)]
   208→            self._spinner_idx += 1
   209→            line2.update(f"  [{CYAN}]{frame}[/] [{DIM_CYAN}]Initializing agent...[/]")
   210→            if self._spinner_idx > 30:
   211→                self._spinner_idx = 0
   212→                self._step = 2
   213→                line2.update(f"  [{GREEN}]✓[/] [{DIM_CYAN}]Agent initialized.[/]")
   214→
   215→        elif self._step == 2:
   216→            # Module names appearing one by one
   217→            if self._module_idx < len(self._module_names):
   218→                visible = self._module_names[: self._module_idx + 1]
   219→                self._module_idx += 1
   220→                modules_str = "  ".join(f"[{CYAN}]○[/] [{DIM_CYAN}]{m}[/]" for m in visible)
   221→                remaining = len(self._module_names) - self._module_idx
   222→                if remaining > 0:
   223→                    modules_str += f"  [{DIM_CYAN}]+{remaining} more...[/]"
   224→                frame = SPINNER_FRAMES[self._spinner_idx % len(SPINNER_FRAMES)]
   225→                self._spinner_idx += 1
   226→                line3.update(
   227→                    f"  [{CYAN}]{frame}[/] [{DIM_CYAN}]Loading {len(self._module_names)} modules...[/]\n  {modules_str}"
   228→                )
   229→            else:
   230→                all_mods = "  ".join(
   231→                    f"[{GREEN}]●[/] [{DIM_CYAN}]{m}[/]" for m in self._module_names
   232→                )
   233→                line3.update(
   234→                    f"  [{GREEN}]✓[/] [{DIM_CYAN}]All {len(self._module_names)} modules loaded.[/]\n  {all_mods}"
   235→                )
   236→                self._step = 3
   237→                self._spinner_idx = 0
   238→
   239→        elif self._step == 3:
   240→            # "Ready." with flash
   241→            if self._spinner_idx == 0:
   242→                line4.update(f"[{YELLOW} bold]  ▶  READY.[/]")
   243→            elif self._spinner_idx == 6:
   244→                line4.update(f"[{GREEN} bold]  ▶  READY.[/]")
   245→            elif self._spinner_idx >= 10:
   246→                self._done = True
   247→                if self._timer:
   248→                    self._timer.stop()
   249→                self.app.call_from_thread(self.app._finish_boot)
   250→            self._spinner_idx += 1
   251→

   253→# ════════════════════════════════════════════════════════════════════════════════
   254→# Module Status Cell Widget
   255→# ════════════════════════════════════════════════════════════════════════════════
   256→
   257→
   258→class ModuleCell(Static):
   259→    """Single module status cell in the grid."""
   260→
   261→    def __init__(self, module_id: str, name: str) -> None:
   262→        super().__init__()
   263→        self.module_id = module_id
   264→        self.module_name = name
   265→        self.status = "idle"  # idle | scanning | done | error
   266→        self.finding_count = 0
   267→        self._spinner_idx = 0
   268→        self._timer: Optional[Timer] = None
   269→
   270→    def on_mount(self) -> None:
   271→        self._update_display()
   272→        self._timer = self.set_interval(0.1, self._spin)
   273→
   274→    def _spin(self) -> None:
   275→        if self.status == "scanning":
   276→            self._spinner_idx = (self._spinner_idx + 1) % len(SPINNER_FRAMES)
   277→            self._update_display()
   278→
   279→    def _update_display(self) -> None:
   280→        name_short = self.module_name[:14]
   281→        if self.status == "idle":
   282→            icon = f"[{DIM_CYAN}]○[/]"
   283→            status_text = f"[{DIM_CYAN}]idle[/]"
   284→        elif self.status == "scanning":
   285→            frame = SPINNER_FRAMES[self._spinner_idx]
   286→            icon = f"[{CYAN}]{frame}[/]"
   287→            status_text = f"[{CYAN}]scanning[/]"
   288→        elif self.status == "done":
   289→            icon = f"[{GREEN}]✓[/]"
   290→            status_text = f"[{GREEN}]done ({self.finding_count})[/]"
   291→        elif self.status == "error":
   292→            icon = f"[{RED}]✗[/]"
   293→            status_text = f"[{RED}]error[/]"
   294→        else:
   295→            icon = "○"
   296→            status_text = "idle"
   297→
   298→        self.update(
   299→            f"{icon} [{CYAN} bold]{name_short}[/]\n  {status_text}"
   300→        )
   301→
   302→    def set_status(self, status: str, count: int = 0) -> None:
   303→        self.status = status
   304→        self.finding_count = count
   305→        self._update_display()
   306→
   307→    def on_unmount(self) -> None:
   308→        if self._timer:
   309→            self._timer.stop()
   310→

   313→# Pulsing Status Dot Widget
   314→# ════════════════════════════════════════════════════════════════════════════════
   315→
   316→
   317→class StatusDot(Static):
   318→    """Pulsing dot indicator for scan status."""
   319→
   320→    scanning: reactive[bool] = reactive(False)
   321→
   322→    def __init__(self) -> None:
   323→        super().__init__("")
   324→        self._phase = 0
   325→
   326→    def on_mount(self) -> None:
   327→        self.set_interval(0.3, self._pulse)
   328→
   329→    def watch_scanning(self, old: bool, new: bool) -> None:
   330→        # Update visual state based on scanning status
   331→        if new:
   332→            self.update(f"[{self._get_pulse_color()}]●[/]")
   333→        else:
   334→            self.update(f"[{GREEN}]●[/]")
   335→
   336→    def _get_pulse_color(self) -> str:
   337→        return ["#003322", "#006644", CYAN, "#006644"][self._phase] if self.scanning else GREEN
   338→
   339→    def _pulse(self) -> None:
   340→        self._phase = (self._phase + 1) % 4
   341→        if self.scanning:
   342→            brightness = self._get_pulse_color()
   343→            self.update(f"[{brightness}]●[/]")
   344→        else:
   345→            self.update(f"[{GREEN}]●[/]")

   711→        """Handle key events — boot skip + history navigation."""

   732→                pass

   735→                inp = self.query_one("#command-input", Input)

   737→                    if self._history_index < len(self._command_history) - 1:
   738→                        self._history_index += 1
   739→                        inp.value = self._command_history[self._history_index]
   740→                    else:

   980→            return
   981→
   982→        if cmd == "help":
   983→            self._show_help()
   984→            return
   985→
   986→        if cmd == "scan":
   987→            self._handle_scan(args)
   988→            return
   989→
   990→        if cmd == "audit":
   991→            self._handle_audit(args)
   992→            return
   993→
   994→        if cmd == "dev":

   996→            return
   997→
   998→        if cmd == "doctor":
   999→            self._handle_doctor()
  1000→            return
  1001→
  1002→        if cmd == "blitz":
  1003→            self._handle_blitz(args)
  1004→            return

  1006→        if cmd == "subdomains":

  1009→
  1010→        if cmd == "agent":

  1106→    def _handle_scan(self, args: List[str]) -> None:
  1107→        if not args:
  1108→            self._chat(f"[{RED}]Usage: scan <target> [with <mod1> <mod2> ...][/]")
  1109→            return
  1110→
  1111→        # Parse "with" clause
  1112→        modules: Optional[List[str]] = None
  1113→        target_parts: List[str] = []
  1114→        found_with = False
  1115→        mod_parts: List[str] = []
  1116→        for a in args:
  1117→            if a.lower() == "with":
  1118→                found_with = True
  1119→                continue
  1120→            if found_with:
  1121→                mod_parts.append(a)
  1122→            else:
  1123→                target_parts.append(a)
  1124→
  1125→        target = " ".join(target_parts)
  1126→        if not target:
  1127→            self._chat(f"[{RED}]No target specified.[/]")
  1128→            return
  1129→
  1130→        if mod_parts:
  1131→            modules = [m.strip().lower() for m in mod_parts]
  1132→            self._chat(f"[{DIM_CYAN}]Scanning [cyan bold]{target}[/] with modules: {', '.join(modules)}[/]")
  1133→        else:
  1134→            self._chat(f"[{DIM_CYAN}]Scanning [cyan bold]{target}[/]...[/]")
  1135→
  1136→        self.current_target = target
  1137→        self._run_scan_worker(target, modules=modules)

  1218→    @work(exclusive=False, thread=True)
  1219→    def _run_scan_worker(
  1220→        self,
  1221→        target: str,
  1222→        modules: Optional[List[str]] = None,
  1223→        is_local: bool = False,
  1224→    ) -> None:
  1225→        """Run a scan in a background thread."""
  1226→        self.call_from_thread(self.is_scanning.set, True)
  1227→
  1228→        # Set modules to scanning
  1229→        if modules:
  1230→            for m in modules:
  1231→                if m in self._module_cells:
  1232→                    self.call_from_thread(self._module_cells[m].set_status, "scanning")
  1233→        else:
  1234→            all_mods = (
  1235→                list(MODULE_NAMES.keys())
  1236→                if is_local
  1237→                else [k for k in MODULE_NAMES if k not in ("host", "dev", "doctor")]
  1238→            )
  1239→            for m in all_mods:
  1240→                if m in self._module_cells:
  1241→                    self.call_from_thread(self._module_cells[m].set_status, "scanning")
  1242→
  1243→        try:
  1244→            from .scanner import scan, audit_scan
  1245→            from .history import save_scan
  1246→
  1247→            self.call_from_thread(
  1248→                self._chat, f"[{DIM_CYAN}]  ⏳ Scan in progress...[/]"
  1249→            )
  1250→
  1251→            if is_local:
  1252→                result = audit_scan(target=target, modules=modules)
  1253→            else:
  1254→                result = scan(target=target, modules=modules)
  1255→
  1256→            # Process findings
  1257→            data = result.to_dict()
  1258→            self.call_from_thread(self._on_scan_complete, data, modules, is_local)
  1259→
  1260→            # Save to history
  1261→            try:
  1262→                save_scan(data, label="nexus")
  1263→            except Exception:
  1264→                pass
  1265→
  1266→        except Exception as e:
  1267→            self.call_from_thread(self._chat, f"[{RED}]  ✗ Scan error: {e}[/]")
  1268→            if modules:
  1269→                for m in modules:
  1270→                    if m in self._module_cells:
  1271→                        self.call_from_thread(self._module_cells[m].set_status, "error")
  1272→            else:
  1273→                self.call_from_thread(self._set_all_modules_status, "error")
  1274→
  1275→        finally:
  1276→            self.call_from_thread(self.is_scanning.set, False)

  1629→    def on_click(self, event: Click) -> None:
  1630→        """Open finding detail modal for the clicked finding.
  1631→
  1632→        Uses the vertical click position to determine which finding index
  1633→        the user clicked on, mapping it to _findings_list.
  1634→        """
  1635→        try:
  1636→            if (
  1637→                isinstance(event.widget, RichLog)
  1638→                and event.widget.id == "findings-feed"
  1639→                and self._findings_list
  1640→            ):
  1641→                # Get the line offset to determine which finding was clicked.
  1642→                # Each finding entry is 1 line in the RichLog.
  1643→                # We use the y coordinate relative to the widget to estimate the index.
