    99→
   100→class FindingDetailModal(ModalScreen):
   101→    """Modal overlay showing full finding details."""
   102→
   103→    def __init__(self, finding: Dict[str, Any]) -> None:
   104→        super().__init__()
   105→        self.finding = finding
   106→
   107→    def compose(self) -> ComposeResult:
   108→        f = self.finding
   109→        sev = f.get("severity", "info").lower()
   110→        sev_color = SEV_STYLES.get(sev, "white")
   111→        title = f.get("title", "Unknown")
   112→        category = f.get("category", "")
   113→        module = f.get("module", "")
   114→        pts = f.get("points_deducted", 0)
   115→        evidence = f.get("evidence", "")
   116→        remediation = f.get("remediation", "No remediation available.")
   117→        description = f.get("description", "")
   118→        asset = f.get("asset", "")
   119→        related_cves = f.get("related_cves", [])
   120→
   121→        lines: List[str] = []
   122→        lines.append(f"[{sev_color}][{sev.upper()}] {title}[/]")
   123→        lines.append("")
   124→        if asset:
   125→            lines.append(f"[dim]Asset:[/] [cyan]{asset}[/]")
   126→        if category:
   127→            lines.append(f"[dim]Category:[/] {category}")
   128→        if module:
   129→            lines.append(f"[dim]Module:[/] {module}")
   130→        lines.append(f"[dim]Points:[/] -{pts}")
   131→        if related_cves:
   132→            cve_ids = ", ".join(
   133→                c.get("cve_id", "") for c in related_cves[:5] if c.get("cve_id")
   134→            )
   135→            if cve_ids:
   136→                lines.append(f"[dim]CVEs:[/] [red]{cve_ids}[/]")
   137→        lines.append("")
   138→        if description:
   139→            lines.append(f"[bold]Description:[/]")
   140→            lines.append(f"[dim]{description}[/]")
   141→            lines.append("")
   142→        if evidence:
   143→            lines.append(f"[bold]Evidence:[/]")
   144→            lines.append(f"[dim]{evidence}[/]")
   145→            lines.append("")
   146→        lines.append(f"[bold]{GREEN}Remediation:[/]")
   147→        lines.append(f"{GREEN}{remediation}[/]")
   148→
   149→        content = "\n".join(lines)
   150→
   151→        with Vertical(classes="modal-container"):
   152→            yield Label(content, classes="modal-content")
   153→            with Horizontal(classes="modal-actions"):
   154→                yield Button("Close", variant="primary", id="close-modal")
   155→
   156→    def on_button_pressed(self, event: Button.Pressed) -> None:
   157→        if event.button.id == "close-modal":
   158→            self.dismiss()
   159→

   161→# ════════════════════════════════════════════════════════════════════════════════
   162→# Boot Screen Widget
   163→# ════════════════════════════════════════════════════════════════════════════════
   164→
   165→
   166→class BootScreen(Widget):
   167→    """Full-screen animated boot sequence."""
   168→
   169→    def __init__(self) -> None:
   170→        super().__init__()
   171→        self._step = 0
   172→        self._spinner_idx = 0
   173→        self._module_idx = 0
   174→        self._module_names = list(MODULE_NAMES.values())
   175→        self._done = False
   176→        self._timer: Optional[Timer] = None
   177→
   178→    def compose(self) -> ComposeResult:
   179→        with Vertical(classes="boot-container"):
   180→            yield Static("", id="boot-line-1", classes="boot-title")
   181→            yield Static("", id="boot-line-2", classes="boot-line")
   182→            yield Static("", id="boot-line-3", classes="boot-line")
   183→            yield Static("", id="boot-line-4", classes="boot-line")
   184→
   185→    def on_mount(self) -> None:
   186→        self._timer = self.set_interval(0.04, self._tick)
   187→
   188→    def _tick(self) -> None:
   189→        app = self.app
   190→        if not isinstance(app, NexusApp):
   191→            return
   192→
   193→        line1 = app.query_one("#boot-line-1", Static)
   194→        line2 = app.query_one("#boot-line-2", Static)
   195→        line3 = app.query_one("#boot-line-3", Static)
   196→        line4 = app.query_one("#boot-line-4", Static)
   197→
   198→        if self._step == 0:
   199→            # Letter-by-letter title
   200→            title = "RECONPRO NEXUS"
   201→            chars = title[: self._spinner_idx]
   202→            self._spinner_idx += 1
   203→            line1.update(f"[{CYAN} bold]{chars}[/]")
   204→            if self._spinner_idx > len(title):
   205→                self._spinner_idx = 0
   206→                self._step = 1
   207→                line1.update(f"[{CYAN} bold]{title}[/]  [{DIM_CYAN}]v{VERSION}[/]")
   208→
   209→        elif self._step == 1:
   210→            # Spinner for "Initializing agent..."
   211→            frame = SPINNER_FRAMES[self._spinner_idx % len(SPINNER_FRAMES)]
   212→            self._spinner_idx += 1
   213→            line2.update(f"  [{CYAN}]{frame}[/] [{DIM_CYAN}]Initializing agent...[/]")
   214→            if self._spinner_idx > 30:
   215→                self._spinner_idx = 0
   216→                self._step = 2
   217→                line2.update(f"  [{GREEN}]✓[/] [{DIM_CYAN}]Agent initialized.[/]")
   218→
   219→        elif self._step == 2:
   220→            # Module names appearing one by one
   221→            if self._module_idx < len(self._module_names):
   222→                visible = self._module_names[: self._module_idx + 1]
   223→                self._module_idx += 1
   224→                modules_str = "  ".join(f"[{CYAN}]○[/] [{DIM_CYAN}]{m}[/]" for m in visible)
   225→                remaining = len(self._module_names) - self._module_idx
   226→                if remaining > 0:
   227→                    modules_str += f"  [{DIM_CYAN}]+{remaining} more...[/]"
   228→                frame = SPINNER_FRAMES[self._spinner_idx % len(SPINNER_FRAMES)]
   229→                self._spinner_idx += 1
   230→                line3.update(
   231→                    f"  [{CYAN}]{frame}[/] [{DIM_CYAN}]Loading {len(self._module_names)} modules...[/]\n  {modules_str}"
   232→                )
   233→            else:
   234→                all_mods = "  ".join(
   235→                    f"[{GREEN}]●[/] [{DIM_CYAN}]{m}[/]" for m in self._module_names
   236→                )
   237→                line3.update(
   238→                    f"  [{GREEN}]✓[/] [{DIM_CYAN}]All {len(self._module_names)} modules loaded.[/]\n  {all_mods}"
   239→                )
   240→                self._step = 3
   241→                self._spinner_idx = 0
   242→
   243→        elif self._step == 3:
   244→            # "Ready." with flash
   245→            if self._spinner_idx == 0:
   246→                line4.update(f"[{YELLOW} bold]  ▶  READY.[/]")
   247→            elif self._spinner_idx == 6:
   248→                line4.update(f"[{GREEN} bold]  ▶  READY.[/]")
   249→            elif self._spinner_idx >= 10:
   250→                self._done = True
   251→                if self._timer:
   252→                    self._timer.stop()
   253→                self.app.call_from_thread(self.app._finish_boot)
   254→            self._spinner_idx += 1
   255→

   257→# ════════════════════════════════════════════════════════════════════════════════
   258→# Module Status Cell Widget
   259→# ════════════════════════════════════════════════════════════════════════════════
   260→
   261→
   262→class ModuleCell(Static):
   263→    """Single module status cell in the grid."""
   264→
   265→    def __init__(self, module_id: str, name: str) -> None:
   266→        super().__init__()
   267→        self.module_id = module_id
   268→        self.module_name = name
   269→        self.status = "idle"  # idle | scanning | done | error
   270→        self.finding_count = 0
   271→        self._spinner_idx = 0
   272→        self._timer: Optional[Timer] = None
   273→
   274→    def on_mount(self) -> None:
   275→        self._update_display()
   276→        self._timer = self.set_interval(0.1, self._spin)
   277→
   278→    def _spin(self) -> None:
   279→        if self.status == "scanning":
   280→            self._spinner_idx = (self._spinner_idx + 1) % len(SPINNER_FRAMES)
   281→            self._update_display()
   282→
   283→    def _update_display(self) -> None:
   284→        name_short = self.module_name[:14]
   285→        if self.status == "idle":
   286→            icon = f"[{DIM_CYAN}]○[/]"
   287→            status_text = f"[{DIM_CYAN}]idle[/]"
   288→        elif self.status == "scanning":
   289→            frame = SPINNER_FRAMES[self._spinner_idx]
   290→            icon = f"[{CYAN}]{frame}[/]"
   291→            status_text = f"[{CYAN}]scanning[/]"
   292→        elif self.status == "done":
   293→            icon = f"[{GREEN}]✓[/]"
   294→            status_text = f"[{GREEN}]done ({self.finding_count})[/]"
   295→        elif self.status == "error":
   296→            icon = f"[{RED}]✗[/]"
   297→            status_text = f"[{RED}]error[/]"
   298→        else:
   299→            icon = "○"
   300→            status_text = "idle"
   301→
   302→        self.update(
   303→            f"{icon} [{CYAN} bold]{name_short}[/]\n  {status_text}"
   304→        )
   305→
   306→    def set_status(self, status: str, count: int = 0) -> None:
   307→        self.status = status
   308→        self.finding_count = count
   309→        self._update_display()
   310→
   311→    def on_unmount(self) -> None:
   312→        if self._timer:
   313→            self._timer.stop()
   314→

   316→# ════════════════════════════════════════════════════════════════════════════════
   317→# Pulsing Status Dot Widget
   318→# ════════════════════════════════════════════════════════════════════════════════
   319→
   320→
   321→class StatusDot(Static):
   322→    """Pulsing dot indicator for scan status."""
   323→
   324→    scanning: reactive[bool] = reactive(False)
   325→
   326→    def __init__(self) -> None:
   327→        super().__init__("")
   328→        self._phase = 0
   329→
   330→    def on_mount(self) -> None:
   331→        self.set_interval(0.3, self._pulse)
   332→
   333→    def watch_scanning(self, old: bool, new: bool) -> None:
   334→        # Update visual state based on scanning status
   335→        if new:
   336→            self.update(f"[{self._get_pulse_color()}]●[/]")
   337→        else:
   338→            self.update(f"[{GREEN}]●[/]")
   339→
   340→    def _get_pulse_color(self) -> str:
   341→        return ["#003322", "#006644", CYAN, "#006644"][self._phase] if self.scanning else GREEN
   342→
   343→    def _pulse(self) -> None:
   344→        self._phase = (self._phase + 1) % 4
   345→        if self.scanning:
   346→            brightness = self._get_pulse_color()
   347→            self.update(f"[{brightness}]●[/]")
   348→        else:
   349→            self.update(f"[{GREEN}]●[/]")

   765→    def on_key(self, event: Key) -> None:
   766→        """Handle key events — boot skip + completer + history + vim-nav."""
   767→        # Skip boot on ANY key press during boot sequence
   768→        if not self._boot_done and not self._boot_skipped:
   769→            self._boot_skipped = True
   770→            try:
   771→                boot = self.query_one("#boot-screen", BootScreen)
   772→                if boot._timer:
   773→                    boot._timer.stop()
   774→            except NoMatches:
   775→                pass
   776→            self._finish_boot()
   777→            return
   778→
   779→        # ── Number keys: quick jump to panels (0=input, 1=chat, 2=findings, 3=modules) ──
   780→        if event.key in ("0", "1", "2", "3") and not self._input_focused():
   781→            jump_map = {"0": "#command-input", "1": "#chat-log", "2": "#findings-feed", "3": "#module-grid"}
   782→            self._focus_widget(jump_map[event.key])
   783→            event.stop()
   784→            return
   785→
   786→        # ── Tab: completer accept/cycle (when input focused + completer visible) ──
   787→        if event.key == "tab":
   788→            try:
   789→                inp = self.query_one("#command-input", Input)
   790→                comp = self.query_one("#cmd-completer", CommandCompleter)
   791→                if inp.has_focus and comp.visible:
   792→                    accepted = comp.accept_top()
   793→                    if accepted:
   794→                        self._apply_completion(inp, comp, accepted)
   795→                    event.stop()
   796→                    return
   797→            except NoMatches:
   798→                pass
   799→            # Fall through to default Tab behavior (cycle focus)
   800→            return
   801→
   802→        # ── Escape: dismiss completer, close modal, or jump to input ──
   803→        if event.key == "escape":
   804→            try:
   805→                comp = self.query_one("#cmd-completer", CommandCompleter)
   806→                if comp.visible:
   807→                    comp.hide()
   808→                    event.stop()
   809→                    return
   810→            except NoMatches:
   811→                pass
   812→            # If any modal screen is open, Escape will dismiss it via Textual
   813→            # Otherwise jump back to input
   814→            if not self._input_focused():
   815→                self._focus_input()
   816→                event.stop()
   817→                return
   818→
   819→        # ── Up/Down in completer: cycle suggestions ──
   820→        if event.key in ("up", "down"):
   821→            try:
   822→                comp = self.query_one("#cmd-completer", CommandCompleter)
   823→                inp = self.query_one("#command-input", Input)
   824→                if comp.visible and inp.has_focus:
   825→                    direction = -1 if event.key == "up" else 1
   826→                    comp.cycle_selection(direction)
   827→                    event.stop()
   828→                    return
   829→            except NoMatches:
   830→                pass
   831→
   832→        # ── j/k vim-style navigation in findings feed ──
   833→        if event.key in ("j", "k"):
   834→            try:
   835→                feed = self.query_one("#findings-feed", RichLog)
   836→                if feed.has_focus and self._findings_list:
   837→                    self._vim_navigate_findings(event.key)
   838→                    event.stop()
   839→                    return
   840→            except NoMatches:
   841→                pass
   842→
   843→        # ── Enter on findings: open detail modal ──
   844→        if event.key == "enter":
   845→            try:
   846→                feed = self.query_one("#findings-feed", RichLog)
   847→                if feed.has_focus and self._findings_list:
   848→                    idx = self._findings_cursor if 0 <= self._findings_cursor < len(self._findings_list) else len(self._findings_list) - 1
   849→                    self.push_screen(FindingDetailModal(self._findings_list[idx]))
   850→                    event.stop()
   851→                    return
   852→            except NoMatches:
   853→                pass
   854→
   855→        # Command history navigation
   856→        if event.key == "up" and self._history_index > 0:
   857→            try:
   858→                inp = self.query_one("#command-input", Input)
   859→                if inp.has_focus:
   860→                    self._history_index -= 1
   861→                    inp.value = self._command_history[self._history_index]
   862→            except NoMatches:
   863→                pass
   864→        elif event.key == "down":
   865→            try:
   866→                inp = self.query_one("#command-input", Input)
   867→                if inp.has_focus:
   868→                    if self._history_index < len(self._command_history) - 1:
   869→                        self._history_index += 1
   870→                        inp.value = self._command_history[self._history_index]
   871→                    else:
   872→                        self._history_index = len(self._command_history)
   873→                        inp.value = ""
   874→            except NoMatches:
   875→                pass

   938→    def _apply_completion(self, inp: Input, comp: CommandCompleter, accepted: str) -> None:
   939→        """Apply a completion suggestion to the input field.
   940→
   941→        Phase C: handles both command-level and arg-level completions.
   942→        For arg mode, appends/replaces the last word instead of the first.
   943→        """
   944→        parts = inp.value.split()
   945→
   946→        if comp._mode == "command":
   947→            # Replace the first word with the accepted command
   948→            rest = parts[1:] if len(parts) > 1 else []
   949→            inp.value = accepted + (" " + " ".join(rest) if rest else "")
   950→        else:
   951→            # Arg-level: replace the last word
   952→            if len(parts) >= 2:
   953→                parts[-1] = accepted
   954→                inp.value = " ".join(parts)
   955→            else:
   956→                inp.value = accepted
   957→
   958→        inp.cursor_position = len(inp.value)

  1090→    def watch_finding_count(self, old: int, new: int) -> None:
  1091→        delta = new - old if old > 0 else 0
  1092→        crit = sum(
  1093→            1 for f in self._findings_list
  1094→            if f.get("severity", "").lower() == "critical"
  1095→        )
  1096→        high = sum(
  1097→            1 for f in self._findings_list
  1098→            if f.get("severity", "").lower() == "high"
  1099→        )
  1100→        med = sum(
  1101→            1 for f in self._findings_list
  1102→            if f.get("severity", "").lower() == "medium"
  1103→        )
  1104→
  1105→        # Update stat counters
  1106→        try:
  1107→            self.query_one("#stat-findings", StatCounter).set(new, delta=delta)
  1108→        except NoMatches:
  1109→            pass
  1110→        try:
  1111→            self.query_one("#stat-critical", StatCounter).set(crit)
  1112→        except NoMatches:
  1113→            pass
  1114→        try:
  1115→            self.query_one("#stat-high", StatCounter).set(high)
  1116→        except NoMatches:
  1117→            pass
  1118→        try:
  1119→            self.query_one("#stat-medium", StatCounter).set(med)
  1120→        except NoMatches:
  1121→            pass
  1122→
  1123→        # Track finding timestamps for true per-minute rate
  1124→        now = time.monotonic()
  1125→        for _ in range(delta):
  1126→            self._finding_timestamps.append(now)
  1127→
  1128→        # Push to findings sparkline (true findings-per-minute)
  1129→        try:
  1130→            spark = self.query_one("#spark-findings", Sparkline)
  1131→            # Calculate actual FPM from timestamps in the last 60s
  1132→            cutoff = now - 60.0
  1133→            self._finding_timestamps = [t for t in self._finding_timestamps if t > cutoff]
  1134→            fpm = len(self._finding_timestamps)
  1135→            spark.push(float(fpm))
  1136→        except NoMatches:
  1137→            pass
  1138→
  1139→        # Wire into velocity meter
  1140→        try:
  1141→            vm = self.query_one("#velocity-meter", VelocityMeter)
  1142→            if delta > 0:
  1143→                vm.record_finding(delta)
  1144→        except NoMatches:
  1145→            pass
  1146→

  1525→    def _handle_scan(self, args: List[str]) -> None:

  1529→
  1530→        # Parse "with" clause

  1544→        target = " ".join(target_parts)

  1546→            self._chat(f"[{RED}]No target specified.[/]")

  1549→        if mod_parts:
  1550→            modules = [m.strip().lower() for m in mod_parts]
  1551→            self._chat(f"[{DIM_CYAN}]Scanning [cyan bold]{target}[/] with modules: {', '.join(modules)}[/]")

  1553→            self._chat(f"[{DIM_CYAN}]Scanning [cyan bold]{target}[/]...[/]")

  1555→        self.current_target = target
  1556→        self._first_scan_done = True
  1557→        self._run_scan_worker(target, modules=modules)

  1638→    @work(exclusive=False, thread=True)
  1639→    def _run_scan_worker(
  1640→        self,
  1641→        target: str,
  1642→        modules: Optional[List[str]] = None,
  1643→        is_local: bool = False,
  1644→    ) -> None:
  1645→        """Run a scan in a background thread."""
  1646→        self.call_from_thread(self.is_scanning.set, True)
  1647→
  1648→        # Set modules to scanning
  1649→        if modules:
  1650→            for m in modules:
  1651→                if m in self._module_cells:
  1652→                    self.call_from_thread(self._module_cells[m].set_status, "scanning")
  1653→        else:
  1654→            all_mods = (
