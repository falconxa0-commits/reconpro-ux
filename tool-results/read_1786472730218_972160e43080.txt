   471→  1220→    def select_payloads(self, category: str) -> list[str]:

   473→  1222→
   474→  1223→        Priority logic:
   475→  1224→          1. If tech_profile is available, prefer framework-specific payloads.
   476→  1225→          2. Language-specific SQL payloads (e.g., Django -> postgres, WordPress -> mysql).
   477→  1226→          3. Framework-specific template payloads (e.g., Django -> jinja2/django SSTI).

   584→  1333→    def fuzz_parameter(
   585→  1334→        self,

   689→  1438→    def analyze_response(
   690→  1439→        self,
   691→  1440→        original_body: str,
   692→  1441→        fuzzed_body: str,
   693→  1442→        status_code: int,
   694→  1443→        response_time: float,
   695→  1444→    ) -> FuzzResult:
   696→  1445→        """Analyze a fuzzed response and classify the result.
   697→  1446→
   698→  1447→        Detection methods:
   699→  1448→          - Reflected: the payload itself appears in the response body.
   700→  1449→          - Error-based: SQL errors, stack traces, or framework errors in the body.
   701→  1450→          - Time-based: response took significantly longer than baseline.
   702→  1451→          - Blind: status code change (e.g., 500 instead of 200) without obvious error.
   703→  1452→        """

   706→  1455→        result.response_time = response_time
   707→  1456→        result.matched_pattern = ""
   708→  1457→
   709→  1458→        # --- Check for reflected payload -----------------------------------
   710→  1459→        if fuzzed_body and original_body:

   712→  1461→            payload_escaped = re.escape(result.payload)
   713→  1462→            if result.payload and re.search(payload_escaped, fuzzed_body):
   714→  1463→                result.classification = "reflected"
   715→  1464→                result.matched_pattern = "Payload reflected in response body"
   716→  1465→
   717→  1466→            # Also check for partial reflection or HTML entity encoding
   718→  1467→            if result.payload and not result.classification:
   719→  1468→                html_encoded = (
   720→  1469→                    result.payload.replace("<", "&lt;")
   721→  1470→                    .replace(">", "&gt;")
   722→  1471→                    .replace('"', "&quot;")
   723→  1472→                    .replace("'", "&#39;")
   724→  1473→                )
   725→  1474→                if html_encoded in fuzzed_body:
   726→  1475→                    result.classification = "reflected"
   727→  1476→                    result.matched_pattern = "Payload HTML-encoded in response"
   728→  1477→
   729→  1478→        # --- Check for SQL errors in the response --------------------------
   730→  1479→        if fuzzed_body:
   731→  1480→            for pattern in _SQL_ERROR_PATTERNS:
   732→  1481→                match = pattern.search(fuzzed_body)
   733→  1482→                if match:
   734→  1483→                    result.classification = "error_based"
   735→  1484→                    result.matched_pattern = match.group(0)
   736→  1485→                    break
   737→  1486→
   738→  1487→        # --- Check for stack traces ----------------------------------------
   739→  1488→        if fuzzed_body and result.classification != "error_based":
   740→  1489→            for pattern in _STACK_TRACE_PATTERNS:
   741→  1490→                match = pattern.search(fuzzed_body)
   742→  1491→                if match:
   743→  1492→                    result.classification = "error_based"
   744→  1493→                    result.matched_pattern = match.group(0)
   745→  1494→                    break

   747→  1496→        # --- Check for command injection output ---------------------------
   748→  1497→        if fuzzed_body and result.classification != "error_based":
   749→  1498→            match = _CMD_INJECTION_PATTERN.search(fuzzed_body)
   750→  1499→            if match:
   751→  1500→                result.classification = "error_based"
   752→  1501→                result.matched_pattern = match.group(0)
   753→  1502→
   754→  1503→        # --- Check for path traversal output --------------------------------
   755→  1504→        if fuzzed_body and result.classification != "error_based":
   756→  1505→            match = _PATH_TRAVERSAL_PATTERN.search(fuzzed_body)
   757→  1506→            if match:
   758→  1507→                result.classification = "error_based"
   759→  1508→                result.matched_pattern = match.group(0)
   760→  1509→
   761→  1510→        # --- Check for XSS indicators -------------------------------------
   762→  1511→        if fuzzed_body and result.classification != "reflected":
   763→  1512→            match = _XSS_PATTERN.search(fuzzed_body)
   764→  1513→            if match:
   765→  1514→                result.classification = "reflected"
   766→  1515→                result.matched_pattern = match.group(0)
   767→  1516→
   768→  1517→        # --- Check for SSTI indicators -------------------------------------
   769→  1518→        if fuzzed_body and result.classification not in ("reflected", "error_based"):
   770→  1519→            match = _SSTI_PATTERN.search(fuzzed_body)
   771→  1520→            if match:
   772→  1521→                result.classification = "reflected"
   773→  1522→                result.matched_pattern = match.group(0)
   774→  1523→
   775→  1524→        # --- Time-based detection ------------------------------------------
   776→  1525→        if self._base_time > 0 and response_time > self._base_time * 3:
   777→  1526→            if response_time > 4.0:
   778→  1527→                if result.classification in ("none", ""):
   779→  1528→                    result.classification = "time_based"
   780→  1529→                    result.matched_pattern = f"Response time {response_time:.2f}s (baseline {self._base_time:.2f}s)"
   781→  1530→                elif result.classification == "reflected":
   782→  1531→                    result.classification = "time_based"
   783→  1532→                    result.matched_pattern = f"Reflected + slow response ({response_time:.2f}s)"
   784→  1533→
   785→  1534→        # --- Status code change detection (blind) ---------------------------
   786→  1535→        if (
   787→  1536→            self._baseline_status > 0
   788→  1537→            and status_code != self._baseline_status
   789→  1538→            and status_code >= 400
   790→  1539→            and result.classification in ("none", "")
   791→  1540→        ):
   792→  1541→            result.classification = "blind"
   793→  1542→            result.matched_pattern = (
   794→  1543→                f"Status changed from {self._baseline_status} to {status_code}"
   795→  1544→            )
   796→  1545→
   797→  1546→        # --- Determine vulnerability ---------------------------------------
   798→  1547→        if result.classification in ("reflected", "error_based", "time_based"):
   799→  1548→            result.is_vulnerable = True
   800→  1549→        elif result.classification == "blind":
   801→  1550→            result.is_vulnerable = True
   802→  1551→
   803→  1552→        return result

   834→  1583→def quick_scan(
