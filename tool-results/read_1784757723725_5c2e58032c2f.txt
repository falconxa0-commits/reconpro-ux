   125→   125→// ─── Mini severity chart (SVG) ───────────────────────────────
   126→   126→
   127→   127→function SeverityDonut({ data }: { data: { name: string; value: number; color: string }[] }) {
   128→   128→  const total = data.reduce((s, d) => s + d.value, 0);
   129→   129→  const segments = useMemo(() => {
   130→   130→    if (total === 0) return [];
   131→   131→    let off = 0;
   132→   132→    return data.map((d) => {
   133→   133→      const pct = d.value / total;
   134→   134→      const seg = { ...d, pct, dasharray: `${pct * 283} ${283}`, offset: -off };
   135→   135→      off += pct * 283;
   136→   136→      return seg;
   137→   137→    });
   138→   138→  }, [data, total]);
   139→   139→  if (total === 0) return <div className="text-xs text-muted-foreground text-center py-4">No data yet</div>;
   140→   140→  return (
   141→   141→    <svg viewBox="0 0 120 120" className="w-32 h-32 mx-auto">
   142→   142→      {segments.map((d) => (
   143→   143→        <circle
   144→   144→          key={d.name}
   145→   145→          cx="60" cy="60" r="45"
   146→   146→          fill="none"
   147→   147→          stroke={d.color}
   148→   148→          strokeWidth="18"
   149→   149→          strokeDasharray={d.dasharray}
   150→   150→          strokeDashoffset={d.offset}
   151→   151→          strokeLinecap="round"
   152→   152→          opacity={0.85}
   153→   153→        />
   154→   154→      ))}
   155→   155→      <text x="60" y="56" textAnchor="middle" fill="#e6edf3" fontSize="22" fontWeight="bold" fontFamily="Geist Sans, sans-serif">{total}</text>
   156→   156→      <text x="60" y="72" textAnchor="middle" fill="#7d8590" fontSize="9" fontFamily="Geist Sans, sans-serif">FINDINGS</text>
   157→   157→    </svg>
   158→   158→  );
   159→   159→}
