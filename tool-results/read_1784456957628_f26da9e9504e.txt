   110→// ─── Mini severity chart (SVG) ───────────────────────────────
   111→
   112→function SeverityDonut({ data }: { data: { name: string; value: number; color: string }[] }) {
   113→  const total = data.reduce((s, d) => s + d.value, 0);
   114→  const segments = useMemo(() => {
   115→    if (total === 0) return [];
   116→    let off = 0;
   117→    return data.map((d) => {
   118→      const pct = d.value / total;
   119→      const seg = { ...d, pct, dasharray: `${pct * 283} ${283}`, offset: -off };
   120→      off += pct * 283;
   121→      return seg;
   122→    });
   123→  }, [data, total]);
   124→  if (total === 0) return <div className="text-xs text-muted-foreground text-center py-4">No data yet</div>;
   125→  return (
   126→    <svg viewBox="0 0 120 120" className="w-32 h-32 mx-auto">
   127→      {segments.map((d) => (
   128→        <circle
   129→          key={d.name}
   130→          cx="60" cy="60" r="45"
   131→          fill="none"
   132→          stroke={d.color}
   133→          strokeWidth="18"
   134→          strokeDasharray={d.dasharray}
   135→          strokeDashoffset={d.offset}
   136→          strokeLinecap="round"
   137→          opacity={0.85}
   138→        />
   139→      ))}
   140→      <text x="60" y="56" textAnchor="middle" fill="#e6edf3" fontSize="22" fontWeight="bold" fontFamily="Geist Sans, sans-serif">{total}</text>
   141→      <text x="60" y="72" textAnchor="middle" fill="#7d8590" fontSize="9" fontFamily="Geist Sans, sans-serif">FINDINGS</text>
   142→    </svg>
   143→  );
   144→}
