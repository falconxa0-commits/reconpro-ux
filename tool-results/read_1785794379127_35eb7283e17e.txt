   103→def heading(text, style="H1", level=0):
   104→    key = f"h_{hash(text) % 10000:04x}"
   105→    p = Paragraph(f'<a name="{key}"/>{text}</a>', style=style)
   106→    p.bookmark_name = key
   107→    p.bookmark_level = level
   108→    p.bookmark_text = text
   109→    p.bookmark_key = key
   110→    return p
   111→

   113→    return Paragraph(text, style="Body")

   116→    return Paragraph(text, style="Bullet")
