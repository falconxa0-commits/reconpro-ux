   254→   283→   283→  1280→        self,
   255→   284→   284→  1281→        target: str,
   256→   285→   285→  1282→        modules: Optional[List[str]] = None,
   257→   286→   286→  1283→        is_local: bool = False,

   261→   290→   290→  1287→

   265→   294→   294→  1291→                if m in self._module_cells:
   266→   295→   295→  1292→                    self.call_from_thread(self._module_cells[m].set_status, "scanning")
   267→   296→   296→  1293→        else:
   268→   297→   297→  1294→            all_mods = (
   269→   298→   298→  1295→                list(MODULE_NAMES.keys())
   270→   299→   299→  1296→                if is_local
   271→   300→   300→  1297→                else [k for k in MODULE_NAMES if k not in ("host", "dev", "doctor")]
   272→   301→   301→  1298→            )
   273→   302→   302→  1299→            for m in all_mods:
   274→   303→   303→  1300→                if m in self._module_cells:
   275→   304→   304→  1301→                    self.call_from_thread(self._module_cells[m].set_status, "scanning")
   276→   305→   305→  1302→
   277→   306→   306→  1303→        try:
   278→   307→   307→  1304→            from .scanner import scan, audit_scan
   279→   308→   308→  1305→            from .history import save_scan
   280→   309→   309→  1306→
   281→   310→   310→  1307→            self.call_from_thread(
   282→   311→   311→  1308→                self._chat, f"[{DIM_CYAN}]  ⏳ Scan in progress...[/]"
   283→   312→   312→  1309→            )
   284→   313→   313→  1310→
   285→   314→   314→  1311→            if is_local:
   286→   315→   315→  1312→                result = audit_scan(target=target, modules=modules)

   290→   319→   319→  1316→            # Process findings

   296→   325→   325→  1322→                save_scan(data, label="nexus")
   297→   326→   326→  1323→            except Exception:
   298→   327→   327→  1324→                pass
   299→   328→   328→  1325→
   300→   329→   329→  1326→        except Exception as e:
   301→   330→   330→  1327→            self.call_from_thread(self._chat, f"[{RED}]  ✗ Scan error: {e}[/]")
   302→   331→   331→  1328→            if modules:
   303→   332→   332→  1329→                for m in modules:
   304→   333→   333→  1330→                    if m in self._module_cells:

   308→   337→   337→  1334→
