    35→def _scan_dir(base_dir: str = ".") -> str:
    36→    """Get absolute path of directory to scan."""
    37→    return os.path.abspath(base_dir)

    40→def _find_files(patterns: List[str], base: str) -> List[str]:
    41→    """Find files matching patterns in base directory."""
    42→    found = []
    43→    for pattern in patterns:
    44→        found.extend(glob.glob(os.path.join(base, "**", pattern), recursive=True))
    45→    return found

    48→def _read_file(path: str, max_size: int = 65536) -> str:
    49→    """Read file contents safely."""
    50→    try:
    51→        size = os.path.getsize(path)
    52→        if size > max_size:
    53→            return f"[file too large: {size} bytes]"
    54→        with open(path, errors="replace") as f:
    55→            return f.read()
    56→    except Exception:
    57→        return ""

    60→def _check_package_json(base: str) -> List[Finding]:
    61→    """Scan package.json for security issues."""
    62→    findings: List[Finding] = []
    63→    asset = base
    64→
    65→    pkg_files = _find_files(["package.json"], base)
    66→    for pkg_path in pkg_files:
    67→        # Skip node_modules
    68→        if "node_modules" in pkg_path:
    69→            continue
    70→
    71→        try:
    72→            with open(pkg_path) as f:
    73→                pkg = json.load(f)
    74→        except Exception:
    75→            continue
    76→
    77→        deps = {}
    78→        deps.update(pkg.get("dependencies", {}))
    79→        deps.update(pkg.get("devDependencies", {}))
    80→
    81→        if not deps:
    82→            continue
    83→
    84→        # Check for known vulnerable/outdated patterns
    85→        risky_pkgs = {
    86→            "lodash": ("low", "Consider replacing individual lodash imports with native JS"),
    87→            "express": ("info", "Ensure Express is >= 4.18.0"),
    88→            "jsonwebtoken": ("medium", "Ensure jsonwebtoken is >= 9.0.0"),
    89→            "react": ("info", "Keep React updated"),
    90→            "next": ("info", "Keep Next.js updated"),
    91→            "mongoose": ("info", "Ensure Mongoose is latest"),
    92→            "minimist": ("low", "minimist had prototype pollution CVEs"),
    93→            "uglify-js": ("low", "uglify-js had ReDoS vulnerabilities"),
    94→            "event-stream": ("high", "event-stream was compromised in 2018"),
    95→        }
    96→
    97→        for pkg_name, (sev, note) in risky_pkgs.items():
    98→            if pkg_name in deps:
    99→                findings.append(Finding(
   100→                    title=f"{pkg_name}@{deps[pkg_name]} in {os.path.basename(os.path.dirname(pkg_path))}",
   101→                    severity=sev, category="dependencies",
   102→                    module="dev",
   103→                    description=f"{note}.",
   104→                    evidence=f"{pkg_name}: {deps[pkg_name]}",
   105→                    asset=asset, points_deducted=3 if sev == "medium" else (8 if sev == "high" else 1),
   106→                    remediation=f"Run: npm audit fix && npm update {pkg_name}",
   107→                ))
   108→
   109→        # Check for scripts that might run untrusted code
   110→        scripts = pkg.get("scripts", {})
   111→        risky_scripts = {"preinstall", "postinstall", "prestart", "poststart"}
   112→        for script_name in risky_scripts:
   113→            if script_name in scripts:
   114→                findings.append(Finding(
   115→                    title=f"Lifecycle script '{script_name}' in {os.path.basename(os.path.dirname(pkg_path))}/package.json",
   116→                    severity="medium", category="supply_chain",
   117→                    module="dev",
   118→                    description="Package lifecycle scripts can run arbitrary code during npm install. Review the script content.",
   119→                    evidence=f"{script_name}: {scripts[script_name][:100]}",
   120→                    asset=asset, points_deducted=4,
   121→                    remediation="Review and minimize lifecycle scripts. Avoid curl | bash patterns.",
   122→                ))
   123→
   124→    return findings

   127→def _check_requirements(base: str) -> List[Finding]:
   128→    """Scan requirements.txt / Pipfile for security issues."""
   129→    findings: List[Finding] = []
   130→    asset = base
   131→
   132→    req_files = _find_files(["requirements.txt", "Pipfile", "pyproject.toml"], base)
   133→    for req_path in req_files:
   134→        if "node_modules" in req_path or ".venv" in req_path or "venv/" in req_path:
   135→            continue
   136→
   137→        content = _read_file(req_path)
   138→        if not content:
   139→            continue
   140→
   141→        risky_py_pkgs = {
   142→            "django": ("info", "Keep Django updated (check for CVEs)"),
   143→            "flask": ("info", "Keep Flask updated"),
   144→            "requests": ("info", "Ensure requests >= 2.31.0"),
   145→            "pyjwt": ("medium", "Ensure PyJWT >= 2.8.0"),
   146→            "cryptography": ("info", "Keep cryptography updated"),
   147→            "sqlalchemy": ("info", "Keep SQLAlchemy updated"),
   148→            "paramiko": ("medium", "Ensure Paramiko is latest"),
   149→            "pickle": ("high", "pickle is unsafe for untrusted data"),
   150→        }
   151→
   152→        for pkg, (sev, note) in risky_py_pkgs.items():
   153→            # Match package name at start of line or after common separators
   154→            if re.search(rf'^{re.escape(pkg)}[=<>!\[]', content, re.MULTILINE | re.IGNORECASE):
   155→                findings.append(Finding(
   156→                    title=f"{pkg} in {os.path.basename(os.path.dirname(req_path))}/{os.path.basename(req_path)}",
   157→                    severity=sev, category="dependencies",
   158→                    module="dev",
   159→                    description=note,
   160→                    evidence=f"Found in {req_path}",
   161→                    asset=asset, points_deducted=3 if sev == "medium" else (8 if sev == "high" else 1),
   162→                    remediation=f"Run: pip audit && pip install --upgrade {pkg}",
   163→                ))
   164→
   165→    return findings

   168→def _check_env_files(base: str) -> List[Finding]:
   169→    """Find .env files and scan for secrets."""
   170→    findings: List[Finding] = []
   171→    asset = base
   172→
   173→    env_files = _find_files([
   174→        ".env", ".env.local", ".env.production", ".env.development",
   175→        ".env.staging", ".env.test", ".env.backup",
   176→    ], base)
   177→
   178→    for env_path in env_files:
   179→        if "node_modules" in env_path or ".venv" in env_path:
   180→            continue
   181→
   182→        content = _read_file(env_path)
   183→        if not content:
   184→            continue
   185→
   186→        # Check if .env is in .gitignore
   187→        parent = os.path.dirname(env_path)
   188→        gitignore = os.path.join(parent, ".gitignore")
   189→        gitignored = False
   190→        if os.path.exists(gitignore):
   191→            gi_content = _read_file(gitignore)
   192→            if ".env" in gi_content:
   193→                gitignored = True
   194→
   195→        if not gitignored:
   196→            findings.append(Finding(
   197→                title=f".env file not in .gitignore: {env_path}",
   198→                severity="critical", category="env_files",
   199→                module="dev",
   200→                description=f"{os.path.basename(env_path)} exists but .env is not in .gitignore. This file may be committed to git.",
   201→                evidence=f"File: {env_path}",
   202→                asset=asset, points_deducted=15,
   203→                remediation=f"Add '.env' and '.env.*' to .gitignore in {parent}",
   204→            ))
   205→
   206→        # Scan for actual secrets in .env
   207→        secret_count = 0
   208→        for line in content.splitlines():
   209→            line = line.strip()
   210→            if not line or line.startswith("#"):
   211→                continue
   212→            if "=" not in line:
   213→                continue
   214→            key = line.split("=", 1)[0].upper()
   215→            val = line.split("=", 1)[1].strip()
   216→
   217→            secret_keywords = [
   218→                "KEY", "SECRET", "TOKEN", "PASSWORD", "PASSWD",
   219→                "CREDENTIAL", "PRIVATE", "CERT", "API_KEY", "AUTH",
   220→            ]
   221→            if any(kw in key for kw in secret_keywords) and val and val != '""' and val != "''":
   222→                secret_count += 1
   223→                if secret_count <= 5:
   224→                    masked = val[:8] + "..." if len(val) > 8 else "***"
   225→                    findings.append(Finding(
   226→                        title=f"Secret in {os.path.basename(env_path)}: {key}",
   227→                        severity="high", category="env_secrets",
   228→                        module="dev",
   229→                        description="Potential secret found in .env file.",
   230→                        evidence=f"{key}={masked}",
   231→                        asset=asset, points_deducted=6,
   232→                        remediation="Use a secrets manager for production. Never commit .env files.",
   233→                    ))
   234→
   235→    return findings

   238→def _check_git_security(base: str) -> List[Finding]:
   239→    """Check git configuration and security."""
   240→    findings: List[Finding] = []
   241→    asset = base
   242→
   243→    # Find git repos
   244→    git_dirs = _find_files([".git"], base)
   245→    git_repos = [os.path.dirname(d) for d in git_dirs if os.path.isdir(d)]
   246→
   247→    for repo in git_repos[:10]:  # Limit to 10 repos
   248→        repo_name = os.path.basename(repo)
   249→
   250→        # Check for committed secrets in git log
   251→        try:
   252→            r = subprocess.run(
   253→                ["git", "-C", repo, "log", "--all", "--oneline", "-20"],
   254→                capture_output=True, text=True, timeout=5
   255→            )
   256→            if r.returncode == 0 and r.stdout.strip():
   257→                findings.append(Finding(
   258→                    title=f"Git repo: {repo_name} ({len(r.stdout.strip().splitlines())} recent commits)",
   259→                    severity="info", category="git",
   260→                    module="dev",
   261→                    description=f"Found git repository at {repo}.",
   262→                    evidence=repo,
   263→                    asset=asset, points_deducted=0,
   264→                    remediation="",
   265→                ))
   266→        except Exception:
   267→            pass
   268→
   269→        # Check for pre-commit hooks
   270→        precommit = os.path.join(repo, ".git", "hooks", "pre-commit")
   271→        if os.path.exists(precommit):
   272→            pass  # pre-commit exists, good
   273→        else:
   274→            findings.append(Finding(
   275→                title=f"No pre-commit hooks in {repo_name}",
   276→                severity="low", category="git",
   277→                module="dev",
   278→                description="No pre-commit hooks found. Consider adding pre-commit for secret scanning and linting.",
   279→                evidence=f"No .git/hooks/pre-commit in {repo}",
   280→                asset=asset, points_deducted=2,
   281→                remediation="Install pre-commit: pip install pre-commit && pre-commit install",
   282→            ))
   283→
   284→        # Check for .gitconfig issues
   285→        gitconfig = os.path.expanduser("~/.gitconfig")
   286→        if os.path.exists(gitconfig):
   287→            gc = _read_file(gitconfig)
   288→            if "credential" in gc.lower() and "store" in gc.lower():
   289→                findings.append(Finding(
   290→                    title="Git credential store is enabled",
   291→                    severity="medium", category="git",
   292→                    module="dev",
   293→                    description="Git credentials are stored in plaintext on disk.",
   294→                    evidence="credential helper = store in ~/.gitconfig",
   295→                    asset=asset, points_deducted=5,
   296→                    remediation="Use git-credential-manager or cache instead of store.",
   297→                ))
   298→
   299→    return findings

   302→def _check_sensitive_files(base: str) -> List[Finding]:
   303→    """Find sensitive files in the project."""
   304→    findings: List[Finding] = []
   305→    asset = base
   306→
   307→    sensitive_patterns = [
   308→        ("*.pem", "high", "PEM certificate/key file"),
   309→        ("*.p12", "high", "PKCS12 certificate file"),
   310→        ("*.pfx", "high", "PFX certificate file"),
   311→        ("*.key", "high", "Private key file"),
   312→        ("*.jks", "high", "Java keystore"),
   313→        ("*.keystore", "high", "Keystore file"),
   314→        ("id_rsa*", "critical", "SSH private key"),
   315→        ("id_ed25519*", "critical", "SSH Ed25519 private key"),
   316→        ("*.credentials", "high", "Credentials file"),
   317→        ("*.htpasswd", "medium", "Apache password file"),
   318→        ("*.sql", "medium", "SQL dump file (may contain data)"),
   319→        ("*.dump", "medium", "Database dump file"),
   320→        ("*.bak", "low", "Backup file"),
   321→        (".DS_Store", "info", "macOS metadata file"),
   322→        ("Thumbs.db", "info", "Windows thumbnail cache"),
   323→    ]
   324→
   325→    for pattern, sev, desc in sensitive_patterns:
   326→        files = _find_files([pattern], base)
   327→        # Skip common non-sensitive locations
   328→        for f in files:
   329→            if any(skip in f for skip in ["node_modules", ".venv", "venv/", ".git/objects"]):
   330→                continue
   331→            findings.append(Finding(
   332→                title=f"Sensitive file: {f}",
   333→                severity=sev, category="sensitive_files",
   334→                module="dev",
   335→                description=f"{desc} found in project. Ensure this is not committed to version control.",
   336→                evidence=f"File: {f}",
   337→                asset=asset, points_deducted=8 if sev in ("critical", "high") else (4 if sev == "medium" else 1),
   338→                remediation="Add to .gitignore. Move secrets to a secure vault.",
   339→            ))
   340→
   341→    return findings

   344→def _check_hardcoded_secrets(base: str) -> List[Finding]:
   345→    """Scan source code for hardcoded secrets."""
   346→    findings: List[Finding] = []
   347→    asset = base
   348→
   349→    # File extensions to scan
   350→    code_exts = [".js", ".ts", ".tsx", ".jsx", ".py", ".rb", ".go", ".rs",
   351→                ".java", ".php", ".yml", ".yaml", ".json", ".toml", ".cfg", ".ini"]
   352→
   353→    secret_patterns = [
   354→        (r'(?i)(?:api[_-]?key|apikey)\s*[:=]\s*["\']([\w\-]{20,})', "Hardcoded API key"),
   355→        (r'(?i)(?:secret[_-]?key|secretkey)\s*[:=]\s*["\']([\w\-]{20,})', "Hardcoded secret key"),
   356→        (r'(?i)password\s*[:=]\s*["\']([^"\']{6,})', "Hardcoded password"),
   357→        (r'(?i)token\s*[:=]\s*["\']([\w\-\.=]{20,})', "Hardcoded token"),
   358→        (r'(?i)aws_access_key_id\s*[:=]\s*["\'](AKIA[\w]{16})', "Hardcoded AWS Access Key"),
   359→        (r'(?i)aws_secret_access_key\s*[:=]\s*["\']([\w/+=]{40})', "Hardcoded AWS Secret Key"),
   360→        (r'ghp_[\w]{36}', "Hardcoded GitHub PAT"),
   361→        (r'gho_[\w]{36}', "Hardcoded GitHub OAuth token"),
   362→        (r'ghs_[\w]{36}', "Hardcoded GitHub App token"),
   363→        (r'glpat-[\w\-]{20,}', "Hardcoded GitLab PAT"),
   364→        (r'xox[bposa]-[\w\-]{10,}', "Hardcoded Slack token"),
   365→        (r'np\.[\w]{36}', "Hardcoded npm token"),
   366→        (r'-----BEGIN (?:RSA |EC |DSA )?PRIVATE KEY-----', "Hardcoded private key"),
   367→    ]
   368→
   369→    for ext in code_exts:
   370→        files = _find_files([f"*{ext}"], base)
   371→        for filepath in files[:50]:  # Limit scan
   372→            if any(skip in filepath for skip in ["node_modules", ".venv", "venv/", ".git/", "dist/", "build/", ".next/"]):
   373→                continue
   374→
   375→            content = _read_file(filepath, max_size=32768)
   376→            if not content:
   377→                continue
   378→
   379→            for pattern, title in secret_patterns:
   380→                matches = re.findall(pattern, content)
   381→                if matches:
   382→                    for match in matches[:2]:
   383→                        if isinstance(match, tuple):
   384→                            match = match[0] if match else "***"
   385→                        masked = match[:8] + "..." + match[-4:] if len(str(match)) > 12 else "***"
   386→                        findings.append(Finding(
   387→                            title=f"{title} in {os.path.basename(filepath)}",
   388→                            severity="critical", category="hardcoded_secrets",
   389→                            module="dev",
   390→                            description=f"Secret detected in source code: {title}",
   391→                            evidence=f"Match: {masked}",
   392→                            asset=asset, points_deducted=15,
   393→                            remediation="Move secrets to environment variables or a secrets manager. Never commit secrets to source code.",
   394→                        ))
   395→                    break  # One match per pattern per file
   396→
   397→    return findings

   400→def _check_docker_compose(base: str) -> List[Finding]:
   401→    """Check Docker Compose files for security issues."""
   402→    findings: List[Finding] = []
   403→    asset = base
   404→
   405→    compose_files = _find_files(["docker-compose.yml", "docker-compose.yaml", "compose.yml", "compose.yaml"], base)
   406→
   407→    for cf in compose_files:
   408→        if "node_modules" in cf:
   409→            continue
   410→
   411→        content = _read_file(cf)
   412→        if not content:
   413→            continue
   414→
   415→        # Check for privileged mode
   416→        if re.search(r'privileged\s*:\s*true', content):
   417→            findings.append(Finding(
   418→                title=f"Privileged container in {os.path.basename(cf)}",
   419→                severity="high", category="docker",
   420→                module="dev",
   421→                description="A container is running in privileged mode, giving it full host access.",
   422→                evidence=f"privileged: true found in {cf}",
   423→                asset=asset, points_deducted=10,
   424→                remediation="Remove privileged: true. Use specific capabilities instead.",
   425→            ))
   426→
   427→        # Check for exposed sensitive ports
   428→        exposed_ports = re.findall(r'(\d{4,5})\s*:\s*\d{4,5}', content)
   429→        sensitive_exposed = []
   430→        for port in exposed_ports:
   431→            p = int(port)
   432→            if p in (22, 3306, 5432, 6379, 27017, 9200):
   433→                sensitive_exposed.append(str(p))
   434→
   435→        if sensitive_exposed:
   436→            findings.append(Finding(
   437→                title=f"Sensitive ports exposed in {os.path.basename(cf)}: {', '.join(sensitive_exposed[:5])}",
   438→                severity="medium", category="docker",
   439→                module="dev",
   440→                description="Database or admin ports are exposed to the host network.",
   441→                evidence=f"Ports: {', '.join(sensitive_exposed[:5])}",
   442→                asset=asset, points_deducted=5,
   443→                remediation="Only expose necessary ports. Bind databases to internal networks only.",
   444→            ))
   445→
   446→        # Check for env vars with secrets
   447→        env_vars = re.findall(r'([A-Z_]*(?:KEY|SECRET|TOKEN|PASSWORD|PASSWD)[A-Z_]*)\s*:\s*(.+)', content)
   448→        for var_name, val in env_vars[:5]:
   449→            val = val.strip()
   450→            if val and val not in ('""', "''", '${', 'false', 'true'):
   451→                findings.append(Finding(
   452→                    title=f"Secret in {os.path.basename(cf)}: {var_name}",
   453→                    severity="medium", category="docker",
   454→                    module="dev",
   455→                    description="Secret value found directly in compose file.",
   456→                    evidence=f"{var_name}={val[:15]}...",
   457→                    asset=asset, points_deducted=4,
   458→                    remediation="Use .env file or Docker secrets instead of inline values.",
   459→                ))
   460→
   461→    return findings

   569→def _version_gte(installed: str, minimum: str) -> bool:
   570→    """Check if installed version >= minimum (both semver-ish strings)."""
   571→    def _parse(v: str) -> Tuple[int, ...]:
   572→        parts = []
   573→        for p in re.split(r'[.\-]', v)[:3]:
   574→            try:
   575→                parts.append(int(p))
   576→            except ValueError:
   577→                parts.append(0)
   578→        return tuple(parts)
   579→    return _parse(installed) >= _parse(minimum)

   582→def _check_lockfile_cves(base: str) -> List[Finding]:
   583→    """Parse lockfiles for known vulnerable packages."""
   584→    findings: List[Finding] = []
   585→    asset = base
   586→
   587→    # Collect all (package, version) pairs from lockfiles
   588→    pkg_versions: Dict[str, List[str]] = {}  # pkg_name -> [version strings]
   589→
   590→    # package-lock.json
   591→    lock_files = _find_files(["package-lock.json"], base)
   592→    for lf in lock_files:
   593→        if "node_modules" in lf:
   594→            continue
   595→        try:
   596→            with open(lf) as f:
   597→                data = json.load(f)
   598→            _extract_npm_lockfile(data, pkg_versions, lf)
   599→        except Exception:
   600→            pass
   601→
   602→    # yarn.lock
   603→    yarn_files = _find_files(["yarn.lock"], base)
   604→    for yf in yarn_files:
   605→        if "node_modules" in yf:
   606→            continue
   607→        try:
   608→            content = _read_file(yf)
   609→            _extract_yarn_lockfile(content, pkg_versions, yf)
   610→        except Exception:
   611→            pass
   612→
   613→    # Pipfile.lock
   614→    pipfile_locks = _find_files(["Pipfile.lock"], base)
   615→    for pf in pipfile_locks:
   616→        if ".venv" in pf:
   617→            continue
   618→        try:
   619→            with open(pf) as f:
   620→                data = json.load(f)
   621→            for section in ("develop", "default"):
   622→                for pkg_name, pkg_info in data.get(section, {}).items():
   623→                    if isinstance(pkg_info, dict) and "version" in pkg_info:
   624→                        ver = pkg_info["version"].lstrip("=")
   625→                        pkg_versions.setdefault(pkg_name.lower(), []).append(ver)
   626→        except Exception:
   627→            pass
   628→
   629→    # Check collected packages against CVE database
   630→    for pkg_name, versions in pkg_versions.items():
   631→        for cve in _LOCKFILE_CVE_DB:
   632→            if cve["pkg"].lower() == pkg_name.lower():
   633→                for ver in versions:
   634→                    if not _version_gte(ver, cve["max_ver"]):
   635→                        findings.append(Finding(
   636→                            title=f"{cve['pkg']}@{ver} vulnerable to {cve['cve']}",
   637→                            severity=cve["severity"], category="lockfile_cve",
   638→                            module="dev",
   639→                            description=f"{cve['desc']}. Installed version {ver} is below fixed version {cve['max_ver']}.",
   640→                            evidence=f"{cve['pkg']}=={ver}, {cve['cve']}",
   641→                            asset=asset, points_deducted=10 if cve["severity"] == "high" else 5,
   642→                            remediation=f"Update {cve['pkg']} to >= {cve['max_ver']}",
   643→                        ))
   644→                        break  # One finding per package
   645→                break  # One CVE match per package
   646→
   647→    return findings

   650→def _extract_npm_lockfile(data: Any, pkg_versions: Dict[str, List[str]], source: str) -> None:
   651→    """Recursively extract package versions from package-lock.json."""
   652→    packages = data.get("packages", data.get("dependencies", {}))
   653→    if isinstance(packages, dict):
   654→        for pkg_path, info in packages.items():
   655→            if isinstance(info, dict) and "version" in info:
   656→                # Extract package name from path like "node_modules/lodash"
   657→                name = pkg_path.split("/")[-1] if "/" in pkg_path else pkg_path
   658→                # Handle scoped packages like "@scope/pkg"
   659→                if pkg_path.startswith("@"):
   660→                    name = pkg_path
   661→                pkg_versions.setdefault(name.lower(), []).append(info["version"])
   662→

   664→def _extract_yarn_lockfile(content: str, pkg_versions: Dict[str, List[str]], source: str) -> None:
   665→    """Extract package versions from yarn.lock format."""
   666→    # yarn.lock format: "pkg_name@version:" followed by version "version X.Y.Z"
   667→    blocks = re.split(r'\n(?=\S)', content)
   668→    for block in blocks:
   669→        header_match = re.match(r'^([^"\s]+|"[^"]+")@', block)
   670→        if not header_match:
   671→            continue
   672→        pkg_name = header_match.group(1).strip('"')
   673→        ver_match = re.search(r'version\s+"([^"]+)"', block)
   674→        if ver_match:
   675→            pkg_versions.setdefault(pkg_name.lower(), []).append(ver_match.group(1))

   796→def _check_git_config_credential_leak(base: str) -> List[Finding]:
   797→    """Check .git/config for embedded credentials in remote URLs."""
   798→    findings: List[Finding] = []
   799→    asset = base
   800→
   801→    # Find all .git/config files
   802→    git_dirs = _find_files([".git"], base)
   803→    git_repos = [os.path.dirname(d) for d in git_dirs if os.path.isdir(d)]
   804→
   805→    for repo in git_repos[:10]:
   806→        config_path = os.path.join(repo, ".git", "config")
   807→        if not os.path.isfile(config_path):
   808→            continue
   809→
   810→        try:
   811→            with open(config_path, errors="replace") as f:
   812→                content = f.read()
   813→        except Exception:
   814→            continue
   815→
   816→        # Check remote URLs for embedded credentials
   817→        # Patterns: https://user:pass@host, ssh://user:pass@host, git://user:pass@host
   818→        cred_patterns = [
   819→            (r'https?://[^\s]+:[^\s]+@[^\s]+', "HTTPS URL with embedded credentials"),
   820→            (r'ssh://[^\s]+:[^\s]+@[^\s]+', "SSH URL with embedded credentials"),
   821→            (r'git://[^\s]+:[^\s]+@[^\s]+', "Git protocol URL with embedded credentials"),
   822→        ]
   823→
   824→        for pattern, description in cred_patterns:
   825→            matches = re.findall(pattern, content, re.IGNORECASE)
   826→            if matches:
   827→                # Mask the actual credential
   828→                masked = re.sub(r':([^@]+)@', ':***@', matches[0])
   829→                findings.append(Finding(
   830→                    title=f"Credential in .git/config remote URL: {os.path.basename(repo)}",
   831→                    severity="critical", category="git_credential_leak",
   832→                    module="dev",
   833→                    description=f"{description} found in .git/config. Credentials should not be embedded in remote URLs.",
   834→                    evidence=f"Remote URL: {masked}",
   835→                    asset=asset, points_deducted=15,
   836→                    remediation="Remove credentials from the URL. Use SSH keys or a credential helper: git remote set-url origin <clean-url>",
   837→                ))
   838→                break  # One finding per repo
   839→
   840→    return findings

   893→# ---------------------------------------------------------------------------
   894→# NEW CHECK 5: GitHub Actions Token Exposure
   895→# ---------------------------------------------------------------------------
   896→
   897→_ACTIONS_SECRET_PATTERNS = [
   898→    (r'(?i)(?:secrets|env)\s*:\s*\n(?:[^\n]*\n)*?(?:([A-Z_]*(?:KEY|SECRET|TOKEN|PASSWORD|CREDENTIAL|AUTH)[A-Z_]*)\s*:\s*([^\n]+))',
   899→     "Secret exposed in workflow env/secrets"),
   900→    (r'(?i)GH_TOKEN\s*:\s*["\']?([\w]{20,})', "GH_TOKEN exposed in workflow"),
   901→    (r'(?i)ACTIONS_DEPLOY_KEY\s*:\s*["\']?([\w\-]{20,})', "Deploy key exposed in workflow"),
   902→    (r'(?i)NPM_TOKEN\s*:\s*["\']?([\w\.]{20,})', "npm token exposed in workflow"),
   903→    (r'(?i)SLACK_WEBHOOK\s*:\s*["\']?(https://hooks\.[^"\'\s]+)', "Slack webhook exposed in workflow"),
   904→]
   905→
   906→
   907→def _check_github_actions_tokens(base: str) -> List[Finding]:
   908→    """Check .github/workflows/ for secrets exposed in env vars."""
   909→    findings: List[Finding] = []
   910→    asset = base
   911→
   912→    workflow_dir = os.path.join(base, ".github", "workflows")
   913→    if not os.path.isdir(workflow_dir):
   914→        return findings
   915→
   916→    workflow_files = _find_files(["*.yml", "*.yaml"], workflow_dir)
   917→    for wf in workflow_files:
   918→        content = _read_file(wf)
   919→        if not content:
   920→            continue
   921→
   922→        # Check for secrets in env: blocks that use plain values instead of ${{ secrets.X }}
   923→        # Look for env: sections with hardcoded values
   924→        env_sections = re.finditer(r'^\s*env:\s*$', content, re.MULTILINE)
   925→        for env_match in env_sections:
   926→            start = env_match.end()
   927→            # Read until next top-level key (same or less indentation)
   928→            env_text = ""
   929→            for line in content[start:].splitlines():
   930→                if line and not line[0].isspace() and line.strip():
   931→                    break
   932→                env_text += line + "\n"
   933→
   934→            # Check for secret-like keys with non-${{ }} values
   935→            for line in env_text.splitlines():
   936→                kv_match = re.match(r'^\s+([A-Z_]*(?:KEY|SECRET|TOKEN|PASSWORD|PASSWD|CREDENTIAL|AUTH|WEBHOOK|API)[A-Z_]*)\s*:\s*(.+)', line)
   937→                if kv_match:
   938→                    key_name = kv_match.group(1)
   939→                    val = kv_match.group(2).strip()
   940→                    # If value is NOT a GitHub secret reference
   941→                    if val and "${{ secrets." not in val and val not in ('""', "''", '"""', "'''", 'true', 'false', 'null'):
   942→                        masked = val[:10] + "..." if len(val) > 10 else "***"
   943→                        findings.append(Finding(
   944→                            title=f"Potential secret exposed in env: {key_name} in {os.path.basename(wf)}",
   945→                            severity="high", category="github_actions",
   946→                            module="dev",
   947→                            description=f"Environment variable '{key_name}' in workflow has a hardcoded value. Use ${{{{ secrets.{key_name} }}}} instead.",
   948→                            evidence=f"{key_name}={masked}",
   949→                            asset=asset, points_deducted=10,
   950→                            remediation=f"Use GitHub Secrets: Set {key_name} in Settings > Secrets and use ${{{{ secrets.{key_name} }}}} in the workflow.",
   951→                        ))
   952→
   953→        # Also check for tokens in run: steps
   954→        for pattern, description in _ACTIONS_SECRET_PATTERNS:
   955→            matches = re.findall(pattern, content)
   956→            if matches:
   957→                for match in matches[:1]:
   958→                    if isinstance(match, tuple):
   959→                        match_str = match[-1] if match else "***"
   960→                    else:
   961→                        match_str = match
   962→                    masked = match_str[:10] + "..." if len(str(match_str)) > 10 else "***"
   963→                    findings.append(Finding(
   964→                        title=f"{description}: {os.path.basename(wf)}",
   965→                        severity="high", category="github_actions",
   966→                        module="dev",
   967→                        description=f"Potential secret or token found directly in GitHub Actions workflow.",
   968→                        evidence=f"Match: {masked}",
   969→                        asset=asset, points_deducted=10,
   970→                        remediation="Move to GitHub Secrets and reference via ${{{{ secrets.NAME }}}}",
   971→                    ))
   972→                break  # One per pattern per file
   973→
   974→    return findings

   981→# Popular public npm packages that are commonly used as internal/private names
   982→# This helps detect when a private package shadows a popular public one
   983→_PUBLIC_PACKAGE_NAMES = {
   984→    "lodash", "underscore", "moment", "axios", "express", "react", "react-dom",
   985→    "vue", "angular", "jquery", "bootstrap", "tailwindcss", "next", "nuxt",

   987→    "mongoose", "prisma", "sequelize", "typeorm", "knex", "bookshelf",
   988→    "passport", "bcrypt", "jsonwebtoken", "joi", "yup", "zod", "dotenv",
   989→    "cors", "helmet", "compression", "morgan", "body-parser", "multer",
   990→    "socket.io", "ws", "redis", "ioredis", "pg", "mysql", "mysql2",
   991→    "mongoose", "mongodb", "amqplib", "kafkajs", "bull", "agenda",
   992→    "nodemailer", "sendgrid", "stripe", "twilio", "aws-sdk", "firebase",
   993→    "google-cloud", "azure", "@google/cloud", "@azure/storage",
   994→    "uuid", "nanoid", "date-fns", "dayjs", "chalk", "ora", "inquirer",
   995→    "commander", "yargs", "inquirer", "prompt", "figlet", "cli-table",
   996→    "winston", "pino", "bunyan", "morgan", "log4js", "debug",
   997→    "sharp", "jimp", "canvas", "pdfkit", "docx", "xlsx", "csv-parser",
   998→    "cron", "node-cron", "agenda", "bree", "toad-scheduler",
   999→    "config", "convict", "dotenv-flow", "rc", "nconf",
  1000→    "request", "got", "node-fetch", "ky", "undici", "superagent",
  1001→    "typeorm", "objection", "mikro-orm", "drizzle-orm", "kysely",
  1002→}
  1003→
  1004→
  1005→def _check_dependency_confusion(base: str) -> List[Finding]:
  1006→    """Check if private packages are named the same as popular public packages."""
  1007→    findings: List[Finding] = []
  1008→    asset = base
  1009→
  1010→    # Check for .npmrc with registry config indicating private registry
  1011→    npmrc_paths = _find_files([".npmrc"], base)
  1012→    private_registry = None
  1013→    for npmrc in npmrc_paths:
  1014→        if "node_modules" in npmrc:
  1015→            continue
  1016→        content = _read_file(npmrc)
  1017→        if content:
  1018→            reg_match = re.search(r'registry\s*=\s*(https?://[^\s]+)', content)
  1019→            if reg_match:
  1020→                reg_url = reg_match.group(1)
  1021→                # If it's NOT the default npm registry, it's a private registry
  1022→                if "registry.npmjs.org" not in reg_url:
  1023→                    private_registry = reg_url
  1024→
  1025→    # Check for private packages in package.json
  1026→    pkg_files = _find_files(["package.json"], base)
  1027→    for pkg_path in pkg_files:
  1028→        if "node_modules" in pkg_path:
  1029→            continue
  1030→        try:
  1031→            with open(pkg_path) as f:
  1032→                pkg = json.load(f)
  1033→        except Exception:
  1034→            continue
  1035→
  1036→        all_deps = {}
  1037→        all_deps.update(pkg.get("dependencies", {}))
  1038→        all_deps.update(pkg.get("devDependencies", {}))
  1039→
  1040→        for dep_name in all_deps:
  1041→            # Strip @scope/ prefix for comparison
  1042→            clean_name = dep_name.split("/")[-1] if "/" in dep_name else dep_name
  1043→            if clean_name.lower() in {n.lower() for n in _PUBLIC_PACKAGE_NAMES} and private_registry:
  1044→                findings.append(Finding(
  1045→                    title=f"Dependency confusion risk: '{dep_name}' on private registry",
  1046→                    severity="high", category="dependency_confusion",
  1047→                    module="dev",
  1048→                    description=f"Package '{dep_name}' is named the same as a popular public npm package, but you're using a private registry ({private_registry}). An attacker could publish a malicious version on the public registry with a higher version number.",
  1049→                    evidence=f"Package: {dep_name}, Registry: {private_registry}",
  1050→                    asset=asset, points_deducted=8,
  1051→                    remediation=f"1) Use scoped packages (e.g., @yourcompany/{clean_name}). 2) Configure .npmrc to always prefer your private registry for this package scope.",
  1052→                ))
  1053→
  1054→    return findings
