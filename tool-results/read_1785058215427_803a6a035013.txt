    81→    81→// ─── Mini severity donut (SVG) ──────────────────────────────
    82→    82→
    83→    83→function SeverityDonut({ data }: { data: { name: string; value: number; color: string }[] }) {
    84→    84→  const total = data.reduce((s, d) => s + d.value, 0);
    85→    85→  const segments = data.map((d, i) => {
    86→    86→    const pct = total === 0 ? 0 : d.value / total;
    87→    87→    const off = data.slice(0, i).reduce((s2, prev) => s2 + (total === 0 ? 0 : prev.value / total), 0);
    88→    88→    return { ...d, pct, dasharray: `${pct * 283} ${283}`, offset: -off * 283 };
    89→    89→  });
    90→    90→  if (total === 0) return <div className="text-xs text-muted-foreground text-center py-4">No data yet</div>;
    91→    91→  return (
    92→    92→    <svg viewBox="0 0 120 120" className="w-32 h-32 mx-auto">
    93→    93→      {segments.map((d) => (
    94→    94→        <circle key={d.name} cx="60" cy="60" r="45" fill="none"
    95→    95→          stroke={d.color} strokeWidth="18" strokeDasharray={d.dasharray}
    96→    96→          strokeDashoffset={d.offset} strokeLinecap="round" opacity={0.85} />
    97→    97→      ))}
    98→    98→      <text x="60" y="56" textAnchor="middle" fill="#e6edf3" fontSize="22" fontWeight="bold" fontFamily="Geist Sans, sans-serif">{total}</text>
    99→    99→      <text x="60" y="72" textAnchor="middle" fill="#7d8590" fontSize="9" fontFamily="Geist Sans, sans-serif">FINDINGS</text>
   100→   100→    </svg>
   101→   101→  );
   102→   102→}
