   836→    def scan(self, target: str, base_url: str, timeout: int = 8) -> Dict[str, Any]:
   837→        """
   838→        Run a full Shadow IT scan against *target* (domain) and *base_url*.
   839→
   840→        Returns a dictionary with keys for every scan category, each containing
   841→        a list of :class:`AssetProfile` dicts.
   842→        """
   843→        self.timeout = timeout
   844→        timestamp = datetime.now(timezone.utc).isoformat()
   845→
   846→        results: Dict[str, Any] = {
   847→            "target": target,
   848→            "base_url": base_url,
   849→            "timestamp": timestamp,
   850→            "scanner": "ReconPro ShadowIT v9.2.0",
   851→            "summary": {
   852→                "total_assets": 0,
   853→                "critical": 0,
   854→                "high": 0,
   855→                "medium": 0,
   856→                "low": 0,
   857→                "info": 0,
   858→                "categories_with_findings": 0,
   859→            },
   860→            "findings": {},
   861→        }
   862→
   863→        categories = [
   864→            ("forgotten_subdomains", self.discover_forgotten_subdomains, [target]),
   865→            ("staging_environments", self.discover_staging_environments, [target]),
   866→            ("abandoned_infrastructure", self.discover_abandoned_infrastructure, [base_url]),
   867→            ("shadow_apis", self.discover_shadow_apis, [base_url]),
   868→            ("orphaned_dns", self.discover_orphaned_dns, [target]),
   869→            ("forgotten_services", self.discover_forgotten_services, [base_url]),
   870→            ("cloud_leaks", self.discover_cloud_leaks, [base_url]),
   871→            ("internal_exposure", self.discover_internal_exposure, [base_url]),
   872→        ]
   873→
   874→        for cat_name, cat_func, cat_args in categories:
   875→            try:
   876→                assets = cat_func(*cat_args)
   877→                if isinstance(assets, list):
   878→                    results["findings"][cat_name] = [a.to_dict() if isinstance(a, AssetProfile) else a for a in assets]
   879→                elif isinstance(assets, dict):
   880→                    results["findings"][cat_name] = assets
   881→                else:
   882→                    results["findings"][cat_name] = assets
   883→            except Exception as exc:
   884→                results["findings"][cat_name] = {"error": str(exc)}
   885→
   886→        # Build summary counts
   887→        total = 0
   888→        risk_counts = {"critical": 0, "high": 0, "medium": 0, "low": 0, "info": 0}
   889→        cats_with_findings = 0
   890→        for cat_key, cat_val in results["findings"].items():
   891→            if isinstance(cat_val, list):
   892→                for item in cat_val:
   893→                    if isinstance(item, dict):
   894→                        total += 1
   895→                        rl = item.get("risk_level", "info")
   896→                        if rl in risk_counts:
   897→                            risk_counts[rl] += 1
   898→                if cat_val:
   899→                    cats_with_findings += 1
   900→            elif isinstance(cat_val, dict) and "assets" in cat_val:
   901→                for item in cat_val["assets"]:
   902→                    if isinstance(item, dict):
   903→                        total += 1
   904→                        rl = item.get("risk_level", "info")
   905→                        if rl in risk_counts:
   906→                            risk_counts[rl] += 1
   907→                if cat_val["assets"]:
   908→                    cats_with_findings += 1
   909→
   910→        results["summary"]["total_assets"] = total
   911→        results["summary"]["critical"] = risk_counts["critical"]
   912→        results["summary"]["high"] = risk_counts["high"]
   913→        results["summary"]["medium"] = risk_counts["medium"]
   914→        results["summary"]["low"] = risk_counts["low"]
   915→        results["summary"]["info"] = risk_counts["info"]
   916→        results["summary"]["categories_with_findings"] = cats_with_findings
   917→
   918→        return results

   924→    def discover_forgotten_subdomains(self, target: str) -> List[AssetProfile]:
   925→        """
   926→        Enumerate subdomains via cert transparency (crt.sh) and common-prefix
   927→        brute-force DNS resolution.
   928→        """
   929→        found: List[AssetProfile] = []
   930→        resolved_hosts: Dict[str, str] = {}
   931→
   932→        # --- Phase 1: crt.sh certificate transparency ---
   933→        ct_hosts = self._query_crtsh(target)
   934→        for host in ct_hosts:
   935→            if host not in resolved_hosts:
   936→                ip = self._resolve_host(host)
   937→                if ip:
   938→                    resolved_hosts[host] = ip
   939→
   940→        # --- Phase 2: Brute-force common prefixes ---
   941→        for prefix in self.COMMON_SUBDOMAIN_PREFIXES:
   942→            fqdn = f"{prefix}.{target}"
   943→            if fqdn not in resolved_hosts:
   944→                ip = self._resolve_host(fqdn)
   945→                if ip:
   946→                    resolved_hosts[fqdn] = ip
   947→
   948→        # --- Phase 3: Profile each discovered subdomain ---
   949→        for host, ip in resolved_hosts.items():
   950→            if host == target or host == f"www.{target}":
   951→                continue
   952→            profile = self._profile_host(host, ip)
   953→            profile.discovered_by = "subdomain_enum"
   954→            found.append(profile)
   955→
   956→        return found

   958→    def _query_crtsh(self, domain: str) -> List[str]:
   959→        """Query crt.sh for certificate transparency subdomains."""
   960→        hosts: List[str] = []
   961→        url = f"https://crt.sh/?q=%.{domain}&output=json"
   962→        try:
   963→            resp_headers, status, body, final_url = _make_request(
   964→                url, timeout=12, headers=self._session_headers
   965→            )
   966→            if body and status == 200:
   967→                try:
   968→                    entries = json.loads(body)
   969→                    for entry in entries:
   970→                        name_value = entry.get("name_value", "")
   971→                        for name in name_value.split("\n"):
   972→                            name = name.strip().lstrip("*.")
   973→                            if name and name.endswith(f".{domain}") and name != domain:
   974→                                if name not in hosts:
   975→                                    hosts.append(name)
   976→                except (json.JSONDecodeError, KeyError, TypeError):
   977→                    pass
   978→                # Fallback: regex extraction
   979→                if not hosts:
   980→                    pattern = re.compile(r'([a-zA-Z0-9][a-zA-Z0-9-]*\.' + re.escape(domain) + r')')
   981→                    matches = pattern.findall(body)
   982→                    for m in matches:
   983→                        m = m.lstrip("*.")
   984→                        if m not in hosts and m != domain:
   985→                            hosts.append(m)
   986→        except Exception:
   987→            pass
   988→        return hosts
   989→
   990→    def _resolve_host(self, host: str) -> Optional[str]:
   991→        """Resolve a hostname to an IP address via DNS. Returns IP or None."""
   992→        try:
   993→            result = socket.getaddrinfo(host, None, socket.AF_INET, socket.SOCK_STREAM)
   994→            if result:
   995→                return result[0][4][0]
   996→        except (socket.gaierror, socket.herror, OSError):
   997→            pass
   998→        return None
   999→
  1000→    def _profile_host(self, host: str, ip: str) -> AssetProfile:
  1001→        """Build an AssetProfile for a discovered host."""
  1002→        cert_info = _check_certificate_age(host)
  1003→        # Try HTTPS first, then HTTP
  1004→        resp_headers = None
  1005→        status_code = None
  1006→        body = ""
  1007→        title = ""
  1008→        for scheme in ("https", "http"):
  1009→            url = f"{scheme}://{host}/"
  1010→            resp_headers, status_code, body, final_url = _make_request(url, self.timeout, headers=self._session_headers)
  1011→            if status_code is not None:
  1012→                break
  1013→        title = _extract_title(body)
  1014→ tech_stack = _detect_tech_stack(resp_headers, body)
  1015→        indicators = _assess_indicators(resp_headers, body, title, cert_info)
  1016→        asset_type = _classify_asset(resp_headers, body, title, cert_info)
  1017→        decay = _calculate_decay_from_indicators(indicators, resp_headers, cert_info)
  1018→        return AssetProfile(
  1019→            url=f"https://{host}/" if status_code else f"http://{host}/",
  1020→            asset_type=asset_type,
  1021→            status_code=status_code,
  1022→            decay_score=decay,
  1023→            indicators=indicators,
  1024→            last_modified=resp_headers.get("last-modified") if resp_headers else None,
  1025→            server_header=resp_headers.get("server") if resp_headers else None,
  1026→            title=title,
  1027→            cert_info=cert_info,
  1028→            tech_stack=tech_stack,
  1029→            discovered_by="dns_resolution",
  1030→        )

  1063→    def discover_abandoned_infrastructure(self, base_url: str) -> List[AssetProfile]:
  1064→        """
  1065→        Check the base URL and common paths for signs of abandonment:
  1066→        old WordPress, default pages, expired certificates, outdated headers,
  1067→        missing HSTS, HTTP-only.
  1068→        """
  1069→        found: List[AssetProfile] = []
  1070→        parsed = urlparse(base_url)
  1071→        host = parsed.hostname or ""
  1072→
  1073→        # Check the base URL itself
  1074→        profile = _probe_path(base_url, "/", self.timeout)
  1075→        if profile.status_code is not None:
  1076→            profile.discovered_by = "abandoned_infrastructure_scan"
  1077→            # Additional: check HTTP-only
  1078→            if parsed.scheme == "http":
  1079→                https_url = f"https://{host}/"
  1080→                _, https_status, _, _ = _make_request(https_url, self.timeout)
  1081→                if https_status is None or https_status >= 400:
  1082→                    if "http_only_no_redirect" not in profile.indicators:
  1083→                        profile.indicators.append("http_only_no_redirect")
  1084→                    profile.decay_score = _calculate_decay_from_indicators(
  1085→                        profile.indicators, None, profile.cert_info
  1086→                    )
  1087→                    # Re-evaluate risk level
  1088→                    if profile.decay_score >= 75:
  1089→                        profile.risk_level = "critical"
  1090→                    elif profile.decay_score >= 55:
  1091→                        profile.risk_level = "high"
  1092→                    elif profile.decay_score >= 35:
  1093→                        profile.risk_level = "medium"
  1094→                    elif profile.decay_score >= 15:
  1095→                        profile.risk_level = "low"
  1096→                    else:
  1097→                        profile.risk_level = "info"
  1098→            found.append(profile)
  1099→
  1100→        # Probe common abandoned paths
  1101→        abandoned_paths = [
  1102→            "/", "/index.html", "/index.php", "/default.html",
  1103→            "/old/", "/archive/", "/backup/", "/legacy/", "/deprecated/",
  1104→            "/wordpress/wp-login.php", "/wp-login.php",
  1105→            "/wp-admin/", "/wp-admin/install.php",
  1106→            "/server-status", "/server-info",
  1107→            "/info.php", "/phpinfo.php", "/test.php",
  1108→            "/README.md", "/README.txt", "/CHANGELOG.md",
  1109→            "/.git/HEAD", "/.git/config", "/.svn/entries",
  1110→            "/.DS_Store", "/Thumbs.db",
  1111→            "/WEB-INF/web.xml", "/META-INF/MANIFEST.MF",
  1112→            "/crossdomain.xml", "/clientaccesspolicy.xml",
  1113→        ]
  1114→        for path in abandoned_paths:
  1115→            if path == "/":
  1116→                continue  # already probed
  1117→            p = _probe_path(base_url, path, self.timeout)
  1118→            if p.status_code is not None and p.status_code < 400:
  1119→                p.discovered_by = "abandoned_infrastructure_scan"
  1120→                found.append(p)
  1121→
  1122→        return found

  1128→    def discover_shadow_apis(self, base_url: str) -> List[AssetProfile]:
  1129→        """
  1130→        Find undocumented / hidden APIs: common API paths, debug endpoints,
  1131→        Swagger/OpenAPI specs, GraphQL introspection.
  1132→        """
  1133→        found: List[AssetProfile] = []
  1134→        probed_urls: set = set()
  1135→
  1136→        for path in self.SHADOW_API_PATHS:
  1137→            url = urljoin(base_url.rstrip("/") + "/", path.lstrip("/"))
  1138→            if url in probed_urls:
  1139→                continue
  1140→            probed_urls.add(url)
  1141→            resp_headers, status_code, body, final_url = _make_request(url, self.timeout, headers=self._session_headers)
  1142→            if status_code is None:
  1143→                continue
  1144→            # Skip 404s, focus on real hits
  1145→            if status_code == 404:
  1146→                continue
  1147→            cert_info = _check_certificate_age(urlparse(url).hostname or "")
  1148→            title = _extract_title(body or "")
  1149→            tech_stack = _detect_tech_stack(resp_headers, body or "")
  1150→            indicators = _assess_indicators(resp_headers, body or "", title, cert_info)
  1151→            asset_type = _classify_asset(resp_headers, body or "", title, cert_info)
  1152→            decay = _calculate_decay_from_indicators(indicators, resp_headers, cert_info)
  1153→            profile = AssetProfile(
  1154→                url=url,
  1155→                asset_type=asset_type,
  1156→                status_code=status_code,
  1157→                decay_score=decay,
  1158→                indicators=indicators,
  1159→                last_modified=resp_headers.get("last-modified") if resp_headers else None,
  1160→                server_header=resp_headers.get("server") if resp_headers else None,
  1161→                title=title,
  1162→                cert_info=cert_info,
  1163→                tech_stack=tech_stack,
  1164→                discovered_by="shadow_api_scan",
  1165→            )
  1166→            found.append(profile)
  1167→
  1168→            # GraphQL introspection probe
  1169→            if "graphql" in path.lower() and status_code == 200:
  1170→                introspection = self._probe_graphql_introspection(url)
  1171→                if introspection:
  1172→                    profile.indicators.append("graphql_introspection_enabled")
  1173→                    profile.tech_stack.append("GraphQL (introspection open)")
  1174→
  1175→            # Swagger schema extraction
  1176→            if status_code == 200 and body and ("swagger" in path.lower() or "openapi" in path.lower()):
  1177→                try:
  1178→                    schema = json.loads(body)
  1179→                    if schema.get("swagger") or schema.get("openapi"):
  1180→                        paths_count = len(schema.get("paths", {}))
  1181→                        profile.indicators.append(f"openapi_spec_exposed ({paths_count} endpoints)")
  1182→                        profile.tech_stack.append(f"OpenAPI ({paths_count} paths)")
  1183→                except (json.JSONDecodeError, TypeError):
  1184→                    pass
  1185→
  1186→        return found

  1188→    def _probe_graphql_introspection(self, graphql_url: str) -> bool:
  1189→        """Send a GraphQL introspection query and check for a valid response."""
  1190→        introspection_query = {
  1191→            "query": """{
  1192→                __schema {
  1193→                    types {
  1194→                        name
  1195→                        kind
  1196→                        fields {
  1197→                            name
  1198→                        }
  1199→                    }
  1200→                }
  1201→            }"""
  1202→        }
  1203→        data = json.dumps(introspection_query).encode("utf-8")
  1204→        try:
  1205→            req = Request(
  1206→                graphql_url,
  1207→                data=data,
  1208→                headers={
  1209→                    **self._session_headers,
  1210→                    "Content-Type": "application/json",
  1211→                },
  1212→                method="POST",
  1213→            )
  1214→            resp = urlopen(req, timeout=self.timeout)
  1215→            body = resp.read(65536).decode("utf-8", errors="replace")
  1216→            result = json.loads(body)
  1217→            return "data" in result and "__schema" in result.get("data", {})
  1218→        except Exception:
  1219→            return False

  1228→        records, and stale MX records.

  1230→        findings: Dict[str, Any] = {
  1231→            "target": target,
  1232→            "records": {},
  1233→            "assets": [],
  1234→        }
  1235→
  1236→        # Resolve A / AAAA
  1237→        ipv4 = self._resolve_host(target)
  1238→        findings["records"]["A"] = ipv4
  1239→
  1240→        # NS records
  1241→        ns_records = self._dns_lookup(target, "NS")
  1242→        findings["records"]["NS"] = ns_records
  1243→
  1244→        # MX records
  1245→        mx_records = self._dns_lookup(target, "MX")
  1246→        findings["records"]["MX"] = mx_records
  1247→
  1248→        # TXT records
  1249→        txt_records = self._dns_lookup(target, "TXT")
  1250→        findings["records"]["TXT"] = txt_records
  1251→
  1252→        # CNAME records
  1253→        cname_records = self._dns_lookup(target, "CNAME")
  1254→        findings["records"]["CNAME"] = cname_records
  1255→
  1256→        # Check for orphaned NS: NS records pointing to hosts that don't resolve
  1257→        orphaned_ns: List[str] = []
  1258→        for ns in ns_records:
  1259→            ns_host = ns.rstrip(".")
  1260→            if not self._resolve_host(ns_host):
  1261→                orphaned_ns.append(ns_host)
  1262→        findings["orphaned_ns"] = orphaned_ns
  1263→
  1264→        # Check for stale MX: MX records pointing to unreachable hosts
  1265→        stale_mx: List[Dict[str, Any]] = []
  1266→        for mx in mx_records:
  1267→            mx_host = mx.rstrip(".")
  1268→            ip = self._resolve_host(mx_host)
  1269→            if ip:
  1270→                # Check if SMTP port is open
  1271→                smtp_open = self._check_port_open(mx_host, 25, timeout=4)
  1272→                if not smtp_open:
  1273→                    stale_mx.append({"host": mx_host, "ip": ip, "issue": "smtp_port_closed"})
  1274→            else:
  1275→                stale_mx.append({"host": mx_host, "ip": None, "issue": "unresolvable"})
  1276→        findings["stale_mx"] = stale_mx
  1277→
  1278→        # Check for dangling CNAME
  1279→        dangling_cname: List[str] = []
  1280→        for cname in cname_records:
  1281→            cname_target = cname.rstrip(".")
  1282→            if not self._resolve_host(cname_target):
  1283→                dangling_cname.append(cname_target)
  1284→        findings["dangling_cname"] = dangling_cname
  1285→
  1286→        # Build asset profiles for orphaned records
  1287→        for ns_host in orphaned_ns:
  1288→            profile = AssetProfile(
  1289→                url=f"dns://{ns_host}",
  1290→                asset_type="orphaned_ns",
  1291→                status_code=None,
  1292→                decay_score=70,
  1293→                indicators=["orphaned_ns_record"],
  1294→                discovered_by="dns_analysis",
  1295→            )
  1296→            findings["assets"].append(profile.to_dict())
  1297→
  1298→        for mx_info in stale_mx:
  1299→            profile = AssetProfile(
  1300→                url=f"dns://{mx_info['host']}",
  1301→                asset_type="stale_mx",
  1302→                status_code=None,
  1303→                decay_score=60 if mx_info["issue"] == "smtp_port_closed" else 80,
  1304→                indicators=[f"stale_mx_{mx_info['issue']}"],
  1305→                discovered_by="dns_analysis",
  1306→            )
  1307→            findings["assets"].append(profile.to_dict())
  1308→
  1309→        for cname_host in dangling_cname:
  1310→            profile = AssetProfile(
  1311→                url=f"dns://{cname_host}",
  1312→                asset_type="dangling_cname",
  1313→                status_code=None,
  1314→                decay_score=75,
  1315→                indicators=["dangling_cname_record"],
  1316→                discovered_by="dns_analysis",
  1317→            )
  1318→            findings["assets"].append(profile.to_dict())
  1319→
  1320→        return findings

  1322→    def _dns_lookup(self, domain: str, record_type: str) -> List[str]:
  1323→        """Perform a DNS lookup using system resolver (dig fallback or getaddrinfo)."""
  1324→        results: List[str] = []
  1325→        if record_type in ("A", "AAAA"):
  1326→            try:
  1327→                family = socket.AF_INET if record_type == "A" else socket.AF_INET6
  1328→                addrs = socket.getaddrinfo(domain, None, family, socket.SOCK_STREAM)
  1329→                for addr in addrs:
  1330→                    ip = addr[4][0]
  1331→                    if ip not in results:
  1332→                        results.append(ip)
  1333→            except (socket.gaierror, OSError):
  1334→                pass
  1335→            return results
  1336→
  1337→        # For NS, MX, TXT, CNAME — try system dig/host commands
  1338→        try:
  1339→            cmd = ["dig", "+short", record_type, domain]
  1340→            proc = subprocess.run(
  1341→                cmd, capture_output=True, text=True, timeout=8,
  1342→                stdin=subprocess.DEVNULL,
  1343→            )
  1344→            if proc.returncode == 0 and proc.stdout.strip():
  1345→                for line in proc.stdout.strip().split("\n"):
  1346→                    line = line.strip()
  1347→                    if line and line not in results:
  1348→                        results.append(line)
  1349→                return results
  1350→        except (FileNotFoundError, subprocess.TimeoutExpired, OSError):
  1351→            pass
  1352→
  1353→        # Fallback: try host command
  1354→        try:
  1355→            cmd = ["host", "-t", record_type, domain]
  1356→            proc = subprocess.run(
  1357→                cmd, capture_output=True, text=True, timeout=8,
  1358→                stdin=subprocess.DEVNULL,
  1359→            )
  1360→            if proc.returncode == 0 and proc.stdout.strip():
  1361→                for line in proc.stdout.strip().split("\n"):
  1362→                    line = line.strip()
  1363→                    # Extract the value from host output (typically after "is an alias for" or "mail is handled by")
  1364→                    parts = line.split()
  1365→                    if len(parts) >= 2:
  1366→                        value = parts[-1].rstrip(".")
  1367→                        if value and value not in results:
  1368→                            results.append(value + ".")
  1369→                return results
  1370→        except (FileNotFoundError, subprocess.TimeoutExpired, OSError):
  1371→            pass
  1372→
  1373→        return results
  1374→
  1375→    def _check_port_open(self, host: str, port: int, timeout: int = 4) -> bool:
  1376→        """Check if a TCP port is open on a host."""
  1377→        try:
  1378→            with socket.create_connection((host, port), timeout=timeout):
  1379→                return True
  1380→        except (socket.timeout, socket.gaierror, OSError):
  1381→            return False

  1620→    def calculate_decay_score(self, asset_info: Dict[str, Any]) -> int:
  1621→        """
  1622→        Calculate how "abandoned" an asset is on a 0-100 scale.
  1623→
  1624→        Factors:
  1625→        - Certificate expiry proximity (expired = +25, near expiry = +12)
  1626→        - Outdated server headers (+15)
  1627→        - Default content pages (+20)
  1628→        - Missing security headers (HSTS, CSP) (+5 each)
  1629→        - HTTP-only with no redirect (+8)
  1630→        - No robots.txt (+3), no favicon (+3)
  1631→        - Old WordPress version signatures (+18)
  1632→        - Outdated JS libraries (jQuery 1.x, Bootstrap 3.x) (+10)
  1633→        - Visible PHP errors (+15)
  1634→        - Stack traces exposed (+20)
  1635→        - Debug mode indicators (+18)
  1636→        - Directory listings (+18)
  1637→        - Self-signed certificates (+15)
  1638→        - Certificate age > 1 year (+5), > 2 years (+8)
  1639→        - No Last-Modified header (+2)
  1640→
  1641→        :param asset_info: Dictionary with keys like ``headers``, ``body``,
  1642→            ``title``, ``cert_info``, ``status_code``.
  1643→        :returns: Integer 0-100 where 100 means fully abandoned/decayed.
  1644→        """
  1645→        headers = asset_info.get("headers") or {}
  1646→        body = asset_info.get("body", "")
  1647→        title = asset_info.get("title", "")
  1648→        cert_info = asset_info.get("cert_info")
  1649→        status_code = asset_info.get("status_code")
  1650→
  1651→        indicators = _assess_indicators(headers, body, title, cert_info)
  1652→        score = _calculate_decay_from_indicators(indicators, headers, cert_info)
  1653→
  1654→        # Additional: check for HTTP-only
  1655→        if asset_info.get("http_only") and not asset_info.get("has_https_redirect"):
  1656→            score += 8
  1657→
  1658→        # Check for no robots.txt
  1659→        if asset_info.get("no_robots_txt"):
  1660→            score += 3
  1661→
  1662→        # High status codes suggest degradation
  1663→        if status_code and status_code >= 500:
  1664→            score += 10
  1665→        elif status_code and status_code == 403:
  1666→            score += 5
  1667→
  1668→        return min(score, 100)
