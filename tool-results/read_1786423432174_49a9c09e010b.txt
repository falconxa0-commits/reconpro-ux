   987→    def _check_memory_usage(self) -> int:
   988→        """Return approximate current memory usage in bytes.
   989→
   990→        Uses resource module if available (Unix), otherwise falls back
   991→        to a basic tracking mechanism.
   992→        """
   993→        try:
   994→            import resource as _resource  # type: ignore[import-untyped]
   995→            return _resource.getrusage(_resource.RUSAGE_SELF).ru_maxrss * 1024
   996→        except (ImportError, AttributeError):
   997→            # Fallback: track based on sys.getsizeof of known objects
   998→            return 0
   999→
  1000→    def _enforce_cpu_limit(self) -> None:
  1001→        """Set CPU time limit on platforms that support it (Unix)."""
  1002→        try:
  1003→            import resource as _resource  # type: ignore[import-untyped]
  1004→            _resource.setrlimit(
  1005→                _resource.RLIMIT_CPU,
  1006→                (int(self.cpu_time_seconds), int(self.cpu_time_seconds * 1.5)),
  1007→            )
  1008→        except (ImportError, AttributeError, OSError, ValueError):
  1009→            pass  # Not available on this platform
  1010→
  1011→    def _restore_cpu_limit(self) -> None:
  1012→        """Restore CPU time limit after sandboxed execution."""
  1013→        try:
  1014→            import resource as _resource  # type: ignore[import-untyped]
  1015→            # Set to a very high value (effectively unlimited)
  1016→            _resource.setrlimit(_resource.RLIMIT_CPU, (0x7FFFFFFF, 0x7FFFFFFF))
  1017→        except (ImportError, AttributeError, OSError, ValueError):
  1018→            pass

  1020→    def _validate_output(self, result: Any) -> List[Dict[str, Any]]:
  1021→        """Validate plugin output is safe and within limits.
  1022→
  1023→        Args:
  1024→            result: The return value from the plugin.
  1025→
  1026→        Returns:
  1027→            List of finding dicts, or raises on invalid output.
  1028→        """
  1029→        if not isinstance(result, list):
  1030→            raise SandboxViolation(
  1031→                "output_validation",
  1032→                f"Plugin must return a list, got {type(result).__name__}",
  1033→            )
  1034→
  1035→        output_str = json.dumps(result, default=str)
  1036→        if len(output_str) > self.max_output_size:
  1037→            raise ResourceLimitExceeded(
  1038→                "output_size",
  1039→                self.max_output_size,
  1040→                len(output_str),
  1041→            )
  1042→
  1043→        validated: List[Dict[str, Any]] = []
  1044→        required_keys = {"title", "severity", "category"}
  1045→        for i, item in enumerate(result):
  1046→            if not isinstance(item, dict):
  1047→                raise SandboxViolation(
  1048→                    f"output_item_{i}",
  1049→                    f"Each finding must be a dict, got {type(item).__name__}",
  1050→                )
  1051→            if not required_keys.issubset(item.keys()):
  1052→                missing = required_keys - item.keys()
  1053→                raise SandboxViolation(
  1054→                    f"output_item_{i}",
  1055→                    f"Finding missing required keys: {missing}",
  1056→                )
  1057→            validated.append(item)
  1058→
  1059→        return validated

  1061→    def execute(
  1062→        self,
  1063→        plugin_func: Callable,
  1064→        plugin_name: str = "unknown",
  1065→        **kwargs: Any,
  1066→    ) -> Dict[str, Any]:
  1067→        """Execute a plugin function inside the sandbox.
  1068→
  1069→        The plugin function is called with **kwargs as arguments.
  1070→        Its execution is wrapped with:
  1071→        - Restricted builtins
  1072→        - CPU time limits (where supported)
  1073→        - Memory usage checks
  1074→        - Output validation
  1075→        - Full audit logging
  1076→
  1077→        Args:
  1078→            plugin_func: The plugin's callable (typically its ``run`` function).
  1079→            plugin_name: Human-readable plugin name for logging.
  1080→            **kwargs: Arguments forwarded to the plugin function.
  1081→
  1082→        Returns:
  1083→            Dict with keys:
  1084→            - success: bool
  1085→            - findings: list of validated finding dicts (on success)
  1086→            - error: str (on failure)
  1087→            - execution_time: float (seconds)
  1088→            - memory_used: int (bytes, 0 if unmeasurable)
  1089→            - operations: count of sandbox-tracked operations
  1090→        """
  1091→        self._log_operation("sandbox_enter", {"plugin": plugin_name})
  1092→        start_time = time.monotonic()
  1093→        memory_before = self._check_memory_usage()
  1094→
  1095→        try:
  1096→            # Set CPU time limit
  1097→            self._enforce_cpu_limit()
  1098→            self._log_operation("cpu_limit_set", {
  1099→                "limit_seconds": self.cpu_time_seconds,
  1100→            })
  1101→
  1102→            # Execute the plugin function
  1103→            # Note: Python does not provide true sandboxing. The restricted
  1104→            # builtins are applied when the plugin code uses exec/eval.
  1105→            # For direct function calls, we wrap and monitor.
  1106→            self._log_operation("plugin_execute_start", {
  1107→                "plugin": plugin_name,
  1108→                "args": list(kwargs.keys()),
  1109→            })
  1110→
  1111→            try:
  1112→                result = plugin_func(**kwargs)
  1113→            except TypeError as e:
  1114→                raise SandboxViolation(
  1115→                    "plugin_call",
  1116→                    f"Plugin call failed: {e}",
  1117→                )
  1118→
  1119→            self._log_operation("plugin_execute_end", {
  1120→                "plugin": plugin_name,
  1121→                "result_type": type(result).__name__,
  1122→            })
  1123→
  1124→            # Validate output
  1125→            findings = self._validate_output(result)
  1126→            self._log_operation("output_validated", {
  1127→                "plugin": plugin_name,
  1128→                "finding_count": len(findings),
  1129→            })
  1130→
  1131→            memory_after = self._check_memory_usage()
  1132→            if (memory_after > 0
  1133→                    and memory_before > 0
  1134→                    and memory_after > self.memory_limit_bytes):
  1135→                raise ResourceLimitExceeded(
  1136→                    "memory",
  1137→                    self.memory_limit_bytes,
  1138→                    memory_after,
  1139→                )
  1140→
  1141→            return {
  1142→                "success": True,
  1143→                "findings": findings,
  1144→                "execution_time": time.monotonic() - start_time,
  1145→                "memory_used": max(0, memory_after - memory_before),
  1146→                "operations": len(self._operation_log),
  1147→            }
  1148→
  1149→        except SandboxViolation as exc:
  1150→            self._log_operation("sandbox_violation", {
  1151→                "plugin": plugin_name,
  1152→                "operation": exc.operation,
  1153→                "reason": exc.reason,
  1154→            })
  1155→            return {
  1156→                "success": False,
  1157→                "error": str(exc),
  1158→                "execution_time": time.monotonic() - start_time,
  1159→                "memory_used": 0,
  1160→                "operations": len(self._operation_log),
  1161→            }
  1162→
  1163→        except ResourceLimitExceeded as exc:
  1164→            self._log_operation("resource_limit_exceeded", {
  1165→                "plugin": plugin_name,
  1166→                "resource": exc.resource,
  1167→                "limit": exc.limit,
  1168→                "actual": exc.actual,
  1169→            })
  1170→            return {
  1171→                "success": False,
  1172→                "error": str(exc),
  1173→                "execution_time": time.monotonic() - start_time,
  1174→                "memory_used": 0,
  1175→                "operations": len(self._operation_log),
  1176→            }
  1177→
  1178→        except Exception as exc:
  1179→            self._log_operation("plugin_error", {
  1180→                "plugin": plugin_name,
  1181→                "error_type": type(exc).__name__,
  1182→                "error": str(exc)[:500],
  1183→            })
  1184→            return {
  1185→                "success": False,
  1186→                "error": f"Plugin execution error: {type(exc).__name__}: {exc}",
  1187→                "execution_time": time.monotonic() - start_time,
  1188→                "memory_used": 0,
  1189→                "operations": len(self._operation_log),
  1190→            }
  1191→
  1192→        finally:
  1193→            # Restore CPU limits
  1194→            self._restore_cpu_limit()
  1195→            self._log_operation("sandbox_exit", {"plugin": plugin_name})
  1196→
  1197→    def get_operation_log(self) -> List[Dict[str, Any]]:
  1198→        """Return a copy of the sandbox operation audit log."""
  1199→        with self._log_lock:
  1200→            return copy.deepcopy(self._operation_log)

  1203→        """Clear the operation log."""
  1204→        with self._log_lock:
  1205→            self._operation_log.clear()

  1207→    @staticmethod
  1208→    def check_code_for_violations(code: str) -> List[Dict[str, Any]]:
  1209→        """Statically analyze code for sandbox violations.
  1210→
  1211→        Checks for:
  1212→        - Use of forbidden builtins (eval, exec, compile, open, __import__)
  1213→        - Import of forbidden modules
  1214→        - Use of dangerous attributes (os.system, subprocess.call, etc.)
  1215→
  1216→        Args:
  1217→            code: Python source code to analyze.
  1218→
  1219→        Returns:
  1220→            List of violation dicts with keys:
  1221→            - type: violation type
  1222→            - match: the matching text
  1223→            - line: line number
  1224→            - severity: "critical" or "high"
  1225→        """
  1226→        violations: List[Dict[str, Any]] = []
  1227→        lines = code.split("\n")
  1228→
  1229→        # Check for forbidden builtins
  1230→        forbidden_builtins_pattern = re.compile(
  1231→            r'\b(' + '|'.join(re.escape(b) for b in _FORBIDDEN_BUILTINS) + r')\b'
  1232→        )
  1233→        for line_num, line in enumerate(lines, start=1):
  1234→            # Skip comments
  1235→            stripped = line.strip()
  1236→            if stripped.startswith("#"):
  1237→                continue
  1238→            # Check for string contexts (simple heuristic)
  1239→            # Remove string literals to avoid false positives
  1240→            clean_line = re.sub(r'"[^"]*"', '""', line)
  1241→            clean_line = re.sub(r"'[^']*'", "''", clean_line)
  1242→
  1243→            for match in forbidden_builtins_pattern.finditer(clean_line):
  1244→                builtin = match.group(1)
  1245→                # Avoid flagging __import__ when it's just a string reference
  1246→                if builtin == "__import__" and "__import__" not in clean_line.split("=")[-1]:
  1247→                    violations.append({
  1248→                        "type": "forbidden_builtin",
  1249→                        "match": builtin,
  1250→                        "line": line_num,
  1251→                        "severity": "critical",
  1252→                    })
  1253→                elif builtin != "__import__":
  1254→                    violations.append({
  1255→                        "type": "forbidden_builtin",
  1256→                        "match": builtin,
  1257→                        "line": line_num,
  1258→                        "severity": "critical",
  1259→                    })
  1260→
  1261→        # Check for forbidden imports
  1262→        import_pattern = re.compile(
  1263→            r'\b(?:import|from)\s+([a-zA-Z_][a-zA-Z0-9_]*)'
  1264→        )
  1265→        for line_num, line in enumerate(lines, start=1):
  1266→            stripped = line.strip()
  1267→            if stripped.startswith("#"):
  1268→                continue
  1269→            for match in import_pattern.finditer(line):
  1270→                module = match.group(1)
  1271→                if module in _FORBIDDEN_MODULES:
  1272→                    violations.append({
  1273→                        "type": "forbidden_module",
  1274→                        "match": module,
  1275→                        "line": line_num,
  1276→                        "severity": "high",
  1277→                    })
  1278→
  1279→        # Check for dangerous attribute access patterns
  1280→        dangerous_patterns = [
  1281→            (r'\bos\.system\b', "os.system", "critical"),
  1282→            (r'\bos\.popen\b', "os.popen", "critical"),
  1283→            (r'\bsubprocess\.', "subprocess", "critical"),
  1284→            (r'\bshutil\.', "shutil", "high"),
  1285→            (r'\bctypes\.', "ctypes", "critical"),
  1286→            (r'\b__builtins__\b', "__builtins__", "high"),
  1287→            (r'\bglobals\(\)', "globals()", "high"),
  1288→            (r'\blocals\(\)', "locals()", "medium"),
  1289→            (r'\bgetattr\s*\(\s*__builtins__', "getattr(__builtins__", "critical"),
  1290→        ]
  1291→        for line_num, line in enumerate(lines, start=1):
  1292→            stripped = line.strip()
  1293→            if stripped.startswith("#"):
  1294→                continue
  1295→            for pattern, name, severity in dangerous_patterns:
  1296→                if re.search(pattern, line):
  1297→                    violations.append({
  1298→                        "type": "dangerous_pattern",
  1299→                        "match": name,
  1300→                        "line": line_num,
  1301→                        "severity": severity,
  1302→                    })
  1303→
  1304→        return violations

  1410→class SecretsManager:

  1414→    - Additional patterns (Slack, Google, Stripe, Twilio, SSH, PGP, etc.)

  1420→
  1421→        sm = SecretsManager()
  1422→        findings = sm.scan_content("some text with AKIAIOSFODNN7EXAMPLE key")

  1425→
  1426→    SEVERITY_ORDER: Dict[str, int] = {
  1427→        "critical": 0,
  1428→        "high": 1,
  1429→        "medium": 2,
  1430→        "low": 3,
  1431→    }
  1432→
  1433→    def __init__(self, audit_logger: Optional[SecurityAuditLogger] = None) -> None:
  1434→        self._findings: List[SecretFinding] = []
  1435→        self._audit = audit_logger
  1436→        self._scan_count = 0
  1437→        self._files_scanned = 0
  1438→        self._lock = threading.Lock()

  1440→    def scan_for_secrets(
  1441→        self,
  1442→        content: str,
  1443→        source: str = "string_input",
  1444→    ) -> List[Dict[str, Any]]:
  1445→        """Scan content for secrets using both built-in and extended patterns.
  1446→
  1447→        This method first uses the existing detect_secrets_in_text() from
  1448→        security.py, then applies additional patterns from this module.
  1449→
  1450→        Args:
  1451→            content: Text content to scan.
  1452→            source: Identifier for where the content came from.
  1453→
  1454→        Returns:
  1455→            List of dicts with keys:
  1456→            - secret_type: str
  1457→            - match: str (redacted for critical/high)
  1458→            - line: int
  1459→            - offset: int
  1460→            - severity: str
  1461→            - description: str
  1462→            - source: str
  1463→            - detector: str ("core" or "extended")
  1464→        """
  1465→        if not content:
  1466→            return []
  1467→
  1468→        self._scan_count += 1
  1469→        results: List[Dict[str, Any]] = []
  1470→        lines = content.split("\n")
  1471→
  1472→        # Phase 1: Use existing core detection from security.py
  1473→        core_findings = detect_secrets_in_text(content)
  1474→        for finding in core_findings:
  1475→            redacted = self._redact_match(finding["match"], finding["confidence"])
  1476→            results.append({
  1477→                "secret_type": finding["type"],
  1478→                "match": redacted,
  1479→                "line": finding["line"],
  1480→                "offset": finding["offset"],
  1481→                "severity": finding["confidence"],
  1482→                "description": self._describe_finding(finding["type"]),
  1483→                "source": source,
  1484→                "detector": "core",
  1485→            })
  1486→
  1487→        # Phase 2: Extended patterns
  1488→        for secret_type, pattern, severity, description in _EXTENDED_SECRET_PATTERNS:
  1489→            for line_num, line in enumerate(lines, start=1):
  1490→                for match in pattern.finditer(line):
  1491→                    matched_text = match.group(0)
  1492→                    redacted = self._redact_match(matched_text, severity)
  1493→                    results.append({
  1494→                        "secret_type": secret_type,
  1495→                        "match": redacted,
  1496→                        "line": line_num,
  1497→                        "offset": match.start(),
  1498→                        "severity": severity,
  1499→                        "description": description,
  1500→                        "source": source,
  1501→                        "detector": "extended",
  1502→                    })
  1503→
  1504→        # Store findings
  1505→        with self._lock:
  1506→            for r in results:
  1507→                self._findings.append(SecretFinding(
  1508→                    secret_type=r["secret_type"],
  1509→                    match=r["match"],
  1510→                    line=r["line"],
  1511→                    offset=r["offset"],
  1512→                    severity=r["severity"],
  1513→                    description=r["description"],
  1514→                    source=source,
  1515→                ))
  1516→
  1517→        # Audit log
  1518→        if results and self._audit:
  1519→            severity_counts: Dict[str, int] = {}
  1520→            for r in results:
  1521→                s = r["severity"]
  1522→                severity_counts[s] = severity_counts.get(s, 0) + 1
  1523→            self._audit.warn(
  1524→                "SECRETS_DETECTED",
  1525→                target=source,
  1526→                details={
  1527→                    "total": len(results),
  1528→                    "severity_counts": severity_counts,
  1529→                    "types": list(set(r["secret_type"] for r in results)),
  1530→                },
  1531→            )
  1532→
  1533→        return results

  1535→    def scan_file(self, filepath: str) -> List[Dict[str, Any]]:
  1536→        """Scan a file for secrets.
  1537→
  1538→        Args:
  1539→            filepath: Path to the file to scan.
  1540→
  1541→        Returns:
  1542→            Same format as scan_for_secrets().

  1545→            return []

  1548→            with open(filepath, "r", encoding="utf-8", errors="ignore") as f:
  1549→                content = f.read()

  1557→    def scan_directory(
  1558→        self,
  1559→        dirpath: str,
  1560→        extensions: Optional[Set[str]] = None,
  1561→        max_files: int = 1000,
  1562→    ) -> List[Dict[str, Any]]:
  1563→        """Recursively scan a directory for secrets.
  1564→
  1565→        Args:
  1566→            dirpath: Root directory to scan.
  1567→            extensions: Set of file extensions to scan (e.g., {".py", ".js"}).
  1568→                      If None, scans all files.
  1569→            max_files: Maximum number of files to scan.
  1570→
  1571→        Returns:
  1572→            Combined list of all findings.
  1573→        """
  1574→        all_results: List[Dict[str, Any]] = []
  1575→        files_scanned = 0
  1576→
  1577→        if not os.path.isdir(dirpath):
  1578→            return []
  1579→
  1580→        for root, _dirs, files in os.walk(dirpath):
  1581→            if files_scanned >= max_files:
  1582→                break
  1583→            for fname in sorted(files):
  1584→                if files_scanned >= max_files:
  1585→                    break
  1586→                # Skip hidden files and common non-text files
  1587→                if fname.startswith("."):
  1588→                    continue
  1589→                if fname.endswith((".pyc", ".pyo", ".so", ".dll", ".exe",
  1590→                                  ".png", ".jpg", ".jpeg", ".gif", ".ico",
  1591→                                  ".zip", ".tar", ".gz", ".bz2", ".xz")):
  1592→                    continue
  1593→
  1594→                filepath = os.path.join(root, fname)
  1595→                if extensions is not None:
  1596→                    _, ext = os.path.splitext(fname)
  1597→                    if ext.lower() not in extensions:
  1598→                        continue
  1599→
  1600→                results = self.scan_file(filepath)
  1601→                all_results.extend(results)
  1602→                files_scanned += 1
  1603→
  1604→        return all_results

  1638→            by_detector[detector] = by_detector.get(detector, 0) + 1

  1640→            if f.severity in ("critical", "high"):

  1645→                    "description": f.description,

  1658→                "files_scanned": self._files_scanned,
  1659→            },
  1660→            "generated_at": datetime.now(timezone.utc).isoformat(),
  1661→        }

  1705→
  1706→class TamperEvidenceLogger:
  1707→    """Tamper-evident security event logger.
  1708→
  1709→    Extends the concept of SecurityAuditLogger with hash-chained entries
  1710→    that make it detectable if log entries have been added, removed, or
  1711→    modified.
  1712→
  1713→    Each log entry includes:
  1714→    - sequence number (monotonically increasing)
  1715→    - timestamp (ISO 8601 UTC)
  1716→    - level (INFO, WARN, ALERT)
  1717→    - event_type
  1718→    - module
  1719→    - target
  1720→    - details (dict)
  1721→    - previous_hash: SHA-256 of the previous entry's canonical form
  1722→    - entry_hash: SHA-256 of this entry (including previous_hash)
  1723→
  1724→    The chain can be verified at any time to detect tampering.
  1725→    """
  1726→
  1727→    def __init__(
  1728→        self,
  1729→        log_dir: Optional[str] = None,
  1730→        module_name: str = "security_hardening",
  1731→        chain_file: Optional[str] = None,
  1732→    ) -> None:
  1733→        self.module_name = module_name
  1734→        if log_dir is None:
  1735→            log_dir = os.path.join(os.path.expanduser("~"), ".reconpro")
  1736→        self.log_dir = log_dir
  1737→        os.makedirs(log_dir, exist_ok=True)
  1738→
  1739→        if chain_file is None:
  1740→            chain_file = os.path.join(log_dir, "security_chain.log")
  1741→        self.chain_file = chain_file
  1742→
  1743→        self._entries: List[Dict[str, Any]] = []
  1744→        self._last_hash: str = "GENESIS"  # Seed for the chain
  1745→        self._sequence: int = 0
  1746→        self._lock = threading.Lock()
  1747→        self._event_counts: Dict[str, int] = {}
  1748→        self._level_counts: Dict[str, int] = {"INFO": 0, "WARN": 0, "ALERT": 0}
  1749→
  1750→        # Try to load existing chain
  1751→        self._load_chain()
  1752→
  1753→    def _load_chain(self) -> None:
  1754→        """Load existing chain from disk to continue the hash sequence."""
  1755→        if not os.path.isfile(self.chain_file):
  1756→            return
  1757→        try:
  1758→            with open(self.chain_file, "r", encoding="utf-8") as f:
  1759→                for line in f:
  1760→                    line = line.strip()
  1761→                    if not line:
  1762→                        continue
  1763→                    try:
  1764→                        entry = json.loads(line)
  1765→                        self._entries.append(entry)
  1766→                        self._sequence = entry.get("sequence", 0)
  1767→                        self._last_hash = entry.get("entry_hash", "GENESIS")
  1768→                        level = entry.get("level", "INFO")
  1769→                        if level in self._level_counts:
  1770→                            self._level_counts[level] += 1
  1771→                        event_type = entry.get("event_type", "")
  1772→                        self._event_counts[event_type] = self._event_counts.get(event_type, 0) + 1
  1773→                    except (json.JSONDecodeError, KeyError):
  1774→                        continue
  1775→        except OSError:
  1776→            pass
  1777→
  1778→    def _compute_hash(self, canonical: str) -> str:
  1779→        """Compute SHA-256 hash of a canonical string."""
  1780→        return hashlib.sha256(canonical.encode("utf-8")).hexdigest()
  1781→
  1782→    def _make_canonical(self, entry: Dict[str, Any]) -> str:
  1783→        """Create a canonical string representation for hashing.
  1784→
  1785→        The canonical form includes all fields in a deterministic order.
  1786→        """
  1787→        # Build a deterministic representation
  1788→        parts = [
  1789→            str(entry.get("sequence", "")),
  1790→            str(entry.get("timestamp", "")),
  1791→            str(entry.get("level", "")),
  1792→            str(entry.get("event_type", "")),
  1793→            str(entry.get("module", "")),
  1794→            sanitize_log(str(entry.get("target", ""))),
  1795→            str(entry.get("previous_hash", "")),
  1796→            json.dumps(entry.get("details", {}), sort_keys=True, default=str),
  1797→        ]
  1798→        return "|".join(parts)
  1799→
  1800→    def log(
  1801→        self,
  1802→        level: str,
  1803→        event_type: str,
  1804→        target: str = "",
  1805→        details: Optional[Dict[str, Any]] = None,
  1806→    ) -> Dict[str, Any]:
  1807→        """Append a tamper-evident log entry.
  1808→
  1809→        Args:
  1810→            level: "INFO", "WARN", or "ALERT".
  1811→            event_type: Category of security event.
  1812→            target: Target associated with the event.
  1813→            details: Optional dict of additional details.
  1814→
  1815→        Returns:
  1816→            The complete log entry dict including hashes.
  1817→        """
  1818→        if level not in ("INFO", "WARN", "ALERT"):
  1819→            level = "INFO"
  1820→
  1821→        with self._lock:
  1822→            self._sequence += 1
  1823→            entry: Dict[str, Any] = {
  1824→                "sequence": self._sequence,
  1825→                "timestamp": datetime.now(timezone.utc).isoformat(),
  1826→                "level": level,
  1827→                "event_type": event_type,
  1828→                "module": self.module_name,
  1829→                "target": sanitize_log(target),
  1830→                "details": details or {},
  1831→                "previous_hash": self._last_hash,
  1832→            }
  1833→
  1834→            # Compute hash of this entry
  1835→            canonical = self._make_canonical(entry)
  1836→            entry["entry_hash"] = self._compute_hash(canonical)
  1837→
  1838→            self._last_hash = entry["entry_hash"]
  1839→            self._entries.append(entry)
  1840→
  1841→            # Update statistics
  1842→            self._level_counts[level] = self._level_counts.get(level, 0) + 1
  1843→            self._event_counts[event_type] = self._event_counts.get(event_type, 0) + 1
  1844→
  1845→            # Persist to disk
  1846→            self._persist_entry(entry)
  1847→
  1848→            return dict(entry)
  1849→
  1850→    def _persist_entry(self, entry: Dict[str, Any]) -> None:
  1851→        """Append a single entry to the chain file on disk."""
  1852→        try:
  1853→            with open(self.chain_file, "a", encoding="utf-8") as f:
  1854→                f.write(json.dumps(entry, default=str) + "\n")
  1855→        except OSError as exc:
  1856→            logger.error("Failed to persist security log entry: %s", exc)

  1858→    def verify_chain(self) -> Dict[str, Any]:
  1859→        """Verify the integrity of the entire log chain.
  1860→
  1861→        Checks that:
  1862→        1. Sequence numbers are monotonically increasing.
  1863→        2. Each entry's hash matches its content.
  1864→        3. Each entry's previous_hash matches the prior entry's hash.
  1865→
  1866→        Returns:
  1867→            Dict with keys:
  1868→            - valid: bool (True if chain is intact)
  1869→            - entries_checked: int
  1870→            - first_violation: dict describing the first issue found (if any)
  1871→            - violations: list of all violation descriptions
  1872→        """
  1873→        violations: List[str] = []
  1874→        first_violation: Optional[Dict[str, Any]] = None
  1875→        prev_hash = "GENESIS"
  1876→        expected_sequence = 1
  1877→
  1878→        for i, entry in enumerate(self._entries):
  1879→            entry_issues: List[str] = []
  1880→
  1881→            # Check sequence
  1882→            if entry.get("sequence") != expected_sequence:
  1883→                issue = (
  1884→                    f"Entry {i}: expected sequence {expected_sequence}, "
  1885→                    f"got {entry.get('sequence')}"
  1886→                )
  1887→                entry_issues.append(issue)
  1888→
  1889→            # Check previous_hash
  1890→            if entry.get("previous_hash") != prev_hash:
  1891→                issue = (
  1892→                    f"Entry {i}: previous_hash mismatch. "
  1893→                    f"Expected {prev_hash[:16]}..., got {str(entry.get('previous_hash', ''))[:16]}..."
  1894→                )
  1895→                entry_issues.append(issue)
  1896→
  1897→            # Check entry_hash
  1898→            entry_copy = {k: v for k, v in entry.items() if k != "entry_hash"}
  1899→            canonical = self._make_canonical(entry_copy)
  1900→            computed_hash = self._compute_hash(canonical)
  1901→            if entry.get("entry_hash") != computed_hash:
  1902→                issue = (
  1903→                    f"Entry {i}: hash mismatch. "
  1904→                    f"Stored {str(entry.get('entry_hash', ''))[:16]}..., "
  1905→                    f"computed {computed_hash[:16]}..."
  1906→                )
  1907→                entry_issues.append(issue)
  1908→
  1909→            if entry_issues:
  1910→                if first_violation is None:
  1911→                    first_violation = {
  1912→                        "entry_index": i,
  1913→                        "sequence": entry.get("sequence"),
  1914→                        "issues": entry_issues,
  1915→                    }
  1916→                violations.extend(entry_issues)
  1917→
  1918→            prev_hash = entry.get("entry_hash", "")
  1919→            expected_sequence += 1
  1920→
  1921→        return {
  1922→            "valid": len(violations) == 0,
  1923→            "entries_checked": len(self._entries),
  1924→            "first_violation": first_violation,
  1925→            "violations": violations,
  1926→            "verified_at": datetime.now(timezone.utc).isoformat(),
  1927→        }
  1928→
  1929→    def get_statistics(self) -> Dict[str, Any]:
  1930→        """Return statistics about logged security events.
  1931→
  1932→        Returns:
  1933→            Dict with keys:
  1934→            - total_entries: int
  1935→            - by_level: dict mapping level -> count
  1936→            - by_event_type: dict mapping event_type -> count
  1937→            - chain_valid: bool
  1938→            - first_entry: ISO timestamp of first entry
  1939→            - last_entry: ISO timestamp of last entry
  1940→        """
  1941→        first_ts = (
  1942→            self._entries[0]["timestamp"]
  1943→            if self._entries
  1944→            else None
  1945→        )
  1946→        last_ts = (
  1947→            self._entries[-1]["timestamp"]
  1948→            if self._entries
  1949→            else None
  1950→        )
  1951→
  1952→        return {
  1953→            "total_entries": len(self._entries),
  1954→            "by_level": dict(self._level_counts),
  1955→            "by_event_type": dict(self._event_counts),
  1956→            "chain_valid": self.verify_chain()["valid"],
  1957→            "first_entry": first_ts,
  1958→            "last_entry": last_ts,
  1959→        }

  1961→    def get_entries(
  1962→        self,
  1963→        level: Optional[str] = None,
  1964→        event_type: Optional[str] = None,
  1965→        limit: int = 100,
  1966→        offset: int = 0,
  1967→    ) -> List[Dict[str, Any]]:
  1968→        """Query log entries with optional filters.
  1969→
  1970→        Args:
  1971→            level: Filter by level (INFO, WARN, ALERT).
  1972→            event_type: Filter by event type.
  1973→            limit: Maximum entries to return.
  1974→            offset: Skip this many entries from the start.
  1975→
  1976→        Returns:
  1977→            List of matching log entry dicts.
  1978→        """
  1979→        filtered: List[Dict[str, Any]] = []
  1980→        for entry in self._entries:
  1981→            if level and entry.get("level") != level:
  1982→                continue
  1983→            if event_type and entry.get("event_type") != event_type:
  1984→                continue
  1985→            filtered.append(dict(entry))
  1986→
  1987→        return filtered[offset:offset + limit]

  1989→    def clear(self) -> None:
  1990→        """Clear all in-memory entries and reset the chain.
  1991→
  1992→        Does NOT delete the on-disk file.
  1993→        """
  1994→        with self._lock:
  1995→            self._entries.clear()
  1996→            self._last_hash = "GENESIS"
  1997→            self._sequence = 0
  1998→            self._event_counts.clear()
  1999→            self._level_counts = {"INFO": 0, "WARN": 0, "ALERT": 0}
