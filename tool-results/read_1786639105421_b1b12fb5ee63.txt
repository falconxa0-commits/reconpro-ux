    18→function tcpProbe(host: string, port: number, timeoutMs: number): Promise<boolean> {
    19→  return new Promise((resolve) => {
    20→    const timer = setTimeout(() => { try { s.destroy(); } catch {} resolve(false); }, timeoutMs);
    21→    const s = net.createConnection({ host, port, timeout: timeoutMs });
    22→    s.on('connect', () => { clearTimeout(timer); try { s.destroy(); } catch {} resolve(true); });
    23→    s.on('timeout', () => { clearTimeout(timer); try { s.destroy(); } catch {} resolve(false); });
    24→    s.on('error', () => { clearTimeout(timer); resolve(false); });
    25→  });
    26→}

    28→function tcpBannerGrab(host: string, port: number, timeoutMs: number, maxBytes = 500): Promise<string> {
    29→  return new Promise((resolve) => {
    30→    let data = '';
    31→    const timer = setTimeout(() => { try { s.destroy(); } catch {} resolve(data); }, timeoutMs);
    32→    const s = net.createConnection({ host, port, timeout: timeoutMs });
    33→    s.on('data', (chunk: Buffer) => {
    34→      data += chunk.toString();
    35→      if (data.length >= maxBytes) { clearTimeout(timer); try { s.destroy(); } catch {} resolve(data.substring(0, maxBytes)); }
    36→    });
    37→    s.on('timeout', () => { clearTimeout(timer); try { s.destroy(); } catch {} resolve(data); });
    38→    s.on('error', () => { clearTimeout(timer); resolve(data); });
    39→    s.on('close', () => { clearTimeout(timer); resolve(data); });
    40→  });
    41→}

    43→async function handleCurl(cmd: string, timeout: number): Promise<string> {
    44→  const isHead = /\b-sI\b|\b-I\b/.test(cmd);
    45→  const followRedirects = /\b-L\b/.test(cmd);
    46→  const timeoutMatch = cmd.match(/--max-time\s+(\d+)/);
    47→  const timeoutSec = timeoutMatch ? parseInt(timeoutMatch[1]) : 8;
    48→  const headers: Record<string, string> = { 'User-Agent': 'Mozilla/5.0' };
    49→  const headerRegex = /-H\s+["']([^"']+)["']/g;
    50→  let m: RegExpExecArray | null;
    51→  while ((m = headerRegex.exec(cmd)) !== null) {
    52→    const ci = m[1].indexOf(':');
    53→    if (ci > 0) headers[m[1].substring(0, ci).trim()] = m[1].substring(ci + 1).trim();
    54→  }
    55→  const urlMatch = cmd.match(/(?:https?:\/\/[^\s"'`]+)/);
    56→  const url = urlMatch ? urlMatch[0] : '';
    57→  if (!url) return '';
    58→
    59→  try {
    60→    if (isHead && !followRedirects) {
    61→      const res = await safeFetch(url, { method: 'HEAD', headers, followRedirects: false, timeout: timeoutSec * 1000 });
    62→      if (!res.ok && res.status === 403) return '';
    63→      let out = `HTTP/1.1 ${res.status}\n`;
    64→      for (const [k, v] of Object.entries(res.headers)) { out += `${k}: ${v}\n`; }
    65→      return out;
    66→    }
    67→    if (isHead && followRedirects) {
    68→      let cur = url; let out = '';
    69→      for (let i = 0; i < 5; i++) {
    70→        const res = await safeFetch(cur, { method: 'HEAD', headers, followRedirects: false, timeout: timeoutSec * 1000 });
    71→        if (!res.ok && res.status === 403) return '';
    72→        out += `HTTP/1.1 ${res.status}\n`;
    73→        for (const [k, v] of Object.entries(res.headers)) { out += `${k}: ${v}\n`; }
    74→        out += '\n';
    75→        if (res.status >= 300 && res.status < 400) {
    76→          const loc = res.headers['location']; if (loc) { cur = new URL(loc, cur).href; continue; }
    77→        }
    78→        break;
    79→      }
    80→      return out;
    81→    }
    82→    const res = await safeFetch(url, { method: 'GET', headers, followRedirects, timeout: timeoutSec * 1000 });
    83→    if (!res.ok && res.status === 403) return '';
    84→    return res.text;
    85→  } catch { return ''; }
    86→}

    88→async function handleDig(cmd: string, timeout: number): Promise<string> {
    89→  if (cmd.includes('axfr')) return '';
    90→  const isPtr = cmd.includes('-x');
    91→  let cleaned = cmd.replace(/\bdig\b/, '').replace(/\+short\b/g, '').replace(/\+time=\d+/g, '').replace(/\+tries=\d+/g, '').replace(/@\S+/g, '').replace(/\+noall\b/g, '').replace(/\+answer\b/g, '').replace(/2>\/>\/dev\/null/g, '').replace(/2>&1/g, '').trim();
    92→  if (isPtr) {
    93→    const pm = cleaned.match(/-x\s+(\S+)/);
    94→    const ip = pm ? pm[1] : '';
    95→    if (!ip) return '';
    96→    try { return (await dns.resolvePtr(ip)).join('\n'); } catch { return ''; }
    97→  }
    98→  const tokens = cleaned.split(/\s+/).filter(Boolean);
    99→  if (tokens.length === 0) return '';
   100→  const recordTypes = ['A','AAAA','NS','CNAME','MX','TXT','PTR','SOA','SRV','ANY'];
   101→  let domain: string; let type = 'A';
   102→  const last = tokens[tokens.length - 1].toUpperCase();
   103→  if (tokens.length >= 2 && recordTypes.includes(last)) {
   104→    type = last; domain = tokens.slice(0, -1).join(' ');
   105→  } else {
   106→    domain = tokens.join(' ');
   107→  }
   108→  if (!domain) return '';
   109→  try {
   110→    let result: string[] = [];
   111→    switch (type) {
   112→      case 'A': result = await dns.resolve4(domain); break;
   113→      case 'NS': result = await dns.resolveNs(domain); break;
   114→      case 'CNAME': result = await dns.resolveCname(domain); break;
   115→      case 'TXT': { const r = await dns.resolveTxt(domain); result = r.map(x => x.join('')); break; }
   116→      case 'PTR': result = await dns.resolvePtr(domain); break;
   117→      case 'ANY': {
   118→        const r = await dns.resolveAny(domain);
   119→        result = r.map(x => { switch (x.type) { case 'A': case 'AAAA': return x.address; case 'MX': return `${x.exchange} ${x.priority}`; case 'TXT': return x.entries?.join('') || ''; case 'NS': case 'CNAME': case 'PTR': return x.value; default: return JSON.stringify(x); } });
   120→        break;
   121→      }
   122→      default: result = await dns.resolve4(domain); break;
   123→    }
   124→    return result.join('\n');
   125→  } catch { return ''; }
   126→}

   128→async function handleOpenssl(cmd: string, timeout: number): Promise<string> {
   129→  const cm = cmd.match(/-connect\s+(\S+:(\d+))/);
   130→  const sm = cmd.match(/-servername\s+(\S+)/);
   131→  const hasTlsextdebug = cmd.includes('-tlsextdebug');
   132→  const hasStatus = cmd.includes('-status');
   133→  let host = '', port = 443;
   134→  if (cm) { const [h, p] = cm[1].split(':'); host = h; port = parseInt(p) || 443; }
   135→  if (!host) return '';
   136→  const servername = sm ? sm[1] : host;
   137→  return new Promise((resolve) => {
   138→    const timer = setTimeout(() => { try { s.destroy(); } catch {} resolve(''); }, timeout);
   139→    const opts: tls.ConnectionOptions = { host, port, servername, rejectUnauthorized: false };
   140→    const s = tls.connect(opts, () => {
   141→      const protocol = s.getProtocol() || '';
   142→      const cipher = s.getCipher();
   143→      const cert = s.getPeerCertificate();
   144→      let out = '';
   145→      out += `    Protocol  : ${protocol}\n`;
   146→      const cipherName = protocol === 'TLSv1.3' ? `ECDHE-RSA-${cipher.name}` : cipher.name;
   147→      out += `    Cipher    : ${cipherName}\n`;
   148→      if (cert) {
   149→        if (cert.valid_from) out += `notBefore=${cert.valid_from}\n`;
   150→        if (cert.valid_to) out += `notAfter=${cert.valid_to}\n`;
   151→      }
   152→      if (hasTlsextdebug) { /* heartbeat not directly exposed in Node.js TLS */ }
   153→      if (hasStatus) { /* OCSP stapling not directly exposed */ }
   154→      clearTimeout(timer); try { s.destroy(); } catch {} resolve(out);
   155→    });
   156→    s.on('timeout', () => { clearTimeout(timer); try { s.destroy(); } catch {} resolve(''); });
   157→    s.on('error', () => { clearTimeout(timer); resolve(''); });
   158→  });
   159→}

   161→async function handleNetcat(cmd: string, timeout: number): Promise<string> {
   162→  const bashMatch = cmd.match(/\/dev\/tcp\/([^/]+)\/(\d+)/);
   163→  if (bashMatch) {
   164→    const open = await tcpProbe(bashMatch[1], parseInt(bashMatch[2]), Math.min(timeout, 3000));
   165→    return open ? 'OPEN' : 'CLOSED';
   166→  }
   167→  const ncMatch = cmd.match(/nc\s+(?:-\w+\s+)*(\S+)\s+(\d+)/);
   168→  if (ncMatch) {
   169→    return tcpBannerGrab(ncMatch[1], parseInt(ncMatch[2]), timeout);
   170→  }
   171→  return '';
   172→}

   174→function applyPipe(input: string, pipeCmd: string): string {
   175→  if (pipeCmd.includes('openssl x509')) {
   176→    return input.split('\n').filter(l => l.includes('notAfter=') || l.includes('notBefore=')).join('\n');
   177→  }
   178→  const headBytes = pipeCmd.match(/head\s+-c\s+(\d+)/);
   179→  if (headBytes) return input.substring(0, parseInt(headBytes[1]));
   180→  const headLines = pipeCmd.match(/head\s+-(\d+)/);
   181→  if (headLines) return input.split('\n').slice(0, parseInt(headLines[1])).join('\n');
   182→  const grepMatch = pipeCmd.match(/^grep\s+(-\w+\s+)?(.+)$/);
   183→  if (grepMatch) {
   184→    const flags = grepMatch[1]?.trim() || '';
   185→    let pattern = grepMatch[2].trim();
   186→    if ((pattern.startsWith('"') && pattern.endsWith('"')) || (pattern.startsWith("'") && pattern.endsWith("'"))) pattern = pattern.slice(1, -1);
   187→    const isOI = flags.includes('o');
   188→    const isI = flags.includes('i');
   189→    try {
   190→      if (isOI) {
   191→        const regex = new RegExp(pattern, isI ? 'gi' : 'g');
   192→        const matches: string[] = []; let match;
   193→        while ((match = regex.exec(input)) !== null) { matches.push(match[0]); if (matches.length > 100) break; }
   194→        return matches.join('\n');
   195→      }
   196→      const regex = new RegExp(pattern, isI ? 'i' : '');
   197→      return input.split('\n').filter(l => regex.test(l)).join('\n');
   198→    } catch { return ''; }
   199→  }
   200→  return input;
   201→}

   203→async function run(cmd: string, timeout = 8000): Promise<string> {
   204→  try {
   205→    if (cmd.includes('/dev/tcp')) {
   206→      const m = cmd.match(/\/dev\/tcp\/([^/]+)\/(\d+)/);
   207→      if (m) { const open = await tcpProbe(m[1], parseInt(m[2]), Math.min(timeout, 3000)); return cmd.includes('OPEN') && cmd.includes('CLOSED') ? (open ? 'OPEN' : 'CLOSED') : ''; }
   208→      return '';
   209→    }
   210→    let effectiveCmd = cmd.replace(/^echo\s*(""\s*)?\|\s*/, '').trim();
   211→    effectiveCmd = effectiveCmd.replace(/\|\|/g, '\x00\x00');
   212→    const segments: string[] = []; let cur = ''; let inQ = false; let qCh = '';
   213→    for (let i = 0; i < effectiveCmd.length; i++) {
   214→      const ch = effectiveCmd[i];
   215→      if (inQ) { cur += ch; if (ch === qCh) inQ = false; }
   216→      else if (ch === '"' || ch === "'") { inQ = true; qCh = ch; cur += ch; }
   217→      else if (ch === '|') { segments.push(cur.trim()); cur = ''; }
   218→      else { cur += ch; }
   219→    }
   220→    segments.push(cur.trim());
   221→    const restored = segments.map(s => s.replace(/\x00\x00/g, '||'));
   222→    const mainCmd = restored[0]; const pipes = restored.slice(1);
   223→    let output = '';
   224→    if (mainCmd.includes('curl')) output = await handleCurl(mainCmd, timeout);
   225→    else if (mainCmd.includes('dig')) output = await handleDig(mainCmd, timeout);
   226→    else if (mainCmd.includes('openssl')) output = await handleOpenssl(mainCmd, timeout);
   227→    else if (mainCmd.includes('nc ') || mainCmd.includes('/dev/tcp')) output = await handleNetcat(mainCmd, timeout);
   228→    else if (mainCmd.includes('whois')) return '';
   229→    else return '';
   230→    for (const pipe of pipes) output = applyPipe(output, pipe);
   231→    return output.trim();
   232→  } catch { return ''; }
   233→}

   342→async function bannerGrab(domain: string, ip: string | null): Promise<{ banners: Array<{ port: number; banner: string; service: string }>; cveMatches: CVEEntry[] }> {
   343→  const target = ip || domain;
   344→  const ports = [21, 22, 25, 80, 110, 143, 443, 445, 993, 995, 1433, 1521, 3306, 3389, 5432, 5900, 6379, 8080, 8443, 8888, 9090, 27017];
   345→  const banners: Array<{ port: number; banner: string; service: string }> = [];
   346→
   347→  for (const port of ports) {
   348→    const banner = await run(
   349→      `echo "" | timeout 4 nc -w3 ${target} ${port} 2>/dev/null | head -c 500`,
   350→      6000
   351→    );
   352→    if (banner && banner.length > 3) {
   353→      const service = detectService(port, banner);
   354→      banners.push({ port, banner: banner.replace(/[\x00-\x1f\x7f]/g, '.').substring(0, 200), service });
   355→    }
   356→  }
   357→
   358→  // HTTP banner via curl
   359→  const httpBanner = await run(`curl -sI --max-time 6 https://${domain} 2>/dev/null | head -20`, 8000);
   360→  if (httpBanner) banners.push({ port: 443, banner: httpBanner.substring(0, 300), service: 'http' });
   361→  const httpBanner2 = await run(`curl -sI --max-time 6 http://${domain} 2>/dev/null | head -20`, 8000);
   362→  if (httpBanner2) banners.push({ port: 80, banner: httpBanner2.substring(0, 300), service: 'http' });
   363→
   364→  // Match banners against CVE database
   365→  const combined = banners.map(b => b.banner).join(' ').toLowerCase();
   366→  const cveMatches = CVE_DATABASE.filter(cve => {
   367→    return cve.affected.some(a => combined.includes(a)) && cve.versionPattern.test(combined);
   368→  });
   369→
   370→  return { banners, cveMatches };
   371→}
   372→
   373→function detectService(port: number, banner: string): string {
   374→  const b = banner.toLowerCase();
   375→  if (port === 22 || b.includes('ssh')) return 'openssh';
   376→  if (port === 21 || b.includes('ftp')) return 'ftp';
   377→  if (port === 25 || b.includes('smtp') || b.includes('postfix') || b.includes('esmtp')) return 'postfix';
   378→  if (port === 80 || port === 8080 || port === 8443 || b.includes('http')) return 'http';
   379→  if (port === 110 || b.includes('pop3')) return 'pop3';
   380→  if (port === 143 || b.includes('imap')) return 'imap';
   381→  if (port === 443 || b.includes('ssl') || b.includes('tls')) return 'https';
   382→  if (port === 3306 || b.includes('mysql')) return 'mysql';
   383→  if (port === 5432 || b.includes('postgresql') || b.includes('postgres')) return 'postgresql';
   384→  if (port === 6379 || b.includes('redis')) return 'redis';
   385→  if (port === 27017 || b.includes('mongodb')) return 'mongodb';
   386→  if (b.includes('nginx')) return 'nginx';
   387→  if (b.includes('apache')) return 'apache';
   388→  if (b.includes('tomcat')) return 'tomcat';
   389→  return 'unknown';
   390→}

   396→async function scanHTTPVulns(domain: string): Promise<Finding[]> {
   397→  const findings: Finding[] = [];
   398→
   399→  // CORS misconfiguration
   400→  const corsResponse = await run(
   401→    `curl -sI --max-time 6 -H "Origin: https://evil.attacker.com" https://${domain} 2>/dev/null | grep -i "access-control-allow-origin"`,
   402→    8000
   403→  );
   404→  if (corsResponse.includes('evil.attacker.com') || corsResponse.includes('*')) {
   405→    findings.push({
   406→      title: 'CRITICAL: CORS Misconfiguration — Reflects Attacker Origin',
   407→      severity: 'critical', category: 'http-vuln',
   408→      description: `Server reflects any Origin header. An attacker at evil.attacker.com can make cross-origin requests and read the response, enabling credential theft and data exfiltration from any authenticated user.`,
   409→      evidence: `curl -H "Origin: https://evil.attacker.com" → ${corsResponse.trim()}`,
   410→      asset: domain,
   411→    });
   412→  }
   413→
   414→  // Open redirect test
   415→  const redirectPaths = ['/redirect?url=https://evil.com', '/login?return=https://evil.com',
   416→    '/logout?redirect=https://evil.com', '/next=https://evil.com', '/goto=https://evil.com',
   417→    '/auth?redirect_uri=https://evil.com', '/callback?redirect=https://evil.com'];
   418→  for (const path of redirectPaths) {
   419→    const redirResult = await run(
   420→      `curl -sI --max-time 4 -L "https://${domain}${path}" 2>/dev/null | grep -i "location"`,
   421→      6000
   422→    );
   423→    if (redirResult.includes('evil.com') || redirResult.includes('evil')) {
   424→      findings.push({
   425→        title: 'Open Redirect Vulnerability Detected',
   426→        severity: 'high', category: 'http-vuln',
   427→        description: `Path ${path} redirects to external domain without validation. Attackers use open redirects for phishing campaigns that appear to come from legitimate domain.`,
   428→        evidence: `GET ${path} → Location: ${redirResult.trim().substring(0, 100)}`,
   429→        asset: `${domain}${path}`,
   430→      });
   431→      break;
   432→    }
   433→  }
   434→
   435→  // Path traversal tests
   436→  const traversalPaths = ['../../../etc/passwd', '/static/../../etc/passwd', '/images/..%2f..%2f..%2fetc/passwd', '/file/..%252f..%252fetc/passwd'];
   437→  for (const path of traversalPaths) {
   438→    const ptResult = await run(
   439→      `curl -s --max-time 4 "https://${domain}${path}" 2>/dev/null | head -c 500`,
   440→      6000
   441→    );
   442→    if (ptResult.includes('root:') || ptResult.includes('nobody:') || ptResult.includes('/bin/bash')) {
   443→      findings.push({
   444→        title: 'CRITICAL: Path Traversal — /etc/passwd Exposed',
   445→        severity: 'critical', category: 'http-vuln',
   446→        description: `Server responds with file system contents. Path traversal allows reading arbitrary files including /etc/passwd, /etc/shadow, and application secrets.`,
   447→        evidence: `GET ${path} → "root:x:0:0:..." found in response`,
   448→        asset: `${domain}${path}`,
   449→      });
   450→      break;
   451→    }
   452→  }
   453→
   454→  // SSRF detection (check if internal IPs are accessible)
   455→  const ssrfPaths = ['/?url=http://169.254.169.254/latest/meta-data/', '/api/fetch?url=http://127.0.0.1:80', '/proxy?url=http://10.0.0.1'];
   456→  for (const path of ssrfPaths) {
   457→    const ssrfResult = await run(
   458→      `curl -s --max-time 4 "https://${domain}${path}" 2>/dev/null | head -c 500`,
   459→      6000
   460→    );
   461→    if (ssrfResult.includes('ami-id') || ssrfResult.includes('instance-id') || ssrfResult.includes('local-hostname') || ssrfResult.includes('127.0.0.1')) {
   462→      findings.push({
   463→        title: 'CRITICAL: SSRF — Internal Network Accessible',
   464→        severity: 'critical', category: 'http-vuln',
   465→        description: `Server makes requests to internal URLs. SSRF can access cloud metadata (AWS/GCP), internal services, and bypass firewalls.`,
   466→        evidence: `GET ${path} → internal response detected`,
   467→        asset: `${domain}${path}`,
   468→      });
   469→      break;
   470→    }
   471→  }
   472→
   473→  // XSS detection (reflected input)
   474→  const xssPayloads = ['<script>alert(1)</script>', '"><img src=x onerror=alert(1)>', "'-alert(1)-'"];
   475→  for (const payload of xssPayloads) {
   476→    const xssResult = await run(
   477→      `curl -s --max-time 5 "https://${domain}/search?q=${encodeURIComponent(payload)}" 2>/dev/null | head -c 2000`,
   478→      7000
   479→    );
   480→    if (xssResult.includes(payload) && !xssResult.includes('alert(1)</script></') === false) {
   481→      const reflected = xssResult.includes('<script>alert(1)</script>') || xssResult.includes('onerror=alert(1)');
   482→      if (reflected) {
   483→        findings.push({
   484→          title: 'Reflected XSS — Input Executed in Response',
   485→          severity: 'high', category: 'http-vuln',
   486→          description: `User input is reflected in page content without sanitization. XSS enables session hijacking, credential theft, and malware delivery.`,
   487→          evidence: `GET /search?q=<payload> → reflected unescaped in HTML`,
   488→          asset: `${domain}/search`,
   489→        });
   490→        break;
   491→      }
   492→    }
   493→  }
   494→
   495→  // Clickjacking (X-Frame-Options)
   496→  const xfo = await run(`curl -sI --max-time 5 https://${domain} 2>/dev/null | grep -i "x-frame-options"`, 7000);
   497→  if (!xfo) {
   498→    const cspFrame = await run(`curl -sI --max-time 5 https://${domain} 2>/dev/null | grep -i "content-security-policy" | grep -i "frame-ancestors"`, 7000);
   499→    if (!cspFrame) {
   500→      findings.push({
   501→        title: 'Clickjacking — No Frame Protection',
   502→        severity: 'medium', category: 'http-vuln',
   503→        description: 'No X-Frame-Options or frame-ancestors CSP directive. Attacker can embed site in iframe to trick users into clicking hidden buttons (account takeover, unauthorized actions).',
   504→        evidence: 'No X-Frame-Options header, no frame-ancestors in CSP',
   505→        asset: domain,
   506→      });
   507→    }
   508→  }
   509→
   510→  // Cookie security analysis
   511→  const setCookies = await run(`curl -sI --max-time 5 https://${domain} 2>/dev/null | grep -i "set-cookie"`, 7000);
   512→  if (setCookies) {
   513→    const insecureCookies: string[] = [];
   514→    if (!setCookies.toLowerCase().includes('secure')) insecureCookies.push('Secure flag missing');
   515→    if (!setCookies.toLowerCase().includes('httponly')) insecureCookies.push('HttpOnly flag missing');
   516→    if (!setCookies.toLowerCase().includes('samesite')) insecureCookies.push('SameSite missing');
   517→
   518→    if (insecureCookies.length > 0) {
   519→      findings.push({
   520→        title: `Insecure Cookie Configuration — ${insecureCookies.length} issues`,
   521→        severity: 'medium', category: 'http-vuln',
   522→        description: `Cookies lack security flags: ${insecureCookies.join(', ')}. Without Secure, cookies sent over HTTP. Without HttpOnly, accessible to JavaScript (XSS target). Without SameSite, vulnerable to CSRF.`,
   523→        evidence: `Set-Cookie: ${setCookies.substring(0, 100)} | Issues: ${insecureCookies.join(', ')}`,
   524→        asset: domain,
   525→      });
   526→    }
   527→  }
   528→
   529→  // Information disclosure — server version
   530→  const serverHeader = await run(`curl -sI --max-time 5 https://${domain} 2>/dev/null | grep -i "^server:"`, 7000);
   531→  if (serverHeader && serverHeader.match(/apache\/\d|nginx\/\d|php\/\d|tomcat|express|gunicorn/i)) {
   532→    findings.push({
   533→      title: 'Information Disclosure — Server Version Leaked',
   534→      severity: 'low', category: 'http-vuln',
   535→      description: `Server header reveals exact software version: ${serverHeader.trim()}. Attackers use version info to find known exploits.`,
   536→      evidence: serverHeader.trim(),
   537→      asset: domain,
   538→    });
   539→  }
   540→
   541→  // CSRF token check
   542→  const csrfCheck = await run(`curl -s --max-time 5 "https://${domain}/login" 2>/dev/null | grep -iE "csrf|_token|authenticity_token|x-csrf" | head -3`, 7000);
   543→  if (!csrfCheck) {
   544→    findings.push({
   545→      title: 'CSRF Protection — No Anti-CSRF Token Detected',
   546→      severity: 'medium', category: 'http-vuln',
   547→      description: 'No CSRF token found in login form. State-changing requests may be forged from other sites, enabling unauthorized actions on behalf of authenticated users.',
   548→      evidence: 'No csrf_token, authenticity_token, or x-csrf-token in login page',
   549→      asset: `${domain}/login`,
   550→    });
   551→  }
   552→
   553→  // Mixed content detection
   554→  const mixedContent = await run(
   555→    `curl -s --max-time 8 https://${domain} 2>/dev/null | grep -oiE 'src="http://[^"]+"|href="http://[^"]+"|url\("http://[^)]+"\)' | head -5`,
   556→    10000
   557→  );
   558→  if (mixedContent) {
   559→    findings.push({
   560→      title: `Mixed Content — ${mixedContent.split('\n').filter(Boolean).length} HTTP Resources on HTTPS Page`,
   561→      severity: 'low', category: 'http-vuln',
   562→      description: 'HTTPS page loads resources via HTTP. Browsers will show warnings, and MITM can inject content into the insecure resources.',
   563→      evidence: mixedContent.trim().substring(0, 150),
   564→      asset: domain,
   565→    });
   566→  }
   567→
   568→  return findings;
   569→}

   572→// 3. SSL/TLS ATTACK VECTOR SCANNER
   573→// ══════════════════════════════════════════════════════════════════════════════

   575→async function scanSSLVulns(domain: string): Promise<Finding[]> {
   576→  const findings: Finding[] = [];
   577→
   578→  // Full SSL scan with all cipher suites
   579→  const sslFull = await run(`echo | openssl s_client -connect ${domain}:443 -servername ${domain} 2>/dev/null`, 10000);
   580→  const sslLines = sslFull.split('\n').map(l => l.trim()).filter(Boolean);
   581→
   582→  // Protocol version checks
   583→  const protocolLine = sslLines.find(l => l.startsWith('Protocol'));
   584→  const proto = protocolLine || '';
   585→
   586→  if (proto.includes('TLSv1.0')) {
   587→    findings.push({
   588→      title: 'BEAST Attack Vulnerable — TLS 1.0 Active',
   589→      severity: 'high', category: 'ssl-vuln',
   590→      description: 'TLS 1.0 is vulnerable to BEAST (Browser Exploit Against SSL/TLS). Block cipher attack can decrypt communication blocks. PCI-DSS prohibits TLS 1.0.',
   591→      evidence: proto,
   592→      asset: `${domain}:443`,
   593→    });
   594→  }
   595→  if (proto.includes('TLSv1.1')) {
   596→    findings.push({
   597→      title: 'Deprecated TLS 1.1 in Use',
   598→      severity: 'medium', category: 'ssl-vuln',
   599→      description: 'TLS 1.1 is deprecated (RFC 8996). Remove support to prevent downgrade attacks.',
   600→      evidence: proto,
   601→      asset: `${domain}:443`,
   602→    });
   603→  }
   604→  if (proto.includes('SSLv3')) {
   605→    findings.push({
   606→      title: 'CRITICAL: POODLE — SSLv3 Detected',
   607→      severity: 'critical', category: 'ssl-vuln',
   608→      description: 'SSLv3 is extremely vulnerable to POODLE (Padding Oracle On Downgraded Legacy Encryption). Complete decryption of encrypted sessions possible.',
   609→      evidence: proto,
   610→      asset: `${domain}:443`,
   611→    });
   612→  }
   613→  if (proto.includes('SSLv2')) {
   614→    findings.push({
   615→      title: 'CRITICAL: DROWN Attack — SSLv2 Detected',
   616→      severity: 'critical', category: 'ssl-vuln',
   617→      description: 'SSLv2 enables DROWN (Decrypting RSA with Obsolete and Weakened eNcryption). Cross-protocol attack can decrypt TLS connections.',
   618→      evidence: proto,
   619→      asset: `${domain}:443`,
   620→    });
   621→  }
   622→
   623→  // Cipher analysis
   624→  const cipherLine = sslLines.find(l => l.includes('Cipher'));
   625→  const cipher = cipherLine || '';
   626→
   627→  // Check for weak ciphers by testing multiple
   628→  const weakCiphers = ['RC4', 'DES', '3DES', 'NULL', 'EXPORT', 'MD5', 'RC2', 'IDEA'];
   629→  const cipherSuite = await run(
   630→    `echo | openssl s_client -connect ${domain}:443 -servername ${domain} -cipher 'ALL:eNULL' 2>/dev/null | grep "Cipher"`,
   631→    10000
   632→  );
   633→
   634→  for (const weak of weakCiphers) {
   635→    if (cipherSuite.toUpperCase().includes(weak)) {
   636→      findings.push({
   637→        title: `CRITICAL: Weak Cipher ${weak} Accepted`,
   638→        severity: 'critical', category: 'ssl-vuln',
   639→        description: `Server accepts ${weak} cipher suite. ${weak === 'RC4' ? 'RC4 has known statistical biases exploitable since 2013.' : weak === 'NULL' ? 'NULL cipher means NO encryption — traffic is plaintext.' : weak === 'EXPORT' ? 'EXPORT-grade ciphers are trivially breakable (40-bit keys).' : `${weak} is cryptographically broken and must be disabled.`}`,
   640→        evidence: `Cipher: ${cipherSuite.trim()}`,
   641→        asset: `${domain}:443`,
   642→      });
   643→      break;
   644→    }
   645→  }
   646→
   647→  if (cipher.includes('CBC') && proto.includes('TLSv1')) {
   648→    findings.push({
   649→      title: 'BEAST Susceptible — CBC Mode + TLS 1.0',
   650→      severity: 'medium', category: 'ssl-vuln',
   651→      description: 'CBC cipher suites with TLS 1.0 are vulnerable to BEAST attack. Use GCM-mode ciphers or upgrade to TLS 1.2+.',
   652→      evidence: `${proto} | ${cipher}`,
   653→      asset: `${domain}:443`,
   654→    });
   655→  }
   656→
   657→  // Heartbleed check (CVE-2014-0160)
   658→  const heartbleedCheck = await run(
   659→    `echo | openssl s_client -connect ${domain}:443 -servername ${domain} -tlsextdebug 2>&1 | grep -i "heartbeat"`,
   660→    8000
   661→  );
   662→  if (heartbleedCheck) {
   663→    findings.push({
   664→      title: 'Heartbleed Vulnerability Possible (CVE-2014-0160)',
   665→      severity: 'critical', category: 'ssl-vuln',
   666→      description: 'Server supports heartbeat extension. If running vulnerable OpenSSL (1.0.1-1.0.1f), up to 64KB of server memory can be leaked per request — including private keys, passwords, session tokens.',
   667→      evidence: `Heartbeat extension: ${heartbleedCheck.trim()}`,
   668→      asset: `${domain}:443`,
   669→    });
   670→  }
   671→
   672→  // Certificate issues
   673→  const certDates = await run(
   674→    `echo | openssl s_client -connect ${domain}:443 -servername ${domain} 2>/dev/null | openssl x509 -noout -dates 2>/dev/null`,
   675→    10000
   676→  );
   677→  if (certDates) {
   678→    const notAfter = certDates.match(/notAfter=(.+)/);
   679→    if (notAfter) {
   680→      const expiry = new Date(notAfter[1].trim());
   681→      const now = new Date();
   682→      const daysLeft = Math.floor((expiry.getTime() - now.getTime()) / 86400000);
   683→      if (daysLeft < 0) {
   684→        findings.push({
   685→          title: 'CRITICAL: SSL Certificate EXPIRED',
   686→          severity: 'critical', category: 'ssl-vuln',
   687→          description: `Certificate expired ${Math.abs(daysLeft)} days ago. Visitors see security warnings. Connections may be intercepted.`,
   688→          evidence: `notAfter: ${notAfter[1].trim()} | Expired ${Math.abs(daysLeft)} days ago`,
   689→          asset: `${domain}:443`,
   690→        });
   691→      } else if (daysLeft < 30) {
   692→        findings.push({
   693→          title: `SSL Certificate Expiring in ${daysLeft} Days`,
   694→          severity: 'high', category: 'ssl-vuln',
   695→          description: `Certificate expires soon. Service disruption and security warnings imminent.`,
   696→          evidence: `notAfter: ${notAfter[1].trim()} | ${daysLeft} days remaining`,
   697→          asset: `${domain}:443`,
   698→        });
   699→      }
   700→    }
   701→  }
   702→
   703→  // OCSP stapling
   704→  const ocsp = await run(
   705→    `echo | openssl s_client -connect ${domain}:443 -servername ${domain} -status 2>/dev/null | grep -i "OCSP"`,
   706→    8000
   707→  );
   708→  if (!ocsp) {
   709→    findings.push({
   710→      title: 'OCSP Stapling Not Enabled',
   711→      severity: 'low', category: 'ssl-vuln',
   712→      description: 'Without OCSP stapling, browsers must contact CA separately to verify certificate revocation, adding latency and privacy exposure.',
   713→      evidence: 'No OCSP response in handshake',
   714→      asset: `${domain}:443`,
   715→    });
   716→  }
   717→
   718→  // Perfect Forward Secrecy
   719→  const pfsCheck = await run(
   720→    `echo | openssl s_client -connect ${domain}:443 -servername ${domain} 2>/dev/null | grep -iE "ECDHE-RSA|ECDHE-ECDSA|DHE-RSA|DHE-DSS"`,
   721→    8000
   722→  );
   723→  if (!pfsCheck) {
   724→    findings.push({
   725→      title: 'Perfect Forward Secrecy — Not Supported',
   726→      severity: 'medium', category: 'ssl-vuln',
   727→      description: 'Without PFS, compromise of server private key allows decryption of ALL past sessions captured via passive surveillance.',
   728→      evidence: 'No ECDHE/DHE cipher negotiated',
   729→      asset: `${domain}:443`,
   730→    });
   731→  }
   732→
   733→  return findings;
   734→}

   737→// 4. DNS VULNERABILITY ASSESSMENT
   738→// ══════════════════════════════════════════════════════════════════════════════
   739→
   740→async function scanDNSVulns(domain: string): Promise<Finding[]> {
   741→  const findings: Finding[] = [];
   742→
   743→  // Zone transfer (AXFR) attempt
   744→  const nsRecords = await run(`dig +short +time=3 ${domain} NS`, 5000);
   745→  const nsList = nsRecords.split('\n').filter(Boolean);
   746→  for (const ns of nsList.slice(0, 3)) {
   747→    const axfrResult = await run(`dig axfr ${domain} @${ns} +time=5 +tries=1 2>/dev/null`, 7000);
   748→    if (axfrResult && axfrResult.includes('ANSWER SECTION') && axfrResult.split('\n').length > 5) {
   749→      findings.push({
   750→        title: 'CRITICAL: DNS Zone Transfer (AXFR) Successful',
   751→        severity: 'critical', category: 'dns-vuln',
   752→        description: `Full zone transfer from ${ns} revealed complete DNS records including internal hostnames, IP addresses, and service records. Complete infrastructure mapping possible.`,
   753→        evidence: `dig axfr ${domain} @${ns} → ${axfrResult.split('\n').length} records transferred`,
   754→        asset: ns,
   755→      });
   756→      break;
   757→    }
   758→  }
   759→
   760→  // Subdomain takeover detection (CNAME to dead services)
   761→  const takeoverSubs = ['staging', 'dev', 'blog', 'cdn', 'assets', 'static', 'mail', 'api', 'app', 'test'];
   762→  const takeoverCandidates: string[] = [];
   763→  for (const sub of takeoverSubs) {
   764→    const cname = await run(`dig +short +time=2 ${sub}.${domain} CNAME`, 4000);
   765→    if (cname) {
   766→      const deadServices = ['herokuapp.com', 'cloudfront.net', 'github.io', 's3.amazonaws.com',
   767→        'shopify.com', 'fastly.net', 'cloudwaysapps.com', 'pantheon.io'];
   768→      for (const dead of deadServices) {
   769→        if (cname.toLowerCase().includes(dead)) {
   770→          const subExists = await run(`curl -sI --max-time 4 https://${sub}.${domain} 2>/dev/null | head -1`, 6000);
   771→          if (subExists && !subExists.includes('200') && !subExists.includes('301') && !subExists.includes('302')) {
   772→            takeoverCandidates.push(`${sub}.${domain} → CNAME: ${cname} (${dead})`);
   773→          }
   774→        }
   775→      }
   776→    }
   777→  }
   778→  if (takeoverCandidates.length > 0) {
   779→    findings.push({
   780→      title: `Subdomain Takeover Risk — ${takeoverCandidates.length} candidate(s)`,
   781→      severity: 'high', category: 'dns-vuln',
   782→      description: `Subdomains point to third-party services via CNAME but may not be properly provisioned. If the third-party resource is deleted, an attacker can claim the subdomain and serve malicious content.`,
   783→      evidence: takeoverCandidates.join(' | '),
   784→      asset: domain,
   785→    });
   786→  }
   787→
   788→  // DNS cache snooping
   789→  const cacheResult = await run(`dig +time=3 ${domain} @8.8.8.8 ANY +noall +answer 2>/dev/null`, 5000);
   790→  if (cacheResult && cacheResult.split('\n').length > 8) {
   791→    findings.push({
   792→      title: 'DNS Cache Snooping Possible — Excessive Records Visible',
   793→      severity: 'medium', category: 'dns-vuln',
   794→      description: `Recursive DNS query reveals extensive record types. DNS cache can be probed to determine which domains are recently visited by the resolver's clients.`,
   795→      evidence: `dig ANY ${domain} → ${cacheResult.split('\n').length} records from cache`,
   796→      asset: domain,
   797→    });
   798→  }
   799→
   800→  // DNS rebinding protection check
   801→  const dnsRebindHeaders = await run(`curl -sI --max-time 5 https://${domain} 2>/dev/null | grep -iE "x-dns-prefetch|x-content-type|dns-prefetch"`, 7000);
   802→  if (!dnsRebindHeaders) {
   803→    findings.push({
   804→      title: 'DNS Rebinding — No Protection Detected',
   805→      severity: 'medium', category: 'dns-vuln',
   806→      description: 'No DNS rebinding protection headers. Attackers can bypass Same-Origin Policy by controlling DNS to oscillate between their server and target IP.',
   807→      evidence: 'No DNS rebinding protection headers found',
   808→      asset: domain,
   809→    });
   810→  }
   811→
   812→  return findings;
   813→}

   815→// ══════════════════════════════════════════════════════════════════════════════
   816→// MAIN POST HANDLER
   817→// ══════════════════════════════════════════════════════════════════════════════
   818→
   819→export async function POST(request: NextRequest) {
   820→  const { allowed } = checkRateLimit(extractClientIP(request), 30, 60000);
   821→  if (!allowed) return NextResponse.json({ error: 'Rate limit exceeded' }, { status: 429 });
   822→
   823→  try {
   824→    const body = await request.json();
   825→    const target = body.target?.toString().trim().replace(/^https?:\/\//, '').replace(/\/.*$/, '');
   826→    const sanitized = sanitizeDomain(target);
   827→    if (!sanitized) {
   828→      return NextResponse.json({ error: 'Invalid target domain' }, { status: 400 });
   829→    }
   830→    if (isBlockedDomain(sanitized)) {
   831→      return NextResponse.json({ error: 'Target domain is not permitted' }, { status: 403 });
   832→    }
   833→
   834→    // Resolve IP first
   835→    const ip = await run(`dig +short +time=3 +tries=1 ${sanitized} A`, 5000);
   836→    const mainIp = ip?.split('\n')[0]?.trim() || null;
   837→    if (mainIp && isPrivateIP(mainIp)) {
   838→      return NextResponse.json({ error: 'Target resolves to a private or reserved IP address' }, { status: 403 });
   839→    }
   840→
   841→    // Run all modules in parallel
   842→    const [
   843→      bannerResult,
   844→      httpVulns,
   845→      sslVulns,
   846→      dnsVulns,
   847→    ] = await Promise.all([
   848→      bannerGrab(sanitized, mainIp),
   849→      scanHTTPVulns(sanitized),
   850→      scanSSLVulns(sanitized),
   851→      scanDNSVulns(sanitized),
   852→    ]);
   853→
   854→    // Build response
   855→    const allFindings = [...httpVulns, ...sslVulns, ...dnsVulns];
   856→    const c = allFindings.filter(f => f.severity === 'critical').length;
   857→    const h = allFindings.filter(f => f.severity === 'high').length;
   858→    const m = allFindings.filter(f => f.severity === 'medium').length;
   859→    const exploitSummary = {
   860→      totalCVEs: bannerResult.cveMatches.length,
   861→      criticalCVEs: bannerResult.cveMatches.filter(v => v.severity === 'critical').length,
   862→      weaponized: bannerResult.cveMatches.filter(v => v.exploitAvailable === 'weaponized').length,
   863→      theoretical: bannerResult.cveMatches.filter(v => v.exploitAvailable === 'theoretical').length,
   864→      avgCVSS: bannerResult.cveMatches.length > 0
   865→        ? Math.round(bannerResult.cveMatches.reduce((a, b) => a + b.cvss, 0) / bannerResult.cveMatches.length * 10) / 10
   866→        : 0,
   867→    };
   868→
   869→    return applySecurityHeaders(NextResponse.json({
   870→      success: true,
   871→      scan: {
   872→        target,
   873→        ip: mainIp,
   874→        cveFindings: bannerResult.cveMatches.map(v => ({
   875→          cve: v.id, title: v.title, cvss: v.cvss, severity: v.severity,
   876→          affected: v.affected.join(', '), evidence: `Version pattern match via banner grab`,
   877→          remediation: v.remediation, exploitAvailable: v.exploitAvailable, epss: v.epss,
   878→        })),
   879→        httpVulns: httpVulns.map(f => ({
   880→          ...f, proof: f.evidence,
   881→        })),
   882→        sslVulns: sslVulns.map(f => ({
   883→          ...f, risk: f.severity,
   884→        })),
   885→        dnsVulns,
   886→        banners: bannerResult.banners,
   887→        exploitSummary,
   888→        attackSurface: {
   889→          score: Math.min(100, c * 25 + h * 15 + m * 8 + exploitSummary.weaponized * 20),
   890→          vectors: [
   891→            ...(bannerResult.cveMatches.length > 0 ? ['Known CVE Exploits'] : []),
   892→            ...(httpVulns.some(f => f.severity === 'critical') ? ['Web Application Exploits'] : []),
   893→            ...(sslVulns.some(f => f.severity === 'critical') ? ['Cryptographic Attacks'] : []),
   894→            ...(dnsVulns.some(f => f.severity === 'critical') ? ['DNS Infrastructure Attacks'] : []),
   895→            ...(exploitSummary.weaponized > 0 ? ['Weaponized Exploits Available'] : []),
   896→          ],
   897→        },
   898→      },
   899→    }));
   900→  } catch (error) {
   901→    return safeErrorResponse(error, 500, 'vuln-scan');
   902→  }
   903→}
