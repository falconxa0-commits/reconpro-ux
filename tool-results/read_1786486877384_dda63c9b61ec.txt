   365→   365→    def _build_risk_matrix(
   366→   366→        self,
   367→   367→        findings: List[Dict[str, Any]],
   368→   368→        confidence_scores: List[Dict[str, Any]],
   369→   369→    ) -> List[Dict[str, Any]]:
   370→   370→        """Build risk matrix entries (severity x likelihood)."""
   371→   371→        # Build a confidence lookup by finding title.
   372→   372→        conf_map: Dict[str, float] = {}
   373→   373→        for cs in confidence_scores:
   374→   374→            title = cs.get("title", "")
   375→   375→            conf_val = cs.get("confidence", 0.5)
   376→   376→            if isinstance(conf_val, (int, float)):
   377→   377→                conf_map[title] = float(conf_val)
   378→   378→            else:
   379→   379→                conf_map[title] = 0.5
   380→   380→
   381→   381→        entries: List[RiskMatrixEntry] = []
   382→   382→        for f in findings:
   383→   383→            title = f.get("title", "Untitled Finding")
   384→   384→            severity = str(f.get("severity", "info")).strip().lower()
   385→   385→            category = str(f.get("category", "general")).strip().lower()
   386→   386→
   387→   387→            likelihood = conf_map.get(title, self._default_likelihood(severity))
   388→   388→            impact = _SEVERITY_IMPACT.get(severity, 0.2)
   389→   389→            risk_score = likelihood * impact
   390→   390→            recommendation = _REMEDIATION_TEMPLATES.get(
   391→   391→                category, f"Review and remediate the {category} finding."
   392→   392→            )
   393→   393→
   394→   394→            entries.append(RiskMatrixEntry(
   395→   395→                finding_title=title,
   396→   396→                severity=severity,
   397→   397→                likelihood=likelihood,
   398→   398→                impact=impact,
   399→   399→                risk_score=risk_score,
   400→   400→                category=_CATEGORY_LABELS.get(category, category),
   401→   401→                recommendation=recommendation,
   402→   402→            ))
   403→   403→
   404→   404→        # Sort by risk_score descending.
   405→   405→        entries.sort(key=lambda e: e.risk_score, reverse=True)
   406→   406→        return [e.to_dict() for e in entries]

   408→   408→    @staticmethod
   409→   409→    def _default_likelihood(severity: str) -> float:
   410→   410→        """Estimate likelihood from severity when no confidence score exists."""
   411→   411→        mapping = {
   412→   412→            "critical": 0.9,
   413→   413→            "high": 0.75,
   414→   414→            "medium": 0.5,
   415→   415→            "low": 0.3,
   416→   416→            "info": 0.1,
   417→   417→        }
   418→   418→        return mapping.get(severity, 0.5)

   431→   431→        seen_titles: set = set()

   439→   439→                    timeline.append({

   447→   447→
   448→   448→        # Phase 2: Access / auth findings.

   450→   450→            cat = str(f.get("category", "")).lower()
   451→   451→            if cat in ("auth", "session", "csrf"):
   452→   452→                title = f.get("title", "")
   453→   453→                if title not in seen_titles:
   454→   454→                    timeline.append({
   455→   455→                        "phase": 2,

   457→   457→                        "finding": title,

   459→   459→                        "description": f.get("description", f.get("evidence", ""))[:200],
   460→   460→                    })
   461→   461→                    seen_titles.add(title)
   462→   462→
   463→   463→        # Phase 3: Exploitation findings.
   464→   464→        for f in findings:
   465→   465→            cat = str(f.get("category", "")).lower()
   466→   466→            if cat in ("injection", "xss", "ssti", "ssrf", "command_injection",
   467→   467→                         "path_traversal", "unsafe_deserialization", "dangerous_eval"):
   468→   468→                title = f.get("title", "")
   469→   469→                if title not in seen_titles:
   470→   470→                    timeline.append({
   471→   471→                        "phase": 3,
   472→   472→                        "phase_name": "Exploitation",
   473→   473→                        "finding": title,
   474→   474→                        "severity": f.get("severity", "info"),
   475→   475→                        "description": f.get("description", f.get("evidence", ""))[:200],

   515→   515→    def _prioritize_findings(
   516→   516→        self,
   517→   517→        findings: List[Dict[str, Any]],
   518→   518→        confidence_scores: List[Dict[str, Any]],
   519→   519→    ) -> List[Dict[str, Any]]:
   520→   520→        """Rank findings by severity_weight * confidence, take top 10."""
   521→   521→        conf_map: Dict[str, float] = {}
   522→   522→        for cs in confidence_scores:
   523→   523→            title = cs.get("title", "")
   524→   524→            conf_val = cs.get("confidence", 0.5)
   525→   525→            conf_map[title] = float(conf_val) if isinstance(conf_val, (int, float)) else 0.5
   526→   526→
   527→   527→        ranked: List[Dict[str, Any]] = []
   528→   528→        for f in findings:
   529→   529→            title = f.get("title", "")
   530→   530→            severity = str(f.get("severity", "info")).strip().lower()
   531→   531→            sev_weight = _SEVERITY_WEIGHT.get(severity, 0.0)
   532→   532→            confidence = conf_map.get(title, self._default_likelihood(severity))
   533→   533→            priority = sev_weight * confidence
   534→   534→
   535→   535→            ranked.append({
   536→   536→                "title": title,
   537→   537→                "severity": severity,
   538→   538→                "category": f.get("category", ""),
   539→   539→                "confidence": round(confidence, 3),
   540→   540→                "priority_score": round(priority, 3),
   541→   541→                "description": f.get("description", "")[:300],
   542→   542→                "remediation": f.get("remediation", ""),
   543→   543→                "evidence": str(f.get("evidence", ""))[:200],
   544→   544→            })
   545→   545→
   546→   546→        ranked.sort(key=lambda x: x["priority_score"], reverse=True)
   547→   547→        return ranked[:10]

   553→   553→    def _build_remediation_plan(
   554→   554→        self,
   555→   555→        findings: List[Dict[str, Any]],
   556→   556→        eng_score: Dict[str, Any],
   557→   557→    ) -> List[Dict[str, Any]]:
   558→   558→        """Build prioritized remediation plan grouped by category."""
   559→   559→        if not findings:
   560→   560→            return []
   561→   561→
   562→   562→        # Group findings by category.
   563→   563→        grouped: Dict[str, List[Dict[str, Any]]] = defaultdict(list)
   564→   564→        for f in findings:
   565→   565→            cat = str(f.get("category", "unknown")).strip().lower()
   566→   566→            grouped[cat].append(f)
   567→   567→
   568→   568→        # Try to use the recommendation engine for detailed steps.
   569→   569→        try:
   570→   570→            from .recommendation_engine import RecommendationEngine
   571→   571→            engine = RecommendationEngine()
   572→   572→            rec_report = engine.recommend(findings)
   573→   573→            rec_map: Dict[str, Dict[str, Any]] = {}
   574→   574→            for cs in rec_report.category_summary.values():
   575→   575→                rec_map[cs.get("category", "").lower()] = cs
   576→   576→        except Exception:
   577→   577→            rec_map = {}
   578→   578→
   579→   579→        # Severity ordering.
   580→   580→        sev_order = {"critical": 0, "high": 1, "medium": 2, "low": 3, "info": 4}
   581→   581→
   582→   582→        # Build plan items sorted by max severity within each category.
   583→   583→        plan: List[Dict[str, Any]] = []
   584→   584→        for cat, cat_findings in grouped.items():
   585→   585→            max_sev = max(
   586→   586→                cat_findings,
   587→   587→                key=lambda f: sev_order.get(str(f.get("severity", "info")).lower(), 4),
   588→   588→            )
   589→   589→            severity = str(max_sev.get("severity", "info")).strip().lower()
   590→   590→
   591→   591→            # Get remediation details from recommendation engine or template.
   592→   592→            rec_info = rec_map.get(cat, {})
   593→   593→            steps = rec_info.get("steps", [])
   594→   594→            summary = rec_info.get("summary", _REMEDIATION_TEMPLATES.get(
   595→   595→                cat, f"Review and remediate {cat} findings."
   596→   596→            ))
   597→   597→            effort = rec_info.get("effort", "medium")
   598→   598→
   599→   599→            if not steps:
   600→   600→                # Fallback: generate steps from template.
   601→   601→                template_text = _REMEDIATION_TEMPLATES.get(cat, "")
   602→   602→                if template_text:
   603→   603→                    steps = [template_text]
   604→   604→                else:
   605→   605→                    steps = [f"Investigate and resolve {cat} issues identified in the scan."]
   606→   606→
   607→   607→            plan.append({
   608→   608→                "category": cat,
   609→   609→                "category_label": _CATEGORY_LABELS.get(cat, cat),
   610→   610→                "severity": severity,
   611→   611→                "findings_count": len(cat_findings),
   612→   612→                "summary": summary,
   613→   613→                "steps": steps,
   614→   614→                "effort": effort,
   615→   615→                "finding_titles": [f.get("title", "") for f in cat_findings],
   616→   616→            })
   617→   617→
   618→   618→        # Sort: critical first, then high, medium, low, info.
   619→   619→        plan.sort(key=lambda x: sev_order.get(x["severity"], 4))
   620→   620→        return plan
