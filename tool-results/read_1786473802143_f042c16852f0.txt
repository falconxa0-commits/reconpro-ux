   140→class TocDocTemplate(SimpleDocTemplate):
   141→    def afterFlowable(self, flowable):
   142→        if hasattr(flowable, 'bookmark_name'):
   143→            level = getattr(flowable, 'bookmark_level', 0)
   144→            text = getattr(flowable, 'bookmark_text', '')
   145→            key = getattr(flowable, 'bookmark_key', '')
   146→            self.notify('TOCEntry', (level, text, self.page, key))

   148→def add_heading(text, style, level=0):
   149→    key = 'h_%s' % hashlib.md5(text.encode()).hexdigest()[:8]
   150→    p = Paragraph('<a name="%s"/>%s' % (key, text), style)
   151→    p.bookmark_name = text
   152→    p.bookmark_level = level
   153→    p.bookmark_text = text
   154→    p.bookmark_key = key
   155→    return p

   157→def add_major_section(text, style):
   158→    return [CondPageBreak(H1_ORPHAN_THRESHOLD), add_heading(text, style, level=0)]

   160→def make_table(data, col_ratios, has_header=True):
   161→    col_widths = [r * AVAILABLE_W for r in col_ratios]
   162→    t = Table(data, colWidths=col_widths, hAlign='CENTER')
   163→    style_cmds = [
   164→        ('VALIGN', (0, 0), (-1, -1), 'MIDDLE'),
   165→        ('LEFTPADDING', (0, 0), (-1, -1), 8),
   166→        ('RIGHTPADDING', (0, 0), (-1, -1), 8),
   167→        ('TOPPADDING', (0, 0), (-1, -1), 6),
   168→        ('BOTTOMPADDING', (0, 0), (-1, -1), 6),
   169→        ('GRID', (0, 0), (-1, -1), 0.5, BORDER),
   170→    ]
   171→    if has_header:
   172→        style_cmds.append(('BACKGROUND', (0, 0), (-1, 0), HEADER_FILL))
   173→        style_cmds.append(('TEXTCOLOR', (0, 0), (-1, 0), colors.white))
   174→        for i in range(1, len(data)):
   175→            bg = colors.white if i % 2 == 1 else TABLE_STRIPE
   176→            style_cmds.append(('BACKGROUND', (0, i), (-1, i), bg))
   177→    t.setStyle(TableStyle(style_cmds))
   178→    return t
