    13→
    14→function mulberry32(seed: number) {
    15→  return function () {
    16→    let t = (seed += 0x6d2b79f5);
    17→    t = Math.imul(t ^ (t >>> 15), t | 1);
    18→    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    19→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
    20→  };
    21→}

    23→function hashCode(s: string): number {
    24→  let h = 0;
    25→  for (let i = 0; i < s.length; i++) {
    26→    h = (Math.imul(31, h) + s.charCodeAt(i)) | 0;
    27→  }
    28→  return Math.abs(h);
    29→}

   445→function pickSeverity(rng: () => number): Severity {
   446→  const r = rng() * 100;
   447→  let cum = 0;
   448→  for (const [sev, weight] of SEVERITY_WEIGHTS) {
   449→    cum += weight;
   450→    if (r < cum) return sev;
   451→  }
   452→  return 'low';
   453→}
   454→
   455→function pickIndustry(rng: () => number): Industry {
   456→  return INDUSTRIES[Math.floor(rng() * INDUSTRIES.length)];
   457→}
   458→
   459→function pickFindingType(rng: () => number): FindingType {
   460→  return FINDING_TYPES[Math.floor(rng() * FINDING_TYPES.length)];
   461→}
   462→
   463→function pickRegion(rng: () => number): Region {
   464→  return REGIONS[Math.floor(rng() * REGIONS.length)];
   465→}
   466→
   467→function pickEntity(rng: () => number, industry: Industry): string {
   468→  const entities = ENTITY_TEMPLATES[industry];
   469→  return entities[Math.floor(rng() * entities.length)];
   470→}
   471→
   472→function pickDescription(rng: () => number, ft: FindingType, entity: string): string {
   473→  const templates = FINDING_TEMPLATES[ft];
   474→  const tpl = templates[Math.floor(rng() * templates.length)];
   475→  return tpl(entity);
   476→}
   477→
   478→function pickAssetType(rng: () => number, ft: FindingType): string {
   479→  const types = ASSET_TYPES[ft];
   480→  return types[Math.floor(rng() * types.length)];
   481→}

   483→function generateIncident(rng: () => number, timestamp: Date, id: string): Incident {
   484→  const industry = pickIndustry(rng);
   485→  const findingType = pickFindingType(rng);
   486→  const severity = pickSeverity(rng);
   487→  const entity = pickEntity(rng, industry);
   488→  const region = pickRegion(rng);
   489→
   490→  return {
   491→    id,
   492→    timestamp: timestamp.toISOString(),
   493→    severity,
   494→    industry,
   495→    findingType,
   496→    anonymizedDescription: pickDescription(rng, findingType, entity),
   497→    assetType: pickAssetType(rng, findingType),
   498→    region,
   499→    verifiable: rng() > 0.25,
   500→  };
   501→}

   503→/**
   504→ * Generate incidents for a given time window.
   505→ * Uses minute-level seed so it's deterministic per-minute (no flicker on refresh).
   506→ * Average rate: ~4.2 incidents/hour weighted by business hours.
   507→ */
   508→function generateIncidents(
   509→  since: Date,
   510→  until: Date,
   511→  seedPrefix: string = 'wos',
   512→): Incident[] {
   513→  const incidents: Incident[] = [];
   514→  const minute = 60 * 1000;
   515→  let cursor = new Date(since);
   516→  cursor.setSeconds(0, 0);
   517→  cursor.setMinutes(cursor.getMinutes() + 1);
   518→
   519→  while (cursor < until) {
   520→    const minuteKey = `wos:${Math.floor(cursor.getTime() / minute)}`;
   521→    const seed = hashCode(minuteKey);
   522→    const rng = mulberry32(seed);
   523→
   524→    const hour = cursor.getUTCHours();
   525→    const hourWeight = HOUR_WEIGHTS[hour];
   526→    const expectedIncidents = 0.07 * hourWeight;
   527→
   528→    if (rng() < expectedIncidents) {
   529→      const id = `inc-${Math.floor(cursor.getTime() / minute)}-${Math.floor(rng() * 1000)}`;
   530→      const jitteredTs = new Date(cursor.getTime() + Math.floor(rng() * 60000));
   531→      incidents.push(generateIncident(rng, jitteredTs, id));
   532→    }
   533→
   534→    cursor = new Date(cursor.getTime() + minute);
   535→  }
   536→
   537→  return incidents;
   538→}

   541→
   542→function computeStats(weekIncidents: Incident[]): Stats {
   543→  const now = new Date();
   544→  const todayStart = new Date(now.getUTCFullYear(), now.getUTCMonth(), now.getUTCDate());
   545→
   546→  const todayIncidents = weekIncidents.filter(i => new Date(i.timestamp) >= todayStart);
   547→
   548→  const byIndustry = {} as Record<Industry, number>;
   549→  const byType = {} as Record<FindingType, number>;
   550→
   551→  for (const ind of INDUSTRIES) byIndustry[ind] = 0;
   552→  for (const ft of FINDING_TYPES) byType[ft] = 0;
   553→
   554→  for (const inc of weekIncidents) {
   555→    byIndustry[inc.industry]++;
   556→    byType[inc.findingType]++;
   557→  }
   558→
   559→  return {
   560→    totalToday: todayIncidents.length,
   561→    totalWeek: weekIncidents.length,
   562→    criticalThisWeek: weekIncidents.filter(i => i.severity === 'critical').length,
   563→    byIndustry,
   564→    byType,
   565→  };
   566→}

   568→// ── Route Handler ──────────────────────────────────────────────────────
   569→
   570→export async function GET(request: NextRequest) {
   571→  const { allowed } = checkRateLimit(extractClientIP(request), 30, 60000);
   572→  if (!allowed) return NextResponse.json({ error: 'Rate limit exceeded' }, { status: 429 });
   573→
   574→  const { searchParams } = request.nextUrl;
   575→  const limitParam = searchParams.get('limit');
   576→  const severityFilter = searchParams.get('severity') as Severity | null;
   577→  const industryFilter = searchParams.get('industry') as Industry | null;
   578→
   579→  const limit = limitParam ? Math.min(Math.max(parseInt(limitParam, 10) || 50, 1), 200) : 50;
   580→
   581→  const now = new Date();
   582→  const weekAgo = new Date(now.getTime() - 7 * 24 * 60 * 60 * 1000);
   583→
   584→  const allIncidents = generateIncidents(weekAgo, now);
   585→
   586→  let filtered = allIncidents;
   587→  if (severityFilter && ['critical', 'high', 'medium', 'low'].includes(severityFilter)) {
   588→    filtered = filtered.filter(i => i.severity === severityFilter);
   589→  }
   590→  if (industryFilter && INDUSTRIES.includes(industryFilter)) {
   591→    filtered = filtered.filter(i => i.industry === industryFilter);
   592→  }
   593→
   594→  filtered.sort((a, b) => new Date(b.timestamp).getTime() - new Date(a.timestamp).getTime());
   595→
   596→  const incidents = filtered.slice(0, limit);
   597→  const stats = computeStats(allIncidents);
   598→
   599→  return NextResponse.json({ incidents, stats, simulated: true });
   600→}
