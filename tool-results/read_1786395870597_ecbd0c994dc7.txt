  1518→# ---------------------------------------------------------------------------
  1519→# TTPMatcher — matches observed TTPs against APT group databases
  1520→# ---------------------------------------------------------------------------
  1521→
  1522→class TTPMatcher:
  1523→    """Matches observed tactics, techniques, and procedures against the
  1524→    APT group database to identify potential threat actors.
  1525→
  1526→    The matcher uses a weighted scoring algorithm that considers:
  1527→      - Jaccard similarity of technique sets
  1528→      - Rarity weighting (less common techniques score higher)
  1529→      - Campaign overlap bonuses
  1530→      - Temporal activity signals
  1531→    """
  1532→
  1533→    def __init__(self, group_db: Optional[Dict[str, Any]] = None) -> None:
  1534→        self._groups = group_db if group_db is not None else APT_GROUPS
  1535→        self._technique_frequency: Dict[str, int] = {}
  1536→        self._compute_technique_frequency()
  1537→
  1538→    def _compute_technique_frequency(self) -> None:
  1539→        """Pre-compute how many groups use each technique for rarity weighting."""
  1540→        for group_data in self._groups.values():
  1541→            for tech in group_data.get("techniques", []):
  1542→                self._technique_frequency[tech] = (
  1543→                    self._technique_frequency.get(tech, 0) + 1
  1544→                )
  1545→
  1546→    def _technique_rarity(self, technique: str, total_groups: int) -> float:
  1547→        """Compute rarity score for a technique. Rarer = more discriminating.
  1548→
  1549→        Uses inverse frequency weighting: techniques used by fewer groups
  1550→        contribute more to attribution confidence.
  1551→        """
  1552→        freq = self._technique_frequency.get(technique, 0)
  1553→        if freq == 0:
  1554→            return 1.0
  1555→        # Inverse frequency: 1 for unique, approaching 0 for universal
  1556→        return math.log2(total_groups / (freq + 1)) / math.log2(total_groups)
  1557→
  1558→    def match_observed_ttps(
  1559→        self, observed_techniques: List[str],
  1560→    ) -> List[APTAttribution]:
  1561→        """Match observed TTPs against all APT groups and return a ranked list.
  1562→
  1563→        Args:
  1564→            observed_techniques: List of MITRE ATT&CK technique IDs observed
  1565→                                 in findings (e.g., ["T1566.001", "T1059.001"]).
  1566→
  1567→        Returns:
  1568→            Ranked list of :class:`APTAttribution` sorted by confidence
  1569→            (highest first).
  1570→        """
  1571→        total_groups = len(self._groups)
  1572→        results: List[APTAttribution] = []
  1573→
  1574→        if not observed_techniques:
  1575→            return results
  1576→
  1577→        observed_set = set(observed_techniques)
  1578→
  1579→        for group_id, group_data in self._groups.items():
  1580→            group_techs = set(group_data.get("techniques", []))
  1581→            overlap = observed_set & group_techs
  1582→
  1583→            if not overlap:
  1584→                continue
  1585→
  1586→            # --- Jaccard similarity ---
  1587→            jaccard = len(overlap) / len(observed_set | group_techs)
  1588→
  1589→            # --- Coverage: what fraction of observed techniques are explained ---
  1590→            coverage = len(overlap) / len(observed_set)
  1591→
  1592→            # --- Weighted rarity score ---
  1593→            rarity_sum = sum(
  1594→                self._technique_rarity(t, total_groups) for t in overlap
  1595→            )
  1596→            rarity_avg = rarity_sum / len(overlap)
  1597→
  1598→            # --- Composite confidence ---
  1599→            confidence = 0.35 * jaccard + 0.40 * coverage + 0.25 * rarity_avg
  1600→            confidence = min(confidence, 1.0)
  1601→
  1602→            reasoning: List[str] = []
  1603→            reasoning.append(
  1604→                f"Jaccard similarity: {jaccard:.4f} "
  1605→                f"({len(overlap)} shared techniques)"
  1606→            )
  1607→            reasoning.append(f"Coverage: {coverage:.2%} of observed TTPs")
  1608→            reasoning.append(f"Technique rarity factor: {rarity_avg:.4f}")
  1609→
  1610→            # Campaign overlap bonus
  1611→            matching_campaigns = self._find_matching_campaigns(
  1612→                group_id, overlap
  1613→            )
  1614→            if matching_campaigns:
  1615→                bonus = min(len(matching_campaigns) * 0.05, 0.15)
  1616→                confidence = min(confidence + bonus, 1.0)
  1617→                reasoning.append(
  1618→                    f"Campaign overlap bonus (+{bonus:.2f}): "
  1619→                    f"{', '.join(matching_campaigns)}"
  1620→                )
  1621→
  1622→            results.append(
  1623→                APTAttribution(
  1624→                    group_id=group_id,
  1625→                    group_name=group_data.get("name", group_id),
  1626→                    confidence=confidence,
  1627→                    confidence_level=self._confidence_level(confidence),
  1628→                    matching_techniques=sorted(overlap),
  1629→                    matching_campaigns=matching_campaigns,
  1630→                    country=group_data.get("country", "Unknown"),
  1631→                    motivation=group_data.get("motivation", "Unknown"),
  1632→                    sophistication=group_data.get("sophistication", "Unknown"),
  1633→                    reasoning=reasoning,
  1634→                )
  1635→            )
  1636→
  1637→        results.sort(key=lambda a: a.confidence, reverse=True)
  1638→        return results
  1639→
  1640→    def calculate_confidence(
  1641→        self,
  1642→        group_profile: Dict[str, Any],
  1643→        observed_findings: List[Dict[str, Any]],
  1644→    ) -> float:
  1645→        """Calculate a confidence score (0.0–1.0) that the observed findings
  1646→        align with the given APT group profile.
  1647→
  1648→        Args:
  1649→            group_profile: APT group dict from the database.
  1650→            observed_findings: List of finding dicts with at least ``"type"``
  1651→                               and ``"evidence"`` keys.
  1652→
  1653→        Returns:
  1654→            Float confidence between 0 and 1.
  1655→        """
  1656→        if not observed_findings or not group_profile:
  1657→            return 0.0
  1658→
  1659→        # Extract technique IDs from findings if present
  1660→        observed_techs: Set[str] = set()
  1661→        observed_sectors: Set[str] = set()
  1662→        for finding in observed_findings:
  1663→            tech = finding.get("technique_id") or finding.get("ttp") or ""
  1664→            if tech:
  1665→                observed_techs.add(tech)
  1666→            if "sector" in finding:
  1667→                observed_sectors.add(finding["sector"])
  1668→
  1669→        group_techs = set(group_profile.get("techniques", []))
  1670→        group_sectors = set(
  1671→            s.lower() for s in group_profile.get("sectors", [])
  1672→        )
  1673→
  1674→        tech_overlap = len(observed_techs & group_techs)
  1675→        total_relevant = len(observed_techs | group_techs)
  1676→
  1677→        if total_relevant == 0:
  1678→            return 0.0
  1679→
  1680→        jaccard = tech_overlap / total_relevant
  1681→
  1682→        # Sector alignment bonus
  1683→        sector_bonus = 0.0
  1684→        if observed_sectors:
  1685→            matching_sectors = len(
  1686→                observed_sectors & group_sectors
  1687→            )
  1688→            sector_bonus = 0.1 * (
  1689→                matching_sectors / max(len(observed_sectors), 1)
  1690→            )
  1691→
  1692→        # Sophistication factor
  1693→        soph = group_profile.get("sophistication", "moderate")
  1694→        soph_weight = {"minimal": 0.6, "moderate": 0.75,
  1695→                       "high": 0.9, "very_high": 1.0}
  1696→        soph_factor = soph_weight.get(soph, 0.75)
  1697→
  1698→        confidence = (0.8 * jaccard + sector_bonus) * soph_factor
  1699→        return round(min(max(confidence, 0.0), 1.0), 4)

  1724→    def _generate_patterns(self, target: str) -> List[str]:
  1725→        """Generate regex patterns for phishing and malware IOC detection."""
  1726→        domain = target.replace("https://", "").replace("http://", "").split("/")[0]
  1727→        patterns: List[str] = []
  1728→        patterns.append(
  1729→            rf"(?i)login.*{re.escape(domain)}"  # Phishing login page
  1730→        )
  1731→        patterns.append(
  1732→            rf"(?i)account.*verify.*{re.escape(domain)}"
  1733→        )
  1734→        patterns.append(
  1735→            rf"(?i)update.*{re.escape(domain)}.*\.exe"
  1736→        )
  1737→        patterns.append(
  1738→            rf"(?i){re.escape(domain)}.*\.php\?.*=.{8,}"
  1739→        )
  1740→        patterns.append(
  1741→            rf"(?i)base64.*[A-Za-z0-9+/]{{200,}}"  # Encoded payload
  1742→        )

  1745→    def _generate_watch_domains(self, target: str) -> List[str]:
  1746→        """Generate domain watchlist entries."""
  1747→        base = target.replace("https://", "").replace("http://", "").split("/")[0]
  1748→        parts = base.split(".")
  1749→        domains: List[str] = []
  1750→        if len(parts) >= 2:
  1751→            # Typosquatting variants
  1752→            domains.append(
  1753→                parts[0] + "-" + ".".join(parts[1:])
  1754→            )
  1755→            domains.append(
  1756→                parts[0] + parts[0] + "." + ".".join(parts[1:])
  1757→            )
  1758→            for swap in ("rn", "m", "vv", "l1", "0o"):
  1759→                domains.append(
  1760→                    parts[0].replace("n", swap[0]).replace("m", swap)
  1761→                    + "." + ".".join(parts[1:])
  1762→                )
  1763→        return list(set(domains))

  1806→# ---------------------------------------------------------------------------
  1807→# CampaignTracker — cross-reference findings with known campaigns
  1808→# ---------------------------------------------------------------------------
  1809→
  1810→class CampaignTracker:
  1811→    """Tracks and identifies campaign signatures by cross-referencing
  1812→    observed findings against the known campaigns database.
  1813→
  1814→    Campaign identification uses multi-factor analysis:
  1815→      - Technique overlap (MITRE ATT&CK)
  1816→      - Sector alignment
  1817→      - Temporal correlation
  1818→      - Known infrastructure reuse
  1819→    """
  1820→
  1821→    def __init__(
  1822→        self,
  1823→        campaigns: Optional[List[Dict[str, Any]]] = None,
  1824→        groups: Optional[Dict[str, Any]] = None,
  1825→    ) -> None:
  1826→        self._campaigns = campaigns if campaigns is not None else KNOWN_CAMPAIGNS
  1827→        self._groups = groups if groups is not None else APT_GROUPS
  1828→
  1829→    def identify_campaign(
  1830→        self,
  1831→        target: str,
  1832→        findings: List[Dict[str, Any]],
  1833→    ) -> List[CampaignMatch]:
  1834→        """Identify which known campaigns the target's findings align with.
  1835→
  1836→        Args:
  1837→            target: The target being analyzed.
  1838→            findings: List of finding dicts. Each should ideally contain
  1839→                      ``"technique_id"``, ``"sector"``, and ``"evidence"``.
  1840→
  1841→        Returns:
  1842→            Ranked list of :class:`CampaignMatch` objects.
  1843→        """
  1844→        observed_techs: Set[str] = set()
  1845→        observed_sectors: Set[str] = set()
  1846→        current_year = datetime.now().year
  1847→
  1848→        for finding in findings:
  1849→            tech = finding.get("technique_id") or finding.get("ttp") or ""
  1850→            if tech:
  1851→                observed_techs.add(tech)
  1852→            sector = finding.get("sector") or finding.get("industry") or ""
  1853→            if sector:
  1854→                observed_sectors.add(sector.lower())
  1855→
  1856→        if not observed_techs:
  1857→            return []
  1858→
  1859→        results: List[CampaignMatch] = []
  1860→
  1861→        for campaign in self._campaigns:
  1862→            camp_techs = set(campaign.get("techniques", []))
  1863→            tech_overlap = list(observed_techs & camp_techs)
  1864→
  1865→            if not tech_overlap:
  1866→                continue
  1867→
  1868→            # Technique overlap score
  1869→            tech_score = len(tech_overlap) / len(observed_techs | camp_techs)
  1870→
  1871→            # Sector overlap
  1872→            camp_sectors = set(
  1873→                s.lower() for s in campaign.get("sectors", [])
  1874→            )
  1875→            sector_overlap = list(observed_sectors & camp_sectors)
  1876→
  1877→            # Temporal factor: campaigns active in the last 5 years score higher
  1878→            year = int(campaign.get("year", 2010))
  1879→            age = current_year - year
  1880→            temporal = max(0.0, 1.0 - (age / 20.0))
  1881→
  1882→            # Group attribution overlap
  1883→            attributed = campaign.get("attributed_to", [])
  1884→
  1885→            # Composite score
  1886→            sector_bonus = 0.1 if sector_overlap else 0.0
  1887→            confidence = (0.5 * tech_score + 0.3 * temporal + sector_bonus)
  1888→            confidence = min(confidence, 1.0)
  1889→
  1890→            results.append(
  1891→                CampaignMatch(
  1892→                    campaign_id=campaign.get("id", ""),
  1893→                    campaign_name=campaign.get("name", ""),
  1894→                    attributed_groups=attributed,
  1895→                    confidence=confidence,
  1896→                    technique_overlap=tech_overlap,
  1897→                    sector_overlap=sector_overlap,
  1898→                    description=campaign.get("description", ""),
  1899→                )
  1900→            )
  1901→
  1902→        results.sort(key=lambda m: m.confidence, reverse=True)
  1903→        return results
  1904→
  1905→    def track_campaign_signature(
  1906→        self, findings: List[Dict[str, Any]],
  1907→    ) -> Dict[str, Any]:
  1908→        """Generate a unique campaign signature hash from findings.
  1909→
  1910→        This fingerprint can be used to cluster similar attack campaigns
  1911→        and detect repetitions across different targets.
  1912→
  1913→        Args:
  1914→            findings: List of finding dicts.
  1915→
  1916→        Returns:
  1917→            Dict with ``signature_hash``, ``technique_fingerprint``,
  1918→            ``sector_fingerprint``, and ``composite_hash``.
  1919→        """
  1920→        techs: List[str] = sorted(
  1921→            {f.get("technique_id", "") or f.get("ttp", "")
  1922→             for f in findings if f.get("technique_id") or f.get("ttp")}
  1923→        )
  1924→        sectors: List[str] = sorted(
  1925→            {f.get("sector", "") for f in findings if f.get("sector")}
  1926→        )
  1927→
  1928→        tech_str = "|".join(techs)
  1929→        sector_str = "|".join(sectors)
  1930→
  1931→        tech_hash = hashlib.sha256(tech_str.encode()).hexdigest()[:16]
  1932→        sector_hash = hashlib.sha256(sector_str.encode()).hexdigest()[:16]
  1933→        composite = hashlib.sha256(
  1934→            (tech_str + "||" + sector_str).encode()
  1935→        ).hexdigest()[:20]
  1936→
  1937→        return {
  1938→            "signature_hash": composite,
  1939→            "technique_fingerprint": tech_hash,
  1940→            "sector_fingerprint": sector_hash,
  1941→            "technique_count": len(techs),
  1942→            "sector_count": len(sectors),
  1943→            "techniques": techs,
  1944→            "sectors": sectors,
  1945→        }
  1946→
  1947→    def cross_reference_active_campaigns(
  1948→        self, target: str, findings: List[Dict[str, Any]],
  1949→    ) -> Dict[str, Any]:
  1950→        """Cross-reference findings against all active campaigns and
  1951→        return a combined analysis.
  1952→
  1953→        Returns:
  1954→            Dict with ``campaign_matches``, ``signature``, and ``summary``.
  1955→        """
  1956→        matches = self.identify_campaign(target, findings)
  1957→        signature = self.track_campaign_signature(findings)
  1958→
  1959→        active_campaigns = [m for m in matches if m.confidence >= 0.3]
  1960→
  1961→        return {
  1962→            "campaign_matches": [m.to_dict() for m in matches],
  1963→            "active_campaigns": [m.to_dict() for m in active_campaigns],
  1964→            "signature": signature,
  1965→            "summary": (
  1966→                f"{len(matches)} campaign(s) identified, "
  1967→                f"{len(active_campaigns)} with moderate or higher confidence. "
  1968→                f"Signature: {signature['signature_hash']}"
  1969→            ),
  1970→        }
