   440→        files = [FileFingerprint.from_dict(f) for f in data.get("files", [])]
   441→        return cls(
   442→            name=data.get("name", ""),
   443→            timestamp=data.get("timestamp", ""),
   444→            repo_path=data.get("repo_path", ""),
   445→            files=files,
   446→            total_files=int(data.get("total_files", 0)),
   447→            total_lines=int(data.get("total_lines", 0)),
   448→            total_size_bytes=int(data.get("total_size_bytes", 0)),
   449→            avg_complexity=float(data.get("avg_complexity", 0.0)),
   450→            max_complexity=float(data.get("max_complexity", 0.0)),
   451→            composite_hash=data.get("composite_hash", ""),
   452→        )
   453→
   454→
   455→@dataclass
   456→class DriftReport:
   457→    """Result of comparing current state against a baseline."""
   458→    baseline_name: str = ""
   459→    baseline_timestamp: str = ""
   460→    current_timestamp: str = ""
   461→    files_added: List[str] = field(default_factory=list)
   462→    files_removed: List[str] = field(default_factory=list)
   463→    files_modified: List[str] = field(default_factory=list)
   464→    drift_events: List[Dict[str, Any]] = field(default_factory=list)
   465→    drift_score: float = 0.0
   466→    regression_detected: bool = False
   467→    regression_details: List[str] = field(default_factory=list)
   468→
   469→    def to_dict(self) -> Dict[str, Any]:
   470→        return {
   471→            "baseline_name": self.baseline_name,
   472→            "baseline_timestamp": self.baseline_timestamp,
   473→            "current_timestamp": self.current_timestamp,
   474→            "files_added": self.files_added,
   475→            "files_removed": self.files_removed,
   476→            "files_modified": self.files_modified,
   477→            "total_changes": (
   478→                len(self.files_added) + len(self.files_removed)
   479→                + len(self.files_modified)
   480→            ),
   481→            "drift_events": self.drift_events,
   482→            "drift_score": round(self.drift_score, 2),
   483→            "regression_detected": self.regression_detected,
   484→            "regression_details": self.regression_details,
   485→        }

   487→
   488→class EngineeringBaseline:
   489→    """Manage engineering baselines for a repository.
   490→
   491→    Captures snapshots of repository state (file manifests with
   492→    hashes and complexity estimates), compares current state against
   493→    baselines, and tracks regression status.
   494→
   495→    Baselines are persisted to ``RECONPRO_HOME/baselines/<name>.json``.
   496→
   497→    Parameters
   498→    ----------
   499→    repo_path : str | Path
   500→        Root directory of the repository.
   501→    baseline_dir : Path | None
   502→        Directory for persisted baselines.
   503→    """
   504→
   505→    def __init__(
   506→        self,
   507→        repo_path: str | Path = ".",
   508→        baseline_dir: Optional[Path] = None,
   509→    ) -> None:
   510→        self._repo = Path(repo_path).resolve()
   511→        self._baseline_dir = baseline_dir or BASELINE_DIR
   512→        self._baseline_dir.mkdir(parents=True, exist_ok=True)
   513→
   514→    # -- Public API -------------------------------------------------------
   515→
   516→    def capture_snapshot(self, name: str = "") -> BaselineSnapshot:
   517→        """Capture a baseline snapshot of the current repository state.
   518→
   519→        Parameters
   520→        ----------
   521→        name : str
   522→            Baseline name. Defaults to ``"auto_<timestamp>"``.
   523→
   524→        Returns
   525→        -------
   526→        BaselineSnapshot
   527→            The captured baseline.
   528→        """
   529→        if not name:
   530→            name = f"auto_{datetime.now(timezone.utc).strftime('%Y%m%d_%H%M%S')}"
   531→
   532→        fingerprints: List[FileFingerprint] = []
   533→        total_lines = 0
   534→        total_size = 0
   535→        complexity_values: List[float] = []
   536→        hash_parts: List[str] = []
   537→
   538→        metrics = EngineeringMetrics(repo_path=self._repo)
   539→        source_files = metrics._walk_source_files()
   540→
   541→        for fpath in source_files:
   542→            rel_path = str(fpath.relative_to(self._repo))
   543→            try:
   544→                content = fpath.read_bytes()
   545→            except OSError:
   546→                continue
   547→
   548→            sha = hashlib.sha256(content).hexdigest()
   549→            text_content = content.decode("utf-8", errors="ignore")
   550→            lines = text_content.splitlines()
   551→            line_count = len(lines)
   552→
   553→            complexity = 0.0
   554→            if fpath.suffix == ".py":
   555→                complexity = EngineeringMetrics._estimate_complexity(lines)
   556→
   557→            fp = FileFingerprint(
   558→                path=rel_path,
   559→                size_bytes=len(content),
   560→                line_count=line_count,
   561→                sha256=sha,
   562→                complexity=complexity,
   563→            )
   564→            fingerprints.append(fp)
   565→            total_lines += line_count
   566→            total_size += len(content)
   567→            if complexity > 0:
   568→                complexity_values.append(complexity)
   569→            hash_parts.append(f"{rel_path}:{sha[:16]}")
   570→
   571→        avg_complexity = (
   572→            sum(complexity_values) / len(complexity_values)
   573→            if complexity_values else 0.0
   574→        )
   575→        max_complexity = max(complexity_values) if complexity_values else 0.0
   576→        composite_hash = hashlib.sha256(
   577→            "\n".join(sorted(hash_parts)).encode()
   578→        ).hexdigest()[:32]
   579→
   580→        snapshot = BaselineSnapshot(
   581→            name=name,
   582→            timestamp=datetime.now(timezone.utc).isoformat(),
   583→            repo_path=str(self._repo),
   584→            files=sorted(fingerprints, key=lambda f: f.path),
   585→            total_files=len(fingerprints),
   586→            total_lines=total_lines,
   587→            total_size_bytes=total_size,
   588→            avg_complexity=avg_complexity,
   589→            max_complexity=max_complexity,
   590→            composite_hash=composite_hash,
   591→        )
   592→
   593→        self._save_baseline(snapshot)
   594→        return snapshot
   595→
   596→    def compare(
   597→        self,
   598→        baseline_name: Optional[str] = None,
   599→        baseline: Optional[BaselineSnapshot] = None,
   600→    ) -> DriftReport:
   601→        """Compare current repository state against a baseline.
   602→
   603→        Parameters
   604→        ----------
   605→        baseline_name : str | None
   606→            Name of a previously saved baseline.
   607→        baseline : BaselineSnapshot | None
   608→            An in-memory baseline object. Takes precedence over
   609→            *baseline_name*.
   610→
   611→        Returns
   612→        -------
   613→        DriftReport
   614→            Detailed comparison results.
   615→        """
   616→        if baseline is None:
   617→            if baseline_name is None:
   618→                # Use the most recent baseline
   619→                baseline = self._load_latest_baseline()
   620→            else:
   621→                baseline = self.load_baseline(baseline_name)
   622→
   623→        if baseline is None:
   624→            return DriftReport(
   625→                baseline_name=baseline_name or "",
   626→                current_timestamp=datetime.now(timezone.utc).isoformat(),
   627→                regression_details=["No baseline found for comparison"],
   628→            )
   629→
   630→        # Build current state index
   631→        current_fps: Dict[str, FileFingerprint] = {}
   632→        metrics = EngineeringMetrics(repo_path=self._repo)
   633→        for fpath in metrics._walk_source_files():
   634→            rel_path = str(fpath.relative_to(self._repo))
   635→            try:
   636→                content = fpath.read_bytes()
   637→            except OSError:
   638→                continue
   639→            text_content = content.decode("utf-8", errors="ignore")
   640→            lines = text_content.splitlines()
   641→            complexity = 0.0
   642→            if fpath.suffix == ".py":
   643→                complexity = EngineeringMetrics._estimate_complexity(lines)
   644→            current_fps[rel_path] = FileFingerprint(
   645→                path=rel_path,
   646→                size_bytes=len(content),
   647→                line_count=len(lines),
   648→                sha256=hashlib.sha256(content).hexdigest(),
   649→                complexity=complexity,
   650→            )
   651→
   652→        # Build baseline index
   653→        baseline_map: Dict[str, FileFingerprint] = {
   654→            f.path: f for f in baseline.files
   655→        }
   656→
   657→        current_paths = set(current_fps.keys())
   658→        baseline_paths = set(baseline_map.keys())
   659→
   660→        added = sorted(current_paths - baseline_paths)
   661→        removed = sorted(baseline_paths - current_paths)
   662→        common = current_paths & baseline_paths
   663→
   664→        modified: List[str] = []
   665→        drift_events: List[Dict[str, Any]] = []
   666→        regression_details: List[str] = []
   667→
   668→        for path in sorted(common):
   669→            cur = current_fps[path]
   670→            base = baseline_map[path]
   671→
   672→            if cur.sha256 != base.sha256:
   673→                modified.append(path)
   674→                event: Dict[str, Any] = {
   675→                    "path": path,
   676→                    "type": "CONTENT_CHANGE",
   677→                    "severity": "low",
   678→                }
   679→
   680→                # Detect regression: significant complexity increase
   681→                if cur.complexity > 0 and base.complexity > 0:
   682→                    cx_delta = cur.complexity - base.complexity
   683→                    if cx_delta > 10:
   684→                        event["severity"] = "high"
   685→                        event["detail"] = (
   686→                            f"Complexity surged by {cx_delta:.1f} "
   687→                            f"(from {base.complexity:.1f} to {cur.complexity:.1f})"
   688→                        )
   689→                        regression_details.append(event["detail"])
   690→                    elif cx_delta > 5:
   691→                        event["severity"] = "medium"
   692→                        event["detail"] = (
   693→                            f"Complexity increased by {cx_delta:.1f} "
   694→                            f"(from {base.complexity:.1f} to {cur.complexity:.1f})"
   695→                        )
   696→
   697→                # Detect regression: significant line count increase
   698→                line_delta = cur.line_count - base.line_count
   699→                if line_delta > 100:
   700→                    event["severity"] = max(
   701→                        event.get("severity", "low"), "medium"
   702→                    )
   703→                    if "detail" not in event:
   704→                        event["detail"] = (
   705→                            f"File grew by {line_delta} lines"
   706→                        )
   707→
   708→                drift_events.append(event)
   709→
   710→        # New files add complexity
   711→        for path in added:
   712→            cur = current_fps[path]
   713→            if cur.complexity > 15:
   714→                regression_details.append(
   715→                    f"New file {path} has high complexity ({cur.complexity:.1f})"
   716→                )
   717→
   718→        # Removed files might be regression (deleted tests, docs)
   719→        for path in removed:
   720→            base = baseline_map[path]
   721→            if "test" in path.lower():
   722→                regression_details.append(
   723→                    f"Test file removed: {path}"
   724→                )
   725→
   726→        # Calculate drift score
   727→        total_changes = len(added) + len(removed) + len(modified)
   728→        drift_score = self._calculate_drift_score(
   729→            added=added,
   730→            removed=removed,
   731→            modified=modified,
   732→            drift_events=drift_events,
   733→            regression_details=regression_details,
   734→        )
   735→
   736→        return DriftReport(
   737→            baseline_name=baseline.name,
   738→            baseline_timestamp=baseline.timestamp,
   739→            current_timestamp=datetime.now(timezone.utc).isoformat(),
   740→            files_added=added,
   741→            files_removed=removed,
   742→            files_modified=modified,
   743→            drift_events=drift_events,
   744→            drift_score=drift_score,
   745→            regression_detected=len(regression_details) > 0,
   746→            regression_details=regression_details,
   747→        )
   748→
   749→    def load_baseline(self, name: str) -> Optional[BaselineSnapshot]:
   750→        """Load a named baseline from disk."""
   751→        path = self._baseline_dir / f"{name}.json"
   752→        if not path.exists():
   753→            return None
   754→        try:
   755→            with open(path, "r", encoding="utf-8") as f:
   756→                data = json.load(f)
   757→            return BaselineSnapshot.from_dict(data)
   758→        except (json.JSONDecodeError, OSError, KeyError, TypeError):
   759→            logger.debug("Could not load baseline %s", name, exc_info=True)
   760→            return None
   761→
   762→    def list_baselines(self) -> List[Dict[str, str]]:
   763→        """List all saved baselines with name and timestamp."""
   764→        results: List[Dict[str, str]] = []
   765→        if not self._baseline_dir.exists():
   766→            return results
   767→        for fpath in sorted(self._baseline_dir.glob("*.json")):
   768→            baseline = self.load_baseline(fpath.stem)
   769→            if baseline:
   770→                results.append({
   771→                    "name": baseline.name,
   772→                    "timestamp": baseline.timestamp,
   773→                    "total_files": str(baseline.total_files),
   774→                    "total_lines": str(baseline.total_lines),
   775→                })
   776→        return results
   777→
   778→    def delete_baseline(self, name: str) -> bool:
   779→        """Delete a saved baseline. Returns True on success."""
   780→        path = self._baseline_dir / f"{name}.json"
   781→        if not path.exists():
   782→            return False
   783→        try:
   784→            path.unlink()
   785→            return True
   786→        except OSError:
   787→            return False
   788→
   789→    # -- Internal ---------------------------------------------------------
   790→
   791→    def _save_baseline(self, snapshot: BaselineSnapshot) -> None:
   792→        """Persist a baseline to disk."""
   793→        path = self._baseline_dir / f"{snapshot.name}.json"
   794→        try:
   795→            with open(path, "w", encoding="utf-8") as f:
   796→                json.dump(snapshot.to_dict(), f, indent=2, ensure_ascii=False)
   797→        except OSError:
   798→            logger.debug("Could not save baseline %s", snapshot.name, exc_info=True)
   799→
   800→    def _load_latest_baseline(self) -> Optional[BaselineSnapshot]:
   801→        """Load the most recently created baseline."""
   802→        baselines = self.list_baselines()
   803→        if not baselines:
   804→            return None
   805→        # list_baselines returns sorted by filename; get the last one
   806→        latest = baselines[-1]
   807→        return self.load_baseline(latest["name"])
   808→
   809→    @staticmethod
   810→    def _calculate_drift_score(
   811→        added: List[str],
   812→        removed: List[str],
   813→        modified: List[str],
   814→        drift_events: List[Dict[str, Any]],
   815→        regression_details: List[str],
   816→    ) -> float:
   817→        """Calculate a 0-100 drift score.
   818→
   819→        Higher score = more drift. Regression indicators boost the score.
   820→        """
   821→        if not added and not removed and not modified:
   822→            return 0.0
   823→
   824→        # Base score from change volume
   825→        total_changes = len(added) + len(removed) + len(modified)
   826→        volume_score = min(total_changes * 2.0, 40.0)
   827→
   828→        # Severity weighting
   829→        severity_weights = {"low": 1.0, "medium": 2.0, "high": 3.0, "critical": 5.0}
   830→        event_score = 0.0
   831→        for event in drift_events:
   832→            sev = event.get("severity", "low")
   833→            event_score += severity_weights.get(sev, 1.0)
   834→        event_score = min(event_score, 30.0)
   835→
   836→        # Regression penalty
   837→        regression_penalty = min(len(regression_details) * 5.0, 30.0)
   838→
   839→        return min(volume_score + event_score + regression_penalty, 100.0)

   870→class EngineeringPipeline:
   871→    """Orchestrate automated engineering workflows.
   872→
   873→    Combines health checks, drift detection, quality gates, and
   874→    metrics collection into a unified pipeline. Integrates with
   875→    existing ReconPro modules (diagnostics, drift_monitor, cross_validator,
   876→    observability) rather than duplicating their functionality.
   877→
   878→    Parameters
   879→    ----------
   880→    repo_path : str | Path
   881→        Repository root to analyze.
   882→    thresholds : dict[str, float] | None
   883→        Quality gate thresholds. Defaults to ``DEFAULT_QUALITY_THRESHOLDS``.
   884→    metrics_dir : Path | None
   885→        Override for metrics storage directory.
   886→    baseline_dir : Path | None
   887→        Override for baseline storage directory.
   888→    """
   889→
   890→    def __init__(
   891→        self,
   892→        repo_path: str | Path = ".",
   893→        thresholds: Optional[Dict[str, float]] = None,
   894→        metrics_dir: Optional[Path] = None,
   895→        baseline_dir: Optional[Path] = None,
   896→    ) -> None:
   897→        self._repo = Path(repo_path).resolve()
   898→        self._thresholds = thresholds or dict(DEFAULT_QUALITY_THRESHOLDS)
   899→        self._metrics = EngineeringMetrics(
   900→            repo_path=self._repo,
   901→            metrics_dir=metrics_dir,
   902→        )
   903→        self._baseline = EngineeringBaseline(
   904→            repo_path=self._repo,
   905→            baseline_dir=baseline_dir,
   906→        )
   907→
   908→    # -- Public API -------------------------------------------------------
   909→
   910→    def health_check(self) -> Dict[str, Any]:
   911→        """Run repository health diagnostics.
   912→
   913→        Delegates to ``diagnostics.health_check()``,
   914→        ``diagnostics.validate_config()``, and
   915→        ``diagnostics.run_diagnostics()`` for comprehensive results.
   916→
   917→        Returns
   918→        -------
   919→        dict
   920→            Structured health report with status per category,
   921→            config validation issues, and summary statistics.
   922→        """
   923→        result: Dict[str, Any] = {
   924→            "timestamp": datetime.now(timezone.utc).isoformat(),
   925→            "repo_path": str(self._repo),
   926→        }
   927→
   928→        # 1. Quick health check from diagnostics module
   929→        try:
   930→            from .diagnostics import health_check as diag_health
   931→            result["health"] = diag_health()
   932→        except Exception as exc:
   933→            logger.debug("Health check error: %s", exc, exc_info=True)
   934→            result["health"] = {"error": str(exc)}
   935→
   936→        # 2. Config validation from diagnostics module
   937→        try:
   938→            from .diagnostics import validate_config
   939→            result["config_issues"] = validate_config()
   940→        except Exception as exc:
   941→            logger.debug("Config validation error: %s", exc, exc_info=True)
   942→            result["config_issues"] = []
   943→
   944→        # 3. Full diagnostics from diagnostics module
   945→        try:
   946→            from .diagnostics import run_diagnostics
   947→            diag_result = run_diagnostics()
   948→            result["diagnostics"] = {
   949→                "total": diag_result.get("total_checks", 0),
   950→                "passed": diag_result.get("passed", 0),
   951→                "failed": diag_result.get("failed", 0),
   952→                "checks": diag_result.get("checks", []),
   953→            }
   954→            result["health_score"] = self._compute_health_score(diag_result)
   955→        except Exception as exc:
   956→            logger.debug("Full diagnostics error: %s", exc, exc_info=True)
   957→            result["diagnostics"] = {"error": str(exc)}
   958→            result["health_score"] = 0.0
   959→
   960→        # 4. Repository-specific checks
   961→        repo_health = self._check_repository_health()
   962→        result["repository"] = repo_health
   963→
   964→        return result
   965→
   966→    def drift_detection(
   967→        self,
   968→        baseline_name: Optional[str] = None,
   969→        create_baseline_if_missing: bool = True,
   970→    ) -> Dict[str, Any]:
   971→        """Detect code drift from baselines.
   972→
   973→        Compares the current repository state against the most recent
   974→        baseline (or a named one). If no baseline exists and
   975→        *create_baseline_if_missing* is True, captures one automatically.
   976→
   977→        Parameters
   978→        ----------
   979→        baseline_name : str | None
   980→            Specific baseline to compare against.
   981→        create_baseline_if_missing : bool
   982→            Auto-capture a baseline if none exists.
   983→
   984→        Returns
   985→        -------
   986→        dict
   987→            Drift detection results including drift score, changed files,
   988→            and regression details.
   989→        """
   990→        result: Dict[str, Any] = {
   991→            "timestamp": datetime.now(timezone.utc).isoformat(),
   992→            "repo_path": str(self._repo),
   993→        }
   994→
   995→        # Check if we have any baselines
   996→        existing = self._baseline.list_baselines()
   997→        if not existing and create_baseline_if_missing:
   998→            snapshot = self._baseline.capture_snapshot(
   999→                name=f"initial_{datetime.now(timezone.utc).strftime('%Y%m%d_%H%M%S')}"
  1000→            )
  1001→            result["auto_baseline_created"] = True
  1002→            result["baseline"] = snapshot.to_dict()
  1003→            result["drift"] = {
  1004→                "drift_score": 0.0,
  1005→                "message": "Initial baseline captured; no drift to report",
  1006→            }
  1007→            return result
  1008→
  1009→        # Run drift comparison
  1010→        drift_report = self._baseline.compare(baseline_name=baseline_name)
  1011→        result["drift"] = drift_report.to_dict()
  1012→
  1013→        # Also check infrastructure drift if drift_monitor is available
  1014→        # (only for remote targets, not local repos)
  1015→        try:
  1016→            from .drift_monitor import InfrastructureDriftMonitor
  1017→            result["infra_drift_available"] = True
  1018→        except ImportError:
  1019→            result["infra_drift_available"] = False
  1020→
  1021→        return result

  1023→    def quality_gate(
  1024→        self,
  1025→        metrics_snapshot: Optional[MetricsSnapshot] = None,
  1026→        drift_report: Optional[DriftReport] = None,
  1027→        health_result: Optional[Dict[str, Any]] = None,
  1028→    ) -> QualityGateResult:
  1029→        """Enforce quality thresholds before merge.
  1030→
  1031→        Evaluates metrics, drift, and health against configured
  1032→        thresholds. Returns a detailed pass/fail result.
  1033→
  1034→        Parameters
  1035→        ----------
  1036→        metrics_snapshot : MetricsSnapshot | None
  1037→            Current metrics. Collected automatically if None.
  1038→        drift_report : DriftReport | None
  1039→            Drift comparison result. Run automatically if None.
  1040→        health_result : dict | None
  1041→            Health check result. Run automatically if None.
  1042→
  1043→        Returns
  1044→        -------
  1045→        QualityGateResult
  1046→            Pass/fail result with per-threshold evaluations.
  1047→        """
  1048→        # Collect data if not provided
  1049→        if metrics_snapshot is None:
  1050→            metrics_snapshot = self._metrics.collect_current_metrics()
  1051→
  1052→        if drift_report is None:
  1053→            drift_report = self._baseline.compare()
  1054→
  1055→        if health_result is None:
  1056→            health_result = self.health_check()
  1057→
  1058→        evaluations: Dict[str, Dict[str, Any]] = {}
  1059→        failures: List[str] = []
  1060→        warnings: List[str] = []
  1061→        score_deductions = 0.0
  1062→
  1063→        # 1. Test pass rate gate
  1064→        test_rate = metrics_snapshot.test_pass_rate
  1065→        min_test_rate = self._thresholds.get("min_test_pass_rate", 80.0)
  1066→        test_pass = test_rate >= min_test_rate or test_rate == 0.0  # 0 = no tests run, not a failure
  1067→        evaluations["test_pass_rate"] = {
  1068→            "value": test_rate,
  1069→            "threshold": min_test_rate,
  1070→            "passed": test_pass,
  1071→        }
  1072→        if not test_pass and test_rate > 0:
  1073→            failures.append(
  1074→                f"Test pass rate {test_rate:.1f}% below threshold {min_test_rate:.1f}%"
  1075→            )
  1076→            score_deductions += 25.0
  1077→
  1078→        # 2. Debt ratio gate
  1079→        debt = metrics_snapshot.debt_ratio
  1080→        max_debt = self._thresholds.get("max_debt_ratio", 30.0)
  1081→        debt_pass = debt <= max_debt
  1082→        evaluations["debt_ratio"] = {
  1083→            "value": debt,
  1084→            "threshold": max_debt,
  1085→            "passed": debt_pass,
  1086→        }
  1087→        if not debt_pass:
  1088→            failures.append(
  1089→                f"Debt ratio {debt:.1f}% exceeds threshold {max_debt:.1f}%"
  1090→            )
  1091→            score_deductions += 20.0
  1092→        elif debt > max_debt * 0.7:
  1093→            warnings.append(
  1094→                f"Debt ratio {debt:.1f}% approaching threshold {max_debt:.1f}%"
  1095→            )
  1096→
  1097→        # 3. Complexity gate
  1098→        complexity = metrics_snapshot.complexity_max
  1099→        max_cx = self._thresholds.get("max_complexity", 20.0)
  1100→        cx_pass = complexity <= max_cx
  1101→        evaluations["complexity"] = {
  1102→            "value": complexity,
  1103→            "threshold": max_cx,
  1104→            "passed": cx_pass,
  1105→        }
  1106→        if not cx_pass:
  1107→            failures.append(
  1108→                f"Max complexity {complexity:.1f} exceeds threshold {max_cx:.1f}"
  1109→            )
  1110→            score_deductions += 15.0
  1111→
  1112→        # 4. Drift score gate
  1113→        drift_score = drift_report.drift_score
  1114→        max_drift = self._thresholds.get("max_drift_score", 50.0)
  1115→        drift_pass = drift_score <= max_drift
  1116→        evaluations["drift_score"] = {
  1117→            "value": drift_score,
  1118→            "threshold": max_drift,
  1119→            "passed": drift_pass,
  1120→        }
  1121→        if not drift_pass:
  1122→            failures.append(
  1123→                f"Drift score {drift_score:.1f} exceeds threshold {max_drift:.1f}"
  1124→            )
  1125→            score_deductions += 20.0
  1126→        elif drift_score > max_drift * 0.7:
  1127→            warnings.append(
  1128→                f"Drift score {drift_score:.1f} approaching threshold {max_drift:.1f}"
  1129→            )
  1130→
  1131→        # 5. Drift events volume gate
  1132→        total_drift_events = (
  1133→            len(drift_report.files_added)
  1134→            + len(drift_report.files_removed)
  1135→            + len(drift_report.files_modified)
  1136→        )
  1137→        max_events = self._thresholds.get("max_drift_events", 10)
  1138→        events_pass = total_drift_events <= max_events
  1139→        evaluations["drift_events"] = {
  1140→            "value": total_drift_events,
  1141→            "threshold": max_events,
  1142→            "passed": events_pass,
  1143→        }
  1144→        if not events_pass:
  1145→            failures.append(
  1146→                f"Drift events ({total_drift_events}) exceed threshold ({max_events})"
  1147→            )
  1148→            score_deductions += 10.0
  1149→
  1150→        # 6. Health check gate
  1151→        health_score = health_result.get("health_score", 0.0)
  1152→        min_health = self._thresholds.get("min_health_checks_passed", 0.7)
  1153→        health_pass = health_score >= min_health
  1154→        evaluations["health_checks"] = {
  1155→            "value": health_score,
  1156→            "threshold": min_health,
  1157→            "passed": health_pass,
  1158→        }
  1159→        if not health_pass:
  1160→            failures.append(
  1161→                f"Health score {health_score:.1f}% below threshold {min_health:.1f}%"
  1162→            )
  1163→            score_deductions += 10.0
  1164→
  1165→        # 7. Regression check
  1166→        if drift_report.regression_detected:
  1167→            failures.append(
  1168→                f"Regression detected: {len(drift_report.regression_details)} issue(s)"
  1169→            )
  1170→            score_deductions += 15.0
  1171→            for detail in drift_report.regression_details:
  1172→                warnings.append(f"Regression: {detail}")
  1173→
  1174→        # Calculate aggregate score
  1175→        score = max(0.0, 100.0 - score_deductions)
  1176→        passed = len(failures) == 0
  1177→
  1178→        return QualityGateResult(
  1179→            passed=passed,
  1180→            score=score,
  1181→            thresholds=dict(self._thresholds),
  1182→            evaluations=evaluations,
  1183→            failures=failures,
  1184→            warnings=warnings,
  1185→        )
  1186→
  1187→    def engineering_workflow(
  1188→        self,
  1189→        create_baseline_if_missing: bool = True,
  1190→        record_metrics: bool = True,
  1191→    ) -> Dict[str, Any]:
  1192→        """Run the full engineering pipeline.
  1193→
  1194→        Executes health check, drift detection, metrics collection,
  1195→        and quality gate evaluation in sequence.
  1196→
  1197→        Parameters
  1198→        ----------
  1199→        create_baseline_if_missing : bool
  1200→            Auto-capture a baseline if none exists during drift detection.
  1201→        record_metrics : bool
  1202→            Whether to persist metrics snapshot to history.
  1203→
  1204→        Returns
  1205→        -------
  1206→        dict
  1207→            Complete pipeline result with all sub-results and
  1208→            aggregate quality gate verdict.
  1209→        """
  1210→        pipeline_start = time.perf_counter()
  1211→        result: Dict[str, Any] = {
  1212→            "timestamp": datetime.now(timezone.utc).isoformat(),
  1213→            "repo_path": str(self._repo),
  1214→            "pipeline": "engineering_workflow",
  1215→        }
  1216→
  1217→        # Step 1: Health check
  1218→        t0 = time.perf_counter()
  1219→        health = self.health_check()
  1220→        result["health_check"] = {
  1221→            "duration_ms": round((time.perf_counter() - t0) * 1000, 2),
  1222→            "result": health,
  1223→        }
  1224→
  1225→        # Step 2: Drift detection
  1226→        t0 = time.perf_counter()
  1227→        drift = self.drift_detection(
  1228→            create_baseline_if_missing=create_baseline_if_missing,
  1229→        )
  1230→        result["drift_detection"] = {
  1231→            "duration_ms": round((time.perf_counter() - t0) * 1000, 2),
  1232→            "result": drift,
  1233→        }
  1234→
  1235→        # Step 3: Metrics collection
  1236→        t0 = time.perf_counter()
  1237→        metrics_snapshot = self._metrics.collect_current_metrics()
  1238→        if record_metrics:
  1239→            self._metrics.record_snapshot(metrics_snapshot)
  1240→        result["metrics"] = {
  1241→            "duration_ms": round((time.perf_counter() - t0) * 1000, 2),
  1242→            "snapshot": metrics_snapshot.to_dict(),
  1243→            "trends": {
  1244→                "coverage": self._metrics.get_coverage_trend(),
  1245→                "debt": self._metrics.get_debt_trend(),
  1246→                "complexity": self._metrics.get_complexity_trend(),
  1247→                "test_pass_rate": self._metrics.get_test_pass_rate_trend(),
  1248→            },
  1249→        }
  1250→
  1251→        # Step 4: Quality gate
  1252→        t0 = time.perf_counter()
  1253→        drift_report = DriftReport()
  1254→        drift_data = drift.get("drift", {})
  1255→        if isinstance(drift_data, dict):
  1256→            drift_report = DriftReport(
  1257→                baseline_name=drift_data.get("baseline_name", ""),
  1258→                baseline_timestamp=drift_data.get("baseline_timestamp", ""),
  1259→                current_timestamp=drift_data.get("current_timestamp", ""),
  1260→                files_added=drift_data.get("files_added", []),
  1261→                files_removed=drift_data.get("files_removed", []),
  1262→                files_modified=drift_data.get("files_modified", []),
  1263→                drift_events=drift_data.get("drift_events", []),
  1264→                drift_score=drift_data.get("drift_score", 0.0),
  1265→                regression_detected=drift_data.get("regression_detected", False),
  1266→                regression_details=drift_data.get("regression_details", []),
  1267→            )
  1268→
  1269→        gate = self.quality_gate(
  1270→            metrics_snapshot=metrics_snapshot,
  1271→            drift_report=drift_report,
  1272→            health_result=health,
  1273→        )
  1274→        result["quality_gate"] = {
  1275→            "duration_ms": round((time.perf_counter() - t0) * 1000, 2),
  1276→            "result": gate.to_dict(),
  1277→        }
  1278→
  1279→        # Aggregate summary
  1280→        total_ms = (time.perf_counter() - pipeline_start) * 1000.0
  1281→        result["summary"] = {
  1282→            "total_duration_ms": round(total_ms, 2),
  1283→            "quality_gate_passed": gate.passed,
  1284→            "quality_gate_score": gate.score,
  1285→            "drift_score": drift_report.drift_score,
  1286→            "health_score": health.get("health_score", 0.0),
  1287→            "regression_detected": drift_report.regression_detected,
  1288→            "files_analyzed": metrics_snapshot.file_count,
  1289→        }
  1290→
  1291→        # Emit observability event via plugin hook if available
  1292→        try:
  1293→            from .plugins import HookManager
  1294→            HookManager.fire(
  1295→                "engineering_workflow_complete",
  1296→                result=result,
  1297→            )
  1298→        except Exception:
  1299→            pass
  1300→
  1301→        return result

  1317→    @staticmethod
  1318→    def _compute_health_score(diag_result: Dict[str, Any]) -> float:
  1319→        """Compute a 0-100 health score from diagnostics results."""
  1320→        total = diag_result.get("total_checks", 1)

  1323→            return 100.0
  1324→        return round(passed / total * 100.0, 2)

  1326→    def _check_repository_health(self) -> Dict[str, Any]:
  1327→        """Run repository-specific health checks.
  1328→
  1329→        Checks:
  1330→        - Repository directory exists and is accessible
  1331→        - Source file count
  1332→        - Presence of test files
  1333→        - Presence of configuration files
  1334→        - Presence of documentation
  1335→        """
  1336→        checks: Dict[str, Any] = {}
  1337→
  1338→        # Directory exists
  1339→        checks["repo_exists"] = self._repo.is_dir()
  1340→
  1341→        # Count source files
  1342→        source_files = self._metrics._walk_source_files()
  1343→        checks["source_file_count"] = len(source_files)
  1344→
  1345→        # Test file detection
  1346→        test_files = [
  1347→            f for f in source_files
  1348→            if "test" in f.name.lower()
  1349→            or "tests" in f.parts
  1350→            or "test_" in f.name
  1351→            or f.name.startswith("test_")
  1352→            or f.name.endswith("_test.py")
  1353→        ]
  1354→        checks["test_file_count"] = len(test_files)
  1355→        checks["has_tests"] = len(test_files) > 0
  1356→
  1357→        # Configuration file detection
  1358→        config_patterns = [
  1359→            "setup.py", "setup.cfg", "pyproject.toml",
  1360→            "package.json", "Cargo.toml", "go.mod",
  1361→            "Makefile", "CMakeLists.txt", "build.gradle",
  1362→            ".github", ".gitlab-ci.yml", "Jenkinsfile",
  1363→        ]
  1364→        found_configs = []
  1365→        for pattern in config_patterns:
  1366→            if (self._repo / pattern).exists():
  1367→                found_configs.append(pattern)
  1368→        checks["config_files"] = found_configs
  1369→        checks["has_build_config"] = len(found_configs) > 0
  1370→
  1371→        # Documentation detection
  1372→        doc_patterns = ["README.md", "README.rst", "README.txt", "docs/"]
  1373→        found_docs = []
  1374→        for pattern in doc_patterns:
  1375→            if (self._repo / pattern).exists():
  1376→                found_docs.append(pattern)
  1377→        checks["documentation_files"] = found_docs
  1378→        checks["has_documentation"] = len(found_docs) > 0
  1379→
  1380→        # VCS detection
  1381→        checks["has_vcs"] = (self._repo / ".git").is_dir()
  1382→
  1383→        return checks
