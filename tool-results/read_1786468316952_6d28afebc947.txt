     1→"""Async scan orchestration engine for ReconPro v10.0.
     2→
     3→Runs modules concurrently with an event-driven architecture.  The public
     4→``scan()`` function is a drop-in replacement for ``scanner.scan()`` — it
     5→accepts the same arguments and returns an identical ``ReconProResult`` so
     6→that every existing consumer (CLI, TUI, reports, tests) works unchanged.
     7→
     8→Components
     9→──────────
    10→1. **ScanEvent** — lightweight dataclass emitted at every scan milestone.
    11→2. **ScanEngine** — async orchestrator that resolves modules, runs them
    12→   with bounded concurrency, and aggregates results.
    13→3. **EventCollector** — convenience helper that buffers events for later
    14→   inspection (useful in tests and TUI dashboards).
    15→4. **concurrent_scan()** — scan many targets in parallel.
    16→"""
    17→
    18→from __future__ import annotations
    19→
    20→import asyncio
    21→import json
    22→import logging
    23→import time
    24→from dataclasses import dataclass, field
    25→from typing import Any, Callable, Dict, List, Optional
    26→
    27→logger = logging.getLogger(__name__)
    28→
    29→from .registry import (
    30→    ALL_MODULES,
    31→    DEFAULT_LOCAL_MODULES,
    32→    DEFAULT_MODULES,
    33→    LOCAL_MODULES,
    34→    MODULE_REGISTRY,
    35→)
    36→from .scanner import ReconProResult
    37→from .http_layer import Finding
    38→from .utils import compute_grade, badge_markdown, compute_score

    40→
    41→# ── Scan Event ──────────────────────────────────────────────────────────
    42→
    43→
    44→@dataclass
    45→class ScanEvent:
    46→    """Event emitted during scan execution.
    47→
    48→    Attributes
    49→    ----------
    50→    type : str
    51→        One of ``SCAN_START``, ``MODULE_START``, ``FINDING``,
    52→        ``MODULE_COMPLETE``, ``SCAN_COMPLETE``.
    53→    module_id : str | None
    54→        The module that produced the event (``None`` for scan-level events).
    55→    target : str | None
    56→        The scan target (``None`` for module-level events).
    57→    finding : Finding | None
    58→        Present only for ``FINDING`` events.
    59→    findings_count : int
    60→        Present only for ``MODULE_COMPLETE`` events.
    61→    duration : float
    62→        Elapsed seconds for the module (``MODULE_COMPLETE``) or the
    63→        whole scan (``SCAN_COMPLETE``).
    64→    result : ReconProResult | None
    65→        Present only for ``SCAN_COMPLETE``.
    66→    modules : List[str]
    67→        Present only for ``SCAN_START`` — the resolved module list.
    68→    timestamp : float
    69→        ``time.monotonic()`` when the event was created.
    70→    """
    71→
    72→    type: str
    73→    module_id: Optional[str] = None
    74→    target: Optional[str] = None
    75→    finding: Optional[Finding] = None
    76→    findings_count: int = 0
    77→    duration: float = 0.0
    78→    result: Optional[ReconProResult] = None
    79→    modules: List[str] = field(default_factory=list)
    80→    timestamp: float = field(default_factory=time.monotonic)
    81→
    82→
    83→# ── Event Collector ─────────────────────────────────────────────────────
    84→
    85→
    86→class EventCollector:
    87→    """Buffer that stores every ``ScanEvent`` for later inspection.
    88→
    89→    Usage::
    90→
    91→        collector = EventCollector()
    92→        engine = ScanEngine(event_callback=collector)
    93→        await engine.run("example.com")
    94→        print(collector.timeline())
    95→        for f in collector.findings:
    96→            print(f.title)
    97→    """
    98→
    99→    def __init__(self) -> None:
   100→        self.events: List[ScanEvent] = []
   101→        self._start: float = 0.0
   102→
   103→    def __call__(self, event: ScanEvent) -> None:
   104→        """Append *event*; record scan start time for timeline."""
   105→        if event.type == "SCAN_START" and not self._start:
   106→            self._start = event.timestamp
   107→        self.events.append(event)
   108→
   109→    # -- derived views -----------------------------------------------------
   110→
   111→    @property
   112→    def findings(self) -> List[Finding]:
   113→        """All findings extracted from ``FINDING`` events, in order."""
   114→        return [e.finding for e in self.events if e.type == "FINDING" and e.finding is not None]
   115→
   116→    def by_module(self) -> Dict[str, List[ScanEvent]]:
   117→        """Events grouped by ``module_id``.
   118→
   119→        Scan-level events (``SCAN_START``, ``SCAN_COMPLETE``) are stored
   120→        under the key ``"_scan"``.
   121→        """
   122→        out: Dict[str, List[ScanEvent]] = {}
   123→        for e in self.events:
   124→            key = e.module_id if e.module_id else "_scan"
   125→            out.setdefault(key, []).append(e)
   126→        return out
   127→
   128→    def timeline(self) -> List[str]:
   129→        """Human-readable timeline of events relative to scan start.
   130→
   131→        Returns a list of formatted strings suitable for printing.
   132→        """
   133→        lines: List[str] = []
   134→        anchor = self._start or (self.events[0].timestamp if self.events else 0.0)
   135→        for e in self.events:
   136→            dt = e.timestamp - anchor
   137→            tag = f"[{dt:6.2f}s] {e.type}"
   138→            if e.module_id:
   139→                tag += f" ({e.module_id})"
   140→            if e.target and e.type in ("SCAN_START", "SCAN_COMPLETE"):
   141→                tag += f" target={e.target}"
   142→            if e.type == "FINDING" and e.finding:
   143→                tag += f" [{e.finding.severity}] {e.finding.title}"
   144→            if e.type == "MODULE_COMPLETE":
   145→                tag += f" {e.findings_count} findings in {e.duration:.2f}s"
   146→            if e.type == "SCAN_COMPLETE" and e.result:
   147→                tag += f" score={e.result.total_score} grade={e.result.grade}"
   148→            lines.append(tag)
   149→        return lines
   150→

   152→# ── Scan Engine ─────────────────────────────────────────────────────────
   153→
   154→
   155→class ScanEngine:
   156→    """Async scan orchestrator with bounded concurrency and event emission.
   157→
   158→    Parameters
   159→    ----------
   160→    event_callback : Callable[[ScanEvent], None] | None
   161→        If provided, called for every ``ScanEvent`` produced during the scan.
   162→    concurrency : int
   163→        Maximum number of modules that run simultaneously.
   164→    rate_limit : float
   165→        Default requests-per-second cap (overridden per-scan).
   166→    use_async : bool
   167→        When ``True`` the engine calls ``async_run()`` on modules that
   168→        expose it; otherwise every module is wrapped in
   169→        ``asyncio.to_thread``.
   170→    run_engineering : bool
   171→        When ``True``, the engineering pipeline runs automatically after
   172→        each scan completes.  Default ``False`` to preserve existing
   173→        behaviour.
   174→    engineering_repo_path : str
   175→        Repository path passed to the
   176→        ``ContinuousEngineeringOrchestrator`` when engineering is enabled.
   177→    """
   178→
   179→    def __init__(
   180→        self,
   181→        event_callback: Optional[Callable[[ScanEvent], None]] = None,
   182→        concurrency: int = 5,
   183→        rate_limit: float = 50.0,
   184→        use_async: bool = True,
   185→        run_engineering: bool = False,
   186→        engineering_repo_path: str = ".",
   187→        run_intelligence: bool = True,
   188→        run_quality: bool = False,
   189→        run_defense: bool = False,
   190→    ) -> None:
   191→        self._callback = event_callback
   192→        self._concurrency = concurrency
   193→        self._default_rate_limit = rate_limit
   194→        self._use_async = use_async
   195→        self._run_engineering = run_engineering
   196→        self._engineering_repo_path = engineering_repo_path
   197→        self._rate_limit = rate_limit
   198→        self._run_intelligence = run_intelligence
   199→        self._run_quality = run_quality
   200→        self._run_defense = run_defense
   201→
   202→    # -- internal helpers --------------------------------------------------
   203→
   204→    def _emit(self, event: ScanEvent) -> None:
   205→        """Fire the event callback (if any)."""
   206→        if self._callback is not None:
   207→            try:
   208→                self._callback(event)
   209→            except Exception:
   210→                logger.debug("Event callback error", exc_info=True)
   211→
   212→    @staticmethod
   213→    def _resolve_remote_modules(
   214→        modules: Optional[List[str]],
   215→        all_modules: bool,
   216→    ) -> List[str]:
   217→        """Determine which remote modules to run (mirrors scanner.py logic)."""
   218→        if all_modules:
   219→            return [m for m in ALL_MODULES if m in MODULE_REGISTRY]
   220→        if modules:
   221→            return [m.strip().lower() for m in modules if m.strip().lower() in MODULE_REGISTRY]
   222→        return list(DEFAULT_MODULES)
   223→
   224→    @staticmethod
   225→    def _resolve_local_modules(
   226→        modules: Optional[List[str]],
   227→        all_modules: bool,
   228→    ) -> List[str]:
   229→        """Determine which local modules to run (mirrors scanner.py logic)."""
   230→        if all_modules:
   231→            return list(DEFAULT_LOCAL_MODULES)
   232→        if modules:
   233→            return [m.strip().lower() for m in modules if m.strip().lower() in LOCAL_MODULES]
   234→        return list(DEFAULT_LOCAL_MODULES)
   235→
   236→    def _build_result(
   237→        self,
   238→        target: str,
   239→        mods: List[str],
   240→        all_findings: List[Finding],
   241→        module_results: Dict[str, Dict[str, Any]],
   242→        vibesec_score: Optional[int],
   243→        vibesec_grade: Optional[str],
   244→    ) -> ReconProResult:
   245→        """Aggregate findings into a ``ReconProResult`` identical to scanner.py."""
   246→        total_score = compute_score(all_findings)
   247→        grade = compute_grade(total_score)
   248→        host = target.replace("https://", "").replace("http://", "").split("/")[0]
   249→        badge = badge_markdown(host, grade)
   250→
   251→        sev_counts: Dict[str, int] = {}
   252→        for f in all_findings:
   253→            sev_counts[f.severity] = sev_counts.get(f.severity, 0) + 1
   254→
   255→        return ReconProResult(
   256→            target=host,
   257→            modules_run=mods,
   258→            findings=[f.to_dict() for f in all_findings],
   259→            severity_counts=sev_counts,
   260→            total_score=total_score,
   261→            grade=grade,
   262→            badge_markdown=badge,
   263→            vibesec_score=vibesec_score,
   264→            vibesec_grade=vibesec_grade,
   265→            module_results=module_results,
   266→        )
   267→
   268→    # -- single module execution -------------------------------------------
   269→
   270→    @staticmethod
   271→    def _is_plugin_module(mod_id: str, runner: Any, entry: Any) -> bool:
   272→        """Determine if a module is a third-party plugin (vs built-in)."""
   273→        if entry and isinstance(entry, dict) and entry.get("is_plugin"):
   274→            return True
   275→        if hasattr(runner, "__module__") and "plugins" in (runner.__module__ or ""):
   276→            return True
   277→        return False
   278→
   279→    async def _run_module(
   280→        self,
   281→        mod_id: str,
   282→        runner: Any,
   283→        target: str,
   284→        base_url: str,
   285→        timeout: int,
   286→        verify_tls: bool,
   287→        limiter: Any,
   288→        semaphore: asyncio.Semaphore,
   289→        findings_lock: asyncio.Lock,
   290→        all_findings: List[Finding],
   291→        module_results: Dict[str, Dict[str, Any]],
   292→        vibesec_state: Dict[str, Any],
   293→        _entry: Any = None,
   294→    ) -> None:
   295→        """Execute one module under the concurrency semaphore."""
   296→        # Security hardening: validate plugin modules through PluginSandbox
   297→        if self._is_plugin_module(mod_id, runner, _entry):
   298→            try:
   299→                from .security_hardening import PluginSandbox
   300→                sandbox = PluginSandbox(cpu_time_seconds=float(timeout))
   301→                sandbox_result = sandbox.execute(
   302→                    runner,
   303→                    plugin_name=mod_id,
   304→                    target=target,
   305→                    base_url=base_url,
   306→                    timeout=timeout,
   307→                    verify_tls=verify_tls,
   308→                    limiter=limiter,
   309→                )
   310→                if not sandbox_result.get("success"):
   311→                    _module_health.record_failure(mod_id)
   312→                    logger.error(
   313→                        "Plugin module '%s' blocked by sandbox: %s",
   314→                        mod_id, sandbox_result.get("error", "unknown"),
   315→                    )
   316→                    return
   317→            except Exception:
   318→                logger.debug("PluginSandbox check for '%s' failed, running unsandboxed", mod_id, exc_info=True)
   319→
   320→        async with semaphore:
   321→            self._emit(ScanEvent(type="MODULE_START", module_id=mod_id))
   322→            t0 = time.monotonic()
   323→            try:
   324→                if self._use_async and hasattr(runner, "async_run"):
   325→                    result = await runner.async_run(
   326→                        target=target,
   327→                        base_url=base_url,
   328→                        timeout=timeout,
   329→                        verify_tls=verify_tls,
   330→                        limiter=limiter,
   331→                    )
   332→                else:
   333→                    result = await asyncio.to_thread(
   334→                        runner,
   335→                        target=target,
   336→                        base_url=base_url,
   337→                        timeout=timeout,
   338→                        verify_tls=verify_tls,
   339→                        limiter=limiter,
   340→                    )
   341→
   342→                duration = time.monotonic() - t0
   343→
   344→                # Dispatch based on return type.
   345→                # vibesec returns (findings, score, grade, badge_md).
   346→                # All other modules return list[Finding].
   347→                if isinstance(result, tuple) and len(result) == 4:
   348→                    mod_findings, score, grade, badge_md = result
   349→                    vibesec_state["score"] = score
   350→                    vibesec_state["grade"] = grade
   351→                    async with findings_lock:
   352→                        all_findings.extend(mod_findings)
   353→                    module_results[mod_id] = {
   354→                        "findings": [f.to_dict() for f in mod_findings],
   355→                        "score": score,
   356→                        "grade": grade,
   357→                        "badge": badge_md,
   358→                    }
   359→                else:
   360→                    mod_findings = result
   361→                    async with findings_lock:
   362→                        all_findings.extend(mod_findings)
   363→                        for f in mod_findings:
   364→                            self._emit(
   365→                                ScanEvent(type="FINDING", module_id=mod_id, finding=f)
   366→                            )
   367→                            try:
   368→                                from .plugins import HookManager
   369→                                HookManager.fire("post_finding", finding=f)
   370→                            except Exception:
   371→                                logger.debug("Hook 'post_finding' error", exc_info=True)
   372→                    module_results[mod_id] = {
   373→                        "findings": [f.to_dict() for f in mod_findings],
   374→                        "count": len(mod_findings),
   375→                    }
   376→
   377→                _module_health.record_success(mod_id)
   378→
   379→                self._emit(
   380→                    ScanEvent(
   381→                        type="MODULE_COMPLETE",
   382→                        module_id=mod_id,
   383→                        findings_count=len(mod_findings),
   384→                        duration=duration,
   385→                    )
   386→                )
   387→            except Exception as exc:
   388→                duration = time.monotonic() - t0
   389→                _module_health.record_failure(mod_id)
   390→                logger.error("Module '%s' failed: %s", mod_id, exc, exc_info=True)
   391→                # Emit a finding-like error so the scan doesn't silently
   392→                # swallow module failures.
   393→                err_finding = Finding(
   394→                    title=f"Module {mod_id} error: {exc}",
   395→                    severity="info",
   396→                    category="engine_error",
   397→                    module=mod_id,
   398→                    description=f"Module {mod_id} raised an exception during execution.",
   399→                    evidence=str(exc),
   400→                    asset=target,
   401→                )
   402→                async with findings_lock:
   403→                    all_findings.append(err_finding)
   404→                module_results[mod_id] = {
   405→                    "findings": [err_finding.to_dict()],
   406→                    "count": 1,
   407→                    "error": str(exc),
   408→                }
   409→                self._emit(
   410→                    ScanEvent(
   411→                        type="MODULE_COMPLETE",
   412→                        module_id=mod_id,
   413→                        findings_count=1,
   414→                        duration=duration,
   415→                    )
   416→                )
   417→
   418→    # -- main async entry point --------------------------------------------
   419→
   420→    async def run(
   421→        self,
   422→        target: str,
   423→        modules: Optional[List[str]] = None,
   424→        all_modules: bool = False,
   425→        timeout: int = 10,
   426→        verify_tls: bool = True,
   427→        rate_limit: Optional[float] = None,
   428→        is_local: bool = False,
   429→    ) -> ReconProResult:
   430→        """Run a scan against *target* with concurrent module execution.
   431→
   432→        This is the async counterpart of :func:`scanner.scan` and produces
   433→        an identical ``ReconProResult``.  The only difference is that modules
   434→        are executed concurrently (up to ``self._concurrency`` at a time)
   435→        instead of sequentially.
   436→
   437→        Parameters
   438→        ----------
   439→        target : str
   440→            Domain, URL, or directory path (for local scans).
   441→        modules : list[str] | None
   442→            Explicit module IDs.  ``None`` → defaults.
   443→        all_modules : bool
   444→            Run every registered module for the scan type.
   445→        timeout : int
   446→            Per-request timeout in seconds.
   447→        verify_tls : bool
   448→            Whether to verify TLS certificates.
   449→        rate_limit : float | None
   450→            Requests-per-second cap.  Falls back to the engine default.
   451→        is_local : bool
   452→            If ``True``, use ``LOCAL_MODULES`` instead of
   453→            ``MODULE_REGISTRY`` (same as ``audit_scan``).
   454→
   455→        Returns
   456→        -------
   457→        ReconProResult
   458→            Identical shape to what ``scanner.scan`` returns.
   459→        """
   460→        effective_rate = rate_limit if rate_limit is not None else self._default_rate_limit
   461→
   462→        # Resolve modules list.
   463→        if is_local:
   464→            mods = self._resolve_local_modules(modules, all_modules)
   465→        else:
   466→            mods = self._resolve_remote_modules(modules, all_modules)
   467→
   468→        base_url = target if target.startswith("http") else f"https://{target}"
   469→        host = target.replace("https://", "").replace("http://", "").split("/")[0]
   470→
   471→        self._emit(ScanEvent(type="SCAN_START", target=target, modules=list(mods)))
   472→        try:
   473→            from .plugins import HookManager
   474→            HookManager.fire("pre_scan", target=target, modules=mods)
   475→        except Exception:
   476→            logger.debug("Hook 'pre_scan' error", exc_info=True)
   477→        scan_t0 = time.monotonic()
   478→
   479→        semaphore = asyncio.Semaphore(self._concurrency)
   480→        findings_lock = asyncio.Lock()
   481→        all_findings: List[Finding] = []
   482→        module_results: Dict[str, Dict[str, Any]] = {}
   483→        vibesec_state: Dict[str, Any] = {}
   484→
   485→        # Create rate limiter for modules that accept it.
   486→        from .http_layer import RateLimiter
   487→        limiter = RateLimiter(effective_rate)
   488→
   489→        # Build coroutine tasks for each module.
   490→        tasks: List[asyncio.Task[None]] = []
   491→        for mod_id in mods:
   492→            if is_local:
   493→                entry = LOCAL_MODULES.get(mod_id)
   494→            else:
   495→                entry = MODULE_REGISTRY.get(mod_id)
   496→
   497→            if entry is None:
   498→                continue
   499→
   500→            if not _module_health.is_healthy(mod_id):
   501→                logger.warning("Module '%s' is unhealthy (circuit breaker open), skipping", mod_id)
   502→                continue
   503→
   504→            runner = entry.get("runner")
   505→            if runner is None:
   506→                # Special case: vibesec — lazy import.
   507→                if mod_id == "vibesec" and not is_local:
   508→                    from .modules.vibesec import run_vibesec  # type: ignore[no-redef]
   509→                    runner = run_vibesec
   510→                else:
   511→                    continue
   512→
   513→            tasks.append(
   514→                asyncio.create_task(
   515→                    self._run_module(
   516→                        mod_id=mod_id,
   517→                        runner=runner,
   518→                        target=target,
   519→                        base_url=base_url,
   520→                        timeout=timeout,
   521→                        verify_tls=verify_tls,
   522→                        limiter=limiter,
   523→                        semaphore=semaphore,
   524→                        findings_lock=findings_lock,
   525→                        all_findings=all_findings,
   526→                        module_results=module_results,
   527→                        vibesec_state=vibesec_state,
   528→                        _entry=entry,
   529→                    )
   530→                )
   531→            )
   532→
   533→        # Run all module tasks concurrently (bounded by semaphore).
   534→        if tasks:
   535→            await asyncio.gather(*tasks, return_exceptions=True)
   536→
   537→        scan_duration = time.monotonic() - scan_t0
   538→
   539→        vibesec_score = vibesec_state.get("score")
   540→        vibesec_grade = vibesec_state.get("grade")
   541→
   542→        result = self._build_result(
   543→            target=host if not is_local else (target if target != "." else "local-audit"),
   544→            mods=mods,
   545→            all_findings=all_findings,
   546→            module_results=module_results,
   547→            vibesec_score=vibesec_score,
   548→            vibesec_grade=vibesec_grade,
   549→        )
   550→
   551→        # ── Post-scan Secret Detection & Tamper-Evident Logging ──
   552→        secret_findings: List[Dict[str, Any]] = []
   553→        if all_findings:
   554→            try:
   555→                from .security_hardening import SecretsManager, TamperEvidenceLogger
   556→                secrets_mgr = SecretsManager()
   557→                for f in all_findings:
   558→                    finding_text = json.dumps(f.to_dict(), default=str)
   559→                    sf = secrets_mgr.scan_for_secrets(finding_text, source=f"finding:{f.module}")
   560→                    secret_findings.extend(sf)
   561→                if secret_findings:
   562→                    logger.warning(
   563→                        "Secret detection found %d potential secret(s) in scan findings",
   564→                        len(secret_findings),
   565→                    )
   566→            except Exception as sec_err:
   567→                logger.debug("Post-scan secret detection error: %s", sec_err, exc_info=True)
   568→
   569→            # Tamper-evident logging of scan result
   570→            try:
   571→                from .security_hardening import TamperEvidenceLogger
   572→                tamper_logger = TamperEvidenceLogger(module_name="engine")
   573→                tamper_logger.log(
   574→                    level="INFO",
   575→                    event_type="SCAN_COMPLETE",
   576→                    target=target,
   577→                    details={
   578→                        "findings_count": len(all_findings),
   579→                        "score": result.total_score,
   580→                        "grade": result.grade,
   581→                        "modules_run": mods,
   582→                        "secrets_found": len(secret_findings),
   583→                    },
   584→                )
   585→            except Exception as te_err:
   586→                logger.debug("Tamper-evident logging failed: %s", te_err)
   587→
   588→        # ── Post-scan Intelligence Pipeline ──────────────────────
   589→        intelligence_data = None
   590→        if all_findings and self._run_intelligence:
   591→            try:
   592→                # Apply prompt defense to finding text before AI analyst processes it
   593→                # Always runs when intelligence pipeline is active
   594→                try:
   595→                    from .prompt_defense import PromptDefense
   596→                    _pd = PromptDefense(enable_logging=False)
   597→                    _sanitized_findings = []
   598→                    for _f in all_findings:
   599→                        _title = _f.title or ""
   600→                        _desc = _f.description or ""
   601→                        _evidence = _f.evidence or ""
   602→                        _pd_result = _pd.scan(f"{_title} {_desc} {_evidence}", source="engine-intel")
   603→                        if _pd_result.is_safe:
   604→                            _sanitized_findings.append(_f)
   605→                        else:
   606→                            # Replace description with sanitized version to prevent prompt injection
   607→                            logger.warning(
   608→                                "Prompt defense sanitized finding '%s': %s",
   609→                                _f.title, _pd_result.max_severity.value,
   610→                            )
   611→                            _safe_f = Finding(
   612→                                title=_f.title,
   613→                                severity=_f.severity,
   614→                                category=_f.category,
   615→                                module=_f.module,
   616→                                description=_pd_result.sanitized_input,
   617→                                evidence=_f.evidence,
   618→                                asset=_f.asset,
   619→                                points_deducted=_f.points_deducted,
   620→                                remediation=_f.remediation,
   621→                                references=_f.references,
   622→                            )
   623→                            _sanitized_findings.append(_safe_f)
   624→                except Exception:
   625→                    logger.debug("Prompt defense check skipped", exc_info=True)
   626→                    _sanitized_findings = all_findings
   627→
   628→                from .intelligence_pipeline import IntelligencePipeline
   629→                intel_pipeline = IntelligencePipeline(
   630→                    enable_ai_analyst=True,
   631→                    enable_attack_graph=True,
   632→                    enable_threat_intel=True,
   633→                )
   634→                intel_result = intel_pipeline.analyze(_sanitized_findings, result.to_dict())
   635→                intelligence_data = intel_result.to_dict()
   636→
   637→                # Scan AI-generated intelligence data for secrets
   638→                try:
   639→                    _intel_text = json.dumps(intelligence_data, default=str)
   640→                    from .security_hardening import SecretsManager as _SM2
   641→                    _intel_sm = _SM2()
   642→                    _intel_sf = _intel_sm.scan_for_secrets(_intel_text, source="intelligence_pipeline")
   643→                    if _intel_sf:
   644→                        logger.warning(
   645→                            "Secret detection found %d potential secret(s) in AI-generated content",
   646→                            len(_intel_sf),
   647→                        )
   648→                        secret_findings.extend(_intel_sf)
   649→                except Exception:
   650→                    pass
   651→
   652→                # Inject intelligence into result
   653→                result.intelligence = intelligence_data
   654→                logger.info(
   655→                    "Intelligence pipeline completed: risk=%.1f exposure=%.1f chains=%d cves=%d",
   656→                    intel_result.executive_risk_score,
   657→                    intel_result.exposure_score,
   658→                    len(intel_result.attack_chains),
   659→                    len(intel_result.cve_matches),
   660→                )
   661→            except Exception as e:
   662→                logger.warning("Intelligence pipeline error: %s", e, exc_info=True)
   663→
   664→        self._emit(
   665→            ScanEvent(
   666→                type="SCAN_COMPLETE",
   667→                target=target,
   668→                result=result,
   669→                duration=scan_duration,
   670→            )
   671→        )
   672→        try:
   673→            from .plugins import HookManager
   674→            HookManager.fire("post_scan", target=target, result=result, duration=scan_duration)
   675→        except Exception:
   676→            logger.debug("Hook 'post_scan' error", exc_info=True)
   677→
   678→        # ── Post-scan Engineering Pipeline (if enabled) ──────────────
   679→        if self._run_engineering and all_findings:
   680→            try:
   681→                from .engineering_workflow import ContinuousEngineeringOrchestrator
   682→                eng_orchestrator = ContinuousEngineeringOrchestrator(self._engineering_repo_path)
   683→                eng_result = eng_orchestrator.run_full_cycle(self._engineering_repo_path)
   684→                # Store engineering result metadata in result for later retrieval
   685→                result.engineering = eng_result.to_dict()
   686→                logger.info(
   687→                    "Engineering pipeline completed: status=%s stages=%d duration=%.2fs",
   688→                    eng_result.overall_status,
   689→                    len(eng_result.stages),
   690→                    eng_result.total_duration_s,
   691→                )
   692→            except Exception as e:
   693→                logger.warning("Engineering pipeline error: %s", e, exc_info=True)
   694→
   695→        # ── Post-engineering Quality Intelligence ──
   696→        if self._run_quality or self._run_engineering:
   697→            try:
   698→                from .quality_intelligence import QualityIntelligence
   699→                qi = QualityIntelligence(repository_path=self._engineering_repo_path)
   700→                qi.analyze_repository_quality(self._engineering_repo_path)
   701→                quality_score = qi.get_quality_score()
   702→                quality_dims = qi.get_quality_dimensions()
   703→                result.quality = {
   704→                    "composite_score": quality_score,
   705→                    "dimensions": quality_dims,
   706→                }
   707→                logger.info(
   708→                    "Quality intelligence completed: score=%.1f", quality_score,
   709→                )
   710→            except Exception as e:
   711→                logger.debug("Quality intelligence skipped: %s", e)
   712→
   713→        return result

   715→    # -- sync wrapper ------------------------------------------------------
   716→
   717→    def scan_one(
   718→        self,
   719→        target: str,
   720→        modules: Optional[List[str]] = None,
   721→        timeout: int = 10,
   722→        verify_tls: bool = True,
   723→        rate_limit: Optional[float] = None,
   724→        is_local: bool = False,
   725→        run_engineering: bool = False,
   726→        engineering_repo_path: str = ".",
   727→        run_intelligence: bool = True,
   728→        run_quality: bool = False,
   729→        run_defense: bool = False,
   730→    ) -> ReconProResult:
   731→        """Synchronous wrapper around :meth:`run`.
   732→
   733→        Creates a fresh event loop (or reuses the running one) so callers
   734→        don't need to care about asyncio.
   735→
   736→        Parameters are identical to :meth:`run` except ``all_modules`` is
   737→        not exposed — pass an explicit ``modules`` list instead.
   738→        """
   739→        # Override instance-level settings if caller provides them.
   740→        if run_engineering:
   741→            self._run_engineering = True
   742→        if engineering_repo_path != ".":
   743→            self._engineering_repo_path = engineering_repo_path
   744→        if not run_intelligence:
   745→            self._run_intelligence = False
   746→        if run_quality:
   747→            self._run_quality = True
   748→        if run_defense:
   749→            self._run_defense = True
   750→        try:
   751→            loop = asyncio.get_running_loop()
   752→        except RuntimeError:
   753→            loop = None
   754→
   755→        if loop is not None and loop.is_running():
   756→            # Already inside an async context — use nest_asyncio-safe
   757→            # approach or fallback to a new thread.
   758→            import concurrent.futures
   759→
   760→            with concurrent.futures.ThreadPoolExecutor(max_workers=1) as pool:
   761→                future = pool.submit(
   762→                    asyncio.run,
   763→                    self.run(
   764→                        target=target,
   765→                        modules=modules,
   766→                        timeout=timeout,
   767→                        verify_tls=verify_tls,
   768→                        rate_limit=rate_limit,
   769→                        is_local=is_local,
   770→                    ),
   771→                )
   772→                return future.result()
   773→
   774→        return asyncio.run(
   775→            self.run(
   776→                target=target,
   777→                modules=modules,
   778→                timeout=timeout,
   779→                verify_tls=verify_tls,
   780→                rate_limit=rate_limit,
   781→                is_local=is_local,
   782→            )
   783→        )

   786→# ── Module Health Tracking ────────────────────────────────────────────
   787→
   788→
   789→class ModuleHealthState:
   790→    """Tracks health of scanner modules for circuit-breaker logic."""
   791→
   792→    def __init__(self, failure_threshold: int = 3, cooldown_seconds: float = 60.0):
   793→        self._failure_counts: Dict[str, int] = {}
   794→        self._last_failure_time: Dict[str, float] = {}
   795→        self._failure_threshold = failure_threshold
   796→        self._cooldown_seconds = cooldown_seconds
   797→
   798→    def record_success(self, module_id: str) -> None:
   799→        self._failure_counts.pop(module_id, None)
   800→        self._last_failure_time.pop(module_id, None)
   801→
   802→    def record_failure(self, module_id: str) -> None:
   803→        self._failure_counts[module_id] = self._failure_counts.get(module_id, 0) + 1
   804→        self._last_failure_time[module_id] = time.monotonic()
   805→
   806→    def is_healthy(self, module_id: str) -> bool:
   807→        """Check if a module is healthy enough to run."""
   808→        count = self._failure_counts.get(module_id, 0)
   809→        if count < self._failure_threshold:
   810→            return True
   811→        # In cooldown?
   812→        last_fail = self._last_failure_time.get(module_id, 0)
   813→        if time.monotonic() - last_fail > self._cooldown_seconds:
   814→            # Cooldown expired, reset
   815→            self.record_success(module_id)
   816→            return True
   817→        return False
   818→
   819→    def get_status(self) -> Dict[str, Dict[str, Any]]:
   820→        """Return health status for all tracked modules."""
   821→        return {
   822→            mid: {
   823→                "failure_count": count,
   824→                "is_healthy": self.is_healthy(mid),
   825→                "in_cooldown": not self.is_healthy(mid),
   826→            }
   827→            for mid, count in self._failure_counts.items()
   828→        }
   829→
   830→
   831→# Global module health tracker
   832→_module_health = ModuleHealthState()

   957→async def concurrent_scan(
   958→    targets: List[str],
   959→    modules: Optional[List[str]] = None,
   960→    all_modules: bool = False,
   961→    timeout: int = 10,
   962→    verify_tls: bool = True,
   963→    rate_limit: float = 50.0,
   964→    max_concurrency: int = 10,
   965→    is_local: bool = False,
   966→    event_callback: Optional[Callable[[ScanEvent], None]] = None,
   967→) -> Dict[str, ReconProResult]:
   968→    """Scan multiple targets concurrently with bounded total parallelism.
   969→
   970→    Spawns one ``ScanEngine`` per target but shares a single
   971→    ``asyncio.Semaphore`` so the total number of in-flight modules
   972→    across *all* targets never exceeds ``max_concurrency``.
   973→
   974→    Parameters
   975→    ----------
   976→    targets : list[str]
   977→        Domains / URLs to scan.
   978→    modules, all_modules, timeout, verify_tls, rate_limit
   979→        Forwarded to each per-target scan.
   980→    max_concurrency : int
   981→        Upper bound on simultaneous module executions across all targets.
   982→    is_local : bool
   983→        If ``True``, run local audit modules instead of remote ones.
   984→    event_callback : callable | None
   985→        Receives ``ScanEvent`` from every target scan.
   986→
   987→    Returns
   988→    -------
   989→    dict[str, ReconProResult]
   990→        Mapping of target → result for every target.
   991→    """
   992→    results: Dict[str, ReconProResult] = {}
   993→    lock = asyncio.Lock()
   994→
   995→    async def _scan_one(t: str) -> None:
   996→        engine = ScanEngine(
   997→            event_callback=event_callback,
   998→            concurrency=max_concurrency,
   999→            rate_limit=rate_limit,
  1000→            use_async=True,
  1001→        )
  1002→        result = await engine.run(
  1003→            target=t,
  1004→            modules=modules,
  1005→            all_modules=all_modules,
  1006→            timeout=timeout,
  1007→            verify_tls=verify_tls,
  1008→            rate_limit=rate_limit,
  1009→            is_local=is_local,
  1010→        )
  1011→        async with lock:
  1012→            results[t] = result
  1013→
  1014→    tasks = [asyncio.create_task(_scan_one(t)) for t in targets]
  1015→    if tasks:
  1016→        await asyncio.gather(*tasks, return_exceptions=True)
  1017→
  1018→    return results
