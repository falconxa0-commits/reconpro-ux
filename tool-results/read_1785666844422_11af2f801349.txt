   985→    def _handle_scan(self, args: List[str]) -> None:
   986→        if not args:
   987→            self._chat(f"[{RED}]Usage: scan <target> [with <mod1> <mod2> ...][/]")
   988→            return
   989→
   990→        # Parse "with" clause
   991→        modules: Optional[List[str]] = None
   992→        target_parts: List[str] = []
   993→        found_with = False
   994→        mod_parts: List[str] = []
   995→        for a in args:
   996→            if a.lower() == "with":
   997→                found_with = True
   998→                continue
   999→            if found_with:
  1000→                mod_parts.append(a)
  1001→            else:
  1002→                target_parts.append(a)
  1003→
  1004→        target = " ".join(target_parts)
  1005→        if not target:
  1006→            self._chat(f"[{RED}]No target specified.[/]")
  1007→            return
  1008→
  1009→        if mod_parts:
  1010→            modules = [m.strip().lower() for m in mod_parts]
  1011→            self._chat(f"[{DIM_CYAN}]Scanning [cyan bold]{target}[/] with modules: {', '.join(modules)}[/]")
  1012→        else:
  1013→            self._chat(f"[{DIM_CYAN}]Scanning [cyan bold]{target}[/]...[/]")
  1014→
  1015→        self.current_target = target
  1016→        self._run_scan_worker(target, modules=modules)

  1097→    @work(exclusive=False, thread=True)
  1098→    def _run_scan_worker(
  1099→        self,
  1100→        target: str,
  1101→        modules: Optional[List[str]] = None,
  1102→        is_local: bool = False,
  1103→    ) -> None:
  1104→        """Run a scan in a background thread."""
  1105→        self.call_from_thread(self.is_scanning.set, True)
  1106→
  1107→        # Set modules to scanning
  1108→        if modules:
  1109→            for m in modules:
  1110→                if m in self._module_cells:
  1111→                    self.call_from_thread(self._module_cells[m].set_status, "scanning")
  1112→        else:
  1113→            all_mods = (
  1114→                list(MODULE_NAMES.keys())
  1115→                if is_local
  1116→                else [k for k in MODULE_NAMES if k not in ("host", "dev", "doctor")]
  1117→            )
  1118→            for m in all_mods:
  1119→                if m in self._module_cells:
  1120→                    self.call_from_thread(self._module_cells[m].set_status, "scanning")
  1121→
  1122→        try:
  1123→            from .scanner import scan, audit_scan
  1124→            from .history import save_scan
  1125→
  1126→            self.call_from_thread(
  1127→                self._chat, f"[{DIM_CYAN}]  ⏳ Scan in progress...[/]"
  1128→            )
  1129→
  1130→            if is_local:
  1131→                result = audit_scan(target=target, modules=modules)
  1132→            else:
  1133→                result = scan(target=target, modules=modules)
  1134→
  1135→            # Process findings
  1136→            data = result.to_dict()
  1137→            self.call_from_thread(self._on_scan_complete, data, modules, is_local)
  1138→
  1139→            # Save to history
  1140→            try:
  1141→                save_scan(data, label="nexus")
  1142→            except Exception:
  1143→                pass
  1144→
  1145→        except Exception as e:
  1146→            self.call_from_thread(self._chat, f"[{RED}]  ✗ Scan error: {e}[/]")
  1147→            if modules:
  1148→                for m in modules:
  1149→                    if m in self._module_cells:
  1150→                        self.call_from_thread(self._module_cells[m].set_status, "error")
  1151→            else:
  1152→                self.call_from_thread(self._set_all_modules_status, "error")
  1153→
  1154→        finally:
  1155→            self.call_from_thread(self.is_scanning.set, False)
  1156→
