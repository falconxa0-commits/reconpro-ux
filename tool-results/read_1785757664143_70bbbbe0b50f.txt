   563→   582→   871→  1220→        self,
   564→   583→   872→  1221→        target: str,

   575→   594→   883→  1232→                    self.call_from_thread(self._module_cells[m].set_status, "scanning")
   576→   595→   884→  1233→        else:

   578→   597→   886→  1235→                list(MODULE_NAMES.keys())
   579→   598→   887→  1236→                if is_local
   580→   599→   888→  1237→                else [k for k in MODULE_NAMES if k not in ("host", "dev", "doctor")]
   581→   600→   889→  1238→            )
   582→   601→   890→  1239→            for m in all_mods:
   583→   602→   891→  1240→                if m in self._module_cells:
   584→   603→   892→  1241→                    self.call_from_thread(self._module_cells[m].set_status, "scanning")
   585→   604→   893→  1242→
   586→   605→   894→  1243→        try:
   587→   606→   895→  1244→            from .scanner import scan, audit_scan
   588→   607→   896→  1245→            from .history import save_scan

   590→   609→   898→  1247→            self.call_from_thread(
   591→   610→   899→  1248→                self._chat, f"[{DIM_CYAN}]  ⏳ Scan in progress...[/]"

   594→   613→   902→  1251→            if is_local:

   609→   628→   917→  1266→        except Exception as e:

   612→   631→   920→  1269→                for m in modules:

   975→   994→  1283→  1632→        Uses the vertical click position to determine which finding index
