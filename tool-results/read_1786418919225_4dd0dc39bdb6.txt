   882→# ──────────────────────────────────────────────────────────────────────────────
   883→# TECH_DETECTOR — Fingerprinting engine
   884→# ──────────────────────────────────────────────────────────────────────────────
   885→
   886→class TechDetector:
   887→    """Analyzes HTTP responses, headers, HTML, and URLs to identify
   888→    the technology stack of a target application."""
   889→
   890→    def __init__(self) -> None:
   891→        self._reset()
   892→
   893→    def _reset(self) -> None:
   894→        self._frameworks: set[str] = set()
   895→        self._language: Optional[str] = None
   896→        self._server: Optional[str] = None
   897→        self._waf: Optional[str] = None
   898→        self._cms: Optional[str] = None
   899→        self._has_graphql: bool = False
   900→        self._has_rest_api: bool = False
   901→        self._confidence: float = 0.0
   902→        self._hits: int = 0
   903→
   904→    def analyze(
   905→        self,
   906→        target: str,
   907→        response_headers: dict[str, str],
   908→        html_body: str = "",
   909→        js_files: Optional[list[str]] = None,
   910→    ) -> TechProfile:
   911→        """Run full analysis and return a TechProfile."""
   912→        self._reset()
   913→        if js_files is None:
   914→            js_files = []
   915→
   916→        self._analyze_headers(response_headers)
   917→        self._analyze_html(html_body)
   918→        self._analyze_cookies(response_headers)
   919→        self._analyze_url(target)
   920→        self._analyze_js_files(js_files)
   921→        self._analyze_url_paths(target)
   922→
   923→        if self._hits > 0:
   924→            self._confidence = min(self._hits * 0.12, 1.0)
   925→
   926→        return TechProfile(
   927→            frameworks=list(self._frameworks),
   928→            language=self._language,
   929→            server=self._server,
   930→            waf=self._waf,
   931→            cms=self._cms,
   932→            has_graphql=self._has_graphql,
   933→            has_rest_api=self._has_rest_api,
   934→            confidence=self._confidence,
   935→        )
   936→
   937→    def _analyze_headers(self, headers: dict[str, str]) -> None:
   938→        header_signatures = TECH_SIGNATURES["headers"]
   939→        header_map = TECH_SIGNATURES["waf_signatures"]
   940→
   941→        for header_key, mapping in header_signatures.items():
   942→            header_val = headers.get(header_key, "")
   943→            if not isinstance(mapping, dict):
   944→                continue
   945→            for pattern, tech in mapping.items():
   946→                if isinstance(tech, str) and tech and pattern:
   947→                    if pattern.lower() in header_val.lower():
   948→                        if tech == "Drupal" or tech == "WordPress" or tech == "Joomla":
   949→                            self._cms = tech
   950→                            self._hits += 2
   951→                        elif tech in ("nginx", "Apache", "IIS", "Caddy", "LiteSpeed", "OpenResty"):
   952→                            self._server = tech
   953→                            self._hits += 1
   954→                        elif tech == "PHP":
   955→                            self._language = "PHP"
   956→                            self._hits += 1

  1212→class FuzzSession:
  1213→    """Context-aware fuzzing session that selects payloads based on the
  1214→    target's detected technology stack."""
  1215→
  1216→    def __init__(
  1217→        self,
  1218→        target: str,
  1219→        tech_profile: Optional[TechProfile] = None,
  1220→        timeout: float = 10.0,
  1221→        user_agent: str = "ReconPro/7.5 (Security Scanner)",
  1222→        verify_ssl: bool = False,
  1223→    ) -> None:
  1224→        self.target = target.rstrip("/")
  1225→        self.tech_profile = tech_profile
  1226→        self.timeout = timeout
  1227→        self.user_agent = user_agent
  1228→        self.verify_ssl = verify_ssl
  1229→        self.results: list[FuzzResult] = []
  1230→        self._base_time: float = 0.0
  1231→        self._baseline_status: int = 0
  1232→        self._opener = self._build_opener()
  1233→
  1234→    def _build_opener(self) -> urllib.request.OpenerDirector:
  1235→        """Build a URL opener with optional SSL bypass."""
  1236→        if not self.verify_ssl:
  1237→            import ssl
  1238→            ctx = ssl._create_unverified_context()
  1239→            https_handler = urllib.request.HTTPSHandler(context=ctx)
  1240→            return urllib.request.build_opener(https_handler)
  1241→        return urllib.request.build_opener()
  1242→
  1243→    def _make_request(
  1244→        self,
  1245→        url: str,
  1246→        params: Optional[dict[str, str]] = None,
  1247→        headers: Optional[dict[str, str]] = None,
  1248→        method: str = "GET",
  1249→    ) -> tuple[int, str, float, dict[str, str]]:
  1250→        """Make an HTTP request and return (status_code, body, time, headers)."""
  1251→        if params and method == "GET":
  1252→            query = urllib.parse.urlencode(params)
  1253→            url = f"{url}?{query}"
  1254→
  1255→        req_headers = {
  1256→            "User-Agent": self.user_agent,
  1257→            "Accept": "text/html,application/xhtml+xml,application/xml;q=0.9,*/*;q=0.8",
  1258→            "Accept-Language": "en-US,en;q=0.5",
  1259→        }
  1260→        if headers:
  1261→            req_headers.update(headers)
  1262→
  1263→        body_data: Optional[bytes] = None
  1264→        if params and method == "POST":
  1265→            body_data = urllib.parse.urlencode(params).encode("utf-8")
  1266→            req_headers["Content-Type"] = "application/x-www-form-urlencoded"
  1267→
  1268→        req = urllib.request.Request(url, data=body_data, headers=req_headers, method=method)
  1269→
  1270→        try:
  1271→            start = time.monotonic()
  1272→            resp = self._opener.open(req, timeout=self.timeout)
  1273→            elapsed = time.monotonic() - start
  1274→            resp_body = resp.read().decode("utf-8", errors="replace")
  1275→            resp_headers = dict(resp.headers)
  1276→            return resp.status, resp_body, elapsed, resp_headers
  1277→        except urllib.error.HTTPError as e:
  1278→            elapsed = time.monotonic() - time.monotonic() + 0.001
  1279→            try:
  1280→                resp_body = e.read().decode("utf-8", errors="replace")
  1281→            except Exception:
  1282→                resp_body = ""
  1283→            resp_headers = dict(e.headers) if hasattr(e, "headers") else {}
  1284→            return e.code, resp_body, elapsed, resp_headers
  1285→        except urllib.error.URLError:
  1286→            elapsed = 0.0
  1287→            return 0, "", elapsed, {}
  1288→        except Exception:
  1289→            elapsed = 0.0
  1290→            return 0, "", elapsed, {}
  1291→
  1292→    def calibrate(self) -> None:
  1293→        """Send a baseline request to establish normal response time and status."""
  1294→        status, _, elapsed, _ = self._make_request(self.target)
  1295→        self._baseline_status = status
  1296→        self._base_time = elapsed

  1298→    def get_all_categories(self) -> list[str]:
  1299→        """Return a list of all available payload categories."""
  1300→        return list(PAYLOAD_DATABASE.keys())
  1301→
  1302→    def select_payloads(self, category: str) -> list[str]:
  1303→        """Select the most relevant payloads for the detected tech stack.
  1304→
  1305→        Priority logic:
  1306→          1. If tech_profile is available, prefer framework-specific payloads.
  1307→          2. Language-specific SQL payloads (e.g., Django -> postgres, WordPress -> mysql).
  1308→          3. Framework-specific template payloads (e.g., Django -> jinja2/django SSTI).
  1309→          4. CMS-specific extras for WordPress/Drupal/Joomla.
  1310→          5. Generic payloads always included as fallback.
  1311→        """
  1312→        if category not in PAYLOAD_DATABASE:
  1313→            return []
  1314→
  1315→        cat_payloads = PAYLOAD_DATABASE[category]
  1316→        selected: list[str] = []
  1317→        seen: set[str] = set()
  1318→
  1319→        def _add(payloads: list[str]) -> None:
  1320→            for p in payloads:
  1321→                if p not in seen:
  1322→                    selected.append(p)
  1323→                    seen.add(p)
  1324→
  1325→        # -- If we have a tech profile, prioritize matching payloads --------
  1326→        if self.tech_profile:
  1327→            tp = self.tech_profile
  1328→            fw_set = set(tp.frameworks)
  1329→            lang = tp.language
  1330→
  1331→            # --- SQL injection: route by detected language/backend ----------
  1332→            if category == "sqli":
  1333→                if any(fw in fw_set for fw in ("Django", "Flask", "Python", "FastAPI")):
  1334→                    if "postgres" in cat_payloads:
  1335→                        _add(cat_payloads["postgres"])
  1336→                if lang == "PHP" or tp.cms in ("WordPress", "Drupal", "Joomla"):
  1337→                    if "mysql" in cat_payloads:
  1338→                        _add(cat_payloads["mysql"])
  1339→                if lang == "Java" or any(fw in fw_set for fw in ("Spring Boot", "Tomcat", "Jetty")):
  1340→                    if "oracle" in cat_payloads:
  1341→                        _add(cat_payloads["oracle"])
  1342→                    if "mssql" in cat_payloads:
  1343→                        _add(cat_payloads["mssql"])
  1344→                if lang == "ASP.NET" or "ASP.NET" in fw_set:
  1345→                    if "mssql" in cat_payloads:
  1346→                        _add(cat_payloads["mssql"])
  1347→                if lang == "Ruby" or "Rails" in fw_set:
  1348→                    if "sqlite" in cat_payloads:
  1349→                        _add(cat_payloads["sqlite"])
  1350→                    if "postgres" in cat_payloads:
  1351→                        _add(cat_payloads["postgres"])
  1352→
  1353→            # --- XSS: route by detected frontend framework -----------------
  1354→            elif category == "xss":
  1355→                if "Angular" in fw_set:
  1356→                    _add(cat_payloads.get("angular", []))
  1357→                if "React" in fw_set:
  1358→                    _add(cat_payloads.get("react", []))
  1359→                if "Vue" in fw_set or "Nuxt" in fw_set:
  1360→                    _add(cat_payloads.get("vue", []))
  1361→                _add(cat_payloads.get("html", []))
  1362→
  1363→            # --- SSTI: route by detected backend template engine ------------
  1364→            elif category == "ssti":
  1365→                if any(fw in fw_set for fw in ("Django", "Python", "Flask", "FastAPI")):
  1366→                    _add(cat_payloads.get("jinja2", []))
  1367→                    _add(cat_payloads.get("django", []))
  1368→                if "Ruby" in fw_set or "Rails" in fw_set:
  1369→                    _add(cat_payloads.get("mako", []))
  1370→                if "PHP" in lang or tp.cms == "Drupal":
  1371→                    _add(cat_payloads.get("twig", []))
  1372→
  1373→            # --- SSRF: prefer cloud provider based on headers --------------
  1374→            elif category == "ssrf":
  1375→                if tp.server and "amazon" in tp.server.lower():
  1376→                    _add(cat_payloads.get("aws_metadata", []))
  1377→                elif tp.server and "google" in tp.server.lower():
  1378→                    _add(cat_payloads.get("gcp_metadata", []))
  1379→                elif "azure" in (tp.server or "").lower():
  1380→                    _add(cat_payloads.get("azure_metadata", []))
  1381→                else:
  1382→                    for sub in ("aws_metadata", "gcp_metadata", "azure_metadata"):
  1383→                        _add(cat_payloads.get(sub, []))
  1384→
  1385→            # --- Path traversal: OS-specific based on server ---------------
  1386→            elif category == "path_traversal":
  1387→                if tp.server and "iis" in tp.server.lower():
  1388→                    _add(cat_payloads.get("windows", []))
  1389→                else:
  1390→                    _add(cat_payloads.get("unix", []))
  1391→
  1392→            # --- Command injection: OS-specific based on server -------------
  1393→            elif category == "command_injection":
  1394→                if tp.server and "iis" in tp.server.lower():
  1395→                    _add(cat_payloads.get("windows", []))
  1396→                else:
  1397→                    _add(cat_payloads.get("unix", []))
  1398→
  1399→            # --- LFI: OS-specific based on server ----------------------------
  1400→            elif category == "lfi":
  1401→                _add(cat_payloads.get("generic", []))
  1402→
  1403→            # --- RFI: all payloads (remote URIs) -----------------------------
  1404→            elif category == "rfi":
  1405→                _add(cat_payloads.get("generic", []))
  1406→
  1407→        # -- Always add generic payloads as fallback -------------------------
  1408→        if "generic" in cat_payloads:
  1409→            _add(cat_payloads["generic"])
  1410→
  1411→        # -- Add any remaining sub-categories not yet covered ---------------
  1412→        for sub_key, sub_payloads in cat_payloads.items():
  1413→            if sub_key != "generic" and sub_key not in ("mysql", "postgres", "mssql", "sqlite", "oracle",
  1414→                                                        "html", "angular", "react", "vue",
  1415→                                                        "jinja2", "django", "mako", "twig",
  1416→                                                        "aws_metadata", "gcp_metadata", "azure_metadata",
  1417→                                                        "unix", "windows"):
  1418→                if not self.tech_profile:
  1419→                    _add(sub_payloads)
  1420→
  1421→        return selected

  1423→    def fuzz_parameter(
  1424→        self,
  1425→        url: str,
  1426→        param: str,
  1427→        payloads: Optional[list[str]] = None,
  1428→        category: str = "generic",
  1429→        method: str = "GET",
  1430→    ) -> list[FuzzResult]:
  1431→        """Fuzz a single URL parameter with selected payloads.
  1432→
  1433→        For each payload:
  1434→          1. Substitute the payload into the parameter value.
  1435→          2. Send the request.
  1436→          3. Analyze the response for anomalies.
  1437→          4. Return a list of FuzzResult objects.
  1438→        """
  1439→        if payloads is None:
  1440→            payloads = self.select_payloads(category)
  1441→
  1442→        if not payloads:
  1443→            return []
  1444→
  1445→        # Fetch the original (clean) response for comparison
  1446→        original_status, original_body, _, _ = self._make_request(url)
  1447→
  1448→        results: list[FuzzResult] = []
  1449→
  1450→        for payload in payloads:
  1451→            params = {param: payload}
  1452→
  1453→            status, body, elapsed, resp_headers = self._make_request(url, params=params, method=method)
  1454→
  1455→            result = self._analyze_response(
  1456→                original_body=original_body,
  1457→                fuzzed_body=body,
  1458→                status_code=status,
  1459→                response_time=elapsed,
  1460→            )
  1461→            result.payload = payload
  1462→            result.category = category
  1463→            result.url = url
  1464→            result.param = param
  1465→            result.status_code = status
  1466→
  1467→            results.append(result)
  1468→
  1469→        self.results.extend(results)
  1470→        return results

  1528→    def analyze_response(
  1529→        self,
  1530→        original_body: str,
  1531→        fuzzed_body: str,
  1532→        status_code: int,
  1533→        response_time: float,
  1534→    ) -> FuzzResult:
  1535→        """Analyze a fuzzed response and classify the result.
  1536→
  1537→        Detection methods:
  1538→          - Reflected: the payload itself appears in the response body.
  1539→          - Error-based: SQL errors, stack traces, or framework errors in the body.
  1540→          - Time-based: response took significantly longer than baseline.
  1541→          - Blind: status code change (e.g., 500 instead of 200) without obvious error.
  1542→        """
  1543→        result = FuzzResult()
  1544→        result.status_code = status_code
  1545→        result.response_time = response_time
  1546→        result.matched_pattern = ""
  1547→
  1548→        # --- Check for reflected payload -----------------------------------
  1549→        if fuzzed_body and original_body:
  1550→            # Find the payload in the response (may be HTML-encoded)
  1551→            payload_escaped = re.escape(result.payload)
  1552→            if result.payload and re.search(payload_escaped, fuzzed_body):
  1553→                result.classification = "reflected"
  1554→                result.matched_pattern = "Payload reflected in response body"
  1555→
  1556→            # Also check for partial reflection or HTML entity encoding
  1557→            if result.payload and not result.classification:
  1558→                html_encoded = (
  1559→                    result.payload.replace("<", "&lt;")
  1560→                    .replace(">", "&gt;")
  1561→                    .replace('"', "&quot;")
  1562→                    .replace("'", "&#39;")
  1563→                )
  1564→                if html_encoded in fuzzed_body:
  1565→                    result.classification = "reflected"
  1566→                    result.matched_pattern = "Payload HTML-encoded in response"
  1567→
  1568→        # --- Check for SQL errors in the response --------------------------
  1569→        if fuzzed_body:
  1570→            for pattern in _SQL_ERROR_PATTERNS:
  1571→                match = pattern.search(fuzzed_body)
  1572→                if match:
  1573→                    result.classification = "error_based"
  1574→                    result.matched_pattern = match.group(0)
  1575→                    break
  1576→
  1577→        # --- Check for stack traces ----------------------------------------
  1578→        if fuzzed_body and result.classification != "error_based":
  1579→            for pattern in _STACK_TRACE_PATTERNS:
  1580→                match = pattern.search(fuzzed_body)
  1581→                if match:
  1582→                    result.classification = "error_based"
  1583→                    result.matched_pattern = match.group(0)
  1584→                    break
  1585→
  1586→        # --- Check for command injection output ---------------------------
  1587→        if fuzzed_body and result.classification != "error_based":
  1588→            match = _CMD_INJECTION_PATTERN.search(fuzzed_body)
  1589→            if match:
  1590→                result.classification = "error_based"
  1591→                result.matched_pattern = match.group(0)
  1592→
  1593→        # --- Check for path traversal output --------------------------------
  1594→        if fuzzed_body and result.classification != "error_based":
  1595→            match = _PATH_TRAVERSAL_PATTERN.search(fuzzed_body)
  1596→            if match:
  1597→                result.classification = "error_based"
  1598→                result.matched_pattern = match.group(0)
  1599→
  1600→        # --- Check for XSS indicators -------------------------------------
  1601→        if fuzzed_body and result.classification != "reflected":
  1602→            match = _XSS_PATTERN.search(fuzzed_body)
  1603→            if match:
  1604→                result.classification = "reflected"
  1605→                result.matched_pattern = match.group(0)
  1606→
  1607→        # --- Check for SSTI indicators -------------------------------------
  1608→        if fuzzed_body and result.classification not in ("reflected", "error_based"):
  1609→            match = _SSTI_PATTERN.search(fuzzed_body)
  1610→            if match:
  1611→                result.classification = "reflected"
  1612→                result.matched_pattern = match.group(0)
  1613→
  1614→        # --- Check for LFI indicators ---------------------------------------
  1615→        if fuzzed_body and result.classification not in ("reflected", "error_based"):
  1616→            match = _LFI_PATTERN.search(fuzzed_body)
  1617→            if match:
  1618→                result.classification = "error_based"
  1619→                result.matched_pattern = match.group(0)
  1620→
  1621→        # --- Check for RFI indicators ---------------------------------------
  1622→        if fuzzed_body and result.classification not in ("reflected", "error_based"):
  1623→            match = _RFI_PATTERN.search(fuzzed_body)
  1624→            if match:
  1625→                result.classification = "error_based"
  1626→                result.matched_pattern = match.group(0)
  1627→
  1628→        # --- Time-based detection ------------------------------------------
  1629→        if self._base_time > 0 and response_time > self._base_time * 3:
  1630→            if response_time > 4.0:
  1631→                if result.classification in ("none", ""):
  1632→                    result.classification = "time_based"
  1633→                    result.matched_pattern = f"Response time {response_time:.2f}s (baseline {self._base_time:.2f}s)"
  1634→                elif result.classification == "reflected":
  1635→                    result.classification = "time_based"
  1636→                    result.matched_pattern = f"Reflected + slow response ({response_time:.2f}s)"
  1637→
  1638→        # --- Status code change detection (blind) ---------------------------
  1639→        if (
  1640→            self._baseline_status > 0
  1641→            and status_code != self._baseline_status
  1642→            and status_code >= 400
  1643→            and result.classification in ("none", "")
  1644→        ):
  1645→            result.classification = "blind"
  1646→            result.matched_pattern = (
  1647→                f"Status changed from {self._baseline_status} to {status_code}"
  1648→            )
  1649→
  1650→        # --- Determine vulnerability ---------------------------------------
  1651→        if result.classification in ("reflected", "error_based", "time_based"):
  1652→            result.is_vulnerable = True
  1653→        elif result.classification == "blind":
  1654→            result.is_vulnerable = True
  1655→
  1656→        return result

  1687→def quick_scan(
  1688→    target: str,
  1689→    params: list[str],
  1690→    categories: Optional[list[str]] = None,
  1691→    timeout: float = 10.0,
  1692→) -> tuple[TechProfile, list[FuzzResult]]:
  1693→    """Convenience function: detect tech, calibrate, and fuzz all params.
  1694→
  1695→    Returns (TechProfile, list_of_results).
  1696→    """
  1697→    if categories is None:
  1698→        categories = ["sqli", "xss", "ssti", "command_injection", "path_traversal", "lfi", "rfi"]
  1699→
  1700→    # Quick tech detection
  1701→    status, body, _, headers = 0, "", 0.0, {}
  1702→    try:
  1703→        parsed = urllib.parse.urlparse(target)
  1704→        base_url = f"{parsed.scheme}://{parsed.netloc}{parsed.path}"
  1705→        ctx = None
  1706→        import ssl
  1707→        try:
  1708→            ctx = ssl._create_unverified_context()
  1709→            handler = urllib.request.HTTPSHandler(context=ctx)
  1710→            opener = urllib.request.build_opener(handler)
  1711→        except Exception:
  1712→            opener = urllib.request.build_opener()
  1713→        req = urllib.request.Request(
  1714→            base_url,
  1715→            headers={"User-Agent": "ReconPro/7.5 (Security Scanner)"},
  1716→        )
  1717→        resp = opener.open(req, timeout=timeout)
  1718→        body = resp.read().decode("utf-8", errors="replace")
  1719→        headers = dict(resp.headers)
  1720→        status = resp.status
  1721→    except Exception:
  1722→        pass
  1723→
  1724→    detector = TechDetector()
  1725→    tech_profile = detector.analyze(target, headers, body)
  1726→
  1727→    session = FuzzSession(target, tech_profile=tech_profile, timeout=timeout)
  1728→    session._baseline_status = status
  1729→    results = session.fuzz_all_parameters(target, params, categories)
  1730→    return tech_profile, results
