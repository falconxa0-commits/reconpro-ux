  1125→    def _apply_completion(self, inp: Input, comp: CommandCompleter, accepted: str) -> None:
  1126→        """Apply a completion suggestion to the input field.
  1127→
  1128→        Phase C: handles both command-level and arg-level completions.
  1129→        For arg mode, appends/replaces the last word instead of the first.
  1130→        """
  1131→        parts = inp.value.split()
  1132→
  1133→        if comp._mode == "command":
  1134→            # Replace the first word with the accepted command
  1135→            rest = parts[1:] if len(parts) > 1 else []
  1136→            inp.value = accepted + (" " + " ".join(rest) if rest else "")
  1137→        else:
  1138→            # Arg-level: replace the last word
  1139→            if len(parts) >= 2:
  1140→                parts[-1] = accepted
  1141→                inp.value = " ".join(parts)
  1142→            else:
  1143→                inp.value = accepted
  1144→
  1145→        inp.cursor_position = len(inp.value)

  1671→    def on_resize(self, event) -> None:
  1672→        """Adapt layout to terminal size changes.
  1673→
  1674→        Responsive breakpoints:
  1675→          - < 80 cols:  compact mode (stack panels vertically, 2-col grid)
  1676→          - 80–119 cols: standard (3-col grid, default 40:60 split)
  1677→          - 120–159 cols: wide (3-col grid, expanded stats)
  1678→          - >= 160 cols:  ultra-wide (4-col grid, 35:65 split)
  1679→          - >= 200 cols:  cinematic (4-col grid, 30:70 split)
  1680→        """
  1681→        # Debounce to avoid flicker during rapid resize
  1682→        if self._resize_timer is not None:
  1683→            self._resize_timer.stop()
  1684→        self._resize_timer = self.set_timer(0.15, lambda: self._apply_responsive_layout(event.width, event.height))

  1686→    def _apply_responsive_layout(self, width: int, height: int) -> None:
  1687→        """Apply layout changes based on terminal dimensions.
  1688→
  1689→        Dynamically syncs ``_GRID_COLS`` with the actual CSS grid columns
  1690→        so arrow-key navigation on the module grid stays correct.
  1691→        Adjusts split ratio defaults for wider terminals.
  1692→        """
  1693→        try:
  1694→            content = self.query_one("#content", Widget)
  1695→            grid = self.query_one("#module-grid", Container)
  1696→        except NoMatches:
  1697→            return
  1698→
  1699→        # ── Compact mode: < 80 cols ──
  1700→        if width < 80:
  1701→            if not self._compact_mode:
  1702→                content.set_class(True, "compact")
  1703→                grid.set_class(True, "grid-cols-2")
  1704→                grid.remove_class("grid-cols-4")
  1705→                self._compact_mode = True
  1706→                self._GRID_COLS = 2
  1707→                # Force-collapse left panel in very narrow terminals
  1708→                if width < 60 and not self._left_collapsed:
  1709→                    self._left_collapsed_before_compact = True
  1710→                    self._toggle_left_panel()
  1711→                self._flash_split_indicator(f"COMPACT {width}x{height}")
  1712→            return
  1713→
  1714→        # ── Ultra-wide: >= 160 cols ──
  1715→        if width >= 160:
  1716→            if self._compact_mode:
  1717→                content.remove_class("compact")
  1718→                self._compact_mode = False
  1719→                # Restore auto-collapsed left panel
  1720→                if getattr(self, '_left_collapsed_before_compact', False) and self._left_collapsed:
  1721→                    self._toggle_left_panel()
  1722→                    self._left_collapsed_before_compact = False
  1723→            grid.set_class(True, "grid-cols-4")
  1724→            grid.remove_class("grid-cols-2")
  1725→            self._GRID_COLS = 4
  1726→            # Auto-adjust split for cinematic widths
  1727→            if width >= 200:
  1728→                optimal_split = 30
  1729→            else:
  1730→                optimal_split = 35
  1731→            # Only auto-adjust if user hasn't manually set split
  1732→            if not self._user_manually_resized:
  1733→                self._split_ratio = optimal_split
  1734→                self._apply_split_ratio_silent()
  1735→            label = "CINEMATIC" if width >= 200 else "ULTRA"
  1736→            self._flash_split_indicator(f"{label} {width}x{height}")
  1737→            return
  1738→
  1739→        # ── Wide: 120–159 cols ──
  1740→        if width >= 120:
  1741→            if self._compact_mode:
  1742→                content.remove_class("compact")
  1743→                self._compact_mode = False
  1744→                if getattr(self, '_left_collapsed_before_compact', False) and self._left_collapsed:
  1745→                    self._toggle_left_panel()
  1746→                    self._left_collapsed_before_compact = False
  1747→            grid.remove_class("grid-cols-4")
  1748→            grid.remove_class("grid-cols-2")
  1749→            self._GRID_COLS = 3
  1750→            if not self._user_manually_resized:
  1751→                self._split_ratio = 38
  1752→                self._apply_split_ratio_silent()
  1753→            self._flash_split_indicator(f"WIDE {width}x{height}")
  1754→            return
  1755→
  1756→        # ── Standard: 80–119 cols ──
  1757→        if self._compact_mode:
  1758→            content.remove_class("compact")
  1759→            self._compact_mode = False
  1760→            if getattr(self, '_left_collapsed_before_compact', False) and self._left_collapsed:
  1761→                self._toggle_left_panel()
  1762→                self._left_collapsed_before_compact = False
  1763→        grid.remove_class("grid-cols-4")
  1764→        grid.remove_class("grid-cols-2")
  1765→        self._GRID_COLS = 3
  1766→        if not self._user_manually_resized:
  1767→            self._split_ratio = 40
  1768→            self._apply_split_ratio_silent()
  1769→        self._flash_split_indicator(f"STD {width}x{height}")
  1770→

  2438→    # Background Workers (concurrent)
  2439→    # ════════════════════════════════════════════════════════════════════════
  2440→
  2441→    @work(exclusive=False, thread=True)
  2442→    def _run_scan_worker(
  2443→        self,
  2444→        target: str,
  2445→        modules: Optional[List[str]] = None,
  2446→        is_local: bool = False,
  2447→    ) -> None:
  2448→        """Run a scan in a background thread."""
  2449→        self.call_from_thread(self.is_scanning.set, True)
  2450→
  2451→        # Set modules to scanning + push module-specific hints
  2452→        if modules:
  2453→            for m in modules:
  2454→                if m in self._module_cells:
  2455→                    self.call_from_thread(self._module_cells[m].set_status, "scanning")
  2456→                    self.call_from_thread(self._push_module_scanning_hint, m)
  2457→        else:
  2458→            all_mods = (
  2459→                list(MODULE_NAMES.keys())
  2460→                if is_local
  2461→                else [k for k in MODULE_NAMES if k not in ("host", "dev", "doctor")]
  2462→            )
  2463→            for m in all_mods:
  2464→                if m in self._module_cells:
  2465→                    self.call_from_thread(self._module_cells[m].set_status, "scanning")
  2466→
  2467→        try:
  2468→            from .scanner import scan, audit_scan
  2469→            from .history import save_scan
  2470→
  2471→            self.call_from_thread(
  2472→                self._chat, f"[{DIM_CYAN}]  ⏳ Scan in progress...[/]"
  2473→            )
  2474→
  2475→            if is_local:
  2476→                result = audit_scan(target=target, modules=modules)
  2477→            else:
  2478→                result = scan(target=target, modules=modules)
  2479→
