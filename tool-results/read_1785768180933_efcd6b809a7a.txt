  2185→    def _run_scan_worker(
  2186→        self,
  2187→        target: str,
  2188→        modules: Optional[List[str]] = None,
  2189→        is_local: bool = False,
  2190→    ) -> None:
  2191→        """Run a scan in a background thread."""
  2192→        self.call_from_thread(self.is_scanning.set, True)
  2193→
  2194→        # Set modules to scanning + push module-specific hints
  2195→        if modules:
  2196→            for m in modules:
  2197→                if m in self._module_cells:
  2198→                    self.call_from_thread(self._module_cells[m].set_status, "scanning")
  2199→                    self.call_from_thread(self._push_module_scanning_hint, m)
  2200→        else:
  2201→            all_mods = (
  2202→                list(MODULE_NAMES.keys())
  2203→                if is_local
  2204→                else [k for k in MODULE_NAMES if k not in ("host", "dev", "doctor")]
  2205→            )
  2206→            for m in all_mods:
  2207→                if m in self._module_cells:
  2208→                    self.call_from_thread(self._module_cells[m].set_status, "scanning")
  2209→
  2210→        try:
  2211→            from .scanner import scan, audit_scan
  2212→            from .history import save_scan
  2213→
  2214→            self.call_from_thread(
  2215→                self._chat, f"[{DIM_CYAN}]  ⏳ Scan in progress...[/]"
  2216→            )
  2217→
  2218→            if is_local:
  2219→                result = audit_scan(target=target, modules=modules)
  2220→            else:
  2221→                result = scan(target=target, modules=modules)
  2222→
  2223→            # Process findings
  2224→            data = result.to_dict()
  2225→            self.call_from_thread(self._on_scan_complete, data, modules, is_local)
  2226→
  2227→            # Save to history
  2228→            try:
  2229→                save_scan(data, label="nexus")
  2230→            except Exception:
  2231→                pass
  2232→
  2233→        except Exception as e:
  2234→            self.call_from_thread(self._chat, f"[{RED}]  ✗ Scan error: {e}[/]")
  2235→            # Phase C: push error_state hints
  2236→            self.call_from_thread(self._push_error_hint)
  2237→            if modules:
  2238→                for m in modules:
  2239→                    if m in self._module_cells:
  2240→                        self.call_from_thread(self._module_cells[m].set_status, "error")
  2241→            else:
  2242→                self.call_from_thread(self._set_all_modules_status, "error")
  2243→
  2244→        finally:
  2245→            self.call_from_thread(self.is_scanning.set, False)

  2677→    def on_click(self, event: Click) -> None:
  2678→        """Open finding detail modal for the clicked finding.
  2679→
  2680→        Uses the vertical click position to determine which finding index
  2681→        the user clicked on, mapping it to _findings_list.
  2682→        """
  2683→        try:
  2684→            if (
  2685→                isinstance(event.widget, RichLog)
  2686→                and event.widget.id == "findings-feed"
  2687→                and self._findings_list
  2688→            ):
  2689→                # Get the line offset to determine which finding was clicked.
  2690→                # Each finding entry is 1 line in the RichLog.
  2691→                # We use the y coordinate relative to the widget to estimate the index.
  2692→                region = event.widget.region
  2693→                if region is not None:
  2694→                    # Calculate which line was clicked relative to the scroll position
  2695→                    scroll_offset = event.widget.scroll_y
  2696→                    click_y = event.y - region.y
  2697→                    # Each finding takes 1 line; estimate index from click position
  2698→                    # Account for scroll position
  2699→                    line_height = 1
  2700→                    if hasattr(event.widget, "_line_height"):
  2701→                        line_height = event.widget._line_height
  2702→                    clicked_index = int((click_y + scroll_offset * line_height) / line_height)
  2703→                    # Clamp to valid range
  2704→                    clicked_index = max(0, min(clicked_index, len(self._findings_list) - 1))
  2705→                    finding = self._findings_list[clicked_index]
  2706→                else:
  2707→                    finding = self._findings_list[-1]
  2708→                self.push_screen(FindingDetailModal(finding))
  2709→        except (NoMatches, Exception):
  2710→            pass
