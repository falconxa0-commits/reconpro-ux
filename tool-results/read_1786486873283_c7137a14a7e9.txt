   149→    def generate(
   150→        self,
   151→        scan_result: Any,
   152→        intelligence_report: Optional[Any] = None,
   153→        correlation_result: Optional[Any] = None,
   154→    ) -> ExecutiveReport:
   155→        """Generate a full ExecutiveReport from scan data.
   156→
   157→        Parameters
   158→        ----------
   159→        scan_result : ReconProResult | dict
   160→            Scan result (dataclass or dict with findings, scores, etc.).
   161→        intelligence_report : IntelligenceReport | dict | None
   162→            Optional enriched intelligence from the pipeline.
   163→        correlation_result : CorrelationResult | dict | None
   164→            Optional evidence correlation chains.
   165→
   166→        Returns
   167→        -------
   168→        ExecutiveReport
   169→            Complete executive-level report.
   170→        """
   171→        # Normalise scan data to dict.
   172→        if hasattr(scan_result, "to_dict"):
   173→            scan_data = scan_result.to_dict()
   174→        elif isinstance(scan_result, dict):
   175→            scan_data = dict(scan_result)
   176→        else:
   177→            scan_data = {}
   178→
   179→        target = scan_data.get("target", "unknown")
   180→        findings = scan_data.get("findings", [])
   181→        modules_run = scan_data.get("modules_run", [])
   182→        severity_counts = scan_data.get("severity_counts", {})
   183→        total_score = scan_data.get("total_score", 100)
   184→        grade = scan_data.get("grade", "A")
   185→        total_findings = scan_data.get("total_findings", len(findings))
   186→
   187→        timestamp = datetime.now(timezone.utc).isoformat()
   188→
   189→        # Extract confidence scores from intelligence_report.
   190→        confidence_scores: List[Dict[str, Any]] = []
   191→        eng_report_dict: Dict[str, Any] = {}
   192→        if intelligence_report is not None:
   193→            if hasattr(intelligence_report, "to_dict"):
   194→                ir_dict = intelligence_report.to_dict()
   195→            elif isinstance(intelligence_report, dict):
   196→                ir_dict = intelligence_report
   197→            else:
   198→                ir_dict = {}
   199→            confidence_scores = ir_dict.get("confidence_scores", [])
   200→            eng_report_dict = ir_dict.get("engineering_report", {})
   201→
   202→        # If no engineering report from pipeline, compute one.
   203→        if not eng_report_dict:
   204→            try:
   205→                from .engineering_score import EngineeringScorer
   206→                scorer = EngineeringScorer()
   207→                eng = scorer.score(target, findings)
   208→                eng_report_dict = eng.to_dict()
   209→            except Exception:
   210→                eng_report_dict = {"overall_score": total_score, "dimensions": {}}
   211→
   212→        # Extract evidence chains from correlation_result.
   213→        chains: List[Dict[str, Any]] = []
   214→        if correlation_result is not None:
   215→            if hasattr(correlation_result, "to_dict"):
   216→                cr_dict = correlation_result.to_dict()
   217→            elif isinstance(correlation_result, dict):
   218→                cr_dict = correlation_result
   219→            else:
   220→                cr_dict = {}
   221→            chains = cr_dict.get("chains", cr_dict.get("evidence_chains", []))
   222→
   223→        # Build report sections.
   224→        scan_summary = {
   225→            "score": total_score,
   226→            "grade": grade,
   227→            "modules_run": modules_run,
   228→            "findings_count": total_findings,
   229→            "severity_counts": severity_counts,
   230→        }
   231→
   232→        report_data = {
   233→            "target": target,
   234→            "total_findings": total_findings,
   235→            "severity_counts": severity_counts,
   236→            "total_score": total_score,
   237→            "grade": grade,
   238→            "modules_run": modules_run,
   239→        }
   240→
   241→        executive_summary = self._build_summary(report_data)
   242→        risk_matrix = self._build_risk_matrix(findings, confidence_scores)
   243→        attack_timeline = self._build_attack_timeline(findings, chains)
   244→        top_findings = self._prioritize_findings(findings, confidence_scores)
   245→        remediation_plan = self._build_remediation_plan(findings, eng_report_dict)
   246→        evidence_chain_dicts = self._extract_evidence_chains(chains, findings)
   247→
   248→        # Machine-readable JSON.
   249→        machine_json = {
   250→            "target": target,
   251→            "timestamp": timestamp,
   252→            "scan_summary": scan_summary,
   253→            "risk_matrix": risk_matrix,
   254→            "attack_timeline": attack_timeline,
   255→            "top_findings": top_findings,
   256→            "remediation_plan": remediation_plan,
   257→            "engineering_report": eng_report_dict,
   258→            "evidence_chains": evidence_chain_dicts,
   259→            "meta": {
   260→                "generator": "ReconPro ExecutiveIntelligence",
   261→                "council": "Council Theta",
   262→                "version": "1.0.0",
   263→            },
   264→        }
   265→
   266→        return ExecutiveReport(
   267→            target=target,
   268→            timestamp=timestamp,
   269→            scan_summary=scan_summary,
   270→            executive_summary=executive_summary,
   271→            risk_matrix=risk_matrix,
   272→            attack_timeline=attack_timeline,
   273→            top_findings=top_findings,
   274→            remediation_plan=remediation_plan,
   275→            engineering_report=eng_report_dict,
   276→            evidence_chains=evidence_chain_dicts,
   277→            machine_json=machine_json,
   278→        )

   284→    def _build_summary(self, report_data: Dict[str, Any]) -> str:
   285→        """Generate a 2-3 paragraph executive summary."""
   286→        target = report_data.get("target", "unknown")
   287→        score = report_data.get("total_score", 100)
   288→        grade = report_data.get("grade", "A")
   289→        total = report_data.get("total_findings", 0)
   290→        sev_counts = report_data.get("severity_counts", {})
   291→        modules = report_data.get("modules_run", [])
   292→
   293→        critical = sev_counts.get("critical", 0)
   294→        high = sev_counts.get("high", 0)
   295→        medium = sev_counts.get("medium", 0)
   296→        low = sev_counts.get("low", 0)
   297→
   298→        # Paragraph 1: Overview.
   299→        p1 = (
   300→            f"A security assessment of {target} was completed using "
   301→            f"{len(modules)} scan module{'s' if len(modules) != 1 else ''} "
   302→            f"({', '.join(modules[:5])}{'...' if len(modules) > 5 else ''}). "
   303→            f"The target received an overall security score of {score}/100 "
   304→            f"(grade {grade}) with {total} finding{'s' if total != 1 else ''} identified."
   305→        )
   306→
   307→        # Paragraph 2: Severity breakdown.
   308→        severity_parts: List[str] = []
   309→        if critical:
   310→            severity_parts.append(f"{critical} critical")
   311→        if high:
   312→            severity_parts.append(f"{high} high")
   313→        if medium:
   314→            severity_parts.append(f"{medium} medium")
   315→        if low:
   316→            severity_parts.append(f"{low} low")
   317→
   318→        if severity_parts:
   319→            sev_text = ", ".join(severity_parts)
   320→            p2_parts = [f"The severity distribution is: {sev_text}. "]
   321→            if critical or high:
   322→                p2_parts.append(
   323→                    "Critical and high-severity issues require immediate attention "
   324→                    "and should be addressed before the next deployment cycle. "
   325→                )
   326→            else:
   327→                p2_parts.append(
   328→                    "No critical or high-severity vulnerabilities were detected, "
   329→                    "indicating a strong baseline security posture. "
   330→                )
   331→            if medium:
   332→                p2_parts.append(
   333→                    "Medium-severity findings represent areas for improvement "
   334→                    "in the next sprint. "
   335→                )
   336→            p2 = "".join(p2_parts)
   337→        else:
   338→            p2 = "No vulnerabilities were detected during this assessment."
   339→
   340→        # Paragraph 3: Risk posture.
   341→        if score >= 80:
   342→            posture = "strong"
   343→            action = "Continue monitoring and address medium-severity items in upcoming sprints."
   344→        elif score >= 60:
   345→            posture = "moderate"
   346→            action = "Prioritize high-severity remediation and schedule a follow-up assessment."
   347→        elif score >= 40:
   348→            posture = "elevated"
   349→            action = "Initiate an immediate remediation program with weekly progress reviews."
   350→        else:
   351→            posture = "critical"
   352→            action = "Escalate to leadership and begin emergency remediation immediately."
   353→
   354→        p3 = (
   355→            f"Overall risk posture for {target} is {posture}. {action} "
   356→            f"A detailed remediation plan with prioritized actions is provided below."
   357→        )
   358→
   359→        return f"{p1}\n\n{p2}\n\n{p3}"

   365→    def _build_risk_matrix(
   366→        self,
   367→        findings: List[Dict[str, Any]],
   368→        confidence_scores: List[Dict[str, Any]],
   369→    ) -> List[Dict[str, Any]]:
   370→        """Build risk matrix entries (severity x likelihood)."""
   371→        # Build a confidence lookup by finding title.
   372→        conf_map: Dict[str, float] = {}
   373→        for cs in confidence_scores:
   374→            title = cs.get("title", "")
   375→            conf_val = cs.get("confidence", 0.5)
   376→            if isinstance(conf_val, (int, float)):
   377→                conf_map[title] = float(conf_val)
   378→            else:
   379→                conf_map[title] = 0.5
   380→
   381→        entries: List[RiskMatrixEntry] = []
   382→        for f in findings:
   383→            title = f.get("title", "Untitled Finding")
   384→            severity = str(f.get("severity", "info")).strip().lower()
   385→            category = str(f.get("category", "general")).strip().lower()
   386→
   387→            likelihood = conf_map.get(title, self._default_likelihood(severity))
   388→            impact = _SEVERITY_IMPACT.get(severity, 0.2)
   389→            risk_score = likelihood * impact
   390→            recommendation = _REMEDIATION_TEMPLATES.get(
   391→                category, f"Review and remediate the {category} finding."
   392→            )
   393→
   394→            entries.append(RiskMatrixEntry(
   395→                finding_title=title,
   396→                severity=severity,
   397→                likelihood=likelihood,
   398→                impact=impact,
   399→                risk_score=risk_score,
   400→                category=_CATEGORY_LABELS.get(category, category),
   401→                recommendation=recommendation,
   402→            ))
   403→
   404→        # Sort by risk_score descending.
   405→        entries.sort(key=lambda e: e.risk_score, reverse=True)
   406→        return [e.to_dict() for e in entries]
   407→
   408→    @staticmethod
   409→    def _default_likelihood(severity: str) -> float:
   410→        """Estimate likelihood from severity when no confidence score exists."""
   411→        mapping = {
   412→            "critical": 0.9,
   413→            "high": 0.75,
   414→            "medium": 0.5,
   415→            "low": 0.3,
   416→            "info": 0.1,
   417→        }
   418→        return mapping.get(severity, 0.5)

   424→    def _build_attack_timeline(
   425→        self,
   426→        findings: List[Dict[str, Any]],
   427→        chains: List[Dict[str, Any]],
   428→    ) -> List[Dict[str, Any]]:
   429→        """Build chronological attack path from findings and chains."""
   430→        timeline: List[Dict[str, Any]] = []
   431→        seen_titles: set = set()
   432→
   433→        # Phase 1: Reconnaissance findings.
   434→        for f in findings:
   435→            cat = str(f.get("category", "")).lower()
   436→            if cat in ("info_disclosure", "headers", "network"):
   437→                title = f.get("title", "")
   438→                if title not in seen_titles:
   439→                    timeline.append({
   440→                        "phase": 1,
   441→                        "phase_name": "Reconnaissance",
   442→                        "finding": title,
   443→                        "severity": f.get("severity", "info"),
   444→                        "description": f.get("description", f.get("evidence", ""))[:200],
   445→                    })
   446→                    seen_titles.add(title)
   447→
   448→        # Phase 2: Access / auth findings.
   449→        for f in findings:
   450→            cat = str(f.get("category", "")).lower()
   451→            if cat in ("auth", "session", "csrf"):
   452→                title = f.get("title", "")
   453→                if title not in seen_titles:
   454→                    timeline.append({
   455→                        "phase": 2,
   456→                        "phase_name": "Access & Authentication",
   457→                        "finding": title,
   458→                        "severity": f.get("severity", "info"),
   459→                        "description": f.get("description", f.get("evidence", ""))[:200],
   460→                    })
   461→                    seen_titles.add(title)
   462→
   463→        # Phase 3: Exploitation findings.
   464→        for f in findings:
   465→            cat = str(f.get("category", "")).lower()
   466→            if cat in ("injection", "xss", "ssti", "ssrf", "command_injection",
   467→                         "path_traversal", "unsafe_deserialization", "dangerous_eval"):
   468→                title = f.get("title", "")
   469→                if title not in seen_titles:
   470→                    timeline.append({
   471→                        "phase": 3,
   472→                        "phase_name": "Exploitation",
   473→                        "finding": title,
   474→                        "severity": f.get("severity", "info"),
   475→                        "description": f.get("description", f.get("evidence", ""))[:200],
   476→                    })
   477→                    seen_titles.add(title)
   478→
   479→        # Phase 4: Persistence / crypto / secrets.
   480→        for f in findings:
   481→            cat = str(f.get("category", "")).lower()
   482→            if cat in ("secrets", "crypto", "outdated", "dependency",
   483→                         "misconfiguration", "container", "iac"):
   484→                title = f.get("title", "")
   485→                if title not in seen_titles:
   486→                    timeline.append({
   487→                        "phase": 4,
   488→                        "phase_name": "Persistence & Infrastructure",
   489→                        "finding": title,
   490→                        "severity": f.get("severity", "info"),
   491→                        "description": f.get("description", f.get("evidence", ""))[:200],
   492→                    })
   493→                    seen_titles.add(title)
   494→
   495→        # Append evidence chain correlations if available.
   496→        for chain in chains:
   497→            chain_title = chain.get("title", chain.get("finding", ""))
   498→            if chain_title and chain_title not in seen_titles:
   499→                timeline.append({
   500→                    "phase": 5,
   501→                    "phase_name": "Corroborated Intelligence",
   502→                    "finding": chain_title,
   503→                    "severity": chain.get("severity", "info"),
   504→                    "description": chain.get("summary", chain.get("description", ""))[:200],
   505→                    "corroboration": chain.get("corroboration_count", chain.get("sources", 1)),
   506→                })
   507→                seen_titles.add(chain_title)
   508→
   509→        return timeline

   511→    # ------------------------------------------------------------------
   512→    #  Top findings
   513→    # ------------------------------------------------------------------
   514→
   515→    def _prioritize_findings(
   516→        self,
   517→        findings: List[Dict[str, Any]],
   518→        confidence_scores: List[Dict[str, Any]],
   519→    ) -> List[Dict[str, Any]]:
   520→        """Rank findings by severity_weight * confidence, take top 10."""
   521→        conf_map: Dict[str, float] = {}
   522→        for cs in confidence_scores:
   523→            title = cs.get("title", "")
   524→            conf_val = cs.get("confidence", 0.5)
   525→            conf_map[title] = float(conf_val) if isinstance(conf_val, (int, float)) else 0.5
   526→
   527→        ranked: List[Dict[str, Any]] = []
   528→        for f in findings:
   529→            title = f.get("title", "")
   530→            severity = str(f.get("severity", "info")).strip().lower()
   531→            sev_weight = _SEVERITY_WEIGHT.get(severity, 0.0)
   532→            confidence = conf_map.get(title, self._default_likelihood(severity))
   533→            priority = sev_weight * confidence
   534→
   535→            ranked.append({
   536→                "title": title,
   537→                "severity": severity,
   538→                "category": f.get("category", ""),
   539→                "confidence": round(confidence, 3),
   540→                "priority_score": round(priority, 3),
   541→                "description": f.get("description", "")[:300],
   542→                "remediation": f.get("remediation", ""),
   543→                "evidence": str(f.get("evidence", ""))[:200],
   544→            })
   545→
   546→        ranked.sort(key=lambda x: x["priority_score"], reverse=True)
   547→        return ranked[:10]

   549→    # ------------------------------------------------------------------
   550→    #  Remediation plan
   551→    # ------------------------------------------------------------------
   552→
   553→    def _build_remediation_plan(
   554→        self,
   555→        findings: List[Dict[str, Any]],
   556→        eng_score: Dict[str, Any],
   557→    ) -> List[Dict[str, Any]]:
   558→        """Build prioritized remediation plan grouped by category."""
   559→        if not findings:
   560→            return []
   561→
   562→        # Group findings by category.
   563→        grouped: Dict[str, List[Dict[str, Any]]] = defaultdict(list)
   564→        for f in findings:
   565→            cat = str(f.get("category", "unknown")).strip().lower()
   566→            grouped[cat].append(f)
   567→
   568→        # Try to use the recommendation engine for detailed steps.
   569→        try:
   570→            from .recommendation_engine import RecommendationEngine
   571→            engine = RecommendationEngine()
   572→            rec_report = engine.recommend(findings)
   573→            rec_map: Dict[str, Dict[str, Any]] = {}
   574→            for cs in rec_report.category_summary.values():
   575→                rec_map[cs.get("category", "").lower()] = cs
   576→        except Exception:
   577→            rec_map = {}
   578→
   579→        # Severity ordering.
   580→        sev_order = {"critical": 0, "high": 1, "medium": 2, "low": 3, "info": 4}
   581→
   582→        # Build plan items sorted by max severity within each category.
   583→        plan: List[Dict[str, Any]] = []
   584→        for cat, cat_findings in grouped.items():
   585→            max_sev = max(
   586→                cat_findings,
   587→                key=lambda f: sev_order.get(str(f.get("severity", "info")).lower(), 4),
   588→            )
   589→            severity = str(max_sev.get("severity", "info")).strip().lower()
   590→
   591→            # Get remediation details from recommendation engine or template.
   592→            rec_info = rec_map.get(cat, {})
   593→            steps = rec_info.get("steps", [])
   594→            summary = rec_info.get("summary", _REMEDIATION_TEMPLATES.get(
   595→                cat, f"Review and remediate {cat} findings."
   596→            ))
   597→            effort = rec_info.get("effort", "medium")
   598→
   599→            if not steps:
   600→                # Fallback: generate steps from template.
   601→                template_text = _REMEDIATION_TEMPLATES.get(cat, "")
   602→                if template_text:
   603→                    steps = [template_text]
   604→                else:
   605→                    steps = [f"Investigate and resolve {cat} issues identified in the scan."]
   606→
   607→            plan.append({
   608→                "category": cat,
   609→                "category_label": _CATEGORY_LABELS.get(cat, cat),
   610→                "severity": severity,
   611→                "findings_count": len(cat_findings),
   612→                "summary": summary,
   613→                "steps": steps,
   614→                "effort": effort,
   615→                "finding_titles": [f.get("title", "") for f in cat_findings],
   616→            })
   617→
   618→        # Sort: critical first, then high, medium, low, info.
   619→        plan.sort(key=lambda x: sev_order.get(x["severity"], 4))
   620→        return plan

   622→    # ------------------------------------------------------------------
   623→    #  Evidence chains

   626→    def _extract_evidence_chains(
   627→        self,
   628→        chains: List[Dict[str, Any]],
   629→        findings: List[Dict[str, Any]],
   630→    ) -> List[Dict[str, Any]]:
   631→        """Extract corroborated evidence chains for the report."""
   632→        if chains:
   633→            return [
   634→                {
   635→                    "finding": c.get("title", c.get("finding", "")),
   636→                    "severity": c.get("severity", "info"),
   637→                    "sources": c.get("sources", c.get("corroboration_count", 1)),
   638→                    "confidence": c.get("confidence", c.get("score", 0.5)),
   639→                    "summary": c.get("summary", c.get("description", ""))[:300],
   640→                }
   641→                for c in chains
   642→                if c.get("title") or c.get("finding")
   643→            ]
   644→
   645→        # Fallback: build pseudo-chains from findings with high corroboration.
   646→        result: List[Dict[str, Any]] = []
   647→        for f in findings:
   648→            if f.get("corroboration_count", 0) > 1 or f.get("confidence", 0) >= 0.8:
   649→                result.append({
   650→                    "finding": f.get("title", ""),
   651→                    "severity": f.get("severity", "info"),
   652→                    "sources": f.get("corroboration_count", 1),
   653→                    "confidence": f.get("confidence", 0.5),
   654→                    "summary": f.get("description", "")[:300],
   655→                })
   656→        return result
