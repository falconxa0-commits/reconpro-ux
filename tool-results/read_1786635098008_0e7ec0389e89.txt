   200→   529→   529→  // RSA
   201→   530→   530→  if (algo.includes('RSA')) {
   202→   531→   531→    if (keySize >= 8192) return 'rsa_8192';
   203→   532→   532→    if (keySize >= 4096) return 'rsa_4096';
   204→   533→   533→    if (keySize >= 2048) return 'rsa_2048';
   205→   534→   534→    if (keySize >= 1024) return 'rsa_1024';
   206→   535→   535→    return 'rsa_2048'; // default assumption
   207→   536→   536→  }
   208→   537→   537→
   209→   538→   538→  // Elliptic Curve
   210→   539→   539→  if (algo.includes('ECDH') || algo.includes('ECDHE') || algo.includes('X25519') || algo.includes('X448')) {
   211→   540→   540→    if (algo.includes('P-384') || algo.includes('SECP384') || keySize >= 384) return 'ec_p384';
   212→   541→   541→    if (algo.includes('P-521') || algo.includes('SECP521') || keySize >= 521) return 'ec_p521';
   213→   542→   542→    if (algo.includes('25519')) return 'ec_curve25519';
   214→   543→   543→    if (keySize >= 384) return 'ec_p384';
   215→   544→   544→    return 'ec_p256'; // default assumption
   216→   545→   545→  }
   217→   546→   546→
   218→   547→   547→  if (algo.includes('ECDSA') || algo.includes('ED25519') || algo.includes('EDDSA')) {
   219→   548→   548→    if (algo.includes('P-384') || algo.includes('SECP384') || keySize >= 384) return 'ec_p384';
   220→   549→   549→    if (algo.includes('P-521') || algo.includes('SECP521') || keySize >= 521) return 'ec_p521';
   221→   550→   550→    if (algo.includes('25519')) return 'ec_ed25519';
   222→   551→   551→    if (keySize >= 384) return 'ec_p384';
   223→   552→   552→    return 'ec_p256';
   224→   553→   553→  }
   225→   554→   554→
   226→   555→   555→  // AES / Symmetric
   227→   556→   556→  if (algo.includes('AES')) {
   228→   557→   557→    if (keySize >= 256) return 'aes_256';
   229→   558→   558→    return 'aes_128';
   230→   559→   559→  }
   231→   560→   560→
   232→   561→   561→  if (algo.includes('3DES') || algo.includes('DES')) {
   233→   562→   562→    return 'aes_128'; // 3DES effectively worse than AES-128
   234→   563→   563→  }
   235→   564→   564→
   236→   565→   565→  if (algo.includes('SHA')) {
   237→   566→   566→    if (algo.includes('384') || keySize >= 384) return 'sha_384';
   238→   567→   567→    return 'sha_256';
   239→   568→   568→  }
   240→   569→   569→
   241→   570→   570→  // Default fallback — assume RSA-2048 as most common
   242→   571→   571→  return 'rsa_2048';
   243→   572→   572→}

   329→   658→   658→/**

   336→   665→   665→  companyName?: string

   339→   668→   668→  const nonPQCAssets = assets.filter(a => a.algorithmFamily !== 'pqc');
   340→   669→   669→
   341→   670→   670→  if (nonPQCAssets.length === 0) {
   342→   671→   671→    return {
   343→   672→   672→      score: 0,
   344→   673→   673→      description: 'All assets use post-quantum cryptography. No HNDL risk detected.',

   357→   686→   686→  // Score based on capture window, number of assets, and data sensitivity
   358→   687→   687→  let baseScore = 0;
   359→   688→   688→

   362→   691→   691→  else if (captureWindow <= 5) baseScore += 25;

   366→   695→   695→  // Asset count component

   368→   697→   697→
   369→   698→   698→  // Industry sensitivity component

   371→   700→   700→    finance: 1.3,

   391→   720→   720→  // Regulatory exposure

   393→   722→   722→
   394→   723→   723→  // Generate description

   396→   725→   725→    80: `CRITICAL HNDL THREAT: ${companyName ? `${companyName}'s` : 'Your'} encrypted traffic is actively being captured by state-sponsored adversaries. With quantum decryption expected within ${captureWindow} years, intercepted data will be fully readable. Immediate PQC migration is non-negotiable.`,
   397→   726→   726→    60: `SEVERE HNDL RISK: ${companyName ? `${companyName}'s` : 'Your'} encrypted communications are vulnerable to "store now, decrypt later" attacks. Nation-state threat actors routinely bulk-capture TLS traffic. With ${captureWindow} years until quantum decryption, time is running out.`,
   398→   727→   727→    40: `MODERATE HNDL EXPOSURE: ${companyName ? `${companyName}` : 'Your organization'} has ${nonPQCAssets.length} assets using quantum-vulnerable cryptography. Adversaries may be stockpiling encrypted data for future quantum decryption within ${captureWindow} years.`,

   401→   730→   730→  };

   414→   743→   743→    captureWindowYears: Math.max(0, captureWindow),
   415→   744→   744→    estimatedRecords,
   416→   745→   745→    regulatoryExposure,
   417→   746→   746→  };

   492→   821→   821→      (asset.algorithmFamily === 'ec' && asset.keyExchange.includes('ECDH'))
   493→   822→   822→    ) {

   495→   824→   824→    } else if (asset.algorithmFamily === 'ec') {

   517→   846→   846→      urgency: migration.urgency,
   518→   847→   847→    });

   521→   850→   850→  // Digital signature migrations

   523→   852→   852→    const migration = PQC_MIGRATIONS.signatures;

   531→   860→   860→      costEstimate: '$75K-$250K',
   532→   861→   861→      category: 'Digital Signatures',
   533→   862→   862→      urgency: migration.urgency,
   534→   863→   863→    });
   535→   864→   864→  }

   539→   868→   868→    const migration = PQC_MIGRATIONS.symmetric_upgrade;
   540→   869→   869→    steps.push({
   541→   870→   870→      priority: priority++,

   545→   874→   874→      nistLevel: migration.nistLevel,

   551→   880→   880→  }
   552→   881→   881→
   553→   882→   882→  // Sort by priority (earliest break year first within same category)

   557→   886→   886→    if (aAsset && bAsset) {
   558→   887→   887→      return aAsset.estimatedBreakYear - bAsset.estimatedBreakYear;
   559→   888→   888→    }
   560→   889→   889→    return a.priority - b.priority;
   561→   890→   890→  });
   562→   891→   891→
   563→   892→   892→  // Re-number priorities after sort
   564→   893→   893→  steps.forEach((step, i) => {
   565→   894→   894→    step.priority = i + 1;
   566→   895→   895→  });
   567→   896→   896→
   568→   897→   897→  return steps;
   569→   898→   898→}

   654→   983→   983→function getAlgorithmDisplayName(
   655→   984→   984→  keyExchange: string,
   656→   985→   985→  cipherSuite: string,
   657→   986→   986→  keySize: number
   658→   987→   987→): string {
