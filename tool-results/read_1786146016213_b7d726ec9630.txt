   109→   109→class ToolRegistry:
   110→   110→    """Registry of all available tools."""
   111→   111→
   112→   112→    def __init__(self):
   113→   113→        self.tools: Dict[str, Tool] = {}
   114→   114→
   115→   115→    def register(self, tool: Tool) -> None:
   116→   116→        self.tools[tool.name] = tool
   117→   117→
   118→   118→    def get(self, name: str) -> Optional[Tool]:
   119→   119→        return self.tools.get(name)
   120→   120→
   121→   121→    def execute(self, name: str, **kwargs) -> ToolResult:
   122→   122→        tool = self.tools.get(name)
   123→   123→        if not tool:
   124→   124→            return ToolResult(f"Tool '{name}' not found.", success=False)
   125→   125→        try:
   126→   126→            return tool.execute(**kwargs)
   127→   127→        except Exception as e:
   128→   128→            return ToolResult(f"Tool error: {e}", success=False)
   129→   129→
   130→   130→    def list_tools(self) -> List[Dict]:
   131→   131→        return [
   132→   132→            {"name": t.name, "description": t.description, "category": t.category}
   133→   133→            for t in self.tools.values()
   134→   134→        ]
   135→   135→

   139→   139→    def to_anthropic_tools(self) -> List[Dict]:
   140→   140→        return [t.to_anthropic_tool() for t in self.tools.values()]

   148→   148→class AgentMemory:
   149→   149→    """Persistent memory for the Nexus agent across sessions."""
   150→   150→
   151→   151→    def __init__(self):
   152→   152→        MEMORY_DIR.mkdir(parents=True, exist_ok=True)
   153→   153→        self._data = self._load()

   155→   155→    def _load(self) -> Dict:
   156→   156→        if CONTEXT_FILE.exists():
   157→   157→            try:
   158→   158→                return json.loads(CONTEXT_FILE.read_text())
   159→   159→            except Exception:
   160→   160→                pass
   161→   161→        return {
   162→   162→            "sessions": [],
   163→   163→            "target_contexts": {},
   164→   164→            "learned_patterns": [],
   165→   165→            "created": datetime.now(timezone.utc).isoformat(),
   166→   166→        }
   167→   167→
   168→   168→    def _save(self) -> None:
   169→   169→        # Keep only the last 50 sessions
   170→   170→        if len(self._data.get("sessions", [])) > 50:
   171→   171→            self._data["sessions"] = self._data["sessions"][-50:]
   172→   172→        try:
   173→   173→            CONTEXT_FILE.write_text(json.dumps(self._data, indent=2, default=str))
   174→   174→        except OSError:
   175→   175→            pass
   176→   176→
   177→   177→    def save_session(
   178→   178→        self,
   179→   179→        goal: str,
   180→   180→        target: str,
   181→   181→        plan: List[Dict],
   182→   182→        tool_results: List[Dict],
   183→   183→        summary: str,
   184→   184→    ) -> None:
   185→   185→        session = {
   186→   186→            "goal": goal,
   187→   187→            "target": target,
   188→   188→            "plan": plan,
   189→   189→            "tool_results": tool_results,
   190→   190→            "summary": summary,
   191→   191→            "timestamp": datetime.now(timezone.utc).isoformat(),
   192→   192→        }
   193→   193→        self._data.setdefault("sessions", []).append(session)
   194→   194→        # Update target context
   195→   195→        self._data.setdefault("target_contexts", {})[target] = {
   196→   196→            "last_score": self._extract_score(tool_results),
   197→   197→            "last_grade": self._extract_grade(tool_results),
   198→   198→            "last_run": session["timestamp"],
   199→   199→            "finding_count": self._extract_finding_count(tool_results),
   200→   200→            "known_issues": self._extract_issues(tool_results),
   201→   201→        }
   202→   202→        self._save()
   203→   203→
   204→   204→    def get_target_context(self, target: str) -> Optional[Dict]:
   205→   205→        return self._data.get("target_contexts", {}).get(target)
   206→   206→
   207→   207→    def get_recent_sessions(self, limit: int = 5) -> List[Dict]:
   208→   208→        return self._data.get("sessions", [])[-limit:]

   210→   210→    @staticmethod
   211→   211→    def _extract_score(results: List[Dict]) -> Optional[int]:
   212→   212→        for r in results:
   213→   213→            d = r.get("data")
   214→   214→            if isinstance(d, dict) and "total_score" in d:
   215→   215→                return d["total_score"]
   216→   216→        return None
   217→   217→
   218→   218→    @staticmethod
   219→   219→    def _extract_grade(results: List[Dict]) -> Optional[str]:
   220→   220→        for r in results:
   221→   221→            d = r.get("data")
   222→   222→            if isinstance(d, dict) and "grade" in d:
   223→   223→                return d["grade"]
   224→   224→        return None
   225→   225→
   226→   226→    @staticmethod
   227→   227→    def _extract_finding_count(results: List[Dict]) -> int:
   228→   228→        total = 0
   229→   229→        for r in results:
   230→   230→            d = r.get("data")
   231→   231→            if isinstance(d, dict) and "findings" in d:
   232→   232→                total += len(d["findings"])
   233→   233→        return total
   234→   234→
   235→   235→    @staticmethod
   236→   236→    def _extract_issues(results: List[Dict]) -> List[str]:
   237→   237→        issues = []
   238→   238→        for r in results:
   239→   239→            d = r.get("data")
   240→   240→            if isinstance(d, dict) and "findings" in d:
   241→   241→                for f in d["findings"]:
   242→   242→                    if isinstance(f, dict) and f.get("severity") in ("critical", "high"):
   243→   243→                        issues.append(f.get("title", ""))
   244→   244→        return issues[:20]

   465→   465→    return ToolResult(

   889→   889→        modifier = 1.0
   890→   890→        for w in words:
   891→   891→            if w in self._AMPLIFIERS:
   892→   892→                modifier = max(modifier, self._AMPLIFIERS[w])
   893→   893→            if w in self._DAMPENERS:
   894→   894→                modifier = min(modifier, self._DAMPENERS[w])
   895→   895→
   896→   896→        for intent, keywords in self._INTENT_KEYWORDS.items():
   897→   897→            score = 0.0
   898→   898→            for pattern, weight in keywords:
   899→   899→                # Use whole-word matching to avoid false positives
   900→   900→                # (e.g., "repo" matching inside "report")
   901→   901→                if len(pattern.split()) > 1:
   902→   902→                    # Multi-word pattern: check substring (already specific enough)
   903→   903→                    matched = pattern in goal_lower
   904→   904→                else:
   905→   905→                    # Single-word pattern: check whole word with padding
   906→   906→                    matched = f' {pattern} ' in padded
   907→   907→                if matched:
   908→   908→                    # Boost if near a target-like word
   909→   909→                    proximity_boost = 1.0
   910→   910→                    pattern_idx = goal_lower.find(pattern)
   911→   911→                    for tw in (".com", ".io", ".net", ".org", "http", "localhost"):
   912→   912→                        tw_idx = goal_lower.find(tw)
   913→   913→                        if tw_idx >= 0 and abs(pattern_idx - tw_idx) < 40:
   914→   914→                            proximity_boost = 1.3
   915→   915→                            break
   916→   916→                    score += weight * proximity_boost
   917→   917→            scores[intent] = min(score * modifier, 2.0)
   918→   918→
   919→   919→        return scores

   921→   921→    def extract_targets(self, goal: str) -> List[str]:
   922→   922→        """Extract targets (domains, IPs, URLs, paths) from the goal."""
   923→   923→        urls = re.findall(r'https?://[\w\-._~:/?#\[\]@!$&\'()*+,;=%]+', goal)

   925→   925→        skip = {
   926→   926→            'the', 'and', 'for', 'with', 'all', 'of', 'to', 'from', 'that', 'this',
   927→   927→            'scan', 'audit', 'check', 'find', 'get', 'show', 'me', 'my',
   928→   928→        }
   929→   929→        targets = list(set(
   930→   930→            urls + [d for d in domains if d.lower() not in skip and len(d) > 4]
   931→   931→        ))
   932→   932→        return targets

   934→   934→    def detect_target_type(self, goal: str, targets: List[str]) -> str:
   935→   935→        """Classify the target type from context."""
   936→   936→        goal_lower = goal.lower()
   937→   937→        local_indicators = [
   938→   938→            'machine', 'laptop', 'computer', 'local', 'my system', 'this machine',
   939→   939→            'localhost', 'my mac', 'my pc', 'my server',
   940→   940→        ]
   941→   941→        for ind in local_indicators:
   942→   942→            if ind in goal_lower:
   943→   943→                return "local"
   944→   944→        dev_indicators = ['project', 'codebase', 'repo', 'source code', 'code review']
   945→   945→        padded = f' {goal_lower} '
   946→   946→        for ind in dev_indicators:
   947→   947→            if f' {ind} ' in padded and not targets:
   948→   948→                return "codebase"
   949→   949→        if targets:
   950→   950→            for t in targets:
   951→   951→                if t.startswith("http"):
   952→   952→                    return "remote"
   953→   953→                if re.match(r'^\d+\.\d+\.\d+\.\d+$', t):
   954→   954→                    return "remote"
   955→   955→            return "remote"
   956→   956→        return "unknown"

   958→   958→    def detect_modules(self, goal: str) -> Optional[List[str]]:

   962→   962→            for kw in keywords:
   963→   963→                if kw in goal_lower:

  1131→  1131→    def adapt_plan(
  1132→  1132→        self,
  1133→  1133→        plan: List[Dict[str, Any]],
  1134→  1134→        completed_results: List[Tuple[Dict, ToolResult]],
  1135→  1135→        current_idx: int,
  1136→  1136→    ) -> List[Dict[str, Any]]:
  1137→  1137→        """Adapt the remaining plan based on results seen so far.
  1138→  1138→
  1139→  1139→        This is the 'intelligence' of the rule engine — it inspects intermediate
  1140→  1140→        results and inserts new steps or modifies remaining ones.
  1141→  1141→        """
  1142→  1142→        additions: List[Dict[str, Any]] = []
  1143→  1143→        domain = ""
  1144→  1144→        last_score = None
  1145→  1145→        has_critical = False
  1146→  1146→        total_findings = 0
  1147→  1147→        all_findings: List[Dict] = []
  1148→  1148→        subdomains_found = False
  1149→  1149→
  1150→  1150→        for step, result in completed_results:
  1151→  1151→            d = result.data
  1152→  1152→            if isinstance(d, dict):
  1153→  1153→                # Extract score from scan results
  1154→  1154→                if "total_score" in d:
  1155→  1155→                    last_score = d["total_score"]
  1156→  1156→                if "vibesec_score" in d:
  1157→  1157→                    last_score = d["vibesec_score"]
  1158→  1158→                # Collect findings
  1159→  1159→                findings = d.get("findings", [])
  1160→  1160→                if isinstance(findings, list):
  1161→  1161→                    all_findings.extend(findings)
  1162→  1162→                    total_findings += len(findings)
  1163→  1163→                    for f in findings:
  1164→  1164→                        if isinstance(f, dict) and f.get("severity") in ("critical", "high"):
  1165→  1165→                            has_critical = True
  1166→  1166→                # Check for subdomains

  1168→  1168→                    subdomains_found = True
  1169→  1169→                # Get target from data
  1170→  1170→                if not domain:
  1171→  1171→                    domain = d.get("target", "")
  1172→  1172→
  1173→  1173→        # If any findings found → add CVE enrichment (if not already planned)
  1174→  1174→        has_any_findings = total_findings > 0
  1175→  1175→        if has_any_findings and not any(s["tool"] == "cve_enrich" for s in plan[current_idx:]):
  1176→  1176→            if all_findings:
  1177→  1177→                try:
  1178→  1178→                    findings_json = json.dumps(all_findings[:20])
  1179→  1179→                except (TypeError, ValueError):
  1180→  1180→                    findings_json = "[]"

  1188→  1188→                })
  1189→  1189→
  1190→  1190→        # If many subdomains found and not already doing blitz → suggest blitz
  1191→  1191→        if subdomains_found and not any(s["tool"] == "blitz_scan" for s in plan[current_idx:]):
  1192→  1192→            # Don't add blitz automatically (too aggressive), but add knowledge graph

  1194→  1194→                additions.append({
  1195→  1195→                    "tool": "knowledge_graph",
  1196→  1196→                    "args": {"action": "build", "target": domain},
  1197→  1197→                    "reason": "Large attack surface detected — building knowledge graph",
  1198→  1198→                })

  1202→  1202→            if domain and not any(s["tool"] == "knowledge_graph" for s in plan[current_idx:]):
  1203→  1203→                additions.append({
  1204→  1204→                    "tool": "knowledge_graph",

  1207→  1207→                })
  1208→  1208→

  1210→  1210→        if last_score is not None and last_score < 50:

  1364→  1364→    def _execute_llm(self, goal: str, backend: str) -> Dict[str, Any]:
  1365→  1365→        """Execute using LLM tool-calling loop."""
  1366→  1366→        tool_descs = "\n".join(
  1367→  1367→            f"  - {t.name}: {t.description}" for t in self.registry.tools.values()
  1368→  1368→        )
  1369→  1369→        prior = ""
  1370→  1370→        for session in self.memory.get_recent_sessions(limit=3):
  1371→  1371→            prior += f"  - {session.get('target', '?')}: {session.get('summary', '')[:100]}\n"
  1372→  1372→        context_summary = f"Recent sessions:\n{prior}" if prior else "No prior sessions."
  1373→  1373→
  1374→  1374→        system_prompt = _build_system_prompt(tool_descs, context_summary)
  1375→  1375→        messages = [{"role": "system", "content": system_prompt}]
  1376→  1376→        messages.append({"role": "user", "content": goal})
  1377→  1377→
  1378→  1378→        self._print_header(goal, "LLM reasoning")
  1379→  1379→
  1380→  1380→        for iteration in range(MAX_ITERATIONS):
  1381→  1381→            self._emit_status(f"LLM reasoning (step {iteration + 1})...")
  1382→  1382→
  1383→  1383→            try:
  1384→  1384→                if backend == "openai":
  1385→  1385→                    response = self._call_openai(messages)
  1386→  1386→                else:
  1387→  1387→                    response = self._call_anthropic(messages)
  1388→  1388→            except Exception as e:
  1389→  1389→                self._emit(f"[red]LLM error: {e} — falling back to rule engine.[/]")
  1390→  1390→                return self._execute_rules(goal)
  1391→  1391→
  1392→  1392→            if not response:
  1393→  1393→                self._emit("[yellow]LLM returned empty response — falling back to rules.[/]")
  1394→  1394→                return self._execute_rules(goal)
  1395→  1395→
  1396→  1396→            content = response.get("content", "")
  1397→  1397→            tool_calls = response.get("tool_calls", [])
  1398→  1398→
  1399→  1399→            # If no tool calls, the LLM is done
  1400→  1400→            if not tool_calls:
  1401→  1401→                if content:
  1402→  1402→                    self._emit(f"  [dim]{content[:500]}[/]")
  1403→  1403→                break
  1404→  1404→
  1405→  1405→            # Execute each tool call
  1406→  1406→            for tc in tool_calls:
  1407→  1407→                fn_name = tc.get("function", {}).get("name", "")
  1408→  1408→                fn_args_str = tc.get("function", {}).get("arguments", "{}")
  1409→  1409→                try:
  1410→  1410→                    fn_args = json.loads(fn_args_str) if isinstance(fn_args_str, str) else fn_args_str
  1411→  1411→                except json.JSONDecodeError:
  1412→  1412→                    fn_args = {}
  1413→  1413→
  1414→  1414→                self._emit(f"  [bold]{fn_name}[/]({json.dumps(fn_args)[:100]})")
  1415→  1415→                self._emit_status(f"Executing: {fn_name}")
  1416→  1416→
  1417→  1417→                result = self.registry.execute(fn_name, **fn_args)
  1418→  1418→                success_mark = "[green]✓[/]" if result.success else "[red]✗[/]"
  1419→  1419→                self._emit(f"    {success_mark} {result.output[:200]}")
  1420→  1420→
  1421→  1421→                self._steps.append(fn_name)
  1422→  1422→                self._collect_result(fn_name, result)
  1423→  1423→
  1424→  1424→                # Feed result back to LLM
  1425→  1425→                if backend == "openai":
  1426→  1426→                    messages.append({"role": "assistant", "content": None, "tool_calls": [tc]})
  1427→  1427→                    messages.append({
  1428→  1428→                        "role": "tool",
  1429→  1429→                        "tool_call_id": tc.get("id", f"call_{fn_name}"),
  1430→  1430→                        "content": result.output[:4000],
  1431→  1431→                    })
  1432→  1432→                else:
  1433→  1433→                    messages.append({
  1434→  1434→                        "role": "user",
  1435→  1435→                        "content": f"Tool {fn_name} result: {result.output[:3000]}",
  1436→  1436→                    })
  1437→  1437→
  1438→  1438→        summary = self._build_summary()
  1439→  1439→        self._print_footer(summary)
  1440→  1440→        return {
  1441→  1441→            "goal": goal,
  1442→  1442→            "steps": self._steps,
  1443→  1443→            "targets": self._targets,
  1444→  1444→            "total_findings": len(self._findings),
  1445→  1445→            "findings": self._findings,

  1516→  1516→    def _execute_rules(self, goal: str) -> Dict[str, Any]:
  1517→  1517→        """Execute using the intelligent rule-based planner."""
  1518→  1518→        # THINK phase
  1519→  1519→        scores = self.planner.classify_intent(goal)
  1520→  1520→        targets = self.planner.extract_targets(goal)
  1521→  1521→        target_type = self.planner.detect_target_type(goal, targets)
  1522→  1522→
  1523→  1523→        domain = ""
  1524→  1524→        if targets:
  1525→  1525→            domain = targets[0].replace("https://", "").replace("http://", "").split("/")[0]
  1526→  1526→
  1527→  1527→        # Build intent display
  1528→  1528→        top_intents = sorted(scores.items(), key=lambda x: x[1], reverse=True)[:4]
  1529→  1529→        intent_str = " + ".join(f"{n} ({s:.1f})" for n, s in top_intents if s > 0.1)
  1530→  1530→
  1531→  1531→        self._print_header(goal, "rule-based planner")
  1532→  1532→        self._emit(f"  [THINK] Analyzing goal...")
  1533→  1533→        self._emit(f"    [dim]→ Target: {domain or target_type} ({target_type})[/]")
  1534→  1534→        if intent_str:
  1535→  1535→            self._emit(f"    [dim]→ Intent: {intent_str}[/]")
  1536→  1536→        self._emit(f"    [dim]→ Strategy: plan → execute → adapt[/]")
  1537→  1537→
  1538→  1538→        # PLAN phase
  1539→  1539→        plan = self.planner.generate_plan(goal, scores, targets, target_type)
  1540→  1540→        self._emit(f"")
  1541→  1541→        self._emit(f"  [PLAN] {len(plan)} step(s):")
  1542→  1542→        for i, step in enumerate(plan, 1):
  1543→  1543→            self._emit(f"    {i}. [bold]{step['tool']}[/] — {step['reason']}")
  1544→  1544→
  1545→  1545→        # EXECUTE phase with adaptation
  1546→  1546→        completed: List[Tuple[Dict, ToolResult]] = []
  1547→  1547→        step_idx = 0
  1548→  1548→        total_steps = len(plan)
  1549→  1549→
  1550→  1550→        while step_idx < total_steps and step_idx < MAX_ITERATIONS:
  1551→  1551→            step = plan[step_idx]
  1552→  1552→            tool_name = step["tool"]
  1553→  1553→            args = step["args"]
  1554→  1554→            current_display = step_idx + 1
  1555→  1555→
  1556→  1556→            self._emit(f"")
  1557→  1557→            self._emit(f"  [EXEC] Step {current_display}/{total_steps}: [bold]{tool_name}[/]...")
  1558→  1558→            self._emit_status(f"Executing step {current_display}/{total_steps}: {tool_name}")
  1559→  1559→
  1560→  1560→            result = self.registry.execute(tool_name, **args)
  1561→  1561→            success_mark = "[green]✓[/]" if result.success else "[red]✗[/]"
  1562→  1562→            self._emit(f"    {success_mark} {result.output[:300]}")
  1563→  1563→
  1564→  1564→            self._steps.append(tool_name)
  1565→  1565→            self._collect_result(tool_name, result)
  1566→  1566→            completed.append((step, result))
  1567→  1567→
  1568→  1568→            # Track targets
  1569→  1569→            if tool_name in ("remote_scan", "vibesec_benchmark", "subdomain_discover",
  1570→  1570→                             "adversarial_loop", "swarm_run"):
  1571→  1571→                if domain:
  1572→  1572→                    self._targets.append(domain)
  1573→  1573→            if tool_name == "blitz_scan" and result.data:
  1574→  1574→                for t in result.data.get("results", {}):
  1575→  1575→                    self._targets.append(t)
  1576→  1576→
  1577→  1577→            # VERIFY & ADAPT phase (after each step)
  1578→  1578→            old_len = len(plan)
  1579→  1579→            plan = self.planner.adapt_plan(plan, completed, step_idx)
  1580→  1580→            new_steps = len(plan) - old_len
  1581→  1581→            if new_steps > 0:
  1582→  1582→                total_steps = len(plan)
  1583→  1583→                added = plan[step_idx + 1: step_idx + 1 + new_steps]
  1584→  1584→                self._emit(f"")
  1585→  1585→                self._emit(f"  [ADAPT] Added {new_steps} step(s) based on results:")
  1586→  1586→                for a in added:
  1587→  1587→                    self._emit(f"    + [bold]{a['tool']}[/] — {a['reason']}")
  1588→  1588→
  1589→  1589→            step_idx += 1
  1590→  1590→
  1591→  1591→        summary = self._build_summary()
  1592→  1592→        self._print_footer(summary)
  1593→  1593→        return {
  1594→  1594→            "goal": goal,
  1595→  1595→            "steps": self._steps,
  1596→  1596→            "targets": self._targets,
  1597→  1597→            "total_findings": len(self._findings),
  1598→  1598→            "findings": self._findings,
  1599→  1599→            "summary": summary,
  1600→  1600→            "backend": "rules",
  1601→  1601→        }

  1619→  1619→                        for f in td.get("findings", []):

  1621→  1621→                                self._emit_finding(f)

  1676→  1676→def run_nexus_agent(goal: str, verbose: bool = True) -> dict:
  1677→  1677→    """Run the Nexus Agent on a goal.
  1678→  1678→
  1679→  1679→    This is the main entry point for the agent system. It creates an agent,
  1680→  1680→    executes the goal, and returns a results dictionary.
  1681→  1681→
  1682→  1682→    Args:
  1683→  1683→        goal: Natural language goal (e.g., "Fully recon example.com and find all vulnerabilities").
  1684→  1684→        verbose: If True, print rich output. If False, run silently.
