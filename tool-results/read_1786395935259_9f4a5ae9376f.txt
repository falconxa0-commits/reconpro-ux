  2531→    def _calculate_risk_score(
  2532→        self,
  2533→        ttp_results: List[APTAttribution],
  2534→        campaign_results: List[CampaignMatch],
  2535→        sovereignty: SovereigntyProfile,
  2536→    ) -> float:
  2537→        """Calculate an overall risk score (0–100) for the target."""
  2538→        risk = 0.0
  2539→
  2540→        # TTP risk component (0–40)
  2541→        if ttp_results:
  2542→            top_conf = ttp_results[0].confidence
  2543→            risk += top_conf * 40
  2544→            if len(ttp_results) > 1:
  2545→                risk += min(len(ttp_results) - 1, 5) * 2  # up to 10
  2546→
  2547→        # Campaign risk component (0–25)
  2548→        if campaign_results:
  2549→            top_campaign = campaign_results[0].confidence
  2550→            risk += top_campaign * 25
  2551→
  2552→        # Sovereignty risk component (0–25)
  2553→        risk += sovereignty.confidence * 25
  2554→
  2555→        # Sophistication bonus (0–10)
  2556→        if ttp_results:
  2557→            soph = ttp_results[0].sophistication
  2558→            soph_bonus = {
  2559→                "minimal": 1, "moderate": 3,
  2560→                "high": 6, "very_high": 10,
  2561→            }
  2562→            risk += soph_bonus.get(soph, 3)
  2563→
  2564→        return round(min(risk, 100.0), 2)

  2613→    def _passive_recon(
  2614→        self, url: str, timeout: int,
  2615→    ) -> Dict[str, Any]:
  2616→        """Attempt a lightweight HTTP fetch to gather passive signals
  2617→        (headers, WAF banners, etc.) without being intrusive.
  2618→
  2619→        Returns a dict with ``"headers"``, ``"body"``, ``"tls"``, ``"geo"``,
  2620→        ``"waf"`` fields. Never raises — returns partial data on failure.
  2621→        """
  2622→        result: Dict[str, Any] = {
  2623→            "headers": {},
  2624→            "body": "",
  2625→            "tls": {},
  2626→            "geo": {},
  2627→            "waf": {},
  2628→        }
  2629→
  2630→        try:
  2631→            parsed = urllib.parse.urlparse(url)
  2632→            hostname = parsed.hostname or ""
  2633→            port = parsed.port or (443 if parsed.scheme == "https" else 80)
  2634→
  2635→            # TLS certificate probe
  2636→            if parsed.scheme == "https":
  2637→                try:
  2638→                    context = ssl.create_default_context()
  2639→                    context.check_hostname = False
  2640→                    context.verify_mode = ssl.CERT_NONE
  2641→                    with socket.create_connection(
  2642→                        (hostname, port), timeout=timeout
  2643→                    ) as sock:
  2644→                        with context.wrap_socket(sock, server_hostname=hostname) as ssock:
  2645→                            cert = ssock.getpeercert(binary_form=True)
  2646→                            if cert:
  2647→                                cert_text = ssl.DER_cert_to_PEM_cert(cert)
  2648→                                result["tls"]["raw"] = cert_text[:2000]
  2649→                                # Extract org, issuer etc. via regex
  2650→                                for field in ("O=", "CN=", "OU=", "ST=", "L="):
  2651→                                    matches = re.findall(
  2652→                                        rf"{field}([^,\n]+)", cert_text
  2653→                                    )
  2654→                                    if matches:
  2655→                                        result["tls"][field.rstrip("=").lower()] = (
  2656→                                            matches[0].strip()
  2657→                                        )
  2658→                            # Resolve IP for geo hints
  2659→                            ip = sock.getpeername()[0]
  2660→                            result["geo"]["resolved_ip"] = ip
  2661→                            try:
  2662→                                # Extract ASN-style info from reverse DNS
  2663→                                rev = socket.gethostbyaddr(ip)[0]
  2664→                                result["geo"]["reverse_dns"] = rev
  2665→                                asn_hint = rev.split(".")
  2666→                                if len(asn_hint) > 1:
  2667→                                    for part in asn_hint:
  2668→                                        if part.upper().startswith("AS"):
  2669→                                            result["geo"]["asn"] = part.upper()
  2670→                                            break
  2671→                            except (socket.herror, socket.gaierror):
  2672→                                pass
  2673→                except (socket.timeout, socket.error, ssl.SSLError):
  2674→                    pass
  2675→
  2676→            # HTTP header fetch
  2677→            try:
  2678→                req = urllib.request.Request(
  2679→                    url,
  2680→                    headers={"User-Agent": "ReconPro/9.2.0 AttributionEngine"},
  2681→                )
  2682→                with urllib.request.urlopen(req, timeout=timeout) as resp:
  2683→                    headers_dict = dict(resp.headers)
  2684→                    result["headers"] = headers_dict
  2685→                    # Read body (first 8KB for language analysis)
  2686→                    body = resp.read(8192)
  2687→                    try:
  2688→                        result["body"] = body.decode("utf-8", errors="replace")
  2689→                    except Exception:
  2690→                        result["body"] = body.decode("latin-1", errors="replace")
  2691→
  2692→                    # Detect WAF from headers
  2693→                    server = headers_dict.get("server", "")
  2694→                    waf_headers = (
  2695→                        headers_dict.get("x-waf", "")
  2696→                        + " " + headers_dict.get("cf-ray", "")
  2697→                        + " " + headers_dict.get("x-cdn", "")
  2698→                    )
  2699→                    if server or waf_headers.strip():
  2700→                        result["waf"] = {
  2701→                            "waf_name": server,
  2702→                            "waf_vendor": waf_headers.strip(),
  2703→                        }
  2704→
  2705→            except (urllib.error.URLError, socket.timeout, ValueError):
  2706→                pass
  2707→
  2708→        except Exception:
  2709→            pass
  2710→
  2711→        return result

  3117→        cls, group_ids: Optional[List[str]] = None,
  3118→    ) -> Dict[str, Dict[str, int]]:
  3119→        """Compute a pairwise technique overlap matrix for APT groups.
  3120→
  3121→        Args:
  3122→            group_ids: Optional list of group IDs. If ``None``, uses all groups.
  3123→
  3124→        Returns:
  3125→            Nested dict mapping ``group_id -> group_id -> overlap_count``.
  3126→        """
  3127→        ids = group_ids or list(APT_GROUPS.keys())
  3128→        matrix: Dict[str, Dict[str, int]] = {}
  3129→
  3130→        for gid_a in ids:
  3131→            matrix[gid_a] = {}
  3132→            techs_a = set(APT_GROUPS.get(gid_a, {}).get("techniques", []))
  3133→            for gid_b in ids:
  3134→                techs_b = set(APT_GROUPS.get(gid_b, {}).get("techniques", []))
  3135→                matrix[gid_a][gid_b] = len(techs_a & techs_b)
  3136→
  3137→        return matrix
