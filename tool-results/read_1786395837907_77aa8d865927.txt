   929→def _is_same_domain(url: str, base_url: str) -> bool:
   930→    """Check if a URL belongs to the same domain as the target."""
   931→    try:
   932→        parsed_url = urllib.parse.urlparse(url)
   933→        parsed_base = urllib.parse.urlparse(base_url)
   934→        return parsed_url.hostname == parsed_base.hostname
   935→    except Exception:
   936→        return False

   994→def _count_matching_patterns(
   995→    text: str, patterns: List[re.Pattern]
   996→) -> List[str]:
   997→    """Return matched strings for all matching patterns."""
   998→    matches: List[str] = []
   999→    seen: set = set()
  1000→    for pat in patterns:
  1001→        for m in pat.finditer(text):
  1002→            matched_text = m.group(0)
  1003→            # Truncate long matches for evidence
  1004→            if len(matched_text) > 200:
  1005→                matched_text = matched_text[:200] + "..."
  1006→            if matched_text not in seen:
  1007→                seen.add(matched_text)
  1008→                matches.append(matched_text)
  1009→    return matches

  1033→        "\u2063": "INVISIBLE SEPARATOR (U+2063)",
  1034→        "\u2064": "INVISIBLE PLUS (U+2064)",
  1035→        "\u180e": "MONGOLIAN VOWEL SEPARATOR (U+180E)",
  1036→        "\u202a": "LEFT-TO-RIGHT EMBEDDING (U+202A)",
  1037→        "\u202b": "RIGHT-TO-LEFT EMBEDDING (U+202B)",
  1038→        "\u202c": "POP DIRECTIONAL FORMATTING (U+202C)",
  1039→        "\u202d": "LEFT-TO-RIGHT OVERRIDE (U+202D)",
  1040→        "\u202e": "RIGHT-TO-LEFT OVERRIDE (U+202E)",
  1041→        "\u206a": "INHIBIT SYMMETRIC SWAPPING (U+206A)",
  1042→        "\u206b": "ACTIVATE SYMMETRIC SWAPPING (U+206B)",
  1043→        "\u206c": "INHIBIT ARABIC FORM SHAPING (U+206C)",
  1044→        "\u206d": "ACTIVATE ARABIC FORM SHAPING (U+206D)",
  1045→        "\u206e": "NATIONAL DIGIT SHAPES (U+206E)",

  1049→    found_zwc: List[str] = []
  1050→    zwc_count = 0
  1051→    for char, name in zwc_map.items():
  1052→        count = body.count(char)

  1061→            and body.startswith("\ufeff")
  1062→            and "\u200b" not in body
  1063→            and "\u200c" not in body
  1064→            and "\u200d" not in body

  1067→        if not is_bom_only:

  1069→                title="Steganographic Watermark — Invisible Characters Detected",

  1072→                module=MODULE_NAME,
  1073→                description=(
  1074→                    f"Document contains {zwc_count} invisible/zero-width characters that can "
  1075→                    f"encode hidden data (steganographic watermarking). Found: "
  1076→                    f"{', '.join(found_zwc[:5])}. "
  1077→                    f"{'This may be a legitimate BOM.' if is_bom_only else 'This strongly suggests intentional data hiding.'}"
  1078→                ),
  1079→                evidence=f"ZWC types: {'; '.join(found_zwc[:5])}; total count: {zwc_count}",
  1080→                asset="",
  1081→                points_deducted=12 if not is_bom_only else 0,
  1082→                remediation=(
  1083→                    "Sanitize documents to remove zero-width characters before distribution. "
  1084→                    "Use unicode-normalization (NFKC) to strip invisible chars."
  1085→                ),
  1086→                dread_score=0.7 if not is_bom_only else 0.1,
  1087→            ))
  1088→
  1089→    # ── Base64-encoded image data (possible hidden payload) ──
  1090→    b64_matches = _RE_BASE64_DATA_URL.findall(body)
  1091→    suspicious_b64: List[str] = []
  1092→    for b64data in b64_matches:
  1093→        if len(b64data) < 64:
  1094→            continue
  1095→        try:
  1096→            decoded = base64.b64decode(b64data)
  1097→            # Check if decoded content has unusual structure
  1098→            # Very large base64 images with little visible content is suspicious
  1099→            if len(decoded) > 50000:
  1100→                suspicious_b64.append(
  1101→                    f"base64 image: {len(decoded)} bytes decoded from {len(b64data)} chars"
  1102→                )
  1103→        except Exception:
  1104→            # Invalid base64 — could be intentional obfuscation
  1105→            if len(b64data) > 100:
  1106→                suspicious_b64.append(f"invalid base64 data: {len(b64data)} chars")
  1107→
  1108→    if suspicious_b64:
  1109→        findings.append(Finding(
  1110→            title="Steganographic Watermark — Suspicious Base64-Embedded Content",
  1111→            severity="medium",
  1112→            category="stego-watermark",
  1113→            module=MODULE_NAME,
  1114→            description=(
  1115→                f"Document contains {len(suspicious_b64)} suspicious base64-embedded data blocks. "
  1116→                f"Large or malformed base64 in images can hide tracking payloads."
  1117→            ),
  1118→            evidence=_truncate_evidence("; ".join(suspicious_b64[:3])),
  1119→            asset="",
  1120→            points_deducted=8,
  1121→            remediation="Inspect base64-embedded images for hidden payloads. Compare against known-good versions.",
  1122→            dread_score=0.5,
  1123→        ))
  1124→
  1125→    # ── Trailing whitespace steganography ──
  1126→    lines = body.split("\n")
  1127→    trailing_space_lines = 0
  1128→    trailing_tab_lines = 0
  1129→    for line in lines:
  1130→        stripped = line.rstrip("\r")
  1131→        if stripped != stripped.rstrip():
  1132→            trailing_space_lines += 1
  1133→        if stripped != stripped.rstrip("\t"):
  1134→            trailing_tab_lines += 1
  1135→
  1136→    total_lines = len(lines)
  1137→    if total_lines > 10:
  1138→        trailing_ratio = (trailing_space_lines + trailing_tab_lines) / total_lines
  1139→        if trailing_ratio > 0.3 and (trailing_space_lines + trailing_tab_lines) > 20:
  1140→            findings.append(Finding(
  1141→                title="Steganographic Watermark — Trailing Whitespace Encoding",
  1142→                severity="medium",
  1143→                category="stego-watermark",
  1144→                module=MODULE_NAME,
  1145→                description=(
  1146→                    f"{trailing_ratio:.1%} of lines ({trailing_space_lines} spaces, "
  1147→                    f"{trailing_tab_lines} tabs out of {total_lines}) have trailing whitespace. "
  1148→                    f"Trailing whitespace can encode binary data (space=0, tab=1 or vice versa)."
  1149→                ),
  1150→                evidence=(
  1151→                    f"Trailing spaces: {trailing_space_lines}, trailing tabs: {trailing_tab_lines}, "
  1152→                    f"total lines: {total_lines}, ratio: {trailing_ratio:.2%}"
  1153→                ),
  1154→                asset="",
  1155→                points_deducted=6,
  1156→                remediation="Strip trailing whitespace from documents. Use automated linters to prevent whitespace steganography.",
  1157→                dread_score=0.4,
  1158→            ))
  1159→
  1160→    # ── Line-ending inconsistency (CRLF vs LF steganography) ──
  1161→    crlf_count = body.count("\r\n")
  1162→    lf_only = body.replace("\r\n", "").count("\n")
  1163→    if crlf_count > 0 and lf_only > 0 and total_lines > 20:
  1164→        mixed_ratio = crlf_count / (crlf_count + lf_only)
  1165→        if 0.1 < mixed_ratio < 0.9:
  1166→            findings.append(Finding(
  1167→                title="Steganographic Watermark — Mixed Line Ending Encoding",
  1168→                severity="low",
  1169→                category="stego-watermark",
  1170→                module=MODULE_NAME,
  1171→                description=(
  1172→                    f"Document has mixed line endings: {crlf_count} CRLF, {lf_only} LF-only. "
  1173→                    f"Mixed ratio: {mixed_ratio:.2%}. This can encode binary data "
  1174→                    f"(CRLF=1, LF=0)."
  1175→                ),
  1176→                evidence=f"CRLF: {crlf_count}, LF: {lf_only}, mixed ratio: {mixed_ratio:.2%}",
  1177→                asset="",
  1178→                points_deducted=4,
  1179→                remediation="Normalize all line endings to a consistent format (LF or CRLF).",
  1180→                dread_score=0.3,
  1181→            ))
  1182→
  1183→    return findings

  1186→def _check_tracking_pixels(
  1187→    body: str, base_url: str, target: str
  1188→) -> List[Finding]:
  1189→    """Cat 1: Detect 1x1 tracking pixels, web bugs, and beacon images."""
  1190→    findings: List[Finding] = []
  1191→    if not body:
  1192→        return findings
  1193→
  1194→    img_tags = _RE_IMG_TAG.findall(body)
  1195→    for img_attrs in img_tags:
  1196→        # Check dimensions
  1197→        dims = _RE_DIM_ATTR.findall(img_attrs)
  1198→        width = int(dims[0]) if len(dims) > 0 else 0
  1199→        height = int(dims[1]) if len(dims) > 1 else 0
  1200→
  1201→        is_tiny = (width == 1 and height == 1) or (width == 0 and height == 0)
  1202→        is_hidden = _RE_INVISIBLE_STYLE.search(img_attrs)
  1203→
  1204→        if not (is_tiny or is_hidden):
  1205→            continue
  1206→
  1207→        # Extract src
  1208→        src_match = re.search(r"""src\s*=\s*["']([^"']+)["']""", img_attrs, re.IGNORECASE)
  1209→        if not src_match:
  1210→            continue
  1211→
  1212→        src = src_match.group(1)
  1213→        full_url = _normalize_url(base_url, src)
  1214→        domain = _extract_domain(full_url)
  1215→
  1216→        # Determine evidence
  1217→        dim_str = f"{width}x{height}" if (width or height) else "hidden"
  1218→        evidence_parts = [f"<{dim_str}>"]
  1219→        if is_hidden:
  1220→            evidence_parts.append("invisible style")
  1221→        if is_tiny:
  1222→            evidence_parts.append("tiny dimensions")
  1223→
  1224→        # Check if external tracking domain
  1225→        is_external = _is_external_url(src) and not _is_same_domain(full_url, base_url)
  1226→        has_tracking_params = bool(_RE_UTM_PARAMS.search(src) or _RE_CLICK_ID.search(src))
  1227→        has_tracking_path = bool(_RE_SUSPICIOUS_HOST.search(full_url))
  1228→
  1229→        if is_tiny and (is_external or has_tracking_path or has_tracking_params):
  1230→            severity = "high"
  1231→            desc = (
  1232→                f"Classic web bug / tracking pixel detected: {dim_str} image pointing to "
  1233→                f"{'external domain ' + domain if is_external else ''}"
  1234→                f"{'tracking URL path' if has_tracking_path else ''}"
  1235→                f"{' with tracking parameters (UTM/click IDs)' if has_tracking_params else ''}. "
  1236→                f"This image loads when the document is opened, signaling access to a third party."
  1237→            )
  1238→            evidence_parts.append(f"src: {src}")
  1239→        elif is_hidden and is_external:
  1240→            severity = "medium"
  1241→            desc = (
  1242→                f"Hidden image ({dim_str}) with invisible styling loads from external domain "
  1243→                f"'{domain}'. May be a tracking pixel using CSS to hide rather than tiny dimensions."
  1244→            )
  1245→            evidence_parts.append(f"src: {src}")
  1246→        elif is_tiny:
  1247→            severity = "low"
  1248→            desc = (
  1249→                f"Tiny image ({dim_str}) found. While it may be legitimate spacer, "
  1250→                f"1x1 images are commonly used as tracking pixels."
  1251→            )
  1252→            evidence_parts.append(f"src: {src}")
  1253→        else:
  1254→            continue
  1255→
  1256→        findings.append(Finding(
  1257→            title="Tracking Pixel — Web Bug / Beacon Image Detected",
  1258→            severity=severity,
  1259→            category="tracking-pixel",
  1260→            module=MODULE_NAME,
  1261→            description=desc,
  1262→            evidence="; ".join(evidence_parts),
  1263→            asset=domain or target,
  1264→            points_deducted=_severity_points(severity),
  1265→            remediation=(
  1266→                "Remove tracking pixels from documents. If analytics are needed, use "
  1267→                "server-side logging that doesn't leak reader identity to third parties."
  1268→            ),
  1269→            dread_score=_severity_dread(severity),
  1270→        ))
  1271→
  1272→    # Check for empty/hidden div-based pixels
  1273→    hidden_div_pattern = re.compile(
  1274→        r"""<div[^>]+style\s*=\s*["'][^"']*(?:width\s*:\s*[01]px|height\s*:\s*[01]px|"""
  1275→        r"""display\s*:\s*none|visibility\s*:\s*hidden|opacity\s*:\s*0)[^"']*""'>"""
  1276→        r"[^<]*(?:<img[^>]+>)?[^<]*</div>",
  1277→        re.IGNORECASE | re.DOTALL,
  1278→    )
  1279→    hidden_divs = hidden_div_pattern.findall(body)
  1280→    if len(hidden_divs) > 3:
  1281→        findings.append(Finding(
  1282→            title="Tracking Pixel — Multiple Hidden Container Elements",
  1283→            severity="medium",
  1284→            category="tracking-pixel",
  1285→            module=MODULE_NAME,
  1286→            description=(
  1287→                f"Found {len(hidden_divs)} hidden container elements (divs) with zero/tiny dimensions "
  1288→                f"or invisible styling. This pattern is commonly used to wrap tracking pixels."
  1289→            ),
  1290→            evidence=f"Hidden div count: {len(hidden_divs)}",
  1291→            asset=target,
  1292→            points_deducted=8,
  1293→            remediation="Audit hidden container elements. Remove any that serve no functional purpose.",
  1294→            dread_score=0.5,
  1295→        ))
  1296→
  1297→    return findings

  1300→def _check_beaconing(
  1301→    body: str, base_url: str, target: str, headers: Dict[str, str]
  1302→) -> List[Finding]:
  1303→    """Cat 2: Detect beaconing URLs in document metadata and content."""
  1304→    findings: List[Finding] = []
  1305→    if not body:
  1306→        return findings
  1307→
  1308→    # ── Meta refresh redirects (phone-home on open) ──
  1309→    refresh_matches = _RE_REFRESH_META.findall(body)
  1310→    for refresh_url in refresh_matches:
  1311→        clean_url = refresh_url.strip().split("url=", 1)[-1].strip("'\" ")
  1312→        if _is_external_url(clean_url):
  1313→            domain = _extract_domain(clean_url)
  1314→            findings.append(Finding(
  1315→                title="Beaconing — Meta Refresh Redirect to External Domain",
  1316→                severity="critical",
  1317→                category="document-beacon",
  1318→                module=MODULE_NAME,
  1319→                description=(
  1320→                    f"Document contains a <meta http-equiv=refresh> that redirects to external domain "
  1321→                    f"'{domain}'. This triggers an automatic request when the document is opened, "
  1322→                    f"potentially leaking access information."
  1323→                ),
  1324→                evidence=f"Refresh URL: {clean_url}",
  1325→                asset=domain or target,
  1326→                points_deducted=25,
  1327→                remediation=(
  1328→                    "Remove meta refresh redirects. If timed navigation is needed, use "
  1329→                    "JavaScript with user consent."
  1330→                ),
  1331→                dread_score=0.9,
  1332→            ))
  1333→
  1334→    # ── Ping attributes ──
  1335→    ping_matches = _RE_PING_ATTR.findall(body)
  1336→    for ping_url in ping_matches:
  1337→        full_url = _normalize_url(base_url, ping_url)
  1338→        domain = _extract_domain(full_url)
  1339→        is_external = not _is_same_domain(full_url, base_url)
  1340→        severity = "high" if is_external else "medium"
  1341→        findings.append(Finding(
  1342→            title="Beaconing — HTML Ping Attribute Detected",
  1343→            severity=severity,
  1344→            category="html-beacon",
  1345→            module=MODULE_NAME,
  1346→            description=(
  1347→                f"HTML element with ping attribute sends notification to '{domain}' on interaction. "
  1348→                f"{'External domain — recipient can track link engagement.' if is_external else ''}"
  1349→            ),
  1350→            evidence=f"ping URL: {ping_url}",
  1351→            asset=domain or target,
  1352→            points_deducted=_severity_points(severity),
  1353→            remediation="Remove ping attributes. Use server-side redirect logging if click tracking is required.",
  1354→            dread_score=_severity_dread(severity),
  1355→        ))
  1356→
  1357→    # ── Hidden iframes ──
  1358→    iframe_srcs = _RE_IFRAME_SRC.findall(body)
  1359→    for src in iframe_srcs:
  1360→        full_url = _normalize_url(base_url, src)
  1361→        # Check if the iframe is hidden (look for surrounding context)
  1362→        is_hidden = bool(re.search(
  1363→            r'<iframe[^>]*' + re.escape(src[:80]) + r'[^>]*' +
  1364→            r'(?:display\s*:\s*none|visibility\s*:\s*hidden|opacity\s*:\s*0|'
  1365→            r'width\s*:\s*[01]px|height\s*:\s*[01]px|'
  1366→            r'position\s*:\s*absolute[^;]*left\s*:\s*-)',
  1367→            body[:body.index(src) + len(src) + 200],
  1368→            re.IGNORECASE,
  1369→        )) if src in body else False
  1370→
  1371→        domain = _extract_domain(full_url)
  1372→        is_external = not _is_same_domain(full_url, base_url)
  1373→
  1374→        if is_hidden and is_external:
  1375→            findings.append(Finding(
  1376→                title="Beaconing — Hidden IFRAME Loading External Resource",
  1377→                severity="critical",
  1378→                category="document-beacon",
  1379→                module=MODULE_NAME,
  1380→                description=(
  1381→                    f"Hidden iframe loads content from external domain '{domain}'. "
  1382→                    f"Hidden iframes can execute tracking scripts, load exploits, or exfiltrate data "
  1383→                    f"without the user's knowledge."
  1384→                ),
  1385→                evidence=f"iframe src: {src}",
  1386→                asset=domain or target,
  1387→                points_deducted=25,
  1388→                remediation=(
  1389→                    "Remove hidden iframes. All iframes should be visible and serve a clear purpose. "
  1390→                    "Audit all cross-origin iframe inclusions."
  1391→                ),
  1392→                dread_score=0.9,
  1393→            ))
  1394→        elif is_external:
  1395→            findings.append(Finding(
  1396→                title="Beaconing — IFRAME Loading External Resource",
  1397→                severity="medium",
  1398→                category="document-beacon",
  1399→                module=MODULE_NAME,
  1400→                description=(
  1401→                    f"Iframe loads content from external domain '{domain}'. "
  1402→                    f"Cross-origin iframes can leak referrer information and enable tracking."
  1403→                ),
  1404→                evidence=f"iframe src: {src}",
  1405→                asset=domain or target,
  1406→                points_deducted=8,
  1407→                remediation="Review cross-origin iframe usage. Consider sandbox attribute or CSP restrictions.",
  1408→                dread_score=0.5,
  1409→            ))
  1410→
  1411→    # ── Object/embed tags loading external resources ──
  1412→    for tag_name, pattern, label in [
  1413→        ("object", _RE_OBJECT_DATA, "data"),
  1414→        ("embed", _RE_EMBED_SRC, "src"),
  1415→    ]:
  1416→        for src in pattern.findall(body):
  1417→            full_url = _normalize_url(base_url, src)
  1418→            domain = _extract_domain(full_url)
  1419→            is_external = not _is_same_domain(full_url, base_url)
  1420→            if is_external:
  1421→                findings.append(Finding(
  1422→                    title=f"Beaconing — External {tag_name.upper()} Tag",
  1423→                    severity="high",
  1424→                    category="document-beacon",
  1425→                    module=MODULE_NAME,
  1426→                    description=(
  1427→                        f"<{tag_name}> tag with {label}='{src}' loads from external domain '{domain}'. "
  1428→                        f"Object/embed tags can execute plugins and scripts from remote sources."
  1429→                    ),
  1430→                    evidence=f"<{tag_name} {label}={src}",
  1431→                    asset=domain or target,
  1432→                    points_deducted=15,
  1433→                    remediation=f"Remove external <{tag_name}> tags or host the resource locally.",
  1434→                    dread_score=0.7,
  1435→                ))
  1436→
  1437→    # ── WebSocket connections ──
  1438→    ws_matches = _RE_WS_CONN.findall(body)
  1439→    for ws_url in ws_matches:
  1440→        domain = _extract_domain(ws_url.replace("wss://", "https://").replace("ws://", "http://"))
  1441→        is_external = not _is_same_domain(
  1442→            ws_url.replace("wss://", "https://").replace("ws://", "http://"),
  1443→            base_url,
  1444→        )
  1445→        severity = "critical" if is_external else "high"
  1446→        findings.append(Finding(
  1447→            title="Beaconing — WebSocket Connection Detected",
  1448→            severity=severity,
  1449→            category="ws-beacon",
  1450→            module=MODULE_NAME,
  1451→            description=(
  1452→                f"JavaScript opens a WebSocket connection to '{ws_url}'. "
  1453→                f"{'External WebSocket — enables real-time data exfiltration.' if is_external else ''} "
  1454→                f"WebSockets bypass standard HTTP logging and can maintain persistent tracking channels."
  1455→            ),
  1456→            evidence=f"WebSocket URL: {ws_url}",
  1457→            asset=domain or target,
  1458→            points_deducted=_severity_points(severity),
  1459→            remediation=(
  1460→                "Remove or justify all WebSocket connections. Ensure WebSocket traffic is encrypted (WSS) "
  1461→                "and serves a documented functional purpose."
  1462→            ),
  1463→            dread_score=_severity_dread(severity),
  1464→        ))
  1465→
  1466→    # ── Link rel=prefetch/prerender beacons ──
  1467→    prefetch_pattern = re.compile(
  1468→        r"""<link[^>]+rel\s*=\s*["\'](?:prefetch|prerender)["\'][^>]+href\s*=\s*["\']([^"']+)["']""",
  1469→        re.IGNORECASE,
  1470→    )
  1471→    prefetch_urls = prefetch_pattern.findall(body)
  1472→    for pf_url in prefetch_urls:
  1473→        full_url = _normalize_url(base_url, pf_url)
  1474→        domain = _extract_domain(full_url)
  1475→        has_tracking_path = bool(_RE_SUSPICIOUS_HOST.search(full_url))
  1476→        if has_tracking_path:
  1477→            findings.append(Finding(
  1478→                title="Beaconing — Prefetch/Prerender Used as Tracking Beacon",
  1479→                severity="medium",
  1480→                category="resource-beacon",
  1481→                module=MODULE_NAME,
  1482→                description=(
  1483→                    f"Link with rel=prefetch/prerender points to tracking URL '{pf_url}' on domain '{domain}'. "
  1484→                    f"Browsers automatically fetch these URLs, creating a tracking beacon without JavaScript."
  1485→                ),
  1486→                evidence=f"Prefetch URL: {pf_url}",
  1487→                asset=domain or target,
  1488→                points_deducted=8,
  1489→                remediation="Remove prefetch links to tracking endpoints. Use standard analytics if needed.",
  1490→                dread_score=0.5,
  1491→            ))
  1492→
  1493→    # ── DNS-based beaconing (encoded data in subdomains) ──
  1494→    all_urls = _extract_urls_from_text(body)
  1495→    for url in all_urls:
  1496→        parsed = urllib.parse.urlparse(url)
  1497→        hostname = parsed.hostname or ""
  1498→        # Check for hex-encoded subdomains (common in DNS tunneling)
  1499→        parts = hostname.split(".")
  1500→        for part in parts:
  1501→            if len(part) >= 16 and re.match(r"^[a-f0-9]+$", part, re.IGNORECASE):
  1502→                findings.append(Finding(
  1503→                    title="Beaconing — Potential DNS Tunneling / Encoded Subdomain",
  1504→                    severity="critical",
  1505→                    category="dns-beacon",
  1506→                    module=MODULE_NAME,
  1507→                    description=(
  1508→                        f"URL '{url}' contains a hex-encoded subdomain '{part}' ({len(part)} chars). "
  1509→                        f"DNS queries for encoded subdomains are a classic C2 beaconing technique used to "
  1510→                        f"exfiltrate data through DNS, bypassing most firewalls and proxies."
  1511→                    ),
  1512→                    evidence=f"URL: {url}; encoded subdomain: {part}",
  1513→                    asset=hostname or target,
  1514→                    points_deducted=25,
  1515→                    remediation=(
  1516→                        "Investigate the source of this URL. DNS tunneling indicates a compromised or "
  1517→                        "maliciously crafted document. Block the domain and analyze network DNS logs."
  1518→                    ),
  1519→                    dread_score=0.9,
  1520→                ))
  1521→                break
  1522→
  1523→    return findings

  1526→def _check_malicious_links(
  1527→    body: str, base_url: str, target: str
  1528→) -> List[Finding]:
  1529→    """Cat 4: Analyze URLs in served content for malicious/tracking destinations."""
  1530→    findings: List[Finding] = []
  1531→    if not body:
  1532→        return findings
  1533→
  1534→    all_urls = _extract_urls_from_text(body)
  1535→    if not all_urls:
  1536→        return findings
  1537→
  1538→    malicious_categories: List[Dict[str, Any]] = []
  1539→    tracking_domains_found: List[str] = []
  1540→    utm_tracked: List[str] = []
  1541→    click_id_tracked: List[str] = []
  1542→    suspicious_param_urls: List[str] = []
  1543→
  1544→    for url in all_urls:
  1545→        parsed = urllib.parse.urlparse(url)
  1546→        hostname = parsed.hostname or ""
  1547→        path = parsed.path or ""
  1548→        query = parsed.query or ""
  1549→
