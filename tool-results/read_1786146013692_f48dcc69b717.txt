    52→class ToolResult:
    53→    """Result of a tool execution."""
    54→
    55→    def __init__(self, output: str, success: bool = True, data: Any = None):
    56→        self.output = output
    57→        self.success = success
    58→        self.data = data
    59→
    60→    def to_dict(self) -> Dict:
    61→        return {"output": self.output, "success": self.success, "data": self.data}
    62→
    63→
    64→class Tool:
    65→    """A callable tool the agent can use."""
    66→
    67→    def __init__(
    68→        self,
    69→        name: str,
    70→        description: str,
    71→        parameters: Dict[str, Any],
    72→        execute: Callable[..., ToolResult],
    73→        category: str = "scan",
    74→        dangerous: bool = False,
    75→    ):
    76→        self.name = name
    77→        self.description = description
    78→        self.parameters = parameters
    79→        self.execute = execute
    80→        self.category = category
    81→        self.dangerous = dangerous
    82→
    83→    def to_openai_tool(self) -> Dict:
    84→        return {
    85→            "type": "function",
    86→            "function": {
    87→                "name": self.name,
    88→                "description": self.description,
    89→                "parameters": {
    90→                    "type": "object",
    91→                    "properties": self.parameters,
    92→                    "required": list(self.parameters.keys()),
    93→                },
    94→            },
    95→        }
    96→
    97→    def to_anthropic_tool(self) -> Dict:
    98→        return {
    99→            "name": self.name,
   100→            "description": self.description,
   101→            "input_schema": {
   102→                "type": "object",
   103→                "properties": self.parameters,
   104→                "required": list(self.parameters.keys()),
   105→            },
   106→        }

   108→
   109→class ToolRegistry:
   110→    """Registry of all available tools."""
   111→
   112→    def __init__(self):
   113→        self.tools: Dict[str, Tool] = {}
   114→
   115→    def register(self, tool: Tool) -> None:
   116→        self.tools[tool.name] = tool
   117→
   118→    def get(self, name: str) -> Optional[Tool]:
   119→        return self.tools.get(name)
   120→
   121→    def execute(self, name: str, **kwargs) -> ToolResult:
   122→        tool = self.tools.get(name)
   123→        if not tool:
   124→            return ToolResult(f"Tool '{name}' not found.", success=False)
   125→        try:
   126→            return tool.execute(**kwargs)
   127→        except Exception as e:
   128→            return ToolResult(f"Tool error: {e}", success=False)
   129→
   130→    def list_tools(self) -> List[Dict]:
   131→        return [
   132→            {"name": t.name, "description": t.description, "category": t.category}
   133→            for t in self.tools.values()
   134→        ]
   135→
   136→    def to_openai_tools(self) -> List[Dict]:
   137→        return [t.to_openai_tool() for t in self.tools.values()]
   138→
   139→    def to_anthropic_tools(self) -> List[Dict]:
   140→        return [t.to_anthropic_tool() for t in self.tools.values()]

   147→
   148→class AgentMemory:
   149→    """Persistent memory for the Nexus agent across sessions."""
   150→
   151→    def __init__(self):
   152→        MEMORY_DIR.mkdir(parents=True, exist_ok=True)
   153→        self._data = self._load()
   154→
   155→    def _load(self) -> Dict:
   156→        if CONTEXT_FILE.exists():
   157→            try:
   158→                return json.loads(CONTEXT_FILE.read_text())
   159→            except Exception:
   160→                pass
   161→        return {
   162→            "sessions": [],
   163→            "target_contexts": {},
   164→            "learned_patterns": [],
   165→            "created": datetime.now(timezone.utc).isoformat(),
   166→        }
   167→
   168→    def _save(self) -> None:
   169→        # Keep only the last 50 sessions
   170→        if len(self._data.get("sessions", [])) > 50:
   171→            self._data["sessions"] = self._data["sessions"][-50:]
   172→        try:
   173→            CONTEXT_FILE.write_text(json.dumps(self._data, indent=2, default=str))
   174→        except OSError:
   175→            pass
   176→
   177→    def save_session(
   178→        self,
   179→        goal: str,
   180→        target: str,
   181→        plan: List[Dict],
   182→        tool_results: List[Dict],
   183→        summary: str,
   184→    ) -> None:
   185→        session = {
   186→            "goal": goal,
   187→            "target": target,
   188→            "plan": plan,
   189→            "tool_results": tool_results,
   190→            "summary": summary,
   191→            "timestamp": datetime.now(timezone.utc).isoformat(),
   192→        }
   193→        self._data.setdefault("sessions", []).append(session)
   194→        # Update target context
   195→        self._data.setdefault("target_contexts", {})[target] = {
   196→            "last_score": self._extract_score(tool_results),
   197→            "last_grade": self._extract_grade(tool_results),
   198→            "last_run": session["timestamp"],
   199→            "finding_count": self._extract_finding_count(tool_results),
   200→            "known_issues": self._extract_issues(tool_results),
   201→        }
   202→        self._save()
   203→
   204→    def get_target_context(self, target: str) -> Optional[Dict]:
   205→        return self._data.get("target_contexts", {}).get(target)
   206→
   207→    def get_recent_sessions(self, limit: int = 5) -> List[Dict]:
   208→        return self._data.get("sessions", [])[-limit:]
   209→
   210→    @staticmethod
   211→    def _extract_score(results: List[Dict]) -> Optional[int]:
   212→        for r in results:
   213→            d = r.get("data")
   214→            if isinstance(d, dict) and "total_score" in d:
   215→                return d["total_score"]
   216→        return None
   217→
   218→    @staticmethod
   219→    def _extract_grade(results: List[Dict]) -> Optional[str]:
   220→        for r in results:
   221→            d = r.get("data")
   222→            if isinstance(d, dict) and "grade" in d:
   223→                return d["grade"]
   224→        return None
   225→
   226→    @staticmethod
   227→    def _extract_finding_count(results: List[Dict]) -> int:
   228→        total = 0
   229→        for r in results:
   230→            d = r.get("data")
   231→            if isinstance(d, dict) and "findings" in d:
   232→                total += len(d["findings"])
   233→        return total
   234→
   235→    @staticmethod
   236→    def _extract_issues(results: List[Dict]) -> List[str]:
   237→        issues = []
   238→        for r in results:
   239→            d = r.get("data")
   240→            if isinstance(d, dict) and "findings" in d:
   241→                for f in d["findings"]:
   242→                    if isinstance(f, dict) and f.get("severity") in ("critical", "high"):
   243→                        issues.append(f.get("title", ""))
   244→        return issues[:20]

   252→def _tool_remote_scan(
   253→    target: str = "",
   254→    modules: str = "",
   255→    all_modules: bool = False,
   256→    timeout: int = 10,
   257→) -> ToolResult:
   258→    """Full remote scan of a target domain/URL."""
   259→    from .scanner import scan
   260→    if not target:
   261→        return ToolResult("No target specified.", success=False)
   262→    mod_list = [m.strip().lower() for m in modules.split(",") if m.strip()] if modules else None
   263→    result = scan(target, modules=mod_list, all_modules=all_modules, timeout=timeout)
   264→    return ToolResult(
   265→        f"Scanned {target}. Score: {result.total_score}/100 ({result.grade}). "
   266→        f"{len(result.findings)} findings.",
   267→        success=True,
   268→        data=result.to_dict(),
   269→    )

   428→def _tool_knowledge_graph(action: str = "build", target: str = "") -> ToolResult:
   429→    """Build or query the security knowledge graph."""
   430→    from .knowledge_graph import SecurityKnowledgeGraph
   431→    if not target:
   432→        return ToolResult("No target specified.", success=False)
   433→    kg = SecurityKnowledgeGraph()
   434→    if action == "build":
   435→        from .history import get_latest
   436→        latest = get_latest(target=target)
   437→        if not latest:
   438→            latest = get_latest()
   439→        if latest:
   440→            kg.add_scan_result(latest)
   441→        n_count = len(list(kg._g.nodes(data=False)))
   442→        e_count = len(list(kg._g.edges(data=False)))
   443→        return ToolResult(
   444→            f"Knowledge graph built: {n_count} nodes, {e_count} edges for {target}.",
   445→            success=True,
   446→            data={"node_count": n_count, "edge_count": e_count, "target": target},
   447→        )
   448→    elif action == "attack_chains":
   449→        chains = kg.find_attack_chains(max_depth=4)
   450→        return ToolResult(
   451→            f"Found {len(chains)} attack chain(s).", success=True,
   452→            data={"attack_chains": chains},
   453→        )
   454→    else:
   455→        return ToolResult(f"Unknown action: {action}. Use 'build' or 'attack_chains'.", success=False)

   503→def _tool_generate_report(format: str = "html") -> ToolResult:
   504→    """Generate a report (HTML, MD, SARIF, JSON) from the last scan."""
   505→    from .history import get_latest
   506→    from .formats import export
   507→    latest = get_latest()
   508→    if not latest:
   509→        return ToolResult("No scan data to export. Run a scan first.", success=False)
   510→    target = latest.get("target", "unknown").replace(",", "").replace("/", "_")[:40]
   511→    ts = datetime.now().strftime("%Y%m%d_%H%M%S")
   512→    ext_map = {"html": ".html", "md": ".md", "sarif": ".sarif", "json": ".json"}
   513→    ext = ext_map.get(format.lower(), ".html")
   514→    filename = f"reconpro_report_{ts}_{target}{ext}"
   515→    filepath = str(MEMORY_DIR / filename)
   516→    path = export(latest, filepath, format=format.lower())
   517→    return ToolResult(
   518→        f"Report generated: {path}",
   519→        success=True,
   520→        data={"path": path, "format": format},

   563→    dangerous_patterns = ["rm -rf /", "mkfs", "dd if=", ":(){ :|:& };:", "chmod -R 777 /"]
   564→    for dp in dangerous_patterns:
   565→        if dp in command:
   566→            return ToolResult(f"Blocked dangerous command pattern: {dp}", success=False)
   567→    try:
   568→        result = subprocess.run(
   569→            command, shell=True, capture_output=True, text=True, timeout=30,
   570→        )
   571→        output = (result.stdout + result.stderr)[:2000]
   572→        return ToolResult(
   573→            f"Exit code: {result.returncode}\n{output}",
   574→            success=result.returncode == 0,
   575→        )
   576→    except subprocess.TimeoutExpired:
   577→        return ToolResult("Command timed out (30s).", success=False)
   578→    except Exception as e:
   579→        return ToolResult(f"Command error: {e}", success=False)

   779→#  RULE-BASED PLANNING ENGINE
   780→# ══════════════════════════════════════════════════════════════════════

   782→
   783→class RuleBasedPlanner:
   784→    """Smart rule-based planner that classifies intent, generates plans,
   785→    and adapts based on results — no LLM required.
   786→
   787→    The planner uses a weighted intent scoring system that considers:
   788→      - Direct keyword matches with configurable weights
   789→      - Modifier words that boost or reduce intent scores
   790→      - Target type classification (domain, IP, local, codebase)
   791→      - Proximity of intent words to target words
   792→      - Prior context from persistent memory
   793→    """
   794→
   795→    # Intent keyword weights: (word_pattern, base_weight)
   796→    _INTENT_KEYWORDS: Dict[str, List[Tuple[str, float]]] = {
   797→        "full_scan": [
   798→            ("scan", 0.4), ("audit", 0.3), ("assess", 0.3), ("pentest", 0.5),
   799→            ("security", 0.2), ("vuln", 0.4), ("test", 0.2),
   800→        ],
   801→        "subdomain_recon": [
   802→            ("subdomain", 0.8), ("sub-domain", 0.8), ("attack surface", 0.6),
   803→            ("enumerate", 0.5), ("discover", 0.3), ("surface map", 0.7),
   804→        ],
   805→        "local_audit": [
   806→            ("machine", 0.6), ("laptop", 0.6), ("computer", 0.5), ("local", 0.6),
   807→            ("my system", 0.7), ("this machine", 0.6), ("localhost", 0.7),
   808→        ],
   809→        "dev_security": [
   810→            ("project", 0.5), ("codebase", 0.5), ("repo", 0.4), ("source code", 0.5),
   811→            ("dependencies", 0.4), ("package", 0.3), ("code", 0.3),
   812→        ],
   813→        "secret_hunting": [
   814→            ("secrets", 0.7), ("secret", 0.7), ("credential", 0.7), ("api key", 0.6), ("token", 0.5),
   815→            ("password", 0.5), ("leak", 0.5), ("exposed", 0.3),
   816→        ],
   817→        "compliance": [
   818→            ("compliance", 0.7), ("health", 0.5), ("doctor", 0.5), ("diagnos", 0.5),
   819→            ("checkup", 0.5), ("hardening", 0.5), ("cis", 0.6), ("policy", 0.4),
   820→        ],
   821→        "attack_simulation": [
   822→            ("attack", 0.6), ("exploit", 0.6), ("hack", 0.5), ("swarm", 0.7),
   823→            ("adversarial", 0.7), ("red team", 0.6), ("offensive", 0.5),
   824→        ],
   825→        "code_analysis": [
   826→            (" ast ", 0.7), ("static analysis", 0.5), ("code review", 0.5), ("sast", 0.7),
   827→            ("analyze code", 0.6), ("code quality", 0.4), ("static code", 0.5),
   828→        ],
   829→        "threat_intel": [
   830→            ("cve", 0.7), ("threat", 0.6), ("intel", 0.6), ("enrich", 0.5),
   831→            ("knowledge graph", 0.7), ("graph", 0.4), ("chain", 0.3),
   832→        ],
   833→        "reporting": [
   834→            ("report", 0.7), ("html", 0.4), ("sarif", 0.6), ("pdf", 0.4),
   835→            ("markdown", 0.4), ("export", 0.5), ("document", 0.4), ("generate report", 0.7),
   836→        ],
   837→        "comparison": [
   838→            ("compare", 0.7), ("diff", 0.7), ("vs", 0.5), ("versus", 0.5),
   839→            ("trend", 0.5), ("progress", 0.3),
   840→        ],
   841→        "benchmarking": [
   842→            ("benchmark", 0.7), ("score", 0.4), ("grade", 0.4), ("rating", 0.5),
   843→            ("vibesec", 0.6), ("quick", 0.2),
   844→        ],
   845→    }
   846→
   847→    # Modifier words that amplify intent
   848→    _AMPLIFIERS = {
   849→        "full": 1.5, "deep": 1.5, "complete": 1.4, "everything": 1.5,
   850→        "all": 1.3, "thorough": 1.4, "comprehensive": 1.4, "maximum": 1.5,
   851→        "aggressive": 1.3, "exhaustive": 1.5, "entire": 1.3,
   852→    }
   853→
   854→    # Modifier words that reduce intent
   855→    _DAMPENERS = {
   856→        "quick": 0.7, "fast": 0.7, "brief": 0.6, "simple": 0.7,
   857→        "just": 0.7, "only": 0.6, "basic": 0.6, "lightweight": 0.6,
   858→    }
   859→
   860→    # Module-specific keywords
   861→    _MODULE_KEYWORDS: Dict[str, List[str]] = {
   862→        "recon": ["recon", "reconnaissance", "probe", "fingerprint"],
   863→        "auth": ["auth", "login", "bypass", "session", "cookie"],
   864→        "chain": ["ssrf", "chain", "redirect", "open redirect"],
   865→        "gorgon": ["sqli", "xss", "injection", "gorgon"],
   866→        "oblivion": ["oblivion", "dread", "deep scan", "ultimate"],
   867→        "bot": ["bot", "crawler", "c2", "automation"],
   868→        "nhi": ["nhi", "cloud", "identity", "oauth"],
   869→        "vibesec": ["vibesec", "vibe", "benchmark", "score"],
   870→    }
   871→
   872→    def __init__(self, registry: ToolRegistry, memory: AgentMemory):
   873→        self.registry = registry
   874→        self.memory = memory
   875→
   876→    def classify_intent(self, goal: str) -> Dict[str, float]:
   877→        """Score each intent based on the goal text.
   878→
   879→        Uses weighted keyword matching, proximity scoring, and modifiers.
   880→        Returns a dict mapping intent names to float scores in [0, 1+].
   881→        """
   882→        goal_lower = goal.lower()
   883→        # Pad for whole-word matching
   884→        padded = f' {goal_lower} '
   885→        scores: Dict[str, float] = {}
   886→        words = goal_lower.split()
   887→
   888→        # Calculate global modifier
   889→        modifier = 1.0
   890→        for w in words:
   891→            if w in self._AMPLIFIERS:
   892→                modifier = max(modifier, self._AMPLIFIERS[w])
   893→            if w in self._DAMPENERS:
   894→                modifier = min(modifier, self._DAMPENERS[w])
   895→
   896→        for intent, keywords in self._INTENT_KEYWORDS.items():
   897→            score = 0.0
   898→            for pattern, weight in keywords:
   899→                # Use whole-word matching to avoid false positives
   900→                # (e.g., "repo" matching inside "report")
   901→                if len(pattern.split()) > 1:
   902→                    # Multi-word pattern: check substring (already specific enough)
   903→                    matched = pattern in goal_lower
   904→                else:
   905→                    # Single-word pattern: check whole word with padding
   906→                    matched = f' {pattern} ' in padded
   907→                if matched:
   908→                    # Boost if near a target-like word
   909→                    proximity_boost = 1.0
   910→                    pattern_idx = goal_lower.find(pattern)
   911→                    for tw in (".com", ".io", ".net", ".org", "http", "localhost"):
   912→                        tw_idx = goal_lower.find(tw)
   913→                        if tw_idx >= 0 and abs(pattern_idx - tw_idx) < 40:
   914→                            proximity_boost = 1.3
   915→                            break
   916→                    score += weight * proximity_boost
   917→            scores[intent] = min(score * modifier, 2.0)
   918→
   919→        return scores
   920→
   921→    def extract_targets(self, goal: str) -> List[str]:
   922→        """Extract targets (domains, IPs, URLs, paths) from the goal."""
   923→        urls = re.findall(r'https?://[\w\-._~:/?#\[\]@!$&\'()*+,;=%]+', goal)
   924→        domains = re.findall(r'[\w][\w\-]*\.[\w]{2,}(?:\.[\w]{2,})?', goal)
   925→        skip = {
   926→            'the', 'and', 'for', 'with', 'all', 'of', 'to', 'from', 'that', 'this',
   927→            'scan', 'audit', 'check', 'find', 'get', 'show', 'me', 'my',
   928→        }
   929→        targets = list(set(
   930→            urls + [d for d in domains if d.lower() not in skip and len(d) > 4]
   931→        ))
   932→        return targets
   933→
   934→    def detect_target_type(self, goal: str, targets: List[str]) -> str:
   935→        """Classify the target type from context."""
   936→        goal_lower = goal.lower()
   937→        local_indicators = [
   938→            'machine', 'laptop', 'computer', 'local', 'my system', 'this machine',
   939→            'localhost', 'my mac', 'my pc', 'my server',
   940→        ]
   941→        for ind in local_indicators:
   942→            if ind in goal_lower:
   943→                return "local"
   944→        dev_indicators = ['project', 'codebase', 'repo', 'source code', 'code review']
   945→        padded = f' {goal_lower} '
   946→        for ind in dev_indicators:
   947→            if f' {ind} ' in padded and not targets:
   948→                return "codebase"
   949→        if targets:
   950→            for t in targets:
   951→                if t.startswith("http"):
   952→                    return "remote"
   953→                if re.match(r'^\d+\.\d+\.\d+\.\d+$', t):
   954→                    return "remote"
   955→            return "remote"
   956→        return "unknown"
   957→
   958→    def detect_modules(self, goal: str) -> Optional[List[str]]:
   959→        """Detect specific module requests from the goal text."""
   960→        goal_lower = goal.lower()
   961→        for mod_name, keywords in self._MODULE_KEYWORDS.items():
   962→            for kw in keywords:
   963→                if kw in goal_lower:
   964→                    return [mod_name]
   965→        return None
   966→
   967→    def generate_plan(
   968→        self, goal: str, scores: Dict[str, float],
   969→        targets: List[str], target_type: str,
   970→    ) -> List[Dict[str, Any]]:
   971→        """Generate an ordered execution plan based on classified intents.
   972→
   973→        Returns a list of step dicts: {"tool", "args", "reason"}
   974→        """
   975→        plan: List[Dict[str, Any]] = []
   976→        domain = ""
   977→        if targets:
   978→            domain = targets[0].replace("https://", "").replace("http://", "").split("/")[0]
   979→
   980→        # Sort intents by score, take significant ones
   981→        sorted_intents = sorted(scores.items(), key=lambda x: x[1], reverse=True)
   982→        significant = [(name, s) for name, s in sorted_intents if s > 0.2]
   983→
   984→        # Check prior context for this target
   985→        prior = None
   986→        if domain:
   987→            prior = self.memory.get_target_context(domain)
   988→
   989→        # ── Subdomain recon phase ──
   990→        if any(n == "subdomain_recon" and s > 0.3 for n, s in significant):
   991→            if domain:
   992→                plan.append({
   993→                    "tool": "subdomain_discover",
   994→                    "args": {"domain": domain},
   995→                    "reason": "User requested subdomain reconnaissance",
   996→                })
   997→
   998→        # ── Benchmarking ──
   999→        if any(n == "benchmarking" and s > 0.3 for n, s in significant):
  1000→            if domain:
  1001→                plan.append({
  1002→                    "tool": "vibesec_benchmark",
  1003→                    "args": {"target": domain},
  1004→                    "reason": "User wants a quick security benchmark",
  1005→                })
  1006→
  1007→        # ── Local audit ──
  1008→        if target_type == "local":
  1009→            if any(n == "compliance" and s > 0.3 for n, s in significant):
  1010→                plan.append({"tool": "doctor_check", "args": {}, "reason": "Health/compliance check requested"})
  1011→            if any(n == "local_audit" and s > 0.2 for n, s in significant):
  1012→                plan.append({"tool": "local_audit", "args": {"target_path": "."}, "reason": "Local machine audit requested"})
  1013→
  1014→        # ── Dev security ──
  1015→        if target_type == "codebase" or any(n == "dev_security" and s > 0.3 for n, s in significant):
  1016→            plan.append({"tool": "dev_scan", "args": {"path": "."}, "reason": "Developer project scan requested"})
  1017→
  1018→        # ── Secret hunting ──
  1019→        if any(n == "secret_hunting" and s > 0.3 for n, s in significant):
  1020→            plan.append({"tool": "secret_hunt", "args": {"path": "."}, "reason": "Secret hunting requested"})
  1021→
  1022→        # ── Code analysis ──
  1023→        if any(n == "code_analysis" and s > 0.3 for n, s in significant):
  1024→            plan.append({"tool": "ast_analyze", "args": {"path": "."}, "reason": "Static code analysis requested"})
  1025→
  1026→        # ── Full remote scan ──
  1027→        is_full = any(n == "full_scan" and s > 0.2 for n, s in significant)
  1028→        if targets and target_type == "remote":
  1029→            modules = self.detect_modules(goal)
  1030→            if len(targets) > 1:
  1031→                # Multiple targets → blitz (even if full_scan is set)
  1032→                plan.append({
  1033→                    "tool": "blitz_scan",
  1034→                    "args": {"targets_str": ",".join(targets)},
  1035→                    "reason": f"Multiple targets detected ({len(targets)}), using parallel blitz scan",
  1036→                })
  1037→            elif is_full or not plan:
  1038→                plan.append({
  1039→                    "tool": "remote_scan",
  1040→                    "args": {
  1041→                        "target": targets[0],
  1042→                        "modules": ",".join(modules) if modules else "",
  1043→                        "all_modules": is_full and not modules,
  1044→                    },
  1045→                    "reason": "Full remote security scan" +
  1046→                              (" with all modules" if is_full and not modules else ""),
  1047→                })
  1048→
  1049→        # ── Attack simulation ──
  1050→        if any(n == "attack_simulation" and s > 0.3 for n, s in significant):
  1051→            if domain:
  1052→                plan.append({
  1053→                    "tool": "adversarial_loop",

  1131→    def adapt_plan(
  1132→        self,
  1133→        plan: List[Dict[str, Any]],
  1134→        completed_results: List[Tuple[Dict, ToolResult]],
  1135→        current_idx: int,
  1136→    ) -> List[Dict[str, Any]]:
  1137→        """Adapt the remaining plan based on results seen so far.
  1138→
  1139→        This is the 'intelligence' of the rule engine — it inspects intermediate
  1140→        results and inserts new steps or modifies remaining ones.
  1141→        """
  1142→        additions: List[Dict[str, Any]] = []
  1143→        domain = ""
  1144→        last_score = None
  1145→        has_critical = False
  1146→        total_findings = 0
  1147→        all_findings: List[Dict] = []
  1148→        subdomains_found = False
  1149→
  1150→        for step, result in completed_results:
  1151→            d = result.data
  1152→            if isinstance(d, dict):
  1153→                # Extract score from scan results
  1154→                if "total_score" in d:
  1155→                    last_score = d["total_score"]
  1156→                if "vibesec_score" in d:
  1157→                    last_score = d["vibesec_score"]
  1158→                # Collect findings
  1159→                findings = d.get("findings", [])
  1160→                if isinstance(findings, list):
  1161→                    all_findings.extend(findings)
  1162→                    total_findings += len(findings)
  1163→                    for f in findings:
  1164→                        if isinstance(f, dict) and f.get("severity") in ("critical", "high"):
  1165→                            has_critical = True
  1166→                # Check for subdomains
  1167→                if "subdomains" in d and len(d["subdomains"]) > 3:
  1168→                    subdomains_found = True
  1169→                # Get target from data
  1170→                if not domain:
  1171→                    domain = d.get("target", "")
  1172→
  1173→        # If any findings found → add CVE enrichment (if not already planned)
  1174→        has_any_findings = total_findings > 0
  1175→        if has_any_findings and not any(s["tool"] == "cve_enrich" for s in plan[current_idx:]):
  1176→            if all_findings:
  1177→                try:
  1178→                    findings_json = json.dumps(all_findings[:20])
  1179→                except (TypeError, ValueError):
  1180→                    findings_json = "[]"
  1181→                reason = "Enriching findings with CVE data from NVD"
  1182→                if has_critical:
  1183→                    reason = "Critical/high findings detected — enriching with CVE data"
  1184→                additions.append({
  1185→                    "tool": "cve_enrich",
  1186→                    "args": {"findings_json": findings_json},
  1187→                    "reason": reason,
  1188→                })
  1189→
  1190→        # If many subdomains found and not already doing blitz → suggest blitz
  1191→        if subdomains_found and not any(s["tool"] == "blitz_scan" for s in plan[current_idx:]):
  1192→            # Don't add blitz automatically (too aggressive), but add knowledge graph
  1193→            if domain and not any(s["tool"] == "knowledge_graph" for s in plan[current_idx:]):
  1194→                additions.append({
  1195→                    "tool": "knowledge_graph",
  1196→                    "args": {"action": "build", "target": domain},
  1197→                    "reason": "Large attack surface detected — building knowledge graph",
  1198→                })
  1199→
  1200→        # If score is very low → add knowledge graph
  1201→        if last_score is not None and last_score < 40:
  1202→            if domain and not any(s["tool"] == "knowledge_graph" for s in plan[current_idx:]):
  1203→                additions.append({
  1204→                    "tool": "knowledge_graph",
  1205→                    "args": {"action": "build", "target": domain},
  1206→                    "reason": f"Critical score ({last_score}/100) — analyzing attack surface via knowledge graph",
  1207→                })
  1208→
  1209→        # If score is low and no adversarial loop → add one
  1210→        if last_score is not None and last_score < 50:
  1211→            if domain and not any(s["tool"] in ("adversarial_loop", "swarm_run") for s in plan + additions):
  1212→                additions.append({
  1213→                    "tool": "adversarial_loop",
  1214→                    "args": {"target": domain, "rounds": 2},
  1215→                    "reason": f"Low score ({last_score}/100) — running adversarial loop for remediation guidance",
  1216→                })
  1217→
  1218→        # Insert additions after current step
  1219→        remaining = plan[current_idx + 1:]
  1220→        plan = plan[:current_idx + 1] + additions + remaining
  1221→
  1222→        return plan

  1365→        """Execute using LLM tool-calling loop."""
  1366→        tool_descs = "\n".join(
  1367→            f"  - {t.name}: {t.description}" for t in self.registry.tools.values()

  1373→

  1386→                else:

  1391→

  1393→                self._emit("[yellow]LLM returned empty response — falling back to rules.[/]")
  1394→                return self._execute_rules(goal)
  1395→
  1396→            content = response.get("content", "")
  1397→            tool_calls = response.get("tool_calls", [])
  1398→
  1399→            # If no tool calls, the LLM is done
  1400→            if not tool_calls:
  1401→                if content:
  1402→                    self._emit(f"  [dim]{content[:500]}[/]")
  1403→                break
  1404→
  1405→            # Execute each tool call

  1407→                fn_name = tc.get("function", {}).get("name", "")
  1408→                fn_args_str = tc.get("function", {}).get("arguments", "{}")

  1428→                        "role": "tool",
  1429→                        "tool_call_id": tc.get("id", f"call_{fn_name}"),
  1430→                        "content": result.output[:4000],
  1431→                    })
  1432→                else:
  1433→                    messages.append({
  1434→                        "role": "user",

  1436→                    })

  1514→    # ── RULE-BASED EXECUTION ──────────────────────────────────────────

  1516→    def _execute_rules(self, goal: str) -> Dict[str, Any]:
  1517→        """Execute using the intelligent rule-based planner."""
  1518→        # THINK phase
  1519→        scores = self.planner.classify_intent(goal)
  1520→        targets = self.planner.extract_targets(goal)
  1521→        target_type = self.planner.detect_target_type(goal, targets)
  1522→
  1523→        domain = ""
  1524→        if targets:
  1525→            domain = targets[0].replace("https://", "").replace("http://", "").split("/")[0]
  1526→
  1527→        # Build intent display
  1528→        top_intents = sorted(scores.items(), key=lambda x: x[1], reverse=True)[:4]
  1529→        intent_str = " + ".join(f"{n} ({s:.1f})" for n, s in top_intents if s > 0.1)
  1530→
  1531→        self._print_header(goal, "rule-based planner")
  1532→        self._emit(f"  [THINK] Analyzing goal...")
  1533→        self._emit(f"    [dim]→ Target: {domain or target_type} ({target_type})[/]")
  1534→        if intent_str:
  1535→            self._emit(f"    [dim]→ Intent: {intent_str}[/]")
  1536→        self._emit(f"    [dim]→ Strategy: plan → execute → adapt[/]")
  1537→
  1538→        # PLAN phase
  1539→        plan = self.planner.generate_plan(goal, scores, targets, target_type)
  1540→        self._emit(f"")
  1541→        self._emit(f"  [PLAN] {len(plan)} step(s):")
  1542→        for i, step in enumerate(plan, 1):
  1543→            self._emit(f"    {i}. [bold]{step['tool']}[/] — {step['reason']}")
  1544→
  1545→        # EXECUTE phase with adaptation
  1546→        completed: List[Tuple[Dict, ToolResult]] = []
  1547→        step_idx = 0
  1548→        total_steps = len(plan)
  1549→
