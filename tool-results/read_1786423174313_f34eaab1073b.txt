   100→   349→        # Tech stack changes
   101→   350→        events.extend(self._diff_list(
   102→   351→            ts, "TECH_CHANGE", snapshot_a.tech_stack, snapshot_b.tech_stack
   103→   352→        ))
   104→   353→
   105→   354→        # Port changes
   106→   355→        events.extend(self._diff_list(
   107→   356→            ts, "PORT_CHANGE",
   108→   357→            [str(p) for p in snapshot_a.open_ports],
   109→   358→            [str(p) for p in snapshot_b.open_ports],
   110→   359→        ))
   111→   360→
   112→   361→        # Security header changes
   113→   362→        events.extend(self._diff_security_headers(
   114→   363→            ts, snapshot_a.security_headers, snapshot_b.security_headers
   115→   364→        ))
   116→   365→
   117→   366→        # Content hash change
   118→   367→        if (snapshot_a.page_hash and snapshot_b.page_hash
   119→   368→                and snapshot_a.page_hash != snapshot_b.page_hash):
   120→   369→            events.append(DriftEvent(
   121→   370→                timestamp=ts,
   122→   371→                field_changed="CONTENT_CHANGE",
   123→   372→                old_value=snapshot_a.page_hash,
   124→   373→                new_value=snapshot_b.page_hash,
   125→   374→                severity="low",
   126→   375→                risk_impact="Page content changed",
   127→   376→            ))
   128→   377→
   129→   378→        # Status code change
   130→   379→        if snapshot_a.status_code != snapshot_b.status_code:
   131→   380→            sev = "high" if snapshot_b.status_code >= 500 else "medium"
   132→   381→            events.append(DriftEvent(
   133→   382→                timestamp=ts,
   134→   383→                field_changed="STATUS_CHANGE",
   135→   384→                old_value=str(snapshot_a.status_code),
   136→   385→                new_value=str(snapshot_b.status_code),
   137→   386→                severity=sev,
   138→   387→                risk_impact=(
   139→   388→                    f"HTTP status changed from {snapshot_a.status_code} "
   140→   389→                    f"to {snapshot_b.status_code}"
   141→   390→                ),
   142→   391→            ))
   143→   392→
   144→   393→        # Body length change (significant)
   145→   394→        if snapshot_a.body_length > 0 and snapshot_b.body_length > 0:
   146→   395→            ratio = snapshot_b.body_length / max(snapshot_a.body_length, 1)
   147→   396→            if ratio > 1.5 or ratio < 0.5:
   148→   397→                events.append(DriftEvent(
   149→   398→                    timestamp=ts,
   150→   399→                    field_changed="CONTENT_CHANGE",
   151→   400→                    old_value=str(snapshot_a.body_length),
   152→   401→                    new_value=str(snapshot_b.body_length),
   153→   402→                    severity="medium",
   154→   403→                    risk_impact=(
   155→   404→                        f"Body length changed by {abs(ratio - 1) * 100:.0f}%"
   156→   405→                    ),
   157→   406→                ))
   158→   407→
   159→   408→        return events

   161→   410→    def detect_drift(
   162→   411→        self, target: str, base_url: str, timeout: int = 8
   163→   412→    ) -> Dict[str, Any]:
   164→   413→        """Take a snapshot and compare against the last saved one."""
   165→   414→        current = self.take_snapshot(target, base_url, timeout=timeout)
   166→   415→        self.save_snapshot(current)
   167→   416→
   168→   417→        previous = self._load_latest_snapshot(target, exclude_current=True)
   169→   418→        if previous is None:
   170→   419→            return {
   171→   420→                "target": target,
   172→   421→                "drift_detected": False,
   173→   422→                "message": "No previous snapshot to compare against",
   174→   423→                "current_snapshot": current.to_dict(),
   175→   424→                "drift_events": [],
   176→   425→                "risk_score": 0.0,
   177→   426→            }
   178→   427→
   179→   428→        events = self.compare_snapshots(previous, current)
   180→   429→        risk_score = self._scorer.score_drift(events)
   181→   430→
   182→   431→        return {
   183→   432→            "target": target,
   184→   433→            "drift_detected": len(events) > 0,
   185→   434→            "drift_event_count": len(events),
   186→   435→            "previous_timestamp": previous.timestamp,
   187→   436→            "current_timestamp": current.timestamp,
   188→   437→            "drift_events": [e.to_dict() for e in events],
   189→   438→            "risk_score": round(risk_score, 1),
   190→   439→            "current_snapshot": current.to_dict(),
   191→   440→        }

   202→   451→    def load_snapshots(self, target: str) -> List[DriftSnapshot]:
   203→   452→        """Load all saved snapshots for a target, newest first."""
   204→   453→        target_dir = self._target_dir(target)

   206→   455→            return []

   208→   457→        for p in sorted(target_dir.glob("snapshot_*.json"), reverse=True):

   217→   466→    def analyze_drift_trend(self, target: str) -> Dict[str, Any]:
   218→   467→        """Analyze drift patterns over time.
   219→   468→
   220→   469→        Returns trend analysis: increasing, decreasing, cyclical,
   221→   470→        or stable, along with category breakdowns.
   222→   471→        """
   223→   472→        snapshots = self.load_snapshots(target)
   224→   473→        if len(snapshots) < 2:
   225→   474→            return {
   226→   475→                "target": target,
   227→   476→                "trend": "insufficient_data",
   228→   477→                "snapshot_count": len(snapshots),
   229→   478→                "category_breakdown": {},
   230→   479→                "details": [],
   231→   480→            }
   232→   481→
   233→   482→        all_events: List[DriftEvent] = []
   234→   483→        for i in range(len(snapshots) - 1):
   235→   484→            events = self.compare_snapshots(snapshots[i + 1], snapshots[i])
   236→   485→            all_events.extend(events)
   237→   486→
   238→   487→        # Category breakdown
   239→   488→        category_counts: Dict[str, int] = {}
   240→   489→        for ev in all_events:
   241→   490→            cat = ev.field_changed
   242→   491→            category_counts[cat] = category_counts.get(cat, 0) + 1
   243→   492→
   244→   493→        # Group events by snapshot pair to detect trends
   245→   494→        pairs: List[int] = []
   246→   495→        for i in range(len(snapshots) - 1):
   247→   496→            pair_events = self.compare_snapshots(
   248→   497→                snapshots[i + 1], snapshots[i]
   249→   498→            )
   250→   499→            pairs.append(len(pair_events))
   251→   500→
   252→   501→        # Determine trend
   253→   502→        trend = self._determine_trend(pairs)
   254→   503→
   255→   504→        # Time between changes
   256→   505→        change_intervals: List[str] = []
   257→   506→        for i in range(len(snapshots) - 1):
   258→   507→            events = self.compare_snapshots(snapshots[i + 1], snapshots[i])
   259→   508→            if events:
   260→   509→                try:
   261→   510→                    ts_a = datetime.fromisoformat(snapshots[i + 1].timestamp)
   262→   511→                    ts_b = datetime.fromisoformat(snapshots[i].timestamp)
   263→   512→                    delta = abs((ts_b - ts_a).total_seconds())
   264→   513→                    if delta < 3600:
   265→   514→                        interval = f"{int(delta // 60)} minutes"
   266→   515→                    elif delta < 86400:
   267→   516→                        interval = f"{delta / 3600:.1f} hours"
   268→   517→                    else:
   269→   518→                        interval = f"{delta / 86400:.1f} days"
   270→   519→                    change_intervals.append(interval)
   271→   520→                except (ValueError, TypeError):
   272→   521→                    pass
   273→   522→
   274→   523→        return {
   275→   524→            "target": target,
   276→   525→            "trend": trend,
   277→   526→            "snapshot_count": len(snapshots),
   278→   527→            "total_drift_events": len(all_events),
   279→   528→            "category_breakdown": category_counts,
   280→   529→            "events_per_snapshot_pair": pairs,
   281→   530→            "change_intervals": change_intervals,
   282→   531→            "risk_trend": self._risk_trend(snapshots),
   283→   532→            "details": [e.to_dict() for e in all_events],
   284→   533→        }

   360→   609→    def _capture_dns(
   361→   610→        self, host: str, timeout: int = 8
   362→   611→    ) -> Dict[str, Any]:
   363→   612→        """Resolve DNS and return structured records."""
   364→   613→        records: Dict[str, Any] = {}
   365→   614→        try:
   366→   615→            answers = socket.getaddrinfo(host, None)
   367→   616→            a_records = sorted({addr[4][0] for addr in answers if addr[4]})
   368→   617→            records["A"] = a_records
   369→   618→            records["AAAA"] = [
   370→   619→                addr[4][0] for addr in answers
   371→   620→                if addr[0] == socket.AF_INET6 and addr[4]
   372→   621→            ]
   373→   622→        except (socket.gaierror, OSError) as exc:
   374→   623→            records["error"] = str(exc)
   375→   624→
   376→   625→        # Check for CNAME-like patterns
   377→   626→        records["hostname"] = host
   378→   627→        return records

   380→   629→    def _capture_tls_cert(
   381→   630→        self, host: str, timeout: int = 8
   382→   631→    ) -> Dict[str, Any]:
   383→   632→        """Fetch the TLS certificate details."""
   384→   633→        cert_info: Dict[str, Any] = {}
   385→   634→        try:
   386→   635→            ctx = ssl.create_default_context()
   387→   636→            ctx.check_hostname = False
   388→   637→            ctx.verify_mode = ssl.CERT_NONE
   389→   638→            with socket.create_connection((host, 443), timeout=timeout) as sock:
   390→   639→                with ctx.wrap_socket(sock, server_hostname=host) as tls:
   391→   640→                    cert = tls.getpeercert()
   392→   641→                    if cert:
   393→   642→                        cert_info["subject"] = {
   394→   643→                            k: v for item in cert.get("subject", [])
   395→   644→                            for k, v in item
   396→   645→                        }
   397→   646→                        cert_info["issuer"] = {
   398→   647→                            k: v for item in cert.get("issuer", [])
   399→   648→                            for k, v in item
   400→   649→                        }
   401→   650→                        cert_info["not_before"] = cert.get("notBefore", "")
   402→   651→                        cert_info["not_after"] = cert.get("notAfter", "")
   403→   652→                        cert_info["serial"] = cert.get("serialNumber", "")
   404→   653→                        cert_info["version"] = cert.get("version", "")
   405→   654→                        cert_info["san"] = [
   406→   655→                            v for item in cert.get("subjectAltName", [])
   407→   656→                            for v in item[1:]  # skip the type tag
   408→   657→                        ]
   409→   658→        except (socket.timeout, socket.gaierror, ssl.SSLError, OSError) as exc:
   410→   659→            cert_info["error"] = str(exc)
   411→   660→        return cert_info

   479→   728→                events.append(DriftEvent(

   505→   754→        for item in sorted(added):

   512→   761→                risk_impact=f"{item} added",

   568→   817→        first_half = pairs[: len(pairs) // 2]
   569→   818→        second_half = pairs[len(pairs) // 2 :]
   570→   819→        avg_first = sum(first_half) / len(first_half)
   571→   820→        avg_second = sum(second_half) / len(second_half)
   572→   821→
   573→   822→        if avg_first == 0 and avg_second == 0:
   574→   823→            return "stable"
   575→   824→        if avg_first == 0:
   576→   825→            return "increasing"
   577→   826→
   578→   827→        ratio = avg_second / avg_first
   579→   828→        if ratio > 1.5:
   580→   829→            return "increasing"
   581→   830→        elif ratio < 0.5:
   582→   831→            return "decreasing"
   583→   832→        else:
   584→   833→            # Check for cyclical pattern (alternating high/low)
   585→   834→            if len(pairs) >= 4:
   586→   835→                alternating = 0
   587→   836→                for i in range(1, len(pairs)):
   588→   837→                    if pairs[i] == pairs[i - 1]:
   589→   838→                        continue
   590→   839→                    if i == 1:
   591→   840→                        alternating += 1
   592→   841→                        continue
   593→   842→                    curr_up = pairs[i] > pairs[i - 1]
   594→   843→                    prev_up = pairs[i - 1] > pairs[i - 2]
   595→   844→                    if prev_up != curr_up:
   596→   845→                        alternating += 1
   597→   846→                if alternating > len(pairs) * 0.6:
   598→   847→                    return "cyclical"
   599→   848→            return "stable"

   601→   850→    def _risk_trend(self, snapshots: List[DriftSnapshot]) -> str:
   602→   851→        """Determine if overall risk is trending up or down."""
   603→   852→        if len(snapshots) < 3:
   604→   853→            return "insufficient_data"
   605→   854→
   606→   855→        scores: List[float] = []
   607→   856→        for i in range(len(snapshots) - 1):
   608→   857→            events = self.compare_snapshots(snapshots[i + 1], snapshots[i])
   609→   858→            scores.append(self._scorer.score_drift(events))
   610→   859→
   611→   860→        recent = scores[: max(1, len(scores) // 2)]
   612→   861→        older = scores[max(1, len(scores) // 2) :]
   613→   862→        avg_recent = sum(recent) / len(recent)
   614→   863→        avg_older = sum(older) / len(older) if older else 0.0
   615→   864→
   616→   865→        if avg_recent > avg_older * 1.3:
   617→   866→            return "increasing_risk"
   618→   867→        elif avg_recent < avg_older * 0.7:
   619→   868→            return "decreasing_risk"
   620→   869→        return "stable_risk"
