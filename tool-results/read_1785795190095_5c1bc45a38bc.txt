    85→    findings: List[Dict[str, Any]],
    86→) -> Dict[str, Dict[str, int]]:
    87→    """Count findings per module per severity."""
    88→    matrix: Dict[str, Dict[str, int]] = {}
    89→    for f in findings:

    97→def _aggregate_dread(findings: List[Dict[str, Any]]) -> List[float]:
    98→    """Aggregate DREAD components across all findings.
    99→
   100→    Some modules store dread as a float (average score), others as a dict
   101→    with component breakdowns. We handle both cases.
   102→
   103→    Returns list of 5 floats: [damage, reproducibility, exploitability, affected_users, discoverability]
   104→    Averaged across all findings that have dread data.
   105→    """
   106→    sums = [0.0] * 5
   107→    counts = [0] * 5
   108→
   109→    for f in findings:
   110→        dread = f.get("dread")
   111→        if dread is None:
   112→            continue
   113→
   114→        if isinstance(dread, dict):
   115→            # Dict form: {"damage": 10, "reproducibility": 9, ...}
   116→            for i, key in enumerate(_DREAD_LABELS):
   117→                v = dread.get(key)
   118→                if v is not None and isinstance(v, (int, float)):
   119→                    sums[i] += float(v)
   120→                    counts[i] += 1
   121→        elif isinstance(dread, (int, float)):
   122→            # Float form: overall DREAD score — distribute evenly
   123→            val = float(dread)
   124→            if val > 0:
   125→                for i in range(5):
   126→                    sums[i] += val
   127→                    counts[i] += 1
   128→
   129→    # Average each component
   130→    avg = []
   131→    for i in range(5):
   132→        avg.append(round(sums[i] / counts[i], 1) if counts[i] > 0 else 0.0)
   133→    return avg

   136→def _top_finding_dread(findings: List[Dict[str, Any]], n: int = 8) -> List[Dict[str, Any]]:
   137→    """Get top N findings with their DREAD breakdowns for grouped bar chart."""
   138→    # Sort by severity, then by points deducted
   139→    sev_order = {"critical": 0, "high": 1, "medium": 2, "low": 3, "info": 4}
   140→    sorted_f = sorted(
   141→        findings,
   142→        key=lambda f: (sev_order.get(f.get("severity", "info").lower(), 99), -f.get("points_deducted", 0)),
   143→    )
   144→
   145→    result = []
   146→    for f in sorted_f[:n]:
   147→        dread = f.get("dread")
   148→        if isinstance(dread, dict):
   149→            components = [dread.get(key, 0) for key in _DREAD_LABELS]
   150→        elif isinstance(dread, (int, float)) and float(dread) > 0:
   151→            # Distribute evenly
   152→            val = float(dread)
   153→            components = [val] * 5
   154→        else:
   155→            components = None
   156→
   157→        if components:
   158→            result.append({
   159→                "title": f.get("title", "")[:30],
   160→                "severity": f.get("severity", "info").lower(),
   161→                "components": components,
   162→            })
   163→    return result

   205→def _risk_posture(score: int, grade: str, sc: Dict[str, int]) -> Tuple[str, str]:
   206→    """Determine executive risk posture label and description."""
   207→    crit = sc.get("critical", 0)
   208→    high = sc.get("high", 0)
   209→
   210→    if score >= 90 and crit == 0 and high == 0:
   211→        return ("Excellent", "Target demonstrates strong security posture with no critical or high-severity findings. Minor informational items may be addressed opportunistically.")
   212→    elif score >= 75 and crit == 0:
   213→        return ("Good", "Overall security posture is sound. A small number of high-severity findings should be prioritized for remediation to reach best-in-class standing.")
   214→    elif score >= 50 and crit <= 2:
   215→        return ("Moderate", "Target presents a moderate attack surface. Critical and high findings require immediate attention. A structured remediation plan is recommended.")
   216→    elif score >= 25:
   217→        return ("Elevated", "Significant security weaknesses detected across multiple categories. Exploit chain potential exists. Prioritize critical findings and conduct a focused remediation sprint.")
   218→    else:
   219→        return ("Critical", "Severe security posture with widespread vulnerabilities. Immediate containment and remediation is required before any production exposure. Consider a dedicated security engagement.")
