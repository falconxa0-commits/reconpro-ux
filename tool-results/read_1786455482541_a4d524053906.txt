  1519→# ---------------------------------------------------------------------------
  1520→# TTPMatcher — matches observed TTPs against APT group databases
  1521→# ---------------------------------------------------------------------------
  1522→
  1523→class TTPMatcher:
  1524→    """Matches observed tactics, techniques, and procedures against the
  1525→    APT group database to identify potential threat actors.
  1526→
  1527→    The matcher uses a weighted scoring algorithm that considers:
  1528→      - Jaccard similarity of technique sets
  1529→      - Rarity weighting (less common techniques score higher)
  1530→      - Campaign overlap bonuses
  1531→      - Temporal activity signals
  1532→    """
  1533→
  1534→    def __init__(self, group_db: Optional[Dict[str, Any]] = None) -> None:
  1535→        self._groups = group_db if group_db is not None else APT_GROUPS
  1536→        self._technique_frequency: Dict[str, int] = {}
  1537→        self._compute_technique_frequency()
  1538→
  1539→    def _compute_technique_frequency(self) -> None:
  1540→        """Pre-compute how many groups use each technique for rarity weighting."""
  1541→        for group_data in self._groups.values():
  1542→            for tech in group_data.get("techniques", []):
  1543→                self._technique_frequency[tech] = (
  1544→                    self._technique_frequency.get(tech, 0) + 1
  1545→                )
  1546→
  1547→    def _technique_rarity(self, technique: str, total_groups: int) -> float:
  1548→        """Compute rarity score for a technique. Rarer = more discriminating.
  1549→
  1550→        Uses inverse frequency weighting: techniques used by fewer groups
  1551→        contribute more to attribution confidence.
  1552→        """
  1553→        freq = self._technique_frequency.get(technique, 0)
  1554→        if freq == 0:
  1555→            return 1.0
  1556→        # Inverse frequency: 1 for unique, approaching 0 for universal
  1557→        return math.log2(total_groups / (freq + 1)) / math.log2(total_groups)
  1558→
  1559→    def match_observed_ttps(
  1560→        self, observed_techniques: List[str],
  1561→    ) -> List[APTAttribution]:
  1562→        """Match observed TTPs against all APT groups and return a ranked list.
  1563→
  1564→        Args:
  1565→            observed_techniques: List of MITRE ATT&CK technique IDs observed
  1566→                                 in findings (e.g., ["T1566.001", "T1059.001"]).
  1567→
  1568→        Returns:
  1569→            Ranked list of :class:`APTAttribution` sorted by confidence
  1570→            (highest first).
  1571→        """
  1572→        total_groups = len(self._groups)
  1573→        results: List[APTAttribution] = []
  1574→
  1575→        if not observed_techniques:
  1576→            return results
  1577→
  1578→        observed_set = set(observed_techniques)
  1579→
  1580→        for group_id, group_data in self._groups.items():
  1581→            group_techs = set(group_data.get("techniques", []))
  1582→            overlap = observed_set & group_techs
  1583→
  1584→            if not overlap:
  1585→                continue
  1586→
  1587→            # --- Jaccard similarity ---
  1588→            jaccard = len(overlap) / len(observed_set | group_techs)
  1589→
  1590→            # --- Coverage: what fraction of observed techniques are explained ---
  1591→            coverage = len(overlap) / len(observed_set)
  1592→
  1593→            # --- Weighted rarity score ---
  1594→            rarity_sum = sum(
  1595→                self._technique_rarity(t, total_groups) for t in overlap
  1596→            )
  1597→            rarity_avg = rarity_sum / len(overlap)
  1598→
  1599→            # --- Composite confidence ---
  1600→            confidence = 0.35 * jaccard + 0.40 * coverage + 0.25 * rarity_avg
  1601→            confidence = min(confidence, 1.0)
  1602→
  1603→            reasoning: List[str] = []
  1604→            reasoning.append(
  1605→                f"Jaccard similarity: {jaccard:.4f} "
  1606→                f"({len(overlap)} shared techniques)"
  1607→            )
  1608→            reasoning.append(f"Coverage: {coverage:.2%} of observed TTPs")
  1609→            reasoning.append(f"Technique rarity factor: {rarity_avg:.4f}")
  1610→
  1611→            # Campaign overlap bonus
  1612→            matching_campaigns = self._find_matching_campaigns(
  1613→                group_id, overlap
  1614→            )
  1615→            if matching_campaigns:
  1616→                bonus = min(len(matching_campaigns) * 0.05, 0.15)
  1617→                confidence = min(confidence + bonus, 1.0)
  1618→                reasoning.append(
  1619→                    f"Campaign overlap bonus (+{bonus:.2f}): "
  1620→                    f"{', '.join(matching_campaigns)}"
  1621→                )
  1622→
  1623→            results.append(
  1624→                APTAttribution(
  1625→                    group_id=group_id,
  1626→                    group_name=group_data.get("name", group_id),
  1627→                    confidence=confidence,
  1628→                    confidence_level=self._confidence_level(confidence),
  1629→                    matching_techniques=sorted(overlap),
  1630→                    matching_campaigns=matching_campaigns,
  1631→                    country=group_data.get("country", "Unknown"),
  1632→                    motivation=group_data.get("motivation", "Unknown"),
  1633→                    sophistication=group_data.get("sophistication", "Unknown"),
  1634→                    reasoning=reasoning,
  1635→                )
  1636→            )
  1637→
  1638→        results.sort(key=lambda a: a.confidence, reverse=True)
  1639→        return results
  1640→
  1641→    def calculate_confidence(
  1642→        self,
  1643→        group_profile: Dict[str, Any],
  1644→        observed_findings: List[Dict[str, Any]],
  1645→    ) -> float:
  1646→        """Calculate a confidence score (0.0–1.0) that the observed findings
  1647→        align with the given APT group profile.
  1648→
  1649→        Args:
  1650→            group_profile: APT group dict from the database.
  1651→            observed_findings: List of finding dicts with at least ``"type"``
  1652→                               and ``"evidence"`` keys.
  1653→
  1654→        Returns:
  1655→            Float confidence between 0 and 1.
  1656→        """
  1657→        if not observed_findings or not group_profile:
  1658→            return 0.0
  1659→
  1660→        # Extract technique IDs from findings if present
  1661→        observed_techs: Set[str] = set()
  1662→        observed_sectors: Set[str] = set()
  1663→        for finding in observed_findings:
  1664→            tech = finding.get("technique_id") or finding.get("ttp") or ""
  1665→            if tech:
  1666→                observed_techs.add(tech)
  1667→            if "sector" in finding:
  1668→                observed_sectors.add(finding["sector"])
  1669→
  1670→        group_techs = set(group_profile.get("techniques", []))
  1671→        group_sectors = set(
  1672→            s.lower() for s in group_profile.get("sectors", [])
  1673→        )
  1674→
  1675→        tech_overlap = len(observed_techs & group_techs)
  1676→        total_relevant = len(observed_techs | group_techs)
  1677→
  1678→        if total_relevant == 0:
  1679→            return 0.0
  1680→
  1681→        jaccard = tech_overlap / total_relevant
  1682→
  1683→        # Sector alignment bonus
  1684→        sector_bonus = 0.0
  1685→        if observed_sectors:
  1686→            matching_sectors = len(
  1687→                observed_sectors & group_sectors
  1688→            )
  1689→            sector_bonus = 0.1 * (
  1690→                matching_sectors / max(len(observed_sectors), 1)
  1691→            )
  1692→
  1693→        # Sophistication factor
  1694→        soph = group_profile.get("sophistication", "moderate")
  1695→        soph_weight = {"minimal": 0.6, "moderate": 0.75,
  1696→                       "high": 0.9, "very_high": 1.0}
  1697→        soph_factor = soph_weight.get(soph, 0.75)
  1698→
  1699→        confidence = (0.8 * jaccard + sector_bonus) * soph_factor
  1700→        return round(min(max(confidence, 0.0), 1.0), 4)

  1703→        """Generate an IOC watch profile for a target based on top

  1708→
  1709→        Returns:
  1710→            An :class:`IOCProfile` with recommended IOCs to monitor.

  1713→        return IOCProfile(

  1718→            watch_ips=[],
  1719→            watch_hashes=[],

  1723→        )

  1725→    def _generate_patterns(self, target: str) -> List[str]:
  1726→        """Generate regex patterns for phishing and malware IOC detection."""
  1727→        domain = target.replace("https://", "").replace("http://", "").split("/")[0]
  1728→        patterns: List[str] = []
  1729→        patterns.append(
  1730→            rf"(?i)login.*{re.escape(domain)}"  # Phishing login page
  1731→        )
  1732→        patterns.append(
  1733→            rf"(?i)account.*verify.*{re.escape(domain)}"
  1734→        )
  1735→        patterns.append(
  1736→            rf"(?i)update.*{re.escape(domain)}.*\.exe"
  1737→        )
  1738→        patterns.append(
  1739→            rf"(?i){re.escape(domain)}.*\.php\?.*=.{8,}"
  1740→        )
  1741→        patterns.append(
  1742→            rf"(?i)base64.*[A-Za-z0-9+/]{{200,}}"  # Encoded payload
  1743→        )
  1744→        return patterns
  1745→
  1746→    def _generate_watch_domains(self, target: str) -> List[str]:
  1747→        """Generate domain watchlist entries."""
  1748→        base = target.replace("https://", "").replace("http://", "").split("/")[0]
  1749→        parts = base.split(".")
  1750→        domains: List[str] = []
  1751→        if len(parts) >= 2:
  1752→            # Typosquatting variants
  1753→            domains.append(
  1754→                parts[0] + "-" + ".".join(parts[1:])
  1755→            )
  1756→            domains.append(
  1757→                parts[0] + parts[0] + "." + ".".join(parts[1:])
  1758→            )
  1759→            for swap in ("rn", "m", "vv", "l1", "0o"):
  1760→                domains.append(
  1761→                    parts[0].replace("n", swap[0]).replace("m", swap)
  1762→                    + "." + ".".join(parts[1:])
  1763→                )
  1764→        return list(set(domains))

  1766→    def _generate_detection_rules(self) -> List[str]:

  1768→        return [

  1772→            "Watch for TLS certificate changes on target infrastructure",

  1775→            "Monitor for credential dump tool execution (Mimikatz, LSASS access)",
  1776→            "Detect unusual process injection into legitimate system binaries",

  1779→    def _find_matching_campaigns(

  1782→        """Find known campaigns that match both the group and technique set."""

  1786→                camp_techs = set(campaign.get("techniques", []))

  1796→        elif confidence >= 0.70:

  1798→        elif confidence >= 0.40:
  1799→            return ConfidenceLevel.MODERATE
  1800→        elif confidence >= 0.20:
  1801→            return ConfidenceLevel.LOW
  1802→        elif confidence > 0.0:
  1803→            return ConfidenceLevel.MINIMAL
  1804→        return ConfidenceLevel.NONE

  1807→# ---------------------------------------------------------------------------
  1808→# CampaignTracker — cross-reference findings with known campaigns
  1809→# ---------------------------------------------------------------------------
  1810→
  1811→class CampaignTracker:
  1812→    """Tracks and identifies campaign signatures by cross-referencing
  1813→    observed findings against the known campaigns database.
  1814→
  1815→    Campaign identification uses multi-factor analysis:
  1816→      - Technique overlap (MITRE ATT&CK)
  1817→      - Sector alignment
  1818→      - Temporal correlation
  1819→      - Known infrastructure reuse
  1820→    """
  1821→
  1822→    def __init__(
  1823→        self,
  1824→        campaigns: Optional[List[Dict[str, Any]]] = None,
  1825→        groups: Optional[Dict[str, Any]] = None,
  1826→    ) -> None:
  1827→        self._campaigns = campaigns if campaigns is not None else KNOWN_CAMPAIGNS
  1828→        self._groups = groups if groups is not None else APT_GROUPS
  1829→
  1830→    def identify_campaign(
  1831→        self,
  1832→        target: str,
  1833→        findings: List[Dict[str, Any]],
  1834→    ) -> List[CampaignMatch]:
  1835→        """Identify which known campaigns the target's findings align with.
  1836→
  1837→        Args:
  1838→            target: The target being analyzed.
  1839→            findings: List of finding dicts. Each should ideally contain
  1840→                      ``"technique_id"``, ``"sector"``, and ``"evidence"``.
  1841→
  1842→        Returns:
  1843→            Ranked list of :class:`CampaignMatch` objects.
  1844→        """
  1845→        observed_techs: Set[str] = set()
  1846→        observed_sectors: Set[str] = set()
  1847→        current_year = datetime.now().year
  1848→
  1849→        for finding in findings:
  1850→            tech = finding.get("technique_id") or finding.get("ttp") or ""
  1851→            if tech:
  1852→                observed_techs.add(tech)
  1853→            sector = finding.get("sector") or finding.get("industry") or ""
  1854→            if sector:
  1855→                observed_sectors.add(sector.lower())
  1856→
  1857→        if not observed_techs:
  1858→            return []
  1859→
  1860→        results: List[CampaignMatch] = []
  1861→
  1862→        for campaign in self._campaigns:
  1863→            camp_techs = set(campaign.get("techniques", []))
  1864→            tech_overlap = list(observed_techs & camp_techs)
  1865→
  1866→            if not tech_overlap:
  1867→                continue
  1868→
  1869→            # Technique overlap score
  1870→            tech_score = len(tech_overlap) / len(observed_techs | camp_techs)
  1871→
  1872→            # Sector overlap
  1873→            camp_sectors = set(
  1874→                s.lower() for s in campaign.get("sectors", [])
  1875→            )
  1876→            sector_overlap = list(observed_sectors & camp_sectors)
  1877→
  1878→            # Temporal factor: campaigns active in the last 5 years score higher
  1879→            year = int(campaign.get("year", 2010))
  1880→            age = current_year - year
  1881→            temporal = max(0.0, 1.0 - (age / 20.0))
  1882→
  1883→            # Group attribution overlap
  1884→            attributed = campaign.get("attributed_to", [])
  1885→
  1886→            # Composite score
  1887→            sector_bonus = 0.1 if sector_overlap else 0.0
  1888→            confidence = (0.5 * tech_score + 0.3 * temporal + sector_bonus)
  1889→            confidence = min(confidence, 1.0)
  1890→
  1891→            results.append(
  1892→                CampaignMatch(
  1893→                    campaign_id=campaign.get("id", ""),
  1894→                    campaign_name=campaign.get("name", ""),
  1895→                    attributed_groups=attributed,
  1896→                    confidence=confidence,
  1897→                    technique_overlap=tech_overlap,
  1898→                    sector_overlap=sector_overlap,
  1899→                    description=campaign.get("description", ""),
  1900→                )
  1901→            )
  1902→
  1903→        results.sort(key=lambda m: m.confidence, reverse=True)
  1904→        return results
  1905→
  1906→    def track_campaign_signature(
  1907→        self, findings: List[Dict[str, Any]],
  1908→    ) -> Dict[str, Any]:
  1909→        """Generate a unique campaign signature hash from findings.
  1910→
  1911→        This fingerprint can be used to cluster similar attack campaigns
  1912→        and detect repetitions across different targets.
  1913→
  1914→        Args:
  1915→            findings: List of finding dicts.
  1916→
  1917→        Returns:
  1918→            Dict with ``signature_hash``, ``technique_fingerprint``,
  1919→            ``sector_fingerprint``, and ``composite_hash``.
  1920→        """
  1921→        techs: List[str] = sorted(
  1922→            {f.get("technique_id", "") or f.get("ttp", "")
  1923→             for f in findings if f.get("technique_id") or f.get("ttp")}
  1924→        )
  1925→        sectors: List[str] = sorted(
  1926→            {f.get("sector", "") for f in findings if f.get("sector")}
  1927→        )
  1928→
  1929→        tech_str = "|".join(techs)
  1930→        sector_str = "|".join(sectors)
  1931→
  1932→        tech_hash = hashlib.sha256(tech_str.encode()).hexdigest()[:16]
  1933→        sector_hash = hashlib.sha256(sector_str.encode()).hexdigest()[:16]
  1934→        composite = hashlib.sha256(
  1935→            (tech_str + "||" + sector_str).encode()
  1936→        ).hexdigest()[:20]
  1937→
  1938→        return {
  1939→            "signature_hash": composite,
  1940→            "technique_fingerprint": tech_hash,
  1941→            "sector_fingerprint": sector_hash,
  1942→            "technique_count": len(techs),
  1943→            "sector_count": len(sectors),
  1944→            "techniques": techs,
  1945→            "sectors": sectors,
  1946→        }
  1947→
  1948→    def cross_reference_active_campaigns(
  1949→        self, target: str, findings: List[Dict[str, Any]],
  1950→    ) -> Dict[str, Any]:
  1951→        """Cross-reference findings against all active campaigns and
  1952→        return a combined analysis.
  1953→
  1954→        Returns:
  1955→            Dict with ``campaign_matches``, ``signature``, and ``summary``.
  1956→        """
  1957→        matches = self.identify_campaign(target, findings)
  1958→        signature = self.track_campaign_signature(findings)
  1959→
  1960→        active_campaigns = [m for m in matches if m.confidence >= 0.3]
  1961→
  1962→        return {
  1963→            "campaign_matches": [m.to_dict() for m in matches],
  1964→            "active_campaigns": [m.to_dict() for m in active_campaigns],
  1965→            "signature": signature,
  1966→            "summary": (
  1967→                f"{len(matches)} campaign(s) identified, "
  1968→                f"{len(active_campaigns)} with moderate or higher confidence. "
  1969→                f"Signature: {signature['signature_hash']}"
  1970→            ),
  1971→        }
