  1040→    (re.compile(r"credentials\.json|service-account", re.I), "GCP service account credentials"),
  1041→]
  1042→
  1043→# Secret environment variable patterns
  1044→_SECRET_ENV_PATTERNS = re.compile(
  1045→    r"(?:PASSWORD|SECRET|TOKEN|API_KEY|PRIVATE_KEY|AUTH_TOKEN|ACCESS_KEY)"
  1046→    r"\s*=\s*[\"'](?!").+[\"']",
  1047→    re.I,
  1048→)
  1049→
  1050→# Dangerous ports to expose
  1051→_DANGEROUS_PORTS: Dict[str, str] = {
  1052→    "22": "SSH",
  1053→    "2375": "Docker daemon (unencrypted)",
  1054→    "2376": "Docker daemon (TLS)",
  1055→    "6379": "Redis",
  1056→    "27017": "MongoDB",
  1057→    "5432": "PostgreSQL",
  1058→    "3306": "MySQL",
  1059→    "9200": "Elasticsearch",
  1060→    "11211": "Memcached",
  1061→}

  1063→
  1064→def parse_dockerfile(path: str) -> List[Dict[str, Any]]:
  1065→    """Parse a Dockerfile and return a list of issue dicts."""
  1066→    issues: List[Dict[str, Any]] = []
  1067→    fname = os.path.basename(path)
  1068→
  1069→    try:
  1070→        text = Path(path).read_text(encoding="utf-8", errors="replace")
  1071→    except OSError:
  1072→        return issues
  1073→
  1074→    lines = text.splitlines()
  1075→    has_user = False
  1076→    has_healthcheck = False
  1077→    has_from = False
  1078→    from_line = 0
  1079→    user_line_num = 0
  1080→
  1081→    for idx, line in enumerate(lines):
  1082→        stripped = line.strip()
  1083→        if not stripped or stripped.startswith("#"):
  1084→            continue
  1085→
  1086→        ln = idx + 1
  1087→
  1088→        # ── FROM checks ───────────────────────────────────────────────
  1089→        if stripped.upper().startswith("FROM "):
  1090→            has_from = True
  1091→            from_line = ln
  1092→            from_val = stripped[5:].strip()
  1093→
  1094→            # Check: FROM scratch
  1095→            if from_val.lower() == "scratch":
  1096→                issues.append({
  1097→                    "title": "Dockerfile Uses FROM scratch",
  1098→                    "severity": "medium", "category": "container_image",
  1099→                    "description": "FROM scratch produces a minimal image but provides no package management or shell for debugging.",
  1100→                    "evidence": _ev(fname, ln),
  1101→                    "remediation": "Use a minimal base like alpine or distroless if possible.",
  1102→                    "dread_score": _dread("medium"), "points_deducted": _pts("medium"),
  1103→                })
  1104→            # Check: FROM without tag (mutable image)
  1105→            elif ":" not in from_val.split("@")[0]:
  1106→                issues.append({
  1107→                    "title": "Dockerfile FROM Without Specific Tag",
  1108→                    "severity": "medium", "category": "container_image",
  1109→                    "description": f"Base image '{from_val}' has no specific tag, using 'latest' which is mutable.",
  1110→                    "evidence": _ev(fname, ln),
  1111→                    "remediation": f"Pin the image: {from_val}:<specific-tag> or use a digest @{from_val}@sha256:...",
  1112→                    "dread_score": _dread("medium"), "points_deducted": _pts("medium"),
  1113→                })
  1114→            # Check: FROM using :latest tag
  1115→            elif from_val.endswith(":latest"):
  1116→                issues.append({
  1117→                    "title": "Dockerfile FROM Uses :latest Tag",
  1118→                    "severity": "medium", "category": "container_image",
  1119→                    "description": f"Base image uses ':latest' tag which is mutable and non-reproducible.",
  1120→                    "evidence": _ev(fname, ln),
  1121→                    "remediation": f"Pin to a specific version tag instead of :latest.",
  1122→                    "dread_score": _dread("medium"), "points_deducted": _pts("medium"),
  1123→                })
  1124→
  1125→        # ── USER checks ───────────────────────────────────────────────
  1126→        if stripped.upper().startswith("USER "):
  1127→            user_val = stripped[5:].strip()
  1128→            has_user = True
  1129→            user_line_num = ln
  1130→
  1131→            if user_val in ("root", "0"):
  1132→                issues.append({
  1133→                    "title": "Dockerfile Explicitly Runs as Root",
  1134→                    "severity": "high", "category": "container_privilege",
  1135→                    "description": "USER root/0 is explicitly set, running the container with full root privileges.",
  1136→                    "evidence": _ev(fname, ln),
  1137→                    "remediation": "Create a non-root user and set USER <username>.",
  1138→                    "dread_score": _dread("high"), "points_deducted": _pts("high"),
  1139→                })
  1140→
  1141→        # ── COPY / ADD checks ─────────────────────────────────────────
  1142→        if stripped.upper().startswith(("COPY ", "ADD ")):
  1143→            # Check for sensitive files
  1144→            for pat, desc in _SENSITIVE_FILE_PATTERNS:
  1145→                if pat.search(stripped):
  1146→                    issues.append({
  1147→                        "title": f"Dockerfile Copies Sensitive File ({desc})",
  1148→                        "severity": "critical", "category": "secrets_leak",
  1149→                        "description": f"COPY or ADD instruction includes {desc} in the image.",
  1150→                        "evidence": _ev(fname, ln),
  1151→                        "remediation": f"Do not copy {desc} into the image. Use runtime secrets or multi-stage builds.",
  1152→                        "dread_score": _dread("critical"), "points_deducted": _pts("critical"),
  1153→                    })
  1154→                    break
  1155→
  1156→            # Check: ADD with remote URL (supply chain risk)
  1157→            if re.search(r'ADD\s+https?://', stripped, re.I):
  1158→                issues.append({
  1159→                    "title": "Dockerfile ADD from Remote URL",
  1160→                    "severity": "high", "category": "supply_chain",
  1161→                    "description": "ADD instruction fetches from a remote URL, creating a supply chain risk.",
  1162→                    "evidence": _ev(fname, ln),
  1163→                    "remediation": "Use RUN curl/wget with checksum verification instead of ADD for remote URLs.",
  1164→                    "dread_score": _dread("high"), "points_deducted": _pts("high"),
  1165→                })
  1166→
  1167→        # ── ENV checks ────────────────────────────────────────────────
  1168→        if stripped.upper().startswith("ENV "):
  1169→            env_val = stripped[4:].strip()
  1170→            if _SECRET_ENV_PATTERNS.search(env_val):
  1171→                issues.append({
  1172→                    "title": "Dockerfile ENV Contains Secret",
  1173→                    "severity": "high", "category": "hardcoded_secret",
  1174→                    "description": "An ENV variable appears to contain a hardcoded secret/password/token.",
  1175→                    "evidence": _ev(fname, ln),
  1176→                    "remediation": "Use Docker secrets, Kubernetes secrets, or runtime environment injection.",
  1177→                    "dread_score": _dread("high"), "points_deducted": _pts("high"),
  1178→                })
  1179→
  1180→        # ── RUN checks ────────────────────────────────────────────────
  1181→        if stripped.upper().startswith("RUN "):
  1182→            run_val = stripped[4:].strip()
  1183→
  1184→            # curl|wget piped to bash/sh
  1185→            if re.search(r'(?:curl|wget)\s+.*\|\s*(?:ba)?sh', run_val):
  1186→                issues.append({
  1187→                    "title": "Dockerfile RUN Pipes Remote Script to Shell",
  1188→                    "severity": "high", "category": "supply_chain",
  1189→                    "description": "RUN instruction pipes curl/wget output directly to shell — supply chain risk.",
  1190→                    "evidence": _ev(fname, ln),
  1191→                    "remediation": "Download the script first, verify its checksum, then execute it.",
  1192→                    "dread_score": _dread("high"), "points_deducted": _pts("high"),
  1193→                })
  1194→
  1195→            # --privileged in RUN (docker run inside docker)
  1196→            if "--privileged" in run_val:
  1197→                issues.append({
  1198→                    "title": "Dockerfile RUN with --privileged Flag",
  1199→                    "severity": "critical", "category": "container_privilege",
  1200→                    "description": "RUN instruction uses --privileged, granting full host access to the build container.",
  1201→                    "evidence": _ev(fname, ln),
  1202→                    "remediation": "Remove --privileged flag. Grant only specific capabilities.",
  1203→                    "dread_score": _dread("critical"), "points_deducted": _pts("critical"),
  1204→                })
  1205→
  1206→            # Running as root in RUN (sudo/su)
  1207→            if re.search(r'\b(?:sudo|su\s)\b', run_val):
  1208→                issues.append({
  1209→                    "title": "Dockerfile RUN Uses sudo/su",
  1210→                    "severity": "medium", "category": "container_privilege",
  1211→                    "description": "RUN uses sudo/su, indicating the container may need root privileges at runtime.",
  1212→                    "evidence": _ev(fname, ln),
  1213→                    "remediation": "Avoid sudo/su. If root is needed for install, switch back to non-root user.",
  1214→                    "dread_score": _dread("medium"), "points_deducted": _pts("medium"),
  1215→                })
  1216→
  1217→            # apt-get without no-install-recommends
  1218→            if re.search(r'apt-get\s+install', run_val) and "--no-install-recommends" not in run_val:
  1219→                issues.append({
  1220→                    "title": "Dockerfile apt-get Without --no-install-recommends",
  1221→                    "severity": "low", "category": "image_size",
  1222→                    "description": "apt-get install without --no-install-recommends increases image size and attack surface.",
  1223→                    "evidence": _ev(fname, ln),
  1224→                    "remediation": "Add --no-install-recommends to apt-get install and combine RUN layers.",
  1225→                    "dread_score": _dread("low"), "points_deducted": _pts("low"),
  1226→                })
  1227→
  1228→            # apt-get update without cache cleanup in same layer
  1229→            if re.search(r'apt-get\s+update', run_val) and not re.search(r'rm\s+-rf\s+/var/lib/apt', run_val):
  1230→                issues.append({
  1231→                    "title": "Dockerfile apt-get Cache Not Cleaned",
  1232→                    "severity": "low", "category": "image_size",
  1233→                    "description": "apt-get update without cache cleanup in the same layer increases image size.",
  1234→                    "evidence": _ev(fname, ln),
  1235→                    "remediation": "Add '&& rm -rf /var/lib/apt/lists/*' in the same RUN instruction.",
  1236→                    "dread_score": _dread("low"), "points_deducted": _pts("low"),
  1237→                })
  1238→
  1239→        # ── EXPOSE checks ─────────────────────────────────────────────
  1240→        if stripped.upper().startswith("EXPOSE "):
  1241→            ports = stripped[7:].strip().split()
  1242→            for port in ports:
  1243→                port_clean = port.split("/")[0]
  1244→                if port_clean in _DANGEROUS_PORTS:
  1245→                    svc = _DANGEROUS_PORTS[port_clean]
  1246→                    issues.append({
  1247→                        "title": f"Dockerfile Exposes {svc} (Port {port_clean})",
  1248→                        "severity": "medium", "category": "network_exposure",
  1249→                        "description": f"Dockerfile exposes port {port_clean} ({svc}) which should not be publicly accessible.",
  1250→                        "evidence": _ev(fname, ln),
  1251→                        "remediation": f"Do not EXPOSE {svc} in production. Use Docker networks for internal communication.",
  1252→                        "dread_score": _dread("medium"), "points_deducted": _pts("medium"),
  1253→                    })
  1254→
  1255→        # ── HEALTHCHECK checks ────────────────────────────────────────
  1256→        if stripped.upper().startswith("HEALTHCHECK"):
  1257→            has_healthcheck = True
  1258→
  1259→    # ── Post-scan checks ──────────────────────────────────────────────
  1260→
  1261→    # Check: No USER directive (runs as root)
  1262→    if has_from and not has_user:
  1263→        issues.append({
  1264→            "title": "Dockerfile Has No USER Directive (Runs as Root)",
  1265→            "severity": "high", "category": "container_privilege",
  1266→            "description": "No USER instruction found — the container process runs as root by default.",
  1267→            "evidence": _ev(fname, from_line or 1),
  1268→            "remediation": "Add 'USER <non-root-user>' before the ENTRYPOINT/CMD.",
  1269→            "dread_score": _dread("high"), "points_deducted": _pts("high"),
  1270→        })
  1271→
  1272→    # Check: No HEALTHCHECK
  1273→    if has_from and not has_healthcheck:
  1274→        issues.append({
  1275→            "title": "Dockerfile Has No HEALTHCHECK Instruction",
  1276→            "severity": "low", "category": "monitoring",
  1277→            "description": "No HEALTHCHECK instruction — the orchestrator cannot determine if the container is healthy.",
  1278→            "evidence": _ev(fname, 1),
  1279→            "remediation": "Add a HEALTHCHECK instruction (e.g., HEALTHCHECK CMD curl -f http://localhost:8080/ || exit 1).",
  1280→            "dread_score": _dread("low"), "points_deducted": _pts("low"),
  1281→        })
  1282→
  1283→    # Check: No .dockerignore (info-level — check if .dockerignore exists alongside)
  1284→    parent = Path(path).parent
  1285→    dockerignore = parent / ".dockerignore"
  1286→    if parent.exists() and not dockerignore.exists():
  1287→        issues.append({
  1288→            "title": "No .dockerignore File Detected",
  1289→            "severity": "info", "category": "best_practice",
  1290→            "description": "No .dockerignore found in the Dockerfile's directory — sensitive files may be included in the build context.",
  1291→            "evidence": _ev(fname, 1),
  1292→            "remediation": "Create a .dockerignore file excluding .git, .env, *.pem, node_modules, etc.",
  1293→            "dread_score": _dread("info"), "points_deducted": _pts("info"),
  1294→        })
  1295→
  1296→    return issues

  1304→def _k8s_get_containers(doc: Dict[str, Any]) -> List[Dict[str, Any]]:
  1305→    """Extract all container definitions from a K8s manifest."""
  1306→    containers: List[Dict[str, Any]] = []
  1307→    spec = doc.get("spec", {})
  1308→    if not isinstance(spec, dict):
  1309→        return containers
  1310→
  1311→    for c in spec.get("containers", []):
  1312→        if isinstance(c, dict):
  1313→            containers.append(c)
  1314→    for c in spec.get("initContainers", []):
  1315→        if isinstance(c, dict):
  1316→            containers.append(c)
  1317→    return containers

  1320→def _k8s_get_pods(doc: Dict[str, Any]) -> List[Dict[str, Any]]:
  1321→    """Extract pod specs from a K8s manifest (handles Deployments, StatefulSets, etc.)."""
  1322→    specs: List[Dict[str, Any]] = []
  1323→    spec = doc.get("spec", {})
  1324→    if not isinstance(spec, dict):
  1325→        return specs
  1326→
  1327→    # Direct Pod spec
  1328→    if spec.get("containers") or spec.get("initContainers"):
  1329→        specs.append(spec)
  1330→    # Template spec (Deployment, StatefulSet, DaemonSet, Job, CronJob)
  1331→    template = spec.get("template", {})
  1332→    if isinstance(template, dict):
  1333→        pod_spec = template.get("spec", {})
  1334→        if isinstance(pod_spec, dict) and (pod_spec.get("containers") or pod_spec.get("initContainers")):
  1335→            specs.append(pod_spec)
  1336→        # CronJob nested template
  1337→        job_spec = template.get("spec", {})
  1338→        if isinstance(job_spec, dict):
  1339→            inner_template = job_spec.get("template", {})
  1340→            if isinstance(inner_template, dict):
  1341→                inner_spec = inner_template.get("spec", {})
  1342→                if isinstance(inner_spec, dict) and (inner_spec.get("containers") or inner_spec.get("initContainers")):
  1343→                    specs.append(inner_spec)
  1344→
  1345→    return specs

  1348→def _k8s_check_security_context(pod_spec: Dict[str, Any], doc_name: str, fname: str) -> List[Dict[str, Any]]:
  1349→    """Check pod and container security contexts."""
  1350→    issues: List[Dict[str, Any]] = []
  1351→
  1352→    # Pod-level security context
  1353→    pod_sc = pod_spec.get("securityContext", {})
  1354→    if not isinstance(pod_sc, dict):
  1355→        pod_sc = {}
  1356→
  1357→    # Check: privileged at pod level
  1358→    if pod_sc.get("privileged") is True:
  1359→        issues.append({
  1360→            "title": f"K8s: '{doc_name}' Pod securityContext.privileged=true",
  1361→            "severity": "critical", "category": "container_privilege",
  1362→            "description": f"Pod '{doc_name}' has privileged=true at pod level — full host access.",
  1363→            "evidence": f"{fname}: {doc_name} securityContext.privileged: true",
  1364→            "remediation": "Remove privileged: true. Grant only specific Linux capabilities.",
  1365→            "dread_score": _dread("critical"), "points_deducted": _pts("critical"),
  1366→        })
  1367→
  1368→    # Check: allowPrivilegeEscalation at pod level
  1369→    if pod_sc.get("allowPrivilegeEscalation") is True:
  1370→        issues.append({
  1371→            "title": f"K8s: '{doc_name}' Pod allowPrivilegeEscalation=true",
  1372→            "severity": "high", "category": "container_privilege",
  1373→            "description": f"Pod '{doc_name}' allows privilege escalation.",
  1374→            "evidence": f"{fname}: {doc_name} allowPrivilegeEscalation: true",
  1375→            "remediation": "Set allowPrivilegeEscalation: false.",
  1376→            "dread_score": _dread("high"), "points_deducted": _pts("high"),
  1377→        })
  1378→
  1379→    # Check: runAsUser: 0 at pod level
  1380→    if pod_sc.get("runAsUser") == 0:
  1381→        issues.append({
  1382→            "title": f"K8s: '{doc_name}' Pod Runs as Root (runAsUser: 0)",
  1383→            "severity": "high", "category": "container_privilege",
  1384→            "description": f"Pod '{doc_name}' explicitly runs as root (runAsUser: 0).",
  1385→            "evidence": f"{fname}: {doc_name} runAsUser: 0",
  1386→            "remediation": "Set runAsUser to a non-zero UID (e.g., 1000).",
  1387→            "dread_score": _dread("high"), "points_deducted": _pts("high"),
  1388→        })
  1389→
  1390→    # Check: readOnlyRootFilesystem not true at pod level
  1391→    if pod_sc.get("readOnlyRootFilesystem") is not True and "readOnlyRootFilesystem" in str(pod_sc):
  1392→        pass  # explicitly false — handled below
  1393→    elif "readOnlyRootFilesystem" not in pod_sc and pod_sc:
  1394→        issues.append({
  1395→            "title": f"K8s: '{doc_name}' Pod Writable Root Filesystem",
  1396→            "severity": "medium", "category": "container_hardening",
  1397→            "description": f"Pod '{doc_name}' does not set readOnlyRootFilesystem: true.",
  1398→            "evidence": f"{fname}: {doc_name} missing readOnlyRootFilesystem: true",
  1399→            "remediation": "Set readOnlyRootFilesystem: true in securityContext.",
  1400→            "dread_score": _dread("medium"), "points_deducted": _pts("medium"),
  1401→        })
  1402→
  1403→    # Container-level checks
  1404→    for container in _k8s_get_containers_from_spec(pod_spec):
  1405→        cname = container.get("name", "<unnamed>")
  1406→        csc = container.get("securityContext", {})
  1407→        if not isinstance(csc, dict):
  1408→            csc = {}
  1409→
  1410→        # Privileged at container level
  1411→        if csc.get("privileged") is True:
  1412→            issues.append({
  1413→                "title": f"K8s: '{doc_name}/{cname}' Container privileged=true",
  1414→                "severity": "critical", "category": "container_privilege",
  1415→                "description": f"Container '{cname}' in '{doc_name}' has privileged=true — full host access.",
  1416→                "evidence": f"{fname}: {doc_name}/{cname} securityContext.privileged: true",
  1417→                "remediation": "Remove privileged: true. Grant only specific Linux capabilities.",
  1418→                "dread_score": _dread("critical"), "points_deducted": _pts("critical"),
  1419→            })
  1420→
  1421→        # allowPrivilegeEscalation at container level
  1422→        if csc.get("allowPrivilegeEscalation") is True:
  1423→            issues.append({
  1424→                "title": f"K8s: '{doc_name}/{cname}' allowPrivilegeEscalation=true",
  1425→                "severity": "high", "category": "container_privilege",
  1426→                "description": f"Container '{cname}' allows privilege escalation.",
  1427→                "evidence": f"{fname}: {doc_name}/{cname} allowPrivilegeEscalation: true",
  1428→                "remediation": "Set allowPrivilegeEscalation: false.",
  1429→                "dread_score": _dread("high"), "points_deducted": _pts("high"),
  1430→            })
  1431→
  1432→        # runAsUser: 0 at container level
  1433→        if csc.get("runAsUser") == 0:
  1434→            issues.append({
  1435→                "title": f"K8s: '{doc_name}/{cname}' Runs as Root (runAsUser: 0)",
  1436→                "severity": "high", "category": "container_privilege",
  1437→                "description": f"Container '{cname}' explicitly runs as root.",
  1438→                "evidence": f"{fname}: {doc_name}/{cname} runAsUser: 0",
  1439→                "remediation": "Set runAsUser to a non-zero UID.",
  1440→                "dread_score": _dread("high"), "points_deducted": _pts("high"),
  1441→                })
  1442→
  1443→        # Writable root filesystem at container level
  1444→        if csc.get("readOnlyRootFilesystem") is not True and pod_sc.get("readOnlyRootFilesystem") is not True:
  1445→            issues.append({
  1446→                "title": f"K8s: '{doc_name}/{cname}' Writable Root Filesystem",
  1447→                "severity": "medium", "category": "container_hardening",
  1448→                "description": f"Container '{cname}' has a writable root filesystem.",
  1449→                "evidence": f"{fname}: {doc_name}/{cname} missing readOnlyRootFilesystem: true",
  1450→                "remediation": "Set readOnlyRootFilesystem: true and use emptyDir for writable paths.",
  1451→                "dread_score": _dread("medium"), "points_deducted": _pts("medium"),
  1452→            })
  1453→
  1454→        # Capabilities check
  1455→        cap = csc.get("capabilities", {})
  1456→        if isinstance(cap, dict):
  1457→            adds = cap.get("add", [])
  1458→            if isinstance(adds, list):
  1459→                dangerous_caps = ["ALL", "NET_RAW", "SYS_ADMIN", "SYS_PTRACE", "SYS_MODULE",
  1460→                                  "DAC_OVERRIDE", "NET_ADMIN", "SYSLOG", "KILL"]
  1461→                for c in adds:
  1462→                    if c in dangerous_caps:
  1463→                        sev = "critical" if c == "ALL" else "high"
  1464→                        issues.append({
  1465→                            "title": f"K8s: '{doc_name}/{cname}' Adds Capability {c}",
  1466→                            "severity": sev, "category": "container_privilege",
  1467→                            "description": f"Container '{cname}' adds Linux capability {c}, expanding its privilege.",
  1468→                            "evidence": f"{fname}: {doc_name}/{cname} capabilities.add: [{c}]",

  1472→
  1473→        # No resource limits

  1507→
  1508→        # Secrets in env (plain text) vs envFrom/secretRef
  1509→        env_list = container.get("env", [])
  1510→        if isinstance(env_list, list):
  1511→            secret_env_names = []

  1514→                    name = env_var.get("name", "")

  1618→            # Root filesystem mounts
  1619→            dangerous_roots = ["/", "/etc", "/root", "/var", "/usr", "/bin", "/sbin", "/lib"]

  1680→def parse_k8s_yaml(path: str) -> List[Dict[str, Any]]:
  1681→    """Parse a Kubernetes YAML manifest and return a list of issue dicts."""
  1682→    issues: List[Dict[str, Any]] = []
  1683→    fname = os.path.basename(path)
  1684→
  1685→    try:
  1686→        text = Path(path).read_text(encoding="utf-8", errors="replace")
  1687→    except OSError:
  1688→        return issues
  1689→
  1690→    docs = _yaml_load_all(text)
  1691→
  1692→    for doc in docs:
  1693→        if not isinstance(doc, dict):
  1694→            continue
  1695→
  1696→        kind = str(doc.get("kind", ""))
  1697→        metadata = doc.get("metadata", {})
  1698→        doc_name = metadata.get("name", "<unnamed>") if isinstance(metadata, dict) else "<unnamed>"
  1699→
  1700→        pod_specs = _k8s_get_pods(doc)
  1701→        if not pod_specs:
  1702→            continue
  1703→
  1704→        for pod_spec in pod_specs:
  1705→            # Security context checks
  1706→            issues.extend(_k8s_check_security_context(pod_spec, doc_name, fname))
  1707→            # Host namespace checks
  1708→            issues.extend(_k8s_check_host_settings(pod_spec, doc_name, fname))
  1709→            # Volume mount checks
  1710→            issues.extend(_k8s_check_volumes(pod_spec, doc_name, fname))
  1711→            # Service account checks
  1712→            issues.extend(_k8s_check_service_account(pod_spec, doc_name, fname))
  1713→            # Toleration checks
  1714→            issues.extend(_k8s_check_tolerations(pod_spec, doc_name, fname))
  1715→
  1716→    return issues

  1731→def _is_likely_cloudformation(path: Path, text: str) -> bool:
  1732→    """Heuristic to determine if a YAML/JSON file is a CloudFormation template."""
  1733→    if re.search(r'AWSTemplateFormatVersion|AWS::', text):
  1734→        return True
  1735→    if re.search(r'Resources:\s*$', text, re.M):
  1736→        # Could be K8s or CF — check for CF-specific keys
  1737→        if re.search(r'Type:\s*AWS::|Description:\s*', text):
  1738→            return True
  1739→    return False

  1742→def _is_likely_k8s(path: Path, text: str) -> bool:
  1743→    """Heuristic to determine if a YAML file is a Kubernetes manifest."""
  1744→    return bool(re.search(r'apiVersion:\s*(?:v1|apps/|extensions/|batch/|networking\.k8s\.io|rbac\.authorization)', text))
  1745→

  1747→def _is_likely_docker_compose(path: Path, text: str) -> bool:
  1748→    """Heuristic to determine if a YAML file is a docker-compose file."""
  1749→    return bool(re.search(r'version:\s*["\']?3|services:\s*$', text, re.M))

  1752→def run(target: str, base_url: str, **kwargs) -> Tuple[List[Finding], int, str, str]:
  1753→    """Scan a directory for IaC security issues.
  1754→
  1755→    Recursively discovers Terraform, CloudFormation, Dockerfile, and
  1756→    Kubernetes manifest files, runs the appropriate parser, and
  1757→    converts all issues to Finding objects.
  1758→
  1759→    Returns:
  1760→        (findings, max_points, module_name, category)
  1761→    """
  1762→    findings: List[Finding] = []
  1763→    target_path = Path(target)
  1764→
  1765→    max_points = 100
  1766→    module_name = "iac_audit"
  1767→    category = "infrastructure_as_code"
  1768→
  1769→    if not target_path.is_dir():
  1770→        # Single file mode
  1771→        _scan_file(target_path, findings)
  1772→    else:
  1773→        # Recursive directory scan
  1774→        for p in sorted(target_path.rglob("*")):
  1775→            if not p.is_file():
  1776→                continue
  1777→            # Skip hidden dirs and common non-IaC dirs
  1778→            parts = p.parts
  1779→            if any(part.startswith(".") for part in parts if part != "."):
  1780→                continue
  1781→            if any(part in ("node_modules", "vendor", "__pycache__", ".git", "venv") for part in parts):
  1782→                continue
  1783→            _scan_file(p, findings)
  1784→
  1785→    # Deduplicate by title + evidence
  1786→    seen: set = set()
  1787→    deduped: List[Finding] = []
  1788→    for f in findings:
  1789→        key = (f.title, f.evidence)
  1790→        if key not in seen:
  1791→            seen.add(key)
  1792→            deduped.append(f)
  1793→
  1794→    return (deduped, max_points, module_name, category)

  1799→    name = p.name

  1801→
  1802→    try:
  1803→        text = p.read_text(encoding="utf-8", errors="replace")
  1804→    except OSError:
  1805→        return
  1806→
  1807→    # ── Terraform ─────────────────────────────────────────────────────
  1808→    if ext in _TF_EXTS:
  1809→        for issue in parse_tf_file(str(p)):
  1810→            findings.append(Finding(
  1811→                title=issue["title"],
  1812→                severity=issue["severity"],
  1813→                category=issue["category"],
  1814→                module="iac_audit",
  1815→                description=issue["description"],
  1816→                evidence=issue["evidence"],
  1817→                asset=str(p),
  1818→                points_deducted=issue["points_deducted"],
  1819→                remediation=issue["remediation"],
  1820→                dread_score=issue["dread_score"],
  1821→            }))
  1822→        return
  1823→
  1824→    # ── Dockerfile ─────────────────────────────────────────────────────
  1825→    if name in _DOCKERFILE_NAMES:
  1826→        for issue in parse_dockerfile(str(p)):
  1827→            findings.append(Finding(
  1828→                title=issue["title"],
  1829→                severity=issue["severity"],
  1830→                category=issue["category"],
  1831→                module="iac_audit",
  1832→                description=issue["description"],
  1833→                evidence=issue["evidence"],
  1834→                asset=str(p),
  1835→                points_deducted=issue["points_deducted"],
  1836→                remediation=issue["remediation"],
  1837→                dread_score=issue["dread_score"],
  1838→            }))
  1839→        return
  1840→
  1841→    # ── Docker Compose ────────────────────────────────────────────────
  1842→    if name in _DOCKER_COMPOSE_NAMES:
  1843→        # Analyze docker-compose files for security issues
  1844→        for issue in _parse_docker_compose(str(p), text):
  1845→            findings.append(Finding(
  1846→                title=issue["title"],
  1847→                severity=issue["severity"],
  1848→                category=issue["category"],
  1849→                module="iac_audit",
  1850→                description=issue["description"],
  1851→                evidence=issue["evidence"],
  1852→                asset=str(p),
  1853→                points_deducted=issue["points_deducted"],
  1854→                remediation=issue["remediation"],
  1855→                dread_score=issue["dread_score"],
  1856→            ))
  1857→        return
  1858→
  1859→    # ── YAML / JSON: CloudFormation or Kubernetes ──────────────────────
  1860→    if ext in _CF_EXTS:
  1861→        if _is_likely_cloudformation(p, text):
  1862→            for issue in parse_cf_file(str(p)):
  1863→                findings.append(Finding(
  1864→                    title=issue["title"],
  1865→                    severity=issue["severity"],
  1866→                    category=issue["category"],
  1867→                    module="iac_audit",
  1868→                    description=issue["description"],
  1869→                    evidence=issue["evidence"],
  1870→                    asset=str(p),
  1871→                    points_deducted=issue["points_deducted"],
  1872→                    remediation=issue["remediation"],
  1873→                    dread_score=issue["dread_score"],
  1874→                }))
  1875→        elif _is_likely_k8s(p, text):
  1876→            for issue in parse_k8s_yaml(str(p)):
  1877→                findings.append(Finding(
  1878→                    title=issue["title"],
  1879→                    severity=issue["severity"],
  1880→                    category=issue["category"],
  1881→                    module="iac_audit",
  1882→                    description=issue["description"],
  1883→                    evidence=issue["evidence"],
  1884→                    asset=str(p),
  1885→                    points_deducted=issue["points_deducted"],
  1886→                    remediation=issue["remediation"],
  1887→                    dread_score=issue["dread_score"],
  1888→                }))
  1889→        # If neither CF nor K8s, skip silently (not an IaC file)
  1890→

  1895→def _parse_docker_compose(path: str, text: str) -> List[Dict[str, Any]]:
  1896→    """Analyze docker-compose files for security issues."""
  1897→    issues: List[Dict[str, Any]] = []
  1898→    fname = os.path.basename(path)
  1899→    doc = _simple_yaml_load(text)
  1900→    if not isinstance(doc, dict):
  1901→        return issues
  1902→
  1903→    services = doc.get("services", {})
  1904→    if not isinstance(services, dict):
  1905→        return issues
  1906→
  1907→    for svc_name, svc_config in services.items():
  1908→        if not isinstance(svc_config, dict):
  1909→            continue
  1910→
  1911→        # Check: privileged
  1912→        if svc_config.get("privileged") is True:
  1913→            issues.append({
  1914→                "title": f"Docker Compose: '{svc_name}' Runs Privileged",
  1915→                "severity": "critical", "category": "container_privilege",
  1916→                "description": f"Service '{svc_name}' runs with --privileged flag.",
  1917→                "evidence": f"{fname}: services.{svc_name}.privileged: true",
  1918→                "remediation": "Remove privileged: true from the service.",
  1919→                "dread_score": _dread("critical"), "points_deducted": _pts("critical"),
  1920→            })
  1921→
  1922→        # Check: security_opt with no-new-privileges missing
  1923→        sec_opts = svc_config.get("security_opt", [])
  1924→        if isinstance(sec_opts, list) and "no-new-privileges:true" not in sec_opts and "no-new-privileges" not in sec_opts:
  1925→            issues.append({
  1926→                "title": f"Docker Compose: '{svc_name}' Missing no-new-privileges",
  1927→                "severity": "medium", "category": "container_privilege",
  1928→                "description": f"Service '{svc_name}' does not set security_opt: no-new-privileges:true.",
