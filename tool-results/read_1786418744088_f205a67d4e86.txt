   163→@dataclass
   164→class ChainPattern:
   165→    """A rule that describes how specific finding categories combine
   166→    into a compound attack path.
   167→
   168→    Parameters
   169→    ----------
   170→    name:
   171→        Human-readable name for the compound attack.
   172→    description:
   173→        One-line description of the attack scenario.
   174→    severity:
   175→        Resulting severity of the compound attack (uppercase).
   176→    required_categories:
   177→        The finding categories that must all be present to trigger this chain.
   178→        A finding matches if its ``category`` or ``type`` field contains
   179→        the pattern as a substring (case-insensitive).
   180→    required_tags:
   181→        Optional additional tags that at least one finding must carry.
   182→        Tags are checked against the ``tags`` list field of findings.
   183→    severity_multiplier:
   184→        Score multiplier applied on top of the base severity.
   185→    narrative_template:
   186→        A template string describing the compound attack.  May contain
   187→        ``{findings}`` placeholder (replaced with finding titles).
   188→    steps:
   189→        Ordered attack steps describing the exploit sequence.
   190→    remediation:
   191→        How to break this specific chain.
   192→    """
   193→
   194→    name: str
   195→    description: str
   196→    severity: str
   197→    required_categories: List[str]
   198→    required_tags: List[str] = field(default_factory=list)
   199→    severity_multiplier: float = 1.5
   200→    narrative_template: str = ""
   201→    steps: List[str] = field(default_factory=list)
   202→    remediation: str = ""
   203→
   204→
   205→@dataclass
   206→class ChainReport:
   207→    """A realized compound attack path, produced when a ChainPattern
   208→    matches against actual findings.
   209→
   210→    Parameters
   211→    ----------
   212→    name:
   213→        Chain name (from pattern or LLM).
   214→    severity:
   215→        Computed severity (one of info/low/medium/high/critical).
   216→    score:
   217→        Numeric risk score for ranking.
   218→    findings:
   219→        The individual finding dicts that form this chain.
   220→    narrative:
   221→        Human-readable attack story.
   222→    steps:
   223→        Ordered attack steps.
   224→    remediation:
   225→        How to break the chain.
   226→    confidence:
   227→        0.0 – 1.0 confidence that this chain is real and exploitable.
   228→    """
   229→
   230→    name: str
   231→    severity: str
   232→    score: float
   233→    findings: List[Dict[str, Any]]
   234→    narrative: str
   235→    steps: List[str]
   236→    remediation: str
   237→    confidence: float = 0.8
   238→
   239→    def to_dict(self) -> Dict[str, Any]:
   240→        """Serialise to a plain dict for JSON output / display."""
   241→        return {
   242→            "name": self.name,
   243→            "severity": self.severity,
   244→            "score": round(self.score, 2),
   245→            "finding_count": len(self.findings),
   246→            "finding_titles": [f.get("title", f.get("name", "")) for f in self.findings],
   247→            "narrative": self.narrative,
   248→            "steps": self.steps,
   249→            "remediation": self.remediation,
   250→            "confidence": round(self.confidence, 2),
   251→            "is_chain": True,
   252→        }

  1427→def _normalise_severity(raw: Any) -> str:
  1428→    """Return a normalised lowercase severity string."""
  1429→    if raw is None:
  1430→        return "info"
  1431→    s = str(raw).strip().lower()
  1432→    return s if s in SEVERITY_ORDER else "info"
  1433→
  1434→
  1435→def _sev_score(severity: str) -> float:
  1436→    """Map a severity string to a numeric score."""
  1437→    return _SEVERITY_SCORES.get(_normalise_severity(severity), 1.0)

  1439→
  1440→def _finding_category(finding: Dict[str, Any]) -> str:
  1441→    """Extract the category or type from a finding dict."""
  1442→    cat = finding.get("category", "") or finding.get("type", "") or finding.get("finding_type", "")
  1443→    return str(cat).strip().lower()
  1444→
  1445→
  1446→def _finding_tags(finding: Dict[str, Any]) -> List[str]:
  1447→    """Extract tags from a finding dict."""
  1448→    tags = finding.get("tags", [])
  1449→    if isinstance(tags, str):
  1450→        return [t.strip().lower() for t in tags.split(",") if t.strip()]
  1451→    return [str(t).strip().lower() for t in tags if t]

  1454→def _finding_title(finding: Dict[str, Any]) -> str:
  1455→    """Get a human-readable title for a finding."""
  1456→    return (
  1457→        finding.get("title", "")
  1458→        or finding.get("name", "")
  1459→        or finding.get("description", "")[:80]
  1460→        or "Unknown finding"
  1461→    )

  1464→def _format_narrative(template: str, matched_findings: List[Dict[str, Any]]) -> str:
  1465→    """Fill in a narrative template with finding titles.
  1466→
  1467→    Supports ``{findings[N]}`` placeholders where N is a 0-based index.
  1468→    Falls back to a plain concatenation if the template is empty.
  1469→    """
  1470→    if not template:
  1471→        titles = [_finding_title(f) for f in matched_findings]
  1472→        return "Combined findings: " + ", ".join(titles)
  1473→    try:
  1474→        return template.format(findings=[_finding_title(f) for f in matched_findings])
  1475→    except (KeyError, IndexError):
  1476→        titles = [_finding_title(f) for f in matched_findings]
  1477→        return "Combined findings: " + ", ".join(titles)

  1485→class AttackPathFinder:
  1486→    """Discovers compound attack paths by combining rule-based pattern
  1487→    matching, graph-based traversal, and optional LLM-assisted analysis.
  1488→
  1489→    Parameters
  1490→    ----------
  1491→    memory_store :
  1492→        Optional :class:`~reconpro.memory.UnifiedMemoryStore` instance for
  1493→        graph-based chain discovery.  When ``None``, only rule-based
  1494→        analysis is available.
  1495→    """
  1496→
  1497→    def __init__(self, memory_store: UnifiedMemoryStore | None = None) -> None:
  1498→        self._memory = memory_store
  1499→
  1500→    # ------------------------------------------------------------------
  1501→    # Public API
  1502→    # ------------------------------------------------------------------
  1503→
  1504→    def find_chains(
  1505→        self,
  1506→        target: str,
  1507→        findings: List[Dict[str, Any]],
  1508→    ) -> List[ChainReport]:
  1509→        """Run rule-based chain discovery against *findings* for *target*.
  1510→
  1511→        Groups findings by category, matches against all ``CHAIN_RULES``,
  1512→        and returns a ranked list of :class:`ChainReport` objects.
  1513→        """
  1514→        # Index findings by normalised category
  1515→        by_cat: Dict[str, List[Dict[str, Any]]] = defaultdict(list)
  1516→        for f in findings:
  1517→            cat = _finding_category(f)
  1518→            if cat:
  1519→                by_cat[cat].append(f)
  1520→
  1521→        # Also index by tags
  1522→        by_tag: Dict[str, List[Dict[str, Any]]] = defaultdict(list)
  1523→        for f in findings:
  1524→            for tag in _finding_tags(f):
  1525→                by_tag[tag].append(f)
  1526→
  1527→        reports: List[ChainReport] = []
  1528→        seen_rule_names: Set[str] = set()
  1529→
  1530→        for pattern in CHAIN_RULES:
  1531→            if pattern.name in seen_rule_names:
  1532→                continue
  1533→
  1534→            # Check all required categories are present
  1535→            matched_findings: List[Dict[str, Any]] = []
  1536→            all_cats_present = True
  1537→            for req_cat in pattern.required_categories:
  1538→                req_lower = req_cat.lower()
  1539→                # Check if any finding category contains the required category as substring
  1540→                found = False
  1541→                for existing_cat, cat_findings in by_cat.items():
  1542→                    if req_lower in existing_cat or existing_cat in req_lower:
  1543→                        if cat_findings:
  1544→                            matched_findings.append(cat_findings[0])
  1545→                            found = True
  1546→                            break
  1547→                if not found:
  1548→                    all_cats_present = False
  1549→                    break
  1550→
  1551→            if not all_cats_present:
  1552→                continue
  1553→
  1554→            # Check required tags (at least one finding must have at least one tag)
  1555→            if pattern.required_tags:
  1556→                req_tags_lower = [t.lower() for t in pattern.required_tags]
  1557→                tag_match = False
  1558→                for f in matched_findings:
  1559→                    f_tags = _finding_tags(f)
  1560→                    for rt in req_tags_lower:
  1561→                        if rt in f_tags or any(rt in ft for ft in f_tags):
  1562→                            tag_match = True
  1563→                            break
  1564→                    if tag_match:
  1565→                        break
  1566→                if not tag_match:
  1567→                    # Also check all findings, not just matched ones
  1568→                    for f in findings:
  1569→                        f_tags = _finding_tags(f)
  1570→                        for rt in req_tags_lower:
  1571→                            if rt in f_tags or any(rt in ft for ft in f_tags):
  1572→                                tag_match = True
  1573→                                break
  1574→                        if tag_match:
  1575→                            break
  1576→                if not tag_match:
  1577→                    continue
  1578→
  1579→            # Compute score
  1580→            max_child_sev = max(_sev_score(f.get("severity", "info")) for f in matched_findings)
  1581→            path_length_bonus = (len(matched_findings) - 2) * 1.5  # bonus for longer chains
  1582→            score = max_child_sev * pattern.severity_multiplier + max(path_length_bonus, 0.0)
  1583→
  1584→            # Build confidence based on how specific the match is
  1585→            confidence = 0.7 + (0.05 * len(pattern.required_categories))
  1586→            if pattern.required_tags:
  1587→                confidence += 0.1
  1588→            confidence = min(confidence, 0.95)
  1589→
  1590→            # Build the narrative
  1591→            narrative = _format_narrative(pattern.narrative_template, matched_findings)
  1592→
  1593→            # Build steps (use pattern steps if available)
  1594→            steps = pattern.steps if hasattr(pattern, "steps") and pattern.steps else []
  1595→            if not steps:
  1596→                steps = [f"Step {i+1}: Exploit {_finding_title(f)}" for i, f in enumerate(matched_findings)]
  1597→
  1598→            # Build remediation
  1599→            remediation = pattern.remediation if hasattr(pattern, "remediation") and pattern.remediation else (
  1600→                "Address each individual finding in the chain to break the compound attack path."
  1601→            )
  1602→
  1603→            report = ChainReport(
  1604→                name=pattern.name,
  1605→                severity=_normalise_severity(pattern.severity),
  1606→                score=round(score, 2),
  1607→                findings=list(matched_findings),
  1608→                narrative=narrative,
  1609→                steps=steps,
  1610→                remediation=remediation,
  1611→                confidence=round(confidence, 2),
  1612→            )
  1613→            reports.append(report)
  1614→            seen_rule_names.add(pattern.name)
  1615→
  1616→        # Sort by score descending
  1617→        reports.sort(key=lambda r: r.score, reverse=True)
  1618→        return reports
  1619→
  1620→    def find_compound_paths(self, target: str) -> List[ChainReport]:
  1621→        """Graph-based chain discovery using the memory store's knowledge graph.
  1622→
  1623→        Combines structural graph chains with rule-based matching to
  1624→        produce compound path reports.
  1625→
  1626→        Returns an empty list if no memory store is available.
  1627→        """
  1628→        if self._memory is None:
  1629→            return []
  1630→
  1631→        reports: List[ChainReport] = []
  1632→
  1633→        # 1. Get graph-based chains from the knowledge graph
  1634→        try:
  1635→            graph_chains = self._memory.find_chains(target)
  1636→        except Exception:
  1637→            graph_chains = []
  1638→
  1639→        # 2. Get all findings for the target
  1640→        try:
  1641→            all_findings = self._memory.get_findings(target)
  1642→        except Exception:
  1643→            all_findings = []
  1644→
  1645→        # Filter to actual vulnerability findings (not scan summaries)
  1646→        vuln_findings = [
  1647→            f for f in all_findings
  1648→            if f.get("severity", "info") not in ("info",)
  1649→            and f.get("title", "").find("Scan completed") == -1
  1650→        ]
  1651→
  1652→        if not vuln_findings:
  1653→            # Try rule-based analysis on whatever we have
  1654→            return self.find_chains(target, all_findings)
  1655→
  1656→        # 3. Rule-based chains
  1657→        rule_reports = self.find_chains(target, vuln_findings)
  1658→        reports.extend(rule_reports)
  1659→
  1660→        # 4. Graph-based chain enrichment
  1661→        seen_rule_names = {r.name for r in rule_reports}
  1662→        for chain in graph_chains:
  1663→            if not chain or len(chain) < 2:
  1664→                continue
  1665→
  1666→            # Extract finding nodes from the chain
  1667→            chain_findings = [
  1668→                step for step in chain
  1669→                if step.get("node_type") in ("finding", "vulnerability", "cve")
  1670→            ]
  1671→
  1672→            if len(chain_findings) < 2:
  1673→                continue
  1674→
  1675→            # Check if this chain is already covered by a rule
  1676→            chain_title_parts = [f.get("title", "") for f in chain_findings]
  1677→            covered = False
  1678→            for existing_name in seen_rule_names:
  1679→                # Simple check: if any two titles contain words from an existing rule's categories
  1680→                for part in chain_title_parts[:2]:
  1681→                    part_lower = part.lower()
  1682→                    if any(cat.lower() in part_lower for rule in CHAIN_RULES if rule.name == existing_name for cat in rule.required_categories):
  1683→                        covered = True
  1684→                        break
  1685→                if covered:
  1686→                    break
  1687→
  1688→            if covered:
  1689→                continue
  1690→
  1691→            # Build a generic graph-based chain report
  1692→            max_sev = max(_sev_score(f.get("severity", "info")) for f in chain_findings)
  1693→            path_len = len(chain_findings)
  1694→            score = max_sev * 1.3 + (path_len - 2) * 1.0
  1695→
  1696→            # Determine chain type from the path
  1697→            types_in_chain = [
  1698→                f.get("finding_type", f.get("type", "unknown"))
  1699→                for f in chain_findings
  1700→            ]
  1701→
  1702→            report = ChainReport(
  1703→                name=f"Graph Path: {' → '.join(types_in_chain[:3])}",
  1704→                severity=_normalise_severity(chain_findings[0].get("severity", "medium")),
  1705→                score=round(score, 2),
  1706→                findings=chain_findings,
  1707→                narrative=(
  1708→                    "Graph analysis discovered a multi-hop attack path: "
  1709→                    + " → ".join(f.get("title", "unknown") for f in chain_findings)
  1710→                    + ". These vulnerabilities are connected through shared assets "
  1711→                    "and can be chained for greater impact."
  1712→                ),
  1713→                steps=[
  1714→                    f"Step {i+1}: Exploit {_finding_title(f)} ({f.get('severity', '?')} severity)"
  1715→                    for i, f in enumerate(chain_findings)
  1716→                ],
  1717→                remediation=(
  1718→                    "Break this attack path by remediating the highest-severity finding "
  1719→                    f"in the chain: {chain_findings[0].get('title', 'unknown')}."
  1720→                ),
  1721→                confidence=0.5,  # Graph-based chains have lower confidence
  1722→            )
  1723→            reports.append(report)
  1724→
  1725→        # Deduplicate and sort
  1726→        reports.sort(key=lambda r: r.score, reverse=True)
  1727→        return reports

  1729→    def llm_assisted_chains(
  1730→        self,
  1731→        target: str,
  1732→        findings: List[Dict[str, Any]],
  1733→        llm_client: Any = None,
  1734→    ) -> List[ChainReport]:
  1735→        """Use an LLM (OpenAI or Anthropic) to discover additional chains.
  1736→
  1737→        Parameters
  1738→        ----------
  1739→        target :
  1740→            The target being analysed.
  1741→        findings :
  1742→            List of finding dicts to analyse.
  1743→        llm_client :
  1744→            Optional pre-configured LLM client.  If ``None``, the method
  1745→            attempts to create one from environment variables
  1746→            (``OPENAI_API_KEY`` or ``ANTHROPIC_API_KEY``).  Falls back to
  1747→            rule-based analysis if no LLM is available.
  1748→
  1749→        Returns
  1750→        -------
  1751→        List of :class:`ChainReport` — LLM chains first, then rule-based.
  1752→        """
  1753→        rule_reports = self.find_chains(target, findings)
  1754→
  1755→        if not findings:
  1756→            return rule_reports
  1757→
  1758→        llm_reports = self._call_llm(target, findings, llm_client)
  1759→        if llm_reports:
  1760→            # Merge: LLM chains first, then rule-based chains that aren't duplicates
  1761→            llm_names = {r.name.lower() for r in llm_reports}
  1762→            for rr in rule_reports:
  1763→                if rr.name.lower() not in llm_names:
  1764→                    llm_reports.append(rr)
  1765→            return llm_reports
  1766→
  1767→        return rule_reports

  1769→    # ------------------------------------------------------------------
  1770→    # Internal: LLM interaction

  1773→    def _call_llm(
  1774→        self,
  1775→        target: str,
  1776→        findings: List[Dict[str, Any]],
  1777→        llm_client: Any = None,
  1778→    ) -> List[ChainReport]:
  1779→        """Attempt to call an LLM and parse the response into ChainReport objects.
  1780→
  1781→        Returns an empty list on any failure (silently falls back to rules).
  1782→        """
  1783→        # Prepare findings for the prompt (truncate to avoid token limits)
  1784→        trimmed = []
  1785→        for f in findings[:30]:  # cap at 30 findings
  1786→            trimmed.append({
  1787→                "title": _finding_title(f),
  1788→                "severity": _normalise_severity(f.get("severity", "info")),
  1789→                "category": _finding_category(f),
  1790→                "description": str(f.get("description", ""))[:200],
  1791→            })
  1792→
  1793→        prompt = _LLIM_PROMPT.format(
  1794→            target=target,
  1795→            findings_json=json.dumps(trimmed, indent=2),
  1796→        )
  1797→
  1798→        response_text = ""
  1799→
  1800→        # Try OpenAI
  1801→        if llm_client is not None and hasattr(llm_client, "chat"):  # type: ignore[union-attr]
  1802→            try:
  1803→                resp = llm_client.chat.completions.create(  # type: ignore[union-attr]
  1804→                    model="gpt-4o-mini",
  1805→                    messages=[{"role": "user", "content": prompt}],
  1806→                    temperature=0.3,
  1807→                    max_tokens=2000,
  1808→                )
  1809→                response_text = resp.choices[0].message.content or ""
  1810→            except Exception:
  1811→                response_text = ""
  1812→        elif HAS_OPENAI:
  1813→            try:
  1814→                import os
  1815→                api_key = os.environ.get("OPENAI_API_KEY")
  1816→                if not api_key:
  1817→                    return []
  1818→                client = _openai_mod.OpenAI(api_key=api_key)
  1819→                resp = client.chat.completions.create(
  1820→                    model="gpt-4o-mini",
  1821→                    messages=[{"role": "user", "content": prompt}],
  1822→                    temperature=0.3,
  1823→                    max_tokens=2000,
  1824→                )
  1825→                response_text = resp.choices[0].message.content or ""
  1826→            except Exception:
  1827→                response_text = ""
  1828→
  1829→        # Try Anthropic if OpenAI didn't work
  1830→        if not response_text and HAS_ANTHROPIC:
  1831→            try:
  1832→                import os
  1833→                api_key = os.environ.get("ANTHROPIC_API_KEY")
  1834→                if not api_key:
  1835→                    return []
  1836→                client = _anthropic_mod.Anthropic(api_key=api_key)
  1837→                resp = client.messages.create(
  1838→                    model="claude-haiku-4-5-20241022",
  1839→                    max_tokens=2000,
  1840→                    messages=[{"role": "user", "content": prompt}],
  1841→                )
  1842→                response_text = resp.content[0].text if resp.content else ""
  1843→            except Exception:
  1844→                response_text = ""
  1845→
  1846→        if not response_text:
  1847→            return []
  1848→
  1849→        return self._parse_llm_response(response_text, findings)
  1850→
  1851→    @staticmethod
  1852→    def _parse_llm_response(
  1853→        response_text: str,
  1854→        findings: List[Dict[str, Any]],
  1855→    ) -> List[ChainReport]:
  1856→        """Parse an LLM JSON response into ChainReport objects.
  1857→
  1858→        Gracefully handles malformed JSON, missing fields, and out-of-range indices.
  1859→        """
  1860→        reports: List[ChainReport] = []
  1861→
  1862→        # Try to extract JSON from the response (handle markdown code blocks)
  1863→        json_match = re.search(r"```(?:json)?\s*\n?(.*?)\n?```", response_text, re.DOTALL)
  1864→        if json_match:
  1865→            json_str = json_match.group(1).strip()
  1866→        else:
  1867→            # Try to find a top-level JSON object
  1868→            brace_match = re.search(r"\{.*\}", response_text, re.DOTALL)
  1869→            json_str = brace_match.group(0) if brace_match else response_text.strip()
  1870→
  1871→        try:
  1872→            data = json.loads(json_str)
  1873→        except (json.JSONDecodeError, AttributeError):
  1874→            return reports
  1875→
  1876→        chains = data.get("chains", [])
  1877→        if not isinstance(chains, list):
  1878→            return reports
  1879→
  1880→        for chain in chains:
  1881→            if not isinstance(chain, dict):
  1882→                continue
  1883→
  1884→            try:
  1885→                name = str(chain.get("name", "LLM-Discovered Chain"))
  1886→                severity = _normalise_severity(chain.get("severity", "medium"))
  1887→                confidence = float(chain.get("confidence", 0.6))
  1888→                confidence = max(0.0, min(1.0, confidence))
  1889→                narrative = str(chain.get("narrative", "LLM-identified compound attack path."))
  1890→                steps = chain.get("steps", [])
  1891→                if not isinstance(steps, list):
  1892→                    steps = []
  1893→                steps = [str(s) for s in steps]
  1894→                remediation = str(chain.get("remediation", "Address the identified vulnerability chain."))
  1895→
  1896→                # Resolve findings by index
  1897→                indices = chain.get("findings_indices", [])
  1898→                if not isinstance(indices, list):
  1899→                    indices = []
  1900→                matched_findings = []
  1901→                for idx in indices:
  1902→                    if isinstance(idx, (int, float)) and 0 <= int(idx) < len(findings):
  1903→                        matched_findings.append(findings[int(idx)])
  1904→
  1905→                if not matched_findings:
  1906→                    continue
  1907→
  1908→                score = max(_sev_score(f.get("severity", "info")) for f in matched_findings) * 1.4
  1909→
  1910→                reports.append(ChainReport(
  1911→                    name=name,
  1912→                    severity=severity,
  1913→                    score=round(score, 2),
  1914→                    findings=matched_findings,
  1915→                    narrative=narrative,
  1916→                    steps=steps if steps else [
  1917→                        f"Step {i+1}: Exploit {_finding_title(f)}"
  1918→                        for i, f in enumerate(matched_findings)
  1919→                    ],
  1920→                    remediation=remediation,
  1921→                    confidence=round(confidence, 2),
  1922→                ))
  1923→            except Exception:
  1924→                continue
  1925→
  1926→        return reports
  1927→

  1935→
  1936→class MitreMapper:
  1937→    """Maps scan findings to MITRE ATT&CK techniques and Kill Chain phases."""
  1938→
  1939→    def __init__(self, findings: List[Dict[str, Any]]) -> None:
  1940→        self.findings = findings
  1941→        self._technique_hits: Dict[str, List[str]] = defaultdict(list)
  1942→        self._tactic_hits: Dict[str, List[str]] = defaultdict(list)
  1943→        self._kill_chain_hits: Dict[str, List[str]] = defaultdict(list)
  1944→        self._map_findings()
  1945→
  1946→    def _map_findings(self) -> None:
  1947→        """Map each finding to MITRE techniques."""
  1948→        for f in self.findings:
  1949→            cat = f.get("category", "").lower()
  1950→            if cat in MITRE_TECHNIQUE_MAP:
  1951→                tech = MITRE_TECHNIQUE_MAP[cat]
  1952→                self._technique_hits[tech["technique"]].append(f.get("title", ""))
  1953→                self._tactic_hits[tech["tactic"]].append(f.get("title", ""))
  1954→                phase_idx = MITRE_TACTICS.get(tech["tactic"], {}).get("phase", 0)
  1955→                if 0 < phase_idx <= len(KILL_CHAIN_PHASES):
  1956→                    self._kill_chain_hits[KILL_CHAIN_PHASES[phase_idx - 1]].append(f.get("title", ""))
  1957→
  1958→    @property
  1959→    def techniques(self) -> Dict[str, List[str]]:
  1960→        """Technique IDs mapped to finding titles."""
  1961→        return dict(self._technique_hits)
  1962→
  1963→    @property
  1964→    def tactics(self) -> Dict[str, List[str]]:
  1965→        """Tactic IDs mapped to finding titles."""
  1966→        return dict(self._tactic_hits)
  1967→
  1968→    @property
  1969→    def kill_chain(self) -> Dict[str, List[str]]:
  1970→        """Kill Chain phase names mapped to finding titles."""
  1971→        return dict(self._kill_chain_hits)
  1972→
  1973→    def technique_summary(self) -> List[Dict[str, Any]]:
  1974→        """Return list of {technique_id, name, tactic, tactic_name, finding_titles, severity}."""
  1975→        summary: List[Dict[str, Any]] = []
  1976→        seen: Set[str] = set()
  1977→        for f in self.findings:
  1978→            cat = f.get("category", "").lower()
  1979→            if cat not in MITRE_TECHNIQUE_MAP:
  1980→                continue
  1981→            tech = MITRE_TECHNIQUE_MAP[cat]
  1982→            tid = tech["technique"]
  1983→            if tid in seen:
  1984→                continue
  1985→            seen.add(tid)
  1986→            tactic_id = tech["tactic"]
  1987→            tactic_name = MITRE_TACTICS.get(tactic_id, {}).get("name", tactic_id)
  1988→            # Determine worst severity among findings that map to this technique
  1989→            titles = self._technique_hits.get(tid, [])
  1990→            sev = "info"
