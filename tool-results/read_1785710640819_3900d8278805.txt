   640→        for tmpl in cat_entry.get("waf_rules", []):
   641→            rules.append(tmpl)
   642→
   643→        # 2. Generate AWS WAF JSON rule for categories that have WAF applicability
   644→        if category in ("sqli", "xss", "command_injection", "path_traversal", "ssrf"):
   645→            rules.append(self._aws_waf_rule(category, severity, target_uri))
   646→
   647→        # 3. Generate category-specific Nginx rules not already in the map
   648→        if category == "security_headers":
   649→            rules.append(self._nginx_security_headers())
   650→        elif category == "cors":
   651→            rules.append(self._nginx_cors_config(finding))
   652→        elif category == "csrf":
   653→            rules.append(self._nginx_csrf_check())
   654→
   655→        return rules

   657→    def _aws_waf_rule(self, category: str, severity: str, uri: str) -> str:
   658→        """Generate an AWS WAF JSON rule statement."""
   659→        visual_filters: dict[str, list[str]] = {
   660→            "sqli": [
   661→                {
   662→                    "filter": {
   663→                        "type": "SQL_INJECTION_MATCHING_STATEMENT",
   664→                    }
   665→                }
   666→            ],
   667→            "xss": [
   668→                {
   669→                    "filter": {
   670→                        "type": "XSS_MATCHING_STATEMENT",
   671→                    }
   672→                }
   673→            ],
   674→            "command_injection": [
   675→                {
   676→                    "filter": {
   677→                        "type": "REGEX_MATCH",
   678→                        "regexString": ";.*\\b(cat|ls|id|whoami|bash|sh|wget|curl|nc|ping|nmap)\\b",
   679→                        "fieldToMatch": {"type": "QUERY_STRING"},
   680→                        "textTransformations": ["NONE"],
   681→                    }
   682→                }
   683→            ],
   684→            "path_traversal": [
   685→                {
   686→                    "filter": {
   687→                        "type": "REGEX_MATCH",
   688→                        "regexString": "(\\.\\./|%2e%2e|%252e%252e)",
   689→                        "fieldToMatch": {"type": "URI"},
   690→                        "textTransformations": ["URL_DECODE", "HTML_ENTITY_DECODE"],
   691→                    }
   692→                }
   693→            ],
   694→            "ssrf": [
   695→                {
   696→                    "filter": {
   697→                        "type": "REGEX_MATCH",
   698→                        "regexString": "(127\\.0\\.0\\.1|localhost|10\\.\\d{1,3}\\.\\d{1,3}\\.\\d{1,3}|192\\.168\\.\\d{1,3}\\.\\d{1,3}|169\\.254\\.169\\.254)",
   699→                        "fieldToMatch": {"type": "QUERY_STRING"},
   700→                        "textTransformations": ["NONE"],
   701→                    }
   702→                }
   703→            ],
   704→        }
   705→
   706→        filters = visual_filters.get(category, [])
   707→        priority_map = {"CRITICAL": 0, "HIGH": 1, "MEDIUM": 2, "LOW": 3}
   708→        priority = priority_map.get(severity, 1)
   709→
   710→        rule: dict[str, Any] = {
   711→            "Name": f"ReconPro-{category}-{uuid.uuid4().hex[:8]}",
   712→            "Priority": priority,
   713→            "Statement": {
   714→                "ManagedRuleGroupStatement": {
   715→                    "VendorName": "AWS",
   716→                    "Name": f"{category.upper() if category != 'sqli' else 'SQLInjection'}_Prevention",
   717→                    "ExcludedRules": [],
   718→                }
   719→            } if category in ("sqli", "xss") else {
   720→                "RateBasedStatement": {
   721→                    "Limit": 100,
   722→                    "AggregateKeyType": "IP",
   723→                }
   724→            } if not filters else {
   725→                "OrStatement": {
   726→                    "Statements": [
   727→                        {
   728→                            "ByteMatchStatement": {
   729→                                "SearchString": f.get("filter", {}).get("regexString", "").encode() if "regexString" in f.get("filter", {}) else b"",
   730→                                "FieldToMatch": f.get("filter", {}).get("fieldToMatch", {"type": "QUERY_STRING"}),
   731→                                "TextTransformations": f.get("filter", {}).get("textTransformations", [{"type": "NONE"}]),
   732→                                "PositionalConstraint": "CONTAINS",
   733→                            }
   734→                        }
   735→                        for f in filters
   736→                    ],
   737→                }
   738→            },
   739→            "Action": {
   740→                "Block": {
   741→                    "CustomResponse": {
   742→                        "ResponseCode": 403,
   743→                        "ResponseBody": {"Text": "Request blocked by ReconPro WAF rule."},
   744→                    }
   745→                }
   746→            },
   747→            "VisibilityConfig": {
   748→                "SampledRequestsEnabled": True,
   749→                "CloudWatchMetricsEnabled": True,
   750→                "MetricName": f"ReconPro{category.capitalize()}Rule",
   751→            },
   752→        }
   753→
   754→        # For SQLi and XSS, use managed rule groups (cleaner and more complete)
   755→        if category == "sqli":
   756→            rule["Statement"] = {
   757→                "ManagedRuleGroupStatement": {
   758→                    "VendorName": "AWS",
   759→                    "Name": "SQLInjectionRuleSet",
   760→                }
   761→            }
   762→        elif category == "xss":
   763→            rule["Statement"] = {
   764→                "ManagedRuleGroupStatement": {
   765→                    "VendorName": "AWS",
   766→                    "Name": "XSSRuleSet",
   767→                }
   768→            }
   769→
   770→        return "AWS_WAF_JSON:" + json.dumps(rule, indent=2)

   829→    # -- Code patch generation ---------------------------------------------
   830→
   831→    def generate_code_patch(self, finding: dict) -> str:
   832→        """Generate a unified diff patch for *finding*.
   833→
   834→        The finding must contain at least ``file`` (path) and ``line`` (1-based
   835→        line number).  If ``original_line`` is provided it is used as the
   836→        removed line; otherwise a placeholder is shown.
   837→        """
   838→        filepath = finding.get("file", "unknown.py")
   839→        line_no = finding.get("line", 1)
   840→        category = finding.get("category", "").lower()
   841→        original = finding.get("original_line", "")
   842→        severity = finding.get("severity", "high").upper()
   843→
   844→        # Determine the fix based on category or a pattern match
   845→        fix = self._resolve_code_fix(finding)
   846→
   847→        # Build unified diff
   848→        ctx_before = 3
   849→        start = max(1, line_no - ctx_before)
   850→        end = line_no + ctx_before
   851→
   852→        lines: list[str] = []
   853→        for i in range(start, end + 1):
   854→            if i == line_no:
   855→                if original:
   856→                    lines.append(f"-{original}")
   857→                else:
   858→                    lines.append(f"-# [REMOVED] vulnerable code at line {line_no}")
   859→                for fix_line in fix.split("\n"):
   860→                    lines.append(f"+{fix_line}")
   861→            else:
   862→                lines.append(f" # (context line {i})")
   863→
   864→        hunk_header = f"@@ -{start},{end - start + 1} +{start},{end - start + 1 + max(fix.count('\n'), 0)} @@"
   865→
   866→        diff = f"--- a/{filepath}\n"
   867→        diff += f"+++ b/{filepath}\n"
   868→        diff += f"{hunk_header}\n"
   869→        diff += "\n".join(lines) + "\n"
   870→        return diff
   871→
   872→    def _resolve_code_fix(self, finding: dict) -> str:
   873→        """Return the replacement code lines for a finding."""
   874→        category = finding.get("category", "").lower()
   875→        original = finding.get("original_line", "")
   876→
   877→        # Pattern-based detection on the original line
   878→        if "eval(" in original:
   879→            return (
   880→                "import ast\n"
   881→                "result = ast.literal_eval(sanitized_input)  # safe alternative to eval()"
   882→            )
   883→        if "exec(" in original:
   884→            return (
   885→                "# exec() removed — use a proper task queue, plugin system, or\n"
   886→                "# sandboxed interpreter for dynamic code execution"
   887→            )
   888→        if "shell=True" in original:
   889→            # Convert to list-based subprocess call
   890→            return (
   891→                "subprocess.run(\n"
   892→                "    [\"command\", \"arg1\", \"arg2\"],\n"
   893→                "    capture_output=True,\n"
   894→                "    text=True,\n"
   895→                "    shell=False,\n"
   896→                ")"
   897→            )
   898→        if any(kw in original.lower() for kw in ("select", "insert", "update", "delete")) and ("f\"" in original or "f'" in original):
   899→            return (
   900→                "cursor.execute(\n"
   901→                "    \"SELECT ... WHERE col = %s\",\n"
   902→                "    (user_input,),\n"
   903→                ")  # parameterized query"
   904→            )
   905→        if category == "hardcoded_secret" or any(s in original for s in ('"' in original and any(k in original.lower() for k in ("secret", "password", "key", "token", "api_key")),)):
   906→            # Check for hardcoded string that looks like a secret
   907→            import re
   908→            if re.search(r'("[^"]{8,}")\s*$', original.strip()):
   909→                var_match = re.match(r'(\w+)\s*=\s*'["\']', original)
   910→                var_name = var_match.group(1) if var_match else "CONFIG_VALUE"
   911→                env_key = var_name.upper()
   912→                return (
   913→                    f"import os\n"
   914→                    f"{var_name} = os.environ.get(\"{env_key}\")\n"
   915→                    f"if not {var_name}:\n"
   916→                    f"    raise RuntimeError(\"{env_key} environment variable is not set\")"
   917→                )
   918→        if "DEBUG = True" in original or "DEBUG=True" in original:
   919→            return "DEBUG = False"
   920→        if "csrf_exempt" in original:
   921→            return "# @csrf_exempt removed — CSRF protection is required for this endpoint"
   922→        if category == "security_headers":
   923→            return (
   924→                "response.headers['X-Content-Type-Options'] = 'nosniff'\n"
   925→                "response.headers['X-Frame-Options'] = 'DENY'\n"
   926→                "response.headers['Strict-Transport-Security'] = 'max-age=31536000; includeSubDomains'"
   927→            )
   928→
   929→        # Fall back to category-based template from CATEGORY_PATCH_MAP
   930→        cat_entry = CATEGORY_PATCH_MAP.get(category, {})
   931→        code_fix = cat_entry.get("code_fix", "")
   932→        if code_fix:
   933→            return code_fix.strip()
   934→
   935→        return "# Apply manual fix: review the vulnerability description and remediation guidance"
   936→

  1253→class DefenseBundle:
  1254→    """Orchestrate defense generation across all findings for a target."""
  1255→
  1256→    def __init__(self, target: str, findings: list[dict]) -> None:
  1257→        self.target = target
  1258→        self.findings = findings
  1259→        self._generator = PatchGenerator()
  1260→        self._bundle_id = uuid.uuid4().hex[:12]
  1261→        self._generated_at = datetime.now(timezone.utc).isoformat()
  1262→
  1263→    def generate_all(self) -> dict[str, str | list[str] | dict[str, str]]:
  1264→        """Generate all defense artifacts.
  1265→
  1266→        Returns:
  1267→            {
  1268→                "waf_rules": [...],
  1269→                "code_patches": {...},
  1270→                "infra_fixes": {...},
  1271→                "cicd_additions": str,
  1272→                "apply_script": str,
  1273→                "rollback_script": str,
  1274→            }
  1275→        """
  1276→        waf_rules: list[str] = []
  1277→        code_patches: dict[str, str] = {}
  1278→        infra_fixes: dict[str, str] = {}
  1279→
  1280→        for finding in self.findings:
  1281→            cat = finding.get("category", "unknown")
  1282→            fid = finding.get("id", hashlib.sha256(str(finding).encode()).hexdigest()[:8])
  1283→
  1284→            # WAF rules
  1285→            rules = self._generator.generate_waf_rules(finding)
  1286→            waf_rules.extend(rules)
  1287→
  1288→            # Code patches (only if file is specified)
  1289→            if finding.get("file"):
  1290→                patch = self._generator.generate_code_patch(finding)
  1291→                key = f"{cat}:{fid}"
  1292→                code_patches[key] = patch
  1293→
  1294→            # Infrastructure fixes
  1295→            infra = self._generator.generate_infra_fix(finding)
  1296→            if infra:
  1297→                infra_fixes[f"{cat}:{fid}"] = infra
  1298→
  1299→        # CI/CD additions (generated once for all findings)
  1300→        cicd = self._generator.generate_cicd_additions(self.findings)
  1301→
  1302→        # Apply and rollback scripts
  1303→        apply_script = self.generate_apply_script()
  1304→        rollback_script = self.generate_rollback_script()
  1305→
  1306→        return {
  1307→            "waf_rules": waf_rules,
  1308→            "code_patches": code_patches,
  1309→            "infra_fixes": infra_fixes,
  1310→            "cicd_additions": cicd,
  1311→            "apply_script": apply_script,
  1312→            "rollback_script": rollback_script,
  1313→        }
  1314→
  1315→    def generate_apply_script(self) -> str:
