   653→def _ip_in_range(ip: str, start: str, end: str) -> bool:

   662→def _ip_to_int(ip: str) -> int:
   663→    parts = ip.strip().split(".")
   664→    if len(parts) != 4:
   665→        raise ValueError(f"Invalid IP: {ip}")
   666→    return (int(parts[0]) << 24) + (int(parts[1]) << 16) + (int(parts[2]) << 8) + int(parts[3])

   669→def _extract_domain(target: str) -> str:
   670→    """Extract base domain from a target string."""

   674→    t = t.rstrip("/")
   675→    # Remove port
   676→    if ":" in t:

   678→    return t

   681→def _fingerprint_hash(host: str, techs: List[str], ips: List[str], cert_orgs: List[str]) -> str:
   682→    """Create a stable fingerprint hash for infrastructure drift comparison."""
   683→    blob = "|".join(sorted(set(techs))) + "||" + "|".join(sorted(set(ips))) + "||" + "|".join(sorted(set(cert_orgs)))
   684→    return hashlib.sha256(blob.encode("utf-8", errors="replace")).hexdigest()[:32]

   687→def _name_similarity(a: str, b: str) -> float:
   688→    """Compute a 0-1 similarity score between two domain/subdomain names."""
   689→    a, b = a.lower().strip(), b.lower().strip()
   690→    if a == b:
   691→        return 1.0
   692→    # Strip common suffix
   693→    parts_a = a.rsplit(".", 2)
   694→    parts_b = b.rsplit(".", 2)
   695→    if len(parts_a) >= 2 and len(parts_b) >= 2 and parts_a[-2:] == parts_b[-2:]:
   696→        a_stem, b_stem = parts_a[0], parts_b[0]
   697→    else:
   698→        a_stem, b_stem = a, b
   699→    # Common prefix length
   700→    prefix_len = 0
   701→    for ca, cb in zip(a_stem, b_stem):
   702→        if ca == cb:
   703→            prefix_len += 1
   704→        else:
   705→            break
   706→    max_len = max(len(a_stem), len(b_stem), 1)
   707→    return prefix_len / max_len

   714→def _resolve_dns(host: str) -> List[str]:
   715→    """Resolve a hostname to all A/AAAA records via stdlib socket."""
   716→    ips: List[str] = []
   717→    for family in (socket.AF_INET, socket.AF_INET6):
   718→        try:
   719→            results = socket.getaddrinfo(host, None, family, socket.SOCK_STREAM)
   720→            for r in results:
   721→                ip = r[4][0]
   722→                if ip not in ips:
   723→                    ips.append(ip)
   724→        except (socket.gaierror, OSError):
   725→            continue
   726→    return ips

   729→def _discover_ips_from_redirects(base_url: str, timeout: int, verify_tls: bool) -> List[str]:
   730→    """Follow HTTP redirects and collect IPs from Location headers."""
   731→    ips: List[str] = []
   732→    url = base_url
   733→    seen: Set[str] = set()
   734→    for _ in range(10):  # max redirect depth
   735→        if url in seen:
   736→            break
   737→        seen.add(url)
   738→        parsed = urllib.parse.urlparse(url)
   739→        host = parsed.hostname or ""
   740→        # Resolve the redirect target
   741→        for ip in _resolve_dns(host):
   742→            if ip not in ips:
   743→                ips.append(ip)
   744→        resp = http_probe(url, timeout=timeout, verify_tls=verify_tls, limiter=default_limiter)
   745→        status = resp.get("status", 0)
   746→        if status in (301, 302, 303, 307, 308):
   747→            location = resp.get("headers", {}).get("Location", "" or resp.get("headers", {}).get("location", ""))
   748→            if not location:
   749→                break
   750→            # Handle relative redirects
   751→            if location.startswith("/"):
   752→                location = f"{parsed.scheme}://{parsed.netloc}{location}"
   753→            elif not location.startswith("http"):
   754→                location = f"{parsed.scheme}://{parsed.netloc}/{location.lstrip('/')}"
   755→            url = location
   756→        else:
   757→            break
   758→    return ips

   761→def _discover_ips_from_certs(host: str, port: int = 443, timeout: int = 8) -> List[str]:
   762→    """Extract subject/alt-name IPs from TLS certificate."""
   763→    ips: List[str] = []
   764→    try:
   765→        ctx = ssl.create_default_context()
   766→        ctx.check_hostname = False
   767→        ctx.verify_mode = ssl.CERT_NONE
   768→        with socket.create_connection((host, port), timeout=timeout) as sock:
   769→            with ctx.wrap_socket(sock, server_hostname=host) as tls:
   770→                cert_der = tls.getpeercert(binary_form=True)
   771→                if cert_der:
   772→                    # Decode using stdlib — extract SANs
   773→                    import ssl as _ssl_mod
   774→                    cert_dict = ssl.DER_cert_to_PEM_cert(cert_der)
   775→                    # Parse PEM for subjectAltName
   776→                    in_san = False
   777→                    for line in cert_dict.splitlines():
   778→                        line = line.strip()
   779→                        if "Subject Alternative Name" in line:
   780→                            in_san = True
   781→                            continue
   782→                        if in_san:
   783→                            if line.startswith("---"):
   784→                                in_san = False
   785→                                continue
   786→                            # Extract IPs from SAN
   787→                            for part in line.split(","):
   788→                                part = part.strip()
   789→                                if part.startswith("IP Address:"):
   790→                                    ip_addr = part[len("IP Address:"):].strip()
   791→                                    if ip_addr not in ips:
   792→                                        ips.append(ip_addr)
   793→    except (ssl.SSLError, socket.error, OSError, Exception):
   794→        pass
   795→    return ips

   798→def _run_ip_discovery(target: str, base_url: str, timeout: int, verify_tls: bool) -> Dict[str, Any]:
   799→    """Phase 1: Enumerate all IPs associated with the target."""
   800→    host = _extract_domain(target)

   821→    for ip in cert_ips:
   822→        if ip not in all_ips:

   825→    return {
   826→        "target": host,
   827→        "total_ips": len(all_ips),
   828→        "ips": all_ips,

   858→def _get_certificate_info(host: str, port: int = 443, timeout: int = 8) -> Dict[str, Any]:
   859→    """Extract certificate details: issuer, subject, SANs, org."""
   860→    info: Dict[str, Any] = {
   861→        "issuer": "",
   862→        "subject": "",
   863→        "sans": [],
   864→        "org": "",
   865→        "serial": "",
   866→        "not_before": "",
   867→        "not_after": "",
   868→        "error": None,
   869→    }
   870→    try:
   871→        ctx = ssl.create_default_context()
   872→        ctx.check_hostname = False
   873→        ctx.verify_mode = ssl.CERT_NONE
   874→        with socket.create_connection((host, port), timeout=timeout) as sock:
   875→            with ctx.wrap_socket(sock, server_hostname=host) as tls:
   876→                cert = tls.getpeercert()
   877→                if not cert:
   878→                    info["error"] = "no_certificate"
   879→                    return info
   880→                # Subject
   881→                subject_parts = []
   882→                for rdn in cert.get("subject", ()):
   883→                    for attr_type, attr_val in rdn:
   884→                        subject_parts.append(f"{attr_type}={attr_val}")
   885→                info["subject"] = ", ".join(subject_parts)
   886→                # Issuer
   887→                issuer_parts = []
   888→                for rdn in cert.get("issuer", ()):
   889→                    for attr_type, attr_val in rdn:
   890→                        issuer_parts.append(f"{attr_type}={attr_val}")
   891→                info["issuer"] = ", ".join(issuer_parts)
   892→                # Organization
   893→                for rdn in cert.get("subject", ()):
   894→                    for attr_type, attr_val in rdn:
   895→                        if attr_type == "organizationName":
   896→                            info["org"] = attr_val
   897→                # Serial
   898→                info["serial"] = cert.get("serialNumber", "")
   899→                # Validity
   900→                info["not_before"] = cert.get("notBefore", "")
   901→                info["not_after"] = cert.get("notAfter", "")
   902→                # SANs
   903→                for ext in cert.get("extensions", []):
   904→                    if ext[0] == "subjectAltName":
   905→                        for san in ext[1]:
   906→                            info["sans"].append(san[1])
   907→    except (ssl.SSLError, socket.error, OSError) as exc:
   908→        info["error"] = str(exc)
   909→    except Exception as exc:
   910→        info["error"] = str(exc)
   911→    return info

   914→def _detect_tech_stack(headers: Dict[str, str], body: str) -> List[str]:
   915→    """Detect technologies from HTTP response headers and body."""
   916→    detected: List[str] = []
   917→    body_lower = body[:8192].lower()
   918→    header_str = "\n".join(f"{k}: {v}" for k, v in headers.items())
   919→    cookie_str = headers.get("set-cookie", "")
   920→
   921→    for tech_name, tech_info in TECH_STACK_DB.items():
   922→        matched = False
   923→        # Header value patterns (e.g., Server: nginx)
   924→        for pat in tech_info.get("header_patterns", []):
   925→            if pat.search(header_str):
   926→                matched = True
   927→                break
   928→        if not matched:
   929→            for pat in tech_info.get("header_name_patterns", []):
   930→                for h_key in headers:
   931→                    if pat.search(h_key):
   932→                        matched = True
   933→                        break
   934→                if matched:
   935→                    break
   936→        if not matched:
   937→            for pat in tech_info.get("body_patterns", []):
   938→                if pat.search(body_lower):
   939→                    matched = True
   940→                    break
   941→        if not matched:
   942→            for pat in tech_info.get("cookie_patterns", []):
   943→                if pat.search(cookie_str):
   944→                    matched = True
   945→                    break
   946→        if matched and tech_name not in detected:
   947→            detected.append(tech_name)
   948→
   949→    return detected

   952→def _scan_subdomain_infrastructure(
   953→    domain: str, subdomain: str, timeout: int, verify_tls: bool
   954→) -> Dict[str, Any]:
   955→    """Scan a single subdomain for infrastructure details."""
   956→    fqdn = f"{subdomain}.{domain}" if subdomain else domain
   957→    result: Dict[str, Any] = {
   958→        "host": fqdn,
   959→        "resolves": False,
   960→        "ips": [],
   961→        "http_reachable": False,
   962→        "status": 0,
   963→        "tech_stack": [],
   964→        "cert": {},
   965→        "response_time_ms": 0,
   966→    }
   967→
   968→    # DNS resolution
   969→    ips = _resolve_dns(fqdn)
   970→    if not ips:
   971→        return result
   972→    result["resolves"] = True
   973→    result["ips"] = ips
   974→
   975→    # HTTP probe
   976→    start = time.monotonic()
   977→    resp = http_probe(f"https://{fqdn}", timeout=timeout, verify_tls=verify_tls, limiter=default_limiter)
   978→    elapsed_ms = int((time.monotonic() - start) * 1000)
   979→    result["response_time_ms"] = elapsed_ms
   980→
   981→    if resp.get("status", 0) > 0:
   982→        result["http_reachable"] = True
   983→        result["status"] = resp["status"]
   984→        result["tech_stack"] = _detect_tech_stack(resp.get("headers", {}), resp.get("body", ""))
   985→    else:
   986→        # Try HTTP
   987→        resp = http_probe(f"http://{fqdn}", timeout=timeout, verify_tls=False, limiter=default_limiter)
   988→        if resp.get("status", 0) > 0:
   989→            result["http_reachable"] = True
   990→            result["status"] = resp["status"]
   991→            result["tech_stack"] = _detect_tech_stack(resp.get("headers", {}), resp.get("body", ""))
   992→
   993→    # Certificate
   994→    cert_info = _get_certificate_info(fqdn, 443, timeout)
   995→    if not cert_info.get("error"):
   996→        result["cert"] = cert_info
   997→
   998→    return result

  1012→    for sub in sorted(all_subs):
  1013→        info = _scan_subdomain_infrastructure(domain, sub, timeout, verify_tls)
  1014→        if info["resolves"] or info["http_reachable"]:
  1015→            results.append(info)

  1028→def _mine_ct_logs(domain: str, timeout: int) -> Dict[str, Any]:
  1029→    """Phase 3: Extract subdomains, org details, cert chains from CT logs."""
  1030→    result: Dict[str, Any] = {
  1031→        "subdomains": [],
  1032→        "organizations": [],
  1033→        "issuers": [],
  1034→        "cert_count": 0,
  1035→        "wildcard_count": 0,
  1036→        "expired_count": 0,
  1037→        "entries": [],
  1038→        "error": None,
  1039→    }
  1040→
  1041→    url = CT_LOG_ENDPOINTS[0].format(domain=urllib.parse.quote(domain))
  1042→    resp = http_probe(url, timeout=max(timeout, 15), verify_tls=True, limiter=default_limiter)
  1043→
  1044→    if not resp.get("ok") or resp.get("status") != 200:
  1045→        result["error"] = f"CT log query failed: status={resp.get('status')} {resp.get('reason')}"
  1046→        return result
  1047→
  1048→    try:
  1049→        entries = json.loads(resp.get("body", ""))
  1050→    except (json.JSONDecodeError, ValueError):
  1051→        result["error"] = "Failed to parse CT log response"
  1052→        return result
  1053→
  1054→    subdomains: Set[str] = set()
  1055→    orgs: Set[str] = set()
  1056→    issuers: Set[str] = set()
  1057→
  1058→    for entry in entries:
  1059→        name_value = entry.get("name_value", "")
  1060→        if not name_value:
  1061→            continue
  1062→
  1063→        # Count certs
  1064→        result["cert_count"] += 1
  1065→
  1066→        # Check wildcard
  1067→        if name_value.startswith("*"):
  1068→            result["wildcard_count"] += 1
  1069→
  1070→        # Check expiry
  1071→        not_after = entry.get("not_after", "")
  1072→        if not_after:
  1073→            try:
  1074→                from datetime import datetime
  1075→                exp = datetime.strptime(not_after.split("T")[0], "%Y-%m-%d")
  1076→                if exp < datetime.utcnow():
  1077→                    result["expired_count"] += 1
  1078→            except (ValueError, IndexError):
  1079→                pass
  1080→
  1081→        # Extract names
  1082→        for name in name_value.split("\n"):
  1083→            name = name.strip().lstrip("*.")
  1084→            if name and name != domain:
  1085→                subdomains.add(name)
  1086→
  1087→        # Organization
  1088→        issuer_ca = entry.get("issuer_ca_id", "")
  1089→        issuer_name = entry.get("issuer_name", "")
  1090→        if issuer_name:
  1091→            issuers.add(issuer_name)
  1092→
  1093→        # Common name org
  1094→        common_name = entry.get("common_name", "")
  1095→        if common_name and "." in common_name:
  1096→            # Look for org in issuer
  1097→            for part in issuer_name.split(","):
  1098→                part = part.strip()
  1099→                if "O=" in part or "O =" in part:
  1100→                    org = part.split("=")[1].strip()
  1101→                    if org:
  1102→                        orgs.add(org)
  1103→
  1104→        # Store simplified entry
  1105→        result["entries"].append({
  1106→            "name_value": name_value[:256],
  1107→            "issuer_name": issuer_name[:256] if issuer_name else "",
  1108→            "not_before": entry.get("not_before", "")[:10],
  1109→            "not_after": entry.get("not_after", "")[:10],
  1110→        })
  1111→
  1112→    result["subdomains"] = sorted(subdomains)
  1113→    result["organizations"] = sorted(orgs)
  1114→    result["issuers"] = sorted(issuers)
  1115→
  1116→    return result

  1123→def _detect_cdn(headers: Dict[str, str], body: str) -> List[str]:
  1124→    """Detect CDN presence from response headers and body."""
  1125→    detected: List[str] = []
  1126→    body_lower = body[:8192].lower()
  1127→    header_keys_lower = {k.lower(): v for k, v in headers.items()}
  1128→
  1129→    for cdn_key, cdn_info in CDN_FINGERPRINTS.items():
  1130→        matched = False
  1131→        # Header matching
  1132→        for hdr_name, hdr_val_pat in cdn_info.get("headers", {}).items():
  1133→            hdr_lookup = hdr_name.lower()
  1134→            if hdr_lookup in header_keys_lower:
  1135→                if hdr_val_pat is None:
  1136→                    # Presence of header is enough
  1137→                    matched = True
  1138→                    break
  1139→                elif isinstance(hdr_val_pat, re.Pattern) and hdr_val_pat.search(header_keys_lower[hdr_lookup]):
  1140→                    matched = True
  1141→                    break
  1142→        if not matched:
  1143→            for pat in cdn_info.get("body_patterns", []):
  1144→                if pat.search(body_lower):
  1145→                    matched = True
  1146→                    break
  1147→        if matched and cdn_key not in detected:
  1148→            detected.append(cdn_key)
  1149→
  1150→    return detected

  1152→
  1153→def _detect_cloud_provider(
  1154→    headers: Dict[str, str], body: str, ip: str, cname: str = ""
  1155→) -> List[str]:
  1156→    """Detect cloud provider from headers, body, IP, and CNAME."""
  1157→    detected: List[str] = []
  1158→    body_lower = body[:8192].lower()
  1159→    header_str = "\n".join(f"{k}: {v}" for k, v in headers.items())
  1160→
  1161→    for prov_key, prov_info in CLOUD_PROVIDER_SIGNS.items():
  1162→        matched = False
  1163→        # IP range check (strongest signal)
  1164→        for start, end in prov_info.get("ip_ranges", []):
  1165→            if _ip_in_range(ip, start, end):
  1166→                matched = True
  1167→                break
  1168→        # Header patterns
  1169→        if not matched:
  1170→            for pat in prov_info.get("header_patterns", []):
  1171→                if pat.search(header_str):
  1172→                    matched = True
  1173→                    break
  1174→        # Body patterns
  1175→        if not matched:
  1176→            for pat in prov_info.get("body_patterns", []):
  1177→                if pat.search(body_lower):
  1178→                    matched = True
  1179→                    break
  1180→        # CNAME patterns
  1181→        if not matched and cname:
  1182→            for pat in prov_info.get("cname_patterns", []):
  1183→                if pat.search(cname):
  1184→                    matched = True
  1185→                    break
  1186→        if matched and prov_key not in detected:
  1187→            detected.append(prov_key)
  1188→
  1189→    return detected

  1241→def _run_tech_clustering(subdomain_results: List[Dict[str, Any]]) -> Dict[str, Any]:
  1242→    """Phase 5: Group infrastructure by shared technologies."""
  1243→    # Build tech -> hosts mapping
  1244→    tech_hosts: Dict[str, List[str]] = {}
  1245→    host_techs: Dict[str, List[str]] = {}
  1246→
  1247→    for r in subdomain_results:
  1248→        host = r["host"]
  1249→        techs = r.get("tech_stack", [])
  1250→        if not techs:
  1251→            continue
  1252→        host_techs[host] = techs
  1253→        for t in techs:
  1254→            if t not in tech_hosts:
  1255→                tech_hosts[t] = []
  1256→            tech_hosts[t].append(host)
  1257→
  1258→    # Build clusters: groups of hosts sharing >=2 technologies
  1259→    clusters: List[Dict[str, Any]] = []
  1260→    visited: Set[str] = set()
  1261→    hosts_list = list(host_techs.keys())
  1262→
  1263→    for i, host_a in enumerate(hosts_list):
  1264→        if host_a in visited:
  1265→            continue
  1266→        cluster: Dict[str, Any] = {
  1267→            "hosts": [host_a],
  1268→            "shared_techs": list(host_techs[host_a]),
  1269→        }
  1270→        visited.add(host_a)
  1271→        for j in range(i + 1, len(hosts_list)):
  1272→            host_b = hosts_list[j]
  1273→            if host_b in visited:
  1274→                continue
  1275→            shared = set(host_techs[host_a]) & set(host_techs[host_b])
  1276→            if len(shared) >= 2:
  1277→                cluster["hosts"].append(host_b)
  1278→                cluster["shared_techs"] = sorted(set(cluster["shared_techs"]) & shared | set(cluster["shared_techs"]) & set(host_techs[host_b]))
  1279→                visited.add(host_b)
  1280→        if len(cluster["hosts"]) > 1:
  1281→            clusters.append(cluster)
  1282→
  1283→    # Technology prevalence
  1284→    tech_prevalence = sorted(
  1285→        [(t, len(hosts)) for t, hosts in tech_hosts.items()],
  1286→        key=lambda x: -x[1],
  1287→    )
  1288→
  1289→    return {
  1290→        "clusters": clusters,
  1291→        "cluster_count": len(clusters),
  1292→        "tech_hosts": {k: v for k, v in sorted(tech_hosts.items())},
  1293→        "tech_prevalence": tech_prevalence,
  1294→        "unique_techs": sorted(tech_hosts.keys()),
  1295→    }

  1313→def _run_lookalike_detection(
  1314→    domain: str,
  1315→    subdomain_results: List[Dict[str, Any]],
  1316→    ct_data: Dict[str, Any],
  1317→    timeout: int, verify_tls: bool,
  1318→) -> Dict[str, Any]:
  1319→    """Phase 6: Find domains/hosts with similar naming, identical tech, or shared certs."""
  1320→    lookalikes: List[Dict[str, Any]] = []
  1321→    base_domain = _extract_domain(domain)
  1322→    base_parts = base_domain.split(".")
  1323→    base_stem = base_parts[0] if len(base_parts) > 2 else ""
  1324→    tld = ".".join(base_parts[-2:]) if len(base_parts) >= 2 else base_domain
  1325→
  1326→    # 6a. Name-based lookalikes
  1327→    if base_stem:
  1328→        candidates_name: List[str] = []
  1329→        for suffix in LOOKALIKE_SUFFIXES:
  1330→            candidates_name.append(f"{base_stem}{suffix}.{tld}")
  1331→            # Also try with hyphen between stem and known subdomains
  1332→        # Check a subset (rate-limit aware)
  1333→        checked = 0
  1334→        for candidate in candidates_name:
  1335→            if checked >= 30:
  1336→                break
  1337→            checked += 1
  1338→            ips = _resolve_dns(candidate)
  1339→            if ips:
  1340→                resp = http_probe(
  1341→                    f"https://{candidate}", timeout=timeout, verify_tls=verify_tls, limiter=default_limiter
  1342→                )
  1343→                if resp.get("status", 0) == 0:
  1344→                    resp = http_probe(
  1345→                        f"http://{candidate}", timeout=timeout, verify_tls=False, limiter=default_limiter
  1346→                    )
  1347→                techs = []
  1348→                if resp.get("status", 0) > 0:
  1349→                    techs = _detect_tech_stack(resp.get("headers", {}), resp.get("body", ""))
  1350→                sim = _name_similarity(candidate, base_domain)
  1351→                lookalikes.append({
  1352→                    "host": candidate,
  1353→                    "type": "naming_pattern",
  1354→                    "similarity": round(sim, 2),
  1355→                    "ips": ips,
  1356→                    "tech_stack": techs,
  1357→                    "status": resp.get("status", 0),
  1358→                })
  1359→
  1360→    # 6b. Same-tech-stack lookalikes from subdomain scan
  1361→    all_techs: Dict[str, List[str]] = {}  # tech fingerprint -> list of hosts
  1362→    for r in subdomain_results:
  1363→        host = r["host"]
  1364→        techs = tuple(sorted(r.get("tech_stack", [])))
  1365→        if len(techs) >= 2:
  1366→            key = "|".join(techs)
  1367→            if key not in all_techs:
  1368→                all_techs[key] = []
  1369→            all_techs[key].append(host)
  1370→    for key, hosts in all_techs.items():
  1371→        if len(hosts) > 1:
  1372→            for host in hosts:
  1373→                # Don't add duplicates
  1374→                if not any(l["host"] == host and l["type"] == "shared_tech" for l in lookalikes):
  1375→                    lookalikes.append({
  1376→                        "host": host,
  1377→                        "type": "shared_tech",
  1378→                        "similarity": 0.0,
  1379→                        "tech_fingerprint": key,
  1380→                        "peer_count": len(hosts) - 1,
  1381→                    })
  1382→
  1383→    # 6c. Shared certificate lookalikes
  1384→    cert_org_hosts: Dict[str, List[str]] = {}
  1385→    for r in subdomain_results:
  1386→        host = r["host"]
  1387→        org = r.get("cert", {}).get("org", "")
  1388→        issuer = r.get("cert", {}).get("issuer", "")
  1389→        if org:
  1390→            cert_org_hosts.setdefault(org, []).append(host)
  1391→        elif issuer:
  1392→            cert_org_hosts.setdefault(issuer, []).append(host)
  1393→    for org, hosts in cert_org_hosts.items():
  1394→        if len(hosts) > 1:
  1395→            for host in hosts:
  1396→                if not any(l["host"] == host and l["type"] == "shared_cert" for l in lookalikes):
  1397→                    lookalikes.append({
  1398→                        "host": host,
  1399→                        "type": "shared_cert",
  1400→                        "similarity": 0.0,
  1401→                        "shared_cert_id": org[:128],
  1402→                        "peer_count": len(hosts) - 1,
  1403→                    })
  1404→
  1405→    return {
  1406→        "lookalikes": lookalikes,
  1407→        "total_lookalikes": len(lookalikes),
  1408→        "naming_count": sum(1 for l in lookalikes if l.get("type") == "naming_pattern"),
  1409→        "shared_tech_count": sum(1 for l in lookalikes if l.get("type") == "shared_tech"),
  1410→        "shared_cert_count": sum(1 for l in lookalikes if l.get("type") == "shared_cert"),
  1411→    }

  1449→def _run_drift_detection(target: str, current_fp: Dict[str, Any]) -> Dict[str, Any]:
  1450→    """Phase 7: Compare current fingerprint against historical."""
  1451→    domain = _extract_domain(target)
  1452→    result: Dict[str, Any] = {
  1453→        "drift_detected": False,
  1454→        "drift_details": [],
  1455→        "historical_snapshots": 0,
  1456→        "current_fingerprint": current_fp["fingerprint"],
  1457→    }
  1458→
  1459→    if domain not in _DRIFT_HISTORY:
  1460→        _DRIFT_HISTORY[domain] = current_fp
  1461→        result["historical_snapshots"] = 1
  1462→        return result
  1463→
  1464→    previous = _DRIFT_HISTORY[domain]
  1465→    result["historical_snapshots"] = len(_DRIFT_HISTORY)
  1466→    drifts: List[Dict[str, str]] = []
  1467→
  1468→    # Fingerprint changed
  1469→    if previous["fingerprint"] != current_fp["fingerprint"]:
  1470→        result["drift_detected"] = True
  1471→        drifts.append({
  1472→            "type": "fingerprint_change",
  1473→            "detail": f"Infrastructure fingerprint changed: {previous['fingerprint'][:16]}... -> {current_fp['fingerprint'][:16]}...",
  1474→        })
  1475→
  1476→    # IP count changed
  1477→    prev_ips = previous.get("ip_count", 0)
  1478→    curr_ips = current_fp.get("ip_count", 0)
  1479→    if prev_ips != curr_ips:
  1480→        result["drift_detected"] = True
  1481→        drifts.append({
  1482→            "type": "ip_count_change",
  1483→            "detail": f"IP count changed: {prev_ips} -> {curr_ips} (delta={curr_ips - prev_ips:+d})",
  1484→        })
  1485→
  1486→    # Tech count changed
  1487→    prev_techs = set(previous.get("unique_techs", []))
  1488→    curr_techs = set(current_fp.get("unique_techs", []))
  1489→    added_techs = curr_techs - prev_techs
  1490→    removed_techs = prev_techs - curr_techs
  1491→    if added_techs or removed_techs:
  1492→        result["drift_detected"] = True
  1493→        if added_techs:
  1494→            drifts.append({
  1495→                "type": "tech_added",
  1496→                "detail": f"New technologies detected: {', '.join(sorted(added_techs))}",
  1497→            })
  1498→        if removed_techs:
  1499→            drifts.append({
  1500→                "type": "tech_removed",
  1501→                "detail": f"Technologies no longer detected: {', '.join(sorted(removed_techs))}",
  1502→            })
  1503→
  1504→    # Subdomain count changed
  1505→    prev_subs = previous.get("subdomain_count", 0)
  1506→    curr_subs = current_fp.get("subdomain_count", 0)
  1507→    if prev_subs != curr_subs:
  1508→        result["drift_detected"] = True
  1509→        drifts.append({
  1510→            "type": "subdomain_count_change",
  1511→            "detail": f"Subdomain count changed: {prev_subs} -> {curr_subs} (delta={curr_subs - prev_subs:+d})",
  1512→        })
  1513→
  1514→    # Org changes
  1515→    prev_orgs = set(previous.get("orgs", []))
  1516→    curr_orgs = set(current_fp.get("orgs", []))
  1517→    new_orgs = curr_orgs - prev_orgs
  1518→    if new_orgs:
  1519→        result["drift_detected"] = True
  1520→        drifts.append({
  1521→            "type": "new_organizations",
  1522→            "detail": f"New certificate organizations: {', '.join(sorted(new_orgs))}",
  1523→        })
  1524→
  1525→    result["drift_details"] = drifts
  1526→    # Update history
  1527→    _DRIFT_HISTORY[domain] = current_fp
  1528→
  1529→    return result
  1530→
  1531→
  1532→# ════════════════════════════════════════════════════════════════════════
  1533→# 8. ATTACK SURFACE SCORING
  1534→# ════════════════════════════════════════════════════════════════════════
  1535→
  1536→
  1537→def _score_attack_surface(
  1538→    ip_data: Dict[str, Any],
  1539→    subdomain_results: List[Dict[str, Any]],
  1540→    ct_data: Dict[str, Any],
  1541→    cdn_cloud_data: Dict[str, Any],
  1542→    tech_cluster_data: Dict[str, Any],
  1543→    lookalike_data: Dict[str, Any],
  1544→    drift_data: Dict[str, Any],
  1545→) -> Dict[str, Any]:
  1546→    """Phase 8: Score the total infrastructure attack surface."""
  1547→    score = 0.0
  1548→    max_score = 100.0
  1549→    factors: List[Dict[str, Any]] = []
  1550→
  1551→    # Factor 1: IP exposure (more IPs = larger attack surface)
  1552→    ip_count = ip_data.get("total_ips", 0)
  1553→    ip_score = min(15.0, ip_count * 2.5)
  1554→    score += ip_score
  1555→    factors.append({
  1556→        "name": "ip_exposure",
  1557→        "points": round(ip_score, 1),
  1558→        "max": 15.0,
  1559→        "detail": f"{ip_count} unique IPs discovered",
  1560→    })
  1561→
  1562→    # Factor 2: Subdomain sprawl
  1563→    sub_count = len(subdomain_results)
  1564→    sub_reachable = sum(1 for r in subdomain_results if r.get("http_reachable"))
  1565→    sub_score = min(15.0, sub_count * 0.8 + sub_reachable * 0.5)
  1566→    score += sub_score
  1567→    factors.append({
  1568→        "name": "subdomain_spread",
  1569→        "points": round(sub_score, 1),
  1570→        "max": 15.0,
  1571→        "detail": f"{sub_count} subdomains ({sub_reachable} HTTP-reachable)",
  1572→    })
  1573→
  1574→    # Factor 3: Certificate sprawl
  1575→    cert_count = ct_data.get("cert_count", 0)
  1576→    wildcard_count = ct_data.get("wildcard_count", 0)
  1577→    cert_score = min(10.0, cert_count * 0.3 + wildcard_count * 2.0)
  1578→    score += cert_score
  1579→    factors.append({
  1580→        "name": "certificate_spread",
  1581→        "points": round(cert_score, 1),
  1582→        "max": 10.0,
  1583→        "detail": f"{cert_count} certificates ({wildcard_count} wildcards)",
  1584→    })
  1585→
  1586→    # Factor 4: Technology diversity (more tech = more attack vectors)
  1587→    tech_count = tech_cluster_data.get("unique_techs", [])
  1588→    tech_n = len(tech_count)
  1589→    tech_score = min(15.0, tech_n * 2.0)
  1590→    score += tech_score
  1591→    factors.append({
  1592→        "name": "technology_diversity",
  1593→        "points": round(tech_score, 1),
  1594→        "max": 15.0,
  1595→        "detail": f"{tech_n} unique technologies detected",
  1596→    })
  1597→
  1598→    # Factor 5: Missing CDN/WAF (reduces score if absent)
  1599→    cdns = cdn_cloud_data.get("all_cdns", [])
  1600→    has_waf = any("cloudflare" in c or "sucuri" in c or "incapsula" in c or "akamai" in c for c in cdns)
  1601→    if not has_waf:
  1602→        no_waf_score = 8.0
  1603→        score += no_waf_score
  1604→        factors.append({
  1605→            "name": "missing_waf",
  1606→            "points": round(no_waf_score, 1),
  1607→            "max": 8.0,
  1608→            "detail": "No WAF detected on any infrastructure",
  1609→        })
  1610→    else:
  1611→        factors.append({
  1612→            "name": "missing_waf",
  1613→            "points": 0.0,
  1614→            "max": 8.0,
  1615→            "detail": f"WAF present: {', '.join(cdns)}",
  1616→        })
  1617→
  1618→    # Factor 6: Cloud provider concentration risk
  1619→    clouds = cdn_cloud_data.get("all_clouds", [])
  1620→    if len(clouds) == 1:
  1621→        cloud_score = 5.0  # single provider = concentration risk
  1622→    elif len(clouds) == 0:
  1623→        cloud_score = 3.0  # unknown hosting
  1624→    else:
  1625→        cloud_score = 1.0  # multi-cloud = more resilient but larger surface
  1626→    score += cloud_score
  1627→    factors.append({
  1628→        "name": "cloud_concentration",
  1629→        "points": round(cloud_score, 1),
  1630→        "max": 5.0,
  1631→        "detail": f"{len(clouds)} cloud provider(s): {', '.join(clouds) or 'none detected'}",
  1632→    })
  1633→
  1634→    # Factor 7: Lookalike risk
  1635→    lookalike_count = lookalike_data.get("total_lookalikes", 0)
  1636→    lookalike_score = min(10.0, lookalike_count * 1.5)
  1637→    score += lookalike_score
  1638→    factors.append({
  1639→        "name": "lookalike_risk",
  1640→        "points": round(lookalike_score, 1),
  1641→        "max": 10.0,
  1642→        "detail": f"{lookalike_count} lookalike infrastructure assets",
  1643→    })
  1644→
  1645→    # Factor 8: Drift (recent changes indicate instability)
  1646→    if drift_data.get("drift_detected"):
  1647→        drift_score = 7.0
  1648→    else:
  1649→        drift_score = 0.0
  1650→    score += drift_score
  1651→    factors.append({
  1652→        "name": "infrastructure_drift",
  1653→        "points": round(drift_score, 1),
  1654→        "max": 7.0,
  1655→        "detail": f"{'Drift detected with ' + str(len(drift_data.get('drift_details', []))) + ' change(s)' if drift_data.get('drift_detected') else 'No infrastructure drift detected'}",
  1656→    })
  1657→
  1658→    # Factor 9: Exposed sensitive subdomains
  1659→    sensitive_subs = [
  1660→        r for r in subdomain_results
  1661→        if any(s in r["host"] for s in ("admin", "staging", "dev", "test", "internal", "db", "git"))
  1662→        and r.get("http_reachable")
  1663→    ]
  1664→    sensitive_score = min(10.0, len(sensitive_subs) * 3.0)
  1665→    score += sensitive_score
  1666→    factors.append({
  1667→        "name": "sensitive_subdomains",
  1668→        "points": round(sensitive_score, 1),
  1669→        "max": 10.0,
  1670→        "detail": f"{len(sensitive_subs)} potentially sensitive subdomains exposed",
  1671→    })
  1672→
  1673→    # Factor 10: Expired certificates
  1674→    expired = ct_data.get("expired_count", 0)
  1675→    expired_score = min(5.0, expired * 1.5)
  1676→    score += expired_score
  1677→    factors.append({
  1678→        "name": "expired_certificates",
  1679→        "points": round(expired_score, 1),
  1680→        "max": 5.0,
  1681→        "detail": f"{expired} expired certificate(s) in CT logs",
  1682→    })
  1683→
  1684→    final_score = min(max_score, score)
  1685→
  1686→    # Risk level
  1687→    if final_score >= 70:
  1688→        risk_level = "critical"
  1689→    elif final_score >= 50:
  1690→        risk_level = "high"
  1691→    elif final_score >= 30:
  1692→        risk_level = "medium"
  1693→    elif final_score >= 15:
  1694→        risk_level = "low"
  1695→    else:
  1696→        risk_level = "info"
  1697→
  1698→    return {
  1699→        "total_score": round(final_score, 1),
  1700→        "max_score": max_score,
  1701→        "risk_level": risk_level,
  1702→        "factors": factors,
  1703→    }
  1704→
  1705→
  1706→# ════════════════════════════════════════════════════════════════════════
  1707→# MAIN ORCHESTRATOR
  1708→# ════════════════════════════════════════════════════════════════════════
  1709→
  1710→def run_infrastructure_ghost(
  1711→    target: str,
  1712→    base_url: str,
  1713→    timeout: int = 8,
  1714→    verify_tls: bool = True,
  1715→) -> List[Finding]:
  1716→    """Create a complete digital ghost of the target's infrastructure.
  1717→
  1718→    Parameters
  1719→    ----------
  1720→    target : str
  1721→        Target domain (e.g. ``example.com``).
  1722→    base_url : str
  1723→        Base URL for HTTP probing (e.g. ``https://example.com``).
  1724→    timeout : int
  1725→        Per-request timeout in seconds (default 8).
  1726→    verify_tls : bool
  1727→        Whether to verify TLS certificates (default True).
  1728→
  1729→    Returns
  1730→    -------
  1731→    List[Finding]
  1732→        All findings from the 8-phase infrastructure ghost scan.
  1733→    """
  1734→    findings: List[Finding] = []
  1735→    domain = _extract_domain(target)
  1736→    deductions = 0
  1737→
  1738→    def add(
  1739→        title: str, severity: str, category: str,
  1740→        desc: str, evidence: str, pts: int,
  1741→        dread: float = 0.0, remediation: str = "",
  1742→    ) -> None:
  1743→        nonlocal deductions
  1744→        deductions += pts
  1745→        findings.append(Finding(
  1746→            title=title,
  1747→            severity=severity,
  1748→            category=category,
  1749→            module=MODULE_NAME,
  1750→            description=desc,
  1751→            evidence=evidence,
  1752→            asset=domain,
  1753→            points_deducted=pts,
  1754→            dread_score=dread,
  1755→            remediation=remediation,
  1756→        ))
  1757→
  1758→    # ── Phase 1: IP Discovery & Mapping ─────────────────────────────────
  1759→    ip_data = _run_ip_discovery(target, base_url, timeout, verify_tls)
  1760→    ip_list = ip_data.get("ips", [])
  1761→    if ip_list:
  1762→        add(
  1763→            f"IP infrastructure mapped: {len(ip_list)} address(es)",
  1764→            "info", "ip_discovery",
  1765→            f"Discovered {len(ip_list)} unique IP addresses associated with {domain} "
  1766→            f"via DNS ({len(ip_data['sources'].get('dns', []))}), "
  1767→            f"HTTP redirects ({len(ip_data['sources'].get('http_redirects', []))}), and "
  1768→            f"certificate SANs ({len(ip_data['sources'].get('certificates', []))}).",
  1769→            ", ".join(ip_list[:20]), 0,
  1770→            remediation="Review which IPs are publicly exposed and ensure "
  1771→                        "only necessary services are accessible.",
  1772→        )
  1773→    else:
  1774→        add(
  1775→            "No IPs resolved for target",
  1776→            "low", "ip_discovery",
  1777→            f"Could not resolve any IP addresses for {domain}.",
  1778→            "DNS resolution returned empty", 2,
  1779→            remediation="Verify the target domain is correct and DNS is properly configured.",
  1780→        )
  1781→
  1782→    # Identify cloud-hosted IPs
  1783→    cloud_ips: Dict[str, List[str]] = {}
  1784→    for ip in ip_list:
  1785→        for prov_key, prov_info in CLOUD_PROVIDER_SIGNS.items():
  1786→            for start, end in prov_info.get("ip_ranges", []):
  1787→                if _ip_in_range(ip, start, end):
  1788→                    cloud_ips.setdefault(prov_key, []).append(ip)
  1789→                    break
  1790→    if cloud_ips:
  1791→        providers_str = "; ".join(f"{k} ({len(v)} IPs)" for k, v in cloud_ips.items())
  1792→        add(
  1793→            "Cloud-hosted infrastructure detected",
  1794→            "info", "cloud_detection",
  1795→            f"IP addresses hosted on cloud providers: {providers_str}.",
  1796→            json.dumps({k: v for k, v in cloud_ips.items()}, default=str), 0,
  1797→        )
  1798→
  1799→    # ── Phase 3: CT Log Mining (run before subdomain scan to get subs) ─
  1800→    ct_data = _mine_ct_logs(domain, timeout)
  1801→    ct_subs = ct_data.get("subdomains", [])
  1802→
  1803→    if ct_data.get("error"):
  1804→        add(
  1805→            "Certificate Transparency log query failed",
  1806→            "low", "ct_mining",
  1807→            f"Could not query CT logs for {domain}: {ct_data['error']}",
  1808→            ct_data["error"], 2,
  1809→            remediation="CT logs may be temporarily unavailable. Retry later.",
  1810→        )
  1811→    else:
  1812→        ct_cert_count = ct_data.get("cert_count", 0)
  1813→        ct_sub_count = len(ct_subs)
  1814→        ct_wildcards = ct_data.get("wildcard_count", 0)
  1815→        ct_expired = ct_data.get("expired_count", 0)
  1816→        add(
  1817→            f"CT log mining: {ct_cert_count} certs, {ct_sub_count} subdomains",
  1818→            "info", "ct_mining",
  1819→            f"Certificate Transparency logs reveal {ct_cert_count} certificates for {domain}, "
  1820→            f"{ct_sub_count} unique subdomains, {ct_wildcards} wildcard certs, "
  1821→            f"{ct_expired} expired certs.",
  1822→            ", ".join(ct_subs[:30]), 0,
  1823→        )
  1824→
  1825→        # Organizations
  1826→        ct_orgs = ct_data.get("organizations", [])
  1827→        if ct_orgs:
  1828→            add(
  1829→                f"Certificate organizations disclosed: {len(ct_orgs)}",
  1830→                "low", "ct_mining",
  1831→                f"CT logs expose {len(ct_orgs)} organization names associated with "
  1832→                f"certificates for {domain}: {', '.join(ct_orgs[:10])}.",
  1833→                ", ".join(ct_orgs[:20]), 2,
  1834→                _dread(3, 8, 3, 4, 7),
  1835→                remediation="Consider using Organization Validated (OV) or Extended Validation (EV) "
  1836→                            "certificates to limit information leakage.",
  1837→            )
  1838→
  1839→        # Expired certs
  1840→        if ct_expired > 0:
  1841→            add(
  1842→                f"Expired certificates in CT logs: {ct_expired}",
  1843→                "medium", "ct_mining",
  1844→                f"Found {ct_expired} expired certificates for {domain} in CT logs. "
  1845→                f"Expired certificates may indicate maintenance gaps or "
  1846→                f"abandoned infrastructure that could be hijacked.",
  1847→                f"{ct_expired} expired certs in CT logs", 5,
  1848→                _dread(5, 7, 4, 5, 6),
  1849→                remediation="Audit expired certificates and ensure all active "
  1850→                            "infrastructure has valid, non-expired certificates.",
  1851→            )
  1852→
  1853→        # Wildcard certs
  1854→        if ct_wildcards > 3:
  1855→            add(
  1856→                f"Excessive wildcard certificates: {ct_wildcards}",
  1857→                "medium", "ct_mining",
  1858→                f"Found {ct_wildcards} wildcard certificates for {domain}. "
  1859→                f"Wildcards cover all subdomains, increasing the blast radius "
  1860→                f"if the private key is compromised.",
  1861→                f"{ct_wildcards} wildcard certs", 5,
  1862→                _dread(7, 6, 5, 6, 7),
  1863→                remediation="Minimize wildcard certificate usage. Use individual "
  1864→                            "certificates for high-security subdomains.",
  1865→            )
  1866→
  1867→    # ── Phase 2: Subdomain Infrastructure Scan ──────────────────────────
  1868→    subdomain_results = _run_subdomain_scan(domain, ct_subs, timeout, verify_tls)
  1869→    reachable = [r for r in subdomain_results if r.get("http_reachable")]
  1870→    add(
  1871→        f"Subdomain scan: {len(subdomain_results)} discovered, {len(reachable)} reachable",
  1872→        "info", "subdomain_scan",
  1873→        f"Scanned {len(subdomain_results)} subdomains (combining CT logs + wordlist). "
  1874→        f"{len(reachable)} are HTTP-reachable with active services.",
  1875→        ", ".join(r["host"] for r in reachable[:30]), 0,
  1876→    )
  1877→
  1878→    # Sensitive subdomain exposure
  1879→    sensitive_names = ("admin", "staging", "dev", "test", "internal", "db", "git",
  1880→                       "jenkins", "grafana", "kibana", "phpmyadmin", "manager")
  1881→    exposed_sensitive = [
  1882→        r for r in reachable
  1883→        if any(s in r["host"] for s in sensitive_names)
  1884→    ]
  1885→    if exposed_sensitive:
  1886→        for r in exposed_sensitive:
  1887→            tech_str = ", ".join(r.get("tech_stack", [])) or "unknown"
  1888→            add(
  1889→                f"Sensitive subdomain exposed: {r['host']}",
  1890→                "high", "subdomain_scan",
  1891→                f"Potentially sensitive subdomain {r['host']} is publicly accessible "
  1892→                f"(HTTP {r['status']}). Detected technologies: {tech_str}. "
  1893→                f"IPs: {', '.join(r.get('ips', []))}.",
  1894→                f"{r['host']} -> {r['status']} | IPs: {', '.join(r.get('ips', []))}",
  1895→                10, _dread(8, 8, 8, 7, 8),
  1896→                remediation=f"Restrict access to {r['host']}. Use IP allowlisting, "
  1897→                            f"VPN, or authentication to protect sensitive services.",
  1898→            )
  1899→
  1900→    # Certificate chain issues
  1901→    for r in subdomain_results:
  1902→        cert = r.get("cert", {})
  1903→        if not cert:
  1904→            continue
  1905→        if cert.get("issuer") and "Let's Encrypt" not in cert.get("issuer", "") and "DigiCert" not in cert.get("issuer", "") and "Sectigo" not in cert.get("issuer", ""):
  1906→            # Check if it's a self-signed or unusual issuer
  1907→            issuer = cert.get("issuer", "")
  1908→            if "CN=" in issuer:
  1909→                cn = issuer.split("CN=")[1].split(",")[0].strip()
  1910→                if cn and cn != r["host"] and "." in cn and cn.endswith(domain):
  1911→                    add(
  1912→                        f"Internal CA certificate on public host: {r['host']}",
  1913→                        "medium", "subdomain_scan",
  1914→                        f"{r['host']} uses a certificate issued by an internal CA: {issuer[:128]}. "
  1915→                        f"This may indicate a misconfigured or leaked internal certificate.",
  1916→                        f"Issuer: {issuer[:200]}",
  1917→                        5, _dread(6, 5, 6, 5, 7),
  1918→                        remediation="Use publicly trusted CAs for all internet-facing services. "
  1919→                                    "Rotate any potentially leaked internal certificates.",
  1920→                    )
  1921→
  1922→    # ── Phase 4: CDN / Cloud Provider Detection ─────────────────────────
  1923→    cdn_cloud_data = _run_cdn_cloud_detection(subdomain_results, base_url, timeout, verify_tls)
  1924→    all_cdns = cdn_cloud_data.get("all_cdns", [])
  1925→    all_clouds = cdn_cloud_data.get("all_clouds", [])
  1926→
  1927→    if all_cdns:
  1928→        add(
  1929→            f"CDN detected: {', '.join(all_cdns)}",
  1930→            "info", "cdn_detection",
  1931→            f"CDN services identified across infrastructure: {', '.join(all_cdns)}. "
  1932→            f"Hosts using CDN: {len(cdn_cloud_data.get('cdn_map', {}))}.",
  1933→            json.dumps(cdn_cloud_data.get("cdn_map", {}), default=str)[:512], 0,
  1934→        )
  1935→    else:
  1936→        add(
  1937→            "No CDN detected on any infrastructure",
  1938→            "low", "cdn_detection",
  1939→            f"No CDN fingerprints found on {domain} or any of its subdomains. "
  1940→            f"This may indicate direct origin exposure.",
  1941→            "No CDN headers or body fingerprints detected", 2,
  1942→            _dread(4, 5, 3, 4, 6),
  1943→            remediation="Consider deploying a CDN to add DDoS protection, improve "
  1944→                        "performance, and hide origin infrastructure.",
  1945→        )
  1946→
  1947→    if all_clouds:
  1948→        add(
  1949→            f"Cloud hosting detected: {', '.join(all_clouds)}",
  1950→            "info", "cloud_detection",
  1951→            f"Infrastructure hosted on: {', '.join(all_clouds)}. "
  1952→            f"Host mappings: {json.dumps(cdn_cloud_data.get('cloud_map', {}))[:512]}",
  1953→            json.dumps(cdn_cloud_data.get("cloud_map", {}), default=str)[:512], 0,
  1954→        )
  1955→
  1956→    # ── Phase 5: Technology Stack Clustering ────────────────────────────
  1957→    tech_cluster_data = _run_tech_clustering(subdomain_results)
  1958→    clusters = tech_cluster_data.get("clusters", [])
  1959→    prevalence = tech_cluster_data.get("tech_prevalence", [])
  1960→    unique_techs = tech_cluster_data.get("unique_techs", [])
  1961→
  1962→    if unique_techs:
  1963→        top_techs = prevalence[:8]
  1964→        tech_summary = "; ".join(f"{t} ({c} hosts)" for t, c in top_techs)
  1965→        add(
  1966→            f"Technology fingerprint: {len(unique_techs)} technologies across infrastructure",
  1967→            "info", "tech_clustering",
  1968→            f"Detected {len(unique_techs)} unique technologies. Top: {tech_summary}. "
  1969→            f"{len(clusters)} technology cluster(s) found (hosts sharing >=2 technologies).",
  1970→            tech_summary, 0,
  1971→        )
  1972→
  1973→    if clusters:
  1974→        for cl in clusters[:5]:
  1975→            hosts = cl["hosts"]
  1976→            shared = cl.get("shared_techs", [])
  1977→            add(
  1978→                f"Tech cluster: {len(hosts)} hosts sharing {', '.join(shared[:4])}",
  1979→                "info", "tech_clustering",
  1980→                f"Infrastructure cluster of {len(hosts)} hosts sharing technologies: "
  1981→                f"{', '.join(shared[:6])}. Hosts: {', '.join(hosts[:8])}.",
  1982→                f"Hosts: {', '.join(hosts[:15])} | Shared: {', '.join(shared[:10])}", 0,
  1983→                remediation="Review clustered infrastructure for consistent security "
  1984→                            "hardening across all cluster members.",
