     1→    65→# ---------------------------------------------------------------------------
     2→    66→# Fallback dict-based directed graph (used when networkx is absent)
     3→    67→# ---------------------------------------------------------------------------
     4→    68→
     5→    69→class _FallbackDiGraph:
     6→    70→    """Minimal directed-graph implementation backed by plain dicts."""
     7→    71→
     8→    72→    def __init__(self) -> None:
     9→    73→        self._nodes: Dict[str, Dict[str, Any]] = {}
    10→    74→        self._succ: Dict[str, Dict[str, Dict[str, Any]]] = defaultdict(dict)
    11→    75→        self._pred: Dict[str, Dict[str, Dict[str, Any]]] = defaultdict(dict)
    12→    76→
    13→    77→    # -- mutation ---------------------------------------------------------
    14→    78→    def add_node(self, node: str, **attr: Any) -> None:
    15→    79→        if node not in self._nodes:
    16→    80→            self._nodes[node] = {}
    17→    81→        self._nodes[node].update(attr)
    18→    82→
    19→    83→    def add_edge(self, u: str, v: str, **attr: Any) -> None:
    20→    84→        self.add_node(u)
    21→    85→        self.add_node(v)
    22→    86→        self._succ[u][v] = attr
    23→    87→        self._pred[v][u] = attr
    24→    88→
    25→    89→    def remove_node(self, node: str) -> None:
    26→    90→        self._nodes.pop(node, None)
    27→    91→        for neighbour in list(self._succ.get(node, {})):
    28→    92→            self._pred[neighbour].pop(node, None)
    29→    93→        self._succ.pop(node, None)
    30→    94→        for neighbour in list(self._pred.get(node, {})):
    31→    95→            self._succ[neighbour].pop(node, None)
    32→    96→        self._pred.pop(node, None)
    33→    97→
    34→    98→    # -- queries ----------------------------------------------------------
    35→    99→    def nodes(self, data: bool = False):
    36→   100→        if data:
    37→   101→            return list(self._nodes.items())
    38→   102→        return list(self._nodes.keys())
    39→   103→
    40→   104→    def edges(self, data: bool = False):
    41→   105→        result = []
    42→   106→        for u, nbrs in self._succ.items():
    43→   107→            for v, attr in nbrs.items():
    44→   108→                if data:
    45→   109→                    result.append((u, v, attr))
    46→   110→                else:
    47→   111→                    result.append((u, v))
    48→   112→        return result
    49→   113→
    50→   114→    def successors(self, node: str):
    51→   115→        return list(self._succ.get(node, {}).keys())
    52→   116→
    53→   117→    def predecessors(self, node: str):
    54→   118→        return list(self._pred.get(node, {}).keys())
    55→   119→
    56→   120→    def out_edges(self, node: str, data: bool = False):
    57→   121→        if data:
    58→   122→            return [(node, v, a) for v, a in self._succ.get(node, {}).items()]
    59→   123→        return [(node, v) for v in self._succ.get(node, {}).keys()]
    60→   124→
    61→   125→    def in_edges(self, node: str, data: bool = False):
    62→   126→        if data:
    63→   127→            return [(u, node, a) for u, a in self._pred.get(node, {}).items()]
    64→   128→        return [(u, node) for u in self._pred.get(node, {}).keys()]
    65→   129→
    66→   130→    def has_node(self, node: str) -> bool:
    67→   131→        return node in self._nodes
    68→   132→
    69→   133→    def get_node_data(self, node: str) -> Dict[str, Any]:
    70→   134→        return dict(self._nodes.get(node, {}))
    71→   135→
    72→   136→    def subgraph(self, nodes: Set[str]):
    73→   137→        g = _FallbackDiGraph()
    74→   138→        for n in nodes:
    75→   139→            if n in self._nodes:
    76→   140→                g.add_node(n, **self._nodes[n])
    77→   141→        for u, nbrs in self._succ.items():
    78→   142→            if u in nodes:
    79→   143→                for v, attr in nbrs.items():
    80→   144→                    if v in nodes:
    81→   145→                        g.add_edge(u, v, **attr)
    82→   146→        return g
    83→   147→
    84→   148→    def to_dict(self) -> Dict[str, Any]:
    85→   149→        return {
    86→   150→            "nodes": dict(self._nodes),
    87→   151→            "edges": [
    88→   152→                {"source": u, "target": v, **a}
    89→   153→                for u, nbrs in self._succ.items()
    90→   154→                for v, a in nbrs.items()
    91→   155→            ],
    92→   156→        }
    93→   157→
    94→   158→    @classmethod
    95→   159→    def from_dict(cls, data: Dict[str, Any]):
    96→   160→        g = cls()
    97→   161→        for node, attr in data.get("nodes", {}).items():
    98→   162→            g.add_node(node, **attr)
    99→   163→        for edge in data.get("edges", []):
   100→   164→            src = edge.pop("source")
   101→   165→            tgt = edge.pop("target")
   102→   166→            g.add_edge(src, tgt, **edge)
   103→   167→        return g

   106→   170→def _make_graph():
   107→   171→    """Return a NetworkX DiGraph if available, else the fallback."""
   108→   172→    if HAS_NX:
   109→   173→        return nx.DiGraph()
   110→   174→    return _FallbackDiGraph()

   133→   197→    def _node_id(self, ntype: str, label: str) -> str:

   147→   211→    def _safe_sev(self, raw) -> str:
   148→   212→        """Normalise a severity value to lowercase string."""
   149→   213→        if raw is None:
   150→   214→            return "info"
   151→   215→        s = str(raw).strip().lower()
   152→   216→        if s in SEVERITY_ORDER:
   153→   217→            return s
   154→   218→        return "info"

   160→   224→    def add_scan_result(self, result_dict: Dict[str, Any]) -> None:

   162→   226→
   163→   227→        Creates nodes/edges for the target, every finding, associated
   164→   228→        technologies, and open ports.

   166→   230→        target = result_dict.get("target") or result_dict.get("host", "unknown")

   168→   232→
   169→   233→        # Subdomains
   170→   234→        for sub in result_dict.get("subdomains", []):
   171→   235→            self.add_subdomain(sub, target)
   172→   236→
   173→   237→        # Ports & services

   178→   242→                state = port_entry.get("state", "open")
   179→   243→            else:
   180→   244→                port_num = port_entry
   181→   245→                service = "unknown"
   182→   246→                state = "open"

   184→   248→            if port_num is None:
   185→   249→                continue

   190→   254→
   191→   255→            svc_id = self._add_indexed_node("service", f"{target}:{port_num}:{service}", name=service, port=int(port_num))
   192→   256→            self._add_indexed_edge(port_id, svc_id, "runs_on")
   193→   257→
   194→   258→        # Technologies
   195→   259→        for tech in result_dict.get("technologies", []):
   196→   260→            tech_name = tech if isinstance(tech, str) else tech.get("name", str(tech))
   197→   261→            tech_id = self._add_indexed_node("technology", tech_name)
   198→   262→            target_id = self._node_id("target", target)
   199→   263→            self._add_indexed_edge(target_id, tech_id, "uses_tech")
   200→   264→
   201→   265→        # Findings
   202→   266→        for finding in result_dict.get("findings", []):
   203→   267→            self.add_finding(finding)

   207→   271→            if isinstance(cert, dict):
   208→   272→                cn = cert.get("common_name", cert.get("cn", "unknown"))
   209→   273→                issuer = cert.get("issuer", "")
   210→   274→                expires = cert.get("expires", "")

   212→   276→                cn = str(cert)
   213→   277→                issuer = ""
   214→   278→                expires = ""

   216→   280→            target_id = self._node_id("target", target)

   218→   282→

   221→   285→            if isinstance(ep, dict):
   222→   286→                path = ep.get("path", ep.get("url", str(ep)))
   223→   287→                method = ep.get("method", "GET")
   224→   288→            else:
   225→   289→                path = str(ep)
   226→   290→                method = "GET"

   228→   292→            target_id = self._node_id("target", target)
   229→   293→            self._add_indexed_edge(target_id, ep_id, "exposes")

   231→   295→    def add_finding(self, finding_dict: Dict[str, Any]) -> None:

   241→   305→        fid = self._add_indexed_node(

   257→   321→        # Link to port if specified
   258→   322→        port_num = finding_dict.get("port")

   270→   334→                affected = cve_ref.get("affected", "")
   271→   335→            else:
   272→   336→                cve_id = str(cve_ref)
   273→   337→                cve_sev = severity
   274→   338→                cve_desc = ""
   275→   339→                affected = ""
   276→   340→            if cve_id:
   277→   341→                self.add_cve(cve_id, cve_sev, cve_desc, affected)

   280→   344→        # Link related technologies
   281→   345→        for tech in finding_dict.get("technologies", []):

   359→   423→    def get_blast_radius(self, asset: str) -> List[Dict[str, Any]]:
   360→   424→        """BFS from *asset* to find all connected vulnerabilities.
   361→   425→
   362→   426→        *asset* can be a node label, an IP, or a partial node id.
   363→   427→        """

   365→   429→        if start is None:

   367→   431→

   369→   433→        vulns: List[Dict[str, Any]] = []
   370→   434→        for nid in visited:
   371→   435→            if HAS_NX:
   372→   436→                data = self._g.nodes[nid]
   373→   437→            else:
   374→   438→                data = self._g.get_node_data(nid)

   379→   443→    def find_chains(self, target: str) -> List[List[Dict[str, Any]]]:
   380→   444→        """Find multi-hop attack chains originating from *target*.
   381→   445→
   382→   446→        Each chain follows a pattern like::
   383→   447→
   384→   448→            target → port → service → vulnerability → CVE
   385→   449→        """
   386→   450→        target_id = self._resolve_node(target)
   387→   451→        if target_id is None:
   388→   452→            return []
   389→   453→
   390→   454→        chains: List[List[Dict[str, Any]]] = []
   391→   455→        # DFS to find paths that end at CVE or finding nodes
   392→   456→        stack: List[Tuple[str, List[str]]] = [(target_id, [target_id])]
   393→   457→        seen_chains: Set[tuple] = set()
   394→   458→
   395→   459→        while stack:
   396→   460→            current, path = stack.pop()
   397→   461→            succs = self._g.successors(current) if HAS_NX else self._g.successors(current)
   398→   462→            for nxt in succs:
   399→   463→                if nxt in path:
   400→   464→                    continue
   401→   465→                new_path = path + [nxt]
   402→   466→                if HAS_NX:
   403→   467→                    nd = self._g.nodes[nxt]
   404→   468→                else:
   405→   469→                    nd = self._g.get_node_data(nxt)
   406→   470→                ntype = nd.get("node_type", "")
   407→   471→                # A chain is complete if it reaches a CVE or high-sev finding
   408→   472→                if ntype == "cve" or (ntype == "finding" and SEVERITY_ORDER.get(self._safe_sev(nd.get("severity")), 0) >= 2):
   409→   473→                    chain_key = tuple(new_path)
   410→   474→                    if chain_key not in seen_chains:
   411→   475→                        seen_chains.add(chain_key)
   412→   476→                        chain_data = []
   413→   477→                        for nid in new_path:
   414→   478→                            if HAS_NX:
   415→   479→                                chain_data.append(dict(self._g.nodes[nid]))
   416→   480→                            else:
   417→   481→                                chain_data.append(self._g.get_node_data(nid))
   418→   482→                        chains.append(chain_data)
   419→   483→                elif len(new_path) < 8:
   420→   484→                    stack.append((nxt, new_path))
   421→   485→
   422→   486→        # Sort by severity of the terminal node
   423→   487→        def _chain_sort_key(chain):
   424→   488→            last = chain[-1]
   425→   489→            return SEVERITY_ORDER.get(self._safe_sev(last.get("severity")), 0)
   426→   490→
   427→   491→        chains.sort(key=_chain_sort_key, reverse=True)
   428→   492→        return chains

   430→   494→    def get_vulnerable_assets(self) -> List[Dict[str, Any]]:
   431→   495→        """Return all assets (targets/subdomains) with HIGH or CRITICAL findings."""
   432→   496→        vulnerable: List[Dict[str, Any]] = []
   433→   497→        seen: Set[str] = set()
   434→   498→
   435→   499→        for nid in self._node_type_index.get("finding", set()):
   436→   500→            if HAS_NX:
   437→   501→                data = self._g.nodes[nid]
   438→   502→            else:
   439→   503→                data = self._g.get_node_data(nid)
   440→   504→            sev = self._safe_sev(data.get("severity"))
   441→   505→            if SEVERITY_ORDER.get(sev, 0) < 3:
   442→   506→                continue
   443→   507→            # Walk predecessors to find the owning target/subdomain
   444→   508→            for pred in (self._g.predecessors(nid) if HAS_NX else self._g.predecessors(nid)):
   445→   509→                if HAS_NX:
   446→   510→                    pdata = self._g.nodes[pred]
   447→   511→                else:
   448→   512→                    pdata = self._g.get_node_data(pred)
   449→   513→                ptype = pdata.get("node_type", "")
   450→   514→                if ptype in ("target", "subdomain", "asset") and pred not in seen:
   451→   515→                    seen.add(pred)
   452→   516→                    vulnerable.append({"node_id": pred, **pdata, "critical_finding": data})
   453→   517→        return vulnerable

   455→   519→    def correlate(self) -> List[Dict[str, Any]]:
   456→   520→        """Cross-reference findings to suggest related vulnerabilities.
   457→   521→
   458→   522→        Returns a list of correlation suggestions, each containing
   459→   523→        the two related node ids, the relationship type, and a
   460→   524→        confidence score.
   461→   525→        """
   462→   526→        suggestions: List[Dict[str, Any]] = []
   463→   527→        # Group findings by technology
   464→   528→        tech_findings: Dict[str, List[str]] = defaultdict(list)
   465→   529→        for nid in self._node_type_index.get("finding", set()):
   466→   530→            if HAS_NX:
   467→   531→                data = self._g.nodes[nid]
   468→   532→            else:
   469→   533→                data = self._g.get_node_data(nid)
   470→   534→            for pred in (self._g.predecessors(nid) if HAS_NX else self._g.predecessors(nid)):
   471→   535→                if HAS_NX:
   472→   536→                    pdata = self._g.nodes[pred]
   473→   537→                else:
   474→   538→                    pdata = self._g.get_node_data(pred)
   475→   539→                if pdata.get("node_type") == "technology":
   476→   540→                    tech_findings[pred].append(nid)
   477→   541→
   478→   542→        for tech_id, finding_ids in tech_findings.items():
   479→   543→            if len(finding_ids) < 2:
   480→   544→                continue
   481→   545→            for i in range(len(finding_ids)):
   482→   546→                for j in range(i + 1, len(finding_ids)):
   483→   547→                    f1 = finding_ids[i]
   484→   548→                    f2 = finding_ids[j]
   485→   549→                    if HAS_NX:
   486→   550→                        d1 = self._g.nodes[f1]
   487→   551→                        d2 = self._g.nodes[f2]
   488→   552→                    else:
   489→   553→                        d1 = self._g.get_node_data(f1)
   490→   554→                        d2 = self._g.get_node_data(f2)
   491→   555→                    # If both share a technology, suggest they may be related
   492→   556→                    conf = 0.6
   493→   557→                    if d1.get("severity") == d2.get("severity"):
   494→   558→                        conf += 0.2
   495→   559→                    if d1.get("finding_type") == d2.get("finding_type"):
   496→   560→                        conf += 0.2
   497→   561→                    suggestions.append({
   498→   562→                        "source": f1,
   499→   563→                        "target": f2,
   500→   564→                        "relationship": "shared_technology",
   501→   565→                        "technology": tech_id,
   502→   566→                        "confidence": min(conf, 1.0),
   503→   567→                        "source_severity": d1.get("severity"),
   504→   568→                        "target_severity": d2.get("severity"),
   505→   569→                    })
   506→   570→
   507→   571→        # Also correlate by port: findings on the same port may be related
   508→   572→        port_findings: Dict[str, List[str]] = defaultdict(list)
   509→   573→        for nid in self._node_type_index.get("finding", set()):
   510→   574→            for pred in (self._g.predecessors(nid) if HAS_NX else self._g.predecessors(nid)):
   511→   575→                if HAS_NX:
   512→   576→                    pdata = self._g.nodes[pred]
   513→   577→                else:
   514→   578→                    pdata = self._g.get_node_data(pred)
   515→   579→                if pdata.get("node_type") == "port":
   516→   580→                    port_findings[pred].append(nid)
   517→   581→
   518→   582→        for port_id, fids in port_findings.items():
   519→   583→            if len(fids) < 2:
   520→   584→                continue
   521→   585→            for i in range(len(fids)):
   522→   586→                for j in range(i + 1, len(fids)):
   523→   587→                    f1, f2 = fids[i], fids[j]
   524→   588→                    suggestions.append({
   525→   589→                        "source": f1,
   526→   590→                        "target": f2,
   527→   591→                        "relationship": "shared_port",
   528→   592→                        "port": port_id,
   529→   593→                        "confidence": 0.7,
   530→   594→                    })
   531→   595→
   532→   596→        suggestions.sort(key=lambda s: s["confidence"], reverse=True)
   533→   597→        return suggestions

   535→   599→    def to_cypher(self) -> str:
   536→   600→        """Export the graph as Cypher-like CREATE statements."""
   537→   601→        lines: List[str] = []
   538→   602→        # Nodes
   539→   603→        for nid in (self._g.nodes() if HAS_NX else self._g.nodes()):
   540→   604→            if HAS_NX:
   541→   605→                data = self._g.nodes[nid]
   542→   606→            else:
   543→   607→                data = self._g.get_node_data(nid)
   544→   608→            ntype = data.pop("node_type", "Unknown")
   545→   609→            label = data.pop("label", nid)
   546→   610→            props = json.dumps(data, default=str)
   547→   611→            safe_id = nid.replace(":", "__").replace("-", "_")
   548→   612→            lines.append(f'CREATE ({safe_id}:{ntype} {{label: "{label}", {props[1:-1]}}})')
   549→   613→            # Restore for edge generation below
   550→   614→            data["node_type"] = ntype
   551→   615→            data["label"] = label
   552→   616→
   553→   617→        # Edges
   554→   618→        for u, v, attr in (self._g.edges(data=True) if HAS_NX else self._g.edges(data=True)):
   555→   619→            etype = attr.get("edge_type", "RELATED_TO")
   556→   620→            props = {k: v for k, v in attr.items() if k != "edge_type"}
   557→   621→            props_str = json.dumps(props, default=str)
   558→   622→            safe_u = u.replace(":", "__").replace("-", "_")
   559→   623→            safe_v = v.replace(":", "__").replace("-", "_")
   560→   624→            if props_str == "{}":
   561→   625→                lines.append(f'CREATE ({safe_u})-[:{etype}]->({safe_v})')
   562→   626→            else:
   563→   627→                lines.append(f'CREATE ({safe_u})-[:{etype} {{{props_str[1:-1]}}}]->({safe_v})')
   564→   628→
   565→   629→        return "\n".join(lines)

   568→   632→        """Return a summary dict with node/edge counts broken down by type."""
   569→   633→        node_counts: Dict[str, int] = {}
   570→   634→        for ntype, nids in self._node_type_index.items():
   571→   635→            node_counts[ntype] = len(nids)

   669→   733→    def _resolve_node(self, label: str) -> Optional[str]:
   670→   734→        """Try to resolve a human-readable label to a full node id."""
   671→   735→        # Exact node id
   672→   736→        if self._g.has_node(label):
   673→   737→            return label
   674→   738→        # Try as target
   675→   739→        candidate = self._node_id("target", label)
   676→   740→        if self._g.has_node(candidate):
   677→   741→            return candidate
   678→   742→        # Try as subdomain
   679→   743→        candidate = self._node_id("subdomain", label)
   680→   744→        if self._g.has_node(candidate):
   681→   745→            return candidate
   682→   746→        # Try as asset
   683→   747→        candidate = self._node_id("asset", label)
   684→   748→        if self._g.has_node(candidate):
   685→   749→            return candidate
   686→   750→        # Fuzzy: search all node labels
   687→   751→        for nid in self._g.nodes():
   688→   752→            if HAS_NX:
   689→   753→                data = self._g.nodes[nid]
   690→   754→            else:
   691→   755→                data = self._g.get_node_data(nid)
   692→   756→            if data.get("label") == label or nid.endswith(f":{label}"):
   693→   757→                return nid
   694→   758→        return None

   696→   760→    def _bfs_collect(self, start: str, max_depth: int = 5) -> Set[str]:
   697→   761→        """BFS from *start* up to *max_depth* hops, return visited nodes."""
   698→   762→        visited: Set[str] = set()
   699→   763→        queue: deque = deque()
   700→   764→        queue.append((start, 0))
   701→   765→        visited.add(start)
   702→   766→        while queue:
   703→   767→            node, depth = queue.popleft()
   704→   768→            if depth >= max_depth:
   705→   769→                continue
   706→   770→            for neighbour in (self._g.successors(node) if HAS_NX else self._g.successors(node)):
   707→   771→                if neighbour not in visited:
   708→   772→                    visited.add(neighbour)
   709→   773→                    queue.append((neighbour, depth + 1))
   710→   774→        return visited
