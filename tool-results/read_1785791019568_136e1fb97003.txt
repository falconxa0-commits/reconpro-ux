  2443→        self,
  2444→        target: str,
  2445→        modules: Optional[List[str]] = None,
  2446→        is_local: bool = False,
  2447→    ) -> None:
  2448→        """Run a scan in a background thread."""

  2450→
  2451→        # Set modules to scanning + push module-specific hints

  2462→            )

  2464→                if m in self._module_cells:
  2465→                    self.call_from_thread(self._module_cells[m].set_status, "scanning")
  2466→
  2467→        try:
  2468→            from .scanner import scan, audit_scan
  2469→            from .history import save_scan
  2470→
  2471→            self.call_from_thread(
  2472→                self._chat, f"[{DIM_CYAN}]  ⏳ Scan in progress...[/]"
  2473→            )
  2474→
  2475→            if is_local:
  2476→                result = audit_scan(target=target, modules=modules)
  2477→            else:
  2478→                result = scan(target=target, modules=modules)
  2479→
  2480→            # Process findings
  2481→            data = result.to_dict()
  2482→            self.call_from_thread(self._on_scan_complete, data, modules, is_local)
  2483→
  2484→            # Save to history
  2485→            try:
  2486→                save_scan(data, label="nexus")
  2487→            except Exception:
  2488→                pass
  2489→
  2490→        except Exception as e:

  2492→            # Phase E: error toast with recovery hint
  2493→            self.call_from_thread(self._toast_error, e, context="Scan")
  2494→            # Phase C: push error_state hints
  2495→            self.call_from_thread(self._push_error_hint)
  2496→            if modules:
  2497→                for m in modules:
  2498→                    if m in self._module_cells:
  2499→                        self.call_from_thread(self._module_cells[m].set_status, "error")
  2500→            else:
  2501→                self.call_from_thread(self._set_all_modules_status, "error")
  2502→
  2503→        finally:
  2504→            self.call_from_thread(self.is_scanning.set, False)

  2961→    def on_click(self, event: Click) -> None:
  2962→        """Open finding detail modal for the clicked finding.
  2963→
  2964→        Uses the vertical click position to determine which finding index
  2965→        the user clicked on, mapping it to _findings_list.
  2966→        """
  2967→        try:
  2968→            if (
  2969→                isinstance(event.widget, RichLog)
  2970→                and event.widget.id == "findings-feed"
  2971→                and self._findings_list
  2972→            ):
  2973→                # Get the line offset to determine which finding was clicked.
  2974→                # Each finding entry is 1 line in the RichLog.
  2975→                # We use the y coordinate relative to the widget to estimate the index.
  2976→                region = event.widget.region
  2977→                if region is not None:
  2978→                    # Calculate which line was clicked relative to the scroll position
  2979→                    scroll_offset = event.widget.scroll_y
  2980→                    click_y = event.y - region.y
  2981→                    # Each finding takes 1 line; estimate index from click position
  2982→                    # Account for scroll position
  2983→                    line_height = 1
  2984→                    if hasattr(event.widget, "_line_height"):
  2985→                        line_height = event.widget._line_height
  2986→                    clicked_index = int((click_y + scroll_offset * line_height) / line_height)
  2987→                    # Clamp to valid range
  2988→                    clicked_index = max(0, min(clicked_index, len(self._findings_list) - 1))
  2989→                    finding = self._findings_list[clicked_index]
  2990→                else:
  2991→                    finding = self._findings_list[-1]
  2992→                self.push_screen(FindingDetailModal(finding))
  2993→        except (NoMatches, Exception):
  2994→            pass
