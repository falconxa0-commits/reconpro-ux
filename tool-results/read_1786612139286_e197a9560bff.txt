   126→   126→async function bannerGrab(domain: string, ip: string | null): Promise<{ banners: Array<{ port: number; banner: string; service: string }>; cveMatches: CVEEntry[] }> {
   127→   127→  const target = ip || domain;
   128→   128→  const ports = [21, 22, 25, 80, 110, 143, 443, 445, 993, 995, 1433, 1521, 3306, 3389, 5432, 5900, 6379, 8080, 8443, 8888, 9090, 27017];
   129→   129→  const banners: Array<{ port: number; banner: string; service: string }> = [];
   130→   130→
   131→   131→  for (const port of ports) {
   132→   132→    const banner = await run(
   133→   133→      `echo "" | timeout 4 nc -w3 ${target} ${port} 2>/dev/null | head -c 500`,
   134→   134→      6000
   135→   135→    );
   136→   136→    if (banner && banner.length > 3) {
   137→   137→      const service = detectService(port, banner);
   138→   138→      banners.push({ port, banner: banner.replace(/[\x00-\x1f\x7f]/g, '.').substring(0, 200), service });
   139→   139→    }
   140→   140→  }
   141→   141→
   142→   142→  // HTTP banner via curl
   143→   143→  const httpBanner = await run(`curl -sI --max-time 6 https://${domain} 2>/dev/null | head -20`, 8000);
   144→   144→  if (httpBanner) banners.push({ port: 443, banner: httpBanner.substring(0, 300), service: 'http' });
   145→   145→  const httpBanner2 = await run(`curl -sI --max-time 6 http://${domain} 2>/dev/null | head -20`, 8000);
   146→   146→  if (httpBanner2) banners.push({ port: 80, banner: httpBanner2.substring(0, 300), service: 'http' });
   147→   147→
   148→   148→  // Match banners against CVE database
   149→   149→  const combined = banners.map(b => b.banner).join(' ').toLowerCase();
   150→   150→  const cveMatches = CVE_DATABASE.filter(cve => {
   151→   151→    return cve.affected.some(a => combined.includes(a)) && cve.versionPattern.test(combined);
   152→   152→  });
   153→   153→
   154→   154→  return { banners, cveMatches };
   155→   155→}
   156→   156→
   157→   157→function detectService(port: number, banner: string): string {
   158→   158→  const b = banner.toLowerCase();
   159→   159→  if (port === 22 || b.includes('ssh')) return 'openssh';
   160→   160→  if (port === 21 || b.includes('ftp')) return 'ftp';
   161→   161→  if (port === 25 || b.includes('smtp') || b.includes('postfix') || b.includes('esmtp')) return 'postfix';
   162→   162→  if (port === 80 || port === 8080 || port === 8443 || b.includes('http')) return 'http';
   163→   163→  if (port === 110 || b.includes('pop3')) return 'pop3';
   164→   164→  if (port === 143 || b.includes('imap')) return 'imap';
   165→   165→  if (port === 443 || b.includes('ssl') || b.includes('tls')) return 'https';
   166→   166→  if (port === 3306 || b.includes('mysql')) return 'mysql';
   167→   167→  if (port === 5432 || b.includes('postgresql') || b.includes('postgres')) return 'postgresql';
   168→   168→  if (port === 6379 || b.includes('redis')) return 'redis';
   169→   169→  if (port === 27017 || b.includes('mongodb')) return 'mongodb';
   170→   170→  if (b.includes('nginx')) return 'nginx';
   171→   171→  if (b.includes('apache')) return 'apache';
   172→   172→  if (b.includes('tomcat')) return 'tomcat';
   173→   173→  return 'unknown';
   174→   174→}

   224→   224→      6000

   235→   235→    }

   250→   250→        evidence: `GET ${path} → internal response detected`,
   251→   251→        asset: `${domain}${path}`,
   252→   252→      });
   253→   253→      break;
   254→   254→    }
   255→   255→  }
   256→   256→

   258→   258→  const xssPayloads = ['<script>alert(1)</script>', '"><img src=x onerror=alert(1)>', "'-alert(1)-'"];
   259→   259→  for (const payload of xssPayloads) {

   261→   261→      `curl -s --max-time 5 "https://${domain}/search?q=${encodeURIComponent(payload)}" 2>/dev/null | head -c 2000`,
   262→   262→      7000

   267→   267→        findings.push({

   271→   271→          evidence: `GET /search?q=<payload> → reflected unescaped in HTML`,
   272→   272→          asset: `${domain}/search`,
   273→   273→        });

   289→   289→        asset: domain,

   291→   291→    }

   322→   322→    });
   323→   323→  }

   327→   327→  if (!csrfCheck) {

   333→   333→      asset: `${domain}/login`,
   334→   334→    });
   335→   335→  }
   336→   336→
   337→   337→  // Mixed content detection
   338→   338→  const mixedContent = await run(

   340→   340→    10000

   348→   348→      asset: domain,
   349→   349→    });
   350→   350→  }
   351→   351→
   352→   352→  return findings;
   353→   353→}

   356→   356→// 3. SSL/TLS ATTACK VECTOR SCANNER
   357→   357→// ══════════════════════════════════════════════════════════════════════════════
   358→   358→
   359→   359→async function scanSSLVulns(domain: string): Promise<Finding[]> {
   360→   360→  const findings: Finding[] = [];
   361→   361→
   362→   362→  // Full SSL scan with all cipher suites
   363→   363→  const sslFull = await run(`echo | openssl s_client -connect ${domain}:443 -servername ${domain} 2>/dev/null`, 10000);
   364→   364→  const sslLines = sslFull.split('\n').map(l => l.trim()).filter(Boolean);
   365→   365→
   366→   366→  // Protocol version checks
   367→   367→  const protocolLine = sslLines.find(l => l.startsWith('Protocol'));
   368→   368→  const proto = protocolLine || '';
   369→   369→
   370→   370→  if (proto.includes('TLSv1.0')) {
   371→   371→    findings.push({
   372→   372→      title: 'BEAST Attack Vulnerable — TLS 1.0 Active',
   373→   373→      severity: 'high', category: 'ssl-vuln',
   374→   374→      description: 'TLS 1.0 is vulnerable to BEAST (Browser Exploit Against SSL/TLS). Block cipher attack can decrypt communication blocks. PCI-DSS prohibits TLS 1.0.',
   375→   375→      evidence: proto,
   376→   376→      asset: `${domain}:443`,
   377→   377→    });
   378→   378→  }
   379→   379→  if (proto.includes('TLSv1.1')) {
   380→   380→    findings.push({
   381→   381→      title: 'Deprecated TLS 1.1 in Use',
   382→   382→      severity: 'medium', category: 'ssl-vuln',
   383→   383→      description: 'TLS 1.1 is deprecated (RFC 8996). Remove support to prevent downgrade attacks.',
   384→   384→      evidence: proto,
   385→   385→      asset: `${domain}:443`,
   386→   386→    });
   387→   387→  }
   388→   388→  if (proto.includes('SSLv3')) {
   389→   389→    findings.push({
   390→   390→      title: 'CRITICAL: POODLE — SSLv3 Detected',
   391→   391→      severity: 'critical', category: 'ssl-vuln',
   392→   392→      description: 'SSLv3 is extremely vulnerable to POODLE (Padding Oracle On Downgraded Legacy Encryption). Complete decryption of encrypted sessions possible.',
   393→   393→      evidence: proto,
   394→   394→      asset: `${domain}:443`,
   395→   395→    });
   396→   396→  }
   397→   397→  if (proto.includes('SSLv2')) {
   398→   398→    findings.push({
   399→   399→      title: 'CRITICAL: DROWN Attack — SSLv2 Detected',
   400→   400→      severity: 'critical', category: 'ssl-vuln',
   401→   401→      description: 'SSLv2 enables DROWN (Decrypting RSA with Obsolete and Weakened eNcryption). Cross-protocol attack can decrypt TLS connections.',
   402→   402→      evidence: proto,
   403→   403→      asset: `${domain}:443`,
   404→   404→    });
   405→   405→  }
   406→   406→
   407→   407→  // Cipher analysis
   408→   408→  const cipherLine = sslLines.find(l => l.includes('Cipher'));
   409→   409→  const cipher = cipherLine || '';
   410→   410→
   411→   411→  // Check for weak ciphers by testing multiple
   412→   412→  const weakCiphers = ['RC4', 'DES', '3DES', 'NULL', 'EXPORT', 'MD5', 'RC2', 'IDEA'];
   413→   413→  const cipherSuite = await run(
   414→   414→    `echo | openssl s_client -connect ${domain}:443 -servername ${domain} -cipher 'ALL:eNULL' 2>/dev/null | grep "Cipher"`,
   415→   415→    10000
   416→   416→  );
   417→   417→
   418→   418→  for (const weak of weakCiphers) {
   419→   419→    if (cipherSuite.toUpperCase().includes(weak)) {
   420→   420→      findings.push({
   421→   421→        title: `CRITICAL: Weak Cipher ${weak} Accepted`,
   422→   422→        severity: 'critical', category: 'ssl-vuln',
   423→   423→        description: `Server accepts ${weak} cipher suite. ${weak === 'RC4' ? 'RC4 has known statistical biases exploitable since 2013.' : weak === 'NULL' ? 'NULL cipher means NO encryption — traffic is plaintext.' : weak === 'EXPORT' ? 'EXPORT-grade ciphers are trivially breakable (40-bit keys).' : `${weak} is cryptographically broken and must be disabled.`}`,
   424→   424→        evidence: `Cipher: ${cipherSuite.trim()}`,
   425→   425→        asset: `${domain}:443`,
   426→   426→      });
   427→   427→      break;
   428→   428→    }
   429→   429→  }
   430→   430→
   431→   431→  if (cipher.includes('CBC') && proto.includes('TLSv1')) {
   432→   432→    findings.push({
   433→   433→      title: 'BEAST Susceptible — CBC Mode + TLS 1.0',
   434→   434→      severity: 'medium', category: 'ssl-vuln',
   435→   435→      description: 'CBC cipher suites with TLS 1.0 are vulnerable to BEAST attack. Use GCM-mode ciphers or upgrade to TLS 1.2+.',
   436→   436→      evidence: `${proto} | ${cipher}`,
   437→   437→      asset: `${domain}:443`,
   438→   438→    });
   439→   439→  }
   440→   440→
   441→   441→  // Heartbleed check (CVE-2014-0160)
   442→   442→  const heartbleedCheck = await run(
   443→   443→    `echo | openssl s_client -connect ${domain}:443 -servername ${domain} -tlsextdebug 2>&1 | grep -i "heartbeat"`,
   444→   444→    8000
   445→   445→  );
   446→   446→  if (heartbleedCheck) {
   447→   447→    findings.push({
   448→   448→      title: 'Heartbleed Vulnerability Possible (CVE-2014-0160)',
   449→   449→      severity: 'critical', category: 'ssl-vuln',
   450→   450→      description: 'Server supports heartbeat extension. If running vulnerable OpenSSL (1.0.1-1.0.1f), up to 64KB of server memory can be leaked per request — including private keys, passwords, session tokens.',
   451→   451→      evidence: `Heartbeat extension: ${heartbleedCheck.trim()}`,
   452→   452→      asset: `${domain}:443`,
   453→   453→    });
   454→   454→  }
   455→   455→
   456→   456→  // Certificate issues
   457→   457→  const certDates = await run(
   458→   458→    `echo | openssl s_client -connect ${domain}:443 -servername ${domain} 2>/dev/null | openssl x509 -noout -dates 2>/dev/null`,
   459→   459→    10000
   460→   460→  );
   461→   461→  if (certDates) {
   462→   462→    const notAfter = certDates.match(/notAfter=(.+)/);
   463→   463→    if (notAfter) {
   464→   464→      const expiry = new Date(notAfter[1].trim());
   465→   465→      const now = new Date();
   466→   466→      const daysLeft = Math.floor((expiry.getTime() - now.getTime()) / 86400000);
   467→   467→      if (daysLeft < 0) {
   468→   468→        findings.push({
   469→   469→          title: 'CRITICAL: SSL Certificate EXPIRED',
   470→   470→          severity: 'critical', category: 'ssl-vuln',
   471→   471→          description: `Certificate expired ${Math.abs(daysLeft)} days ago. Visitors see security warnings. Connections may be intercepted.`,
   472→   472→          evidence: `notAfter: ${notAfter[1].trim()} | Expired ${Math.abs(daysLeft)} days ago`,
   473→   473→          asset: `${domain}:443`,
   474→   474→        });
   475→   475→      } else if (daysLeft < 30) {
   476→   476→        findings.push({
   477→   477→          title: `SSL Certificate Expiring in ${daysLeft} Days`,
   478→   478→          severity: 'high', category: 'ssl-vuln',
   479→   479→          description: `Certificate expires soon. Service disruption and security warnings imminent.`,
   480→   480→          evidence: `notAfter: ${notAfter[1].trim()} | ${daysLeft} days remaining`,
   481→   481→          asset: `${domain}:443`,
   482→   482→        });
   483→   483→      }
   484→   484→    }
   485→   485→  }
   486→   486→
   487→   487→  // OCSP stapling
   488→   488→  const ocsp = await run(
   489→   489→    `echo | openssl s_client -connect ${domain}:443 -servername ${domain} -status 2>/dev/null | grep -i "OCSP"`,
   490→   490→    8000
   491→   491→  );
   492→   492→  if (!ocsp) {
   493→   493→    findings.push({
   494→   494→      title: 'OCSP Stapling Not Enabled',
   495→   495→      severity: 'low', category: 'ssl-vuln',
   496→   496→      description: 'Without OCSP stapling, browsers must contact CA separately to verify certificate revocation, adding latency and privacy exposure.',
   497→   497→      evidence: 'No OCSP response in handshake',
   498→   498→      asset: `${domain}:443`,
   499→   499→    });
   500→   500→  }
   501→   501→
   502→   502→  // Perfect Forward Secrecy
   503→   503→  const pfsCheck = await run(
   504→   504→    `echo | openssl s_client -connect ${domain}:443 -servername ${domain} 2>/dev/null | grep -iE "ECDHE-RSA|ECDHE-ECDSA|DHE-RSA|DHE-DSS"`,
   505→   505→    8000
   506→   506→  );
   507→   507→  if (!pfsCheck) {
   508→   508→    findings.push({
   509→   509→      title: 'Perfect Forward Secrecy — Not Supported',
   510→   510→      severity: 'medium', category: 'ssl-vuln',
   511→   511→      description: 'Without PFS, compromise of server private key allows decryption of ALL past sessions captured via passive surveillance.',
   512→   512→      evidence: 'No ECDHE/DHE cipher negotiated',
   513→   513→      asset: `${domain}:443`,
   514→   514→    });
   515→   515→  }
   516→   516→
   517→   517→  return findings;
   518→   518→}

   521→   521→// 4. DNS VULNERABILITY ASSESSMENT
   522→   522→// ══════════════════════════════════════════════════════════════════════════════

   524→   524→async function scanDNSVulns(domain: string): Promise<Finding[]> {
   525→   525→  const findings: Finding[] = [];
   526→   526→
   527→   527→  // Zone transfer (AXFR) attempt
   528→   528→  const nsRecords = await run(`dig +short +time=3 ${domain} NS`, 5000);
   529→   529→  const nsList = nsRecords.split('\n').filter(Boolean);
   530→   530→  for (const ns of nsList.slice(0, 3)) {
   531→   531→    const axfrResult = await run(`dig axfr ${domain} @${ns} +time=5 +tries=1 2>/dev/null`, 7000);
   532→   532→    if (axfrResult && axfrResult.includes('ANSWER SECTION') && axfrResult.split('\n').length > 5) {
   533→   533→      findings.push({
   534→   534→        title: 'CRITICAL: DNS Zone Transfer (AXFR) Successful',
   535→   535→        severity: 'critical', category: 'dns-vuln',
   536→   536→        description: `Full zone transfer from ${ns} revealed complete DNS records including internal hostnames, IP addresses, and service records. Complete infrastructure mapping possible.`,
   537→   537→        evidence: `dig axfr ${domain} @${ns} → ${axfrResult.split('\n').length} records transferred`,
   538→   538→        asset: ns,
   539→   539→      });
   540→   540→      break;
   541→   541→    }
   542→   542→  }
   543→   543→
   544→   544→  // Subdomain takeover detection (CNAME to dead services)
   545→   545→  const takeoverSubs = ['staging', 'dev', 'blog', 'cdn', 'assets', 'static', 'mail', 'api', 'app', 'test'];
   546→   546→  const takeoverCandidates: string[] = [];
   547→   547→  for (const sub of takeoverSubs) {
   548→   548→    const cname = await run(`dig +short +time=2 ${sub}.${domain} CNAME`, 4000);
   549→   549→    if (cname) {
   550→   550→      const deadServices = ['herokuapp.com', 'cloudfront.net', 'github.io', 's3.amazonaws.com',
   551→   551→        'shopify.com', 'fastly.net', 'cloudwaysapps.com', 'pantheon.io'];
   552→   552→      for (const dead of deadServices) {
   553→   553→        if (cname.toLowerCase().includes(dead)) {
   554→   554→          const subExists = await run(`curl -sI --max-time 4 https://${sub}.${domain} 2>/dev/null | head -1`, 6000);
   555→   555→          if (subExists && !subExists.includes('200') && !subExists.includes('301') && !subExists.includes('302')) {
   556→   556→            takeoverCandidates.push(`${sub}.${domain} → CNAME: ${cname} (${dead})`);
   557→   557→          }
   558→   558→        }
   559→   559→      }
   560→   560→    }
   561→   561→  }
   562→   562→  if (takeoverCandidates.length > 0) {
   563→   563→    findings.push({
   564→   564→      title: `Subdomain Takeover Risk — ${takeoverCandidates.length} candidate(s)`,
   565→   565→      severity: 'high', category: 'dns-vuln',
   566→   566→      description: `Subdomains point to third-party services via CNAME but may not be properly provisioned. If the third-party resource is deleted, an attacker can claim the subdomain and serve malicious content.`,
   567→   567→      evidence: takeoverCandidates.join(' | '),
   568→   568→      asset: domain,
   569→   569→    });
   570→   570→  }
   571→   571→
   572→   572→  // DNS cache snooping
   573→   573→  const cacheResult = await run(`dig +time=3 ${domain} @8.8.8.8 ANY +noall +answer 2>/dev/null`, 5000);
   574→   574→  if (cacheResult && cacheResult.split('\n').length > 8) {
   575→   575→    findings.push({
   576→   576→      title: 'DNS Cache Snooping Possible — Excessive Records Visible',
   577→   577→      severity: 'medium', category: 'dns-vuln',
   578→   578→      description: `Recursive DNS query reveals extensive record types. DNS cache can be probed to determine which domains are recently visited by the resolver's clients.`,
   579→   579→      evidence: `dig ANY ${domain} → ${cacheResult.split('\n').length} records from cache`,
   580→   580→      asset: domain,
   581→   581→    });
   582→   582→  }
   583→   583→
   584→   584→  // DNS rebinding protection check
   585→   585→  const dnsRebindHeaders = await run(`curl -sI --max-time 5 https://${domain} 2>/dev/null | grep -iE "x-dns-prefetch|x-content-type|dns-prefetch"`, 7000);
   586→   586→  if (!dnsRebindHeaders) {
   587→   587→    findings.push({
   588→   588→      title: 'DNS Rebinding — No Protection Detected',
   589→   589→      severity: 'medium', category: 'dns-vuln',
   590→   590→      description: 'No DNS rebinding protection headers. Attackers can bypass Same-Origin Policy by controlling DNS to oscillate between their server and target IP.',

   600→   600→// MAIN POST HANDLER
   601→   601→// ══════════════════════════════════════════════════════════════════════════════
   602→   602→
   603→   603→export async function POST(request: NextRequest) {
   604→   604→  try {
   605→   605→    const body = await request.json();
   606→   606→    const target = body.target?.toString().trim().replace(/^https?:\/\//, '').replace(/\/.*$/, '');
   607→   607→    if (!target || !/^[a-zA-Z0-9][\w.-]+$/.test(target)) {
   608→   608→      return NextResponse.json({ error: 'Invalid target domain' }, { status: 400 });
   609→   609→    }
   610→   610→
   611→   611→    // Resolve IP first
   612→   612→    const ip = await run(`dig +short +time=3 +tries=1 ${target} A`, 5000);
   613→   613→    const mainIp = ip?.split('\n')[0]?.trim() || null;
   614→   614→
   615→   615→    // Run all modules in parallel
   616→   616→    const [
   617→   617→      bannerResult,
   618→   618→      httpVulns,
   619→   619→      sslVulns,
   620→   620→      dnsVulns,
   621→   621→    ] = await Promise.all([
   622→   622→      bannerGrab(target, mainIp),
   623→   623→      scanHTTPVulns(target),
   624→   624→      scanSSLVulns(target),
   625→   625→      scanDNSVulns(target),
   626→   626→    ]);
   627→   627→
   628→   628→    // Build response
   629→   629→    const allFindings = [...httpVulns, ...sslVulns, ...dnsVulns];
   630→   630→    const c = allFindings.filter(f => f.severity === 'critical').length;
   631→   631→    const h = allFindings.filter(f => f.severity === 'high').length;
   632→   632→    const m = allFindings.filter(f => f.severity === 'medium').length;
   633→   633→    const exploitSummary = {
   634→   634→      totalCVEs: bannerResult.cveMatches.length,
   635→   635→      criticalCVEs: bannerResult.cveMatches.filter(v => v.severity === 'critical').length,
   636→   636→      weaponized: bannerResult.cveMatches.filter(v => v.exploitAvailable === 'weaponized').length,
   637→   637→      theoretical: bannerResult.cveMatches.filter(v => v.exploitAvailable === 'theoretical').length,
   638→   638→      avgCVSS: bannerResult.cveMatches.length > 0
   639→   639→        ? Math.round(bannerResult.cveMatches.reduce((a, b) => a + b.cvss, 0) / bannerResult.cveMatches.length * 10) / 10
   640→   640→        : 0,
   641→   641→    };
   642→   642→
   643→   643→    return NextResponse.json({
   644→   644→      success: true,
   645→   645→      scan: {
   646→   646→        target,
   647→   647→        ip: mainIp,
   648→   648→        cveFindings: bannerResult.cveMatches.map(v => ({
   649→   649→          cve: v.id, title: v.title, cvss: v.cvss, severity: v.severity,
   650→   650→          affected: v.affected.join(', '), evidence: `Version pattern match via banner grab`,
   651→   651→          remediation: v.remediation, exploitAvailable: v.exploitAvailable, epss: v.epss,
   652→   652→        })),
   653→   653→        httpVulns: httpVulns.map(f => ({
   654→   654→          ...f, proof: f.evidence,
   655→   655→        })),
   656→   656→        sslVulns: sslVulns.map(f => ({
   657→   657→          ...f, risk: f.severity,
   658→   658→        })),
   659→   659→        dnsVulns,
   660→   660→        banners: bannerResult.banners,
   661→   661→        exploitSummary,
   662→   662→        attackSurface: {
   663→   663→          score: Math.min(100, c * 25 + h * 15 + m * 8 + exploitSummary.weaponized * 20),
   664→   664→          vectors: [
   665→   665→            ...(bannerResult.cveMatches.length > 0 ? ['Known CVE Exploits'] : []),
   666→   666→            ...(httpVulns.some(f => f.severity === 'critical') ? ['Web Application Exploits'] : []),
   667→   667→            ...(sslVulns.some(f => f.severity === 'critical') ? ['Cryptographic Attacks'] : []),
   668→   668→            ...(dnsVulns.some(f => f.severity === 'critical') ? ['DNS Infrastructure Attacks'] : []),
   669→   669→            ...(exploitSummary.weaponized > 0 ? ['Weaponized Exploits Available'] : []),
   670→   670→          ],
   671→   671→        },
   672→   672→      },
   673→   673→    });
   674→   674→  } catch (error) {
   675→   675→    return NextResponse.json({ error: 'Vulnerability scan failed: ' + (error instanceof Error ? error.message : 'unknown') }, { status: 500 });
   676→   676→  }
   677→   677→}
