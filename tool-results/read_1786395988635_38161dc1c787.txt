   770→# ComplianceReport
   771→# ══════════════════════════════════════════════════════════════════════
   772→
   773→@dataclass
   774→class ComplianceReport:
   775→    """Structured output of a compliance mapping run."""
   776→
   777→    per_framework: Dict[str, Dict[str, Any]] = field(default_factory=dict)
   778→    findings_by_control: Dict[str, List[Dict[str, Any]]] = field(default_factory=dict)
   779→    overall_compliance: float = 0.0
   780→    recommendations: List[str] = field(default_factory=list)
   781→
   782→    def to_dict(self) -> Dict[str, Any]:
   783→        return {
   784→            "per_framework": self.per_framework,
   785→            "findings_by_control": {
   786→                k: v for k, v in self.findings_by_control.items()
   787→            },
   788→            "overall_compliance": self.overall_compliance,
   789→            "recommendations": self.recommendations,
   790→        }
   791→
   792→    def summary_text(self) -> str:
   793→        """Human-readable multi-line summary."""
   794→        lines = ["=" * 64, "COMPLIANCE MAPPING REPORT", "=" * 64, ""]
   795→        for fw, data in self.per_framework.items():
   796→            pct = data["compliance_pct"]
   797→            covered = data["covered_controls"]
   798→            total = data["total_controls"]
   799→            lines.append(f"  [{fw}]  {pct:.0f}%  ({covered}/{total} controls covered)")
   800→            if data["gaps"]:
   801→                for g in data["gaps"]:
   802→                    lines.append(f"      ⚠ GAP: {g}")
   803→            lines.append("")
   804→        lines.append(f"  Overall Compliance: {self.overall_compliance:.0f}%")
   805→        lines.append("")
   806→        if self.recommendations:
   807→            lines.append("  RECOMMENDATIONS:")
   808→            for r in self.recommendations:
   809→                lines.append(f"    • {r}")
   810→        lines.append("=" * 64)
   811→        return "\n".join(lines)

   815→# ComplianceMapper
   816→# ══════════════════════════════════════════════════════════════════════
   817→
   818→class ComplianceMapper:
   819→    """Map security findings to compliance controls.
   820→
   821→    Parameters
   822→    ----------
   823→    rules : dict | None
   824→        Override the default ``COMPLIANCE_RULES``.
   825→    """
   826→
   827→    def __init__(self, rules: Optional[Dict[str, Dict[str, Any]]] = None) -> None:
   828→        self._rules = rules if rules is not None else COMPLIANCE_RULES
   829→
   830→    # ------------------------------------------------------------------
   831→    # Public API
   832→    # ------------------------------------------------------------------
   833→
   834→    def map_findings(
   835→        self,
   836→        findings: List[Dict[str, Any]],
   837→        frameworks: Optional[List[str]] = None,
   838→    ) -> ComplianceReport:
   839→        """Map each finding to applicable compliance controls.
   840→
   841→        Parameters
   842→        ----------
   843→        findings : list[dict]
   844→            Each dict must have at least ``category`` (str).  Other useful
   845→            keys: ``title``, ``severity``, ``description``, ``cves``,
   846→            ``remediation``.
   847→        frameworks : list[str] | None
   848→            Restrict to these frameworks (e.g. ``["SOC2", "PCI-DSS"]``).  
   849→            ``None`` means all.
   850→        """
   851→        # Filter rules by framework if requested
   852→        if frameworks:
   853→            fw_set = {f.strip().upper() for f in frameworks}
   854→            active_rules = {
   855→                cid: rule for cid, rule in self._rules.items()
   856→                if rule["framework"].upper() in fw_set
   857→            }
   858→        else:
   859→            active_rules = self._rules
   860→
   861→        # Index rules by their mapping categories
   862→        cat_to_controls: Dict[str, List[str]] = {}
   863→        for cid, rule in active_rules.items():
   864→            for cat in rule["mapping_categories"]:
   865→                cat_to_controls.setdefault(cat, []).append(cid)
   866→
   867→        # Map each finding
   868→        findings_by_control: Dict[str, List[Dict[str, Any]]] = {}
   869→        for f in findings:
   870→            raw_cat = f.get("category", f.get("type", ""))
   871→            canonical = _match_category(raw_cat)
   872→            matched_controls = cat_to_controls.get(canonical, [])
   873→            if not matched_controls:
   874→                # Try exact raw match
   875→                matched_controls = cat_to_controls.get(raw_cat.lower().replace("-","_").replace(" ","_"), [])
   876→            for cid in matched_controls:
   877→                findings_by_control.setdefault(cid, []).append(f)
   878→
   879→        # Group rules by framework
   880→        fw_rules: Dict[str, Dict[str, Dict[str, Any]]] = {}
   881→        for cid, rule in active_rules.items():
   882→            fw = rule["framework"]
   883→            fw_rules.setdefault(fw, {})[cid] = rule
   884→
   885→        # Compute per-framework compliance
   886→        per_framework: Dict[str, Dict[str, Any]] = {}
   887→        fw_pcts: List[float] = []
   888→        for fw, rules_dict in fw_rules.items():
   889→            total = len(rules_dict)
   890→            covered = sum(1 for cid in rules_dict if cid in findings_by_control)
   891→            pct = (covered / total * 100.0) if total else 100.0
   892→            gaps = [cid for cid in rules_dict if cid not in findings_by_control]
   893→            per_framework[fw] = {
   894→                "total_controls": total,
   895→                "covered_controls": covered,
   896→                "compliance_pct": round(pct, 1),
   897→                "gaps": gaps,
   898→            }
   899→            fw_pcts.append(pct)
   900→
   901→        overall = sum(fw_pcts) / len(fw_pcts) if fw_pcts else 100.0
   902→
   903→        # Recommendations
   904→        recommendations = self._generate_recommendations(per_framework, findings_by_control)
   905→
   906→        return ComplianceReport(
   907→            per_framework=per_framework,
   908→            findings_by_control=findings_by_control,
   909→            overall_compliance=round(overall, 1),
   910→            recommendations=recommendations,
   911→        )
   912→
   913→    def generate_evidence(self, finding: Dict[str, Any], control_id: str) -> str:
   914→        """Describe how *finding* provides evidence for (or against) *control_id*.
   915→
   916→        Returns a human-readable evidence statement.
   917→        """
   918→        rule = self._rules.get(control_id)
   919→        if not rule:
   920→            return f"No compliance rule found for {control_id}."
   921→
   922→        title = finding.get("title", "Unnamed finding")
   923→        sev = finding.get("severity", "unknown")
   924→        desc = finding.get("description", "")
   925→        fw = rule["framework"]
   926→        cname = rule["control_name"]
   927→        cid = rule["control_id"]
   928→
   929→        if sev in ("high", "critical"):
   930→            return (
   931→                f"FINDING [{sev.upper()}]: {title}\n"
   932→                f"  The presence of this {sev}-severity finding indicates a likely non-compliance "
   933→                f"with {fw} {cid} ({cname}). {desc}\n"
   934→                f"  This finding was automatically mapped because its category relates to "
   935→                f"the control's scope: {', '.join(rule['mapping_categories'])}."
   936→            )
   937→        elif sev == "medium":
   938→            return (
   939→                f"FINDING [{sev.upper()}]: {title}\n"
   940→                f"  This medium-severity finding is potentially relevant to {fw} {cid} ({cname}). "
   941→                f"While not a definitive non-compliance indicator, it suggests the control "
   942→                f"may not be fully implemented. {desc}"
   943→            )
   944→        else:
   945→            return (
   946→                f"FINDING [{sev.upper()}]: {title}\n"
   947→                f"  This informational finding was mapped to {fw} {cid} ({cname}) by category "
   948→                f"association. It does not indicate non-compliance but provides coverage evidence "
   949→                f"that the control area is being tested."
   950→            )
   951→
   952→    # ------------------------------------------------------------------
   953→    # Private helpers
   954→    # ------------------------------------------------------------------
   955→
   956→    def _generate_recommendations(
   957→        self,
   958→        per_framework: Dict[str, Dict[str, Any]],
   959→        findings_by_control: Dict[str, List[Dict[str, Any]]],
   960→    ) -> List[str]:
   961→        recs: List[str] = []
   962→        seen: set = set()
   963→
   964→        for fw, data in per_framework.items():
   965→            pct = data["compliance_pct"]
   966→            gaps = data["gaps"]
   967→
   968→            if pct < 50:
   969→                recs.append(
   970→                    f"{fw} compliance is critically low at {pct:.0f}%. "
   971→                    f"{len(gaps)} of {data['total_controls']} controls have no coverage."
   972→                )
   973→            elif pct < 80:
   974→                recs.append(
   975→                    f"{fw} compliance is at {pct:.0f}%. Address {len(gaps)} uncovered controls."
   976→                )
   977→
   978→            for gid in gaps:
   979→                rule = self._rules.get(gid)
   980→                if not rule:
   981→                    continue
   982→                cat_str = ", ".join(rule["mapping_categories"])
   983→                key = (fw, gid)
   984→                if key not in seen:
   985→                    seen.add(key)
   986→                    recs.append(
   987→                        f"[{fw}] GAP: {gid} {rule['control_name']} — no findings cover this control. "
   988→                        f"Consider testing categories: {cat_str}."
   989→                    )
   990→
   991→        # Severity-based recommendations
   992→        high_sev_controls: Dict[str, int] = {}
   993→        for cid, f_list in findings_by_control.items():
   994→            rule = self._rules.get(cid)
   995→            if not rule:
   996→                continue
   997→            crit_count = sum(1 for f in f_list if f.get("severity") in ("high", "critical"))
   998→            if crit_count > 0:
   999→                high_sev_controls[cid] = crit_count
  1000→
  1001→        for cid, count in sorted(high_sev_controls.items(), key=lambda x: -x[1])[:5]:
  1002→            rule = self._rules.get(cid, {})
  1003→            if (rule.get("framework"), cid) not in seen:
  1004→                recs.append(
  1005→                    f"[{rule.get('framework','')}] {cid} has {count} high/critical findings — "
  1006→                    f"prioritize remediation of {rule.get('control_name', cid)}."
  1007→                )
  1008→
  1009→        return recs
  1010→

  1012→# Module-level convenience exports
  1013→mapper = ComplianceMapper()
  1014→
