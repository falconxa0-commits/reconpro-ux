   518→# ══════════════════════════════════════════════════════════════════════
   519→
   520→COUNTRY_NAMES: Dict[str, str] = {
   521→    code: data["country_name"]
   522→    for code, data in SURVEILLANCE_DATABASE.items()

   525→
   526→def _country_code_to_name(code: str) -> str:
   527→    """Resolve a 2-letter country code to its full name."""
   528→    return COUNTRY_NAMES.get(code.upper(), code.upper())

   558→# SovereigntyMapper
   559→# ══════════════════════════════════════════════════════════════════════
   560→
   561→class SovereigntyMapper:
   562→    """Maps a target to its digital sovereignty profile.
   563→
   564→    Combines ASN analysis, TLS certificate inspection, DNS nameserver
   565→    geolocation, content language detection, and cloud provider region
   566→    detection to build a comprehensive jurisdictional picture.
   567→    """

   573→
   574→    def map_sovereignty(
   575→        self,
   576→        target: str,
   577→        base_url: str,
   578→        geo_data: Optional[Dict[str, Any]] = None,
   579→        tls_data: Optional[Dict[str, Any]] = None,
   580→        timeout: int = 8,
   581→    ) -> Dict[str, Any]:
   582→        """Build a full sovereignty profile for *target*.
   583→
   584→        Returns a dict with keys: ``jurisdictions``, ``primary_jurisdiction``,
   585→        ``cloud_provider``, ``cloud_region``, ``surveillance_risk``,
   586→        ``data_residency_compliance``, ``overall_risk_score``, ``details``.
   587→        """
   588→        host = self._extract_host(base_url)
   589→
   590→        asn_jur = self.analyze_asn_jurisdiction(host, timeout=timeout)
   591→        cert_jur = self.analyze_certificate_jurisdiction(host, timeout=timeout)
   592→        dns_jur = self.analyze_dns_jurisdiction(host, timeout=timeout)
   593→        content_jur = self.analyze_content_jurisdiction(base_url, timeout=timeout)
   594→        cloud_jur = self.analyze_cloud_jurisdiction(base_url, timeout=timeout)
   595→
   596→        all_jurisdictions: List[Jurisdiction] = [asn_jur, cert_jur, dns_jur, content_jur]
   597→
   598→        # Merge into a primary assessment weighted by confidence
   599→        primary = self._merge_jurisdictions(all_jurisdictions)
   600→
   601→        # Surveillance risk assessment
   602→        surveillance = self.assess_surveillance_risk(primary.country)
   603→
   604→        # Data residency compliance
   605→        involved_countries = list({
   606→            j.country for j in all_jurisdictions if j.country != "Unknown"
   607→        })
   608→        compliance = self.assess_data_residency_compliance(involved_countries)
   609→
   610→        risk_score = self._calculate_overall_risk(primary, surveillance, cloud_jur)
   611→
   612→        return {
   613→            "target": target,
   614→            "base_url": base_url,
   615→            "host": host,
   616→            "jurisdictions": [j.to_dict() for j in all_jurisdictions],
   617→            "primary_jurisdiction": primary.to_dict(),
   618→            "cloud_provider": cloud_jur.get("provider", "Unknown"),
   619→            "cloud_region": cloud_jur.get("region", "Unknown"),
   620→            "surveillance_risk": surveillance,
   621→            "data_residency_compliance": compliance,
   622→            "overall_risk_score": risk_score,
   623→            "details": {
   624→                "geo_data_supplied": geo_data is not None,
   625→                "tls_data_supplied": tls_data is not None,
   626→                "timestamp": datetime.now(timezone.utc).isoformat(),
   627→            },
   628→        }

   633→        self, host: str, timeout: int = 8
   634→    ) -> Jurisdiction:

   636→        jur = Jurisdiction(evidence=[])  # type: ignore[arg-type]
   637→
   638→        try:
   639→            result = self._whois_query(host, timeout=timeout)
   640→            if result:
   641→                country = self._extract_country_from_whois(result)
   642→                if country:
   643→                    jur.country = _country_code_to_name(country)
   644→                    jur.confidence = 0.75

   646→
   647→                org = self._extract_org_from_whois(result)
   648→                if org:
   649→                    jur.evidence.append(f"WHOIS Organization: {org}")
   650→
   651→                # Check for known surveillance countries

   653→                if code in SURVEILLANCE_DATABASE:

   656→                    if db["encryption_backdoor_laws"]:

   658→                            "Mandatory backdoor / key escrow"
   659→                        )
   660→                    jur.data_residency_risk = db["risk_level"]
   661→        except Exception as exc:
   662→            jur.evidence.append(f"ASN lookup failed: {exc}")

   664→        return jur

   666→    # ── Certificate jurisdiction ────────────────────────────────────
   667→
   668→    def analyze_certificate_jurisdiction(
   669→        self, host: str, timeout: int = 8
   670→    ) -> Jurisdiction:
   671→        """Analyze TLS certificate issuer to determine jurisdiction."""
   672→        jur = Jurisdiction(evidence=[])  # type: ignore[arg-type]
   673→
   674→        try:
   675→            ctx = ssl.create_default_context()
   676→            ctx.check_hostname = False
   677→            ctx.verify_mode = ssl.CERT_NONE
   678→
   679→            with socket.create_connection((host, 443), timeout=timeout) as sock:
   680→                with ctx.wrap_socket(sock, server_hostname=host) as tls:
   681→                    cert_der = tls.getpeercert(binary_form=True)
   682→                    if cert_der is None:
   683→                        jur.evidence.append("No TLS certificate returned")
   684→                        return jur
   685→
   686→                    import ssl as _ssl_mod
   687→                    # decode_der_subj takes DER; we use getpeercert() dict instead
   688→                    cert_dict = tls.getpeercert()
   689→                    if not cert_dict:
   690→                        jur.evidence.append("Could not parse certificate")
   691→                        return jur
   692→
   693→                    # Extract issuer
   694→                    issuer_parts = cert_dict.get("issuer", [])
   695→                    issuer_str = ", ".join(
   696→                        f"{k}={v}" for item in issuer_parts for k, v in item
   697→                    )
   698→
   699→                    subject_parts = cert_dict.get("subject", [])
   700→                    subject_str = ", ".join(
   701→                        f"{k}={v}" for item in subject_parts for k, v in item
   702→                    )
   703→
   704→                    jur.evidence.append(f"Certificate Subject: {subject_str}")
   705→                    jur.evidence.append(f"Certificate Issuer: {issuer_str}")
   706→
   707→                    # Map known CAs to jurisdictions
   708→                    issuer_country = self._map_ca_to_jurisdiction(issuer_str)
   709→                    if issuer_country:
   710→                        jur.country = issuer_country
   711→                        jur.confidence = 0.55
   712→                        jur.evidence.append(f"CA jurisdiction: {issuer_country}")
   713→
   714→                    # Check for EV certs (higher identity assurance)
   715→                    if "businessCategory" in subject_str or "jurisdiction" in subject_str.lower():
   716→                        jur.evidence.append("Extended Validation (EV) certificate detected")
   717→                        jur.confidence = min(jur.confidence + 0.1, 1.0)
   718→
   719→        except socket.timeout:
   720→            jur.evidence.append("TLS connection timed out")
   721→        except socket.gaierror as exc:
   722→            jur.evidence.append(f"DNS resolution failed: {exc}")
   723→        except ssl.SSLError as exc:
   724→            jur.evidence.append(f"TLS handshake error: {exc}")
   725→        except OSError as exc:
   726→            jur.evidence.append(f"Connection error: {exc}")
   727→
   728→        return jur

   731→
   732→    def analyze_dns_jurisdiction(
   733→        self, host: str, timeout: int = 8
   734→    ) -> Jurisdiction:
   735→        """Check the countries where a host's nameservers are located."""
   736→        jur = Jurisdiction(evidence=[])  # type: ignore[arg-type]
   737→
   738→        try:
   739→            answers = socket.getaddrinfo(host, None)
   740→            ips = list({addr[4][0] for addr in answers if addr[4]})
   741→            jur.evidence.append(f"Resolved IPs: {', '.join(ips[:5])}")
   742→
   743→            # Try to get NS records via system resolver
   744→            ns_result = self._dig_ns(host, timeout=timeout)
   745→            if ns_result:
   746→                for ns in ns_result:
   747→                    jur.evidence.append(f"Nameserver: {ns}")
   748→                    ns_country = self._infer_ns_country(ns)
   749→                    if ns_country and jur.country == "Unknown":
   750→                        jur.country = ns_country
   751→                        jur.confidence = 0.45
   752→        except socket.gaierror as exc:
   753→            jur.evidence.append(f"DNS resolution failed: {exc}")
   754→        except Exception as exc:
   755→            jur.evidence.append(f"DNS analysis error: {exc}")
   756→
   757→        return jur

   759→    # ── Content jurisdiction ────────────────────────────────────────
   760→
   761→    def analyze_content_jurisdiction(
   762→        self, base_url: str, timeout: int = 8
   763→    ) -> Jurisdiction:
   764→        """Detect page language, cultural markers, and jurisdiction hints."""
   765→        jur = Jurisdiction(evidence=[])  # type: ignore[arg-type]
   766→
   767→        try:
   768→            req = urllib.request.Request(
   769→                base_url,
   770→                headers={"User-Agent": self._ua},
   771→            )
   772→            with urllib.request.urlopen(req, timeout=timeout) as resp:
   773→                body = resp.read(65_536).decode("utf-8", errors="replace")
   774→                ct = resp.headers.get("Content-Language", "")
   775→                if ct:
   776→                    jur.evidence.append(f"Content-Language header: {ct}")
   777→
   778→                # Detect lang attribute
   779→                lang_match = re.search(r'<html[^>]+lang=["\']([a-zA-Z-]+)["\']', body, re.I)
   780→                if lang_match:
   781→                    lang = lang_match.group(1).split("-")[0].upper()
   782→                    jur.evidence.append(f"HTML lang attribute: {lang_match.group(1)}")
   783→                    country = self._lang_to_country(lang)
   784→                    if country and jur.country == "Unknown":
   785→                        jur.country = country
   786→                        jur.confidence = 0.30
   787→
   788→                # Detect cultural / jurisdictional markers
   789→                markers = self._detect_cultural_markers(body)
   790→                jur.evidence.extend(markers)
   791→
   792→                # Detect privacy policy / legal jurisdiction references
   793→                legal_refs = self._detect_legal_references(body)
   794→                jur.evidence.extend(legal_refs)
   795→                for ref in legal_refs:
   796→                    detected_country = self._legal_ref_to_country(ref)
   797→                    if detected_country and jur.country == "Unknown":
   798→                        jur.country = detected_country
   799→                        jur.confidence = 0.60
   800→
   801→        except (urllib.error.URLError, socket.timeout, OSError) as exc:
   802→            jur.evidence.append(f"Content fetch failed: {exc}")
   803→
   804→        return jur

   808→    def analyze_cloud_jurisdiction(
   809→        self, base_url: str, timeout: int = 8
   810→    ) -> Dict[str, str]:
   811→        """Detect AWS, Azure, GCP region from headers, URLs, or page content."""
   812→        result: Dict[str, str] = {"provider": "Unknown", "region": "Unknown"}
   813→
   814→        try:
   815→            req = urllib.request.Request(
   816→                base_url,
   817→                headers={"User-Agent": self._ua},
   818→            )
   819→            with urllib.request.urlopen(req, timeout=timeout) as resp:
   820→                headers = dict(resp.headers)
   821→                body = resp.read(65_536).decode("utf-8", errors="replace")
   822→
   823→            # AWS detection
   824→            if (headers.get("server", "").lower() in ("amazonS3", "awselb", "cloudfront")
   825→                    or "amazonaws.com" in base_url.lower()
   826→                    or "s3.amazonaws.com" in body.lower()
   827→                    or "x-amz-request-id" in headers
   828→                    or "x-amz-cf-id" in headers):
   829→                result["provider"] = "AWS"
   830→                region = self._detect_aws_region(base_url, headers, body)
   831→                if region:
   832→                    result["region"] = region
   833→
   834→            # Azure detection
   835→            elif ("azure" in headers.get("server", "").lower()
   836→                  or ".azurewebsites.net" in base_url.lower()
   837→                  or "x-azure-ref" in headers
   838→                  or "x-ms-request-id" in headers):
   839→                result["provider"] = "Azure"
   840→                region = self._detect_azure_region(headers, body)
   841→                if region:
   842→                    result["region"] = region
   843→
   844→            # GCP detection
   845→            elif ("gke" in headers.get("server", "").lower()
   846→                  or ".appspot.com" in base_url.lower()
   847→                  or ".run.app" in base_url.lower()
   848→                  or "x-goog-request-id" in headers):
   849→                result["provider"] = "GCP"
   850→                region = self._detect_gcp_region(base_url, headers, body)
   851→                if region:
   852→                    result["region"] = region
   853→
   854→            # Cloudflare detection
   855→            elif ("cloudflare" in headers.get("server", "").lower()
   856→                  or "cf-ray" in headers):
   857→                result["provider"] = "Cloudflare"
   858→                cf_ray = headers.get("cf-ray", "")
   859→                if "-" in cf_ray:
   860→                    iata = cf_ray.rsplit("-", 1)[-1]
   861→                    result["region"] = f"Cloudflare edge (IATA: {iata})"
   862→
   863→            # Generic CDN / hosting detection
   864→            elif headers.get("server", ""):
   865→                result["provider"] = f"Hosting: {headers['server']}"
   866→
   867→        except (urllib.error.URLError, socket.timeout, OSError):
   868→            pass
   869→
   870→        return result

   914→    # ── Data residency compliance ───────────────────────────────────
   915→
   916→    def assess_data_residency_compliance(
   917→        self, countries: List[str]
   918→    ) -> Dict[str, Any]:
   919→        """Check GDPR, CCPA, PIPL, LGPD compliance across involved countries.
   920→
   921→        Returns a compliance assessment showing which data protection
   922→        frameworks apply and whether cross-border transfers are at risk.
   923→        """
   924→        applicable_laws: List[Dict[str, Any]] = []
   925→        cross_border_risk = False
   926→        data_localization_required = False
   927→        compliance_gaps: List[str] = []
   928→
   929→        for country in countries:
   930→            code = self._country_name_to_code(country)
   931→            if code and code in DATA_PROTECTION_LAWS:
   932→                law = DATA_PROTECTION_LAWS[code]
   933→                applicable_laws.append({
   934→                    "country": country,
   935→                    "law": law["name"],
   936→                    "full_name": law["full_name"],
   937→                    "compliance_score": law["compliance_score"],
   938→                    "data_localization": law["data_localization"],
   939→                })
   940→                if law["data_localization"]:
   941→                    data_localization_required = True
   942→
   943→        # Check for cross-border transfer issues
   944→        if len(countries) > 1:
   945→            cross_border_risk = True
   946→            compliance_gaps.append(
   947→                "Data spans multiple jurisdictions — cross-border transfer mechanisms required"
   948→            )
   949→
   950→        # Check for conflicting data localization requirements
   951→        localization_countries = [
   952→            c for c, law in zip(countries, applicable_laws)
   953→            if law.get("data_localization")
   954→        ]
   955→        if len(localization_countries) > 1:
   956→            compliance_gaps.append(
   957→                f"Conflicting data localization: {', '.join(localization_countries)}"
   958→            )
   959→
   960→        # Check for surveillance countries
   961→        for country in countries:
   962→            code = self._country_name_to_code(country)
   963→            if code and code in SURVEILLANCE_DATABASE:
   964→                risk = SURVEILLANCE_DATABASE[code]["risk_level"]
   965→                if risk in ("high", "critical"):
   966→                    compliance_gaps.append(
   967→                        f"High surveillance risk in {country} may conflict with data protection obligations"
   968→                    )
   969→
   970→        return {
   971→            "applicable_laws": applicable_laws,
   972→            "cross_border_risk": cross_border_risk,
   973→            "data_localization_required": data_localization_required,
   974→            "compliance_gaps": compliance_gaps,
   975→            "jurisdiction_count": len(countries),
   976→            "assessment": "compliant" if not compliance_gaps else "review_required",
   977→        }
