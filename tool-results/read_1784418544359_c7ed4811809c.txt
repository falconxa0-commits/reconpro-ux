   100→// ─── Mini severity chart (SVG) ───────────────────────────────
   101→
   102→function SeverityDonut({ data }: { data: { name: string; value: number; color: string }[] }) {
   103→  const total = data.reduce((s, d) => s + d.value, 0);
   104→  const segments = useMemo(() => {
   105→    if (total === 0) return [];
   106→    let off = 0;
   107→    return data.map((d) => {
   108→      const pct = d.value / total;
   109→      const seg = { ...d, pct, dasharray: `${pct * 283} ${283}`, offset: -off };
   110→      off += pct * 283;
   111→      return seg;
   112→    });
   113→  }, [data, total]);
   114→  if (total === 0) return <div className="text-xs text-muted-foreground text-center py-4">No data yet</div>;
   115→  return (
   116→    <svg viewBox="0 0 120 120" className="w-32 h-32 mx-auto">
   117→      {segments.map((d) => (
   118→        <circle
   119→          key={d.name}
   120→          cx="60" cy="60" r="45"
   121→          fill="none"
   122→          stroke={d.color}
   123→          strokeWidth="18"
   124→          strokeDasharray={d.dasharray}
   125→          strokeDashoffset={d.offset}
   126→          strokeLinecap="round"
   127→          opacity={0.85}
   128→        />
   129→      ))}
   130→      <text x="60" y="56" textAnchor="middle" fill="#e6edf3" fontSize="22" fontWeight="bold" fontFamily="Geist Sans, sans-serif">{total}</text>
   131→      <text x="60" y="72" textAnchor="middle" fill="#7d8590" fontSize="9" fontFamily="Geist Sans, sans-serif">FINDINGS</text>
   132→    </svg>
   133→  );
   134→}
