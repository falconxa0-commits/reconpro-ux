    75→    75→@dataclass
    76→    76→class LearnedPattern:
    77→    77→    """A single learned pattern with confidence scoring.
    78→    78→
    79→    79→    Attributes:
    80→    80→        pattern_id:    SHA-256 derived unique identifier.
    81→    81→        pattern_type:  One of frequency, correlation, trend, anomaly.
    82→    82→        key:           Primary key identifying the pattern (e.g. module name,
    83→    83→                       error type, correlation pair).
    84→    84→        description:   Human-readable description of the pattern.
    85→    85→        confidence:    0.0–1.0 confidence score.
    86→    86→        support:       Number of observations backing this pattern.
    87→    87→        source:        Origin of the learning event (scan, test, error,
    88→    88→                       code_change).
    89→    89→        created_at:    ISO-8601 timestamp of first observation.
    90→    90→        updated_at:    ISO-8601 timestamp of most recent observation.
    91→    91→        metadata:      Extensible dict for additional pattern data.
    92→    92→    """
    93→    93→    pattern_id: str = ""
    94→    94→    pattern_type: str = PATTERN_FREQUENCY
    95→    95→    key: str = ""
    96→    96→    description: str = ""
    97→    97→    confidence: float = 0.0
    98→    98→    support: int = 0
    99→    99→    source: str = ""
   100→   100→    created_at: str = ""
   101→   101→    updated_at: str = ""
   102→   102→    metadata: Dict[str, Any] = field(default_factory=dict)
   103→   103→
   104→   104→    def __post_init__(self) -> None:
   105→   105→        if not self.created_at:
   106→   106→            self.created_at = _now_iso()
   107→   107→        if not self.updated_at:
   108→   108→            self.updated_at = self.created_at
   109→   109→        if not self.pattern_id:
   110→   110→            self.pattern_id = self._compute_id()
   111→   111→
   112→   112→    def _compute_id(self) -> str:
   113→   113→        raw = f"{self.pattern_type}:{self.key}:{self.source}"
   114→   114→        return hashlib.sha256(raw.encode()).hexdigest()[:24]
   115→   115→
   116→   116→    def age_hours(self) -> float:
   117→   117→        """Hours since last update."""
   118→   118→        try:
   119→   119→            updated = datetime.fromisoformat(self.updated_at)
   120→   120→            delta = datetime.now(timezone.utc) - updated
   121→   121→            return max(0.0, delta.total_seconds() / 3600.0)
   122→   122→        except (ValueError, TypeError):
   123→   123→            return 0.0
   124→   124→
   125→   125→    def decayed_confidence(self, half_life_days: float = DEFAULT_AGE_HALF_LIFE_DAYS) -> float:
   126→   126→        """Confidence reduced by exponential decay based on age."""
   127→   127→        hours = self.age_hours()
   128→   128→        half_life_hours = half_life_days * 24.0
   129→   129→        if half_life_hours <= 0:
   130→   130→            return self.confidence
   131→   131→        decay_factor = 0.5 ** (hours / half_life_hours)
   132→   132→        return max(0.0, self.confidence * decay_factor)
   133→   133→
   134→   134→    def to_dict(self) -> Dict[str, Any]:
   135→   135→        return {
   136→   136→            "pattern_id": self.pattern_id,
   137→   137→            "pattern_type": self.pattern_type,
   138→   138→            "key": self.key,
   139→   139→            "description": self.description,
   140→   140→            "confidence": round(self.confidence, 6),
   141→   141→            "support": self.support,
   142→   142→            "source": self.source,
   143→   143→            "created_at": self.created_at,
   144→   144→            "updated_at": self.updated_at,
   145→   145→            "metadata": self.metadata,
   146→   146→        }
   147→   147→
   148→   148→    @classmethod
   149→   149→    def from_dict(cls, d: Dict[str, Any]) -> LearnedPattern:
   150→   150→        return cls(
   151→   151→            pattern_id=d.get("pattern_id", ""),
   152→   152→            pattern_type=d.get("pattern_type", PATTERN_FREQUENCY),
   153→   153→            key=d.get("key", ""),
   154→   154→            description=d.get("description", ""),
   155→   155→            confidence=float(d.get("confidence", 0.0)),
   156→   156→            support=int(d.get("support", 0)),
   157→   157→            source=d.get("source", ""),
   158→   158→            created_at=d.get("created_at", ""),
   159→   159→            updated_at=d.get("updated_at", ""),
   160→   160→            metadata=dict(d.get("metadata", {})),
   161→   161→        )

   164→   164→# ═══════════════════════════════════════════════════════════════════════════
   165→   165→# PatternStore — Persistent pattern storage with ageing/merging/pruning
   166→   166→# ═══════════════════════════════════════════════════════════════════════════
   167→   167→
   168→   168→
   169→   169→class PatternStore:
   170→   170→    """Persistent pattern storage.
   171→   171→
   172→   172→    Features:
   173→   173→    - Store patterns with confidence scores
   174→   174→    - Age patterns (old patterns lose confidence via exponential decay)
   175→   175→    - Merge similar patterns
   176→   176→    - Prune low-confidence patterns
   177→   177→    - Thread-safe JSON persistence
   178→   178→    """
   179→   179→
   180→   180→    def __init__(
   181→   181→        self,
   182→   182→        path: Optional[Path] = None,
   183→   183→        max_patterns: int = DEFAULT_MAX_PATTERNS,
   184→   184→        half_life_days: float = DEFAULT_AGE_HALF_LIFE_DAYS,
   185→   185→        prune_threshold: float = PRUNE_BELOW_CONFIDENCE,
   186→   186→        merge_threshold: float = MERGE_SIMILARITY_THRESHOLD,
   187→   187→    ) -> None:
   188→   188→        self._path = path or PATTERN_FILE
   189→   189→        self._max_patterns = max(max_patterns, 1)
   190→   190→        self._half_life_days = half_life_days
   191→   191→        self._prune_threshold = prune_threshold
   192→   192→        self._merge_threshold = merge_threshold
   193→   193→        self._lock = threading.RLock()
   194→   194→        self._patterns: Dict[str, LearnedPattern] = {}
   195→   195→        self._loaded: bool = False
   196→   196→
   197→   197→    # ------------------------------------------------------------------
   198→   198→    # Persistence
   199→   199→    # ------------------------------------------------------------------
   200→   200→
   201→   201→    def _ensure_dir(self) -> None:
   202→   202→        self._path.parent.mkdir(parents=True, exist_ok=True)
   203→   203→
   204→   204→    def load(self) -> None:
   205→   205→        """Load patterns from disk."""
   206→   206→        with self._lock:
   207→   207→            self._patterns.clear()
   208→   208→            if not self._path.exists():
   209→   209→                self._loaded = True
   210→   210→                return
   211→   211→            try:
   212→   212→                text = self._path.read_text(encoding="utf-8")
   213→   213→                data = json.loads(text) if text.strip() else {}
   214→   214→                raw_patterns = data.get("patterns", {})
   215→   215→                for pid, pdict in raw_patterns.items():
   216→   216→                    try:
   217→   217→                        self._patterns[pid] = LearnedPattern.from_dict(pdict)
   218→   218→                    except Exception:
   219→   219→                        logger.debug("Skipping corrupt pattern %s", pid)
   220→   220→                self._loaded = True
   221→   221→                logger.info(
   222→   222→                    "Loaded %d learned patterns from %s",
   223→   223→                    len(self._patterns), self._path,
   224→   224→                )
   225→   225→            except (json.JSONDecodeError, OSError) as exc:
   226→   226→                logger.warning("Failed to load patterns from %s: %s", self._path, exc)
   227→   227→                self._loaded = True
   228→   228→
   229→   229→    def save(self) -> None:
   230→   230→        """Persist current patterns to disk."""
   231→   231→        with self._lock:
   232→   232→            self._ensure_dir()
   233→   233→            try:
   234→   234→                data: Dict[str, Any] = {
   235→   235→                    "patterns": {
   236→   236→                        pid: p.to_dict()
   237→   237→                        for pid, p in self._patterns.items()
   238→   238→                    },
   239→   239→                    "saved_at": _now_iso(),
   240→   240→                    "total_patterns": len(self._patterns),
   241→   241→                }
   242→   242→                tmp = self._path.with_suffix(".tmp")
   243→   243→                tmp.write_text(
   244→   244→                    json.dumps(data, indent=2, default=str),
   245→   245→                    encoding="utf-8",
   246→   246→                )
   247→   247→                tmp.replace(self._path)
   248→   248→                logger.debug("Saved %d patterns to %s", len(self._patterns), self._path)
   249→   249→            except OSError as exc:
   250→   250→                logger.error("Failed to save patterns to %s: %s", self._path, exc)
   251→   251→
   252→   252→    def _ensure_loaded(self) -> None:
   253→   253→        if not self._loaded:
   254→   254→            self.load()
   255→   255→
   256→   256→    # ------------------------------------------------------------------
   257→   257→    # CRUD
   258→   258→    # ------------------------------------------------------------------
   259→   259→
   260→   260→    def add(self, pattern: LearnedPattern) -> Optional[LearnedPattern]:
   261→   261→        """Add or update a pattern.  Returns the stored pattern."""
   262→   262→        with self._lock:
   263→   263→            self._ensure_loaded()
   264→   264→            pid = pattern.pattern_id
   265→   265→            existing = self._patterns.get(pid)
   266→   266→            if existing is not None:
   267→   267→                # Merge: weighted average confidence, increment support
   268→   268→                new_conf = _weighted_avg(
   269→   269→                    existing.confidence, existing.support,
   270→   270→                    pattern.confidence, pattern.support,
   271→   271→                )
   272→   272→                existing.confidence = min(1.0, new_conf)
   273→   273→                existing.support += pattern.support
   274→   274→                existing.updated_at = pattern.updated_at or _now_iso()
   275→   275→                if pattern.description and not existing.description:
   276→   276→                    existing.description = pattern.description
   277→   277→                if pattern.metadata:
   278→   278→                    existing.metadata.update(pattern.metadata)
   279→   279→                logger.debug("Updated pattern %s (conf=%.3f, sup=%d)", pid, existing.confidence, existing.support)
   280→   280→                return existing
   281→   281→            else:
   282→   282→                self._patterns[pid] = pattern
   283→   283→                logger.debug("Added pattern %s (type=%s)", pid, pattern.pattern_type)
   284→   284→                return pattern
   285→   285→
   286→   286→    def get(self, pattern_id: str) -> Optional[LearnedPattern]:
   287→   287→        """Retrieve a pattern by ID."""
   288→   288→        with self._lock:
   289→   289→            self._ensure_loaded()
   290→   290→            return self._patterns.get(pattern_id)
   291→   291→
   292→   292→    def remove(self, pattern_id: str) -> bool:
   293→   293→        """Remove a pattern by ID."""
   294→   294→        with self._lock:
   295→   295→            self._ensure_loaded()
   296→   296→            return self._patterns.pop(pattern_id, None) is not None
   297→   297→
   298→   298→    def all_patterns(self) -> List[LearnedPattern]:
   299→   299→        """Return a snapshot of all stored patterns."""
   300→   300→        with self._lock:
   301→   301→            self._ensure_loaded()
   302→   302→            return list(self._patterns.values())
   303→   303→
   304→   304→    def count(self) -> int:
   305→   305→        with self._lock:
   306→   306→            self._ensure_loaded()
   307→   307→            return len(self._patterns)
   308→   308→
   309→   309→    # ------------------------------------------------------------------
   310→   310→    # Ageing & Pruning
   311→   311→    # ------------------------------------------------------------------
   312→   312→
   313→   313→    def decay_all(self) -> int:
   314→   314→        """Apply exponential decay to all pattern confidences.
   315→   315→
   316→   316→        Returns the number of patterns that were decayed.
   317→   317→        """
   318→   318→        decayed = 0
   319→   319→        with self._lock:
   320→   320→            self._ensure_loaded()
   321→   321→            for p in self._patterns.values():
   322→   322→                old_conf = p.confidence
   323→   323→                p.confidence = p.decayed_confidence(self._half_life_days)
   324→   324→                if p.confidence < old_conf - 1e-9:
   325→   325→                    decayed += 1
   326→   326→        return decayed
   327→   327→
   328→   328→    def prune(self) -> int:
   329→   329→        """Remove patterns below the prune confidence threshold.
   330→   330→
   331→   331→        Also enforces the max_patterns cap (LRU by last update).
   332→   332→        Returns count of removed patterns.
   333→   333→        """
   334→   334→        removed = 0
   335→   335→        with self._lock:
   336→   336→            self._ensure_loaded()
   337→   337→            # 1. Remove below threshold
   338→   338→            to_remove: List[str] = []
   339→   339→            for pid, p in self._patterns.items():
   340→   340→                if p.decayed_confidence(self._half_life_days) < self._prune_threshold:
   341→   341→                    to_remove.append(pid)
   342→   342→            for pid in to_remove:
   343→   343→                del self._patterns[pid]
   344→   344→                removed += 1
   345→   345→
   346→   346→            # 2. Enforce cap — remove oldest-updated first
   347→   347→            if len(self._patterns) > self._max_patterns:
   348→   348→                sorted_pids = sorted(
   349→   349→                    self._patterns.keys(),
   350→   350→                    key=lambda k: self._patterns[k].updated_at,
   351→   351→                )
   352→   352→                excess = len(self._patterns) - self._max_patterns
   353→   353→                for pid in sorted_pids[:excess]:
   354→   354→                    del self._patterns[pid]
   355→   355→                    removed += 1
   356→   356→
   357→   357→        if removed:
   358→   358→            logger.info("Pruned %d patterns (store now has %d)", removed, len(self._patterns))
   359→   359→        return removed
   360→   360→
   361→   361→    # ------------------------------------------------------------------
   362→   362→    # Merging
   363→   363→    # ------------------------------------------------------------------
   364→   364→
   365→   365→    def merge_similar(self) -> int:
   366→   366→        """Merge patterns that are sufficiently similar.
   367→   367→
   368→   368→        Two patterns are similar when they share the same type and source,
   369→   369→        and their keys have a Jaccard similarity above the threshold.
   370→   370→        Returns the number of merges performed.
   371→   371→        """
   372→   372→        merged_count = 0
   373→   373→        with self._lock:
   374→   374→            self._ensure_loaded()
   375→   375→            # Group by (type, source)
   376→   376→            groups: Dict[Tuple[str, str], List[str]] = defaultdict(list)
   377→   377→            for pid, p in self._patterns.items():
   378→   378→                groups[(p.pattern_type, p.source)].append(pid)
   379→   379→
   380→   380→            for (ptype, psource), pids in groups.items():
   381→   381→                if len(pids) < 2:
   382→   382→                    continue
   383→   383→                consumed: Set[str] = set()
   384→   384→                for i in range(len(pids)):
   385→   385→                    if pids[i] in consumed:
   386→   386→                        continue
   387→   387→                    for j in range(i + 1, len(pids)):
   388→   388→                        if pids[j] in consumed:
   389→   389→                            continue
   390→   390→                        pa = self._patterns[pids[i]]
   391→   391→                        pb = self._patterns[pids[j]]
   392→   392→                        if self._is_similar(pa, pb):
   393→   393→                            # Merge pb into pa
   394→   394→                            pa.confidence = min(1.0, _weighted_avg(
   395→   395→                                pa.confidence, pa.support,
   396→   396→                                pb.confidence, pb.support,
   397→   397→                            ))
   398→   398→                            pa.support += pb.support
   399→   399→                            pa.updated_at = _now_iso()
   400→   400→                            if not pa.description and pb.description:
   401→   401→                                pa.description = pb.description
   402→   402→                            pa.metadata.update(pb.metadata)
   403→   403→                            del self._patterns[pids[j]]
   404→   404→                            consumed.add(pids[j])
   405→   405→                            merged_count += 1
   406→   406→        if merged_count:
   407→   407→            logger.info("Merged %d similar pattern pairs", merged_count)
   408→   408→        return merged_count
   409→   409→
   410→   410→    def _is_similar(self, a: LearnedPattern, b: LearnedPattern) -> bool:
   411→   411→        """Jaccard similarity on tokenised keys."""
   412→   412→        tokens_a = set(a.key.lower().split(":"))
   413→   413→        tokens_b = set(b.key.lower().split(":"))
   414→   414→        if not tokens_a or not tokens_b:
   415→   415→            return False
   416→   416→        intersection = tokens_a & tokens_b
   417→   417→        union = tokens_a | tokens_b
   418→   418→        return len(intersection) / len(union) >= self._merge_threshold
   419→   419→
   420→   420→    def clear(self) -> int:
   421→   421→        """Remove all patterns. Returns the count removed."""
   422→   422→        with self._lock:
   423→   423→            self._ensure_loaded()
   424→   424→            n = len(self._patterns)
   425→   425→            self._patterns.clear()
   426→   426→            return n
   427→   427→

   429→   429→# ═══════════════════════════════════════════════════════════════════════════
   430→   430→# PatternExtractor — Statistical pattern extraction from raw events
   431→   431→# ═══════════════════════════════════════════════════════════════════════════
   432→   432→
   433→   433→
   434→   434→class PatternExtractor:
   435→   435→    """Extract patterns from raw data using stdlib statistics.
   436→   436→
   437→   437→    Extracts four kinds of patterns:
   438→   438→    1. Frequency — common failures, common findings, recurring events.
   439→   439→    2. Correlation — module A failure co-occurs with module B.
   440→   440→    3. Trend — metrics degrading/improving over time.
   441→   441→    4. Anomaly — events that deviate significantly from the norm.
   442→   442→    """
   443→   443→
   444→   444→    # ── Frequency extraction ──────────────────────────────────────────
   445→   445→
   446→   446→    def extract_frequency(
   447→   447→        self,
   448→   448→        items: List[str],
   449→   449→        source: str = SOURCE_SCAN,
   450→   450→        min_occurrences: int = 2,
   451→   451→        max_patterns: int = 20,
   452→   452→    ) -> List[LearnedPattern]:
   453→   453→        """Extract frequency patterns from a list of item identifiers.
   454→   454→
   455→   455→        Parameters
   456→   456→        ----------
   457→   457→        items:
   458→   458→            List of string identifiers (e.g. finding titles, error messages,
   459→   459→            module names).
   460→   460→        source:
   461→   461→            Origin of the data (scan, test, error, code_change).
   462→   462→        min_occurrences:
   463→   463→            Minimum count to be considered a pattern.
   464→   464→        max_patterns:
   465→   465→            Cap on number of patterns returned.
   466→   466→
   467→   467→        Returns
   468→   468→        -------
   469→   469→        List of LearnedPattern sorted by confidence (descending).
   470→   470→        """
   471→   471→        if not items:
   472→   472→            return []
   473→   473→
   474→   474→        counter = Counter(items)
   475→   475→        total = len(items)
   476→   476→        patterns: List[LearnedPattern] = []
   477→   477→
   478→   478→        for item, count in counter.most_common(max_patterns):
   479→   479→            if count < min_occurrences:
   480→   480→                continue
   481→   481→            freq = count / total
   482→   482→            confidence = min(1.0, freq * 2.0)  # Scale so 50% freq → 1.0
   483→   483→            patterns.append(LearnedPattern(
   484→   484→                pattern_type=PATTERN_FREQUENCY,
   485→   485→                key=f"freq:{item}",
   486→   486→                description=f"'{item}' occurs {count} times ({freq:.1%} of events)",
   487→   487→                confidence=round(confidence, 6),
   488→   488→                support=count,
   489→   489→                source=source,
   490→   490→                metadata={"frequency": round(freq, 6), "total_events": total},
   491→   491→            ))
   492→   492→
   493→   493→        logger.debug("Extracted %d frequency patterns from %d items", len(patterns), total)
   494→   494→        return patterns

   496→   496→    # ── Correlation extraction ────────────────────────────────────────
   497→   497→
   498→   498→    def extract_correlation(
   499→   499→        self,
   500→   500→        event_sets: List[Set[str]],
   501→   501→        source: str = SOURCE_SCAN,
   502→   502→        min_co_occurrence: int = 2,
   503→   503→        max_patterns: int = 20,
   504→   504→    ) -> List[LearnedPattern]:
   505→   505→        """Extract correlation patterns from co-occurring item sets.
   506→   506→
   507→   507→        Parameters
   508→   508→        ----------
   509→   509→        event_sets:
   510→   510→            Each element is a set of items that occurred together in a
   511→   511→            single event (e.g. modules that had findings in one scan).
   512→   512→        source:
   513→   513→            Origin of the data.
   514→   514→        min_co_occurrence:
   515→   515→            Minimum co-occurrence count.
   516→   516→        max_patterns:
   517→   517→            Cap on returned patterns.
   518→   518→
   519→   519→        Returns
   520→   520→        -------
   521→   521→        List of LearnedPattern with correlation data in metadata.
   522→   522→        """
   523→   523→        if not event_sets:
   524→   524→            return []
   525→   525→
   526→   526→        n_events = len(event_sets)
   527→   527→        pair_counts: Dict[Tuple[str, str], int] = Counter()
   528→   528→        item_counts: Counter = Counter()
   529→   529→
   530→   530→        for item_set in event_sets:
   531→   531→            items = sorted(item_set)
   532→   532→            for item in items:
   533→   533→                item_counts[item] += 1
   534→   534→            for i in range(len(items)):
   535→   535→                for j in range(i + 1, len(items)):
   536→   536→                    pair_counts[(items[i], items[j])] += 1
   537→   537→
   538→   538→        patterns: List[LearnedPattern] = []
   539→   539→        for (a, b), co_count in pair_counts.most_common(max_patterns):
   540→   540→            if co_count < min_co_occurrence:
   541→   541→                continue
   542→   542→            # Normalised pointwise mutual information (simplified)
   543→   543→            p_a = item_counts[a] / n_events
   544→   544→            p_b = item_counts[b] / n_events
   545→   545→            p_ab = co_count / n_events
   546→   546→            if p_a > 0 and p_b > 0 and p_ab > 0:
   547→   547→                log_joint = math.log(p_ab)
   548→   548→                log_product = math.log(p_a * p_b)
   549→   549→                log_neg_joint = -log_joint
   550→   550→                if log_neg_joint > 0:
   551→   551→                    npmi = max(-1.0, min(1.0, (log_joint - log_product) / log_neg_joint))
   552→   552→                else:
   553→   553→                    npmi = 0.0
   554→   554→            else:
   555→   555→                npmi = 0.0
   556→   556→
   557→   557→            confidence = min(1.0, (co_count / n_events) * 3.0)
   558→   558→            key = f"corr:{a}:{b}"
   559→   559→            patterns.append(LearnedPattern(
   560→   560→                pattern_type=PATTERN_CORRELATION,
   561→   561→                key=key,
   562→   562→                description=(
   563→   563→                    f"'{a}' and '{b}' co-occur in {co_count}/{n_events} events"
   564→   564→                    f" (NPMI={npmi:.3f})"
   565→   565→                ),
   566→   566→                confidence=round(confidence, 6),
   567→   567→                support=co_count,
   568→   568→                source=source,
   569→   569→                metadata={
   570→   570→                    "item_a": a,
   571→   571→                    "item_b": b,
   572→   572→                    "co_occurrence": co_count,
   573→   573→                    "total_events": n_events,
   574→   574→                    "npmi": round(npmi, 6),
   575→   575→                    "p_a": round(p_a, 6),
   576→   576→                    "p_b": round(p_b, 6),
   577→   577→                },
   578→   578→            ))
   579→   579→
   580→   580→        logger.debug("Extracted %d correlation patterns from %d event sets", len(patterns), n_events)
   581→   581→        return patterns

   583→   583→    # ── Trend extraction ──────────────────────────────────────────────
   584→   584→
   585→   585→    def extract_trend(
   586→   586→        self,
   587→   587→        time_series: List[Tuple[float, float]],
   588→   588→        source: str = SOURCE_SCAN,
   589→   589→        min_points: int = 3,
   590→   590→    ) -> List[LearnedPattern]:
   591→   591→        """Extract trend patterns from a time series.
   592→   592→
   593→   593→        Parameters
   594→   594→        ----------
   595→   595→        time_series:
   596→   596→            List of (timestamp, value) tuples.  Timestamps are assumed
   597→   597→            to be monotonically increasing.
   598→   598→        source:
   599→   599→            Origin of the data.
   600→   600→        min_points:
   601→   601→            Minimum data points required for trend detection.
   602→   602→
   603→   603→        Returns
   604→   604→        -------
   605→   605→        List of LearnedPattern (0 or 1) describing the detected trend.
   606→   606→        """
   607→   607→        if len(time_series) < min_points:
   608→   608→            return []
   609→   609→
   610→   610→        values = [v for _, v in time_series]
   611→   611→        timestamps = [t for t, _ in time_series]
   612→   612→        n = len(values)
   613→   613→
   614→   614→        # Simple linear regression via least squares
   615→   615→        x_mean = statistics.mean(range(n))
   616→   616→        y_mean = statistics.mean(values)
   617→   617→
   618→   618→        ss_xx = sum((i - x_mean) ** 2 for i in range(n))
   619→   619→        ss_xy = sum((i - x_mean) * (v - y_mean) for i, v in enumerate(values))
   620→   620→
   621→   621→        if ss_xx == 0:
   622→   622→            return []
   623→   623→
   624→   624→        slope = ss_xy / ss_xx
   625→   625→        intercept = y_mean - slope * x_mean
   626→   626→
   627→   627→        # R-squared
   628→   628→        ss_res = sum((v - (slope * i + intercept)) ** 2 for i, v in enumerate(values))
   629→   629→        ss_tot = sum((v - y_mean) ** 2 for v in values)
   630→   630→        r_squared = 1.0 - (ss_res / ss_tot) if ss_tot > 0 else 0.0
   631→   631→
   632→   632→        # Standard deviation of residuals for anomaly detection context
   633→   633→        if n > 2:
   634→   634→            residuals = [v - (slope * i + intercept) for i, v in enumerate(values)]
   635→   635→            try:
   636→   636→                residual_std = statistics.stdev(residuals)
   637→   637→            except statistics.StatisticsError:
   638→   638→                residual_std = 0.0
   639→   639→        else:
   640→   640→            residual_std = 0.0
   641→   641→
   642→   642→        # Determine direction
   643→   643→        if abs(slope) < 1e-9:
   644→   644→            direction = "stable"
   645→   645→        elif slope > 0:
   646→   646→            direction = "increasing"
   647→   647→        else:
   648→   648→            direction = "decreasing"
   649→   649→
   650→   650→        # Confidence based on R² and data point count
   651→   651→        confidence = min(1.0, r_squared * (n / 10.0))
   652→   652→
   653→   653→        # Only report meaningful trends
   654→   654→        if confidence < 0.1:
   655→   655→            return []
   656→   656→
   657→   657→        key = f"trend:{direction}:{source}"
   658→   658→        pattern = LearnedPattern(
   659→   659→            pattern_type=PATTERN_TREND,
   660→   660→            key=key,
   661→   661→            description=(
   662→   662→                f"{direction} trend (slope={slope:.4f}, R²={r_squared:.4f})"
   663→   663→                f" over {n} data points"
   664→   664→            ),
   665→   665→            confidence=round(confidence, 6),
   666→   666→            support=n,
   667→   667→            source=source,
   668→   668→            metadata={
   669→   669→                "direction": direction,
   670→   670→                "slope": round(slope, 8),
   671→   671→                "intercept": round(intercept, 8),
   672→   672→                "r_squared": round(r_squared, 6),
   673→   673→                "n_points": n,
   674→   674→                "residual_std": round(residual_std, 8),
   675→   675→                "y_mean": round(y_mean, 6),
   676→   676→                "time_start": round(timestamps[0], 2),
   677→   677→                "time_end": round(timestamps[-1], 2),
   678→   678→            },
   679→   679→        )
   680→   680→
   681→   681→        logger.debug(
   682→   682→            "Extracted trend pattern: %s (conf=%.3f, slope=%.4f)",
   683→   683→            direction, confidence, slope,
   684→   684→        )
   685→   685→        return [pattern]
   686→   686→
   687→   687→    # ── Anomaly extraction ────────────────────────────────────────────
   688→   688→
   689→   689→    def extract_anomaly(
   690→   690→        self,
   691→   691→        values: List[float],
   692→   692→        source: str = SOURCE_SCAN,
   693→   693→        key_prefix: str = "anomaly",
   694→   694→        z_threshold: float = 2.0,
   695→   695→        max_patterns: int = 10,
   696→   696→    ) -> List[LearnedPattern]:
   697→   697→        """Extract anomaly patterns using z-score based outlier detection.
   698→   698→
   699→   699→        Parameters
   700→   700→        ----------
   701→   701→        values:
   702→   702→            List of numeric observations.
   703→   703→        source:
   704→   704→            Origin of the data.
   705→   705→        key_prefix:
   706→   706→            Prefix for the pattern key.
   707→   707→        z_threshold:
   708→   708→            Z-score threshold for anomaly detection.
   709→   709→        max_patterns:
   710→   710→            Cap on returned patterns.
   711→   711→
   712→   712→        Returns
   713→   713→        -------
   714→   714→        List of LearnedPattern for detected anomalies.
   715→   715→        """
   716→   716→        if len(values) < 3:
   717→   717→            return []
   718→   718→
   719→   719→        try:
   720→   720→            mean_val = statistics.mean(values)
   721→   721→            stdev_val = statistics.stdev(values)
   722→   722→        except statistics.StatisticsError:
   723→   723→            return []
   724→   724→
   725→   725→        if stdev_val == 0:
   726→   726→            return []
   727→   727→
   728→   728→        patterns: List[LearnedPattern] = []
   729→   729→        for i, v in enumerate(values):
   730→   730→            z = (v - mean_val) / stdev_val
   731→   731→            if abs(z) >= z_threshold:
   732→   732→                direction = "high" if v > mean_val else "low"
   733→   733→                confidence = min(1.0, abs(z) / 5.0)  # Scale: z=5 → 1.0
   734→   734→                patterns.append(LearnedPattern(
   735→   735→                    pattern_type=PATTERN_ANOMALY,
   736→   736→                    key=f"{key_prefix}:{direction}:{i}",
   737→   737→                    description=(
   738→   738→                        f"Anomalous {direction} value {v:.4f} at index {i}"
   739→   739→                        f" (z={z:.2f}, mean={mean_val:.4f})"
   740→   740→                    ),
   741→   741→                    confidence=round(confidence, 6),
   742→   742→                    support=1,
   743→   743→                    source=source,
   744→   744→                    metadata={
   745→   745→                        "value": v,
   746→   746→                        "z_score": round(z, 6),
   747→   747→                        "mean": round(mean_val, 6),
   748→   748→                        "stdev": round(stdev_val, 6),
   749→   749→                        "index": i,
   750→   750→                        "direction": direction,
   751→   751→                    },
   752→   752→                ))
   753→   753→
   754→   754→        # Cap and sort by confidence
   755→   755→        patterns.sort(key=lambda p: p.confidence, reverse=True)
   756→   756→        patterns = patterns[:max_patterns]
   757→   757→
   758→   758→        logger.debug(
   759→   759→            "Extracted %d anomaly patterns from %d values (threshold=%.1f)",
   760→   760→            len(patterns), len(values), z_threshold,
   761→   761→        )
   762→   762→        return patterns

   803→   803→            seen_combos: Set[frozenset] = set()

   807→   807→                seen_combos.add(frozenset(mods_in_scan))
   808→   808→            if module_sets:
   809→   809→                all_patterns.extend(
   810→   810→                    self.extract_correlation(module_sets, source=SOURCE_SCAN)
   811→   811→                )
   812→   812→
   813→   813→        # 3. Trend: score over time (if we have historical scores in metadata)
   814→   814→        score_history = scan_result.get("score_history", [])
   815→   815→        if not score_history and score:
   816→   816→            # Use single data point — no trend yet
   817→   817→            pass
   818→   818→        if score_history:
   819→   819→            all_patterns.extend(
   820→   820→                self.extract_trend(score_history, source=SOURCE_SCAN)
   821→   821→            )
   822→   822→
   823→   823→        return all_patterns

   948→   948→            files = [change.get("file", "")]
   949→   949→
   950→   950→        # 1. Frequency: commonly changed files/modules
   951→   951→        file_names: List[str] = [f for f in files if isinstance(f, str) and f]
   952→   952→        if file_names:
   953→   953→            all_patterns.extend(
   954→   954→                self.extract_frequency(file_names, source=SOURCE_CODE_CHANGE)
   955→   955→            )
   956→   956→
   957→   957→        # 2. Frequency of change types
   958→   958→        change_types: List[str] = []
   959→   959→        for f in files:
   960→   960→            if isinstance(f, dict):
   961→   961→                ct = f.get("change_type", f.get("action", "modified"))
   962→   962→                change_types.append(ct)
   963→   963→        if change_types:
   964→   964→            all_patterns.extend(
   965→   965→                self.extract_frequency(change_types, source=SOURCE_CODE_CHANGE, max_patterns=5)
   966→   966→            )
   967→   967→
   968→   968→        # 3. Trend: size of changes over time
   969→   969→        size_history = change.get("size_history", [])
   970→   970→        if isinstance(size_history, list) and len(size_history) >= 3:
   971→   971→            ts_pairs: List[Tuple[float, float]] = []
   972→   972→            for entry in size_history:
   973→   973→                if isinstance(entry, (list, tuple)) and len(entry) >= 2:
   974→   974→                    try:
   975→   975→                        ts_pairs.append((float(entry[0]), float(entry[1])))
   976→   976→                    except (TypeError, ValueError):
   977→   977→                        pass
   978→   978→            if len(ts_pairs) >= 3:
   979→   979→                all_patterns.extend(
   980→   980→                    self.extract_trend(ts_pairs, source=SOURCE_CODE_CHANGE)
   981→   981→                )
   982→   982→
   983→   983→        # 4. Anomaly: unusually large changes
   984→   984→        sizes: List[float] = []
   985→   985→        for f in files:
   986→   986→            if isinstance(f, dict):
   987→   987→                for size_key in ("lines_added", "lines_changed", "size"):
   988→   988→                    if size_key in f:
   989→   989→                        try:
   990→   990→                            sizes.append(float(f[size_key]))
   991→   991→                        except (TypeError, ValueError):
   992→   992→                            pass
   993→   993→        if len(sizes) >= 3:
   994→   994→            all_patterns.extend(
   995→   995→                self.extract_anomaly(sizes, source=SOURCE_CODE_CHANGE, key_prefix="change_size")
   996→   996→            )
   997→   997→
   998→   998→        return all_patterns
   999→   999→

  1006→  1006→class PatternQuery:
  1007→  1007→    """Query interface for learned patterns.
  1008→  1008→
  1009→  1009→    Supports filtering by:
  1010→  1010→    - Type (frequency, correlation, trend, anomaly)
  1011→  1011→    - Confidence (minimum threshold)
  1012→  1012→    - Recency (max age in hours)
  1013→  1013→    - Source (scan, test, error, code_change)
  1014→  1014→    - Key substring match
  1015→  1015→    - Correlation with specific items
  1016→  1016→    """
  1017→  1017→
  1018→  1018→    def __init__(self, store: PatternStore) -> None:
  1019→  1019→        self._store = store
  1020→  1020→
  1021→  1021→    def query(
  1022→  1022→        self,
  1023→  1023→        *,
  1024→  1024→        pattern_type: Optional[str] = None,
  1025→  1025→        min_confidence: float = 0.0,
  1026→  1026→        max_age_hours: Optional[float] = None,
  1027→  1027→        source: Optional[str] = None,
  1028→  1028→        key_contains: Optional[str] = None,
  1029→  1029→        limit: int = 100,
  1030→  1030→        sort_by: str = "confidence",
  1031→  1031→        sort_desc: bool = True,
  1032→  1032→    ) -> List[LearnedPattern]:
  1033→  1033→        """Query patterns with optional filters.
  1034→  1034→
  1035→  1035→        Parameters
  1036→  1036→        ----------
  1037→  1037→        pattern_type:
  1038→  1038→            Filter to a specific pattern type.
  1039→  1039→        min_confidence:
  1040→  1040→            Minimum (decayed) confidence threshold.
  1041→  1041→        max_age_hours:
  1042→  1042→            Maximum age of the pattern's last update.
  1043→  1043→        source:
  1044→  1044→            Filter to a specific learning source.
  1045→  1045→        key_contains:
  1046→  1046→            Substring match on pattern key.
  1047→  1047→        limit:
  1048→  1048→            Maximum number of results.
  1049→  1049→        sort_by:
  1050→  1050→            Field to sort by (confidence, support, updated_at).
  1051→  1051→        sort_desc:
  1052→  1052→            Sort descending if True.
  1053→  1053→
  1054→  1054→        Returns
  1055→  1055→        -------
  1056→  1056→        Filtered and sorted list of LearnedPattern.
  1057→  1057→        """
  1058→  1058→        patterns = self._store.all_patterns()
  1059→  1059→
  1060→  1060→        # Apply filters
  1061→  1061→        if pattern_type is not None:
  1062→  1062→            patterns = [p for p in patterns if p.pattern_type == pattern_type]
  1063→  1063→        if source is not None:
  1064→  1064→            patterns = [p for p in patterns if p.source == source]
  1065→  1065→        if key_contains is not None:
  1066→  1066→            lc = key_contains.lower()
  1067→  1067→            patterns = [p for p in patterns if lc in p.key.lower()]
  1068→  1068→        if min_confidence > 0:
  1069→  1069→            patterns = [
  1070→  1070→                p for p in patterns
  1071→  1071→                if p.decayed_confidence(self._store._half_life_days) >= min_confidence
  1072→  1072→            ]
  1073→  1073→        if max_age_hours is not None:
  1074→  1074→            patterns = [
  1075→  1075→                p for p in patterns
  1076→  1076→                if p.age_hours() <= max_age_hours
  1077→  1077→            ]
  1078→  1078→
  1079→  1079→        # Sort
  1080→  1080→        if sort_by == "confidence":
  1081→  1081→            patterns.sort(key=lambda p: p.decayed_confidence(self._store._half_life_days), reverse=sort_desc)
  1082→  1082→        elif sort_by == "support":
  1083→  1083→            patterns.sort(key=lambda p: p.support, reverse=sort_desc)
  1084→  1084→        elif sort_by == "updated_at":
  1085→  1085→            patterns.sort(key=lambda p: p.updated_at, reverse=sort_desc)
  1086→  1086→
  1087→  1087→        return patterns[:limit]
  1088→  1088→
  1089→  1089→    def get_correlations_for(
  1090→  1090→        self,
  1091→  1091→        item: str,
  1092→  1092→        min_confidence: float = 0.1,
  1093→  1093→    ) -> List[LearnedPattern]:
  1094→  1094→        """Find all correlation patterns involving *item*."""
  1095→  1095→        lc = item.lower()
  1096→  1096→        results = [
  1097→  1097→            p for p in self._store.all_patterns()
  1098→  1098→            if p.pattern_type == PATTERN_CORRELATION
  1099→  1099→            and lc in p.key.lower()
  1100→  1100→            and p.decayed_confidence(self._store._half_life_days) >= min_confidence
  1101→  1101→        ]
  1102→  1102→        results.sort(key=lambda p: p.confidence, reverse=True)
  1103→  1103→        return results

  1359→  1359→
  1360→  1360→        Uses frequency and correlation patterns to make predictions

  1362→  1362→

  1367→  1367→            - ``module``: scanner module being used
  1368→  1368→            - ``target``: target being scanned
  1369→  1369→            - ``finding_type``: type of finding expected
  1370→  1370→            - ``source``: learning source to query
  1371→  1371→

  1375→  1375→        - ``predicted_findings``: list of likely finding patterns
  1376→  1376→        - ``predicted_errors``: list of likely error patterns
  1377→  1377→        - ``risk_assessment``: overall risk based on learned data
  1378→  1378→        - ``confidence``: prediction confidence
  1379→  1379→        - ``supporting_patterns``: patterns that support the prediction
  1380→  1380→        """

  1392→  1392→        target = scenario.get("target", "")
  1393→  1393→
  1394→  1394→        # 1. Find frequency patterns relevant to this scenario
  1395→  1395→        freq_patterns = self._query.query(
  1396→  1396→            pattern_type=PATTERN_FREQUENCY,
  1397→  1397→            source=source,
  1398→  1398→            min_confidence=0.15,
  1399→  1399→            limit=10,
  1400→  1400→        )
  1401→  1401→
