   804→# ──────────────────────────────────────────────────────────────────────────────
   805→# TECH_DETECTOR — Fingerprinting engine
   806→# ──────────────────────────────────────────────────────────────────────────────
   807→
   808→class TechDetector:
   809→    """Analyzes HTTP responses, headers, HTML, and URLs to identify
   810→    the technology stack of a target application."""
   811→
   812→    def __init__(self) -> None:
   813→        self._reset()
   814→
   815→    def _reset(self) -> None:
   816→        self._frameworks: set[str] = set()
   817→        self._language: Optional[str] = None
   818→        self._server: Optional[str] = None
   819→        self._waf: Optional[str] = None
   820→        self._cms: Optional[str] = None
   821→        self._has_graphql: bool = False
   822→        self._has_rest_api: bool = False
   823→        self._confidence: float = 0.0
   824→        self._hits: int = 0
   825→
   826→    def analyze(
   827→        self,
   828→        target: str,
   829→        response_headers: dict[str, str],
   830→        html_body: str = "",
   831→        js_files: Optional[list[str]] = None,
   832→    ) -> TechProfile:
   833→        """Run full analysis and return a TechProfile."""
   834→        self._reset()
   835→        if js_files is None:
   836→            js_files = []
   837→
   838→        self._analyze_headers(response_headers)
   839→        self._analyze_html(html_body)
   840→        self._analyze_cookies(response_headers)
   841→        self._analyze_url(target)
   842→        self._analyze_js_files(js_files)
   843→        self._analyze_url_paths(target)
   844→
   845→        if self._hits > 0:
   846→            self._confidence = min(self._hits * 0.12, 1.0)
   847→
   848→        return TechProfile(
   849→            frameworks=list(self._frameworks),
   850→            language=self._language,
   851→            server=self._server,
   852→            waf=self._waf,
   853→            cms=self._cms,
   854→            has_graphql=self._has_graphql,
   855→            has_rest_api=self._has_rest_api,
   856→            confidence=self._confidence,
   857→        )

   859→    def _analyze_headers(self, headers: dict[str, str]) -> None:
   860→        header_signatures = TECH_SIGNATURES["headers"]
   861→        header_map = TECH_SIGNATURES["waf_signatures"]
   862→
   863→        for header_key, mapping in header_signatures.items():
   864→            header_val = headers.get(header_key, "")
   865→            if not isinstance(mapping, dict):
   866→                continue
   867→            for pattern, tech in mapping.items():
   868→                if isinstance(tech, str) and tech and pattern:
   869→                    if pattern.lower() in header_val.lower():
   870→                        if tech == "Drupal" or tech == "WordPress" or tech == "Joomla":
   871→                            self._cms = tech
   872→                            self._hits += 2
   873→                        elif tech in ("nginx", "Apache", "IIS", "Caddy", "LiteSpeed", "OpenResty"):
   874→                            self._server = tech
   875→                            self._hits += 1
   876→                        elif tech == "PHP":
   877→                            self._language = "PHP"
   878→                            self._hits += 1
   879→                        elif tech == "Python":
   880→                            self._language = "Python"
   881→                            self._hits += 1
   882→                        elif tech == "Java":
   883→                            self._language = "Java"
   884→                            self._hits += 1
   885→                        elif tech == "Ruby":
   886→                            self._language = "Ruby"
   887→                            self._hits += 1
   888→                        elif tech == "ASP.NET" or tech == "ASP.NET MVC" or tech == "ASP.NET Core":
   889→                            self._language = "ASP.NET"
   890→                            self._frameworks.add("ASP.NET")
   891→                            self._hits += 1
   892→                        elif tech in ("Express", "Django", "Flask", "Tornado", "Next.js", "React", "Vue", "Nuxt"):
   893→                            self._frameworks.add(tech)
   894→                            self._hits += 1
   895→                        elif tech == "Cloudflare":
   896→                            self._waf = "Cloudflare WAF"
   897→                            self._hits += 1
   898→                        else:
   899→                            self._frameworks.add(tech)
   900→                            self._hits += 1
   901→
   902→                if pattern == "" and mapping.get(header_key, "") == "" and header_val:
   903→                    tech_name = list(mapping.values())[0] if mapping else ""
   904→                    if tech_name and tech_name == "":
   905→                        tech_name = header_key.split("-")[-1].strip()
   906→                        self._hits += 1
   907→
   908→        for sig_header, waf_name in header_map.items():
   909→            if sig_header in headers:
   910→                self._waf = waf_name
   911→                self._hits += 1
   912→                break
   913→
   914→    def _analyze_html(self, html: str) -> None:
   915→        if not html:
   916→            return
   917→        html_lower = html.lower()
   918→
   919→        meta_signatures = TECH_SIGNATURES["html_meta"]
   920→        for meta_type, mapping in meta_signatures.items():
   921→            if meta_type == "generator":
   922→                match = re.search(
   923→                    r'<meta\s+[^>]*?name=["\']generator["\'][^>]*?content=["\']([^"\']+)["\']',
   924→                    html_lower,
   925→                    re.IGNORECASE,
   926→                )
   927→                if match:
   928→                    gen_value = match.group(1).strip().lower()
   929→                    for pattern, tech in mapping.items():
   930→                        if pattern.lower() in gen_value:
   931→                            if tech in ("WordPress", "Joomla", "Drupal", "Ghost", "vBulletin", "phpBB", "MediaWiki"):
   932→                                self._cms = tech
   933→                            else:
   934→                                self._frameworks.add(tech)
   935→                            self._hits += 2
   936→                            break
   937→
   938→            if meta_type == "viewport":
   939→                if 'name="viewport"' in html_lower or 'content="width=device-width' in html_lower:
   940→                    self._hits += 0.5
   941→
   942→        ext_signatures = TECH_SIGNATURES["file_extensions"]
   943→        for ext, lang in ext_signatures.items():
   944→            safe_ext = re.escape(ext)
   945→            pattern = rf'(?:src|href)=["\'][^"\']*{safe_ext}(?:\?[^"\']*)?["\']'
   946→            if re.search(pattern, html_lower):
   947→                if self._language is None:
   948→                    self._language = lang
   949→                self._hits += 1
   950→
   951→    def _analyze_cookies(self, headers: dict[str, str]) -> None:
   952→        cookie_header = headers.get("Set-Cookie", "") or headers.get("Cookie", "")
   953→        cookie_signatures = TECH_SIGNATURES["cookies"]
   954→
   955→        for cookie_name, lang in cookie_signatures.items():
   956→            if cookie_name.lower() in cookie_header.lower():
   957→                if lang == "PHP" and self._language is None:
   958→                    self._language = "PHP"
   959→                elif lang == "Java" and self._language is None:
   960→                    self._language = "Java"
   961→                elif lang == "ASP.NET" and self._language is None:
   962→                    self._language = "ASP.NET"
   963→                elif lang == "Rails" and self._language is None:
   964→                    self._language = "Ruby"
   965→                    self._frameworks.add("Rails")
   966→                elif lang == "Laravel":
   967→                    self._frameworks.add("Laravel")
   968→                elif lang == "CodeIgniter":
   969→                    self._frameworks.add("CodeIgniter")

  1130→class FuzzSession:
  1131→    """Context-aware fuzzing session that selects payloads based on the
  1132→    target's detected technology stack."""
  1133→
  1134→    def __init__(
  1135→        self,
  1136→        target: str,
  1137→        tech_profile: Optional[TechProfile] = None,
  1138→        timeout: float = 10.0,
  1139→        user_agent: str = "ReconPro/7.5 (Security Scanner)",
  1140→        verify_ssl: bool = False,
  1141→    ) -> None:
  1142→        self.target = target.rstrip("/")
  1143→        self.tech_profile = tech_profile
  1144→        self.timeout = timeout
  1145→        self.user_agent = user_agent
  1146→        self.verify_ssl = verify_ssl
  1147→        self.results: list[FuzzResult] = []
  1148→        self._base_time: float = 0.0
  1149→        self._baseline_status: int = 0
  1150→        self._opener = self._build_opener()
  1151→
  1152→    def _build_opener(self) -> urllib.request.OpenerDirector:
  1153→        """Build a URL opener with optional SSL bypass."""
  1154→        if not self.verify_ssl:
  1155→            import ssl
  1156→            ctx = ssl._create_unverified_context()
  1157→            https_handler = urllib.request.HTTPSHandler(context=ctx)
  1158→            return urllib.request.build_opener(https_handler)
  1159→        return urllib.request.build_opener()
  1160→
  1161→    def _make_request(
  1162→        self,
  1163→        url: str,
  1164→        params: Optional[dict[str, str]] = None,
  1165→        headers: Optional[dict[str, str]] = None,
  1166→        method: str = "GET",
  1167→    ) -> tuple[int, str, float, dict[str, str]]:
  1168→        """Make an HTTP request and return (status_code, body, time, headers)."""
  1169→        if params and method == "GET":
  1170→            query = urllib.parse.urlencode(params)
  1171→            url = f"{url}?{query}"
  1172→
  1173→        req_headers = {
  1174→            "User-Agent": self.user_agent,
  1175→            "Accept": "text/html,application/xhtml+xml,application/xml;q=0.9,*/*;q=0.8",
  1176→            "Accept-Language": "en-US,en;q=0.5",
  1177→        }
  1178→        if headers:
  1179→            req_headers.update(headers)
  1180→
  1181→        body_data: Optional[bytes] = None
  1182→        if params and method == "POST":
  1183→            body_data = urllib.parse.urlencode(params).encode("utf-8")
  1184→            req_headers["Content-Type"] = "application/x-www-form-urlencoded"
  1185→
  1186→        req = urllib.request.Request(url, data=body_data, headers=req_headers, method=method)
  1187→
  1188→        try:
  1189→            start = time.monotonic()
  1190→            resp = self._opener.open(req, timeout=self.timeout)
  1191→            elapsed = time.monotonic() - start
  1192→            resp_body = resp.read().decode("utf-8", errors="replace")
  1193→            resp_headers = dict(resp.headers)
  1194→            return resp.status, resp_body, elapsed, resp_headers
  1195→        except urllib.error.HTTPError as e:
  1196→            elapsed = time.monotonic() - time.monotonic() + 0.001
  1197→            try:
  1198→                resp_body = e.read().decode("utf-8", errors="replace")
  1199→            except Exception:
  1200→                resp_body = ""
  1201→            resp_headers = dict(e.headers) if hasattr(e, "headers") else {}
  1202→            return e.code, resp_body, elapsed, resp_headers
  1203→        except urllib.error.URLError:
  1204→            elapsed = 0.0
  1205→            return 0, "", elapsed, {}
  1206→        except Exception:
  1207→            elapsed = 0.0
  1208→            return 0, "", elapsed, {}
  1209→
  1210→    def calibrate(self) -> None:
  1211→        """Send a baseline request to establish normal response time and status."""
  1212→        status, _, elapsed, _ = self._make_request(self.target)
  1213→        self._baseline_status = status
  1214→        self._base_time = elapsed

  1216→    def get_all_categories(self) -> list[str]:
  1217→        """Return a list of all available payload categories."""
  1218→        return list(PAYLOAD_DATABASE.keys())
  1219→
  1220→    def select_payloads(self, category: str) -> list[str]:
  1221→        """Select the most relevant payloads for the detected tech stack.
  1222→
  1223→        Priority logic:
  1224→          1. If tech_profile is available, prefer framework-specific payloads.
  1225→          2. Language-specific SQL payloads (e.g., Django -> postgres, WordPress -> mysql).
  1226→          3. Framework-specific template payloads (e.g., Django -> jinja2/django SSTI).
  1227→          4. CMS-specific extras for WordPress/Drupal/Joomla.
  1228→          5. Generic payloads always included as fallback.
  1229→        """
  1230→        if category not in PAYLOAD_DATABASE:
  1231→            return []
  1232→
  1233→        cat_payloads = PAYLOAD_DATABASE[category]
  1234→        selected: list[str] = []
  1235→        seen: set[str] = set()
  1236→
  1237→        def _add(payloads: list[str]) -> None:
  1238→            for p in payloads:
  1239→                if p not in seen:
  1240→                    selected.append(p)
  1241→                    seen.add(p)
  1242→
  1243→        # -- If we have a tech profile, prioritize matching payloads --------
  1244→        if self.tech_profile:
  1245→            tp = self.tech_profile
  1246→            fw_set = set(tp.frameworks)
  1247→            lang = tp.language
  1248→
  1249→            # --- SQL injection: route by detected language/backend ----------
  1250→            if category == "sqli":
  1251→                if any(fw in fw_set for fw in ("Django", "Flask", "Python", "FastAPI")):
  1252→                    if "postgres" in cat_payloads:
  1253→                        _add(cat_payloads["postgres"])
  1254→                if lang == "PHP" or tp.cms in ("WordPress", "Drupal", "Joomla"):
  1255→                    if "mysql" in cat_payloads:
  1256→                        _add(cat_payloads["mysql"])
  1257→                if lang == "Java" or any(fw in fw_set for fw in ("Spring Boot", "Tomcat", "Jetty")):
  1258→                    if "oracle" in cat_payloads:
  1259→                        _add(cat_payloads["oracle"])
  1260→                    if "mssql" in cat_payloads:
  1261→                        _add(cat_payloads["mssql"])
  1262→                if lang == "ASP.NET" or "ASP.NET" in fw_set:
  1263→                    if "mssql" in cat_payloads:
  1264→                        _add(cat_payloads["mssql"])
  1265→                if lang == "Ruby" or "Rails" in fw_set:
  1266→                    if "sqlite" in cat_payloads:
  1267→                        _add(cat_payloads["sqlite"])
  1268→                    if "postgres" in cat_payloads:
  1269→                        _add(cat_payloads["postgres"])
  1270→
  1271→            # --- XSS: route by detected frontend framework -----------------
  1272→            elif category == "xss":
  1273→                if "Angular" in fw_set:
  1274→                    _add(cat_payloads.get("angular", []))
  1275→                if "React" in fw_set:
  1276→                    _add(cat_payloads.get("react", []))
  1277→                if "Vue" in fw_set or "Nuxt" in fw_set:
  1278→                    _add(cat_payloads.get("vue", []))
  1279→                _add(cat_payloads.get("html", []))
  1280→
  1281→            # --- SSTI: route by detected backend template engine ------------
  1282→            elif category == "ssti":
  1283→                if any(fw in fw_set for fw in ("Django", "Python", "Flask", "FastAPI")):
  1284→                    _add(cat_payloads.get("jinja2", []))
  1285→                    _add(cat_payloads.get("django", []))
  1286→                if "Ruby" in fw_set or "Rails" in fw_set:
  1287→                    _add(cat_payloads.get("mako", []))
  1288→                if "PHP" in lang or tp.cms == "Drupal":
  1289→                    _add(cat_payloads.get("twig", []))
  1290→
  1291→            # --- SSRF: prefer cloud provider based on headers --------------
  1292→            elif category == "ssrf":
  1293→                if tp.server and "amazon" in tp.server.lower():
  1294→                    _add(cat_payloads.get("aws_metadata", []))
  1295→                elif tp.server and "google" in tp.server.lower():
  1296→                    _add(cat_payloads.get("gcp_metadata", []))
  1297→                elif "azure" in (tp.server or "").lower():
  1298→                    _add(cat_payloads.get("azure_metadata", []))
  1299→                else:
  1300→                    for sub in ("aws_metadata", "gcp_metadata", "azure_metadata"):
  1301→                        _add(cat_payloads.get(sub, []))
  1302→
  1303→            # --- Path traversal: OS-specific based on server ---------------
  1304→            elif category == "path_traversal":
  1305→                if tp.server and "iis" in tp.server.lower():
  1306→                    _add(cat_payloads.get("windows", []))
  1307→                else:
  1308→                    _add(cat_payloads.get("unix", []))
  1309→
  1310→            # --- Command injection: OS-specific based on server -------------
  1311→            elif category == "command_injection":
  1312→                if tp.server and "iis" in tp.server.lower():
  1313→                    _add(cat_payloads.get("windows", []))
  1314→                else:
  1315→                    _add(cat_payloads.get("unix", []))
  1316→
  1317→        # -- Always add generic payloads as fallback -------------------------
  1318→        if "generic" in cat_payloads:
  1319→            _add(cat_payloads["generic"])
  1320→
  1321→        # -- Add any remaining sub-categories not yet covered ---------------
  1322→        for sub_key, sub_payloads in cat_payloads.items():
  1323→            if sub_key != "generic" and sub_key not in ("mysql", "postgres", "mssql", "sqlite", "oracle",
  1324→                                                        "html", "angular", "react", "vue",
  1325→                                                        "jinja2", "django", "mako", "twig",
  1326→                                                        "aws_metadata", "gcp_metadata", "azure_metadata",
  1327→                                                        "unix", "windows"):
  1328→                if not self.tech_profile:
  1329→                    _add(sub_payloads)
  1330→
  1331→        return selected
  1332→
  1333→    def fuzz_parameter(
  1334→        self,
  1335→        url: str,
  1336→        param: str,
  1337→        payloads: Optional[list[str]] = None,
  1338→        category: str = "generic",
  1339→        method: str = "GET",
  1340→    ) -> list[FuzzResult]:
  1341→        """Fuzz a single URL parameter with selected payloads.
  1342→
  1343→        For each payload:
  1344→          1. Substitute the payload into the parameter value.
  1345→          2. Send the request.
  1346→          3. Analyze the response for anomalies.
  1347→          4. Return a list of FuzzResult objects.
  1348→        """
  1349→        if payloads is None:
  1350→            payloads = self.select_payloads(category)
  1351→
  1352→        if not payloads:
  1353→            return []
  1354→
  1355→        # Fetch the original (clean) response for comparison
  1356→        original_status, original_body, _, _ = self._make_request(url)
  1357→
  1358→        results: list[FuzzResult] = []
  1359→
  1360→        for payload in payloads:
  1361→            params = {param: payload}
  1362→
  1363→            status, body, elapsed, resp_headers = self._make_request(url, params=params, method=method)
  1364→
  1365→            result = self._analyze_response(
  1366→                original_body=original_body,
  1367→                fuzzed_body=body,
  1368→                status_code=status,
  1369→                response_time=elapsed,
  1370→            )
  1371→            result.payload = payload
  1372→            result.category = category
  1373→            result.url = url
  1374→            result.param = param
  1375→            result.status_code = status
  1376→
  1377→            results.append(result)
  1378→
  1379→        self.results.extend(results)
  1380→        return results
  1381→
  1382→    def fuzz_headers(
  1383→        self,
  1384→        url: str,
  1385→        payloads: Optional[list[str]] = None,
  1386→        category: str = "sqli",
  1387→        header_name: str = "X-Custom-Header",
  1388→    ) -> list[FuzzResult]:
  1389→        """Fuzz HTTP headers with payloads."""
  1390→        if payloads is None:
  1391→            payloads = self.select_payloads(category)
  1392→
  1393→        if not payloads:
  1394→            return []
  1395→
  1396→        original_status, original_body, _, _ = self._make_request(url)
  1397→        results: list[FuzzResult] = []
  1398→
  1399→        for payload in payloads:
  1400→            headers = {header_name: payload}
  1401→            status, body, elapsed, _ = self._make_request(url, headers=headers)
  1402→
  1403→            result = self._analyze_response(
  1404→                original_body=original_body,
  1405→                fuzzed_body=body,
  1406→                status_code=status,
  1407→                response_time=elapsed,
  1408→            )
  1409→            result.payload = payload
  1410→            result.category = category
  1411→            result.url = url
  1412→            result.param = f"Header:{header_name}"
  1413→            result.status_code = status
  1414→
  1415→            results.append(result)
  1416→
  1417→        self.results.extend(results)
  1418→        return results

  1420→    def fuzz_all_parameters(
  1421→        self,
  1422→        url: str,
  1423→        params: list[str],
  1424→        categories: Optional[list[str]] = None,
  1425→    ) -> list[FuzzResult]:
  1426→        """Fuzz multiple parameters across multiple categories."""
  1427→        if categories is None:
  1428→            categories = ["sqli", "xss", "ssti"]
  1429→
  1430→        all_results: list[FuzzResult] = []
  1431→        for param in params:
  1432→            for category in categories:
  1433→                payloads = self.select_payloads(category)
  1434→                results = self.fuzz_parameter(url, param, payloads, category)
  1435→                all_results.extend(results)
  1436→        return all_results

  1438→    def analyze_response(
  1439→        self,
  1440→        original_body: str,
  1441→        fuzzed_body: str,
  1442→        status_code: int,
  1443→        response_time: float,
  1444→    ) -> FuzzResult:
  1445→        """Analyze a fuzzed response and classify the result.
  1446→
  1447→        Detection methods:
  1448→          - Reflected: the payload itself appears in the response body.
  1449→          - Error-based: SQL errors, stack traces, or framework errors in the body.
  1450→          - Time-based: response took significantly longer than baseline.
  1451→          - Blind: status code change (e.g., 500 instead of 200) without obvious error.
  1452→        """
  1453→        result = FuzzResult()
  1454→        result.status_code = status_code
  1455→        result.response_time = response_time
  1456→        result.matched_pattern = ""
  1457→
  1458→        # --- Check for reflected payload -----------------------------------
  1459→        if fuzzed_body and original_body:
  1460→            # Find the payload in the response (may be HTML-encoded)
  1461→            payload_escaped = re.escape(result.payload)
  1462→            if result.payload and re.search(payload_escaped, fuzzed_body):
  1463→                result.classification = "reflected"
  1464→                result.matched_pattern = "Payload reflected in response body"
  1465→
  1466→            # Also check for partial reflection or HTML entity encoding
  1467→            if result.payload and not result.classification:
  1468→                html_encoded = (
  1469→                    result.payload.replace("<", "&lt;")
  1470→                    .replace(">", "&gt;")
  1471→                    .replace('"', "&quot;")
  1472→                    .replace("'", "&#39;")
  1473→                )
  1474→                if html_encoded in fuzzed_body:
  1475→                    result.classification = "reflected"
  1476→                    result.matched_pattern = "Payload HTML-encoded in response"
  1477→
  1478→        # --- Check for SQL errors in the response --------------------------
  1479→        if fuzzed_body:
  1480→            for pattern in _SQL_ERROR_PATTERNS:
  1481→                match = pattern.search(fuzzed_body)
  1482→                if match:
  1483→                    result.classification = "error_based"
  1484→                    result.matched_pattern = match.group(0)
  1485→                    break
  1486→
  1487→        # --- Check for stack traces ----------------------------------------
  1488→        if fuzzed_body and result.classification != "error_based":
  1489→            for pattern in _STACK_TRACE_PATTERNS:
  1490→                match = pattern.search(fuzzed_body)
  1491→                if match:
  1492→                    result.classification = "error_based"
  1493→                    result.matched_pattern = match.group(0)
  1494→                    break
  1495→
  1496→        # --- Check for command injection output ---------------------------
  1497→        if fuzzed_body and result.classification != "error_based":
  1498→            match = _CMD_INJECTION_PATTERN.search(fuzzed_body)
  1499→            if match:
  1500→                result.classification = "error_based"
  1501→                result.matched_pattern = match.group(0)
  1502→
  1503→        # --- Check for path traversal output --------------------------------
  1504→        if fuzzed_body and result.classification != "error_based":
  1505→            match = _PATH_TRAVERSAL_PATTERN.search(fuzzed_body)
  1506→            if match:
  1507→                result.classification = "error_based"
  1508→                result.matched_pattern = match.group(0)
  1509→
  1510→        # --- Check for XSS indicators -------------------------------------
  1511→        if fuzzed_body and result.classification != "reflected":
  1512→            match = _XSS_PATTERN.search(fuzzed_body)
  1513→            if match:
  1514→                result.classification = "reflected"
  1515→                result.matched_pattern = match.group(0)
  1516→
  1517→        # --- Check for SSTI indicators -------------------------------------
  1518→        if fuzzed_body and result.classification not in ("reflected", "error_based"):
  1519→            match = _SSTI_PATTERN.search(fuzzed_body)
  1520→            if match:
  1521→                result.classification = "reflected"
  1522→                result.matched_pattern = match.group(0)
  1523→
  1524→        # --- Time-based detection ------------------------------------------
  1525→        if self._base_time > 0 and response_time > self._base_time * 3:
  1526→            if response_time > 4.0:
  1527→                if result.classification in ("none", ""):
  1528→                    result.classification = "time_based"
  1529→                    result.matched_pattern = f"Response time {response_time:.2f}s (baseline {self._base_time:.2f}s)"
  1530→                elif result.classification == "reflected":
  1531→                    result.classification = "time_based"
  1532→                    result.matched_pattern = f"Reflected + slow response ({response_time:.2f}s)"
  1533→
  1534→        # --- Status code change detection (blind) ---------------------------
  1535→        if (
  1536→            self._baseline_status > 0
  1537→            and status_code != self._baseline_status
  1538→            and status_code >= 400
  1539→            and result.classification in ("none", "")
  1540→        ):
  1541→            result.classification = "blind"
  1542→            result.matched_pattern = (
  1543→                f"Status changed from {self._baseline_status} to {status_code}"
  1544→            )
  1545→
  1546→        # --- Determine vulnerability ---------------------------------------
  1547→        if result.classification in ("reflected", "error_based", "time_based"):
  1548→            result.is_vulnerable = True
  1549→        elif result.classification == "blind":
  1550→            result.is_vulnerable = True
  1551→
  1552→        return result

  1583→def quick_scan(
  1584→    target: str,
  1585→    params: list[str],
  1586→    categories: Optional[list[str]] = None,
  1587→    timeout: float = 10.0,
  1588→) -> tuple[TechProfile, list[FuzzResult]]:
  1589→    """Convenience function: detect tech, calibrate, and fuzz all params.
  1590→
  1591→    Returns (TechProfile, list_of_results).
  1592→    """
  1593→    if categories is None:
  1594→        categories = ["sqli", "xss", "ssti", "command_injection", "path_traversal"]
  1595→
  1596→    # Quick tech detection
  1597→    status, body, _, headers = 0, "", 0.0, {}
  1598→    try:
  1599→        parsed = urllib.parse.urlparse(target)
  1600→        base_url = f"{parsed.scheme}://{parsed.netloc}{parsed.path}"
  1601→        ctx = None
  1602→        import ssl
  1603→        try:
  1604→            ctx = ssl._create_unverified_context()
  1605→            handler = urllib.request.HTTPSHandler(context=ctx)
  1606→            opener = urllib.request.build_opener(handler)
  1607→        except Exception:
  1608→            opener = urllib.request.build_opener()
  1609→        req = urllib.request.Request(
  1610→            base_url,
  1611→            headers={"User-Agent": "ReconPro/7.5 (Security Scanner)"},
  1612→        )
  1613→        resp = opener.open(req, timeout=timeout)
  1614→        body = resp.read().decode("utf-8", errors="replace")
  1615→        headers = dict(resp.headers)
  1616→        status = resp.status
  1617→    except Exception:
  1618→        pass
  1619→
  1620→    detector = TechDetector()
  1621→    tech_profile = detector.analyze(target, headers, body)
  1622→
  1623→    session = FuzzSession(target, tech_profile=tech_profile, timeout=timeout)
  1624→    session._baseline_status = status
  1625→    results = session.fuzz_all_parameters(target, params, categories)
  1626→    return tech_profile, results
