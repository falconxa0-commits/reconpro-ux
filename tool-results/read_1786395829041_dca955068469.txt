   603→# PatchGenerator  --  single-finding defense generator
   604→# ---------------------------------------------------------------------------
   605→
   606→
   607→class PatchGenerator:
   608→    """Generate deployable security fixes from a single vulnerability finding."""
   609→
   610→    def __init__(self) -> None:
   611→        self._rule_counter = 900100
   612→
   613→    def generate_waf_rules(self, finding: dict) -> list[str]:
   614→        """Return a list of WAF rules for *finding*.
   615→
   616→        Formats: ModSecurity SecRule, Nginx location block, Cloudflare WAF expression,
   617→        AWS WAF JSON rule (prefixed with ``AWS_WAF_JSON:``).
   618→        """
   619→        category = finding.get("category", "").lower()
   620→        severity = finding.get("severity", "high").upper()
   621→        rules: list[str] = []
   622→
   623→        # 1. Category-mapped rules from CATEGORY_PATCH_MAP
   624→        cat_entry = CATEGORY_PATCH_MAP.get(category, {})
   625→        for tmpl in cat_entry.get("waf_rules", []):
   626→            rules.append(tmpl)
   627→
   628→        # 2. Generate AWS WAF JSON rule
   629→        if category in ("sqli", "xss", "command_injection", "path_traversal", "ssrf"):
   630→            rules.append(self._aws_waf_rule(category, severity))
   631→
   632→        # 3. Generate category-specific Nginx rules not already in the map
   633→        if category == "security_headers":
   634→            rules.append(self._nginx_security_headers())
   635→        elif category == "cors":
   636→            rules.append(self._nginx_cors_config(finding))
   637→        elif category == "csrf":
   638→            rules.append(self._nginx_csrf_check())
   639→
   640→        return rules
   641→
   642→    def _aws_waf_rule(self, category: str, severity: str) -> str:
   643→        """Generate an AWS WAF JSON rule statement."""
   644→        priority_map = {"CRITICAL": 0, "HIGH": 1, "MEDIUM": 2, "LOW": 3}
   645→        priority = priority_map.get(severity, 1)
   646→        rule_name = f"ReconPro-{category}-{uuid.uuid4().hex[:8]}"
   647→
   648→        managed_rules = {
   649→            "sqli": "AWSManagedRulesSQLiRuleSet",
   650→            "xss": "AWSManagedRulesCommonRuleSet",
   651→        }
   652→
   653→        if category in managed_rules:
   654→            rule: dict[str, Any] = {
   655→                "Name": rule_name,
   656→                "Priority": priority,
   657→                "Statement": {
   658→                    "ManagedRuleGroupStatement": {
   659→                        "VendorName": "AWS",
   660→                        "Name": managed_rules[category],
   661→                        "ExcludedRules": [],
   662→                    }
   663→                },
   664→                "Action": {
   665→                    "Block": {
   666→                        "CustomResponse": {
   667→                            "ResponseCode": 403,
   668→                            "ResponseBody": {
   669→                                "Text": f"Request blocked by ReconPro {category} rule."
   670→                            },
   671→                        }
   672→                    }
   673→                },
   674→                "VisibilityConfig": {
   675→                    "SampledRequestsEnabled": True,
   676→                    "CloudWatchMetricsEnabled": True,
   677→                    "MetricName": f"ReconPro{category.capitalize()}Rule",
   678→                },
   679→            }
   680→        else:
   681→            regex_map = {
   682→                "command_injection": (
   683→                    r";.*\b(cat|ls|id|whoami|bash|sh|wget|curl|nc|ping|nmap)\b"
   684→                ),
   685→                "path_traversal": r"(\.\./|%2e%2e)",
   686→                "ssrf": r"(127\.0\.0\.1|localhost|169\.254\.169\.254)",
   687→            }
   688→            pattern = regex_map.get(category, ".*")
   689→            rule = {
   690→                "Name": rule_name,
   691→                "Priority": priority,
   692→                "Statement": {
   693→                    "RegexMatchStatement": {
   694→                        "RegexString": pattern,
   695→                        "FieldToMatch": {"QueryString": {}},
   696→                        "TextTransformations": [{"Priority": 0, "Type": "NONE"}],
   697→                    }
   698→                },
   699→                "Action": {
   700→                    "Block": {
   701→                        "CustomResponse": {
   702→                            "ResponseCode": 403,
   703→                            "ResponseBody": {
   704→                                "Text": f"Request blocked by ReconPro {category} rule."
   705→                            },
   706→                        }
   707→                    }
   708→                },
   709→                "VisibilityConfig": {
   710→                    "SampledRequestsEnabled": True,
   711→                    "CloudWatchMetricsEnabled": True,
   712→                    "MetricName": f"ReconPro{category.capitalize()}Rule",
   713→                },
   714→            }
   715→
   716→        return "AWS_WAF_JSON:" + json.dumps(rule, indent=2)

   753→    def _nginx_csrf_check(self) -> str:

   771→    # -- Code patch generation ---------------------------------------------
   772→
   773→    def generate_code_patch(self, finding: dict) -> str:
   774→        """Generate a unified diff patch for *finding*.
   775→
   776→        Requires ``file`` (path) and ``line`` (1-based) in finding.
   777→        """
   778→        filepath = finding.get("file", "unknown.py")
   779→        line_no = finding.get("line", 1)
   780→        original = finding.get("original_line", "")
   781→
   782→        fix = self._resolve_code_fix(finding)
   783→        fix_lines = fix.split("\n")
   784→
   785→        ctx_before = 3
   786→        start = max(1, line_no - ctx_before)
   787→        end = line_no + ctx_before
   788→        old_count = end - start + 1
   789→        new_count = old_count + len(fix_lines) - 1
   790→
   791→        diff_lines: list[str] = []
   792→        diff_lines.append(f"--- a/{filepath}")
   793→        diff_lines.append(f"+++ b/{filepath}")
   794→        diff_lines.append(f"@@ -{start},{old_count} +{start},{new_count} @@")
   795→
   796→        for i in range(start, end + 1):
   797→            if i == line_no:
   798→                if original:
   799→                    diff_lines.append(f"-{original}")
   800→                else:
   801→                    diff_lines.append(f"-# [REMOVED] vulnerable code at line {line_no}")
   802→                for fl in fix_lines:
   803→                    diff_lines.append(f"+{fl}")
   804→            else:
   805→                diff_lines.append(f" # (context line {i})")
   806→
   807→        return "\n".join(diff_lines) + "\n"

   809→    def _resolve_code_fix(self, finding: dict) -> str:
   810→        """Return the replacement code lines for a finding."""
   811→        category = finding.get("category", "").lower()
   812→        original = finding.get("original_line", "")
   813→
   814→        # Pattern-based detection on the original line
   815→        if "eval(" in original:
   816→            return (
   817→                "import ast\n"
   818→                "result = ast.literal_eval(sanitized_input)  # safe alternative to eval()"
   819→            )
   820→        if "exec(" in original:
   821→            return (
   822→                "# exec() removed \u2014 use a task queue, plugin system, or\n"
   823→                "# sandboxed interpreter for dynamic code execution"
   824→            )
   825→        if "shell=True" in original:
   826→            return (
   827→                "subprocess.run(\n"
   828→                '    ["command", "arg1"],\n'
   829→                "    capture_output=True,\n"
   830→                "    text=True,\n"
   831→                "    shell=False,\n"
   832→                ")"
   833→            )
   834→        if any(kw in original for kw in ("select ", "insert ", "update ", "delete ")) and ("f\"" in original or "f'" in original):
   835→            return (
   836→                'cursor.execute(\n'
   837→                '    "SELECT ... WHERE col = %s",\n'
   838→                '    (user_input,),\n'
   839→                ")  # parameterized query"
   840→            )
   841→        if category == "hardcoded_secret" or any(
   842→            k in original.lower() for k in ("secret", "password", "api_key", "token")
   843→        ):
   844→            var_match = re.match(r'(\w+)\s*=\s*["\']', original)
   845→            var_name = var_match.group(1) if var_match else "CONFIG_VALUE"
   846→            env_key = var_name.upper()
   847→            return (
   848→                "import os\n"
   849→                f'{var_name} = os.environ.get("{env_key}")\n'
   850→                f'if not {var_name}:\n'
   851→                f'    raise RuntimeError("{env_key} environment variable is not set")'
   852→            )
   853→        if "DEBUG = True" in original or "DEBUG=True" in original:
   854→            return "DEBUG = False"
   855→        if "csrf_exempt" in original:
   856→            return "# @csrf_exempt removed \u2014 CSRF protection required for this endpoint"
   857→
   858→        # Fall back to category template
   859→        cat_entry = CATEGORY_PATCH_MAP.get(category, {})
   860→        code_fix = cat_entry.get("code_fix", "")
   861→        if code_fix:
   862→            return code_fix.strip()
   863→
   864→        return "# Apply manual fix: review the vulnerability description and remediation"

  1372→def generate_defense(finding: dict) -> dict[str, Any]:
  1373→    """Generate all defenses for a single vulnerability finding.

  1378→    result: dict[str, Any] = {

  1380→        "code_patch": "",
  1381→        "infra_fix": "",
  1382→        "category": finding.get("category", "unknown"),
  1383→        "severity": finding.get("severity", "unknown"),
  1384→    }
  1385→    if finding.get("file"):
  1386→        result["code_patch"] = gen.generate_code_patch(finding)
  1387→    result["infra_fix"] = gen.generate_infra_fix(finding)
