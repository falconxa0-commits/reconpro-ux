    36→    36→@dataclass
    37→    37→class BenchResult:
    38→    38→    """Statistical summary of a benchmark run."""
    39→    39→    name: str
    40→    40→    iterations: int
    41→    41→    total_ops: int
    42→    42→    min_us: float
    43→    43→    max_us: float
    44→    44→    mean_us: float
    45→    45→    median_us: float
    46→    46→    ops_per_sec: float
    47→    47→
    48→    48→    def row(self) -> str:
    49→    49→        return (
    50→    50→            f"│ {self.name:<40s} │ {self.iterations:>3d} × {self.total_ops:>6d} │"
    51→    51→            f" {self.min_us:>8.1f} │ {self.max_us:>8.1f} │"
    52→    52→            f" {self.mean_us:>8.1f} │ {self.median_us:>8.1f} │"
    53→    53→            f" {self.ops_per_sec:>12.0f} │"
    54→    54→        )
    55→    55→
    56→    56→
    57→    57→def run_benchmark(
    58→    58→    name: str,
    59→    59→    fn,
    60→    60→    ops_per_iter: int = 1,
    61→    61→    iterations: int = 5,
    62→    62→    warmup: int = 1,
    63→    63→) -> BenchResult:
    64→    64→    """Run *fn* for *iterations*, return min/max/mean/median per-op timing.
    65→    65→
    66→    66→    Parameters
    67→    67→    ----------
    68→    68→    name : str
    69→    69→        Human-readable benchmark name.
    70→    70→    fn : callable
    71→    71→        Function to benchmark. May accept no arguments.
    72→    72→    ops_per_iter : int
    73→    73→        Number of logical operations performed by one call to *fn*.
    74→    74→    iterations : int
    75→    75→        Number of timing iterations (after warmup).
    76→    76→    warmup : int
    77→    77→        Number of un-timed warmup iterations.
    78→    78→    """
    79→    79→    # Warmup
    80→    80→    for _ in range(warmup):
    81→    81→        fn()
    82→    82→
    83→    83→    times: List[float] = []
    84→    84→    for _ in range(iterations):
    85→    85→        t0 = time.perf_counter()
    86→    86→        fn()
    87→    87→        t1 = time.perf_counter()
    88→    88→        times.append(t1 - t0)
    89→    89→
    90→    90→    per_op_us = [t / ops_per_iter * 1_000_000 for t in times]
    91→    91→    total_ops = iterations * ops_per_iter
    92→    92→    total_time = sum(times)
    93→    93→    ops_sec = total_ops / total_time if total_time > 0 else float("inf")
    94→    94→
    95→    95→    return BenchResult(
    96→    96→        name=name,
    97→    97→        iterations=iterations,
    98→    98→        total_ops=total_ops,
    99→    99→        min_us=min(per_op_us),
   100→   100→        max_us=max(per_op_us),
   101→   101→        mean_us=statistics.mean(per_op_us),
   102→   102→        median_us=statistics.median(per_op_us),
   103→   103→        ops_per_sec=ops_sec,
   104→   104→    )
   105→   105→
   106→   106→
   107→   107→def _print_table(results: List[BenchResult], title: str) -> None:
   108→   108→    """Print a formatted benchmark results table."""
   109→   109→    sep = "├" + "─" * 42 + "┼" + "─" * 16 + "┼" + "─" * 11 + "┼" + "─" * 11 + "┼" + "─" * 11 + "┼" + "─" * 11 + "┼" + "─" * 15 + "┤"
   110→   110→    top = "┌" + "─" * 42 + "┬" + "─" * 16 + "┬" + "─" * 11 + "┬" + "─" * 11 + "┬" + "─" * 11 + "┬" + "─" * 11 + "┬" + "─" * 15 + "┐"
   111→   111→    bot = "└" + "─" * 42 + "┴" + "─" * 16 + "┴" + "─" * 11 + "┴" + "─" * 11 + "┴" + "─" * 11 + "┴" + "─" * 11 + "┴" + "─" * 15 + "┘"
   112→   112→    hdr = "│ {:<40s} │ {:>14s} │ {:>9s} │ {:>9s} │ {:>9s} │ {:>9s} │ {:>13s} │".format(
   113→   113→        title[:40], "Iterations", "Min (µs)", "Max (µs)", "Mean (µs)", "Med (µs)", "Ops/sec"
   114→   114→    )
   115→   115→
