   131→class TargetProfiler:
   132→    """Fingerprint a target's technology stack from HTTP responses."""
   133→
   134→    def __init__(self, timeout: int = 8, verify_tls: bool = True) -> None:
   135→        self._timeout = timeout
   136→        self._verify_tls = verify_tls
   137→
   138→    # ── public API ────────────────────────────────────────────────────────
   139→
   140→    def profile(self, target: str, base_url: str) -> TechProfile:
   141→        """Run 3-5 rapid HTTP probes and return a TechProfile."""
   142→        profile = TechProfile()
   143→        base_url = base_url.rstrip("/")
   144→
   145→        # 1. Gather responses from multiple methods
   146→        responses: Dict[str, Dict[str, Any]] = {}
   147→        for method in ("GET", "HEAD", "OPTIONS"):
   148→            resp = http_probe(
   149→                base_url + "/",
   150→                method=method,
   151→                timeout=self._timeout,
   152→                verify_tls=self._verify_tls,
   153→            )
   154→            responses[method] = resp
   155→
   156→        # 2. Probe /api/ and /graphql
   157→        api_resp = http_probe(
   158→            base_url + "/api/",
   159→            timeout=self._timeout,
   160→            verify_tls=self._verify_tls,
   161→        )
   162→        responses["GET_API"] = api_resp
   163→
   164→        gql_resp = http_probe(
   165→            base_url + "/graphql",
   166→            method="POST",
   167→            body=b'{"query":"{__typename}"}',
   168→            headers={"Content-Type": "application/json"},
   169→            timeout=self._timeout,
   170→            verify_tls=self._verify_tls,
   171→        )
   172→        responses["POST_GRAPHQL"] = gql_resp
   173→
   174→        # 3. Merge all headers from non-fatal responses
   175→        merged_headers: Dict[str, str] = {}
   176→        _recoverable = {200, 301, 302, 304, 400, 401, 403, 405}
   177→        for resp in responses.values():
   178→            status = resp.get("status", 0)
   179→            if resp.get("ok") or status in _recoverable:
   180→                for k, v in resp.get("headers", {}).items():
   181→                    merged_headers[k.lower()] = v
   182→        profile.detected_headers = merged_headers
   183→
   184→        # 4. Get the main body (prefer GET /)
   185→        main_body = responses["GET"].get("body", "")
   186→
   187→        # 5. Detect WAF
   188→        profile.waf = self._detect_waf(merged_headers, main_body)
   189→
   190→        # 6. Detect server from headers
   191→        profile.server = merged_headers.get("server")
   192→
   193→        # 7. Detect frameworks from headers and body
   194→        frameworks, language = self._detect_frameworks(merged_headers, main_body)
   195→        profile.frameworks = frameworks
   196→        profile.language = language
   197→
   198→        # 8. Detect CMS from meta generator
   199→        cms, cms_lang = self._detect_cms(main_body)
   200→        if cms:
   201→            profile.cms = cms
   202→            if cms_lang and not profile.language:
   203→                profile.language = cms_lang
   204→            if cms not in profile.frameworks:
   205→                profile.frameworks.append(cms)
   206→
   207→        # 9. Detect GraphQL
   208→        gql_status = responses["POST_GRAPHQL"].get("status", 0)
   209→        gql_body = responses["POST_GRAPHQL"].get("body", "")
   210→        if "graphql" in gql_body.lower() or "__typename" in gql_body:
   211→            if "data" in gql_body and "__typename" in gql_body:
   212→                profile.has_graphql = True
   213→        if any("graphql" in v.lower() for v in merged_headers.values()):
   214→            profile.has_graphql = True
   215→
   216→        # 10. Detect REST API
   217→        api_status = responses["GET_API"].get("status", 0)
   218→        api_body = responses["GET_API"].get("body", "")
   219→        if api_status == 200:
   220→            try:
   221→                json.loads(api_body)
   222→                profile.has_rest_api = True
   223→            except (json.JSONDecodeError, ValueError):
   224→                pass
   225→        if any("/api/" in v.lower() for v in merged_headers.values()):
   226→            profile.has_rest_api = True
   227→
   228→        # 11. Detect app type
   229→        profile.app_type = self._detect_app_type(main_body, merged_headers, profile)
   230→
   231→        # 12. Build detected technologies list
   232→        profile.detected_technologies = list(profile.frameworks)
   233→        if profile.server:
   234→            profile.detected_technologies.insert(0, profile.server)
   235→        if profile.waf:
   236→            profile.detected_technologies.insert(0, profile.waf)
   237→        if profile.has_graphql:
   238→            profile.detected_technologies.append("GraphQL")
   239→        if profile.has_rest_api:
   240→            profile.detected_technologies.append("REST API")
   241→
   242→        # 13. Compute confidence (0.0 - 1.0)
   243→        signals = (
   244→            len(profile.frameworks)
   245→            + (1 if profile.server else 0)
   246→            + (1 if profile.waf else 0)
   247→        )
   248→        profile.confidence = min(signals / 6.0, 1.0)
   249→
   250→        return profile
   251→
   252→    # ── detection helpers ────────────────────────────────────────────────
   253→
   254→    @staticmethod
   255→    def _detect_waf(headers: Dict[str, str], body: str) -> Optional[str]:
   256→        header_blob = "\n".join(f"{k}: {v}" for k, v in headers.items())
   257→        combined = (header_blob + "\n" + body[:4096]).lower()
   258→        for pattern, name in _WAF_SIGNATURES:
   259→            if re.search(pattern, combined, re.IGNORECASE):
   260→                return name
   261→        return None
   262→
   263→    @staticmethod
   264→    def _detect_frameworks(
   265→        headers: Dict[str, str], body: str
   266→    ) -> tuple[List[str], Optional[str]]:
   267→        header_blob = "\n".join(f"{k}: {v}" for k, v in headers.items())
   268→        combined = header_blob + "\n" + body[:16384]
   269→        frameworks: List[str] = []
   270→        language: Optional[str] = None
   271→        for pattern, name, lang in _FRAMEWORK_SIGNATURES:
   272→            if re.search(pattern, combined, re.IGNORECASE):
   273→                if name not in frameworks:
   274→                    frameworks.append(name)
   275→                if lang and language is None:
   276→                    language = lang
   277→        return frameworks, language
   278→
   279→    @staticmethod
   280→    def _detect_cms(body: str) -> tuple[Optional[str], Optional[str]]:
   281→        meta_match = re.search(
   282→            r'<meta\s+name=["\x27]generator["\x27]\s+content=["\x27]([^"\x27>]+)',
   283→            body,
   284→            re.IGNORECASE,
   285→        )
   286→        if not meta_match:
   287→            return None, None
   288→        content = meta_match.group(1).strip().lower()
   289→        for keyword, (name, lang) in _META_GENERATOR_MAP.items():
   290→            if keyword in content:
   291→                return name, lang
   292→        return meta_match.group(1).strip(), None
   293→
   294→    @staticmethod
   295→    def _detect_app_type(
   296→        body: str, headers: Dict[str, str], profile: TechProfile
   297→    ) -> str:
   298→        body_lower = body.lower()
   299→        ct = headers.get("content-type", "")
   300→
   301→        # API: JSON responses
   302→        if "application/json" in ct:
   303→            return "api"
   304→        try:
   305→            json.loads(body[:4096])
   306→            return "api"
   307→        except (json.JSONDecodeError, ValueError):
   308→            pass
   309→        if profile.has_rest_api and len(body.strip()) < 2000:
   310→            return "api"
   311→
   312→        # Microservices: no HTML, small response
   313→        if "html" not in ct and len(body.strip()) < 500:
   314→            return "microservices"
   315→
   316→        # SPA detection signals
   317→        spa_signals = [
   318→            "__NEXT_DATA__" in body,
   319→            "__vue__" in body or "v-cloak" in body,
   320→            re.search(r"ng-version", body) is not None,
   321→            re.search(r"react\.|_reactRootContainer|__REACT", body) is not None,
   322→            bool(re.search(r'chunk\.js|bundle\.js|app\.[a-f0-9]+\.js', body)),
   323→        ]
   324→        if any(spa_signals):
   325→            if "__NEXT_DATA__" in body and "<!doctype html" in body_lower:
   326→                return "ssr"
   327→            return "spa"
   328→
   329→        # SSR: server-rendered content with template markers
   330→        ssr_markers = [
   331→            "csrfmiddlewaretoken" in body,
   332→            "csrf-token" in body,
   333→            bool(re.search(r'\{\{.*?\}\}', body)),
   334→            bool(re.search(r'\$\{.*?\}', body)),
   335→        ]
   336→        if any(ssr_markers) and "<html" in body_lower:
   337→            return "ssr"
   338→
   339→        # Static: plain HTML, minimal or no JavaScript
   340→        script_tags = re.findall(r"<script", body, re.IGNORECASE)
   341→        if len(script_tags) <= 1 and "<html" in body_lower and not profile.has_rest_api:
   342→            return "static"
   343→
   344→        return "unknown"

   346→    # ── security posture estimation ───────────────────────────────────────
   347→
   348→    def estimate_posture(self, profile: TechProfile) -> Dict[str, Any]:
   349→        """Evaluate security posture based on detected headers."""
   350→        headers_lower = {k.lower(): v for k, v in profile.detected_headers.items()}
   351→        score = 0
   352→        present_headers: List[str] = []
   353→        missing_headers: List[str] = []
   354→
   355→        for hdr_name, points in _SECURITY_HEADERS.items():
   356→            if hdr_name in headers_lower:
   357→                score += points
   358→                present_headers.append(hdr_name)
   359→            else:
   360→                missing_headers.append(hdr_name)
   361→
   362→        if profile.waf:
   363→            score += 10
   364→        score = min(score, 100)
   365→
   366→        strengths: List[str] = []
   367→        weaknesses: List[str] = []
   368→
   369→        if "strict-transport-security" in present_headers:
   370→            strengths.append("HSTS enforced")
   371→        else:
   372→            weaknesses.append("Missing HSTS - vulnerable to protocol downgrade")
   373→
   374→        if "content-security-policy" in present_headers:
   375→            strengths.append("CSP header present")
   376→        else:
   377→            weaknesses.append("Missing Content-Security-Policy - XSS risk elevated")
   378→
   379→        if "x-frame-options" in present_headers:
   380→            strengths.append("X-Frame-Options set")
   381→        else:
   382→            weaknesses.append("Missing X-Frame-Options - clickjacking possible")
   383→
   384→        if profile.waf:
   385→            strengths.append(f"WAF detected ({profile.waf})")
   386→        else:
   387→            weaknesses.append("No WAF detected")
   388→
   389→        if "x-content-type-options" in present_headers:
   390→            strengths.append("MIME sniffing mitigated")
   391→        else:
   392→            weaknesses.append("Missing X-Content-Type-Options")
   393→
   394→        if "referrer-policy" in present_headers:
   395→            strengths.append("Referrer-Policy set")
   396→        else:
   397→            weaknesses.append("Missing Referrer-Policy")
   398→
   399→        return {
   400→            "security_score": score,
   401→            "missing_headers": missing_headers,
   402→            "strengths": strengths,
   403→            "weaknesses": weaknesses,
   404→        }

   543→                    "estimated_time": len(modules) * 4,
   544→                }
   545→
   546→        # Fall back to app-type rules
   547→        type_entry = self._TYPE_RULES.get(profile.app_type)
   548→        if type_entry:
   549→            modules, reasoning = type_entry
   550→            return {
   551→                "recommended_modules": modules,
   552→                "reasoning": reasoning,
   553→                "estimated_time": len(modules) * 4,
   554→            }
   555→
   556→        # Default plan for unknown stacks
   557→        modules, reasoning = self._DEFAULT_PLAN
   558→        return {
   559→            "recommended_modules": modules,
   560→            "reasoning": reasoning,
   561→            "estimated_time": len(modules) * 4,
   562→        }

   584→# ── TimingAnalyzer ───────────────────────────────────────────────────────
   585→
   586→
   587→class TimingAnalyzer:
   588→    """Detect timing-based vulnerabilities and side channels.
   589→
   590→    Measures response times across multiple requests to detect:
   591→    - Time-based SQL injection (sleep/BENCHMARK responses)
   592→    - Boolean-based blind SQL injection (timing differences)
   593→    - Time-based blind XSS
   594→    - Error-based information disclosure via timing
   595→    - Rate limiting and brute-force protection
   596→    """
   597→
   598→    def __init__(self, base_url: str, timeout: float = 15.0, samples: int = 5):
   599→        self.base_url = base_url.rstrip("/")
   600→        self.timeout = timeout
   601→        self.samples = samples
   602→        self.findings: List[Finding] = []
   603→        self._baseline_time: float = 0.0
   604→
   605→    def analyze(self) -> Dict[str, Any]:
   606→        """Run timing analysis. Returns results dict."""
   607→        results = {
   608→            "baseline_ms": 0.0,
   609→            "tests_run": 0,
   610→            "vulnerabilities": [],
   611→        }
   612→
   613→        # Establish baseline
   614→        self._baseline_time = self._measure_baseline()
   615→        results["baseline_ms"] = round(self._baseline_time * 1000, 2)
   616→
   617→        # Run timing tests
   618→        self._test_sqli_timing()
   619→        self._test_boolean_blind()
   620→        self._test_error_timing()
   621→        self._test_rate_limiting()
   622→        self._test_auth_timing()
   623→
   624→        results["tests_run"] = len(self.findings)
   625→        results["vulnerabilities"] = [f.to_dict() if hasattr(f, 'to_dict') else f for f in self.findings]
   626→        return results
   627→
   628→    def _measure_baseline(self) -> float:
   629→        """Get average response time for baseline URL."""
   630→        import urllib.request
   631→        times = []
   632→        for _ in range(self.samples):
   633→            try:
   634→                start = time.time()
   635→                req = urllib.request.Request(self.base_url, headers={"User-Agent": "ReconPro/9"})
   636→                with urllib.request.urlopen(req, timeout=self.timeout) as resp:
   637→                    resp.read()
   638→                times.append(time.time() - start)
   639→            except Exception:
   640→                pass
   641→        return sum(times) / len(times) if times else 0.1
   642→
   643→    def _measure_endpoint(self, url: str, method: str = "GET", data: bytes = b"") -> float:
   644→        """Measure response time for a specific request."""
   645→        import urllib.request
   646→        try:
   647→            start = time.time()
   648→            req = urllib.request.Request(url, data=data if method == "POST" else None, method=method,
   649→                                          headers={"User-Agent": "ReconPro/9", "Content-Type": "application/x-www-form-urlencoded"})
   650→            with urllib.request.urlopen(req, timeout=self.timeout) as resp:
   651→                resp.read()
   652→            return time.time() - start
   653→        except Exception:
   654→            return -1.0
   655→
   656→    def _test_sqli_timing(self):
   657→        """Test for time-based SQL injection."""
   658→        sqli_payloads = [
   659→            "1; WAITFOR DELAY '0:0:5'--",
   660→            "1' OR SLEEP(5)--",
   661→            "1' AND pg_sleep(5)--",
   662→            "1' || DBMS_PIPE.RECEIVE_MESSAGE(5,5) || '",
   663→            "1; SELECT BENCHMARK(5000000,SHA1('test'))--",
   664→        ]
   665→        for param in ["id", "user", "page", "q"]:
   666→            for payload in sqli_payloads:
   667→                url = f"{self.base_url}?{param}={urllib.parse.quote(payload)}"
   668→                elapsed = self._measure_endpoint(url)
   669→                if elapsed > 0 and elapsed > self._baseline_time * 3 + 3.0:
   670→                    self.findings.append(Finding(
   671→                        title=f"Potential time-based SQL injection",
   672→                        severity="high",
   673→                        category="timing_sqli",
   674→                        module="profiler",
   675→                        description=f"Parameter '{param}' responded in {elapsed:.2f}s (baseline: {self._baseline_time:.2f}s) "
   676→                                    f"with SQL timing payload, suggesting time-based SQL injection.",
   677→                        evidence=f"URL: {url} | Response time: {elapsed:.2f}s | Baseline: {self._baseline_time:.2f}s",
   678→                        asset=f"{self.base_url}?{param}=...",
   679→                        points_deducted=8,
   680→                        remediation="Use parameterized queries / prepared statements. Never concatenate user input into SQL.",
   681→                    ))
   682→                    return  # Report once per test
   683→
   684→    def _test_boolean_blind(self):
   685→        """Test for boolean-based blind SQL injection via timing."""
   686→        for param in ["id", "user", "page"]:
   687→            true_url = f"{self.base_url}?{param}=1%20AND%201=1"
   688→            false_url = f"{self.base_url}?{param}=1%20AND%201=2"
   689→            t_true = self._measure_endpoint(true_url)
   690→            t_false = self._measure_endpoint(false_url)
   691→            if t_true > 0 and t_false > 0:
   692→                diff = abs(t_true - t_false)
   693→                if diff > self._baseline_time * 2:
   694→                    self.findings.append(Finding(
   695→                        title=f"Potential boolean-based blind SQL injection",
   696→                        severity="medium",
   697→                        category="timing_boolean_blind",
   698→                        module="profiler",
   699→                        description=f"Parameter '{param}' shows {diff:.2f}s timing difference between true/false conditions.",
   700→                        evidence=f"True: {t_true:.2f}s | False: {t_false:.2f}s | Diff: {diff:.2f}s",
   701→                        asset=f"{self.base_url}?{param}=...",
   702→                        points_deducted=6,
   703→                        remediation="Use parameterized queries. Boolean-based blind SQLi can extract data character by character.",
   704→                    ))
   705→                    return
   706→
   707→    def _test_error_timing(self):
   708→        """Test if error messages take significantly longer (information disclosure)."""
   709→        normal_url = self.base_url
   710→        error_url = f"{self.base_url}/nonexistent_path_404_trigger"
   711→        t_normal = self._measure_endpoint(normal_url)
   712→        t_error = self._measure_endpoint(error_url)
   713→        if t_normal > 0 and t_error > 0 and t_error > t_normal * 5:
   714→            self.findings.append(Finding(
   715→                title="Error responses expose timing information",
   716→                severity="low",
   717→                category="timing_error",
   718→                module="profiler",
   719→                description=f"Error responses take {t_error:.2f}s vs normal {t_normal:.2f}s — significant difference "
   720→                            f"may indicate verbose error handling or debugging mode.",
   721→                evidence=f"Normal: {t_normal:.2f}s | Error: {t_error:.2f}s | Ratio: {t_error/t_normal:.1f}x",
   722→                asset=self.base_url,
   723→                points_deducted=2,
   724→                remediation="Ensure error pages are pre-rendered with consistent response times.",
   725→            ))
   726→
   727→    def _test_rate_limiting(self):
   728→        """Test for rate limiting on authentication endpoints."""
   729→        auth_paths = ["/login", "/auth/login", "/api/auth/login", "/signin", "/api/login"]
   730→        for path in auth_paths:
   731→            url = f"{self.base_url}{path}"
   732→            times = []
   733→            for i in range(10):
   734→                t = self._measure_endpoint(url, method="POST", data=b"username=test&password=test")
   735→                if t > 0:
   736→                    times.append(t)
   737→            if len(times) >= 5:
   738→                early_avg = sum(times[:3]) / 3
   739→                late_avg = sum(times[-3:]) / 3
   740→                if late_avg < early_avg * 1.5:  # No significant slowdown
   741→                    self.findings.append(Finding(
   742→                        title=f"No rate limiting detected on {path}",
   743→                        severity="medium",
   744→                        category="rate_limiting",
   745→                        module="profiler",
   746→                        description=f"10 rapid requests to {path} showed no response time degradation. "
   747→                                    f"Authentication endpoint may be vulnerable to brute-force attacks.",
   748→                        evidence=f"First 3 avg: {early_avg:.2f}s | Last 3 avg: {late_avg:.2f}s | Total: 10 requests",
   749→                        asset=url,
   750→                        points_deducted=5,
   751→                        remediation="Implement rate limiting on authentication endpoints (e.g., fail2ban, nginx limit_req, cloudflare rate rules).",
   752→                    ))
   753→                    return
   754→
   755→    def _test_auth_timing(self):
   756→        """Test for timing-based username enumeration."""
   757→        existing_url = f"{self.base_url}/login"
   758→        times_valid = []
   759→        times_invalid = []
   760→        for _ in range(3):
   761→            t1 = self._measure_endpoint(existing_url, method="POST", data=b"username=admin&password=wrong")
   762→            t2 = self._measure_endpoint(existing_url, method="POST", data=b"username=nonexist_user_xyz&password=wrong")
   763→            if t1 > 0: times_valid.append(t1)
   764→            if t2 > 0: times_invalid.append(t2)
   765→        if times_valid and times_invalid:
   766→            avg_valid = sum(times_valid) / len(times_valid)
   767→            avg_invalid = sum(times_invalid) / len(times_invalid)
   768→            diff = abs(avg_valid - avg_invalid)
   769→            if diff > 0.1:  # More than 100ms difference
   770→                self.findings.append(Finding(
   771→                    title="Potential username enumeration via timing",
   772→                    severity="medium",
   773→                    category="timing_auth_enum",
   774→                    module="profiler",
   775→                    description=f"Login responses show {diff:.2f}s timing difference between valid/invalid usernames, "
   776→                                f"suggesting username enumeration is possible.",
   777→                    evidence=f"Valid username avg: {avg_valid:.2f}s | Invalid avg: {avg_invalid:.2f}s | Diff: {diff:.2f}s",
   778→                    asset=f"{self.base_url}/login",
   779→                    points_deducted=5,
   780→                    remediation="Ensure authentication failures return consistent response times regardless of username validity.",
   781→                ))
