  1197→    def _handle_scan(self, args: List[str]) -> None:
  1198→        if not args:
  1199→            self._chat(f"[{RED}]Usage: scan <target> [with <mod1> <mod2> ...][/]")
  1200→            return
  1201→
  1202→        # Parse "with" clause
  1203→        modules: Optional[List[str]] = None
  1204→        target_parts: List[str] = []
  1205→        found_with = False
  1206→        mod_parts: List[str] = []

  1208→            if a.lower() == "with":
  1209→                found_with = True
  1210→                continue

  1212→                mod_parts.append(a)
  1213→            else:
  1214→                target_parts.append(a)
  1215→
  1216→        target = " ".join(target_parts)
  1217→        if not target:
  1218→            self._chat(f"[{RED}]No target specified.[/]")
  1219→            return
  1220→
  1221→        if mod_parts:
  1222→            modules = [m.strip().lower() for m in mod_parts]
  1223→            self._chat(f"[{DIM_CYAN}]Scanning [cyan bold]{target}[/] with modules: {', '.join(modules)}[/]")
  1224→        else:
  1225→            self._chat(f"[{DIM_CYAN}]Scanning [cyan bold]{target}[/]...[/]")
  1226→
  1227→        self.current_target = target
  1228→        self._run_scan_worker(target, modules=modules)

  1311→        self,

  1343→                result = audit_scan(target=target, modules=modules)

  1345→                result = scan(target=target, modules=modules)

  1720→    def on_click(self, event: Click) -> None:
  1721→        """Open finding detail modal for the clicked finding.
  1722→
  1723→        Uses the vertical click position to determine which finding index
  1724→        the user clicked on, mapping it to _findings_list.
  1725→        """
  1726→        try:
  1727→            if (
  1728→                isinstance(event.widget, RichLog)
  1729→                and event.widget.id == "findings-feed"
  1730→                and self._findings_list
  1731→            ):
  1732→                # Get the line offset to determine which finding was clicked.
  1733→                # Each finding entry is 1 line in the RichLog.
  1734→                # We use the y coordinate relative to the widget to estimate the index.
  1735→                region = event.widget.region
  1736→                if region is not None:
  1737→                    # Calculate which line was clicked relative to the scroll position
  1738→                    scroll_offset = event.widget.scroll_y
  1739→                    click_y = event.y - region.y
  1740→                    # Each finding takes 1 line; estimate index from click position
  1741→                    # Account for scroll position
  1742→                    line_height = 1
  1743→                    if hasattr(event.widget, "_line_height"):
  1744→                        line_height = event.widget._line_height
  1745→                    clicked_index = int((click_y + scroll_offset * line_height) / line_height)
  1746→                    # Clamp to valid range
  1747→                    clicked_index = max(0, min(clicked_index, len(self._findings_list) - 1))
  1748→                    finding = self._findings_list[clicked_index]
  1749→                else:
  1750→                    finding = self._findings_list[-1]
  1751→                self.push_screen(FindingDetailModal(finding))
  1752→        except (NoMatches, Exception):
  1753→            pass
