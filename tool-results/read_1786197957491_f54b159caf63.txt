   287→class AIEndpointDiscovery:
   288→    """Discovers AI-specific endpoints on a target."""
   289→
   290→    def discover(self, base_url: str, timeout: int = 8,
   291→                 verify_tls: bool = True) -> List[Dict[str, Any]]:
   292→        """Probe for AI endpoints. Returns list of discovered endpoints."""
   293→        discovered = []
   294→        host = base_url.replace("https://", "").replace("http://", "").split("/")[0]
   295→
   296→        for endpoint in AI_ENDPOINTS:
   297→            url = base_url.rstrip("/") + endpoint
   298→            try:
   299→                resp = http_probe(url, timeout=timeout, verify_tls=verify_tls)
   300→                status = resp.get("status", 0)
   301→                if status in (200, 201, 401, 403, 405):
   302→                    body = resp.get("body", "")[:2048]
   303→                    discovered.append({
   304→                        "endpoint": endpoint,
   305→                        "url": url,
   306→                        "status": status,
   307→                        "response_size": len(resp.get("body", "")),
   308→                        "content_type": resp.get("headers", {}).get("content-type", ""),
   309→                        "has_ai_content": bool(self._has_ai_content(body)),
   310→                        "host": host,
   311→                    })
   312→            except Exception:
   313→                pass
   314→        return discovered
   315→
   316→    def _has_ai_content(self, body: str) -> bool:
   317→        """Check if response body contains AI-related content."""
   318→        ai_terms = [
   319→            "chat", "completion", "embedding", "model", "prompt",
   320→            "token", "temperature", "max_tokens", "top_p",
   321→            "llm", "ai", "gpt", "claude", "llama", "mistral",
   322→            "choices", "messages", "content", "generated",
   323→        ]
   324→        body_lower = body.lower()
   325→        return sum(1 for t in ai_terms if t in body_lower) >= 3

   328→class AIVendorFingerprinter:
   329→    """Detects AI vendor from response patterns."""
   330→
   331→    def fingerprint(self, base_url: str, timeout: int = 8,
   332→                    verify_tls: bool = True) -> List[Dict[str, Any]]:
   333→        """Probe target and identify AI vendors. Returns ranked vendor matches."""
   334→        host = base_url.replace("https://", "").replace("http://", "").split("/")[0]
   335→        results = []
   336→
   337→        # Probe the base URL and common AI endpoints
   338→        probe_urls = [
   339→            base_url,
   340→            base_url.rstrip("/") + "/v1/models",
   341→            base_url.rstrip("/") + "/api/generate",
   342→        ]
   343→
   344→        probe_results = []
   345→        for url in probe_urls:
   346→            try:
   347→                resp = http_probe(url, timeout=timeout, verify_tls=verify_tls)
   348→                probe_results.append(resp)
   349→            except Exception:
   350→                pass
   351→
   352→        if not probe_results:
   353→            return results
   354→
   355→        # Score each vendor
   356→        for vendor_fp in AI_VENDOR_FINGERPRINTS:
   357→            score = 0
   358→            evidence = []
   359→
   360→            for resp in probe_results:
   361→                body = resp.get("body", "")[:8192]
   362→                headers = resp.get("headers", {})
   363→
   364→                for indicator_type, indicator_key, pattern in vendor_fp["indicators"]:
   365→                    try:
   366→                        if indicator_type == "header" and indicator_key:
   367→                            header_val = headers.get(indicator_key, "")
   368→                            if re.search(pattern, header_val, re.IGNORECASE):
   369→                                score += vendor_fp["weight"] // 2
   370→                                evidence.append(f"Header {indicator_key}: {header_val[:60]}")
   371→
   372→                        if indicator_type == "body":
   373→                            if re.search(pattern, body, re.IGNORECASE):
   374→                                score += vendor_fp["weight"]
   375→                                evidence.append(f"Body match: {pattern[:40]}")
   376→                    except re.error:
   377→                        if pattern.lower() in body.lower():
   378→                            score += vendor_fp["weight"]
   379→                            evidence.append(f"Body string match: {pattern[:40]}")
   380→
   381→            if score > 0:
   382→                results.append({
   383→                    "vendor": vendor_fp["vendor"],
   384→                    "score": score,
   385→                    "evidence": evidence[:3],
   386→                    "host": host,
   387→                })
   388→
   389→        results.sort(key=lambda r: r["score"], reverse=True)
   390→        return results

   394→    """Detects cryptographic watermarks and bias patterns in AI responses."""
   395→
   396→    WATERMARK_INDICATORS = [

   398→        (r"\b(?:google|gemini|bard)\b.*(?:watermark|synthid|digital watermark)", "Google SynthID watermark", 8),

   408→        (r"(?:I (?:can|will|must|should) not)", "Restriction pattern", 2),

   412→        """Analyze text for watermark and bias indicators."""
   413→        findings = []
   414→        total_watermark_score = 0
   415→        total_bias_score = 0
   416→
   417→        for pattern, description, score in self.WATERMARK_INDICATORS:
   418→            try:
   419→                if re.search(pattern, text, re.IGNORECASE):
   420→                    findings.append({
   421→                        "type": "watermark",
   422→                        "description": description,
   423→                        "score": score,
   424→                        "pattern": pattern[:60],
   425→                    })
   426→                    total_watermark_score += score
   427→            except re.error:
   428→                pass
   429→
   430→        for pattern, description, score in self.BIAS_PATTERNS:
   431→            try:
   432→                matches = re.findall(pattern, text, re.IGNORECASE)
   433→                if matches:
   434→                    findings.append({
   435→                        "type": "bias",
   436→                        "description": description,
   437→                        "score": score * len(matches),
   438→                        "count": len(matches),
   439→                    })
   440→                    total_bias_score += score * len(matches)
   441→            except re.error:
   442→                pass
   443→
   444→        # Watermark removal difficulty assessment
   445→        removal_difficulty = "LOW"
   446→        if total_watermark_score >= 15:
   447→            removal_difficulty = "HIGH"
   448→        elif total_watermark_score >= 8:
   449→            removal_difficulty = "MEDIUM"
   450→
   451→        return {
   452→            "watermark_score": total_watermark_score,
   453→            "bias_score": total_bias_score,
   454→            "removal_difficulty": removal_difficulty,
   455→            "findings": findings,

   459→class ModelCollapseDetector:
   460→    """Analyzes for recursive generation poisoning and contamination vectors."""
   461→
   462→    COLLAPSE_INDICATORS = [
   463→        (r"(?:repetitive|repetition).*(?:pattern|text|generation)", "Repetition pattern detected", 6),
   464→        (r"(?:degraded|degrading).*(?:quality|output|performance)", "Quality degradation signal", 7),
   465→        (r"(?:model|training).*(?:collapse|degradation|corruption)", "Model collapse reference", 8),
   466→        (r"(?:synthetic|ai-generated).*(?:data|training|poison)", "Synthetic data poisoning", 7),
   467→        (r"(?:contaminated|contamination).*(?:dataset|training|model)", "Dataset contamination", 8),
   468→        (r"(?:adversarial|poison).*(?:injection|attack|pattern)", "Adversarial contamination", 7),
   469→        (r"(?:recursive|loop).*(?:generation|feedback|training)", "Recursive generation risk", 6),
   470→        (r"(?:drift|distribution).*(?:shift|degradation)", "Distribution drift indicator", 5),
   471→    ]
   472→
   473→    def detect(self, text: str) -> Dict[str, Any]:
   474→        """Analyze text for model collapse vectors."""
   475→        findings = []
   476→        collapse_score = 0
   477→
   478→        for pattern, description, score in self.COLLAPSE_INDICATORS:
   479→            try:
   480→                if re.search(pattern, text, re.IGNORECASE):
   481→                    findings.append({
   482→                        "description": description,
   483→                        "score": score,
   484→                        "pattern": pattern[:60],
   485→                    })
   486→                    collapse_score += score
   487→            except re.error:
   488→                pass
   489→
   490→        # Check for repetitive text patterns (actual collapse evidence)
   491→        repetition_score = self._detect_repetition(text)
   492→
   493→        risk_level = "LOW"
   494→        if collapse_score >= 20 or repetition_score >= 8:
   495→            risk_level = "CRITICAL"
   496→        elif collapse_score >= 12 or repetition_score >= 4:
   497→            risk_level = "HIGH"
   498→        elif collapse_score >= 6 or repetition_score >= 2:
   499→            risk_level = "MEDIUM"
   500→
   501→        return {
   502→            "collapse_score": collapse_score,
   503→            "repetition_score": repetition_score,
   504→            "risk_level": risk_level,
   505→            "findings": findings,
   506→        }
   507→
   508→    def _detect_repetition(self, text: str) -> int:
   509→        """Score text for repetitive patterns."""
   510→        score = 0
   511→        # Check for repeated sentences (longer than 10 chars)
   512→        sentences = re.split(r'[.!?]\s+', text)
   513→        seen: Dict[str, int] = {}
   514→        for s in sentences:
   515→            s_clean = s.strip().lower()
   516→            if len(s_clean) > 10:
   517→                seen[s_clean] = seen.get(s_clean, 0) + 1
   518→        duplicates = sum(c - 1 for c in seen.values() if c > 1)
   519→        score += duplicates * 3
   520→
   521→        # Check for repeated n-grams (3-grams)
   522→        words = text.lower().split()
   523→        trigrams: Dict[str, int] = {}
   524→        for i in range(len(words) - 2):
   525→            tg = " ".join(words[i:i + 3])
   526→            trigrams[tg] = trigrams.get(tg, 0) + 1
   527→        repeated_trigrams = sum(c - 1 for c in trigrams.values() if c > 2)
   528→        score += repeated_trigrams
   529→
   530→        return score

   558→                        "canary_name": name,
   559→                        "pattern_matched": pattern,
   560→                        "severity": "critical",
   561→                    })
   562→                    total_score += 10

   576→class SecretExtractor:
   577→    """Extracts secrets, API keys, and credentials from response bodies."""
   578→
   579→    def __init__(self, patterns: Optional[List[tuple]] = None) -> None:
   580→        self._patterns = patterns or SECRET_PATTERNS
   581→
   582→    def extract(self, text: str) -> List[Dict[str, Any]]:
   583→        """Extract secrets from text. Returns list of found secrets."""
   584→        findings = []
   585→        seen: set = set()
   586→
   587→        for pattern, name, severity in self._patterns:
   588→            try:
   589→                matches = re.findall(pattern, text)
   590→                for match in matches:
   591→                    # Truncate long matches for display
   592→                    display = match[:60] + "..." if len(str(match)) > 60 else str(match)
   593→                    if display not in seen:
   594→                        seen.add(display)
   595→                        findings.append({
   596→                            "type": name,
   597→                            "severity": severity,
   598→                            "match": display,
   599→                            "pattern": pattern[:40],
   600→                        })
   601→            except re.error:
   602→                pass
   603→
   604→        return findings
   605→
   606→    def extract_from_response(self, resp: Dict[str, Any]) -> List[Dict[str, Any]]:
   607→        """Extract secrets from an HTTP response dict."""
   608→        body = resp.get("body", "")
   609→        headers_str = " ".join(f"{k}: {v}" for k, v in resp.get("headers", {}).items())
   610→        combined = f"{body}\n{headers_str}"
   611→        return self.extract(combined)
