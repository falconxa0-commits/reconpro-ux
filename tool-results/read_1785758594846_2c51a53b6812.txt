   715→    def on_key(self, event: Key) -> None:
   716→        """Handle key events — boot skip + history navigation."""
   717→        # Skip boot on ANY key press during boot sequence
   718→        if not self._boot_done and not self._boot_skipped:
   719→            self._boot_skipped = True
   720→            try:
   721→                boot = self.query_one("#boot-screen", BootScreen)
   722→                if boot._timer:
   723→                    boot._timer.stop()
   724→            except NoMatches:
   725→                pass
   726→            self._finish_boot()
   727→            return
   728→
   729→        # Command history navigation
   730→        if event.key == "up" and self._history_index > 0:
   731→            try:
   732→                inp = self.query_one("#command-input", Input)
   733→                if inp.has_focus:
   734→                    self._history_index -= 1
   735→                    inp.value = self._command_history[self._history_index]
   736→            except NoMatches:
   737→                pass
   738→        elif event.key == "down":
   739→            try:
   740→                inp = self.query_one("#command-input", Input)
   741→                if inp.has_focus:
   742→                    if self._history_index < len(self._command_history) - 1:
   743→                        self._history_index += 1
   744→                        inp.value = self._command_history[self._history_index]
   745→                    else:
   746→                        self._history_index = len(self._command_history)
   747→                        inp.value = ""
   748→            except NoMatches:
   749→                pass

   810→    def watch_finding_count(self, old: int, new: int) -> None:
   811→        delta = new - old if old > 0 else 0
   812→        crit = sum(
   813→            1 for f in self._findings_list
   814→            if f.get("severity", "").lower() == "critical"
   815→        )
   816→        high = sum(
   817→            1 for f in self._findings_list
   818→            if f.get("severity", "").lower() == "high"
   819→        )
   820→        med = sum(
   821→            1 for f in self._findings_list
   822→            if f.get("severity", "").lower() == "medium"
   823→        )
   824→
   825→        # Update stat counters
   826→        try:
   827→            self.query_one("#stat-findings", StatCounter).set(new, delta=delta)
   828→        except NoMatches:
   829→            pass
   830→        try:
   831→            self.query_one("#stat-critical", StatCounter).set(crit)
   832→        except NoMatches:
   833→            pass
   834→        try:
   835→            self.query_one("#stat-high", StatCounter).set(high)
   836→        except NoMatches:
   837→            pass
   838→        try:
   839→            self.query_one("#stat-medium", StatCounter).set(med)
   840→        except NoMatches:
   841→            pass
   842→
   843→        # Track finding timestamps for true per-minute rate
   844→        now = time.monotonic()
   845→        for _ in range(delta):
   846→            self._finding_timestamps.append(now)
   847→
   848→        # Push to findings sparkline (true findings-per-minute)
   849→        try:
   850→            spark = self.query_one("#spark-findings", Sparkline)
   851→            # Calculate actual FPM from timestamps in the last 60s
   852→            cutoff = now - 60.0
   853→            self._finding_timestamps = [t for t in self._finding_timestamps if t > cutoff]
   854→            fpm = len(self._finding_timestamps)
   855→            spark.push(float(fpm))
   856→        except NoMatches:
   857→            pass
   858→
   859→        # Wire into velocity meter
   860→        try:
   861→            vm = self.query_one("#velocity-meter", VelocityMeter)
   862→            if delta > 0:
   863→                vm.record_finding(delta)
   864→        except NoMatches:
   865→            pass
   866→
   867→        # Keep bottom-info label too (compact view)
   868→        try:
   869→            el = self.query_one("#bottom-info", Label)
   870→            el.update(
   871→                f"[{GREEN}]●[/] {new} findings  [{RED}]{crit}[/]C [{YELLOW}]{high}[/]H [{MUTED}]{med}[/]M"
   872→            )
   873→        except NoMatches:
   874→            pass

  1019→    def _process_command(self, raw: str) -> None:
  1020→        """Parse and execute a user command."""
  1021→        self._chat(f"[{CYAN} bold]▸ {raw}[/]")
  1022→
  1023→        parts = raw.split()

  1035→        if cmd == "help":
  1036→            self._show_help()
  1037→            return
  1038→
  1039→        if cmd == "scan":
  1040→            self._handle_scan(args)
  1041→            return
  1042→
  1043→        if cmd == "audit":
  1044→            self._handle_audit(args)
  1045→            return
  1046→
  1047→        if cmd == "dev":

  1053→            return

  1081→            return
  1082→
  1083→        if cmd == "history":
  1084→            self._handle_history()
  1085→            return
  1086→
  1087→        if cmd == "theme":
  1088→            # Already handled in on_input_submitted before _process_command
  1089→            return
  1090→
  1091→        self._chat(f"[{RED}]Unknown command: {cmd}[/]")

  1094→    def _handle_theme(self, args: List[str]) -> None:
  1095→        """Handle theme switching command."""
  1096→        if not args or args[0].lower() in ("list", "ls"):
  1097→            available = Theme.available_themes()
  1098→            current = Theme.current_name()
  1099→            self._chat(f"[{CYAN} bold]  THEMES[/]")
  1100→            for name in available:
  1101→                marker = f" [{GREEN}]●[/]" if name == current else "  [{DIM_CYAN}]○[/]"
  1102→                display_name = THEMES[name].get("name", name)
  1103→                self._chat(f"  {marker} [{CYAN}]{name}[/] [{MUTED}]- {display_name}[/]")
  1104→            return
  1105→
  1106→        name = args[0].lower()
  1107→        try:
  1108→            Theme.set_theme(name)
  1109→            self._chat(f"[{GREEN}]  ✓ Theme changed to [bold]{name}[/]. Restart nexus to apply fully.[/]")
  1110→        except ValueError as e:
  1111→            self._chat(f"[{RED}]  {e}[/]")

  1278→    @work(exclusive=False, thread=True)
  1279→    def _run_scan_worker(
  1280→        self,
  1281→        target: str,
  1282→        modules: Optional[List[str]] = None,
  1283→        is_local: bool = False,
  1284→    ) -> None:
  1285→        """Run a scan in a background thread."""
  1286→        self.call_from_thread(self.is_scanning.set, True)
  1287→
  1288→        # Set modules to scanning
  1289→        if modules:
  1290→            for m in modules:
  1291→                if m in self._module_cells:
  1292→                    self.call_from_thread(self._module_cells[m].set_status, "scanning")
  1293→        else:
  1294→            all_mods = (
  1295→                list(MODULE_NAMES.keys())
  1296→                if is_local
  1297→                else [k for k in MODULE_NAMES if k not in ("host", "dev", "doctor")]
  1298→            )
  1299→            for m in all_mods:
  1300→                if m in self._module_cells:
  1301→                    self.call_from_thread(self._module_cells[m].set_status, "scanning")
  1302→
  1303→        try:
  1304→            from .scanner import scan, audit_scan
  1305→            from .history import save_scan
  1306→
  1307→            self.call_from_thread(
  1308→                self._chat, f"[{DIM_CYAN}]  ⏳ Scan in progress...[/]"
  1309→            )
  1310→
  1311→            if is_local:
  1312→                result = audit_scan(target=target, modules=modules)
  1313→            else:
  1314→                result = scan(target=target, modules=modules)
  1315→
  1316→            # Process findings
  1317→            data = result.to_dict()
  1318→            self.call_from_thread(self._on_scan_complete, data, modules, is_local)
  1319→
  1320→            # Save to history
  1321→            try:
  1322→                save_scan(data, label="nexus")
  1323→            except Exception:
  1324→                pass
  1325→
  1326→        except Exception as e:
  1327→            self.call_from_thread(self._chat, f"[{RED}]  ✗ Scan error: {e}[/]")
  1328→            if modules:
  1329→                for m in modules:
  1330→                    if m in self._module_cells:
  1331→                        self.call_from_thread(self._module_cells[m].set_status, "error")
  1332→            else:
  1333→                self.call_from_thread(self._set_all_modules_status, "error")
  1334→
  1335→        finally:
  1336→            self.call_from_thread(self.is_scanning.set, False)

  1689→    def on_click(self, event: Click) -> None:
  1690→        """Open finding detail modal for the clicked finding.
  1691→
  1692→        Uses the vertical click position to determine which finding index
  1693→        the user clicked on, mapping it to _findings_list.
  1694→        """
  1695→        try:
  1696→            if (
  1697→                isinstance(event.widget, RichLog)
  1698→                and event.widget.id == "findings-feed"
  1699→                and self._findings_list
  1700→            ):
  1701→                # Get the line offset to determine which finding was clicked.
  1702→                # Each finding entry is 1 line in the RichLog.
  1703→                # We use the y coordinate relative to the widget to estimate the index.
  1704→                region = event.widget.region
  1705→                if region is not None:
  1706→                    # Calculate which line was clicked relative to the scroll position
  1707→                    scroll_offset = event.widget.scroll_y
  1708→                    click_y = event.y - region.y
  1709→                    # Each finding takes 1 line; estimate index from click position
  1710→                    # Account for scroll position
  1711→                    line_height = 1
  1712→                    if hasattr(event.widget, "_line_height"):
  1713→                        line_height = event.widget._line_height
  1714→                    clicked_index = int((click_y + scroll_offset * line_height) / line_height)
  1715→                    # Clamp to valid range
  1716→                    clicked_index = max(0, min(clicked_index, len(self._findings_list) - 1))
  1717→                    finding = self._findings_list[clicked_index]
  1718→                else:
  1719→                    finding = self._findings_list[-1]
  1720→                self.push_screen(FindingDetailModal(finding))
  1721→        except (NoMatches, Exception):
  1722→            pass
