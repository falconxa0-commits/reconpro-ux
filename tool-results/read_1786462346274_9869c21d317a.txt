     1→"""ReconPro Age III — Auto Fix Proposal Engine.
     2→
     3→Analyzes detected issues and proposes automated fixes.  Every fix is a
     4→*proposal* that must be explicitly approved before it is applied.
     5→Dry-run mode is the default — nothing is ever written to disk without
     6→the caller's consent.
     7→
     8→Classes:
     9→    FixProposal      — Dataclass for a single fix proposal.
    10→    FixAnalyzer      — Generates fix proposals from issue patterns.
    11→    FixStore         — Persistent JSON-backed proposal storage.
    12→    AutoFixEngine    — Top-level orchestrator.
    13→
    14→Lifecycle of a proposal:
    15→    proposed → approved → applied → verified  (or  rejected / failed)
    16→"""

    69→# FixProposal dataclass
    70→# ═══════════════════════════════════════════════════════════════════════════
    71→
    72→
    73→@dataclass
    74→class FixProposal:
    75→    """A single proposed fix for a detected issue.
    76→
    77→    Attributes:
    78→        id:               Unique identifier (UUID4-based).
    79→        title:            Short human-readable title.
    80→        description:      Detailed explanation of the fix.
    81→        severity:         Severity of the *underlying issue* (critical/high/…).
    82→        category:         FixCategory enum value.
    83→        affected_file:    Absolute path to the file (or "").
    84→        affected_lines:   1-based line numbers impacted.
    85→        original_code:    Original source lines (may be empty for config fixes).
    86→        proposed_code:    Proposed replacement lines.
    87→        risk_level:       Risk of applying the fix.
    88→        confidence:       0.0–1.0 confidence that the fix is correct.
    89→        automated:        True if the fix can be applied without human review.
    90→        verification_steps: List of steps to confirm the fix worked.
    91→        rollback_plan:    How to undo the fix.
    92→        status:           Current ProposalStatus.
    93→        source_issue:     The original issue dict that triggered this proposal.
    94→        created_at:       ISO-8601 timestamp.
    95→        applied_at:       ISO-8601 timestamp (empty until applied).
    96→        verified_at:      ISO-8601 timestamp (empty until verified).
    97→        error_message:    If status is FAILED, the error text.
    98→    """
    99→    id: str = field(default_factory=lambda: uuid.uuid4().hex[:12])
   100→    title: str = ""
   101→    description: str = ""
   102→    severity: str = "medium"
   103→    category: str = FixCategory.CODE_QUALITY.value
   104→    affected_file: str = ""
   105→    affected_lines: List[int] = field(default_factory=list)
   106→    original_code: str = ""
   107→    proposed_code: str = ""
   108→    risk_level: str = RiskLevel.MEDIUM.value
   109→    confidence: float = 0.5
   110→    automated: bool = False
   111→    verification_steps: List[str] = field(default_factory=list)
   112→    rollback_plan: str = ""
   113→    status: str = ProposalStatus.PROPOSED.value
   114→    source_issue: Dict[str, Any] = field(default_factory=dict)
   115→    created_at: str = field(
   116→        default_factory=lambda: datetime.now(timezone.utc).isoformat(),
   117→    )
   118→    applied_at: str = ""
   119→    verified_at: str = ""
   120→    error_message: str = ""
   121→
   122→    # ── Serialisation ────────────────────────────────────────────────
   123→
   124→    def to_dict(self) -> Dict[str, Any]:
   125→        return {
   126→            "id": self.id,
   127→            "title": self.title,
   128→            "description": self.description,
   129→            "severity": self.severity,
   130→            "category": self.category,
   131→            "affected_file": self.affected_file,
   132→            "affected_lines": self.affected_lines,
   133→            "original_code": self.original_code,
   134→            "proposed_code": self.proposed_code,
   135→            "risk_level": self.risk_level,
   136→            "confidence": self.confidence,
   137→            "automated": self.automated,
   138→            "verification_steps": self.verification_steps,
   139→            "rollback_plan": self.rollback_plan,
   140→            "status": self.status,
   141→            "source_issue": self.source_issue,
   142→            "created_at": self.created_at,
   143→            "applied_at": self.applied_at,
   144→            "verified_at": self.verified_at,
   145→            "error_message": self.error_message,
   146→        }
   147→
   148→    @classmethod
   149→    def from_dict(cls, data: Dict[str, Any]) -> FixProposal:
   150→        return cls(
   151→            id=data.get("id", uuid.uuid4().hex[:12]),
   152→            title=data.get("title", ""),
   153→            description=data.get("description", ""),
   154→            severity=data.get("severity", "medium"),
   155→            category=data.get("category", FixCategory.CODE_QUALITY.value),
   156→            affected_file=data.get("affected_file", ""),
   157→            affected_lines=data.get("affected_lines", []),
   158→            original_code=data.get("original_code", ""),
   159→            proposed_code=data.get("proposed_code", ""),
   160→            risk_level=data.get("risk_level", RiskLevel.MEDIUM.value),
   161→            confidence=data.get("confidence", 0.5),
   162→            automated=data.get("automated", False),
   163→            verification_steps=data.get("verification_steps", []),
   164→            rollback_plan=data.get("rollback_plan", ""),
   165→            status=data.get("status", ProposalStatus.PROPOSED.value),
   166→            source_issue=data.get("source_issue", {}),
   167→            created_at=data.get("created_at", ""),
   168→            applied_at=data.get("applied_at", ""),
   169→            verified_at=data.get("verified_at", ""),
   170→            error_message=data.get("error_message", ""),
   171→        )

   563→        "match_pattern": r"below minimum",
   564→        "title": "Upgrade Python to 3.9 or higher",
   565→        "description": (
   566→            "Python version is below the minimum supported (3.9). "
   567→            "Upgrade Python to the latest stable release."
   568→        ),
   569→        "category": FixCategory.DEPENDENCY.value,
   570→        "risk": RiskLevel.HIGH.value,
   571→        "confidence": 0.95,
   572→        "automated": False,
   573→    },
   574→]
   575→
   576→
   577→# Merge all templates for fast lookup
   578→_ALL_FIX_TEMPLATES: List[Dict[str, Any]] = (
   579→    _SECURITY_FIX_TEMPLATES
   580→    + _PERFORMANCE_FIX_TEMPLATES
   581→    + _CODE_QUALITY_TEMPLATES
   582→    + _CONFIG_FIX_TEMPLATES
   583→)
   584→
   585→
   586→# ═══════════════════════════════════════════════════════════════════════════
   587→# FixAnalyzer
   588→# ═══════════════════════════════════════════════════════════════════════════
   589→
   590→
   591→class FixAnalyzer:
   592→    """Analyses issues and generates FixProposal objects.
   593→
   594→    Uses pattern matching against a knowledge base of known issue → fix
   595→    mappings, plus AST-based code analysis for Python source files.
   596→    """
   597→
   598→    def __init__(self) -> None:
   599→        self._template_cache: Dict[str, List[Dict[str, Any]]] = {}
   600→        self._build_template_index()
   601→
   602→    def _build_template_index(self) -> None:
   603→        """Index templates by match_category for O(1) lookup."""
   604→        for tmpl in _ALL_FIX_TEMPLATES:
   605→            cat = tmpl["match_category"]
   606→            self._template_cache.setdefault(cat, []).append(tmpl)
   607→
   608→    # ── Public API ────────────────────────────────────────────────────
   609→
   610→    def analyze_issue(self, issue: Dict[str, Any]) -> Optional[FixProposal]:
   611→        """Analyse a single issue dict and return a FixProposal (or None)."""
   612→        # Determine which kind of issue this is
   613→        category = issue.get("category", "")
   614→        message = issue.get("message", "")
   615→        evidence = issue.get("evidence", "")
   616→        title = issue.get("title", "")
   617→        severity = issue.get("severity", "medium")
   618→        description = issue.get("description", "")
   619→        file_path = issue.get("asset", issue.get("file", ""))
   620→
   621→        # Combine searchable text
   622→        search_text = " ".join([title, message, evidence, description]).lower()
   623→
   624→        # 1. Try exact category match first
   625→        proposals = self._match_by_category(category, issue)
   626→        if proposals:
   627→            proposal = proposals[0]
   628→            proposal.severity = self._normalise_severity(severity)
   629→            proposal.source_issue = issue
   630→            proposal.affected_file = str(file_path) if file_path else ""
   631→            return proposal
   632→
   633→        # 2. Try pattern match on the text
   634→        proposals = self._match_by_pattern(search_text, issue)
   635→        if proposals:
   636→            proposal = proposals[0]
   637→            proposal.severity = self._normalise_severity(severity)
   638→            proposal.source_issue = issue
   639→            proposal.affected_file = str(file_path) if file_path else ""
   640→            return proposal
   641→
   642→        # 3. Generate a generic proposal if we have any issue at all
   643→        if message or title:
   644→            return self._generic_proposal(issue)
   645→
   646→        return None
   647→
   648→    def propose_code_fixes(
   649→        self,
   650→        file_path: str,
   651→        issues: List[Dict[str, Any]],
   652→    ) -> List[FixProposal]:
   653→        """Generate code-level fix proposals for issues in a file.
   654→
   655→        Uses AST analysis to propose concrete code changes when possible.
   656→        """
   657→        proposals: List[FixProposal] = []
   658→        path = Path(file_path)
   659→
   660→        if not path.is_file():
   661→            logger.warning("Cannot propose code fixes for non-existent file: %s", file_path)
   662→            return proposals
   663→
   664→        try:
   665→            source = path.read_text(encoding="utf-8", errors="replace")
   666→        except OSError as exc:
   667→            logger.error("Failed to read %s: %s", file_path, exc)
   668→            return proposals
   669→
   670→        lines = source.splitlines()
   671→
   672→        for issue in issues:
   673→            proposal = self._generate_code_proposal(file_path, source, lines, issue)
   674→            if proposal is not None:
   675→                proposals.append(proposal)
   676→
   677→        return proposals
   678→
   679→    def propose_config_fixes(
   680→        self,
   681→        config_issues: List[Dict[str, Any]],
   682→    ) -> List[FixProposal]:
   683→        """Generate configuration fix proposals."""
   684→        proposals: List[FixProposal] = []
   685→        for issue in config_issues:
   686→            proposal = self.analyze_issue(issue)
   687→            if proposal is not None:
   688→                proposals.append(proposal)
   689→        return proposals
   690→
   691→    # ── Internal matching ─────────────────────────────────────────────
   692→
   693→    def _match_by_category(
   694→        self,
   695→        category: str,
   696→        issue: Dict[str, Any],
   697→    ) -> List[FixProposal]:
   698→        """Find fix templates matching the issue category."""
   699→        results: List[FixProposal] = []
   700→        templates = self._template_cache.get(category, [])
   701→        for tmpl in templates:
   702→            proposal = self._template_to_proposal(tmpl, issue)
   703→            if proposal is not None:
   704→                results.append(proposal)
   705→        return results
   706→
   707→    def _match_by_pattern(
   708→        self,
   709→        text: str,
   710→        issue: Dict[str, Any],
   711→    ) -> List[FixProposal]:
   712→        """Find fix templates whose regex pattern matches the text."""
   713→        results: List[FixProposal] = []
   714→        seen_categories: Set[str] = set()
   715→        for tmpl in _ALL_FIX_TEMPLATES:
   716→            cat = tmpl["match_category"]
   717→            if cat in seen_categories:
   718→                continue
   719→            try:
   720→                pattern = re.compile(tmpl["match_pattern"], re.IGNORECASE)
   721→                if pattern.search(text):
   722→                    proposal = self._template_to_proposal(tmpl, issue)
   723→                    if proposal is not None:
   724→                        results.append(proposal)
   725→                        seen_categories.add(cat)
   726→            except re.error:
   727→                continue
   728→        return results
   729→
   730→    def _template_to_proposal(
   731→        self,
   732→        tmpl: Dict[str, Any],
   733→        issue: Dict[str, Any],
   734→    ) -> Optional[FixProposal]:
   735→        """Convert a fix template + issue into a FixProposal."""
   736→        title = tmpl.get("title", "")
   737→        if not title:
   738→            return None
   739→
   740→        evidence = issue.get("evidence", "")
   741→        file_path = issue.get("asset", issue.get("file", ""))
   742→
   743→        proposal = FixProposal(
   744→            title=title,
   745→            description=tmpl.get("description", ""),
   746→            severity=self._normalise_severity(issue.get("severity", "medium")),
   747→            category=tmpl.get("category", FixCategory.CODE_QUALITY.value),
   748→            affected_file=str(file_path) if file_path else "",
   749→            original_code=evidence,
   750→            risk_level=tmpl.get("risk", RiskLevel.MEDIUM.value),
   751→            confidence=tmpl.get("confidence", 0.5),
   752→            automated=tmpl.get("automated", False),
   753→            source_issue=issue,
   754→            rollback_plan=self._generate_rollback(tmpl, issue),
   755→            verification_steps=self._generate_verification(tmpl, issue),
   756→        )
   757→        return proposal
   758→
   759→    def _generic_proposal(self, issue: Dict[str, Any]) -> FixProposal:
   760→        """Create a generic fix proposal when no template matches."""
   761→        title = issue.get("title", "") or issue.get("message", "")
   762→        suggestion = issue.get("suggestion", "") or issue.get("remediation", "")
   763→        description = issue.get("description", "")
   764→        category = issue.get("category", "")
   765→
   766→        # Map issue category to fix category
   767→        fix_cat = FixCategory.CODE_QUALITY.value
   768→        if category in ("security", "headers", "tls", "ssl"):
   769→            fix_cat = FixCategory.SECURITY.value
   770→        elif category in ("performance", "speed"):
   771→            fix_cat = FixCategory.PERFORMANCE.value
   772→        elif category in ("paths", "config", "disk", "network", "python", "modules"):
   773→            fix_cat = FixCategory.CONFIG.value
   774→        elif category in ("deps", "dependencies"):
   775→            fix_cat = FixCategory.DEPENDENCY.value
   776→
   777→        return FixProposal(
   778→            title=f"Review and fix: {title}",
   779→            description=description or f"Address the issue: {title}",
   780→            severity=self._normalise_severity(issue.get("severity", "medium")),
   781→            category=fix_cat,
   782→            affected_file=issue.get("asset", ""),
   783→            proposed_code=suggestion,
   784→            risk_level=RiskLevel.MEDIUM.value,
   785→            confidence=0.3,
   786→            automated=False,
   787→            source_issue=issue,
   788→            verification_steps=[
   789→                f"Manually review the issue: {title}",
   790→                f"Apply the suggested fix: {suggestion}" if suggestion else "Determine and apply the appropriate fix",
   791→                "Re-run diagnostics or scan to confirm resolution",
   792→            ],
   793→            rollback_plan="Revert manual changes. No automated rollback available for generic proposals.",
   794→        )
   795→
   796→    # ── Code-level proposal generation ────────────────────────────────
   797→
   798→    def _generate_code_proposal(
   799→        self,
   800→        file_path: str,
   801→        source: str,
   802→        lines: List[str],
   803→        issue: Dict[str, Any],
   804→    ) -> Optional[FixProposal]:
   805→        """Generate a concrete code fix proposal using AST analysis."""
   806→        category = issue.get("category", "")
   807→        evidence = issue.get("evidence", "")
   808→
   809→        # Try AST-based transformations for Python files
   810→        if file_path.endswith(".py"):
   811→            proposal = self._ast_code_proposal(file_path, source, lines, issue)
   812→            if proposal is not None:
   813→                return proposal
   814→
   815→        # Fall back to text-based proposals
   816→        proposal = self.analyze_issue(issue)
   817→        if proposal is not None and not proposal.original_code and evidence:
   818→            proposal.original_code = evidence
   819→            proposal.affected_file = file_path
   820→        return proposal
   821→
   822→    def _ast_code_proposal(
   823→        self,
   824→        file_path: str,
   825→        source: str,
   826→        lines: List[str],
   827→        issue: Dict[str, Any],
   828→    ) -> Optional[FixProposal]:
   829→        """Use AST analysis to generate a concrete code fix."""
   830→        try:
   831→            tree = ast.parse(source, filename=file_path)
   832→        except SyntaxError:
   833→            return None
   834→
   835→        category = issue.get("category", "")
   836→        title = issue.get("title", "")
   837→        evidence = issue.get("evidence", "")
   838→
   839→        # Find the relevant node
   840→        finder = _IssueNodeFinder(category, evidence, title)
   841→        finder.visit(tree)
   842→
   843→        if not finder.matched_nodes:
   844→            return None
   845→
   846→        node = finder.matched_nodes[0]
   847→        start_line = getattr(node, "lineno", 0)
   848→        end_line = getattr(node, "end_lineno", start_line)
   849→
   850→        # Extract original code lines
   851→        orig_lines = lines[start_line - 1 : end_line]
   852→        original_code = "\n".join(orig_lines)
   853→
   854→        # Generate proposed code based on category
   855→        proposed_code = self._transform_node(category, node, lines, start_line, end_line)
   856→        if proposed_code is None:
   857→            return None
   858→
   859→        # Look up template for metadata
   860→        templates = self._template_cache.get(category, [])
   861→        tmpl = templates[0] if templates else {}
   862→
   863→        return FixProposal(
   864→            title=tmpl.get("title", f"Fix: {title}"),
   865→            description=tmpl.get("description", f"Address the {category} issue in {file_path}"),
   866→            severity=self._normalise_severity(issue.get("severity", "medium")),
   867→            category=tmpl.get("category", FixCategory.SECURITY.value),
   868→            affected_file=file_path,
   869→            affected_lines=list(range(start_line, end_line + 1)),
   870→            original_code=original_code,
   871→            proposed_code=proposed_code,
   872→            risk_level=tmpl.get("risk", RiskLevel.MEDIUM.value),
   873→            confidence=tmpl.get("confidence", 0.7),
   874→            automated=tmpl.get("automated", False),
   875→            source_issue=issue,
   876→            verification_steps=[
   877→                f"Verify the fix in {file_path} lines {start_line}-{end_line}",
   878→                "Run the AST analyzer again to confirm the issue is resolved",
   879→                "Run tests to ensure no regressions",
   880→            ],
   881→            rollback_plan=f"Restore original code at lines {start_line}-{end_line} in {file_path}",
   882→        )
   883→
   884→    def _transform_node(
   885→        self,
   886→        category: str,
   887→        node: ast.AST,
   888→        lines: List[str],
   889→        start_line: int,
   890→        end_line: int,
   891→    ) -> Optional[str]:
   892→        """Transform an AST node into fixed code.
   893→
   894→        Returns the proposed replacement code as a string, or None if no
   895→        transformation is possible.
   896→        """
   897→        indent = self._get_indent(lines, start_line - 1)
   898→
   899→        if category == "dangerous_eval" and isinstance(node, ast.Call):
   900→            return self._transform_eval(node, indent, lines, start_line)
   901→
   902→        if category == "dangerous_exec" and isinstance(node, ast.Call):
   903→            return self._transform_exec(node, indent, lines, start_line)
   904→
   905→        if category == "command_injection" and isinstance(node, ast.Call):
   906→            return self._transform_shell_true(node, indent, lines, start_line, end_line)
   907→
   908→        if category == "hardcoded_secret" and isinstance(node, ast.Assign):
   909→            return self._transform_hardcoded_secret(node, indent, lines, start_line)
   910→
   911→        if category == "bare_except":
   912→            return self._transform_bare_except(indent, lines, start_line)
   913→
   914→        if category == "debug_mode":
   915→            return self._transform_debug_mode(indent, lines, start_line)
   916→
   917→        if category == "cleartext_http":
   918→            return self._transform_http_to_https(indent, lines, start_line, end_line)
   919→
   920→        if category == "temp_file_race":
   921→            return self._transform_mktemp(indent, lines, start_line)
   922→
   923→        if category == "csrf_disabled":
   924→            return self._transform_csrf_disabled(indent, lines, start_line)
   925→
   926→        return None
   927→

   930→    @staticmethod
   931→    def _get_indent(lines: List[str], index: int) -> str:
   932→        """Extract leading whitespace from a line."""
   933→        if 0 <= index < len(lines):
   934→            line = lines[index]
   935→            return line[: len(line) - len(line.lstrip())]
   936→        return ""
   937→
   938→    def _transform_eval(
   939→        self,
   940→        node: ast.Call,
   941→        indent: str,
   942→        lines: List[str],
   943→        start_line: int,
   944→    ) -> Optional[str]:
   945→        """Propose replacing eval() with ast.literal_eval()."""
   946→        func_name = _get_call_name(node)
   947→        if func_name != "eval":
   948→            return None
   949→
   950→        # Reconstruct the call argument
   951→        if node.args:
   952→            arg_src = self._extract_arg_source(node.args[0], lines)
   953→        else:
   954→            arg_src = "..."
   955→
   956→        return (
   957→            f"{indent}import ast as _ast  # noqa: added for safe evaluation\n"
   958→            f"{indent}_ast.literal_eval({arg_src})"
   959→        )
   960→
   961→    def _transform_exec(
   962→        self,
   963→        node: ast.Call,
   964→        indent: str,
   965→        lines: List[str],
   966→        start_line: int,
   967→    ) -> Optional[str]:
   968→        """Propose removing exec() or marking it as dangerous."""
   969→        func_name = _get_call_name(node)
   970→        if func_name != "exec":
   971→            return None
   972→
   973→        arg_src = ""
   974→        if node.args:
   975→            arg_src = self._extract_arg_source(node.args[0], lines)
   976→        else:
   977→            arg_src = "..."
   978→
   979→        return (
   980→            f"{indent}# WARNING: exec() removed — arbitrary code execution risk\n"
   981→            f"{indent}# If this is needed, use a restricted sandbox or whitelist approach\n"
   982→            f"{indent}# Original: exec({arg_src})\n"
   983→            f"{indent}raise NotImplementedError(\"exec() was removed for security reasons\")"
   984→        )
   985→
   986→    def _transform_shell_true(
   987→        self,
   988→        node: ast.Call,
   989→        indent: str,
   990→        lines: List[str],
   991→        start_line: int,
   992→        end_line: int,
   993→    ) -> Optional[str]:
   994→        """Remove shell=True from a subprocess call."""
   995→        func_name = _get_call_name(node)
   996→        if func_name not in ("run", "call", "Popen", "check_output", "check_call"):
   997→            return None
   998→
   999→        # Find and remove shell=True keyword
  1000→        new_lines: List[str] = []
  1001→        for i in range(start_line - 1, end_line):
  1002→            line = lines[i]
  1003→            if "shell=True" in line:
  1004→                # Remove the shell=True keyword argument
  1005→                line = re.sub(r",?\s*shell\s*=\s*True\s*,?", ",", line)
  1006→                line = re.sub(r",\s*,", ",", line)  # Clean double commas
  1007→                # If shell=True was the only keyword, clean up
  1008→                line = re.sub(r"\(\s*,", "(", line)
  1009→                line = re.sub(r",\s*\)", ")", line)
  1010→            new_lines.append(line)
  1011→
  1012→        return "\n".join(new_lines)

  1014→    def _transform_hardcoded_secret(
  1015→        self,
  1016→        node: ast.Assign,
  1017→        indent: str,
  1018→        lines: List[str],
  1019→        start_line: int,
  1020→    ) -> Optional[str]:
  1021→        """Replace hardcoded secret with os.environ.get()."""
  1022→        # Get variable name
  1023→        targets = node.targets
  1024→        if not targets or not isinstance(targets[0], ast.Name):
  1025→            return None
  1026→
  1027→        var_name = targets[0].id
  1028→        return (
  1029→            f"{indent}import os  # noqa: added for secret loading\n"
  1030→            f"{indent}{var_name} = os.environ.get(\"{var_name.upper()}\", \"\")  "
  1031→            f"# was hardcoded secret"
  1032→        )

  1034→    def _transform_bare_except(
  1035→        self,
  1036→        indent: str,
  1037→        lines: List[str],
  1038→        start_line: int,
  1039→    ) -> Optional[str]:
  1040→        """Replace bare 'except:' with 'except Exception:'."""
  1041→        if start_line - 1 < len(lines):
  1042→            line = lines[start_line - 1]
  1043→            if re.search(r"except\s*:\s*$", line):
  1044→                new_line = re.sub(r"except\s*:", "except Exception:", line)
  1045→                return new_line
  1046→        return None
  1047→
  1048→    def _transform_debug_mode(
  1049→        self,
  1050→        indent: str,
  1051→        lines: List[str],
  1052→        start_line: int,
  1053→    ) -> Optional[str]:
  1054→        """Replace DEBUG=True with DEBUG=False."""
  1055→        if start_line - 1 < len(lines):
  1056→            line = lines[start_line - 1]
  1057→            new_line = re.sub(
  1058→                r"(DEBUG\s*=\s*)True",
  1059→                r"\1False  # disabled in production",
  1060→                line,
  1061→                flags=re.IGNORECASE,
  1062→            )
  1063→            if new_line != line:
  1064→                return new_line
  1065→        return None

  1067→    def _transform_http_to_https(
  1068→        self,
  1069→        indent: str,
  1070→        lines: List[str],
  1071→        start_line: int,
  1072→        end_line: int,
  1073→    ) -> Optional[str]:
  1074→        """Replace http:// with https:// in the matched lines."""
  1075→        new_lines: List[str] = []
  1076→        changed = False
  1077→        for i in range(start_line - 1, end_line):
  1078→            line = lines[i]
  1079→            new_line = re.sub(r'"http://', '"https://', line)
  1080→            new_line = re.sub(r"'http://", "'https://", new_line)
  1081→            if new_line != line:
  1082→                changed = True
  1083→            new_lines.append(new_line)
  1084→        return "\n".join(new_lines) if changed else None

  1086→    def _transform_mktemp(
  1087→        self,
  1088→        indent: str,
  1089→        lines: List[str],
  1090→        start_line: int,
  1091→    ) -> Optional[str]:
  1092→        """Replace mktemp() with mkstemp() or NamedTemporaryFile()."""
  1093→        if start_line - 1 < len(lines):
  1094→            line = lines[start_line - 1]
  1095→            if re.search(r"mktemp\s*\(", line):
  1096→                new_line = re.sub(
  1097→                    r"mktemp\s*\([^)]*\)",
  1098→                    "NamedTemporaryFile(delete=False).name",
  1099→                    line,
  1100→                )
  1101→                # Add import if needed
  1102→                return (
  1103→                    f"{indent}from tempfile import NamedTemporaryFile  # noqa: safe replacement\n"
  1104→                    f"{new_line}"
  1105→                )
  1106→        return None

  1108→    def _transform_csrf_disabled(
  1109→        self,
  1110→        indent: str,
  1111→        lines: List[str],
  1112→        start_line: int,

  1114→        """Replace CSRF_ENABLED=False with CSRF_ENABLED=True."""
  1115→        if start_line - 1 < len(lines):

  1121→                flags=re.IGNORECASE,
  1122→            )
  1123→            if new_line != line:

  1128→    def _extract_arg_source(node: ast.AST, lines: List[str]) -> str:
  1129→        """Best-effort extraction of source text for an AST node."""
  1130→        start = getattr(node, "lineno", 0)
  1131→        end = getattr(node, "end_lineno", start)
  1132→        col_start = getattr(node, "col_offset", 0)
  1133→        col_end = getattr(node, "end_col_offset", None)
  1134→
  1135→        if 0 < start <= len(lines):
  1136→            line = lines[start - 1]
  1137→            if col_end is not None:
  1138→                return line[col_start:col_end]
  1139→            # Fallback: take from col_start to end of line
  1140→            return line[col_start:]
  1141→        return "..."

  1145→        """Normalise a severity string to a valid value."""
  1146→        sev = severity.lower().strip()
  1147→        if sev in VALID_SEVERITIES:
  1148→            return sev
  1149→        # Map common alternatives
  1150→        mapping = {
  1151→            "error": "high",
  1152→            "warn": "medium",
  1153→            "warning": "medium",
  1154→        }
  1155→        return mapping.get(sev, "medium")

  1163→        cat = tmpl.get("match_category", "")

  1196→class _IssueNodeFinder(ast.NodeVisitor):
  1197→    """Walks the AST and collects nodes that match a given issue."""
  1198→
  1199→    def __init__(self, category: str, evidence: str, title: str) -> None:
  1200→        self.category = category
  1201→        self.evidence = evidence.lower()
  1202→        self.title = title.lower()
  1203→        self.matched_nodes: List[ast.AST] = []
  1204→        self._imported: Set[str] = set()
  1205→
  1206→    def visit_Import(self, node: ast.Import) -> None:
  1207→        for alias in node.names:
  1208→            self._imported.add(alias.name.split(".")[0])
  1209→        self.generic_visit(node)
  1210→
  1211→    def visit_ImportFrom(self, node: ast.ImportFrom) -> None:
  1212→        if node.module:
  1213→            self._imported.add(node.module.split(".")[0])
  1214→        self.generic_visit(node)
  1215→
  1216→    def visit_Call(self, node: ast.Call) -> None:
  1217→        fname = _get_call_name(node)
  1218→        func_full = _get_call_full_name(node)
  1219→
  1220→        if self.category == "dangerous_eval" and fname == "eval":
  1221→            self.matched_nodes.append(node)
  1222→        elif self.category == "dangerous_exec" and fname == "exec":
  1223→            self.matched_nodes.append(node)
  1224→        elif self.category == "command_injection":
  1225→            if fname in ("run", "call", "Popen", "check_output", "check_call"):
  1226→                for kw in node.keywords:
  1227→                    if kw.arg == "shell" and isinstance(kw.value, ast.Constant) and kw.value.value is True:
  1228→                        self.matched_nodes.append(node)
  1229→                        break
  1230→        elif self.category == "unsafe_deserialization":
  1231→            if func_full and ("pickle.loads" in func_full or "marshal.loads" in func_full):
  1232→                self.matched_nodes.append(node)
  1233→            elif func_full and "yaml.load" in func_full:
  1234→                self.matched_nodes.append(node)
  1235→        elif self.category == "jwt_no_verify":
  1236→            if func_full and "jwt.decode" in func_full:
  1237→                self.matched_nodes.append(node)
  1238→        elif self.category == "temp_file_race":
  1239→            if func_full and "mktemp" in func_full:
  1240→                self.matched_nodes.append(node)
  1241→
  1242→        self.generic_visit(node)
  1243→
  1244→    def visit_Assign(self, node: ast.Assign) -> None:
  1245→        if self.category == "hardcoded_secret":
  1246→            # Get the source text for the line
  1247→            start = getattr(node, "lineno", 0)
  1248→            end = getattr(node, "end_lineno", start)
  1249→            # We'll match by the parent logic
  1250→            for target in node.targets:
  1251→                if isinstance(target, ast.Name):
  1252→                    name = target.id.lower()
  1253→                    secret_names = ("password", "passwd", "secret", "token",
  1254→                                    "api_key", "apikey", "private_key")
  1255→                    if any(s in name for s in secret_names):
  1256→                        # Check if value is a string constant
  1257→                        if isinstance(node.value, (ast.Constant,)) and isinstance(node.value.value, str):
  1258→                            self.matched_nodes.append(node)
  1259→
  1260→        self.generic_visit(node)
  1261→
  1262→    def visit_ExceptHandler(self, node: ast.ExceptHandler) -> None:
  1263→        if self.category == "bare_except":
  1264→            if node.type is None:
  1265→                self.matched_nodes.append(node)
  1266→        self.generic_visit(node)
  1267→
  1268→    def visit_JoinedStr(self, node: ast.JoinedStr) -> None:
  1269→        if self.category == "sql_injection":
  1270→            parts: List[str] = []
  1271→            for val in node.values:
  1272→                if isinstance(val, ast.Constant):
  1273→                    parts.append(str(val.value))
  1274→                elif isinstance(val, ast.FormattedValue):
  1275→                    parts.append("{}")
  1276→            combined = " ".join(parts)
  1277→            sql_kw = re.compile(
  1278→                r"\b(?:SELECT|INSERT|UPDATE|DELETE|DROP|UNION|ALTER|CREATE|EXEC)\b", re.I,
  1279→            )
  1280→            if sql_kw.search(combined):
  1281→                self.matched_nodes.append(node)
  1282→        self.generic_visit(node)
  1283→
  1284→    def _line_based_match(self, node: ast.stmt) -> None:
  1285→        """Additional line-based matching for statement nodes."""
  1286→        # Handled in parent methods
  1287→        pass
  1288→
  1289→    def generic_visit(self, node: ast.AST) -> None:
  1290→        if isinstance(node, ast.stmt):
  1291→            # Check for line-based patterns
  1292→            lineno = getattr(node, "lineno", 0)
  1293→            # debug_mode
  1294→            if self.category == "debug_mode" and lineno > 0:
  1295→                pass  # Handled by parent
  1296→        super().generic_visit(node)

  1302→def _get_call_name(node: ast.Call) -> str:
  1303→    """Get the simple function name from a Call node."""
  1304→    func = node.func
  1305→    if isinstance(func, ast.Name):
  1306→        return func.id
  1307→    if isinstance(func, ast.Attribute):
  1308→        return func.attr
  1309→    return ""
  1310→
  1311→
  1312→def _get_call_full_name(node: ast.Call) -> str:
  1313→    """Get the full dotted function name from a Call node."""
  1314→    func = node.func
  1315→    if isinstance(func, ast.Name):
  1316→        return func.id
  1317→    if isinstance(func, ast.Attribute):
  1318→        parts = [func.attr]
  1319→        current = func.value
  1320→        while isinstance(current, ast.Attribute):
  1321→            parts.append(current.attr)
  1322→            current = current.value
  1323→        if isinstance(current, ast.Name):
  1324→            parts.append(current.id)
  1325→        parts.reverse()
  1326→        return ".".join(parts)
  1327→    return ""

  1335→class FixStore:
  1336→    """Persistent storage for fix proposals.
  1337→
  1338→    Stores proposals in ``RECONPRO_HOME/memory/fix_proposals.json``.
  1339→    Tracks the full lifecycle: proposed → approved → applied → verified.
  1340→    """
  1341→
  1342→    def __init__(self, store_path: Optional[Path] = None) -> None:
  1343→        self._path = store_path or (MEMORY_DIR / "fix_proposals.json")
  1344→        self._proposals: Dict[str, FixProposal] = {}
  1345→        self._history: List[Dict[str, Any]] = []
  1346→        self._load()
  1347→
  1348→    # ── Persistence ───────────────────────────────────────────────────
  1349→
  1350→    def _ensure_dir(self) -> None:
  1351→        """Ensure the storage directory exists."""
  1352→        try:
  1353→            self._path.parent.mkdir(parents=True, exist_ok=True)
  1354→        except OSError as exc:
  1355→            logger.error("Cannot create fix store directory %s: %s", self._path.parent, exc)
  1356→
  1357→    def _load(self) -> None:
  1358→        """Load proposals from disk."""
  1359→        if not self._path.exists():
  1360→            self._proposals = {}
  1361→            self._history = []
  1362→            return
  1363→        try:
  1364→            raw = self._path.read_text(encoding="utf-8")
  1365→            data = json.loads(raw)
  1366→            proposals_data = data.get("proposals", {})
  1367→            self._proposals = {
  1368→                pid: FixProposal.from_dict(pdata)
  1369→                for pid, pdata in proposals_data.items()
  1370→            }
  1371→            self._history = data.get("history", [])
  1372→        except (json.JSONDecodeError, OSError, KeyError) as exc:
  1373→            logger.warning("Failed to load fix store from %s: %s", self._path, exc)
  1374→            self._proposals = {}
  1375→            self._history = []
  1376→
  1377→    def _save(self) -> None:
  1378→        """Persist proposals to disk."""
  1379→        self._ensure_dir()
  1380→        try:
  1381→            data = {
  1382→                "proposals": {
  1383→                    pid: p.to_dict() for pid, p in self._proposals.items()
  1384→                },
  1385→                "history": self._history[-500:],  # Keep last 500 history entries
  1386→                "updated_at": datetime.now(timezone.utc).isoformat(),
  1387→            }
  1388→            tmp_path = self._path.with_suffix(".tmp")
  1389→            tmp_path.write_text(json.dumps(data, indent=2, ensure_ascii=False), encoding="utf-8")
  1390→            tmp_path.replace(self._path)
  1391→        except OSError as exc:
  1392→            logger.error("Failed to save fix store to %s: %s", self._path, exc)
  1393→

  1397→        """Add a new proposal to the store."""
  1398→        self._proposals[proposal.id] = proposal
  1399→        self._add_history_entry("added", proposal)
  1400→        self._save()

  1406→    def update(self, proposal: FixProposal) -> None:
  1407→        """Update an existing proposal."""
  1408→        if proposal.id in self._proposals:
  1409→            old_status = self._proposals[proposal.id].status
  1410→            self._proposals[proposal.id] = proposal
  1411→            if old_status != proposal.status:
  1412→                self._add_history_entry(f"status_changed:{old_status}->{proposal.status}", proposal)
  1413→            self._save()
  1414→        else:
  1415→            logger.warning("Attempted to update non-existent proposal: %s", proposal.id)

  1421→            self._add_history_entry("removed", proposal)
  1422→            self._save()

  1442→    def approve(self, proposal_id: str) -> bool:
  1443→        """Approve a proposal. Returns True if successful."""
  1444→        proposal = self._proposals.get(proposal_id)
  1445→        if proposal is None:
  1446→            return False
  1447→        if proposal.status != ProposalStatus.PROPOSED.value:
  1448→            logger.warning("Cannot approve proposal %s in status %s", proposal_id, proposal.status)
  1449→            return False

  1461→            return False

  1516→# AutoFixEngine — Top-level orchestrator
  1517→# ═══════════════════════════════════════════════════════════════════════════
  1518→
  1519→
  1520→class AutoFixEngine:
  1521→    """Top-level engine that analyses scan results and proposes automated fixes.
  1522→
  1523→    The engine is **dry-run by default** — it only *proposes* fixes.
  1524→    Applying a fix requires explicit approval via :meth:`apply_fix` with
  1525→    ``dry_run=False``.
  1526→
  1527→    Usage::
  1528→
  1529→        engine = AutoFixEngine()
  1530→        proposals = engine.propose_fixes(scan_result)
  1531→        for p in proposals:
  1532→            print(f"[{p.severity}] {p.title} (confidence={p.confidence})")
  1533→
  1534→        # Apply a specific approved fix
  1535→        result = engine.apply_fix(proposals[0], dry_run=False)
  1536→        print(result)
  1537→    """
  1538→
  1539→    def __init__(
  1540→        self,
  1541→        store: Optional[FixStore] = None,
  1542→        dry_run: bool = True,
  1543→    ) -> None:
  1544→        self._analyzer = FixAnalyzer()
  1545→        self._store = store or FixStore()
  1546→        self._dry_run = dry_run
  1547→        logger.info("AutoFixEngine initialised (dry_run=%s)", dry_run)
  1548→
  1549→    # ── Public API ────────────────────────────────────────────────────
  1550→
  1551→    def analyze_issue(self, issue: Dict[str, Any]) -> Optional[FixProposal]:
  1552→        """Analyse a single issue and return a fix proposal."""
  1553→        try:
  1554→            proposal = self._analyzer.analyze_issue(issue)
  1555→            if proposal is not None:
  1556→                self._store.add(proposal)
  1557→                logger.info(
  1558→                    "Proposed fix %s: %s (confidence=%.2f)",
  1559→                    proposal.id, proposal.title, proposal.confidence,
  1560→                )
  1561→            return proposal
  1562→        except Exception as exc:
  1563→            logger.error("Failed to analyze issue: %s", exc, exc_info=True)
  1564→            return None
  1565→
  1566→    def propose_fixes(self, scan_result: Dict[str, Any]) -> List[FixProposal]:
  1567→        """Analyse a full scan result and propose fixes for all findings.
  1568→
  1569→        The scan_result dict can contain:
  1570→        - ``findings``: List of finding dicts (from http_layer.Finding.to_dict())
  1571→        - ``diagnostics``: Dict from diagnostics.run_diagnostics()
  1572→        - ``config_issues``: List from diagnostics.validate_config()
  1573→        - ``cross_validation``: Dict from cross_validator
  1574→
  1575→        Any combination of the above keys is supported.
  1576→        """
  1577→        proposals: List[FixProposal] = []
  1578→
  1579→        # 1. Process scan findings
  1580→        findings = scan_result.get("findings", [])
  1581→        if findings:
  1582→            proposals.extend(self._process_findings(findings))
  1583→
  1584→        # 2. Process diagnostics failures
  1585→        diagnostics = scan_result.get("diagnostics", {})
  1586→        if diagnostics:
  1587→            proposals.extend(self._process_diagnostics(diagnostics))
  1588→
  1589→        # 3. Process config issues
  1590→        config_issues = scan_result.get("config_issues", [])
  1591→        if config_issues:
  1592→            proposals.extend(self._process_config_issues(config_issues))
  1593→
  1594→        # 4. Process cross-validation mismatches
  1595→        cv = scan_result.get("cross_validation", {})
  1596→        if cv:
  1597→            proposals.extend(self._process_cross_validation(cv))
  1598→
  1599→        logger.info(
  1600→            "Generated %d fix proposals from scan result (%d findings, %d diagnostics, %d config, %d cv)",
  1601→            len(proposals),
  1602→            len(findings),
  1603→            len(diagnostics.get("checks", [])) if diagnostics else 0,
  1604→            len(config_issues),
  1605→            len(cv.get("mismatches", [])) if cv else 0,
  1606→        )
  1607→
  1608→        return proposals
  1609→
  1610→    def propose_code_fixes(
  1611→        self,
  1612→        file_path: str,
  1613→        issues: List[Dict[str, Any]],
  1614→    ) -> List[FixProposal]:
  1615→        """Propose code-level fixes for issues in a specific file."""
  1616→        try:
  1617→            proposals = self._analyzer.propose_code_fixes(file_path, issues)
  1618→            for p in proposals:

  1621→        except Exception as exc:

  1625→    def propose_config_fixes(
  1626→        self,
  1627→        config_issues: List[Dict[str, Any]],
  1628→    ) -> List[FixProposal]:
  1629→        """Propose configuration fixes."""
