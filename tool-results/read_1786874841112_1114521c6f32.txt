    21→    21→  return new Promise((resolve) => {
    22→    22→    const timer = setTimeout(() => { try { s.destroy(); } catch {} resolve(false); }, timeoutMs);
    23→    23→    const s = net.createConnection({ host, port, timeout: timeoutMs });
    24→    24→    s.on('connect', () => { clearTimeout(timer); try { s.destroy(); } catch {} resolve(true); });

    26→    26→    s.on('error', () => { clearTimeout(timer); resolve(false); });
    27→    27→  });

    30→    30→function tcpBannerGrab(host: string, port: number, timeoutMs: number, maxBytes = 500): Promise<string> {

    45→    45→async function handleCurl(cmd: string, timeout: number): Promise<string> {
    46→    46→  const isHead = /\b-sI\b|\b-I\b/.test(cmd);
    47→    47→  const followRedirects = /\b-L\b/.test(cmd);
    48→    48→  const timeoutMatch = cmd.match(/--max-time\s+(\d+)/);
    49→    49→  const timeoutSec = timeoutMatch ? parseInt(timeoutMatch[1]) : 8;
    50→    50→  const headers: Record<string, string> = { 'User-Agent': 'Mozilla/5.0' };
    51→    51→  const headerRegex = /-H\s+["']([^"']+)["']/g;
    52→    52→  let m: RegExpExecArray | null;
    53→    53→  while ((m = headerRegex.exec(cmd)) !== null) {
    54→    54→    const ci = m[1].indexOf(':');
    55→    55→    if (ci > 0) headers[m[1].substring(0, ci).trim()] = m[1].substring(ci + 1).trim();
    56→    56→  }
    57→    57→  const urlMatch = cmd.match(/(?:https?:\/\/[^\s"'`]+)/);
    58→    58→  const url = urlMatch ? urlMatch[0] : '';
    59→    59→  if (!url) return '';
    60→    60→
    61→    61→  try {
    62→    62→    if (isHead && !followRedirects) {
    63→    63→      const res = await safeFetch(url, { method: 'HEAD', headers, followRedirects: false, timeout: timeoutSec * 1000 });
    64→    64→      if (!res.ok && res.status === 403) return '';
    65→    65→      let out = `HTTP/1.1 ${res.status}\n`;
    66→    66→      for (const [k, v] of Object.entries(res.headers)) { out += `${k}: ${v}\n`; }
    67→    67→      return out;
    68→    68→    }
    69→    69→    if (isHead && followRedirects) {
    70→    70→      let cur = url; let out = '';
    71→    71→      for (let i = 0; i < 5; i++) {
    72→    72→        const res = await safeFetch(cur, { method: 'HEAD', headers, followRedirects: false, timeout: timeoutSec * 1000 });
    73→    73→        if (!res.ok && res.status === 403) return '';
    74→    74→        out += `HTTP/1.1 ${res.status}\n`;
    75→    75→        for (const [k, v] of Object.entries(res.headers)) { out += `${k}: ${v}\n`; }
    76→    76→        out += '\n';
    77→    77→        if (res.status >= 300 && res.status < 400) {
    78→    78→          const loc = res.headers['location']; if (loc) { cur = new URL(loc, cur).href; continue; }
    79→    79→        }
    80→    80→        break;
    81→    81→      }
    82→    82→      return out;
    83→    83→    }
    84→    84→    const res = await safeFetch(url, { method: 'GET', headers, followRedirects, timeout: timeoutSec * 1000 });
    85→    85→    if (!res.ok && res.status === 403) return '';
    86→    86→    return res.text;
    87→    87→  } catch { return ''; }
    88→    88→}

    90→    90→async function handleDig(cmd: string, timeout: number): Promise<string> {
    91→    91→  if (cmd.includes('axfr')) return '';
    92→    92→  const isPtr = cmd.includes('-x');
    93→    93→  let cleaned = cmd.replace(/\bdig\b/, '').replace(/\+short\b/g, '').replace(/\+time=\d+/g, '').replace(/\+tries=\d+/g, '').replace(/@\S+/g, '').replace(/\+noall\b/g, '').replace(/\+answer\b/g, '').replace(/2>\/>\/dev\/null/g, '').replace(/2>&1/g, '').trim();
    94→    94→  if (isPtr) {
    95→    95→    const pm = cleaned.match(/-x\s+(\S+)/);
    96→    96→    const ip = pm ? pm[1] : '';
    97→    97→    if (!ip) return '';
    98→    98→    try { return (await dns.resolvePtr(ip)).join('\n'); } catch { return ''; }
    99→    99→  }
   100→   100→  const tokens = cleaned.split(/\s+/).filter(Boolean);
   101→   101→  if (tokens.length === 0) return '';
   102→   102→  const recordTypes = ['A','AAAA','NS','CNAME','MX','TXT','PTR','SOA','SRV','ANY'];
   103→   103→  let domain: string; let type = 'A';
   104→   104→  const last = tokens[tokens.length - 1].toUpperCase();
   105→   105→  if (tokens.length >= 2 && recordTypes.includes(last)) {
   106→   106→    type = last; domain = tokens.slice(0, -1).join(' ');
   107→   107→  } else {
   108→   108→    domain = tokens.join(' ');
   109→   109→  }
   110→   110→  if (!domain) return '';
   111→   111→  try {
   112→   112→    let result: string[] = [];
   113→   113→    switch (type) {
   114→   114→      case 'A': result = await dns.resolve4(domain); break;
   115→   115→      case 'NS': result = await dns.resolveNs(domain); break;
   116→   116→      case 'CNAME': result = await dns.resolveCname(domain); break;
   117→   117→      case 'TXT': { const r = await dns.resolveTxt(domain); result = r.map(x => x.join('')); break; }
   118→   118→      case 'PTR': result = await dns.resolvePtr(domain); break;
   119→   119→      case 'ANY': {
   120→   120→        const r = await dns.resolveAny(domain);
   121→   121→        result = r.map(x => { switch (x.type) { case 'A': case 'AAAA': return x.address; case 'MX': return `${x.exchange} ${x.priority}`; case 'TXT': return x.entries?.join('') || ''; case 'NS': case 'CNAME': case 'PTR': return x.value; default: return JSON.stringify(x); } });
   122→   122→        break;
   123→   123→      }
   124→   124→      default: result = await dns.resolve4(domain); break;
   125→   125→    }
   126→   126→    return result.join('\n');
   127→   127→  } catch { return ''; }
   128→   128→}

   130→   130→async function handleOpenssl(cmd: string, timeout: number): Promise<string> {
   131→   131→  const cm = cmd.match(/-connect\s+(\S+:(\d+))/);
   132→   132→  const sm = cmd.match(/-servername\s+(\S+)/);
   133→   133→  const hasTlsextdebug = cmd.includes('-tlsextdebug');
   134→   134→  const hasStatus = cmd.includes('-status');
   135→   135→  let host = '', port = 443;
   136→   136→  if (cm) { const [h, p] = cm[1].split(':'); host = h; port = parseInt(p) || 443; }
   137→   137→  if (!host) return '';
   138→   138→  const servername = sm ? sm[1] : host;
   139→   139→  return new Promise((resolve) => {
   140→   140→    const timer = setTimeout(() => { try { s.destroy(); } catch {} resolve(''); }, timeout);
   141→   141→    const opts: tls.ConnectionOptions = { host, port, servername, rejectUnauthorized: false };
   142→   142→    const s = tls.connect(opts, () => {
   143→   143→      const protocol = s.getProtocol() || '';
   144→   144→      const cipher = s.getCipher();
   145→   145→      const cert = s.getPeerCertificate();
   146→   146→      let out = '';
   147→   147→      out += `    Protocol  : ${protocol}\n`;
   148→   148→      const cipherName = protocol === 'TLSv1.3' ? `ECDHE-RSA-${cipher.name}` : cipher.name;
   149→   149→      out += `    Cipher    : ${cipherName}\n`;
   150→   150→      if (cert) {
   151→   151→        if (cert.valid_from) out += `notBefore=${cert.valid_from}\n`;
   152→   152→        if (cert.valid_to) out += `notAfter=${cert.valid_to}\n`;
   153→   153→      }
   154→   154→      if (hasTlsextdebug) { /* heartbeat not directly exposed in Node.js TLS */ }
   155→   155→      if (hasStatus) { /* OCSP stapling not directly exposed */ }
   156→   156→      clearTimeout(timer); try { s.destroy(); } catch {} resolve(out);
   157→   157→    });
   158→   158→    s.on('timeout', () => { clearTimeout(timer); try { s.destroy(); } catch {} resolve(''); });
   159→   159→    s.on('error', () => { clearTimeout(timer); resolve(''); });
   160→   160→  });
   161→   161→}

   176→   176→function applyPipe(input: string, pipeCmd: string): string {
   177→   177→  if (pipeCmd.includes('openssl x509')) {
   178→   178→    return input.split('\n').filter(l => l.includes('notAfter=') || l.includes('notBefore=')).join('\n');
   179→   179→  }
   180→   180→  const headBytes = pipeCmd.match(/head\s+-c\s+(\d+)/);
   181→   181→  if (headBytes) return input.substring(0, parseInt(headBytes[1]));
   182→   182→  const headLines = pipeCmd.match(/head\s+-(\d+)/);
   183→   183→  if (headLines) return input.split('\n').slice(0, parseInt(headLines[1])).join('\n');
   184→   184→  const grepMatch = pipeCmd.match(/^grep\s+(-\w+\s+)?(.+)$/);
   185→   185→  if (grepMatch) {
   186→   186→    const flags = grepMatch[1]?.trim() || '';
   187→   187→    let pattern = grepMatch[2].trim();
   188→   188→    if ((pattern.startsWith('"') && pattern.endsWith('"')) || (pattern.startsWith("'") && pattern.endsWith("'"))) pattern = pattern.slice(1, -1);
   189→   189→    const isOI = flags.includes('o');
   190→   190→    const isI = flags.includes('i');
   191→   191→    try {
   192→   192→      if (isOI) {
   193→   193→        const regex = new RegExp(pattern, isI ? 'gi' : 'g');
   194→   194→        const matches: string[] = []; let match;
   195→   195→        while ((match = regex.exec(input)) !== null) { matches.push(match[0]); if (matches.length > 100) break; }
   196→   196→        return matches.join('\n');
   197→   197→      }
   198→   198→      const regex = new RegExp(pattern, isI ? 'i' : '');
   199→   199→      return input.split('\n').filter(l => regex.test(l)).join('\n');
   200→   200→    } catch { return ''; }
   201→   201→  }
   202→   202→  return input;

   205→   205→async function run(cmd: string, timeout = 8000): Promise<string> {
   206→   206→  try {
   207→   207→    if (cmd.includes('/dev/tcp')) {
   208→   208→      const m = cmd.match(/\/dev\/tcp\/([^/]+)\/(\d+)/);
   209→   209→      if (m) { const open = await tcpProbe(m[1], parseInt(m[2]), Math.min(timeout, 3000)); return cmd.includes('OPEN') && cmd.includes('CLOSED') ? (open ? 'OPEN' : 'CLOSED') : ''; }
   210→   210→      return '';
   211→   211→    }
   212→   212→    let effectiveCmd = cmd.replace(/^echo\s*(""\s*)?\|\s*/, '').trim();
   213→   213→    effectiveCmd = effectiveCmd.replace(/\|\|/g, '\x00\x00');
   214→   214→    const segments: string[] = []; let cur = ''; let inQ = false; let qCh = '';
   215→   215→    for (let i = 0; i < effectiveCmd.length; i++) {
   216→   216→      const ch = effectiveCmd[i];
   217→   217→      if (inQ) { cur += ch; if (ch === qCh) inQ = false; }
   218→   218→      else if (ch === '"' || ch === "'") { inQ = true; qCh = ch; cur += ch; }
   219→   219→      else if (ch === '|') { segments.push(cur.trim()); cur = ''; }
   220→   220→      else { cur += ch; }
   221→   221→    }
   222→   222→    segments.push(cur.trim());
   223→   223→    const restored = segments.map(s => s.replace(/\x00\x00/g, '||'));
   224→   224→    const mainCmd = restored[0]; const pipes = restored.slice(1);
   225→   225→    let output = '';
   226→   226→    if (mainCmd.includes('curl')) output = await handleCurl(mainCmd, timeout);
   227→   227→    else if (mainCmd.includes('dig')) output = await handleDig(mainCmd, timeout);
   228→   228→    else if (mainCmd.includes('openssl')) output = await handleOpenssl(mainCmd, timeout);
   229→   229→    else if (mainCmd.includes('nc ') || mainCmd.includes('/dev/tcp')) output = await handleNetcat(mainCmd, timeout);
   230→   230→    else if (mainCmd.includes('whois')) return '';
   231→   231→    else return '';
   232→   232→    for (const pipe of pipes) output = applyPipe(output, pipe);
   233→   233→    return output.trim();
   234→   234→  } catch { return ''; }
   235→   235→}

   341→   341→// 1. BANNER GRABBING + CVE MATCHING

   344→   344→async function bannerGrab(domain: string, ip: string | null): Promise<{ banners: Array<{ port: number; banner: string; service: string }>; cveMatches: CVEEntry[] }> {
   345→   345→  const target = ip || domain;
   346→   346→  const ports = [21, 22, 25, 80, 110, 143, 443, 445, 993, 995, 1433, 1521, 3306, 3389, 5432, 5900, 6379, 8080, 8443, 8888, 9090, 27017];
   347→   347→  const banners: Array<{ port: number; banner: string; service: string }> = [];
   348→   348→
   349→   349→  for (const port of ports) {
   350→   350→    const banner = await run(
   351→   351→      `echo "" | timeout 4 nc -w3 ${target} ${port} 2>/dev/null | head -c 500`,
   352→   352→      6000
   353→   353→    );
   354→   354→    if (banner && banner.length > 3) {
   355→   355→      const service = detectService(port, banner);
   356→   356→      banners.push({ port, banner: banner.replace(/[\x00-\x1f\x7f]/g, '.').substring(0, 200), service });
   357→   357→    }
   358→   358→  }
   359→   359→
   360→   360→  // HTTP banner via curl
   361→   361→  const httpBanner = await run(`curl -sI --max-time 6 https://${domain} 2>/dev/null | head -20`, 8000);
   362→   362→  if (httpBanner) banners.push({ port: 443, banner: httpBanner.substring(0, 300), service: 'http' });
   363→   363→  const httpBanner2 = await run(`curl -sI --max-time 6 http://${domain} 2>/dev/null | head -20`, 8000);
   364→   364→  if (httpBanner2) banners.push({ port: 80, banner: httpBanner2.substring(0, 300), service: 'http' });
   365→   365→
   366→   366→  // Match banners against CVE database
   367→   367→  const combined = banners.map(b => b.banner).join(' ').toLowerCase();
   368→   368→  const cveMatches = CVE_DATABASE.filter(cve => {
   369→   369→    return cve.affected.some(a => combined.includes(a)) && cve.versionPattern.test(combined);
   370→   370→  });
   371→   371→
   372→   372→  return { banners, cveMatches };
   373→   373→}
   374→   374→
   375→   375→function detectService(port: number, banner: string): string {
   376→   376→  const b = banner.toLowerCase();
   377→   377→  if (port === 22 || b.includes('ssh')) return 'openssh';
   378→   378→  if (port === 21 || b.includes('ftp')) return 'ftp';
   379→   379→  if (port === 25 || b.includes('smtp') || b.includes('postfix') || b.includes('esmtp')) return 'postfix';
   380→   380→  if (port === 80 || port === 8080 || port === 8443 || b.includes('http')) return 'http';
   381→   381→  if (port === 110 || b.includes('pop3')) return 'pop3';
   382→   382→  if (port === 143 || b.includes('imap')) return 'imap';
   383→   383→  if (port === 443 || b.includes('ssl') || b.includes('tls')) return 'https';
   384→   384→  if (port === 3306 || b.includes('mysql')) return 'mysql';
   385→   385→  if (port === 5432 || b.includes('postgresql') || b.includes('postgres')) return 'postgresql';
   386→   386→  if (port === 6379 || b.includes('redis')) return 'redis';
   387→   387→  if (port === 27017 || b.includes('mongodb')) return 'mongodb';
   388→   388→  if (b.includes('nginx')) return 'nginx';
   389→   389→  if (b.includes('apache')) return 'apache';
   390→   390→  if (b.includes('tomcat')) return 'tomcat';
   391→   391→  return 'unknown';
   392→   392→}

   554→   554→
   555→   555→  // Mixed content detection
   556→   556→  const mixedContent = await run(
   557→   557→    `curl -s --max-time 8 https://${domain} 2>/dev/null | grep -oiE 'src="http://[^"]+"|href="http://[^"]+"|url\("http://[^)]+"\)' | head -5`,
   558→   558→    10000
   559→   559→  );
   560→   560→  if (mixedContent) {
   561→   561→    findings.push({
   562→   562→      title: `Mixed Content — ${mixedContent.split('\n').filter(Boolean).length} HTTP Resources on HTTPS Page`,
   563→   563→      severity: 'low', category: 'http-vuln',
   564→   564→      description: 'HTTPS page loads resources via HTTP. Browsers will show warnings, and MITM can inject content into the insecure resources.',
   565→   565→      evidence: mixedContent.trim().substring(0, 150),
   566→   566→      asset: domain,
   567→   567→    });
   568→   568→  }
   569→   569→
   570→   570→  return findings;
   571→   571→}

   573→   573→// ══════════════════════════════════════════════════════════════════════════════
   574→   574→// 3. SSL/TLS ATTACK VECTOR SCANNER
   575→   575→// ══════════════════════════════════════════════════════════════════════════════
   576→   576→
   577→   577→async function scanSSLVulns(domain: string): Promise<Finding[]> {
   578→   578→  const findings: Finding[] = [];
   579→   579→
   580→   580→  // Full SSL scan with all cipher suites
   581→   581→  const sslFull = await run(`echo | openssl s_client -connect ${domain}:443 -servername ${domain} 2>/dev/null`, 10000);
   582→   582→  const sslLines = sslFull.split('\n').map(l => l.trim()).filter(Boolean);
   583→   583→
   584→   584→  // Protocol version checks
   585→   585→  const protocolLine = sslLines.find(l => l.startsWith('Protocol'));
   586→   586→  const proto = protocolLine || '';
   587→   587→
   588→   588→  if (proto.includes('TLSv1.0')) {
   589→   589→    findings.push({
   590→   590→      title: 'BEAST Attack Vulnerable — TLS 1.0 Active',
   591→   591→      severity: 'high', category: 'ssl-vuln',
   592→   592→      description: 'TLS 1.0 is vulnerable to BEAST (Browser Exploit Against SSL/TLS). Block cipher attack can decrypt communication blocks. PCI-DSS prohibits TLS 1.0.',
   593→   593→      evidence: proto,
   594→   594→      asset: `${domain}:443`,
   595→   595→    });
   596→   596→  }
   597→   597→  if (proto.includes('TLSv1.1')) {
   598→   598→    findings.push({
   599→   599→      title: 'Deprecated TLS 1.1 in Use',
   600→   600→      severity: 'medium', category: 'ssl-vuln',
   601→   601→      description: 'TLS 1.1 is deprecated (RFC 8996). Remove support to prevent downgrade attacks.',
   602→   602→      evidence: proto,
   603→   603→      asset: `${domain}:443`,
   604→   604→    });
   605→   605→  }
   606→   606→  if (proto.includes('SSLv3')) {
   607→   607→    findings.push({
   608→   608→      title: 'CRITICAL: POODLE — SSLv3 Detected',
   609→   609→      severity: 'critical', category: 'ssl-vuln',
   610→   610→      description: 'SSLv3 is extremely vulnerable to POODLE (Padding Oracle On Downgraded Legacy Encryption). Complete decryption of encrypted sessions possible.',
   611→   611→      evidence: proto,
   612→   612→      asset: `${domain}:443`,
   613→   613→    });
   614→   614→  }
   615→   615→  if (proto.includes('SSLv2')) {
   616→   616→    findings.push({
   617→   617→      title: 'CRITICAL: DROWN Attack — SSLv2 Detected',
   618→   618→      severity: 'critical', category: 'ssl-vuln',
   619→   619→      description: 'SSLv2 enables DROWN (Decrypting RSA with Obsolete and Weakened eNcryption). Cross-protocol attack can decrypt TLS connections.',
   620→   620→      evidence: proto,
   621→   621→      asset: `${domain}:443`,
   622→   622→    });
   623→   623→  }
   624→   624→
   625→   625→  // Cipher analysis
   626→   626→  const cipherLine = sslLines.find(l => l.includes('Cipher'));
   627→   627→  const cipher = cipherLine || '';
   628→   628→
   629→   629→  // Check for weak ciphers by testing multiple
   630→   630→  const weakCiphers = ['RC4', 'DES', '3DES', 'NULL', 'EXPORT', 'MD5', 'RC2', 'IDEA'];
   631→   631→  const cipherSuite = await run(
   632→   632→    `echo | openssl s_client -connect ${domain}:443 -servername ${domain} -cipher 'ALL:eNULL' 2>/dev/null | grep "Cipher"`,
   633→   633→    10000
   634→   634→  );
   635→   635→
   636→   636→  for (const weak of weakCiphers) {
   637→   637→    if (cipherSuite.toUpperCase().includes(weak)) {
   638→   638→      findings.push({
   639→   639→        title: `CRITICAL: Weak Cipher ${weak} Accepted`,
   640→   640→        severity: 'critical', category: 'ssl-vuln',
   641→   641→        description: `Server accepts ${weak} cipher suite. ${weak === 'RC4' ? 'RC4 has known statistical biases exploitable since 2013.' : weak === 'NULL' ? 'NULL cipher means NO encryption — traffic is plaintext.' : weak === 'EXPORT' ? 'EXPORT-grade ciphers are trivially breakable (40-bit keys).' : `${weak} is cryptographically broken and must be disabled.`}`,
   642→   642→        evidence: `Cipher: ${cipherSuite.trim()}`,
   643→   643→        asset: `${domain}:443`,
   644→   644→      });
   645→   645→      break;
   646→   646→    }
   647→   647→  }
   648→   648→
   649→   649→  if (cipher.includes('CBC') && proto.includes('TLSv1')) {
   650→   650→    findings.push({
   651→   651→      title: 'BEAST Susceptible — CBC Mode + TLS 1.0',
   652→   652→      severity: 'medium', category: 'ssl-vuln',
   653→   653→      description: 'CBC cipher suites with TLS 1.0 are vulnerable to BEAST attack. Use GCM-mode ciphers or upgrade to TLS 1.2+.',
   654→   654→      evidence: `${proto} | ${cipher}`,
   655→   655→      asset: `${domain}:443`,
   656→   656→    });
   657→   657→  }
   658→   658→
   659→   659→  // Heartbleed check (CVE-2014-0160)
   660→   660→  const heartbleedCheck = await run(
   661→   661→    `echo | openssl s_client -connect ${domain}:443 -servername ${domain} -tlsextdebug 2>&1 | grep -i "heartbeat"`,
   662→   662→    8000
   663→   663→  );
   664→   664→  if (heartbleedCheck) {
   665→   665→    findings.push({
   666→   666→      title: 'Heartbleed Vulnerability Possible (CVE-2014-0160)',
   667→   667→      severity: 'critical', category: 'ssl-vuln',
   668→   668→      description: 'Server supports heartbeat extension. If running vulnerable OpenSSL (1.0.1-1.0.1f), up to 64KB of server memory can be leaked per request — including private keys, passwords, session tokens.',
   669→   669→      evidence: `Heartbeat extension: ${heartbleedCheck.trim()}`,
   670→   670→      asset: `${domain}:443`,
   671→   671→    });
   672→   672→  }
   673→   673→
   674→   674→  // Certificate issues
   675→   675→  const certDates = await run(
   676→   676→    `echo | openssl s_client -connect ${domain}:443 -servername ${domain} 2>/dev/null | openssl x509 -noout -dates 2>/dev/null`,
   677→   677→    10000
   678→   678→  );
   679→   679→  if (certDates) {
   680→   680→    const notAfter = certDates.match(/notAfter=(.+)/);
   681→   681→    if (notAfter) {
   682→   682→      const expiry = new Date(notAfter[1].trim());
   683→   683→      const now = new Date();
   684→   684→      const daysLeft = Math.floor((expiry.getTime() - now.getTime()) / 86400000);
   685→   685→      if (daysLeft < 0) {
   686→   686→        findings.push({
   687→   687→          title: 'CRITICAL: SSL Certificate EXPIRED',
   688→   688→          severity: 'critical', category: 'ssl-vuln',
   689→   689→          description: `Certificate expired ${Math.abs(daysLeft)} days ago. Visitors see security warnings. Connections may be intercepted.`,
   690→   690→          evidence: `notAfter: ${notAfter[1].trim()} | Expired ${Math.abs(daysLeft)} days ago`,
   691→   691→          asset: `${domain}:443`,
   692→   692→        });
   693→   693→      } else if (daysLeft < 30) {
   694→   694→        findings.push({
   695→   695→          title: `SSL Certificate Expiring in ${daysLeft} Days`,
   696→   696→          severity: 'high', category: 'ssl-vuln',
   697→   697→          description: `Certificate expires soon. Service disruption and security warnings imminent.`,
   698→   698→          evidence: `notAfter: ${notAfter[1].trim()} | ${daysLeft} days remaining`,
   699→   699→          asset: `${domain}:443`,
   700→   700→        });
   701→   701→      }
   702→   702→    }
   703→   703→  }
   704→   704→
   705→   705→  // OCSP stapling
   706→   706→  const ocsp = await run(
   707→   707→    `echo | openssl s_client -connect ${domain}:443 -servername ${domain} -status 2>/dev/null | grep -i "OCSP"`,
   708→   708→    8000
   709→   709→  );
   710→   710→  if (!ocsp) {
   711→   711→    findings.push({
   712→   712→      title: 'OCSP Stapling Not Enabled',
   713→   713→      severity: 'low', category: 'ssl-vuln',
   714→   714→      description: 'Without OCSP stapling, browsers must contact CA separately to verify certificate revocation, adding latency and privacy exposure.',
   715→   715→      evidence: 'No OCSP response in handshake',
   716→   716→      asset: `${domain}:443`,
   717→   717→    });
   718→   718→  }
   719→   719→
   720→   720→  // Perfect Forward Secrecy
   721→   721→  const pfsCheck = await run(
   722→   722→    `echo | openssl s_client -connect ${domain}:443 -servername ${domain} 2>/dev/null | grep -iE "ECDHE-RSA|ECDHE-ECDSA|DHE-RSA|DHE-DSS"`,
   723→   723→    8000
   724→   724→  );
   725→   725→  if (!pfsCheck) {
   726→   726→    findings.push({
   727→   727→      title: 'Perfect Forward Secrecy — Not Supported',
   728→   728→      severity: 'medium', category: 'ssl-vuln',
   729→   729→      description: 'Without PFS, compromise of server private key allows decryption of ALL past sessions captured via passive surveillance.',
   730→   730→      evidence: 'No ECDHE/DHE cipher negotiated',
   731→   731→      asset: `${domain}:443`,
   732→   732→    });
   733→   733→  }
   734→   734→
   735→   735→  return findings;
   736→   736→}

   738→   738→// ══════════════════════════════════════════════════════════════════════════════
   739→   739→// 4. DNS VULNERABILITY ASSESSMENT
   740→   740→// ══════════════════════════════════════════════════════════════════════════════
   741→   741→
   742→   742→async function scanDNSVulns(domain: string): Promise<Finding[]> {
   743→   743→  const findings: Finding[] = [];
   744→   744→
   745→   745→  // Zone transfer (AXFR) attempt
   746→   746→  const nsRecords = await run(`dig +short +time=3 ${domain} NS`, 5000);
   747→   747→  const nsList = nsRecords.split('\n').filter(Boolean);
   748→   748→  for (const ns of nsList.slice(0, 3)) {
   749→   749→    const axfrResult = await run(`dig axfr ${domain} @${ns} +time=5 +tries=1 2>/dev/null`, 7000);
   750→   750→    if (axfrResult && axfrResult.includes('ANSWER SECTION') && axfrResult.split('\n').length > 5) {
   751→   751→      findings.push({
   752→   752→        title: 'CRITICAL: DNS Zone Transfer (AXFR) Successful',
   753→   753→        severity: 'critical', category: 'dns-vuln',
   754→   754→        description: `Full zone transfer from ${ns} revealed complete DNS records including internal hostnames, IP addresses, and service records. Complete infrastructure mapping possible.`,
   755→   755→        evidence: `dig axfr ${domain} @${ns} → ${axfrResult.split('\n').length} records transferred`,
   756→   756→        asset: ns,
   757→   757→      });
   758→   758→      break;
   759→   759→    }
   760→   760→  }
   761→   761→
   762→   762→  // Subdomain takeover detection (CNAME to dead services)
   763→   763→  const takeoverSubs = ['staging', 'dev', 'blog', 'cdn', 'assets', 'static', 'mail', 'api', 'app', 'test'];
   764→   764→  const takeoverCandidates: string[] = [];
   765→   765→  for (const sub of takeoverSubs) {
   766→   766→    const cname = await run(`dig +short +time=2 ${sub}.${domain} CNAME`, 4000);
   767→   767→    if (cname) {
   768→   768→      const deadServices = ['herokuapp.com', 'cloudfront.net', 'github.io', 's3.amazonaws.com',
   769→   769→        'shopify.com', 'fastly.net', 'cloudwaysapps.com', 'pantheon.io'];
   770→   770→      for (const dead of deadServices) {
   771→   771→        if (cname.toLowerCase().includes(dead)) {
   772→   772→          const subExists = await run(`curl -sI --max-time 4 https://${sub}.${domain} 2>/dev/null | head -1`, 6000);
   773→   773→          if (subExists && !subExists.includes('200') && !subExists.includes('301') && !subExists.includes('302')) {
   774→   774→            takeoverCandidates.push(`${sub}.${domain} → CNAME: ${cname} (${dead})`);
   775→   775→          }
   776→   776→        }
   777→   777→      }
   778→   778→    }
   779→   779→  }
   780→   780→  if (takeoverCandidates.length > 0) {
   781→   781→    findings.push({
   782→   782→      title: `Subdomain Takeover Risk — ${takeoverCandidates.length} candidate(s)`,
   783→   783→      severity: 'high', category: 'dns-vuln',
   784→   784→      description: `Subdomains point to third-party services via CNAME but may not be properly provisioned. If the third-party resource is deleted, an attacker can claim the subdomain and serve malicious content.`,
   785→   785→      evidence: takeoverCandidates.join(' | '),
   786→   786→      asset: domain,
   787→   787→    });
   788→   788→  }
   789→   789→
   790→   790→  // DNS cache snooping
   791→   791→  const cacheResult = await run(`dig +time=3 ${domain} @8.8.8.8 ANY +noall +answer 2>/dev/null`, 5000);
   792→   792→  if (cacheResult && cacheResult.split('\n').length > 8) {
   793→   793→    findings.push({
   794→   794→      title: 'DNS Cache Snooping Possible — Excessive Records Visible',
   795→   795→      severity: 'medium', category: 'dns-vuln',
   796→   796→      description: `Recursive DNS query reveals extensive record types. DNS cache can be probed to determine which domains are recently visited by the resolver's clients.`,
   797→   797→      evidence: `dig ANY ${domain} → ${cacheResult.split('\n').length} records from cache`,
   798→   798→      asset: domain,
   799→   799→    });
   800→   800→  }
   801→   801→
   802→   802→  // DNS rebinding protection check
   803→   803→  const dnsRebindHeaders = await run(`curl -sI --max-time 5 https://${domain} 2>/dev/null | grep -iE "x-dns-prefetch|x-content-type|dns-prefetch"`, 7000);
   804→   804→  if (!dnsRebindHeaders) {
   805→   805→    findings.push({
   806→   806→      title: 'DNS Rebinding — No Protection Detected',
   807→   807→      severity: 'medium', category: 'dns-vuln',
   808→   808→      description: 'No DNS rebinding protection headers. Attackers can bypass Same-Origin Policy by controlling DNS to oscillate between their server and target IP.',
   809→   809→      evidence: 'No DNS rebinding protection headers found',
   810→   810→      asset: domain,
   811→   811→    });
   812→   812→  }
   813→   813→
   814→   814→  return findings;
   815→   815→}
   816→   816→
   817→   817→// ══════════════════════════════════════════════════════════════════════════════
   818→   818→// MAIN POST HANDLER
   819→   819→// ══════════════════════════════════════════════════════════════════════════════
   820→   820→
   821→   821→export async function POST(request: NextRequest) {
   822→   822→  const { error } = await withProtection(request, { requireAuth: true, rateLimit: { maxRequests: 5, windowMs: 60_000 } });
   823→   823→  if (error) return error;
   824→   824→
   825→   825→  const scanStartTime = Date.now();
   826→   826→
   827→   827→  try {
   828→   828→    const body = await request.json();
   829→   829→    const target = body.target?.toString().trim().replace(/^https?:\/\//, '').replace(/\/.*$/, '');
   830→   830→    const sanitized = sanitizeDomain(target);
   831→   831→    if (!sanitized) {
   832→   832→      return NextResponse.json({ error: 'Invalid target domain' }, { status: 400 });
   833→   833→    }
   834→   834→    if (isBlockedDomain(sanitized)) {
   835→   835→      return NextResponse.json({ error: 'Target domain is not permitted' }, { status: 403 });
   836→   836→    }
   837→   837→
   838→   838→    // Resolve IP first
   839→   839→    const ip = await run(`dig +short +time=3 +tries=1 ${sanitized} A`, 5000);
   840→   840→    const mainIp = ip?.split('\n')[0]?.trim() || null;
   841→   841→    if (mainIp && isPrivateIP(mainIp)) {
   842→   842→      return NextResponse.json({ error: 'Target resolves to a private or reserved IP address' }, { status: 403 });
   843→   843→    }
   844→   844→
   845→   845→    // Run all modules in parallel
   846→   846→    const [
   847→   847→      bannerResult,
   848→   848→      httpVulns,
   849→   849→      sslVulns,
   850→   850→      dnsVulns,
   851→   851→    ] = await Promise.all([
   852→   852→      bannerGrab(sanitized, mainIp),
   853→   853→      scanHTTPVulns(sanitized),
   854→   854→      scanSSLVulns(sanitized),
   855→   855→      scanDNSVulns(sanitized),
   856→   856→    ]);
   857→   857→
   858→   858→    // Build findings list (include CVE matches as findings for DB persistence)
   859→   859→    const allFindings: Finding[] = [
   860→   860→      ...httpVulns,
   861→   861→      ...sslVulns,
   862→   862→      ...dnsVulns,
   863→   863→      // Convert CVE matches into findings so they get persisted
   864→   864→      ...bannerResult.cveMatches.map(v => ({
   865→   865→        title: v.title,
   866→   866→        severity: v.severity,
   867→   867→        category: 'vulnerability',
   868→   868→        description: `${v.id} — affects ${v.affected.join(', ')}${v.exploitAvailable === 'weaponized' ? ' (weaponized exploit available)' : ''}`,
   869→   869→        evidence: `Version pattern match via banner grab on ${sanitized}`,
   870→   870→        asset: sanitized,
   871→   871→        cve: v.id,
   872→   872→        cvss: v.cvss,
   873→   873→        remediation: v.remediation,
   874→   874→      })),
   875→   875→    ];
   876→   876→    const c = allFindings.filter(f => f.severity === 'critical').length;
   877→   877→    const h = allFindings.filter(f => f.severity === 'high').length;
   878→   878→    const m = allFindings.filter(f => f.severity === 'medium').length;
   879→   879→    const l = allFindings.filter(f => f.severity === 'low').length;
   880→   880→    const i = allFindings.filter(f => f.severity === 'info').length;
   881→   881→    const riskScore = Math.min(100, Math.round(c * 25 + h * 15 + m * 8 + l * 3 + i * 1));
   882→   882→    const exploitSummary = {
   883→   883→      totalCVEs: bannerResult.cveMatches.length,
   884→   884→      criticalCVEs: bannerResult.cveMatches.filter(v => v.severity === 'critical').length,
   885→   885→      weaponized: bannerResult.cveMatches.filter(v => v.exploitAvailable === 'weaponized').length,
   886→   886→      theoretical: bannerResult.cveMatches.filter(v => v.exploitAvailable === 'theoretical').length,
   887→   887→      avgCVSS: bannerResult.cveMatches.length > 0
   888→   888→        ? Math.round(bannerResult.cveMatches.reduce((a, b) => a + b.cvss, 0) / bannerResult.cveMatches.length * 10) / 10
   889→   889→        : 0,
   890→   890→    };
   891→   891→
   892→   892→    // ── Persist results to database ──────────────────────────────────
   893→   893→    let persistedScanId: string | null = null;
   894→   894→    try {
   895→   895→      const targetRecord = await db.scanTarget.upsert({
   896→   896→        where: { id: sanitized },
   897→   897→        create: {
   898→   898→          id: sanitized,
   899→   899→          domain: sanitized,
   900→   900→          ip: mainIp || null,
   901→   901→        },
   902→   902→        update: { lastScanned: new Date(), ip: mainIp || undefined },
   903→   903→      });
   904→   904→
   905→   905→      const scanRecord = await db.scan.create({
   906→   906→        data: {
   907→   907→          targetId: targetRecord.id,
   908→   908→          scanType: 'vulnerability',
   909→   909→          triggeredBy: 'manual',
   910→   910→          status: 'completed',
   911→   911→          riskScore,
   912→   912→          totalVulns: allFindings.length,
   913→   913→          criticalCount: c,
   914→   914→          highCount: h,
   915→   915→          mediumCount: m,
   916→   916→          lowCount: l,
   917→   917→          infoCount: i,
   918→   918→          startedAt: new Date(scanStartTime),
   919→   919→          completedAt: new Date(),
   920→   920→          duration: Date.now() - scanStartTime,
   921→   921→          findings: {
   922→   922→            create: allFindings.map(f => ({
   923→   923→              title: f.title,
   924→   924→              severity: f.severity,
   925→   925→              category: f.category,
   926→   926→              description: f.description,
   927→   927→              evidence: f.evidence || null,
   928→   928→              asset: f.asset || sanitized,
   929→   929→              remediation: f.remediation || null,
   930→   930→              cve: f.cve || null,
   931→   931→              cvss: f.cvss || null,
   932→   932→            })),
   933→   933→          },
   934→   934→        },
   935→   935→        include: { findings: true, target: true },
   936→   936→      });
   937→   937→
   938→   938→      persistedScanId = scanRecord.id;
   939→   939→    } catch (dbError) {
   940→   940→      console.error('Failed to persist vuln-scan results to database:', dbError);
   941→   941→      // Continue returning results even if DB persistence fails
   942→   942→    }
   943→   943→
   944→   944→    return applySecurityHeaders(NextResponse.json({
   945→   945→      success: true,
   946→   946→      scan: {
   947→   947→        scanId: persistedScanId,
   948→   948→        target,
   949→   949→        ip: mainIp,
   950→   950→        cveFindings: bannerResult.cveMatches.map(v => ({
   951→   951→          cve: v.id, title: v.title, cvss: v.cvss, severity: v.severity,
   952→   952→          affected: v.affected.join(', '), evidence: `Version pattern match via banner grab`,
   953→   953→          remediation: v.remediation, exploitAvailable: v.exploitAvailable, epss: v.epss,
   954→   954→        })),
   955→   955→        httpVulns: httpVulns.map(f => ({
