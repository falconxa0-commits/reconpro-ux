   100→def _aggregate_dread(findings: List[Dict[str, Any]]) -> List[float]:
   101→    """Aggregate DREAD components across all findings.
   102→
   103→    Some modules store dread as a float (average score), others as a dict
   104→    with component breakdowns. We handle both cases.
   105→
   106→    Returns list of 5 floats: [damage, reproducibility, exploitability, affected_users, discoverability]
   107→    Averaged across all findings that have dread data.
   108→    """
   109→    sums = [0.0] * 5
   110→    counts = [0] * 5
   111→
   112→    for f in findings:
   113→        dread = f.get("dread_score") or f.get("dread")
   114→        if dread is None:
   115→            continue
   116→
   117→        if isinstance(dread, dict):
   118→            # Dict form: {"damage": 10, "reproducibility": 9, ...}
   119→            for i, key in enumerate(_DREAD_LABELS):
   120→                v = dread.get(key)
   121→                if v is not None and isinstance(v, (int, float)):
   122→                    sums[i] += float(v)
   123→                    counts[i] += 1
   124→        elif isinstance(dread, (int, float)):
   125→            # Float form: overall DREAD score — distribute evenly
   126→            val = float(dread)
   127→            if val > 0:
   128→                for i in range(5):
   129→                    sums[i] += val
   130→                    counts[i] += 1
   131→
   132→    # Average each component
   133→    avg = []
   134→    for i in range(5):
   135→        avg.append(round(sums[i] / counts[i], 1) if counts[i] > 0 else 0.0)
   136→    return avg

   139→def _top_finding_dread(findings: List[Dict[str, Any]], n: int = 8) -> List[Dict[str, Any]]:
   140→    """Get top N findings with their DREAD breakdowns for grouped bar chart."""
   141→    # Sort by severity, then by points deducted
   142→    sev_order = {"critical": 0, "high": 1, "medium": 2, "low": 3, "info": 4}
   143→    sorted_f = sorted(
   144→        findings,
   145→        key=lambda f: (sev_order.get(f.get("severity", "info").lower(), 99), -f.get("points_deducted", 0)),
   146→    )
   147→
   148→    result = []
   149→    for f in sorted_f[:n]:
   150→        dread = f.get("dread_score") or f.get("dread")
   151→        if isinstance(dread, dict):
   152→            components = [dread.get(key, 0) for key in _DREAD_LABELS]
   153→        elif isinstance(dread, (int, float)) and float(dread) > 0:
   154→            # Distribute evenly
   155→            val = float(dread)
   156→            components = [val] * 5
   157→        else:
   158→            components = None
   159→
   160→        if components:
   161→            result.append({
   162→                "title": f.get("title", "")[:30],
   163→                "severity": f.get("severity", "info").lower(),
   164→                "components": components,
   165→            })
   166→    return result

   208→def _risk_posture(score: int, grade: str, sc: Dict[str, int]) -> Tuple[str, str]:
   209→    """Determine executive risk posture label and description."""
   210→    crit = sc.get("critical", 0)
   211→    high = sc.get("high", 0)
   212→
   213→    if score >= 90 and crit == 0 and high == 0:
   214→        return ("Excellent", "Target demonstrates strong security posture with no critical or high-severity findings. Minor informational items may be addressed opportunistically.")
   215→    elif score >= 75 and crit == 0:
   216→        return ("Good", "Overall security posture is sound. A small number of high-severity findings should be prioritized for remediation to reach best-in-class standing.")
   217→    elif score >= 50 and crit <= 2:
   218→        return ("Moderate", "Target presents a moderate attack surface. Critical and high findings require immediate attention. A structured remediation plan is recommended.")
   219→    elif score >= 25:
   220→        return ("Elevated", "Significant security weaknesses detected across multiple categories. Exploit chain potential exists. Prioritize critical findings and conduct a focused remediation sprint.")
   221→    else:
   222→        return ("Critical", "Severe security posture with widespread vulnerabilities. Immediate containment and remediation is required before any production exposure. Consider a dedicated security engagement.")

  1326→def _findings_to_dicts(findings: Any) -> List[Dict[str, Any]]:
  1327→    """Normalize findings (list of dicts or objects) to list of dicts."""
  1328→    result: List[Dict[str, Any]] = []
  1329→    for f in findings:
  1330→        if isinstance(f, dict):
  1331→            result.append(f)
  1332→        elif hasattr(f, "to_dict"):
  1333→            result.append(f.to_dict())
  1334→        elif hasattr(f, "__dict__"):
  1335→            result.append(vars(f))
