   190→// --- Circular Progress Ring ---
   191→function CircularProgressRing({ score, size = 140, strokeWidth = 8 }: { score: number; size?: number; strokeWidth?: number }) {
   192→  const radius = (size - strokeWidth * 2) / 2;
   193→  const circumference = 2 * Math.PI * radius;
   194→  const offset = circumference - (score / 100) * circumference;
   195→
   196→  const color = score >= 80 ? '#ff3355' : score >= 60 ? '#ff8844' : score >= 40 ? '#ffaa00' : '#00ff88';
   197→  const glowColor = color + '50';
   198→
   199→  return (
   200→    <div className="relative" style={{ width: size, height: size }}>
   201→      <svg viewBox={`0 0 ${size} ${size}`} className="w-full h-full -rotate-90">
   202→        <circle
   203→          cx={size / 2}
   204→          cy={size / 2}
   205→          r={radius}
   206→          fill="none"
   207→          stroke="rgba(255,255,255,0.06)"
   208→          strokeWidth={strokeWidth}
   209→        />
   210→        <motion.circle
   211→          cx={size / 2}
   212→          cy={size / 2}
   213→          r={radius}
   214→          fill="none"
   215→          stroke={color}
   216→          strokeWidth={strokeWidth}
   217→          strokeLinecap="round"
   218→          strokeDasharray={circumference}
   219→          initial={{ strokeDashoffset: circumference }}
   220→          animate={{ strokeDashoffset: offset }}
   221→          transition={{ duration: 2, ease: 'easeOut' as const, delay: 0.3 }}
   222→          style={{ filter: `drop-shadow(0 0 12px ${glowColor})` }}
   223→        />
   224→      </svg>
   225→      <div className="absolute inset-0 flex flex-col items-center justify-center">
   226→        <motion.span
   227→          className="font-bold font-mono text-3xl"
   228→          style={{ color }}
   229→          initial={{ opacity: 0, scale: 0.5 }}
   230→          animate={{ opacity: 1, scale: 1 }}
   231→          transition={{ delay: 0.8, type: 'spring' as const, stiffness: 200 }}
   232→        >
   233→          {score}
   234→        </motion.span>
   235→        <span className="text-[10px] font-semibold uppercase tracking-widest text-[#444444]">
   236→          / 100
   237→        </span>
   238→      </div>
   239→    </div>
   240→  );
   241→}
