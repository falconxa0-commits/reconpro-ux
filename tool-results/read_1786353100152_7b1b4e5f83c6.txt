     1→     1→"""QUANTUM FINGERPRINT — TCP/IP Stack Fingerprinting via HTTP Timing Analysis.
     2→     2→
     3→     3→ReconPro v9.2.0 capability: infer remote operating system, TCP stack
     4→     4→implementation, congestion-control algorithm, MTU, and kernel clock
     5→     5→resolution — entirely through HTTP-level timing observation.
     6→     6→
     7→     7→No raw sockets.  No SYN packets.  No privileged access.
     8→     8→Pure urllib / socket / ssl / time stdlib.
     9→     9→
    10→    10→Methodology
    11→    11→-----------
    12→    12→Each technique sends carefully crafted HTTP requests and measures
    13→    13→sub-millisecond timing characteristics of the response.  By comparing
    14→    14→the observed multi-dimensional timing fingerprint against a database
    15→    15→of known OS / kernel signatures, we produce a probabilistic OS
    16→    16→identification with a confidence score.
    17→    17→
    18→    18→The seven orthogonal signals are:
    19→    19→  1. Initial TTL deduction       — payload-size vs RTT variance
    20→    20→  2. TCP window size              — initial data-burst volume analysis
    21→    21→  3. SYN-ACK timing behaviour     — fresh-connection first-byte latency
    22→    22→  4. HTTP keep-alive persistence  — connection reuse timing degradation
    23→    23→  5. Path MTU detection           — payload escalation fragmentation spike
    24→    24→  6. Congestion control alg.      — burst-recovery timing pattern
    25→    25→  7. Timestamp resolution         — Date-header clock granularity
    26→    26→
    27→    27→References
    28→    28→----------
    29→    29→  [1] Zalewski, M. "p0f — Passive OS Fingerprinting" (2006)
    30→    30→  [2] RFC 1323 — TCP Extensions for High Performance
    31→    31→  [3] Ha, S. et al. "CUBIC: A New TCP-Friendly High-Speed TCP Variant" (2008)
    32→    32→  [4] Cardwell, N. et al. "BBR: Congestion-Based Congestion Control" (2016)

   390→   390→# Internal Data Structures
   391→   391→# ═══════════════════════════════════════════════════════════════════════════
   392→   392→
   393→   393→@dataclass
   394→   394→class TimingSample:
   395→   395→    """A single timing measurement."""
   396→   396→    label: str
   397→   397→    elapsed_ms: float
   398→   398→    payload_size: int = 0
   399→   399→    status_code: int = 0
   400→   400→    response_bytes: int = 0
   401→   401→    error: Optional[str] = None
   402→   402→
   403→   403→
   404→   404→@dataclass
   405→   405→class SubProbeResult:
   406→   406→    """Aggregated result from one sub-probe."""
   407→   407→    probe_name: str
   408→   408→    samples: List[TimingSample] = field(default_factory=list)
   409→   409→    derived_value: Any = None
   410→   410→    description: str = ""
   411→   411→    confidence: float = 0.0
   412→   412→
   413→   413→    @property
   414→   414→    def values(self) -> List[float]:
   415→   415→        return [s.elapsed_ms for s in self.samples if s.error is None]
   416→   416→
   417→   417→    @property
   418→   418→    def mean(self) -> float:
   419→   419→        v = self.values
   420→   420→        return statistics.mean(v) if v else 0.0
   421→   421→
   422→   422→    @property
   423→   423→    def median(self) -> float:
   424→   424→        v = sorted(self.values)
   425→   425→        n = len(v)
   426→   426→        if n == 0:
   427→   427→            return 0.0
   428→   428→        return v[n // 2]
   429→   429→
   430→   430→    @property
   431→   431→    def stdev(self) -> float:
   432→   432→        v = self.values
   433→   433→        return statistics.stdev(v) if len(v) > 1 else 0.0
   434→   434→
   435→   435→    @property
   436→   436→    def cv(self) -> float:
   437→   437→        """Coefficient of variation (jitter ratio)."""
   438→   438→        m = self.mean
   439→   439→        return self.stdev / m if m > 0 else 0.0
   440→   440→

   459→   459→    h = {
   460→   460→        "User-Agent": "ReconPro/9.2 (Quantum Fingerprint Scanner)",
   461→   461→        "Accept": "*/*",
   462→   462→        "Accept-Language": "en-US,en;q=0.9",
   463→   463→        "Connection": "keep-alive",
   464→   464→    }
   465→   465→    if headers:
   466→   466→        h.update(headers)
   467→   467→
   468→   468→    req = urllib.request.Request(url, data=body, method=method, headers=h)
   469→   469→    t0 = time.perf_counter()
   470→   470→
   471→   471→    try:
   472→   472→        if verify_tls:
   473→   473→            ctx = ssl.create_default_context()
   474→   474→        else:
   475→   475→            ctx = ssl._create_unverified_context()
   476→   476→
   477→   477→        with urllib.request.urlopen(req, timeout=timeout, context=ctx) as resp:
   478→   478→            raw = resp.read()
   479→   479→            elapsed = (time.perf_counter() - t0) * 1000.0
   480→   480→            resp_headers = dict(resp.headers.items())
   481→   481→            return TimingSample(
   482→   482→                label=url,
   483→   483→                elapsed_ms=elapsed,
   484→   484→                payload_size=len(body) if body else 0,
   485→   485→                status_code=resp.status,
   486→   486→                response_bytes=len(raw),
   487→   487→                error=None,
   488→   488→            )
   489→   489→    except urllib.error.HTTPError as e:
   490→   490→        elapsed = (time.perf_counter() - t0) * 1000.0
   491→   491→        try:
   492→   492→            e.read()
   493→   493→        except Exception:
   494→   494→            pass
   495→   495→        return TimingSample(
   496→   496→            label=url, elapsed_ms=elapsed, status_code=e.code, error=e.reason
   497→   497→        )
   498→   498→    except Exception as e:
   499→   499→        elapsed = (time.perf_counter() - t0) * 1000.0
   500→   500→        return TimingSample(label=url, elapsed_ms=elapsed, error=str(e)[:120])

   503→   503→def _make_unique_url(base_url: str, seed: str, size: int = 0) -> str:
   504→   504→    """Generate a unique probe URL to avoid caching artifacts."""
   505→   505→    h = hashlib.sha256(f"{seed}-{time.time_ns()}".encode()).hexdigest()[:12]
   506→   506→    url = f"{base_url}/.well-known/quantumfp-{h}"
   507→   507→    if size > 0:
   508→   508→        url += f"?s={size}&r={h}"
   509→   509→    return url

   511→   511→
   512→   512→# ═══════════════════════════════════════════════════════════════════════════
   513→   513→# Sub-Probe 1: Initial TTL Deduction
   514→   514→# ═══════════════════════════════════════════════════════════════════════════
   515→   515→
   516→   516→def _probe_ttl_deduction(
   517→   517→    base_url: str,
   518→   518→    timeout: int = 8,
   519→   519→    verify_tls: bool = True,
   520→   520→    limiter: Optional[Any] = None,
   521→   521→) -> SubProbeResult:
   522→   522→    """Deduce initial TTL through payload-size / RTT variance analysis.
   523→   523→
   524→   524→    Rationale: Different initial TTL values cause packets to traverse
   525→   525→    different numbers of hops before expiry.  When we vary the payload
   526→   526→    size, the number of TCP segments changes.  Segments that trigger
   527→   527→    more TCP-level processing at each hop reveal the initial TTL
   528→   528→    through a measurable correlation between payload size and RTT
   529→   529→    coefficient of variation.
   530→   530→
   531→   531→    Additionally, we look for a step-function pattern: when the
   532→   532→    number of required hops approaches TTL-1, the RTT variance
   533→   533→    increases measurably.
   534→   534→    """
   535→   535→    result = SubProbeResult(probe_name="ttl_deduction")
   536→   536→
   537→   537→    # Phase 1: Collect RTT samples at multiple payload sizes
   538→   538→    size_samples: Dict[int, List[float]] = {}
   539→   539→    for size in _TTL_PAYLOAD_SIZES:
   540→   540→        body = b"A" * size
   541→   541→        samples: List[float] = []
   542→   542→        for i in range(_PROBE_ITERATIONS):
   543→   543→            url = _make_unique_url(base_url, f"ttl-p{size}", size)
   544→   544→            ts = _timed_request(
   545→   545→                url, method="POST", body=body,
   546→   546→                timeout=timeout, verify_tls=verify_tls, limiter=limiter,
   547→   547→            )
   548→   548→            if ts.error is None:
   549→   549→                samples.append(ts.elapsed_ms)
   550→   550→            result.samples.append(ts)
   551→   551→            time.sleep(0.02)
   552→   552→        if samples:
   553→   553→            size_samples[size] = samples
   554→   554→
   555→   555→    if len(size_samples) < 3:
   556→   556→        result.derived_value = None
   557→   557→        result.description = "Insufficient clean samples for TTL deduction"
   558→   558→        return result
   559→   559→
   560→   560→    # Phase 2: Analyze RTT variance pattern vs payload size
   561→   561→    size_variance: List[Tuple[int, float, float]] = []
   562→   562→    for size, vals in sorted(size_samples.items()):
   563→   563→        mean_rtt = statistics.mean(vals)
   564→   564→        cv = statistics.stdev(vals) / mean_rtt if mean_rtt > 0 else 0
   565→   565→        size_variance.append((size, mean_rtt, cv))
   566→   566→
   567→   567→    # Phase 3: Score each candidate TTL
   568→   568→    # A lower TTL means more hops are traversed, which means more
   569→   569→    # network noise → higher CV for larger payloads.
   570→   570→    # We compute a "slope" of CV vs payload-size for each TTL candidate.
   571→   571→    total_mean_rtt = statistics.mean([m for _, m, _ in size_variance])
   572→   572→    avg_cv = statistics.mean([cv for _, _, cv in size_variance])
   573→   573→
   574→   574→    # The TTL deduction heuristic: look at how CV grows with payload size.
   575→   575→    # Larger payloads → more segments → more hop-dependent variance.
   576→   576→    # The *shape* of this growth curve distinguishes TTL values.
   577→   577→    cv_slope = 0.0
   578→   578→    if len(size_variance) > 2:
   579→   579→        x_vals = [math.log2(max(s, 1)) for s, _, _ in size_variance]
   580→   580→        y_vals = [cv for _, _, cv in size_variance]
   581→   581→        n = len(x_vals)
   582→   582→        mx = sum(x_vals) / n
   583→   583→        my = sum(y_vals) / n
   584→   584→        num = sum((x - mx) * (y - my) for x, y in zip(x_vals, y_vals))
   585→   585→        den = math.sqrt(sum((x - mx) ** 2 for x in x_vals)) * math.sqrt(
   586→   586→            sum((y - my) ** 2 for y in y_vals)
   587→   587→        )
   588→   588→        cv_slope = num / den if den > 0 else 0
   589→   589→
   590→   590→    # Score candidate TTLs
   591→   591→    ttl_scores: List[Tuple[int, float]] = []
   592→   592→    for candidate_ttl in _KNOWN_INITIAL_TTL:
   593→   593→        score = 0.0
   594→   594→
   595→   595→        # Factor 1: CV slope matches expected hop-count scaling
   596→   596→        # Higher TTL → fewer exhausted hops → lower slope
   597→   597→        expected_slope = 1.0 / math.log2(max(candidate_ttl, 2))
   598→   598→        slope_diff = abs(cv_slope - expected_slope)
   599→   599→        score += max(0, 1.0 - slope_diff * 3.0) * 0.35
   600→   600→
   601→   601→        # Factor 2: Mean RTT correlation
   602→   602→        # TTL 255 (networking gear) often has higher latency due to
   603→   603→        # additional processing in routers/switches
   604→   604→        if candidate_ttl == 255:
   605→   605→            score += (0.3 if total_mean_rtt > 50 else 0.1)
   606→   606→        elif candidate_ttl == 128:
   607→   607→            score += (0.3 if total_mean_rtt > 20 else 0.2)
   608→   608→        else:
   609→   609→            score += (0.2 if total_mean_rtt < 100 else 0.1)
   610→   610→
   611→   611→        # Factor 3: CV magnitude
   612→   612→        if candidate_ttl <= 64:
   613→   613→            score += max(0, 0.15 - avg_cv * 0.5)
   614→   614→        elif candidate_ttl == 128:
   615→   615→            score += max(0, 0.15 - abs(avg_cv - 0.07) * 2.0)
   616→   616→        else:
   617→   617→            score += max(0, 0.15 - abs(avg_cv - 0.12) * 2.0)
   618→   618→
   619→   619→        # Factor 4: Jumps in RTT at specific payload sizes
   620→   620→        if len(size_variance) > 3:
   621→   621→            # Look for step-function jumps at segment boundaries
   622→   622→            jumps = 0
   623→   623→            for i in range(1, len(size_variance)):
   624→   624→                rt_diff = size_variance[i][1] - size_variance[i - 1][1]
   625→   625→                if rt_diff > total_mean_rtt * 0.15:
   626→   626→                    jumps += 1
   627→   627→            # More jumps suggest more hops (lower TTL)
   628→   628→            if candidate_ttl == 64 and jumps >= 2:
   629→   629→                score += 0.15
   630→   630→            elif candidate_ttl == 128 and jumps <= 1:
   631→   631→                score += 0.10
   632→   632→            elif candidate_ttl == 255 and jumps == 0:
   633→   633→                score += 0.10
   634→   634→
   635→   635→        ttl_scores.append((candidate_ttl, score))
   636→   636→
   637→   637→    ttl_scores.sort(key=lambda x: x[1], reverse=True)
   638→   638→    best_ttl = ttl_scores[0][0] if ttl_scores else 64
   639→   639→    best_score = ttl_scores[0][1] if ttl_scores else 0
   640→   640→
   641→   641→    result.derived_value = {
   642→   642→        "deduced_ttl": best_ttl,
   643→   643→        "confidence": round(min(best_score, 1.0), 3),
   644→   644→        "all_scores": {str(t): round(s, 3) for t, s in ttl_scores},
   645→   645→        "mean_rtt_ms": round(total_mean_rtt, 2),
   646→   646→        "avg_cv": round(avg_cv, 4),
   647→   647→        "cv_slope": round(cv_slope, 4),
   648→   648→        "size_variance": [
   649→   649→            {"size": s, "mean_rtt": round(m, 2), "cv": round(c, 4)}
   650→   650→            for s, m, c in size_variance
   651→   651→        ],
   652→   652→    }
   653→   653→    result.confidence = min(best_score, 1.0)
   654→   654→    result.description = (
   655→   655→        f"Deduced initial TTL = {best_ttl} "
   656→   656→        f"(confidence {round(min(best_score, 1.0) * 100, 1)}%)"
   657→   657→    )
   658→   658→    return result
   659→   659→

   661→   661→# ═══════════════════════════════════════════════════════════════════════════
   662→   662→# Sub-Probe 2: TCP Window Size Fingerprinting
   663→   663→# ═══════════════════════════════════════════════════════════════════════════
   664→   664→
   665→   665→def _probe_tcp_window(
   666→   666→    base_url: str,
   667→   667→    timeout: int = 8,
   668→   668→    verify_tls: bool = True,
   669→   669→    limiter: Optional[Any] = None,
   670→   670→) -> SubProbeResult:
   671→   671→    """Fingerprint TCP window size via response data-burst analysis.
   672→   672→
   673→   673→    Rationale: When a server sends a large response, the TCP window
   674→   674→    size determines how much data is sent in the initial burst before
   675→   675→    waiting for ACKs.  We measure how response bytes arrive by probing
   676→   676→    with a large response endpoint and analyzing the relationship
   677→   677→    between request size, response time, and bytes received.
   678→   678→
   679→   679→    Key signals:
   680→   680→    - Bytes received per millisecond of RTT
   681→   681→    - Ratio of response time for small vs large responses
   682→   682→    - Chunked vs Content-Length transfer patterns
   683→   683→    """
   684→   684→    result = SubProbeResult(probe_name="tcp_window")
   685→   685→
   686→   686→    # Phase 1: Measure small response baseline
   687→   687→    small_rtts: List[float] = []
   688→   688→    small_bytes: List[int] = []
   689→   689→    for i in range(_PROBE_ITERATIONS):
   690→   690→        url = _make_unique_url(base_url, f"tw-sm-{i}")
   691→   691→        ts = _timed_request(url, timeout=timeout, verify_tls=verify_tls,
   692→   692→                            limiter=limiter)
   693→   693→        if ts.error is None:
   694→   694→            small_rtts.append(ts.elapsed_ms)
   695→   695→            small_bytes.append(ts.response_bytes)
   696→   696→        result.samples.append(ts)
   697→   697→        time.sleep(0.02)
   698→   698→
   699→   699→    # Phase 2: Measure large response (via POST to trigger potential
   700→   700→    # echo or error response of different size)
   701→   701→    large_rtts: List[float] = []
   702→   702→    large_bytes: List[int] = []
   703→   703→    large_body = b"X" * 4096
   704→   704→    for i in range(_PROBE_ITERATIONS):
   705→   705→        url = _make_unique_url(base_url, f"tw-lg-{i}")
   706→   706→        ts = _timed_request(
   707→   707→            url, method="POST", body=large_body,
   708→   708→            timeout=timeout, verify_tls=verify_tls, limiter=limiter,
   709→   709→        )
   710→   710→        if ts.error is None:
   711→   711→            large_rtts.append(ts.elapsed_ms)
   712→   712→            large_bytes.append(ts.response_bytes)
   713→   713→        result.samples.append(ts)
   714→   714→        time.sleep(0.02)
   715→   715→
   716→   716→    # Phase 3: Analyze transfer rates
   717→   717→    all_probe_rtts = small_rtts + large_rtts
   718→   718→    if not all_probe_rtts:
   719→   719→        result.derived_value = None
   720→   720→        result.description = "No clean responses for TCP window analysis"
   721→   721→        return result
   722→   722→
   723→   723→    all_bytes = small_bytes + large_bytes
   724→   724→    avg_bytes = statistics.mean(all_bytes) if all_bytes else 0
   725→   725→    avg_small_rtt = statistics.mean(small_rtts) if small_rtts else 0
   726→   726→    avg_large_rtt = statistics.mean(large_rtts) if large_rtts else 0
   727→   727→
   728→   728→    # Phase 4: Also probe the main page for chunked vs content-length
   729→   729→    main_probe = http_probe(base_url, timeout=timeout, verify_tls=verify_tls)
   730→   730→    is_chunked = "transfer-encoding" in {
   731→   731→        k.lower() for k in main_probe.get("headers", {})
   732→   732→    }
   733→   733→    content_length = main_probe.get("headers", {}).get("Content-Length", "")
   734→   734→
   735→   735→    # Transfer rate (bytes/ms) reflects effective window / RTT
   736→   736→    transfer_rate = avg_bytes / avg_large_rtt if avg_large_rtt > 0 else 0
   737→   737→
   738→   738→    # Phase 5: Classify window size
   739→   739→    # Map transfer-rate ranges to window size classes
   740→   740→    window_classification = "unknown"
   741→   741→    estimated_window = 0
   742→   742→
   743→   743→    if transfer_rate > 100:
   744→   744→        window_classification = "large (65535)"
   745→   745→        estimated_window = 65535
   746→   746→    elif transfer_rate > 30:
   747→   747→        window_classification = "medium-large (29200)"
   748→   748→        estimated_window = 29200
   749→   749→    elif transfer_rate > 15:
   750→   750→        window_classification = "medium (16384)"
   751→   751→        estimated_window = 16384
   752→   752→    elif transfer_rate > 5:
   753→   753→        window_classification = "small (8192)"
   754→   754→        estimated_window = 8192
   755→   755→    else:
   756→   756→        window_classification = "very-small (4128)"
   757→   757→        estimated_window = 4128
   758→   758→
   759→   759→    # Adjust for known server hints
   760→   760→    server_header = main_probe.get("headers", {}).get("Server", "").lower()
   761→   761→    if "nginx" in server_header and estimated_window < 29200:
   762→   762→        estimated_window = 29200
   763→   763→        window_classification = "medium-large (29200) [nginx hint]"
   764→   764→    elif "cloudflare" in server_header:
   765→   765→        estimated_window = 65535
   766→   766→        window_classification = "large (65535) [cloudflare hint]"
   767→   767→    elif "apache" in server_header and estimated_window < 29200:
   768→   768→        estimated_window = 29200
   769→   769→        window_classification = "medium-large (29200) [apache hint]"
   770→   770→
   771→   771→    result.derived_value = {
   772→   772→        "estimated_window": estimated_window,
   773→   773→        "classification": window_classification,
   774→   774→        "transfer_rate_bytes_per_ms": round(transfer_rate, 2),
   775→   775→        "avg_small_rtt_ms": round(avg_small_rtt, 2),
   776→   776→        "avg_large_rtt_ms": round(avg_large_rtt, 2),
   777→   777→        "is_chunked": is_chunked,
   778→   778→        "server_hint": server_header or None,
   779→   779→        "avg_response_bytes": round(avg_bytes, 1),
   780→   780→    }
   781→   781→    result.confidence = 0.6 if transfer_rate > 0 else 0.2
   782→   782→    result.description = (
   783→   783→        f"Estimated TCP window ≈ {window_classification} "
   784→   784→        f"(transfer rate: {transfer_rate:.1f} bytes/ms)"
   785→   785→    )
   786→   786→    return result

   789→   789→# ═══════════════════════════════════════════════════════════════════════════
   790→   790→# Sub-Probe 3: SYN-ACK Timing Behaviour
   791→   791→# ═══════════════════════════════════════════════════════════════════════════
   792→   792→
   793→   793→def _probe_synack_timing(
   794→   794→    base_url: str,
   795→   795→    timeout: int = 8,
   796→   796→    verify_tls: bool = True,
   797→   797→    limiter: Optional[Any] = None,
   798→   798→) -> SubProbeResult:
   799→   799→    """Measure SYN-ACK timing via fresh-connection first-byte latency.
   800→   800→
   801→   801→    Rationale: Every time we open a new TCP connection (no keep-alive
   802→   802→    reuse), the observed latency includes:
   803→   803→      - DNS resolution (amortized / cached)
   804→   804→      - TCP SYN → SYN-ACK round trip
   805→   805→      - TLS handshake (if HTTPS)
   806→   806→      - Server request processing → first byte
   807→   807→
   808→   808→    By opening many fresh connections and measuring the minimum RTT
   809→   809→    (the tail-eliminated latency), we isolate the kernel's TCP
   810→   810→    interrupt-handling and timer resolution characteristics.
   811→   811→
   812→   812→    Different kernels have characteristic first-byte latencies due to
   813→   813→    different NAPI polling intervals, interrupt coalescing, and
   814→   814→    TCP backlog processing strategies.
   815→   815→    """
   816→   816→    result = SubProbeResult(probe_name="synack_timing")
   817→   817→
   818→   818→    # Force fresh connections by using a new OpenerDirector each time
   819→   819→    rtts: List[float] = []
   820→   820→
   821→   821→    for i in range(_PROBE_ITERATIONS + 4):
   822→   822→        url = _make_unique_url(base_url, f"synack-{i}")
   823→   823→
   824→   824→        if limiter:
   825→   825→            limiter.acquire()
   826→   826→
   827→   827→        # Build a fresh opener (no connection pooling)
   828→   828→        if verify_tls:
   829→   829→            ctx = ssl.create_default_context()
   830→   830→        else:
   831→   831→            ctx = ssl._create_unverified_context()
   832→   832→
   833→   833→        opener = urllib.request.build_opener(
   834→   834→            urllib.request.HTTPSHandler(context=ctx) if base_url.startswith("https")
   835→   835→            else urllib.request.HTTPHandler()
   836→   836→        )
   837→   837→
   838→   838→        headers = {
   839→   839→            "User-Agent": "ReconPro/9.2 (Quantum Fingerprint Scanner)",
   840→   840→            "Accept": "*/*",
   841→   841→            "Connection": "close",      # force new connection per request
   842→   842→        }
   843→   843→        req = urllib.request.Request(url, headers=headers)
   844→   844→
   845→   845→        t0 = time.perf_counter()
   846→   846→        try:
   847→   847→            with opener.open(req, timeout=timeout) as resp:
   848→   848→                resp.read(512)
   849→   849→                elapsed = (time.perf_counter() - t0) * 1000.0
   850→   850→                rtts.append(elapsed)
   851→   851→                result.samples.append(TimingSample(
   852→   852→                    label=url, elapsed_ms=elapsed, status_code=resp.status
   853→   853→                ))
   854→   854→        except Exception as e:
   855→   855→            elapsed = (time.perf_counter() - t0) * 1000.0
   856→   856→            result.samples.append(TimingSample(
   857→   857→                label=url, elapsed_ms=elapsed, error=str(e)[:80]
   858→   858→            ))
   859→   859→        time.sleep(0.03)
   860→   860→
   861→   861→    if not rtts:
   862→   862→        result.derived_value = None
   863→   863→        result.description = "No clean SYN-ACK timing samples"
   864→   864→        return result
   865→   865→
   866→   866→    rtts_sorted = sorted(rtts)
   867→   867→    n = len(rtts_sorted)
   868→   868→
   869→   869→    # Statistical measures
   870→   870→    # Use trimmed mean (discard top 20% — outliers from GC / context switches)
   871→   871→    trim_idx = max(1, n - n // 5)
   872→   872→    trimmed = rtts_sorted[:trim_idx]
   873→   873→    median_rtt = rtts_sorted[n // 2]
   874→   874→    trimmed_mean = statistics.mean(trimmed)
   875→   875→    minimum = rtts_sorted[0]
   876→   876→    p10 = rtts_sorted[max(0, n // 10)]
   877→   877→    p90 = rtts_sorted[min(n - 1, n * 9 // 10)]
   878→   878→
   879→   879→    # IQR-based jitter
   880→   880→    q1 = rtts_sorted[n // 4] if n >= 4 else median_rtt
   881→   881→    q3 = rtts_sorted[3 * n // 4] if n >= 4 else median_rtt
   882→   882→    iqr = q3 - q1
   883→   883→
   884→   884→    # Classify the kernel interrupt pattern
   885→   885→    kernel_hint = "unknown"
   886→   886→    if trimmed_mean < 3.0 and iqr < 2.0:
   887→   887→        kernel_hint = "low-latency kernel (BBR / modern Linux / CDN)"
   888→   888→    elif trimmed_mean < 5.0 and iqr < 3.0:
   889→   889→        kernel_hint = "standard Linux 5.x+ / macOS / FreeBSD"
   890→   890→    elif trimmed_mean < 8.0 and iqr < 5.0:
   891→   891→        kernel_hint = "Linux 4.x / older macOS / Windows Server"
   892→   892→    elif trimmed_mean < 12.0 and iqr < 8.0:
   893→   893→        kernel_hint = "older Linux / Windows Server / JunOS"
   894→   894→    elif trimmed_mean >= 12.0:
   895→   895→        kernel_hint = "networking equipment / high-latency kernel / proxy chain"
   896→   896→
   897→   897→    result.derived_value = {
   898→   898→        "trimmed_mean_ms": round(trimmed_mean, 3),
   899→   899→        "median_ms": round(median_rtt, 3),
   900→   900→        "minimum_ms": round(minimum, 3),
   901→   901→        "p10_ms": round(p10, 3),
   902→   902→        "p90_ms": round(p90, 3),
   903→   903→        "iqr_ms": round(iqr, 3),
   904→   904→        "stddev": round(statistics.stdev(rtts) if len(rtts) > 1 else 0, 3),
   905→   905→        "sample_count": n,
   906→   906→        "kernel_pattern": kernel_hint,
   907→   907→    }
   908→   908→    result.confidence = 0.7
   909→   909→    result.description = (
   910→   910→        f"SYN-ACK latency: median={median_rtt:.2f}ms, "
   911→   911→        f"trimmed-mean={trimmed_mean:.2f}ms, "
   912→   912→        f"iqr={iqr:.2f}ms ({kernel_hint})"
   913→   913→    )
   914→   914→    return result
   915→   915→
   916→   916→
   917→   917→# ═══════════════════════════════════════════════════════════════════════════
   918→   918→# Sub-Probe 4: HTTP Keep-Alive Behaviour
   919→   919→# ═══════════════════════════════════════════════════════════════════════════
   920→   920→
   921→   921→def _probe_keepalive(
   922→   922→    base_url: str,
   923→   923→    timeout: int = 8,
   924→   924→    verify_tls: bool = True,
   925→   925→    limiter: Optional[Any] = None,
   926→   926→) -> SubProbeResult:
   927→   927→    """Measure connection persistence and keep-alive timing.
   928→   928→
   929→   929→    Rationale: Different TCP stacks keep idle connections open for
   930→   930→    different durations, governed by kernel-level tcp_fin_timeout,
   931→   931→    tcp_keepalive_time, and server-level configuration.  By sending
   932→   932→    a sequence of requests on the same connection with increasing
   933→   933→    idle intervals, we can detect the point at which the server closes
   934→   934→    the connection, revealing the TCP stack's idle timeout.
   935→   935→
   936→   936→    We also measure RTT degradation across sequential requests —
   937→   937→    some stacks show increasing latency due to NIC buffer pressure or
   938→   938→    TCP delayed-ACK timer interactions.
   939→   939→    """
   940→   940→    result = SubProbeResult(probe_name="keepalive")
   941→   941→
   942→   942→    # Use a single opener for connection reuse
   943→   943→    if verify_tls:
   944→   944→        ctx = ssl.create_default_context()
   945→   945→    else:
   946→   946→        ctx = ssl._create_unverified_context()
   947→   947→
   948→   948→    opener = urllib.request.build_opener(
   949→   949→        urllib.request.HTTPSHandler(context=ctx) if base_url.startswith("https")
   950→   950→        else urllib.request.HTTPHandler()
   951→   951→    )
   952→   952→
   953→   953→    # Phase 1: Rapid sequential requests — measure RTT degradation
   954→   954→    seq_rtts: List[float] = []
   955→   955→    conn_alive = True
   956→   956→
   957→   957→    for i in range(_PROBE_ITERATIONS):
   958→   958→        url = _make_unique_url(base_url, f"ka-seq-{i}")
   959→   959→        headers = {
   960→   960→            "User-Agent": "ReconPro/9.2",
   961→   961→            "Accept": "*/*",
   962→   962→            "Connection": "keep-alive",
   963→   963→        }
   964→   964→        req = urllib.request.Request(url, headers=headers)
   965→   965→
   966→   966→        if limiter:
   967→   967→            limiter.acquire()
   968→   968→
   969→   969→        t0 = time.perf_counter()
   970→   970→        try:
   971→   971→            with opener.open(req, timeout=timeout) as resp:
   972→   972→                resp.read(256)
   973→   973→                elapsed = (time.perf_counter() - t0) * 1000.0
   974→   974→                seq_rtts.append(elapsed)
   975→   975→                result.samples.append(TimingSample(
   976→   976→                    label=f"seq-{i}", elapsed_ms=elapsed, status_code=resp.status
   977→   977→                ))
   978→   978→        except Exception as e:
   979→   979→            elapsed = (time.perf_counter() - t0) * 1000.0
   980→   980→            result.samples.append(TimingSample(
   981→   981→                label=f"seq-{i}", elapsed_ms=elapsed, error=str(e)[:80]
   982→   982→            ))
   983→   983→            conn_alive = False
   984→   984→            break
   985→   985→
   986→   986→    # Phase 2: Idle-interval probing
   987→   987→    # Send a request, wait, send another — find when connection drops
   988→   988→    idle_intervals = [0.5, 1.0, 2.0, 3.0, 5.0, 7.5, 10.0, 15.0, 20.0]
   989→   989→    idle_survival: List[Tuple[float, bool]] = []
   990→   990→    conn_refreshed = False
   991→   991→
   992→   992→    for wait_s in idle_intervals:
   993→   993→        # Refresh connection first
   994→   994→        refresh_url = _make_unique_url(base_url, f"ka-refr-{wait_s}")
   995→   995→        ref_headers = {
   996→   996→            "User-Agent": "ReconPro/9.2",
   997→   997→            "Accept": "*/*",
   998→   998→            "Connection": "keep-alive",
   999→   999→        }
  1000→  1000→        ref_req = urllib.request.Request(refresh_url, headers=ref_headers)
  1001→  1001→
  1002→  1002→        if limiter:
  1003→  1003→            limiter.acquire()
  1004→  1004→
  1005→  1005→        try:
  1006→  1006→            with opener.open(ref_req, timeout=timeout) as resp:
  1007→  1007→                resp.read(128)
  1008→  1008→            conn_refreshed = True
  1009→  1009→        except Exception:
  1010→  1010→            # Connection was closed; create fresh one
  1011→  1011→            opener = urllib.request.build_opener(
  1012→  1012→                urllib.request.HTTPSHandler(context=ctx)
  1013→  1013→                if base_url.startswith("https")
  1014→  1014→                else urllib.request.HTTPHandler()
  1015→  1015→            )
  1016→  1016→            try:
  1017→  1017→                with opener.open(ref_req, timeout=timeout) as resp:
  1018→  1018→                    resp.read(128)
  1019→  1019→                conn_refreshed = True
  1020→  1020→            except Exception:
  1021→  1021→                conn_refreshed = False
  1022→  1022→                idle_survival.append((wait_s, False))
  1023→  1023→                continue
  1024→  1024→
  1025→  1025→        if not conn_refreshed:
  1026→  1026→            idle_survival.append((wait_s, False))
  1027→  1027→            continue
  1028→  1028→
  1029→  1029→        # Wait the idle interval
  1030→  1030→        time.sleep(wait_s)
  1031→  1031→
  1032→  1032→        # Try to reuse the connection
  1033→  1033→        test_url = _make_unique_url(base_url, f"ka-test-{wait_s}")
  1034→  1034→        test_headers = {
  1035→  1035→            "User-Agent": "ReconPro/9.2",
  1036→  1036→            "Accept": "*/*",
  1037→  1037→            "Connection": "keep-alive",
  1038→  1038→        }
  1039→  1039→        test_req = urllib.request.Request(test_url, headers=test_headers)
  1040→  1040→
  1041→  1041→        if limiter:
  1042→  1042→            limiter.acquire()
  1043→  1043→
  1044→  1044→        t0 = time.perf_counter()
  1045→  1045→        try:
  1046→  1046→            with opener.open(test_req, timeout=timeout) as resp:
  1047→  1047→                resp.read(128)
  1048→  1048→                elapsed = (time.perf_counter() - t0) * 1000.0
  1049→  1049→                result.samples.append(TimingSample(
  1050→  1050→                    label=f"idle-{wait_s}s", elapsed_ms=elapsed,
  1051→  1051→                    status_code=resp.status
  1052→  1052→                ))
  1053→  1053→                idle_survival.append((wait_s, True))
  1054→  1054→        except Exception as e:
  1055→  1055→            elapsed = (time.perf_counter() - t0) * 1000.0
  1056→  1056→            result.samples.append(TimingSample(
  1057→  1057→                label=f"idle-{wait_s}s", elapsed_ms=elapsed,
  1058→  1058→                error="connection_closed"
  1059→  1059→            ))
  1060→  1060→            idle_survival.append((wait_s, False))
  1061→  1061→
  1062→  1062→    # Phase 3: Analyze results
  1063→  1063→    estimated_timeout = 0.0
  1064→  1064→    for wait_s, alive in idle_survival:
  1065→  1065→        if not alive:
  1066→  1066→            estimated_timeout = wait_s
  1067→  1067→            break
  1068→  1068→    else:
  1069→  1069→        estimated_timeout = max(w for w, _ in idle_survival) if idle_survival else 0
  1070→  1070→
  1071→  1071→    # RTT degradation slope
  1072→  1072→    rtt_degradation = 0.0
  1073→  1073→    if len(seq_rtts) > 2:
  1074→  1074→        x = list(range(len(seq_rtts)))
  1075→  1075→        y = seq_rtts
  1076→  1076→        n = len(x)
  1077→  1077→        mx = sum(x) / n
  1078→  1078→        my = sum(y) / n
  1079→  1079→        num = sum((xi - mx) * (yi - my) for xi, yi in zip(x, y))
  1080→  1080→        den = math.sqrt(sum((xi - mx) ** 2 for xi in x))
  1081→  1081→        rtt_degradation = num / den if den > 0 else 0
  1082→  1082→
  1083→  1083→    # Classify keep-alive behaviour
  1084→  1084→    ka_hint = "unknown"
  1085→  1085→    if estimated_timeout <= 5:
  1086→  1086→        ka_hint = "aggressive timeout (Apache / embedded / Cisco)"
  1087→  1087→    elif estimated_timeout <= 30:
  1088→  1088→        ka_hint = "moderate timeout (Nginx default / Caddy / load balancer)"
  1089→  1089→    elif estimated_timeout <= 75:
  1090→  1090→        ka_hint = "standard Linux kernel (tcp_keepalive_time ≈ 75s)"
  1091→  1091→    elif estimated_timeout <= 120:
  1092→  1092→        ka_hint = "Windows Server default (120s)"
  1093→  1093→    else:
  1094→  1094→        ka_hint = "long-lived (CDN / reverse proxy / custom config)"
  1095→  1095→
  1096→  1096→    if rtt_degradation > 0.5:
  1097→  1097→        ka_hint += " + increasing latency (possible NIC buffer pressure)"
  1098→  1098→
  1099→  1099→    result.derived_value = {
  1100→  1100→        "estimated_timeout_s": round(estimated_timeout, 1),
  1101→  1101→        "keepalive_hint": ka_hint,
  1102→  1102→        "idle_survival": [(round(w, 1), a) for w, a in idle_survival],
  1103→  1103→        "seq_rtt_degradation_ms_per_req": round(rtt_degradation, 3),
  1104→  1104→        "seq_rtts": [round(r, 2) for r in seq_rtts],
  1105→  1105→        "seq_rtts_median": round(
  1106→  1106→            statistics.median(seq_rtts) if seq_rtts else 0, 2
  1107→  1107→        ),
  1108→  1108→    }
  1109→  1109→    result.confidence = 0.6
  1110→  1110→    result.description = (
  1111→  1111→        f"Keep-alive timeout ≈ {estimated_timeout:.1f}s — {ka_hint}"
  1112→  1112→    )
  1113→  1113→    return result
  1114→  1114→
  1115→  1115→
  1116→  1116→# ═══════════════════════════════════════════════════════════════════════════
  1117→  1117→# Sub-Probe 5: Path MTU Detection
  1118→  1118→# ═══════════════════════════════════════════════════════════════════════════
  1119→  1119→
  1120→  1120→def _probe_mtu(
  1121→  1121→    base_url: str,
  1122→  1122→    timeout: int = 8,
  1123→  1123→    verify_tls: bool = True,
  1124→  1124→    limiter: Optional[Any] = None,
  1125→  1125→) -> SubProbeResult:
  1126→  1126→    """Detect path MTU by escalating payload size until fragmentation.
  1127→  1127→
  1128→  1128→    Rationale: When the POST body exceeds the path MTU (after subtracting
  1129→  1129→    TCP + IP headers), the packet must be fragmented.  Fragmentation
  1130→  1130→    causes a measurable RTT spike because:
  1131→  1131→      - Additional packet processing at routers
  1132→  1132→      - Potential retransmission if fragments are lost
  1133→  1133→      - ICMP "Fragmentation Needed" messages (RFC 1191)
  1134→  1134→
  1135→  1135→    By plotting RTT vs payload size and detecting the inflection point,
  1136→  1136→    we can estimate the effective path MTU.
  1137→  1137→
  1138→  1138→    Typical MTU signatures:
  1139→  1139→      - 1500: Standard Ethernet (most common)
  1140→  1140→      - 1472: Without IP header (ICMP payload)
  1141→  1141→      - 1460: Without IP+TCP headers
  1142→  1142→      - 9000: Jumbo frames (data center)
  1143→  1143→      - 1280: IPv6 minimum
  1144→  1144→    """
  1145→  1145→    result = SubProbeResult(probe_name="mtu_detection")
  1146→  1146→
  1147→  1147→    payload_rtts: List[Tuple[int, float]] = []
  1148→  1148→    prev_rtt = None
  1149→  1149→
  1150→  1150→    for size in _MTU_PAYLOAD_SIZES:
  1151→  1151→        body = b"P" * size
  1152→  1152→        rtts_for_size: List[float] = []
  1153→  1153→
  1154→  1154→        for trial in range(3):
  1155→  1155→            url = _make_unique_url(base_url, f"mtu-{size}-{trial}")
  1156→  1156→            ts = _timed_request(
  1157→  1157→                url, method="POST", body=body,
  1158→  1158→                timeout=timeout, verify_tls=verify_tls, limiter=limiter,
  1159→  1159→            )
  1160→  1160→            if ts.error is None:
  1161→  1161→                rtts_for_size.append(ts.elapsed_ms)
  1162→  1162→            result.samples.append(ts)
  1163→  1163→            time.sleep(0.02)
  1164→  1164→
  1165→  1165→        if rtts_for_size:
  1166→  1166→            mean_rtt = statistics.mean(rtts_for_size)
  1167→  1167→            payload_rtts.append((size, mean_rtt))
  1168→  1168→        else:
  1169→  1169→            payload_rtts.append((size, float("inf")))
  1170→  1170→
  1171→  1171→    # Phase 2: Detect MTU inflection point
  1172→  1172→    # Look for the first significant RTT spike
  1173→  1173→    baseline_rtts = [r for s, r in payload_rtts if s <= 1460 and r < float("inf")]
  1174→  1174→    baseline_mean = statistics.mean(baseline_rtts) if baseline_rtts else 50.0
  1175→  1175→    baseline_stdev = statistics.stdev(baseline_rtts) if len(baseline_rtts) > 1 else 5.0
  1176→  1176→    spike_threshold = baseline_mean + baseline_stdev * 2.5
  1177→  1177→
  1178→  1178→    mtu_estimate = 0
  1179→  1179→    spike_detected = False
  1180→  1180→    for size, rtt in payload_rtts:
  1181→  1181→        if rtt > spike_threshold and rtt != float("inf"):
  1182→  1182→            # Account for TCP/IP headers: effective MTU = payload + headers
  1183→  1183→            # We subtract 60 bytes for TCP+IP headers (typical)
  1184→  1184→            mtu_estimate = size + 60
  1185→  1185→            spike_detected = True
  1186→  1186→            break
  1187→  1187→        elif rtt == float("inf"):
  1188→  1188→            mtu_estimate = size + 60
  1189→  1189→            spike_detected = True
  1190→  1190→            break
  1191→  1191→
  1192→  1192→    if not spike_detected:
  1193→  1193→        # No spike detected → path supports all tested sizes → large MTU
  1194→  1194→        max_tested = max(s for s, _ in payload_rtts) if payload_rtts else 1500
  1195→  1195→        mtu_estimate = max(max_tested + 60, 9000)
  1196→  1196→
  1197→  1197→    # Phase 3: Classify MTU
  1198→  1198→    mtu_class = "unknown"
  1199→  1199→    if 1496 <= mtu_estimate <= 1520:
  1200→  1200→        mtu_class = "standard Ethernet (1500 bytes)"
  1201→  1201→    elif 1470 <= mtu_estimate <= 1496:
  1202→  1202→        mtu_class = "Ethernet with encapsulation overhead"
  1203→  1203→    elif 1486 <= mtu_estimate <= 1520:
  1204→  1204→        mtu_class = "PPPoE or VPN encapsulation"
  1205→  1205→    elif mtu_estimate >= 8960:
  1206→  1206→        mtu_class = "jumbo frames (9000 bytes)"
  1207→  1207→    elif 1276 <= mtu_estimate <= 1300:
  1208→  1208→        mtu_class = "IPv6 minimum MTU"
  1209→  1209→    elif mtu_estimate > 0:
  1210→  1210→        mtu_class = f"non-standard ({mtu_estimate} bytes)"
  1211→  1211→
  1212→  1212→    result.derived_value = {
  1213→  1213→        "estimated_mtu": mtu_estimate,
  1214→  1214→        "classification": mtu_class,
  1215→  1215→        "spike_detected": spike_detected,
  1216→  1216→        "baseline_mean_ms": round(baseline_mean, 2),
  1217→  1217→        "spike_threshold_ms": round(spike_threshold, 2),
  1218→  1218→        "payload_rtts": [
  1219→  1219→            {"size": s, "mean_rtt": round(r, 2) if r < float("inf") else None}
  1220→  1220→            for s, r in payload_rtts
  1221→  1221→        ],
  1222→  1222→    }
  1223→  1223→    result.confidence = 0.7 if spike_detected else 0.4
  1224→  1224→    result.description = (
  1225→  1225→        f"Estimated path MTU = {mtu_estimate} bytes — {mtu_class}"
  1226→  1226→    )
  1227→  1227→    return result
  1228→  1228→
  1229→  1229→
  1230→  1230→# ═══════════════════════════════════════════════════════════════════════════
  1231→  1231→# Sub-Probe 6: Congestion Control Algorithm Identification
  1232→  1232→# ═══════════════════════════════════════════════════════════════════════════
  1233→  1233→
  1234→  1234→def _probe_congestion_control(
  1235→  1235→    base_url: str,
  1236→  1236→    timeout: int = 8,
  1237→  1237→    verify_tls: bool = True,
  1238→  1238→    limiter: Optional[Any] = None,
  1239→  1239→) -> SubProbeResult:
  1240→  1240→    """Identify congestion control algorithm via burst-recovery analysis.
  1241→  1241→
  1242→  1242→    Rationale: Different congestion control algorithms recover from
  1243→  1243→    packet loss / network congestion at different rates:
  1244→  1244→
  1245→  1245→      - **CUBIC**: W(t) = C(t - K)³ + W_max — cubic growth function
  1246→  1246→        produces a characteristic "S-curve" recovery pattern.
  1247→  1247→      - **Reno/NewReno**: Linear increase after loss detection.
  1248→  1248→        Slow, steady recovery with predictable slope.
  1249→  1249→      - **BBR**: Attempts to estimate bandwidth and RTT, then
  1250→  1250→        paces packets accordingly.  Recovery is near-instant
  1251→  1251→        with minimal RTT inflation.
  1252→  1252→      - **LEDBAT**: Designed for background traffic, delays
  1253→  1253→        heavily.  Very high RTT after burst.
  1254→  1254→
  1255→  1255→    Methodology:
  1256→  1256→    1. Establish baseline RTT with isolated probes.
  1257→  1257→    2. Fire a burst of requests as fast as possible.
  1258→  1258→    3. Measure RTT inflation during and after burst.
  1259→  1259→    4. Analyze recovery curve shape.
  1260→  1260→    """
  1261→  1261→    result = SubProbeResult(probe_name="congestion_control")
  1262→  1262→
  1263→  1263→    # Phase 1: Baseline
  1264→  1264→    baseline_rtts: List[float] = []
  1265→  1265→    for i in range(_PROBE_ITERATIONS // 2):
  1266→  1266→        url = _make_unique_url(base_url, f"cc-base-{i}")
  1267→  1267→        ts = _timed_request(url, timeout=timeout, verify_tls=verify_tls,
  1268→  1268→                            limiter=limiter)
  1269→  1269→        if ts.error is None:
  1270→  1270→            baseline_rtts.append(ts.elapsed_ms)
  1271→  1271→        result.samples.append(ts)
  1272→  1272→        time.sleep(0.05)
  1273→  1273→
  1274→  1274→    if not baseline_rtts:
  1275→  1275→        result.derived_value = None
  1276→  1276→        result.description = "No baseline for congestion analysis"
  1277→  1277→        return result
  1278→  1278→
  1279→  1279→    baseline_mean = statistics.mean(baseline_rtts)
  1280→  1280→    baseline_stdev = statistics.stdev(baseline_rtts) if len(baseline_rtts) > 1 else 1.0
  1281→  1281→
  1282→  1282→    # Phase 2: Burst
  1283→  1283→    burst_rtts: List[float] = []
  1284→  1284→    for i in range(_CONGESTION_BURST_SIZE):
  1285→  1285→        url = _make_unique_url(base_url, f"cc-burst-{i}")
  1286→  1286→        # Minimal delay between requests to create congestion
  1287→  1287→        ts = _timed_request(url, timeout=timeout, verify_tls=verify_tls,
  1288→  1288→                            limiter=limiter)
  1289→  1289→        burst_rtts.append(ts.elapsed_ms)
  1290→  1290→        result.samples.append(ts)
  1291→  1291→
  1292→  1292→    # Phase 3: Recovery probes
  1293→  1293→    recovery_rtts: List[float] = []
  1294→  1294→    for i in range(_CONGESTION_RECOVERY_PROBES):
  1295→  1295→        url = _make_unique_url(base_url, f"cc-recovery-{i}")
  1296→  1296→        ts = _timed_request(url, timeout=timeout, verify_tls=verify_tls,
  1297→  1297→                            limiter=limiter)
  1298→  1298→        if ts.error is None:
  1299→  1299→            recovery_rtts.append(ts.elapsed_ms)
  1300→  1300→        result.samples.append(ts)
  1301→  1301→        time.sleep(0.08)
  1302→  1302→
  1303→  1303→    # Phase 4: Analysis
  1304→  1304→    burst_mean = statistics.mean(burst_rtts)
  1305→  1305→    burst_inflation = (burst_mean - baseline_mean) / baseline_mean if baseline_mean > 0 else 0
  1306→  1306→
  1307→  1307→    recovery_mean = statistics.mean(recovery_rtts) if recovery_rtts else burst_mean
  1308→  1308→    recovery_ratio = recovery_mean / baseline_mean if baseline_mean > 0 else 1.0
  1309→  1309→
  1310→  1310→    # Recovery curve slope (should be negative — RTT decreasing back to baseline)
  1311→  1311→    recovery_slope = 0.0
  1312→  1312→    if len(recovery_rtts) > 2:
  1313→  1313→        x = list(range(len(recovery_rtts)))
  1314→  1314→        y = recovery_rtts
  1315→  1315→        n = len(x)
  1316→  1316→        mx = sum(x) / n
  1317→  1317→        my = sum(y) / n
  1318→  1318→        num = sum((xi - mx) * (yi - my) for xi, yi in zip(x, y))
  1319→  1319→        den = math.sqrt(sum((xi - mx) ** 2 for xi in x))
  1320→  1320→        recovery_slope = num / den if den > 0 else 0
  1321→  1321→
  1322→  1322→    # Burst pattern analysis — CUBIC shows non-linear (cubic) increase
  1323→  1323→    burst_nonlinearity = 0.0
  1324→  1324→    if len(burst_rtts) > 3:
  1325→  1325→        # Fit quadratic to burst RTTs; residual indicates non-linearity
  1326→  1326→        x = list(range(len(burst_rtts)))
  1327→  1327→        y = burst_rtts
  1328→  1328→        n = len(x)
  1329→  1329→        # Simple: compute variance of successive differences
  1330→  1330→        diffs = [y[i+1] - y[i] for i in range(len(y) - 1)]
  1331→  1331→        if diffs:
  1332→  1332→            diff_mean = statistics.mean(diffs)
  1333→  1333→            diff_var = statistics.variance(diffs) if len(diffs) > 1 else 0
  1334→  1334→            burst_nonlinearity = math.sqrt(diff_var) / (abs(diff_mean) + 1)
  1335→  1335→
  1336→  1336→    # Classify congestion control
  1337→  1337→    algo = "unknown"
  1338→  1338→    algo_confidence = 0.0
  1339→  1339→
  1340→  1340→    if burst_inflation < 0.15 and recovery_ratio < 1.2:
  1341→  1341→        algo = "bbr"
  1342→  1342→        algo_confidence = 0.75
  1343→  1343→    elif burst_inflation < 0.3 and recovery_slope < -1.0 and burst_nonlinearity > 0.3:
  1344→  1344→        algo = "cubic"
  1345→  1345→        algo_confidence = 0.65
  1346→  1346→    elif burst_inflation < 0.5 and -0.5 < recovery_slope < 0.5:
  1347→  1347→        algo = "newreno"
  1348→  1348→        algo_confidence = 0.55
  1349→  1349→    elif burst_inflation < 0.5 and recovery_slope >= -0.3:
  1350→  1350→        algo = "reno"
  1351→  1351→        algo_confidence = 0.45
  1352→  1352→    elif burst_inflation > 0.5 or recovery_ratio > 2.0:
  1353→  1353→        algo = "ledbat"
  1354→  1354→        algo_confidence = 0.50
  1355→  1355→    elif burst_inflation > 0.3:
  1356→  1356→        algo = "cubic (congested)"
  1357→  1357→        algo_confidence = 0.45
  1358→  1358→
  1359→  1359→    result.derived_value = {
  1360→  1360→        "algorithm": algo,
  1361→  1361→        "confidence": round(algo_confidence, 3),
  1362→  1362→        "baseline_mean_ms": round(baseline_mean, 2),
  1363→  1363→        "burst_mean_ms": round(burst_mean, 2),
  1364→  1364→        "burst_inflation_pct": round(burst_inflation * 100, 1),
  1365→  1365→        "recovery_mean_ms": round(recovery_mean, 2),
  1366→  1366→        "recovery_ratio": round(recovery_ratio, 3),
  1367→  1367→        "recovery_slope": round(recovery_slope, 3),
  1368→  1368→        "burst_nonlinearity": round(burst_nonlinearity, 3),
  1369→  1369→        "burst_rtts": [round(r, 2) for r in burst_rtts],
  1370→  1370→        "recovery_rtts": [round(r, 2) for r in recovery_rtts],
  1371→  1371→    }
  1372→  1372→    result.confidence = algo_confidence
  1373→  1373→    result.description = (
  1374→  1374→        f"Congestion control: {algo} "
  1375→  1375→        f"(confidence {round(algo_confidence * 100, 1)}%, "
  1376→  1376→        f"burst inflation: {burst_inflation * 100:.1f}%)"
  1377→  1377→    )
  1378→  1378→    return result
  1379→  1379→
  1380→  1380→
  1381→  1381→# ═══════════════════════════════════════════════════════════════════════════
  1382→  1382→# Sub-Probe 7: Timestamp Resolution
  1383→  1383→# ═══════════════════════════════════════════════════════════════════════════
  1384→  1384→
  1385→  1385→def _probe_timestamp_resolution(
  1386→  1386→    base_url: str,
  1387→  1387→    timeout: int = 8,
  1388→  1388→    verify_tls: bool = True,
  1389→  1389→    limiter: Optional[Any] = None,
  1390→  1390→) -> SubProbeResult:
  1391→  1391→    """Measure server's Date-header clock granularity.
  1392→  1392→
  1393→  1393→    Rationale: The HTTP Date header reflects the server's system
  1394→  1394→    clock.  Different OS kernels have different timer resolutions:
  1395→  1395→      - Linux 2.6+/macOS/FreeBSD: ~1ms (high-resolution timers)
  1396→  1396→      - Windows: ~15.625ms (default timer resolution, 64 Hz)
  1397→  1397→      - Older/embedded systems: ~10ms
  1398→  1398→
  1399→  1399→    By sending rapid sequential requests and analyzing the
  1400→  1400→    inter-request differences in Date timestamps, we can
  1401→  1401→    determine the kernel's clock tick resolution.
  1402→  1402→
  1403→  1403→    This is one of the most reliable OS fingerprinting signals
  1404→  1404→    because it directly reflects kernel configuration rather
  1405→  1405→    than network conditions.
  1406→  1406→    """
  1407→  1407→    result = SubProbeResult(probe_name="timestamp_resolution")
  1408→  1408→
  1409→  1409→    # Also probe the main page for a reference Date header
  1410→  1410→    main_probe = http_probe(base_url, timeout=timeout, verify_tls=verify_tls)
  1411→  1411→    server_header = main_probe.get("headers", {}).get("Server", "")
  1412→  1412→    via_header = main_probe.get("headers", {}).get("Via", "")
  1413→  1413→
  1414→  1414→    timestamps: List[float] = []  # epoch seconds
  1415→  1415→
  1416→  1416→    for i in range(_TIMESTAMP_SAMPLES):
  1417→  1417→        url = _make_unique_url(base_url, f"ts-{i}")
  1418→  1418→        ts = _timed_request(url, timeout=timeout, verify_tls=verify_tls,
  1419→  1419→                            limiter=limiter)
  1420→  1420→        if ts.error is None and ts.status_code in (200, 301, 302, 304, 404, 405, 403):
  1421→  1421→            # We need the Date header — use http_probe for this
  1422→  1422→            probe = http_probe(url, timeout=timeout, verify_tls=verify_tls)
  1423→  1423→            date_str = probe.get("headers", {}).get("Date", "")
  1424→  1424→            if date_str:
  1425→  1425→                try:
  1426→  1426→                    dt = parsedate_to_datetime(date_str)
  1427→  1427→                    epoch = dt.timestamp()
  1428→  1428→                    timestamps.append(epoch)
  1429→  1429→                except (ValueError, TypeError):
  1430→  1430→                    pass
  1431→  1431→        result.samples.append(ts)
  1432→  1432→        time.sleep(0.005)  # 5ms between samples
  1433→  1433→
  1434→  1434→    if len(timestamps) < 3:
  1435→  1435→        # Fallback: use wall-clock deltas between requests
  1436→  1436→        result.derived_value = {
  1437→  1437→            "resolution_ms": 0,
  1438→  1438→            "classification": "insufficient data",
  1439→  1439→            "method": "wall-clock fallback",
  1440→  1440→            "server_header": server_header,
  1441→  1441→        }
  1442→  1442→        result.confidence = 0.1
  1443→  1443→        result.description = "Insufficient Date-header samples for timestamp analysis"
  1444→  1444→        return result
  1445→  1445→
  1446→  1446→    # Phase 2: Analyze inter-timestamp deltas
  1447→  1447→    deltas_ms: List[float] = []
  1448→  1448→    for i in range(1, len(timestamps)):
  1449→  1449→        delta = (timestamps[i] - timestamps[i - 1]) * 1000.0
  1450→  1450→        if delta >= 0:  # Ignore clock skew (negative deltas)
  1451→  1451→            deltas_ms.append(delta)
  1452→  1452→
  1453→  1453→    if not deltas_ms:
  1454→  1454→        result.derived_value = {
  1455→  1455→            "resolution_ms": 0,
  1456→  1456→            "classification": "clock going backwards?",
  1457→  1457→            "server_header": server_header,
  1458→  1458→        }
  1459→  1459→        result.confidence = 0.1
  1460→  1460→        return result
  1461→  1461→
  1462→  1462→    # Phase 3: Determine resolution
  1463→  1463→    # Common resolutions: 1ms, 10ms, 15.625ms (≈16ms), 1000ms
  1464→  1464→    unique_deltas = sorted(set(round(d, 1) for d in deltas_ms if d > 0))
  1465→  1465→    min_delta = min(deltas_ms) if deltas_ms else 1.0
  1466→  1466→
  1467→  1467→    # GCD-based resolution detection
  1468→  1468→    # Find the greatest common divisor of all deltas
  1469→  1469→    def _gcd_floats(vals: List[float], epsilon: float = 0.5) -> float:
  1470→  1470→        """Approximate GCD for floating-point values."""
  1471→  1471→        vals_rounded = [round(v / epsilon) for v in vals]
  1472→  1472→        from math import gcd as _int_gcd
  1473→  1473→        result = vals_rounded[0]
  1474→  1474→        for v in vals_rounded[1:]:
  1475→  1475→            result = _int_gcd(result, v)
  1476→  1476→        return result * epsilon
  1477→  1477→
  1478→  1478→    resolution = _gcd_floats([d for d in deltas_ms if d > 0], epsilon=0.1)
  1479→  1479→
  1480→  1480→    # Match against known resolutions
  1481→  1481→    known_resolutions = {
  1482→  1482→        1.0: "~1ms (Linux / macOS / FreeBSD — high-resolution timer)",
  1483→  1483→        10.0: "~10ms (embedded / older Linux / Juniper)",
  1484→  1484→        15.625: "~15.625ms (Windows — default system timer, 64 Hz)",
  1485→  1485→        16.0: "~16ms (Windows — rounded timer resolution)",
  1486→  1486→        1000.0: "~1000ms (very coarse / virtualized / load-balanced)",
  1487→  1487→    }
  1488→  1488→
  1489→  1489→    classification = "unknown"
  1490→  1490→    best_match = 0
  1491→  1491→    for known_res, label in known_resolutions.items():
  1492→  1492→        if abs(resolution - known_res) < 2.0 or abs(min_delta - known_res) < 3.0:
  1493→  1493→            classification = label
  1494→  1494→            best_match = known_res
  1495→  1495→            break
  1496→  1496→
  1497→  1497→    if classification == "unknown":
  1498→  1498→        classification = f"~{resolution:.1f}ms (non-standard resolution)"
  1499→  1499→
  1500→  1500→    # Phase 4: Compute confidence
  1501→  1501→    res_confidence = 0.0
  1502→  1502→    if best_match == 15.625 or best_match == 16.0:
  1503→  1503→        # Windows timer is very distinctive — 15.625ms is hard to fake
  1504→  1504→        res_confidence = 0.85
  1505→  1505→    elif best_match == 1.0:
  1506→  1506→        res_confidence = 0.7
  1507→  1507→    elif best_match == 10.0:
  1508→  1508→        res_confidence = 0.55
  1509→  1509→    else:
  1510→  1510→        res_confidence = 0.4
  1511→  1511→
  1512→  1512→    result.derived_value = {
  1513→  1513→        "resolution_ms": round(resolution, 3),
  1514→  1514→        "min_delta_ms": round(min_delta, 3),
  1515→  1515→        "classification": classification,
  1516→  1516→        "unique_deltas": unique_deltas[:10],
  1517→  1517→        "delta_stats": {
  1518→  1518→            "count": len(deltas_ms),
  1519→  1519→            "mean": round(statistics.mean(deltas_ms), 2),
  1520→  1520→            "median": round(statistics.median(deltas_ms), 2),
  1521→  1521→            "min": round(min(deltas_ms), 3),
  1522→  1522→        },
  1523→  1523→        "sample_count": len(timestamps),
  1524→  1524→        "server_header": server_header,
  1525→  1525→        "via_header": via_header,
  1526→  1526→    }
  1527→  1527→    result.confidence = res_confidence
  1528→  1528→    result.description = (
  1529→  1529→        f"Timestamp resolution: {classification} "
  1530→  1530→        f"(measured: {resolution:.2f}ms, min delta: {min_delta:.2f}ms)"
  1531→  1531→    )
  1532→  1532→    return result
  1533→  1533→
  1534→  1534→
  1535→  1535→# ═══════════════════════════════════════════════════════════════════════════
  1536→  1536→# OS Confidence Scoring Engine
  1537→  1537→# ═══════════════════════════════════════════════════════════════════════════
  1538→  1538→
  1539→  1539→def _signal_match_score(
  1540→  1540→    observed: Any,
  1541→  1541→    expected: Any,
  1542→  1542→    tolerance: float,
  1543→  1543→    is_categorical: bool = False,
  1544→  1544→) -> float:
  1545→  1545→    """Compute how well an observed signal matches an expected value.
  1546→  1546→
  1547→  1547→    Returns 0.0 (no match) to 1.0 (perfect match).
  1548→  1548→    """
  1549→  1549→    if observed is None:
  1550→  1550→        return 0.0
  1551→  1551→
  1552→  1552→    if is_categorical:
  1553→  1553→        return 1.0 if str(observed).lower() == str(expected).lower() else 0.0
  1554→  1554→
  1555→  1555→    try:
  1556→  1556→        obs = float(observed)
  1557→  1557→        exp = float(expected)
  1558→  1558→        diff = abs(obs - exp)
  1559→  1559→        range_size = max(abs(exp), 1.0)
  1560→  1560→        normalized_diff = diff / (range_size * tolerance + 0.001)
  1561→  1561→        return max(0.0, 1.0 - normalized_diff)
  1562→  1562→    except (TypeError, ValueError):
  1563→  1563→        return 0.0
  1564→  1564→
  1565→  1565→
  1566→  1566→def _compute_os_confidence(
  1567→  1567→    probes: Dict[str, SubProbeResult],
  1568→  1568→) -> List[Dict[str, Any]]:
  1569→  1569→    """Probabilistic OS identification by matching all signals against
  1570→  1570→    the OS_SIGNATURES database.
  1571→  1571→
  1572→  1572→    Each signal contributes a weighted score.  The total confidence
  1573→  1573→    is the weighted average across all matched signals, normalized
  1574→  1574→    by the prior probability (weight field).
  1575→  1575→    """
  1576→  1576→    scores: List[Dict[str, Any]] = []
  1577→  1577→
  1578→  1578→    # Extract observed values from probe results
  1579→  1579→    ttl_data = probes.get("ttl_deduction")
  1580→  1580→    window_data = probes.get("tcp_window")
  1581→  1581→    synack_data = probes.get("synack_timing")
  1582→  1582→    keepalive_data = probes.get("keepalive")
  1583→  1583→    mtu_data = probes.get("mtu_detection")
  1584→  1584→    congestion_data = probes.get("congestion_control")
  1585→  1585→    timestamp_data = probes.get("timestamp_resolution")
  1586→  1586→
  1587→  1587→    observed_ttl = None
  1588→  1588→    if ttl_data and ttl_data.derived_value:
  1589→  1589→        observed_ttl = ttl_data.derived_value.get("deduced_ttl")
  1590→  1590→
  1591→  1591→    observed_window = None
  1592→  1592→    if window_data and window_data.derived_value:
  1593→  1593→        observed_window = window_data.derived_value.get("estimated_window")
  1594→  1594→
  1595→  1595→    observed_synack = None
  1596→  1596→    if synack_data and synack_data.derived_value:
  1597→  1597→        observed_synack = synack_data.derived_value.get("trimmed_mean_ms")
  1598→  1598→
  1599→  1599→    observed_keepalive = None
  1600→  1600→    if keepalive_data and keepalive_data.derived_value:
  1601→  1601→        observed_keepalive = keepalive_data.derived_value.get("estimated_timeout_s")
  1602→  1602→
  1603→  1603→    observed_mtu = None
  1604→  1604→    if mtu_data and mtu_data.derived_value:
  1605→  1605→        observed_mtu = mtu_data.derived_value.get("estimated_mtu")
  1606→  1606→
  1607→  1607→    observed_congestion = None
  1608→  1608→    if congestion_data and congestion_data.derived_value:
  1609→  1609→        observed_congestion = congestion_data.derived_value.get("algorithm")
  1610→  1610→
  1611→  1611→    observed_ts_res = None
  1612→  1612→    if timestamp_data and timestamp_data.derived_value:
  1613→  1613→        observed_ts_res = timestamp_data.derived_value.get("resolution_ms")
  1614→  1614→
  1615→  1615→    for sig in OS_SIGNATURES:
  1616→  1616→        signal_scores: Dict[str, float] = {}
  1617→  1617→        weights: Dict[str, float] = {}
  1618→  1618→
  1619→  1619→        # TTL match (weight: 0.20)
  1620→  1620→        if observed_ttl is not None:
  1621→  1621→            ttl_match = 1.0 if abs(observed_ttl - sig.ttl_initial) <= sig.ttl_tol else 0.0
  1622→  1622→            signal_scores["ttl"] = ttl_match
  1623→  1623→            weights["ttl"] = 0.20
  1624→  1624→
  1625→  1625→        # Window match (weight: 0.12)
  1626→  1626→        if observed_window is not None:
  1627→  1627→            signal_scores["window"] = _signal_match_score(
  1628→  1628→                observed_window, sig.tcp_window, sig.window_tol
  1629→  1629→            )
  1630→  1630→            weights["window"] = 0.12
  1631→  1631→
  1632→  1632→        # SYN-ACK match (weight: 0.15)
  1633→  1633→        if observed_synack is not None:
  1634→  1634→            signal_scores["synack"] = _signal_match_score(
  1635→  1635→                observed_synack, sig.synack_ms, sig.synack_tol
  1636→  1636→            )
  1637→  1637→            weights["synack"] = 0.15
  1638→  1638→
  1639→  1639→        # Keep-alive match (weight: 0.10)
  1640→  1640→        if observed_keepalive is not None:
  1641→  1641→            signal_scores["keepalive"] = _signal_match_score(
  1642→  1642→                observed_keepalive, sig.keepalive_s, sig.keepalive_tol
  1643→  1643→            )
  1644→  1644→            weights["keepalive"] = 0.10
  1645→  1645→
  1646→  1646→        # MTU match (weight: 0.08)
  1647→  1647→        if observed_mtu is not None:
  1648→  1648→            signal_scores["mtu"] = _signal_match_score(
  1649→  1649→                observed_mtu, sig.mtu_default, 200.0
  1650→  1650→            )
  1651→  1651→            weights["mtu"] = 0.08
  1652→  1652→
  1653→  1653→        # Congestion control match (weight: 0.18)
  1654→  1654→        if observed_congestion is not None:
  1655→  1655→            signal_scores["congestion"] = _signal_match_score(
  1656→  1656→                observed_congestion, sig.congestion, 0.5, is_categorical=True
  1657→  1657→            )
  1658→  1658→            weights["congestion"] = 0.18
  1659→  1659→
  1660→  1660→        # Timestamp resolution match (weight: 0.17)
  1661→  1661→        if observed_ts_res is not None:
  1662→  1662→            signal_scores["timestamp"] = _signal_match_score(
  1663→  1663→                observed_ts_res, sig.timestamp_res, sig.timestamp_tol
  1664→  1664→            )
  1665→  1665→            weights["timestamp"] = 0.17
  1666→  1666→
  1667→  1667→        # Compute weighted score
  1668→  1668→        total_weight = sum(weights.values())
  1669→  1669→        if total_weight > 0:
  1670→  1670→            weighted_score = sum(
  1671→  1671→                signal_scores[k] * weights[k] for k in signal_scores
  1672→  1672→            ) / total_weight
  1673→  1673→        else:
  1674→  1674→            weighted_score = 0.0
  1675→  1675→
  1676→  1676→        # Apply prior (prior weight modulates the score)
  1677→  1677→        prior_factor = sig.weight / max(s.weight for s in OS_SIGNATURES)
  1678→  1678→        final_score = weighted_score * (0.7 + 0.3 * prior_factor)
  1679→  1679→
  1680→  1680→        scores.append({
  1681→  1681→            "os": sig.name,
  1682→  1682→            "family": sig.family,
  1683→  1683→            "score": round(final_score, 4),
