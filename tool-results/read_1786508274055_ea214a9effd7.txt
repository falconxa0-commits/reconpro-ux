   560→def _tool_shell_command(command: str = "") -> ToolResult:
   561→    """Run an arbitrary shell command. DANGEROUS — use with caution."""
   562→    if not command:
   563→        return ToolResult("No command specified.", success=False)
   564→    dangerous_patterns = ["rm -rf /", "mkfs", "dd if=", ":(){ :|:& };:", "chmod -R 777 /"]
   565→    for dp in dangerous_patterns:
   566→        if dp in command:
   567→            return ToolResult(f"Blocked dangerous command pattern: {dp}", success=False)
   568→    try:
   569→        result = subprocess.run(
   570→            command, shell=True, capture_output=True, text=True, timeout=30,
   571→        )
   572→        output = (result.stdout + result.stderr)[:2000]
   573→        return ToolResult(
   574→            f"Exit code: {result.returncode}\n{output}",
   575→            success=result.returncode == 0,
   576→        )
   577→    except subprocess.TimeoutExpired:
   578→        return ToolResult("Command timed out (30s).", success=False)
   579→    except Exception as e:
   580→        return ToolResult(f"Command error: {e}", success=False)

   786→

   875→    @classmethod
   876→    def score_finding(cls, finding: Dict) -> Dict[str, Any]:
   877→        """Compute CVSS-like score for a single finding.
   878→
   879→        Returns a dict with: base_score, severity, av, ac, pr, ui, c, i, a, vector_string.
   880→        """
   881→        title = (finding.get("title") or "").lower()
   882→        desc = (finding.get("description") or "").lower()
   883→        combined = f"{title} {desc}"
   884→        original_sev = (finding.get("severity") or "").lower()
   885→
   886→        # Find matching profile
   887→        profile = cls._CATEGORY_PROFILES[-1]  # default
   888→        for prof in cls._CATEGORY_PROFILES:
   889→            if prof["patterns"] and any(p in combined for p in prof["patterns"]):
   890→                profile = prof
   891→                break
   892→
   893→        # Override metrics based on original severity if it's explicitly critical
   894→        av = cls._AV[profile["av"]]
   895→        ac = cls._AC[profile["ac"]]
   896→        pr = cls._PR[profile["pr"]]
   897→        ui = cls._UI[profile["ui"]]
   898→        c = cls._CIA[profile["c"]]
   899→        i = cls._CIA[profile["i"]]
   900→        a = cls._CIA[profile["a"]]
   901→
   902→        # If the original severity is critical, boost impact to High
   903→        if original_sev == "critical":
   904→            c = max(c, cls._CIA["H"])
   905→            i = max(i, cls._CIA["H"])
   906→            a = max(a, cls._CIA["H"])
   907→        elif original_sev == "high":
   908→            c = max(c, cls._CIA["L"])
   909→            i = max(i, cls._CIA["L"])
   910→            a = max(a, cls._CIA["L"])
   911→
   912→        # CVSS base score calculation
   913→        iss = 1.0 - ((1.0 - c) * (1.0 - i) * (1.0 - a))
   914→        impact = 6.42 * iss
   915→        exploitability = 8.22 * av * ac * pr * ui
   916→
   917→        if impact <= 0:
   918→            base_score = 0.0
   919→        else:
   920→            base_score = min(impact + exploitability, 10.0)
   921→
   922→        # Round up to nearest 0.1
   923→        base_score = math.ceil(base_score * 10) / 10
   924→
   925→        # Derive severity from score
   926→        if base_score >= 9.0:
   927→            derived_sev = "critical"
   928→        elif base_score >= 7.0:
   929→            derived_sev = "high"
   930→        elif base_score >= 4.0:
   931→            derived_sev = "medium"
   932→        elif base_score > 0.0:
   933→            derived_sev = "low"
   934→        else:
   935→            derived_sev = "info"
   936→
   937→        # Use the more severe of original vs derived
   938→        severity_order = {"critical": 4, "high": 3, "medium": 2, "low": 1, "info": 0}
   939→        final_sev = original_sev if severity_order.get(original_sev, -1) >= severity_order.get(derived_sev, -1) else derived_sev
   940→
   941→        vector = f"CVSS:3.1/AV:{profile['av']}/AC:{profile['ac']}/PR:{profile['pr']}/UI:{profile['ui']}/C:{cls._cia_label(c)}/I:{cls._cia_label(i)}/A:{cls._cia_label(a)}"
   942→
   943→        return {
   944→            "base_score": base_score,
   945→            "severity": final_sev,
   946→            "cvss_vector": vector,
   947→            "metrics": {
   948→                "av": profile["av"], "ac": profile["ac"], "pr": profile["pr"],
   949→                "ui": profile["ui"], "c": cls._cia_label(c), "i": cls._cia_label(i),
   950→                "a": cls._cia_label(a),
   951→            },
   952→        }
   953→
   954→    @classmethod
   955→    def score_findings(cls, findings: List[Dict]) -> List[Dict]:
   956→        """Score a list of findings, returning enriched list."""
   957→        enriched = []
   958→        for f in findings:
   959→            scored = dict(f)
   960→            cvss = cls.score_finding(f)
   961→            scored["cvss_score"] = cvss["base_score"]
   962→            scored["cvss_severity"] = cvss["severity"]
   963→            scored["cvss_vector"] = cvss["cvss_vector"]
   964→            scored["cvss_metrics"] = cvss["metrics"]
   965→            enriched.append(scored)
   966→        return enriched
   967→
   968→    @staticmethod
   969→    def _cia_label(value: float) -> str:
   970→        if value >= 0.56:
   971→            return "H"
   972→        elif value >= 0.22:
   973→            return "L"
   974→        return "N"

  1285→    @classmethod
  1286→    def correlate(cls, findings: List[Dict]) -> List[Dict]:
  1287→        """Run all correlation rules against findings and return compound findings.
  1288→
  1289→        Each finding is checked as a lowercase string of title + description.
  1290→        A rule fires when ALL trigger groups have at least one pattern matched
  1291→        across the full finding set.
  1292→        """
  1293→        if not findings:
  1294→            return []
  1295→
  1296→        # Build a flat list of all finding text for matching
  1297→        finding_texts: List[str] = []
  1298→        for f in findings:
  1299→            title = (f.get("title") or "").lower()
  1300→            desc = (f.get("description") or "").lower()
  1301→            finding_texts.append(f"{title} {desc}")
  1302→
  1303→        compound_findings: List[Dict] = []
  1304→
  1305→        for rule in cls.CORRELATION_RULES:
  1306→            triggers = rule["triggers"]
  1307→            rule_fired = True
  1308→
  1309→            for trigger_group in triggers:
  1310→                # Each trigger group: at least one pattern must match across all findings
  1311→                group_matched = False
  1312→                for pattern in trigger_group:
  1313→                    for ft in finding_texts:
  1314→                        if pattern in ft:
  1315→                            group_matched = True
  1316→                            break
  1317→                    if group_matched:
  1318→                        break
  1319→                if not group_matched:
  1320→                    rule_fired = False
  1321→                    break
  1322→
  1323→            if rule_fired:
  1324→                compound = {
  1325→                    "title": rule["compound_title"],
  1326→                    "severity": rule["compound_severity"],
  1327→                    "description": rule["compound_description"],
  1328→                    "category": "compound",
  1329→                    "correlation_id": rule["id"],
  1330→                    "is_compound": True,
  1331→                }
  1332→                # Compute CVSS score for the compound finding
  1333→                if "cvss_override" in rule:
  1334→                    override = rule["cvss_override"]
  1335→                    av = CVSSScorer._AV[override["av"]]
  1336→                    ac = CVSSScorer._AC[override["ac"]]
  1337→                    pr = CVSSScorer._PR[override["pr"]]
  1338→                    ui = CVSSScorer._UI[override["ui"]]
  1339→                    c_val = CVSSScorer._CIA[override["c"]]
  1340→                    i_val = CVSSScorer._CIA[override["i"]]
  1341→                    a_val = CVSSScorer._CIA[override["a"]]

  1362→
  1363→class AttackSurfaceMapper:
  1364→    """Computes an attack surface score (0–100, higher = more exposed).
  1365→
  1366→    Factors:
  1367→      1. Open ports (0–25 pts): 0 ports=0, 1–2=5, 3–5=10, 6–10=15, 11+=25
  1368→      2. Exposed paths/endpoints (0–20 pts): 0=0, 1–3=5, 4–7=10, 8–12=15, 13+=20
  1369→      3. Missing security headers (0–20 pts): 0=0, 1=3, 2=6, 3=10, 4=14, 5+=20
  1370→      4. Tech stack complexity (0–15 pts): 1–2 techs=2, 3–4=5, 5–6=8, 7–8=11, 9+=15
  1371→      5. TLS grade (0–20 pts): A=0, B=5, C=10, D=15, F=20
  1372→    """
  1373→
  1374→    @classmethod
  1375→    def compute(cls, findings: List[Dict], scan_data: Optional[Dict] = None) -> Dict[str, Any]:
  1376→        """Compute attack surface score and return breakdown."""
  1377→        findings_lower = []
  1378→        for f in findings:
  1379→            title = (f.get("title") or "").lower()
  1380→            desc = (f.get("description") or "").lower()
  1381→            findings_lower.append(f"{title} {desc}")
  1382→        all_text = " ".join(findings_lower)
  1383→
  1384→        # 1. Open ports
  1385→        port_score = cls._score_ports(findings_lower, scan_data)
  1386→        # 2. Exposed paths
  1387→        path_score = cls._score_exposed_paths(findings_lower)
  1388→        # 3. Missing headers
  1389→        header_score = cls._score_missing_headers(findings_lower)
  1390→        # 4. Tech stack complexity
  1391→        tech_score = cls._score_tech_stack(all_text, scan_data)
  1392→        # 5. TLS grade
  1393→        tls_score = cls._score_tls_grade(findings_lower, scan_data)
  1394→
  1395→        total = min(port_score + path_score + header_score + tech_score + tls_score, 100)
  1396→
  1397→        # Risk rating
  1398→        if total >= 80:
  1399→            rating = "CRITICAL"
  1400→        elif total >= 60:
  1401→            rating = "HIGH"
  1402→        elif total >= 40:
  1403→            rating = "MODERATE"
  1404→        elif total >= 20:
  1405→            rating = "LOW"
  1406→        else:
  1407→            rating = "MINIMAL"
  1408→
  1409→        return {
  1410→            "attack_surface_score": total,
  1411→            "risk_rating": rating,
  1412→            "breakdown": {
  1413→                "open_ports": {"score": port_score, "max": 25},
  1414→                "exposed_paths": {"score": path_score, "max": 20},
  1415→                "missing_headers": {"score": header_score, "max": 20},
  1416→                "tech_complexity": {"score": tech_score, "max": 15},
  1417→                "tls_grade": {"score": tls_score, "max": 20},
  1418→            },
  1419→        }

  1608→    @classmethod

  1610→        """Render the risk matrix as a text string with Unicode box-drawing.

  1641→                    grid_labels[likelihood_idx][impact_idx].append(title)

  1666→                count = grid[y][x]
  1667→                sym = cls._DENSITY_SYMBOLS[min(count, 4)]
  1668→                row1 += f"{sym:^{col_width}}│"
  1669→                labels = grid_labels[y][x]
  1670→                if labels:
  1671→                    row2 += f"{labels[0][:col_width]:^{col_width}}│"
  1672→                else:
  1673→                    row2 += f"{'':^{col_width}}│"
  1674→
  1675→            lines.append(row1)
  1676→            lines.append(row2)
  1677→
  1678→            # Separator or bottom border
  1679→            if y < 4:
  1680→                lines.append(f"{'':<{label_width + 2}}├" + "─" * col_width + ("┼" + "─" * col_width) * 4 + "┤")
  1681→            else:
  1682→                lines.append(f"{'':<{label_width + 2}}└" + "─" * col_width + ("┴" + "─" * col_width) * 4 + "┘")
  1683→
  1684→        # Summary line
  1685→        total = sum(grid[y][x] for y in range(5) for x in range(5))
  1686→        critical_zone = sum(grid[y][x] for y in range(2) for x in range(3, 5))
  1687→        lines.append("")
  1688→        lines.append(f"  Total findings plotted: {total}")
  1689→        lines.append(f"  Critical zone (High/Very High × High/Critical): {critical_zone}")
  1690→

  1693→    @classmethod
  1694→    def _score_to_band_index(cls, score: float) -> int:
  1695→        """Map score to impact band index (0=Negligible, 4=Critical)."""
  1696→        for i, (name, lo, hi) in enumerate(cls._IMPACT_BANDS):
  1697→            if lo <= score < hi:
  1698→                return 4 - i  # Reverse so Critical is rightmost
  1699→        return 0

  1701→    @classmethod
  1702→    def _score_to_likelihood(cls, score: float, finding: Dict) -> int:
  1703→        """Estimate likelihood from score and finding metadata.
  1704→
  1705→        Higher CVSS exploitability metrics → higher likelihood.
  1706→        We approximate from score and severity.
  1707→        """
  1708→        severity = (finding.get("severity") or "").lower()
  1709→        is_compound = finding.get("is_compound", False)
  1710→        is_remote = True  # Most findings in recon are network-based
  1711→
  1712→        # Base likelihood from score
  1713→        base = score / 10.0
  1714→
  1715→        # Boost for compound findings (more likely to be exploitable)
  1716→        if is_compound:
  1717→            base = min(base + 0.15, 1.0)
  1718→
  1719→        # Adjust for severity
  1720→        sev_boost = {"critical": 0.1, "high": 0.05, "medium": 0, "low": -0.1, "info": -0.2}
  1721→        base += sev_boost.get(severity, 0)
  1722→        base = max(0.0, min(base, 1.0))
  1723→
  1724→        # Map to band index (0=Very Low, 4=Very High)
  1725→        if base >= 0.85:
  1726→            return 0
  1727→        elif base >= 0.65:
  1728→            return 1
  1729→        elif base >= 0.40:
  1730→            return 2
  1731→        elif base >= 0.15:
  1732→            return 3
  1733→        else:
  1734→            return 4

  1859→    @classmethod

  1861→        cls,

  1863→        domain: str = "",
  1864→        already_planned: Optional[List[str]] = None,
  1865→    ) -> List[Dict[str, Any]]:
  1866→        """Analyze findings and suggest deepening steps.
  1867→
  1868→        Returns a list of suggestion dicts: {tool, args, reason, trigger_findings}.

  1871→        suggestions: List[Dict[str, Any]] = []
  1872→
  1873→        for rule in cls.DEEPENING_RULES:
  1874→            threshold = cls._SEVERITY_ORDER.get(rule["severity_threshold"], 2)
  1875→            tool = rule["suggested_tool"]
  1876→

  1878→            if tool in already_planned:
  1879→                continue
  1880→
  1881→            # Check if any finding matches the trigger patterns
  1882→            matching_findings: List[str] = []
  1883→            for f in findings:
  1884→                f_sev = cls._SEVERITY_ORDER.get((f.get("severity") or "").lower(), 0)

  1886→                    continue
  1887→                f_text = f"{(f.get('title') or '').lower()} {(f.get('description') or '').lower()}"
  1888→                for pattern in rule["trigger_patterns"]:
  1889→                    if pattern in f_text:
  1890→                        matching_findings.append(f.get("title", ""))
  1891→                        break

  1898→                        args["findings_json"] = json.dumps(findings[:20])
  1899→                    except (TypeError, ValueError):
  1900→                        args["findings_json"] = "[]"
  1901→                if args.get("domain") == "__AUTO__" and domain:
  1902→                    args["domain"] = domain
  1903→

  1906→                    "args": args,
  1907→                    "reason": rule["reason"],
  1908→                    "trigger_findings": matching_findings[:5],
  1909→                    "trigger_count": len(matching_findings),
  1910→                })
