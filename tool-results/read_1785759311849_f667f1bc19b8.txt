    99→class FindingDetailModal(ModalScreen):
   100→    """Modal overlay showing full finding details."""
   101→
   102→    def __init__(self, finding: Dict[str, Any]) -> None:
   103→        super().__init__()
   104→        self.finding = finding
   105→
   106→    def compose(self) -> ComposeResult:
   107→        f = self.finding
   108→        sev = f.get("severity", "info").lower()
   109→        sev_color = SEV_STYLES.get(sev, "white")
   110→        title = f.get("title", "Unknown")
   111→        category = f.get("category", "")
   112→        module = f.get("module", "")
   113→        pts = f.get("points_deducted", 0)
   114→        evidence = f.get("evidence", "")
   115→        remediation = f.get("remediation", "No remediation available.")
   116→        description = f.get("description", "")
   117→        asset = f.get("asset", "")
   118→        related_cves = f.get("related_cves", [])
   119→
   120→        lines: List[str] = []
   121→        lines.append(f"[{sev_color}][{sev.upper()}] {title}[/]")
   122→        lines.append("")
   123→        if asset:
   124→            lines.append(f"[dim]Asset:[/] [cyan]{asset}[/]")
   125→        if category:
   126→            lines.append(f"[dim]Category:[/] {category}")
   127→        if module:
   128→            lines.append(f"[dim]Module:[/] {module}")
   129→        lines.append(f"[dim]Points:[/] -{pts}")
   130→        if related_cves:
   131→            cve_ids = ", ".join(
   132→                c.get("cve_id", "") for c in related_cves[:5] if c.get("cve_id")
   133→            )
   134→            if cve_ids:
   135→                lines.append(f"[dim]CVEs:[/] [red]{cve_ids}[/]")
   136→        lines.append("")
   137→        if description:
   138→            lines.append(f"[bold]Description:[/]")
   139→            lines.append(f"[dim]{description}[/]")
   140→            lines.append("")
   141→        if evidence:
   142→            lines.append(f"[bold]Evidence:[/]")
   143→            lines.append(f"[dim]{evidence}[/]")
   144→            lines.append("")
   145→        lines.append(f"[bold]{GREEN}Remediation:[/]")
   146→        lines.append(f"{GREEN}{remediation}[/]")
   147→
   148→        content = "\n".join(lines)
   149→
   150→        with Vertical(classes="modal-container"):
   151→            yield Label(content, classes="modal-content")
   152→            with Horizontal(classes="modal-actions"):
   153→                yield Button("Close", variant="primary", id="close-modal")
   154→
   155→    def on_button_pressed(self, event: Button.Pressed) -> None:
   156→        if event.button.id == "close-modal":
   157→            self.dismiss()
   158→

   160→# ════════════════════════════════════════════════════════════════════════════════
   161→# Boot Screen Widget
   162→# ════════════════════════════════════════════════════════════════════════════════
   163→
   164→
   165→class BootScreen(Widget):
   166→    """Full-screen animated boot sequence."""
   167→
   168→    def __init__(self) -> None:
   169→        super().__init__()
   170→        self._step = 0
   171→        self._spinner_idx = 0
   172→        self._module_idx = 0
   173→        self._module_names = list(MODULE_NAMES.values())
   174→        self._done = False
   175→        self._timer: Optional[Timer] = None
   176→
   177→    def compose(self) -> ComposeResult:
   178→        with Vertical(classes="boot-container"):
   179→            yield Static("", id="boot-line-1", classes="boot-title")
   180→            yield Static("", id="boot-line-2", classes="boot-line")
   181→            yield Static("", id="boot-line-3", classes="boot-line")
   182→            yield Static("", id="boot-line-4", classes="boot-line")
   183→
   184→    def on_mount(self) -> None:
   185→        self._timer = self.set_interval(0.04, self._tick)
   186→
   187→    def _tick(self) -> None:
   188→        app = self.app
   189→        if not isinstance(app, NexusApp):
   190→            return
   191→
   192→        line1 = app.query_one("#boot-line-1", Static)
   193→        line2 = app.query_one("#boot-line-2", Static)
   194→        line3 = app.query_one("#boot-line-3", Static)
   195→        line4 = app.query_one("#boot-line-4", Static)
   196→
   197→        if self._step == 0:
   198→            # Letter-by-letter title
   199→            title = "RECONPRO NEXUS"
   200→            chars = title[: self._spinner_idx]
   201→            self._spinner_idx += 1
   202→            line1.update(f"[{CYAN} bold]{chars}[/]")
   203→            if self._spinner_idx > len(title):
   204→                self._spinner_idx = 0
   205→                self._step = 1
   206→                line1.update(f"[{CYAN} bold]{title}[/]  [{DIM_CYAN}]v{VERSION}[/]")
   207→
   208→        elif self._step == 1:
   209→            # Spinner for "Initializing agent..."
   210→            frame = SPINNER_FRAMES[self._spinner_idx % len(SPINNER_FRAMES)]
   211→            self._spinner_idx += 1
   212→            line2.update(f"  [{CYAN}]{frame}[/] [{DIM_CYAN}]Initializing agent...[/]")
   213→            if self._spinner_idx > 30:
   214→                self._spinner_idx = 0
   215→                self._step = 2
   216→                line2.update(f"  [{GREEN}]✓[/] [{DIM_CYAN}]Agent initialized.[/]")
   217→
   218→        elif self._step == 2:
   219→            # Module names appearing one by one
   220→            if self._module_idx < len(self._module_names):
   221→                visible = self._module_names[: self._module_idx + 1]
   222→                self._module_idx += 1
   223→                modules_str = "  ".join(f"[{CYAN}]○[/] [{DIM_CYAN}]{m}[/]" for m in visible)
   224→                remaining = len(self._module_names) - self._module_idx
   225→                if remaining > 0:
   226→                    modules_str += f"  [{DIM_CYAN}]+{remaining} more...[/]"
   227→                frame = SPINNER_FRAMES[self._spinner_idx % len(SPINNER_FRAMES)]
   228→                self._spinner_idx += 1
   229→                line3.update(
   230→                    f"  [{CYAN}]{frame}[/] [{DIM_CYAN}]Loading {len(self._module_names)} modules...[/]\n  {modules_str}"
   231→                )
   232→            else:
   233→                all_mods = "  ".join(
   234→                    f"[{GREEN}]●[/] [{DIM_CYAN}]{m}[/]" for m in self._module_names
   235→                )
   236→                line3.update(
   237→                    f"  [{GREEN}]✓[/] [{DIM_CYAN}]All {len(self._module_names)} modules loaded.[/]\n  {all_mods}"
   238→                )
   239→                self._step = 3
   240→                self._spinner_idx = 0
   241→
   242→        elif self._step == 3:
   243→            # "Ready." with flash
   244→            if self._spinner_idx == 0:
   245→                line4.update(f"[{YELLOW} bold]  ▶  READY.[/]")
   246→            elif self._spinner_idx == 6:
   247→                line4.update(f"[{GREEN} bold]  ▶  READY.[/]")
   248→            elif self._spinner_idx >= 10:
   249→                self._done = True
   250→                if self._timer:
   251→                    self._timer.stop()
   252→                self.app.call_from_thread(self.app._finish_boot)
   253→            self._spinner_idx += 1
   254→

   256→# ════════════════════════════════════════════════════════════════════════════════
   257→# Module Status Cell Widget
   258→# ════════════════════════════════════════════════════════════════════════════════
   259→
   260→
   261→class ModuleCell(Static):
   262→    """Single module status cell in the grid."""
   263→
   264→    def __init__(self, module_id: str, name: str) -> None:
   265→        super().__init__()
   266→        self.module_id = module_id
   267→        self.module_name = name
   268→        self.status = "idle"  # idle | scanning | done | error
   269→        self.finding_count = 0
   270→        self._spinner_idx = 0
   271→        self._timer: Optional[Timer] = None
   272→
   273→    def on_mount(self) -> None:
   274→        self._update_display()
   275→        self._timer = self.set_interval(0.1, self._spin)
   276→
   277→    def _spin(self) -> None:
   278→        if self.status == "scanning":
   279→            self._spinner_idx = (self._spinner_idx + 1) % len(SPINNER_FRAMES)
   280→            self._update_display()
   281→
   282→    def _update_display(self) -> None:
   283→        name_short = self.module_name[:14]
   284→        if self.status == "idle":
   285→            icon = f"[{DIM_CYAN}]○[/]"
   286→            status_text = f"[{DIM_CYAN}]idle[/]"
   287→        elif self.status == "scanning":
   288→            frame = SPINNER_FRAMES[self._spinner_idx]
   289→            icon = f"[{CYAN}]{frame}[/]"
   290→            status_text = f"[{CYAN}]scanning[/]"
   291→        elif self.status == "done":
   292→            icon = f"[{GREEN}]✓[/]"
   293→            status_text = f"[{GREEN}]done ({self.finding_count})[/]"
   294→        elif self.status == "error":
   295→            icon = f"[{RED}]✗[/]"
   296→            status_text = f"[{RED}]error[/]"
   297→        else:
   298→            icon = "○"
   299→            status_text = "idle"
   300→
   301→        self.update(
   302→            f"{icon} [{CYAN} bold]{name_short}[/]\n  {status_text}"
   303→        )
   304→
   305→    def set_status(self, status: str, count: int = 0) -> None:
   306→        self.status = status
   307→        self.finding_count = count
   308→        self._update_display()
   309→
   310→    def on_unmount(self) -> None:
   311→        if self._timer:
   312→            self._timer.stop()
   313→

   315→# ════════════════════════════════════════════════════════════════════════════════
   316→# Pulsing Status Dot Widget
   317→# ════════════════════════════════════════════════════════════════════════════════
   318→
   319→
   320→class StatusDot(Static):
   321→    """Pulsing dot indicator for scan status."""
   322→
   323→    scanning: reactive[bool] = reactive(False)
   324→
   325→    def __init__(self) -> None:
   326→        super().__init__("")
   327→        self._phase = 0
   328→
   329→    def on_mount(self) -> None:
   330→        self.set_interval(0.3, self._pulse)
   331→
   332→    def watch_scanning(self, old: bool, new: bool) -> None:
   333→        # Update visual state based on scanning status
   334→        if new:
   335→            self.update(f"[{self._get_pulse_color()}]●[/]")
   336→        else:
   337→            self.update(f"[{GREEN}]●[/]")
   338→
   339→    def _get_pulse_color(self) -> str:
   340→        return ["#003322", "#006644", CYAN, "#006644"][self._phase] if self.scanning else GREEN
   341→
   342→    def _pulse(self) -> None:
   343→        self._phase = (self._phase + 1) % 4
   344→        if self.scanning:
   345→            brightness = self._get_pulse_color()
   346→            self.update(f"[{brightness}]●[/]")
   347→        else:
   348→            self.update(f"[{GREEN}]●[/]")

   746→    def on_key(self, event: Key) -> None:
   747→        """Handle key events — boot skip + completer + history navigation."""
   748→        # Skip boot on ANY key press during boot sequence
   749→        if not self._boot_done and not self._boot_skipped:
   750→            self._boot_skipped = True
   751→            try:
   752→                boot = self.query_one("#boot-screen", BootScreen)
   753→                if boot._timer:
   754→                    boot._timer.stop()
   755→            except NoMatches:
   756→                pass
   757→            self._finish_boot()
   758→            return
   759→
   760→        # ── Tab: completer accept/cycle (when input focused + completer visible) ──
   761→        if event.key == "tab":
   762→            try:
   763→                inp = self.query_one("#command-input", Input)
   764→                comp = self.query_one("#cmd-completer", CommandCompleter)
   765→                if inp.has_focus and comp.visible:
   766→                    accepted = comp.accept_top()
   767→                    if accepted:
   768→                        # Replace the first word with the accepted command
   769→                        parts = inp.value.split()
   770→                        rest = parts[1:] if len(parts) > 1 else []
   771→                        inp.value = accepted + (" " + " ".join(rest) if rest else "")
   772→                        inp.cursor_position = len(inp.value)
   773→                    event.stop()
   774→                    return
   775→            except NoMatches:
   776→                pass
   777→            # Fall through to default Tab behavior (cycle focus)
   778→            return
   779→
   780→        # ── Escape: dismiss completer ──
   781→        if event.key == "escape":
   782→            try:
   783→                comp = self.query_one("#cmd-completer", CommandCompleter)
   784→                if comp.visible:
   785→                    comp.hide()
   786→                    event.stop()
   787→                    return
   788→            except NoMatches:
   789→                pass
   790→
   791→        # ── Up/Down in completer: cycle suggestions ──
   792→        if event.key in ("up", "down"):
   793→            try:
   794→                comp = self.query_one("#cmd-completer", CommandCompleter)
   795→                inp = self.query_one("#command-input", Input)
   796→                if comp.visible and inp.has_focus:
   797→                    direction = -1 if event.key == "up" else 1
   798→                    comp.cycle_selection(direction)
   799→                    event.stop()
   800→                    return
   801→            except NoMatches:
   802→                pass
   803→
   804→        # Command history navigation
   805→        if event.key == "up" and self._history_index > 0:
   806→            try:
   807→                inp = self.query_one("#command-input", Input)
   808→                if inp.has_focus:
   809→                    self._history_index -= 1
   810→                    inp.value = self._command_history[self._history_index]
   811→            except NoMatches:
   812→                pass
   813→        elif event.key == "down":
   814→            try:
   815→                inp = self.query_one("#command-input", Input)
   816→                if inp.has_focus:
   817→                    if self._history_index < len(self._command_history) - 1:
   818→                        self._history_index += 1
   819→                        inp.value = self._command_history[self._history_index]
   820→                    else:
   821→                        self._history_index = len(self._command_history)
   822→                        inp.value = ""
   823→            except NoMatches:
   824→                pass

   921→            self._finding_timestamps.append(now)

   923→        # Push to findings sparkline (true findings-per-minute)
   924→        try:
   925→            spark = self.query_one("#spark-findings", Sparkline)

   935→        try:
   936→            vm = self.query_one("#velocity-meter", VelocityMeter)

   938→                vm.record_finding(delta)

   940→            pass

   942→        # Keep bottom-info label too (compact view)

   944→            el = self.query_one("#bottom-info", Label)

  1150→    def _process_command(self, raw: str) -> None:
  1151→        """Parse and execute a user command."""
  1152→        self._chat(f"[{CYAN} bold]▸ {raw}[/]")
  1153→
  1154→        parts = raw.split()
  1155→        cmd = parts[0].lower()
  1156→        args = parts[1:]

  1305→    def _handle_scan(self, args: List[str]) -> None:
  1306→        if not args:
  1307→            self._chat(f"[{RED}]Usage: scan <target> [with <mod1> <mod2> ...][/]")
  1308→            return
  1309→
  1310→        # Parse "with" clause
  1311→        modules: Optional[List[str]] = None
  1312→        target_parts: List[str] = []
  1313→        found_with = False
  1314→        mod_parts: List[str] = []
  1315→        for a in args:
  1316→            if a.lower() == "with":
  1317→                found_with = True
  1318→                continue
  1319→            if found_with:
  1320→                mod_parts.append(a)
  1321→            else:
  1322→                target_parts.append(a)
  1323→
  1324→        target = " ".join(target_parts)
  1325→        if not target:
  1326→            self._chat(f"[{RED}]No target specified.[/]")
  1327→            return
  1328→
  1329→        if mod_parts:
  1330→            modules = [m.strip().lower() for m in mod_parts]
  1331→            self._chat(f"[{DIM_CYAN}]Scanning [cyan bold]{target}[/] with modules: {', '.join(modules)}[/]")
  1332→        else:
  1333→            self._chat(f"[{DIM_CYAN}]Scanning [cyan bold]{target}[/]...[/]")
  1334→
  1335→        self.current_target = target
  1336→        self._run_scan_worker(target, modules=modules)

  1417→    @work(exclusive=False, thread=True)
  1418→    def _run_scan_worker(
  1419→        self,
  1420→        target: str,
  1421→        modules: Optional[List[str]] = None,
  1422→        is_local: bool = False,
  1423→    ) -> None:
  1424→        """Run a scan in a background thread."""
  1425→        self.call_from_thread(self.is_scanning.set, True)
  1426→
  1427→        # Set modules to scanning
  1428→        if modules:
  1429→            for m in modules:
  1430→                if m in self._module_cells:
  1431→                    self.call_from_thread(self._module_cells[m].set_status, "scanning")
  1432→        else:
  1433→            all_mods = (
  1434→                list(MODULE_NAMES.keys())
  1435→                if is_local
  1436→                else [k for k in MODULE_NAMES if k not in ("host", "dev", "doctor")]
  1437→            )
  1438→            for m in all_mods:
  1439→                if m in self._module_cells:
  1440→                    self.call_from_thread(self._module_cells[m].set_status, "scanning")
  1441→
  1442→        try:
  1443→            from .scanner import scan, audit_scan
  1444→            from .history import save_scan
  1445→
  1446→            self.call_from_thread(
  1447→                self._chat, f"[{DIM_CYAN}]  ⏳ Scan in progress...[/]"
  1448→            )
  1449→
  1450→            if is_local:
  1451→                result = audit_scan(target=target, modules=modules)
  1452→            else:
  1453→                result = scan(target=target, modules=modules)
  1454→
  1455→            # Process findings
  1456→            data = result.to_dict()
  1457→            self.call_from_thread(self._on_scan_complete, data, modules, is_local)
  1458→
  1459→            # Save to history
  1460→            try:
  1461→                save_scan(data, label="nexus")
  1462→            except Exception:
  1463→                pass
  1464→
  1465→        except Exception as e:
  1466→            self.call_from_thread(self._chat, f"[{RED}]  ✗ Scan error: {e}[/]")
  1467→            if modules:
  1468→                for m in modules:
  1469→                    if m in self._module_cells:
  1470→                        self.call_from_thread(self._module_cells[m].set_status, "error")
  1471→            else:
  1472→                self.call_from_thread(self._set_all_modules_status, "error")
  1473→
  1474→        finally:
  1475→            self.call_from_thread(self.is_scanning.set, False)

  1828→    def on_click(self, event: Click) -> None:
  1829→        """Open finding detail modal for the clicked finding.
  1830→
  1831→        Uses the vertical click position to determine which finding index
  1832→        the user clicked on, mapping it to _findings_list.
  1833→        """
  1834→        try:
  1835→            if (
  1836→                isinstance(event.widget, RichLog)
  1837→                and event.widget.id == "findings-feed"
  1838→                and self._findings_list
  1839→            ):
  1840→                # Get the line offset to determine which finding was clicked.
  1841→                # Each finding entry is 1 line in the RichLog.
  1842→                # We use the y coordinate relative to the widget to estimate the index.
  1843→                region = event.widget.region
  1844→                if region is not None:
  1845→                    # Calculate which line was clicked relative to the scroll position
  1846→                    scroll_offset = event.widget.scroll_y
  1847→                    click_y = event.y - region.y
  1848→                    # Each finding takes 1 line; estimate index from click position
  1849→                    # Account for scroll position
  1850→                    line_height = 1
  1851→                    if hasattr(event.widget, "_line_height"):
  1852→                        line_height = event.widget._line_height
  1853→                    clicked_index = int((click_y + scroll_offset * line_height) / line_height)
  1854→                    # Clamp to valid range
  1855→                    clicked_index = max(0, min(clicked_index, len(self._findings_list) - 1))
  1856→                    finding = self._findings_list[clicked_index]
  1857→                else:
  1858→                    finding = self._findings_list[-1]
  1859→                self.push_screen(FindingDetailModal(finding))
  1860→        except (NoMatches, Exception):
  1861→            pass
