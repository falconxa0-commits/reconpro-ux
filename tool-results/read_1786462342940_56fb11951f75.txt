   139→class EngineeringMetrics:
   140→    """Collect, track, and persist engineering metrics.
   141→
   142→    Tracks test pass rate, coverage trend, debt ratio, and complexity
   143→    trend over time. Persists snapshots to JSON files at
   144→    ``RECONPRO_HOME/metrics/``.
   145→
   146→    Parameters
   147→    ----------
   148→    repo_path : str | Path
   149→        Root directory of the repository to analyze.
   150→    metrics_dir : Path | None
   151→        Directory for persisted metrics. Defaults to ``METRICS_DIR``.
   152→    """
   153→
   154→    def __init__(
   155→        self,
   156→        repo_path: str | Path = ".",
   157→        metrics_dir: Optional[Path] = None,
   158→    ) -> None:
   159→        self._repo = Path(repo_path).resolve()
   160→        self._metrics_dir = metrics_dir or METRICS_DIR
   161→        self._metrics_dir.mkdir(parents=True, exist_ok=True)
   162→        self._history_file = self._metrics_dir / "engineering_metrics.json"
   163→        self._snapshots: List[MetricsSnapshot] = []
   164→        self._load_history()
   165→
   166→    # -- Persistence -----------------------------------------------------
   167→
   168→    def _load_history(self) -> None:
   169→        """Load historical metrics from disk."""
   170→        if not self._history_file.exists():
   171→            return
   172→        try:
   173→            with open(self._history_file, "r", encoding="utf-8") as f:
   174→                data = json.load(f)
   175→            raw_list = data.get("snapshots", [])
   176→            self._snapshots = [MetricsSnapshot.from_dict(s) for s in raw_list]
   177→        except (json.JSONDecodeError, OSError, KeyError, TypeError):
   178→            logger.debug("Could not load metrics history", exc_info=True)
   179→            self._snapshots = []
   180→
   181→    def _save_history(self) -> None:
   182→        """Persist current metrics to disk."""
   183→        try:
   184→            data = {
   185→                "repo_path": str(self._repo),
   186→                "updated": datetime.now(timezone.utc).isoformat(),
   187→                "total_snapshots": len(self._snapshots),
   188→                "snapshots": [s.to_dict() for s in self._snapshots],
   189→            }
   190→            with open(self._history_file, "w", encoding="utf-8") as f:
   191→                json.dump(data, f, indent=2, ensure_ascii=False)
   192→        except OSError:
   193→            logger.debug("Could not save metrics history", exc_info=True)
   194→
   195→    # -- Metric Collection ------------------------------------------------
   196→
   197→    def collect_current_metrics(self) -> MetricsSnapshot:
   198→        """Analyze the repository and return a fresh metrics snapshot.
   199→
   200→        Walks the repository tree, counts files and lines, estimates
   201→        cyclomatic complexity for Python files, and calculates a
   202→        synthetic debt ratio.
   203→        """
   204→        t0 = time.perf_counter()
   205→        file_count = 0
   206→        total_lines = 0
   207→        source_lines = 0
   208→        complexity_values: List[float] = []
   209→        debt_markers = 0
   210→
   211→        for fpath in self._walk_source_files():
   212→            file_count += 1
   213→            try:
   214→                content = fpath.read_text(encoding="utf-8", errors="ignore")
   215→            except OSError:
   216→                continue
   217→
   218→            lines = content.splitlines()
   219→            line_count = len(lines)
   220→            total_lines += line_count
   221→
   222→            if fpath.suffix in _SOURCE_EXTENSIONS:
   223→                source_lines += line_count
   224→
   225→                # Estimate complexity for Python files
   226→                if fpath.suffix == ".py":
   227→                    cx = self._estimate_complexity(lines)
   228→                    complexity_values.append(cx)
   229→
   230→                    # Count debt markers (TODO, FIXME, HACK, XXX)
   231→                    for line in lines:
   232→                        stripped = line.strip().upper()
   233→                        if stripped.startswith(("# TODO", "# FIXME", "# HACK", "# XXX")):
   234→                            debt_markers += 1
   235→
   236→        # Calculate aggregate metrics
   237→        complexity_avg = (
   238→            sum(complexity_values) / len(complexity_values)
   239→            if complexity_values else 0.0
   240→        )
   241→        complexity_max = max(complexity_values) if complexity_values else 0.0
   242→        debt_ratio = (
   243→            (debt_markers / source_lines * 100.0)
   244→            if source_lines > 0 else 0.0
   245→        )
   246→
   247→        # Cap debt_ratio at 100
   248→        debt_ratio = min(debt_ratio, 100.0)
   249→
   250→        elapsed_ms = (time.perf_counter() - t0) * 1000.0
   251→
   252→        snapshot = MetricsSnapshot(
   253→            timestamp=datetime.now(timezone.utc).isoformat(),
   254→            debt_ratio=debt_ratio,
   255→            debt_items=debt_markers,
   256→            complexity_avg=complexity_avg,
   257→            complexity_max=complexity_max,
   258→            file_count=file_count,
   259→            total_lines=total_lines,
   260→            source_lines=source_lines,
   261→            pipeline_duration_ms=elapsed_ms,
   262→        )
   263→
   264→        return snapshot
   265→
   266→    def record_snapshot(self, snapshot: MetricsSnapshot) -> None:
   267→        """Append a snapshot to history and persist."""
   268→        self._snapshots.append(snapshot)
   269→        # Keep last 100 snapshots to prevent unbounded growth
   270→        if len(self._snapshots) > 100:
   271→            self._snapshots = self._snapshots[-100:]
   272→        self._save_history()
   273→

   290→
   291→    def get_test_pass_rate_trend(self, n: int = 10) -> List[float]:

   296→    def get_latest_snapshot(self) -> Optional[MetricsSnapshot]:

   304→    # -- Helpers ----------------------------------------------------------
   305→
   306→    def _walk_source_files(self) -> List[Path]:
   307→        """Yield all source files in the repository, skipping _SKIP_DIRS."""
   308→        files: List[Path] = []
   309→        if not self._repo.is_dir():
   310→            return files
   311→        try:
   312→            for root, dirs, filenames in os.walk(self._repo):
   313→                # Prune skipped directories in-place
   314→                dirs[:] = [
   315→                    d for d in dirs
   316→                    if d not in _SKIP_DIRS and not d.startswith(".")
   317→                    if d not in {"__pycache__", "node_modules"}
   318→                ]
   319→                dirs[:] = [
   320→                    d for d in dirs
   321→                    if d not in _SKIP_DIRS
   322→                ]
   323→                for fname in sorted(filenames):
   324→                    fpath = Path(root) / fname
   325→                    if fpath.suffix in _SOURCE_EXTENSIONS:
   326→                        files.append(fpath)
   327→        except OSError:
   328→            logger.debug("Error walking repository", exc_info=True)
   329→        return files
   330→
   331→    @staticmethod
   332→    def _estimate_complexity(lines: List[str]) -> float:
   333→        """Estimate cyclomatic complexity for a Python file.
   334→
   335→        Counts branch-inducing keywords: ``if``, ``elif``, ``for``,
   336→        ``while``, ``except``, ``with``, ``and``, ``or``. Adds 1 for
   337→        the base path.
   338→
   339→        Ignores lines inside comments and strings (best-effort heuristic).
   340→        """
   341→        complexity = 1.0  # Base path
   342→        branch_keywords = re.compile(
   343→            r"\b(if|elif|for|while|except|with)\b|\b(and|or)\b"
   344→        )
   345→        in_multiline_string = False
   346→        for line in lines:
   347→            stripped = line.strip()
   348→            # Simple heuristic for triple-quoted strings
   349→            triple_count = stripped.count('"""') + stripped.count("'''")
   350→            if triple_count % 2 != 0:
   351→                in_multiline_string = not in_multiline_string
   352→            if in_multiline_string:
   353→                continue
   354→            # Skip comment-only lines
   355→            if stripped.startswith("#"):
   356→                continue
   357→            complexity += len(branch_keywords.findall(stripped))
   358→        return complexity
   359→

   361→        """Return the full metrics state as a dictionary."""

   371→            "complexity_trend": self.get_complexity_trend(),
   372→            "test_pass_rate_trend": self.get_test_pass_rate_trend(),
   373→        }

   488→class EngineeringBaseline:
   489→    """Manage engineering baselines for a repository.
   490→
   491→    Captures snapshots of repository state (file manifests with
   492→    hashes and complexity estimates), compares current state against
   493→    baselines, and tracks regression status.
   494→
   495→    Baselines are persisted to ``RECONPRO_HOME/baselines/<name>.json``.
   496→
   497→    Parameters
   498→    ----------
   499→    repo_path : str | Path
   500→        Root directory of the repository.
   501→    baseline_dir : Path | None
   502→        Directory for persisted baselines.
   503→    """
   504→
   505→    def __init__(
   506→        self,
   507→        repo_path: str | Path = ".",
   508→        baseline_dir: Optional[Path] = None,
   509→    ) -> None:
   510→        self._repo = Path(repo_path).resolve()
   511→        self._baseline_dir = baseline_dir or BASELINE_DIR
   512→        self._baseline_dir.mkdir(parents=True, exist_ok=True)
   513→
   514→    # -- Public API -------------------------------------------------------
   515→
   516→    def capture_snapshot(self, name: str = "") -> BaselineSnapshot:
   517→        """Capture a baseline snapshot of the current repository state.
   518→
   519→        Parameters
   520→        ----------
   521→        name : str
   522→            Baseline name. Defaults to ``"auto_<timestamp>"``.
   523→
   524→        Returns
   525→        -------
   526→        BaselineSnapshot
   527→            The captured baseline.
   528→        """
   529→        if not name:
   530→            name = f"auto_{datetime.now(timezone.utc).strftime('%Y%m%d_%H%M%S')}"
   531→
   532→        fingerprints: List[FileFingerprint] = []
   533→        total_lines = 0
   534→        total_size = 0
   535→        complexity_values: List[float] = []
   536→        hash_parts: List[str] = []
   537→
   538→        metrics = EngineeringMetrics(repo_path=self._repo)
   539→        source_files = metrics._walk_source_files()
   540→
   541→        for fpath in source_files:
   542→            rel_path = str(fpath.relative_to(self._repo))
   543→            try:
   544→                content = fpath.read_bytes()
   545→            except OSError:
   546→                continue
   547→
   548→            sha = hashlib.sha256(content).hexdigest()
   549→            text_content = content.decode("utf-8", errors="ignore")
   550→            lines = text_content.splitlines()
   551→            line_count = len(lines)
   552→
   553→            complexity = 0.0
   554→            if fpath.suffix == ".py":
   555→                complexity = EngineeringMetrics._estimate_complexity(lines)
   556→
   557→            fp = FileFingerprint(
   558→                path=rel_path,
   559→                size_bytes=len(content),
   560→                line_count=line_count,
   561→                sha256=sha,
   562→                complexity=complexity,
   563→            )
   564→            fingerprints.append(fp)
   565→            total_lines += line_count
   566→            total_size += len(content)
   567→            if complexity > 0:
   568→                complexity_values.append(complexity)
   569→            hash_parts.append(f"{rel_path}:{sha[:16]}")
   570→
   571→        avg_complexity = (
   572→            sum(complexity_values) / len(complexity_values)
   573→            if complexity_values else 0.0
   574→        )
   575→        max_complexity = max(complexity_values) if complexity_values else 0.0
   576→        composite_hash = hashlib.sha256(
   577→            "\n".join(sorted(hash_parts)).encode()
   578→        ).hexdigest()[:32]
   579→
   580→        snapshot = BaselineSnapshot(
   581→            name=name,
   582→            timestamp=datetime.now(timezone.utc).isoformat(),
   583→            repo_path=str(self._repo),
   584→            files=sorted(fingerprints, key=lambda f: f.path),
   585→            total_files=len(fingerprints),
   586→            total_lines=total_lines,
   587→            total_size_bytes=total_size,
   588→            avg_complexity=avg_complexity,
   589→            max_complexity=max_complexity,
   590→            composite_hash=composite_hash,
   591→        )
   592→
   593→        self._save_baseline(snapshot)
   594→        return snapshot
   595→
   596→    def compare(
   597→        self,
   598→        baseline_name: Optional[str] = None,
   599→        baseline: Optional[BaselineSnapshot] = None,
   600→    ) -> DriftReport:
   601→        """Compare current repository state against a baseline.
   602→
   603→        Parameters
   604→        ----------
   605→        baseline_name : str | None
   606→            Name of a previously saved baseline.
   607→        baseline : BaselineSnapshot | None
   608→            An in-memory baseline object. Takes precedence over
   609→            *baseline_name*.
   610→
   611→        Returns
   612→        -------
   613→        DriftReport
   614→            Detailed comparison results.
   615→        """
   616→        if baseline is None:
   617→            if baseline_name is None:
   618→                # Use the most recent baseline
   619→                baseline = self._load_latest_baseline()
   620→            else:
   621→                baseline = self.load_baseline(baseline_name)
   622→
   623→        if baseline is None:
   624→            return DriftReport(
   625→                baseline_name=baseline_name or "",
   626→                current_timestamp=datetime.now(timezone.utc).isoformat(),
   627→                regression_details=["No baseline found for comparison"],
   628→            )
   629→
   630→        # Build current state index
   631→        current_fps: Dict[str, FileFingerprint] = {}
   632→        metrics = EngineeringMetrics(repo_path=self._repo)
   633→        for fpath in metrics._walk_source_files():
   634→            rel_path = str(fpath.relative_to(self._repo))
   635→            try:
   636→                content = fpath.read_bytes()
   637→            except OSError:
   638→                continue
   639→            text_content = content.decode("utf-8", errors="ignore")
   640→            lines = text_content.splitlines()
   641→            complexity = 0.0
   642→            if fpath.suffix == ".py":
   643→                complexity = EngineeringMetrics._estimate_complexity(lines)
   644→            current_fps[rel_path] = FileFingerprint(
   645→                path=rel_path,
   646→                size_bytes=len(content),
   647→                line_count=len(lines),
   648→                sha256=hashlib.sha256(content).hexdigest(),
   649→                complexity=complexity,
   650→            )
   651→
   652→        # Build baseline index
   653→        baseline_map: Dict[str, FileFingerprint] = {
   654→            f.path: f for f in baseline.files
   655→        }
   656→
   657→        current_paths = set(current_fps.keys())
   658→        baseline_paths = set(baseline_map.keys())
   659→
   660→        added = sorted(current_paths - baseline_paths)
   661→        removed = sorted(baseline_paths - current_paths)
   662→        common = current_paths & baseline_paths
   663→
   664→        modified: List[str] = []
   665→        drift_events: List[Dict[str, Any]] = []
   666→        regression_details: List[str] = []
   667→
   668→        for path in sorted(common):
   669→            cur = current_fps[path]
   670→            base = baseline_map[path]
   671→
   672→            if cur.sha256 != base.sha256:
   673→                modified.append(path)
   674→                event: Dict[str, Any] = {
   675→                    "path": path,
   676→                    "type": "CONTENT_CHANGE",
   677→                    "severity": "low",
   678→                }
   679→
   680→                # Detect regression: significant complexity increase
   681→                if cur.complexity > 0 and base.complexity > 0:
   682→                    cx_delta = cur.complexity - base.complexity
   683→                    if cx_delta > 10:
   684→                        event["severity"] = "high"
   685→                        event["detail"] = (
   686→                            f"Complexity surged by {cx_delta:.1f} "
   687→                            f"(from {base.complexity:.1f} to {cur.complexity:.1f})"
   688→                        )
   689→                        regression_details.append(event["detail"])
   690→                    elif cx_delta > 5:
   691→                        event["severity"] = "medium"
   692→                        event["detail"] = (
   693→                            f"Complexity increased by {cx_delta:.1f} "
   694→                            f"(from {base.complexity:.1f} to {cur.complexity:.1f})"
   695→                        )
   696→
   697→                # Detect regression: significant line count increase
   698→                line_delta = cur.line_count - base.line_count
   699→                if line_delta > 100:
   700→                    event["severity"] = max(
   701→                        event.get("severity", "low"), "medium"
   702→                    )
   703→                    if "detail" not in event:
   704→                        event["detail"] = (
   705→                            f"File grew by {line_delta} lines"
   706→                        )
   707→
   708→                drift_events.append(event)
   709→
   710→        # New files add complexity
   711→        for path in added:
   712→            cur = current_fps[path]
   713→            if cur.complexity > 15:
   714→                regression_details.append(
   715→                    f"New file {path} has high complexity ({cur.complexity:.1f})"
   716→                )
   717→
   718→        # Removed files might be regression (deleted tests, docs)
   719→        for path in removed:
   720→            base = baseline_map[path]
   721→            if "test" in path.lower():
   722→                regression_details.append(
   723→                    f"Test file removed: {path}"
   724→                )
   725→
   726→        # Calculate drift score
   727→        total_changes = len(added) + len(removed) + len(modified)
   728→        drift_score = self._calculate_drift_score(
   729→            added=added,
   730→            removed=removed,
   731→            modified=modified,
   732→            drift_events=drift_events,
   733→            regression_details=regression_details,
   734→        )
   735→
   736→        return DriftReport(
   737→            baseline_name=baseline.name,
   738→            baseline_timestamp=baseline.timestamp,
   739→            current_timestamp=datetime.now(timezone.utc).isoformat(),
   740→            files_added=added,
   741→            files_removed=removed,
   742→            files_modified=modified,
   743→            drift_events=drift_events,
   744→            drift_score=drift_score,
   745→            regression_detected=len(regression_details) > 0,
   746→            regression_details=regression_details,
   747→        )
   748→
   749→    def load_baseline(self, name: str) -> Optional[BaselineSnapshot]:
   750→        """Load a named baseline from disk."""
   751→        path = self._baseline_dir / f"{name}.json"
   752→        if not path.exists():
   753→            return None
   754→        try:
   755→            with open(path, "r", encoding="utf-8") as f:
   756→                data = json.load(f)
   757→            return BaselineSnapshot.from_dict(data)
   758→        except (json.JSONDecodeError, OSError, KeyError, TypeError):
   759→            logger.debug("Could not load baseline %s", name, exc_info=True)
   760→            return None
   761→
   762→    def list_baselines(self) -> List[Dict[str, str]]:
   763→        """List all saved baselines with name and timestamp."""
   764→        results: List[Dict[str, str]] = []
   765→        if not self._baseline_dir.exists():
   766→            return results
   767→        for fpath in sorted(self._baseline_dir.glob("*.json")):
   768→            baseline = self.load_baseline(fpath.stem)
   769→            if baseline:
   770→                results.append({
   771→                    "name": baseline.name,
   772→                    "timestamp": baseline.timestamp,
   773→                    "total_files": str(baseline.total_files),
   774→                    "total_lines": str(baseline.total_lines),
   775→                })
   776→        return results
   777→
   778→    def delete_baseline(self, name: str) -> bool:
   779→        """Delete a saved baseline. Returns True on success."""
   780→        path = self._baseline_dir / f"{name}.json"
   781→        if not path.exists():
   782→            return False
   783→        try:
   784→            path.unlink()
   785→            return True
   786→        except OSError:
   787→            return False
   788→
   789→    # -- Internal ---------------------------------------------------------
   790→
   791→    def _save_baseline(self, snapshot: BaselineSnapshot) -> None:
   792→        """Persist a baseline to disk."""
   793→        path = self._baseline_dir / f"{snapshot.name}.json"
   794→        try:
   795→            with open(path, "w", encoding="utf-8") as f:
   796→                json.dump(snapshot.to_dict(), f, indent=2, ensure_ascii=False)
   797→        except OSError:
   798→            logger.debug("Could not save baseline %s", snapshot.name, exc_info=True)
   799→
   800→    def _load_latest_baseline(self) -> Optional[BaselineSnapshot]:
   801→        """Load the most recently created baseline."""
   802→        baselines = self.list_baselines()
   803→        if not baselines:
   804→            return None
   805→        # list_baselines returns sorted by filename; get the last one
   806→        latest = baselines[-1]
   807→        return self.load_baseline(latest["name"])
   808→
   809→    @staticmethod
   810→    def _calculate_drift_score(
   811→        added: List[str],
   812→        removed: List[str],
   813→        modified: List[str],
   814→        drift_events: List[Dict[str, Any]],
   815→        regression_details: List[str],
   816→    ) -> float:
   817→        """Calculate a 0-100 drift score.
   818→
   819→        Higher score = more drift. Regression indicators boost the score.
   820→        """
   821→        if not added and not removed and not modified:
   822→            return 0.0
   823→
   824→        # Base score from change volume
   825→        total_changes = len(added) + len(removed) + len(modified)
   826→        volume_score = min(total_changes * 2.0, 40.0)
   827→
   828→        # Severity weighting
   829→        severity_weights = {"low": 1.0, "medium": 2.0, "high": 3.0, "critical": 5.0}
   830→        event_score = 0.0
   831→        for event in drift_events:
   832→            sev = event.get("severity", "low")
   833→            event_score += severity_weights.get(sev, 1.0)
   834→        event_score = min(event_score, 30.0)
   835→
   836→        # Regression penalty
   837→        regression_penalty = min(len(regression_details) * 5.0, 30.0)
   838→
   839→        return min(volume_score + event_score + regression_penalty, 100.0)

   870→class EngineeringPipeline:
   871→    """Orchestrate automated engineering workflows.

   874→    metrics collection into a unified pipeline. Integrates with

   878→    Parameters

   884→    metrics_dir : Path | None

   910→    def health_check(self) -> Dict[str, Any]:
   911→        """Run repository health diagnostics.
   912→
   913→        Delegates to ``diagnostics.health_check()``,
   914→        ``diagnostics.validate_config()``, and
   915→        ``diagnostics.run_diagnostics()`` for comprehensive results.
   916→
   917→        Returns
   918→        -------
   919→        dict
   920→            Structured health report with status per category,
   921→            config validation issues, and summary statistics.
   922→        """
   923→        result: Dict[str, Any] = {
   924→            "timestamp": datetime.now(timezone.utc).isoformat(),
   925→            "repo_path": str(self._repo),
   926→        }
   927→
   928→        # 1. Quick health check from diagnostics module
   929→        try:
   930→            from .diagnostics import health_check as diag_health
   931→            result["health"] = diag_health()
   932→        except Exception as exc:
   933→            logger.debug("Health check error: %s", exc, exc_info=True)
   934→            result["health"] = {"error": str(exc)}
   935→
   936→        # 2. Config validation from diagnostics module
   937→        try:
   938→            from .diagnostics import validate_config
   939→            result["config_issues"] = validate_config()
   940→        except Exception as exc:
   941→            logger.debug("Config validation error: %s", exc, exc_info=True)
   942→            result["config_issues"] = []
   943→
   944→        # 3. Full diagnostics from diagnostics module
   945→        try:
   946→            from .diagnostics import run_diagnostics
   947→            diag_result = run_diagnostics()

   949→                "total": diag_result.get("total_checks", 0),
   950→                "passed": diag_result.get("passed", 0),
   951→                "failed": diag_result.get("failed", 0),
   952→                "checks": diag_result.get("checks", []),

   957→            result["diagnostics"] = {"error": str(exc)}
   958→            result["health_score"] = 0.0
   959→
   960→        # 4. Repository-specific checks
   961→        repo_health = self._check_repository_health()
   962→        result["repository"] = repo_health
   963→
   964→        return result

   966→    def drift_detection(
   967→        self,
   968→        baseline_name: Optional[str] = None,
   969→        create_baseline_if_missing: bool = True,
   970→    ) -> Dict[str, Any]:
   971→        """Detect code drift from baselines.
   972→
   973→        Compares the current repository state against the most recent
   974→        baseline (or a named one). If no baseline exists and
   975→        *create_baseline_if_missing* is True, captures one automatically.
   976→
   977→        Parameters
   978→        ----------
   979→        baseline_name : str | None
   980→            Specific baseline to compare against.
   981→        create_baseline_if_missing : bool
   982→            Auto-capture a baseline if none exists.
   983→
   984→        Returns
   985→        -------
   986→        dict
   987→            Drift detection results including drift score, changed files,
   988→            and regression details.
   989→        """
   990→        result: Dict[str, Any] = {
   991→            "timestamp": datetime.now(timezone.utc).isoformat(),
   992→            "repo_path": str(self._repo),
   993→        }
   994→
   995→        # Check infrastructure drift monitor availability
   996→        try:
   997→            from .drift_monitor import InfrastructureDriftMonitor
   998→            result["infra_drift_available"] = True
   999→        except ImportError:
  1000→            result["infra_drift_available"] = False
  1001→
  1002→        # Check if we have any baselines
  1003→        existing = self._baseline.list_baselines()
  1004→        if not existing and create_baseline_if_missing:
  1005→            snapshot = self._baseline.capture_snapshot(
  1006→                name=f"initial_{datetime.now(timezone.utc).strftime('%Y%m%d_%H%M%S')}"
  1007→            )
  1008→            result["auto_baseline_created"] = True
  1009→            result["baseline"] = snapshot.to_dict()
  1010→            result["drift"] = {
  1011→                "drift_score": 0.0,
  1012→                "message": "Initial baseline captured; no drift to report",
  1013→            }
  1014→            return result
  1015→
  1016→        # Run drift comparison
  1017→        drift_report = self._baseline.compare(baseline_name=baseline_name)
  1018→        result["drift"] = drift_report.to_dict()
  1019→
  1020→        return result

  1022→    def quality_gate(
  1023→        self,
  1024→        metrics_snapshot: Optional[MetricsSnapshot] = None,
  1025→        drift_report: Optional[DriftReport] = None,
  1026→        health_result: Optional[Dict[str, Any]] = None,
  1027→    ) -> QualityGateResult:
  1028→        """Enforce quality thresholds before merge.
  1029→
  1030→        Evaluates metrics, drift, and health against configured
  1031→        thresholds. Returns a detailed pass/fail result.
  1032→
  1033→        Parameters
  1034→        ----------
  1035→        metrics_snapshot : MetricsSnapshot | None
  1036→            Current metrics. Collected automatically if None.
  1037→        drift_report : DriftReport | None
  1038→            Drift comparison result. Run automatically if None.
  1039→        health_result : dict | None
  1040→            Health check result. Run automatically if None.
  1041→
  1042→        Returns
  1043→        -------
  1044→        QualityGateResult
  1045→            Pass/fail result with per-threshold evaluations.
  1046→        """
  1047→        # Collect data if not provided
  1048→        if metrics_snapshot is None:
  1049→            metrics_snapshot = self._metrics.collect_current_metrics()
  1050→
  1051→        if drift_report is None:
  1052→            drift_report = self._baseline.compare()
  1053→
  1054→        if health_result is None:
  1055→            health_result = self.health_check()
  1056→
  1057→        evaluations: Dict[str, Dict[str, Any]] = {}
  1058→        failures: List[str] = []
  1059→        warnings: List[str] = []
  1060→        score_deductions = 0.0
  1061→
  1062→        # 1. Test pass rate gate
  1063→        test_rate = metrics_snapshot.test_pass_rate
  1064→        min_test_rate = self._thresholds.get("min_test_pass_rate", 80.0)
  1065→        test_pass = test_rate >= min_test_rate or test_rate == 0.0  # 0 = no tests run, not a failure
  1066→        evaluations["test_pass_rate"] = {
  1067→            "value": test_rate,
  1068→            "threshold": min_test_rate,
  1069→            "passed": test_pass,
  1070→        }
  1071→        if not test_pass and test_rate > 0:
  1072→            failures.append(
  1073→                f"Test pass rate {test_rate:.1f}% below threshold {min_test_rate:.1f}%"
  1074→            )
  1075→            score_deductions += 25.0
  1076→
  1077→        # 2. Debt ratio gate
  1078→        debt = metrics_snapshot.debt_ratio
  1079→        max_debt = self._thresholds.get("max_debt_ratio", 30.0)
  1080→        debt_pass = debt <= max_debt
  1081→        evaluations["debt_ratio"] = {
  1082→            "value": debt,
  1083→            "threshold": max_debt,
  1084→            "passed": debt_pass,
  1085→        }
  1086→        if not debt_pass:
  1087→            failures.append(
  1088→                f"Debt ratio {debt:.1f}% exceeds threshold {max_debt:.1f}%"
  1089→            )
  1090→            score_deductions += 20.0
  1091→        elif debt > max_debt * 0.7:
  1092→            warnings.append(
  1093→                f"Debt ratio {debt:.1f}% approaching threshold {max_debt:.1f}%"
  1094→            )
  1095→
  1096→        # 3. Complexity gate
  1097→        complexity = metrics_snapshot.complexity_max
  1098→        max_cx = self._thresholds.get("max_complexity", 20.0)
  1099→        cx_pass = complexity <= max_cx
  1100→        evaluations["complexity"] = {
  1101→            "value": complexity,
  1102→            "threshold": max_cx,
  1103→            "passed": cx_pass,
  1104→        }
  1105→        if not cx_pass:
  1106→            failures.append(
  1107→                f"Max complexity {complexity:.1f} exceeds threshold {max_cx:.1f}"
  1108→            )
  1109→            score_deductions += 15.0
  1110→
  1111→        # 4. Drift score gate
  1112→        drift_score = drift_report.drift_score
  1113→        max_drift = self._thresholds.get("max_drift_score", 50.0)
  1114→        drift_pass = drift_score <= max_drift
  1115→        evaluations["drift_score"] = {
  1116→            "value": drift_score,
  1117→            "threshold": max_drift,
  1118→            "passed": drift_pass,
  1119→        }
  1120→        if not drift_pass:
  1121→            failures.append(
  1122→                f"Drift score {drift_score:.1f} exceeds threshold {max_drift:.1f}"
  1123→            )
  1124→            score_deductions += 20.0
  1125→        elif drift_score > max_drift * 0.7:
  1126→            warnings.append(
  1127→                f"Drift score {drift_score:.1f} approaching threshold {max_drift:.1f}"
  1128→            )
  1129→
  1130→        # 5. Drift events volume gate
  1131→        total_drift_events = (
  1132→            len(drift_report.files_added)
  1133→            + len(drift_report.files_removed)
  1134→            + len(drift_report.files_modified)
  1135→        )
  1136→        max_events = self._thresholds.get("max_drift_events", 10)
  1137→        events_pass = total_drift_events <= max_events
  1138→        evaluations["drift_events"] = {
  1139→            "value": total_drift_events,
  1140→            "threshold": max_events,
  1141→            "passed": events_pass,
  1142→        }
  1143→        if not events_pass:
  1144→            failures.append(
  1145→                f"Drift events ({total_drift_events}) exceed threshold ({max_events})"
  1146→            )
  1147→            score_deductions += 10.0
  1148→
  1149→        # 6. Health check gate
  1150→        health_score = health_result.get("health_score", 0.0)
  1151→        min_health = self._thresholds.get("min_health_checks_passed", 0.7)
  1152→        health_pass = health_score >= min_health
  1153→        evaluations["health_checks"] = {
  1154→            "value": health_score,
  1155→            "threshold": min_health,
  1156→            "passed": health_pass,
  1157→        }
  1158→        if not health_pass:
  1159→            failures.append(
  1160→                f"Health score {health_score:.1f}% below threshold {min_health:.1f}%"
  1161→            )
  1162→            score_deductions += 10.0
  1163→
  1164→        # 7. Regression check
  1165→        if drift_report.regression_detected:
  1166→            failures.append(
  1167→                f"Regression detected: {len(drift_report.regression_details)} issue(s)"
  1168→            )
  1169→            score_deductions += 15.0
  1170→            for detail in drift_report.regression_details:
  1171→                warnings.append(f"Regression: {detail}")
  1172→
  1173→        # Calculate aggregate score
  1174→        score = max(0.0, 100.0 - score_deductions)
  1175→        passed = len(failures) == 0
  1176→
  1177→        return QualityGateResult(
  1178→            passed=passed,
  1179→            score=score,
  1180→            thresholds=dict(self._thresholds),
  1181→            evaluations=evaluations,
  1182→            failures=failures,
  1183→            warnings=warnings,
  1184→        )
  1185→
  1186→    def engineering_workflow(
  1187→        self,
  1188→        create_baseline_if_missing: bool = True,
  1189→        record_metrics: bool = True,
  1190→    ) -> Dict[str, Any]:
  1191→        """Run the full engineering pipeline.
  1192→
  1193→        Executes health check, drift detection, metrics collection,
  1194→        and quality gate evaluation in sequence.
  1195→
  1196→        Parameters
  1197→        ----------
  1198→        create_baseline_if_missing : bool
  1199→            Auto-capture a baseline if none exists during drift detection.
  1200→        record_metrics : bool
  1201→            Whether to persist metrics snapshot to history.
  1202→
  1203→        Returns
  1204→        -------
  1205→        dict
  1206→            Complete pipeline result with all sub-results and
  1207→            aggregate quality gate verdict.
  1208→        """
  1209→        pipeline_start = time.perf_counter()
  1210→        result: Dict[str, Any] = {
  1211→            "timestamp": datetime.now(timezone.utc).isoformat(),
  1212→            "repo_path": str(self._repo),
  1213→            "pipeline": "engineering_workflow",
  1214→        }
  1215→
  1216→        # Step 1: Health check
  1217→        t0 = time.perf_counter()
  1218→        health = self.health_check()
  1219→        result["health_check"] = {
  1220→            "duration_ms": round((time.perf_counter() - t0) * 1000, 2),
  1221→            "result": health,
  1222→        }
  1223→
  1224→        # Step 2: Drift detection
  1225→        t0 = time.perf_counter()
  1226→        drift = self.drift_detection(
  1227→            create_baseline_if_missing=create_baseline_if_missing,
  1228→        )
  1229→        result["drift_detection"] = {
  1230→            "duration_ms": round((time.perf_counter() - t0) * 1000, 2),
  1231→            "result": drift,
  1232→        }
  1233→
  1234→        # Step 3: Metrics collection
  1235→        t0 = time.perf_counter()
  1236→        metrics_snapshot = self._metrics.collect_current_metrics()
  1237→        if record_metrics:
  1238→            self._metrics.record_snapshot(metrics_snapshot)
  1239→        result["metrics"] = {
  1240→            "duration_ms": round((time.perf_counter() - t0) * 1000, 2),
  1241→            "snapshot": metrics_snapshot.to_dict(),
  1242→            "trends": {
  1243→                "coverage": self._metrics.get_coverage_trend(),
  1244→                "debt": self._metrics.get_debt_trend(),
  1245→                "complexity": self._metrics.get_complexity_trend(),
  1246→                "test_pass_rate": self._metrics.get_test_pass_rate_trend(),
  1247→            },
  1248→        }
  1249→
  1250→        # Step 4: Quality gate
  1251→        t0 = time.perf_counter()
  1252→        drift_report = DriftReport()
  1253→        drift_data = drift.get("drift", {})
  1254→        if isinstance(drift_data, dict):
  1255→            drift_report = DriftReport(
  1256→                baseline_name=drift_data.get("baseline_name", ""),
  1257→                baseline_timestamp=drift_data.get("baseline_timestamp", ""),
  1258→                current_timestamp=drift_data.get("current_timestamp", ""),
  1259→                files_added=drift_data.get("files_added", []),
  1260→                files_removed=drift_data.get("files_removed", []),
  1261→                files_modified=drift_data.get("files_modified", []),
  1262→                drift_events=drift_data.get("drift_events", []),
  1263→                drift_score=drift_data.get("drift_score", 0.0),
  1264→                regression_detected=drift_data.get("regression_detected", False),
  1265→                regression_details=drift_data.get("regression_details", []),
  1266→            )
  1267→
  1268→        gate = self.quality_gate(
  1269→            metrics_snapshot=metrics_snapshot,
  1270→            drift_report=drift_report,
  1271→            health_result=health,
  1272→        )
  1273→        result["quality_gate"] = {
  1274→            "duration_ms": round((time.perf_counter() - t0) * 1000, 2),
  1275→            "result": gate.to_dict(),
  1276→        }
  1277→
  1278→        # Aggregate summary
  1279→        total_ms = (time.perf_counter() - pipeline_start) * 1000.0
  1280→        result["summary"] = {
  1281→            "total_duration_ms": round(total_ms, 2),
  1282→            "quality_gate_passed": gate.passed,
  1283→            "quality_gate_score": gate.score,
  1284→            "drift_score": drift_report.drift_score,
  1285→            "health_score": health.get("health_score", 0.0),
  1286→            "regression_detected": drift_report.regression_detected,
  1287→            "files_analyzed": metrics_snapshot.file_count,
  1288→        }
  1289→
  1290→        # Emit observability event via plugin hook if available
  1291→        try:
  1292→            from .plugins import HookManager
  1293→            HookManager.fire(
  1294→                "engineering_workflow_complete",
  1295→                result=result,
  1296→            )
  1297→        except Exception:
  1298→            pass
  1299→
  1300→        return result

  1316→    @staticmethod
  1317→    def _compute_health_score(diag_result: Dict[str, Any]) -> float:
  1318→        """Compute a 0-100 health score from diagnostics results."""
  1319→        total = diag_result.get("total_checks", 0)
  1320→        passed = diag_result.get("passed", 0)
  1321→        if total == 0:
  1322→            return 100.0
  1323→        return round(passed / total * 100.0, 2)

  1325→    def _check_repository_health(self) -> Dict[str, Any]:
  1326→        """Run repository-specific health checks.
  1327→
  1328→        Checks:
  1329→        - Repository directory exists and is accessible
  1330→        - Source file count
  1331→        - Presence of test files
  1332→        - Presence of configuration files
  1333→        - Presence of documentation
  1334→        """
  1335→        checks: Dict[str, Any] = {}
  1336→
  1337→        # Directory exists
  1338→        checks["repo_exists"] = self._repo.is_dir()
  1339→
  1340→        # Count source files
  1341→        source_files = self._metrics._walk_source_files()
  1342→        checks["source_file_count"] = len(source_files)
  1343→
  1344→        # Test file detection
  1345→        test_files = [
  1346→            f for f in source_files
  1347→            if "test" in f.name.lower()
  1348→            or "tests" in f.parts
  1349→            or "test_" in f.name
  1350→            or f.name.startswith("test_")
  1351→            or f.name.endswith("_test.py")
  1352→        ]
  1353→        checks["test_file_count"] = len(test_files)
  1354→        checks["has_tests"] = len(test_files) > 0
  1355→
  1356→        # Configuration file detection
  1357→        config_patterns = [
  1358→            "setup.py", "setup.cfg", "pyproject.toml",
  1359→            "package.json", "Cargo.toml", "go.mod",
  1360→            "Makefile", "CMakeLists.txt", "build.gradle",
  1361→            ".github", ".gitlab-ci.yml", "Jenkinsfile",
  1362→        ]
  1363→        found_configs = []
  1364→        for pattern in config_patterns:
  1365→            if (self._repo / pattern).exists():
  1366→                found_configs.append(pattern)
  1367→        checks["config_files"] = found_configs
  1368→        checks["has_build_config"] = len(found_configs) > 0
  1369→
  1370→        # Documentation detection
  1371→        doc_patterns = ["README.md", "README.rst", "README.txt", "docs/"]
  1372→        found_docs = []
  1373→        for pattern in doc_patterns:
  1374→            if (self._repo / pattern).exists():
  1375→                found_docs.append(pattern)
  1376→        checks["documentation_files"] = found_docs
  1377→        checks["has_documentation"] = len(found_docs) > 0
  1378→
  1379→        # VCS detection
  1380→        checks["has_vcs"] = (self._repo / ".git").is_dir()
  1381→
  1382→        return checks
