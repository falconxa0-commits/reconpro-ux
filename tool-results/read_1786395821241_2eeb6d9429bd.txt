     1→     1→"""Async HTTP engine for ReconPro v10.0.
     2→     2→
     3→     3→Architecture
     4→     4→────────────
     5→     5→This module provides a fully async HTTP layer that replaces the synchronous
     6→     6→urllib-based approach in ``http_layer.py`` for high-throughput scanning.  It is
     7→     7→designed as a **drop-in enhancement** — callers that prefer the old sync
     8→     8→API can use :func:`probe_sync` which transparently runs the async path
     9→     9→inside a fresh event loop.
    10→    10→
    11→    11→Components
    12→    12→──────────
    13→    13→1. **AsyncSession** — wraps ``aiohttp.ClientSession`` with connection pooling,
    14→    14→   redirect-chain recording, per-target cookie jars, and an automatic sync
    15→    15→   fallback when ``aiohttp`` is not installed.
    16→    16→2. **AdaptiveLimiter** — token-bucket rate limiter with exponential-moving-
    17→    17→   average tracking of response times, error rates, and timeout rates.
    18→    18→   Automatically backs off on 429/503, recovers on fast responses, and
    19→    19→   maintains per-domain state.  Includes hostname cache to avoid
    20→    20→   redundant urlparse calls on repeated URLs.
    21→    21→3. **async_probe()** — async equivalent of ``http_probe()`` returning a
    22→    22→   rich result dict compatible with the existing ``Finding`` dataclass.
    23→    23→4. **probe_sync()** — synchronous wrapper for backward compatibility.
    24→    24→5. **batch_probe()** — concurrent URL probing with ``asyncio.gather`` and
    25→    25→   per-domain rate limiting.
    26→    26→
    27→    27→Performance notes (v10.1)
    28→    28→───────────────────────────
    29→    29→  - DEFAULT_HEADERS.copy() used instead of dict(DEFAULT_HEADERS) for
    30→    30→    faster C-level copy path.
    31→    31→  - Hostname cache in AdaptiveLimiter avoids repeated urlparse() on
    32→    32→    the same URLs during high-throughput scanning.
    33→    33→  - Header merge short-circuits when no extra headers are provided.
    34→    34→
    35→    35→No direct imports from ``.http`` are used to avoid circular dependencies.
    36→    36→The ``Finding`` dataclass can be imported separately where needed.
    37→    37→"""
    38→    38→
    39→    39→from __future__ import annotations

    77→    77→
    78→    78→# ── Adaptive Limiter ────────────────────────────────────────────────────
    79→    79→
    80→    80→
    81→    81→class _DomainLimiter:
    82→    82→    """Per-domain rate-limit state tracked by AdaptiveLimiter."""
    83→    83→
    84→    84→    __slots__ = (
    85→    85→        "tokens",
    86→    86→        "max_tokens",
    87→    87→        "current_rps",
    88→    88→        "last_refill",
    89→    89→        "response_time_ema",
    90→    90→        "error_rate_ema",
    91→    91→        "timeout_count",
    92→    92→        "total_requests",
    93→    93→        "concurrent_limit",
    94→    94→        "last_backoff",
    95→    95→    )
    96→    96→
    97→    97→    def __init__(self, rps: float, max_concurrent: int = 50) -> None:
    98→    98→        self.tokens: float = rps
    99→    99→        self.max_tokens: float = rps * 2.0
   100→   100→        self.current_rps: float = rps
   101→   101→        self.last_refill: float = time.monotonic()
   102→   102→        self.response_time_ema: float = 0.0
   103→   103→        self.error_rate_ema: float = 0.0
   104→   104→        self.timeout_count: int = 0
   105→   105→        self.total_requests: int = 0
   106→   106→        self.concurrent_limit: int = max_concurrent
   107→   107→        self.last_backoff: float = 0.0
   108→   108→
   109→   109→
   110→   110→class AdaptiveLimiter:
   111→   111→    """Token-bucket rate limiter with adaptive back-off and recovery.
   112→   112→
   113→   113→    Tracks per-domain response-time EMA, error-rate EMA, and timeout
   114→   114→    frequency to dynamically adjust throughput.  Thread-safe.
   115→   115→
   116→   116→    Parameters
   117→   117→    ----------
   118→   118→    min_rps : float
   119→   119→        Minimum allowed requests per second per domain.
   120→   120→    max_rps : float
   121→   121→        Maximum allowed requests per second per domain.
   122→   122→    backoff_factor : float
   123→   123→        Multiplier applied to rate on back-pressure signals (429/503).
   124→   124→    recovery_time : float
   125→   125→        Seconds of calm before the limiter begins ramping back up.
   126→   126→    """
   127→   127→
   128→   128→    def __init__(
   129→   129→        self,
   130→   130→        min_rps: float = 1.0,
   131→   131→        max_rps: float = 100.0,
   132→   132→        backoff_factor: float = 0.5,
   133→   133→        recovery_time: float = 5.0,
   134→   134→    ) -> None:
   135→   135→        self.min_rps = min_rps
   136→   136→        self.max_rps = max_rps
   137→   137→        self.backoff_factor = backoff_factor
   138→   138→        self.recovery_time = recovery_time
   139→   139→        self._alpha = 0.3  # EMA smoothing factor
   140→   140→        self._lock = Lock()
   141→   141→        self._domains: Dict[str, _DomainLimiter] = {}
   142→   142→        # Hostname cache: avoid repeated urlparse() on the same URLs.
   143→   143→        self._hostname_cache: Dict[str, str] = {}
   144→   144→
   145→   145→    # ── public helpers ───────────────────────────────────────────────
   146→   146→
   147→   147→    def _get_domain(self, url: str) -> str:
   148→   148→        cached = self._hostname_cache.get(url)
   149→   149→        if cached is not None:
   150→   150→            return cached
   151→   151→        parsed = urlparse(url)
   152→   152→        hostname = parsed.hostname or "unknown"
   153→   153→        self._hostname_cache[url] = hostname
   154→   154→        return hostname
   155→   155→
   156→   156→    def _get_or_create(self, domain: str) -> _DomainLimiter:
   157→   157→        if domain not in self._domains:
   158→   158→            self._domains[domain] = _DomainLimiter(rps=self.max_rps)
   159→   159→        return self._domains[domain]
   160→   160→
   161→   161→    async def acquire(self, url: str) -> None:
   162→   162→        """Wait until a token is available for *url*."""
   163→   163→        domain = self._get_domain(url)
   164→   164→        with self._lock:
   165→   165→            dl = self._get_or_create(domain)
   166→   166→        await self._wait_for_token(dl)
   167→   167→
   168→   168→    def acquire_sync(self, url: str) -> None:
   169→   169→        """Synchronous acquire — blocks until a token is available."""
   170→   170→        domain = self._get_domain(url)
   171→   171→        with self._lock:
   172→   172→            dl = self._get_or_create(domain)
   173→   173→        self._wait_for_token_sync(dl)
   174→   174→
   175→   175→    def record_response(
   176→   176→        self,
   177→   177→        url: str,
   178→   178→        status: int,
   179→   179→        response_time: float,
   180→   180→        timed_out: bool = False,
   181→   181→    ) -> None:
   182→   182→        """Update limiter state based on the response outcome."""
   183→   183→        domain = self._get_domain(url)
   184→   184→        with self._lock:
   185→   185→            dl = self._get_or_create(domain)
   186→   186→            dl.total_requests += 1
   187→   187→
   188→   188→            # Update response time EMA.
   189→   189→            if dl.response_time_ema == 0.0:
   190→   190→                dl.response_time_ema = response_time
   191→   191→            else:
   192→   192→                dl.response_time_ema = (
   193→   193→                    self._alpha * response_time
   194→   194→                    + (1 - self._alpha) * dl.response_time_ema
   195→   195→                )
   196→   196→
   197→   197→            # Update error rate EMA (error = 4xx/5xx excluding 429/503).
   198→   198→            is_error = 400 <= status < 600 and status not in (429, 503)
   199→   199→            if dl.total_requests <= 1:
   200→   200→                dl.error_rate_ema = 1.0 if is_error else 0.0
   201→   201→            else:
   202→   202→                dl.error_rate_ema = (
   203→   203→                    self._alpha * (1.0 if is_error else 0.0)
   204→   204→                    + (1 - self._alpha) * dl.error_rate_ema
   205→   205→                )
   206→   206→
   207→   207→            # Handle back-pressure signals: 429 or 503.
   208→   208→            if status in (429, 503):
   209→   209→                dl.current_rps = max(
   210→   210→                    self.min_rps, dl.current_rps * self.backoff_factor
   211→   211→                )
   212→   212→                dl.last_backoff = time.monotonic()
   213→   213→                logger.debug(
   214→   214→                    "AdaptiveLimiter back-off for %s: rate→%.1f rps (status %d)",
   215→   215→                    domain, dl.current_rps, status,
   216→   216→                )
   217→   217→
   218→   218→            # Handle timeout — reduce concurrency.
   219→   219→            if timed_out:
   220→   220→                dl.timeout_count += 1
   221→   221→                dl.concurrent_limit = max(
   222→   222→                    1, int(dl.concurrent_limit * 0.75)
   223→   223→                )
   224→   224→                dl.last_backoff = time.monotonic()
   225→   225→                logger.debug(
   226→   226→                    "AdaptiveLimiter timeout for %s: concurrency→%d",
   227→   227→                    domain, dl.concurrent_limit,
   228→   228→                )
   229→   229→
   230→   230→            # Recovery: on fast responses, slowly ramp up.
   231→   231→            if response_time < 0.2 and status < 400:
   232→   232→                now = time.monotonic()
   233→   233→                if (now - dl.last_backoff) > self.recovery_time:
   234→   234→                    new_rps = dl.current_rps * 1.1
   235→   235→                    dl.current_rps = min(self.max_rps, new_rps)
   236→   236→                    dl.timeout_count = max(0, dl.timeout_count - 1)
   237→   237→                    dl.concurrent_limit = min(
   238→   238→                        50, dl.concurrent_limit + 1
   239→   239→                    )
   240→   240→
   241→   241→            # Refill tokens to reflect new rate.
   242→   242→            dl.max_tokens = dl.current_rps * 2.0
   243→   243→            if dl.tokens > dl.max_tokens:
   244→   244→                dl.tokens = dl.max_tokens
   245→   245→
   246→   246→    def get_domain_stats(self, domain: str) -> Dict[str, Any]:
   247→   247→        """Return diagnostic stats for a domain."""
   248→   248→        with self._lock:
   249→   249→            dl = self._domains.get(domain)
   250→   250→            if dl is None:
   251→   251→                return {}
   252→   252→            return {
   253→   253→                "current_rps": round(dl.current_rps, 2),
   254→   254→                "response_time_ema": round(dl.response_time_ema, 4),
   255→   255→                "error_rate_ema": round(dl.error_rate_ema, 4),
   256→   256→                "timeout_count": dl.timeout_count,
   257→   257→                "total_requests": dl.total_requests,
   258→   258→                "concurrent_limit": dl.concurrent_limit,
   259→   259→            }
   260→   260→
   261→   261→    # ── internal ─────────────────────────────────────────────────────
   262→   262→
   263→   263→    def _refill(self, dl: _DomainLimiter) -> None:
   264→   264→        """Add tokens based on elapsed time and current rate."""
   265→   265→        now = time.monotonic()
   266→   266→        elapsed = now - dl.last_refill
   267→   267→        if elapsed > 0:
   268→   268→            dl.tokens = min(
   269→   269→                dl.max_tokens, dl.tokens + elapsed * dl.current_rps
   270→   270→            )
   271→   271→            dl.last_refill = now
   272→   272→
   273→   273→    async def _wait_for_token(self, dl: _DomainLimiter) -> None:
   274→   274→        """Async token wait with jitter on recent back-off."""
   275→   275→        while True:
   276→   276→            with self._lock:
   277→   277→                self._refill(dl)
   278→   278→                if dl.tokens >= 1.0:
   279→   279→                    dl.tokens -= 1.0
   280→   280→                    break
   281→   281→                wait = (1.0 - dl.tokens) / max(dl.current_rps, 0.1)
   282→   282→            # Add jitter if we recently backed off.
   283→   283→            if (time.monotonic() - dl.last_backoff) < self.recovery_time:
   284→   284→                wait += random.uniform(0.1, 0.5)
   285→   285→            await asyncio.sleep(min(wait, 2.0))
   286→   286→
   287→   287→    def _wait_for_token_sync(self, dl: _DomainLimiter) -> None:
   288→   288→        """Synchronous token wait with jitter on recent back-off."""
   289→   289→        while True:
   290→   290→            with self._lock:
   291→   291→                self._refill(dl)
   292→   292→                if dl.tokens >= 1.0:
   293→   293→                    dl.tokens -= 1.0
   294→   294→                    break
   295→   295→                wait = (1.0 - dl.tokens) / max(dl.current_rps, 0.1)
   296→   296→            if (time.monotonic() - dl.last_backoff) < self.recovery_time:
   297→   297→                wait += random.uniform(0.1, 0.5)
   298→   298→            time.sleep(min(wait, 2.0))
   299→   299→
   300→   300→    def close(self) -> None:
   301→   301→        """Release resources held by the limiter."""
   302→   302→
   303→   303→
   304→   304→# Module-level default limiter.
   305→   305→default_adaptive_limiter = AdaptiveLimiter()

   308→   308→# ── Async Session ──────────────────────────────────────────────────────
   309→   309→
   310→   310→
   311→   311→class AsyncSession:
   312→   312→    """Async HTTP session with connection pooling, redirect tracking, and
   313→   313→    an automatic sync fallback when ``aiohttp`` is not installed.
   314→   314→
   315→   315→    Parameters
   316→   316→    ----------
   317→   317→    verify_ssl : bool
   318→   318→        Whether to verify TLS certificates.
   319→   319→    timeout : int
   320→   320→        Default request timeout in seconds.
   321→   321→    max_redirects : int
   322→   322→        Maximum number of redirects to follow.
   323→   323→    max_connections : int
   324→   324→        Connection pool size for aiohttp.
   325→   325→    keepalive : int
   326→   326→        TCP keep-alive timeout in seconds.
   327→   327→    """
   328→   328→
   329→   329→    def __init__(
   330→   330→        self,
   331→   331→        verify_ssl: bool = True,
   332→   332→        timeout: int = 10,
   333→   333→        max_redirects: int = 10,
   334→   334→        max_connections: int = 1000,
   335→   335→        keepalive: int = 30,
   336→   336→    ) -> None:
   337→   337→        self.verify_ssl = verify_ssl
   338→   338→        self.timeout = timeout
   339→   339→        self.max_redirects = max_redirects
   340→   340→        self._aio_session: Optional[Any] = None
   341→   341→        self._sync_ssl_ctx: Optional[ssl.SSLContext] = None
   342→   342→
   343→   343→        if _HAS_AIOHTTP:
   344→   344→            cookie_jar = CookieJar(unsafe=True)
   345→   345→            connector = aiohttp.TCPConnector(
   346→   346→                limit=max_connections,
   347→   347→                keepalive_timeout=keepalive,
   348→   348→                ssl=False if not verify_ssl else None,
   349→   349→            )
   350→   350→            self._aio_session = aiohttp.ClientSession(
   351→   351→                connector=connector,
   352→   352→                cookie_jar=cookie_jar,
   353→   353→                timeout=aiohttp.ClientTimeout(total=timeout),
   354→   354→            )
   355→   355→        else:
   356→   356→            # Prepare a reusable ssl context for the sync fallback.
   357→   357→            if verify_ssl:
   358→   358→                self._sync_ssl_ctx = ssl.create_default_context()
   359→   359→            else:
   360→   360→                self._sync_ssl_ctx = ssl._create_unverified_context()
   361→   361→
   362→   362→    # ── public async methods ─────────────────────────────────────────
   363→   363→
   364→   364→    async def get(
   365→   365→        self,
   366→   366→        url: str,
   367→   367→        headers: Optional[Dict[str, str]] = None,
   368→   368→        **kwargs: Any,
   369→   369→    ) -> Dict[str, Any]:
   370→   370→        """Perform an async GET request."""
   371→   371→        return await self.request("GET", url, headers=headers, **kwargs)
   372→   372→
   373→   373→    async def post(
   374→   374→        self,
   375→   375→        url: str,
   376→   376→        headers: Optional[Dict[str, str]] = None,
   377→   377→        body: Optional[bytes] = None,
   378→   378→        **kwargs: Any,
   379→   379→    ) -> Dict[str, Any]:
   380→   380→        """Perform an async POST request."""
   381→   381→        return await self.request(
   382→   382→            "POST", url, headers=headers, data=body, **kwargs
   383→   383→        )
   384→   384→
   385→   385→    async def head(
   386→   386→        self,
   387→   387→        url: str,
   388→   388→        headers: Optional[Dict[str, str]] = None,
   389→   389→        **kwargs: Any,
   390→   390→    ) -> Dict[str, Any]:
   391→   391→        """Perform an async HEAD request."""
   392→   392→        return await self.request("HEAD", url, headers=headers, **kwargs)
   393→   393→
   394→   394→    async def request(
   395→   395→        self,
   396→   396→        method: str,
   397→   397→        url: str,
   398→   398→        headers: Optional[Dict[str, str]] = None,
   399→   399→        body: Optional[bytes] = None,
   400→   400→        **kwargs: Any,
   401→   401→    ) -> Dict[str, Any]:
   402→   402→        """Perform an async HTTP request with redirect-chain tracking.
   403→   403→
   404→   404→        Returns an ``AsyncProbeResult`` dict with keys: ok, status,
   405→   405→        reason, headers, body, url, redirect_chain, response_time,
   406→   406→        ssl_info.
   407→   407→        """
   408→   408→        # Short-circuit: use DEFAULT_HEADERS directly when no extras.
   409→   409→        if headers:
   410→   410→            merged = DEFAULT_HEADERS.copy()
   411→   411→            merged.update(headers)
   412→   412→        else:
   413→   413→            merged = DEFAULT_HEADERS
   414→   414→
   415→   415→        if _HAS_AIOHTTP and self._aio_session is not None:
   416→   416→            return await self._aio_request(
   417→   417→                method, url, merged, body, **kwargs
   418→   418→            )
   419→   419→        else:
   420→   420→            # Sync fallback — run urllib in a thread to keep the API async.
   421→   421→            return await self._sync_request(method, url, merged, body)
   422→   422→
   423→   423→    async def close(self) -> None:
   424→   424→        """Close the underlying aiohttp session (no-op for sync fallback)."""
   425→   425→        if self._aio_session is not None:
   426→   426→            await self._aio_session.close()
   427→   427→            self._aio_session = None
   428→   428→
   429→   429→    async def __aenter__(self) -> AsyncSession:
   430→   430→        return self
   431→   431→
   432→   432→    async def __aexit__(self, *args: Any) -> None:
   433→   433→        await self.close()
   434→   434→
   435→   435→    # ── aiohttp path ─────────────────────────────────────────────────
   436→   436→
   437→   437→    async def _aio_request(
   438→   438→        self,
   439→   439→        method: str,
   440→   440→        url: str,
   441→   441→        headers: Dict[str, str],
   442→   442→        body: Optional[bytes],
   443→   443→        **kwargs: Any,
   444→   444→    ) -> Dict[str, Any]:
   445→   445→        """Execute request via aiohttp, following redirects manually."""
   446→   446→        redirect_chain: List[Dict[str, Any]] = []
   447→   447→        current_url = url
   448→   448→        session = self._aio_session
   449→   449→        assert session is not None  # noqa: S101
   450→   450→
   451→   451→        for _ in range(self.max_redirects + 1):
   452→   452→            t0 = time.monotonic()
   453→   453→            try:
   454→   454→                # Determine if this is the final hop.
   455→   455→                allow_redirects = False
   456→   456→                async with session.request(
   457→   457→                    method,
   458→   458→                    current_url,
   459→   459→                    headers=headers,
   460→   460→                    data=body if method.upper() != "GET" else None,
   461→   461→                    allow_redirects=allow_redirects,
   462→   462→                    ssl=False if not self.verify_ssl else None,
   463→   463→                    **kwargs,
   464→   464→                ) as resp:
   465→   465→                    elapsed = time.monotonic() - t0
   466→   466→                    ssl_info = self._extract_ssl_info(session, current_url)
   467→   467→
   468→   468→                    # Collect response data.
   469→   469→                    resp_body = await resp.read()
   470→   470→                    status = resp.status
   471→   471→                    reason = resp.reason or ""
   472→   472→                    resp_headers = dict(resp.headers.items())
   473→   473→                    body_str = resp_body.decode("utf-8", errors="replace")[
   474→   474→                        :_BODY_LIMIT
   475→   475→                    ]
   476→   476→
   477→   477→                    # Handle redirects.
   478→   478→                    if status in (301, 302, 303, 307, 308):
   479→   479→                        location = resp_headers.get("Location", "")
   480→   480→                        if not location:
   481→   481→                            break
   482→   482→                        # Resolve relative URLs.
   483→   483→                        next_url = urljoin(current_url, location)
   484→   484→                        redirect_chain.append({
   485→   485→                            "url": current_url,
   486→   486→                            "status": status,
   487→   487→                            "location": next_url,
   488→   488→                            "response_time": elapsed,
   489→   489→                        })
   490→   490→                        current_url = next_url
   491→   491→                        # For 303, switch to GET.
   492→   492→                        if status == 303:
   493→   493→                            method = "GET"
   494→   494→                            body = None
   495→   495→                        continue
   496→   496→
   497→   497→                    return self._build_result(
   498→   498→                        ok=status < 400,
   499→   499→                        status=status,
   500→   500→                        reason=reason,
   501→   501→                        headers=resp_headers,
   502→   502→                        body=body_str,
   503→   503→                        url=current_url,
   504→   504→                        redirect_chain=redirect_chain,
   505→   505→                        response_time=elapsed,
   506→   506→                        ssl_info=ssl_info,
   507→   507→                    )
   508→   508→
   509→   509→            except asyncio.TimeoutError:
   510→   510→                elapsed = time.monotonic() - t0
   511→   511→                return self._build_result(
   512→   512→                    ok=False, status=0, reason="Timeout",
   513→   513→                    headers={}, body="", url=current_url,
   514→   514→                    redirect_chain=redirect_chain,
   515→   515→                    response_time=elapsed, ssl_info={},
   516→   516→                )
   517→   517→            except Exception as exc:
   518→   518→                elapsed = time.monotonic() - t0
   519→   519→                return self._build_result(
   520→   520→                    ok=False, status=0, reason=str(exc),
   521→   521→                    headers={}, body="", url=current_url,
   522→   522→                    redirect_chain=redirect_chain,
   523→   523→                    response_time=elapsed, ssl_info={},
   524→   524→                )
   525→   525→
   526→   526→        # Exceeded max redirects.
   527→   527→        return self._build_result(
   528→   528→            ok=False, status=0, reason="Too many redirects",
   529→   529→            headers={}, body="", url=current_url,
   530→   530→            redirect_chain=redirect_chain, response_time=0.0, ssl_info={},
   531→   531→        )

   533→   533→    # ── sync fallback path ───────────────────────────────────────────
   534→   534→
   535→   535→    async def _sync_request(
   536→   536→        self,
   537→   537→        method: str,
   538→   538→        url: str,
   539→   539→        headers: Dict[str, str],
   540→   540→        body: Optional[bytes],
   541→   541→    ) -> Dict[str, Any]:
   542→   542→        """Run a synchronous urllib request in a thread."""
   543→   543→        loop = asyncio.get_running_loop()
   544→   544→        result = await loop.run_in_executor(
   545→   545→            None,
   546→   546→            self._sync_request_impl,
   547→   547→            method, url, headers, body,
   548→   548→        )
   549→   549→        return result
   550→   550→
   551→   551→    def _sync_request_impl(
   552→   552→        self,
   553→   553→        method: str,
   554→   554→        url: str,
   555→   555→        headers: Dict[str, str],
   556→   556→        body: Optional[bytes],
   557→   557→    ) -> Dict[str, Any]:
   558→   558→        """Blocking urllib implementation with manual redirect following."""
   559→   559→        redirect_chain: List[Dict[str, Any]] = []
   560→   560→        current_url = url
   561→   561→        ctx = self._sync_ssl_ctx
   562→   562→        assert ctx is not None  # noqa: S101
   563→   563→
   564→   564→        for _ in range(self.max_redirects + 1):
   565→   565→            t0 = time.monotonic()
   566→   566→            req = urllib.request.Request(
   567→   567→                current_url, data=body, method=method, headers=headers,
   568→   568→            )
   569→   569→            try:
   570→   570→                with urllib.request.urlopen(
   571→   571→                    req, timeout=self.timeout, context=ctx,
   572→   572→                ) as resp:
   573→   573→                    elapsed = time.monotonic() - t0
   574→   574→                    raw = resp.read(_BODY_LIMIT)
   575→   575→                    status = resp.status
   576→   576→                    reason = resp.reason
   577→   577→                    resp_headers = dict(resp.headers.items())
   578→   578→                    body_str = raw.decode("utf-8", errors="replace")
   579→   579→                    return self._build_result(
   580→   580→                        ok=status < 400,
   581→   581→                        status=status,
   582→   582→                        reason=reason or "",
   583→   583→                        headers=resp_headers,
   584→   584→                        body=body_str,
   585→   585→                        url=current_url,
   586→   586→                        redirect_chain=redirect_chain,
   587→   587→                        response_time=elapsed,
   588→   588→                        ssl_info=self._sync_ssl_info(current_url),
   589→   589→                    )
   590→   590→            except urllib.error.HTTPError as exc:
   591→   591→                elapsed = time.monotonic() - t0
   592→   592→                try:
   593→   593→                    raw = exc.read(_BODY_LIMIT)
   594→   594→                    body_str = raw.decode("utf-8", errors="replace")
   595→   595→                except Exception:
   596→   596→                    body_str = ""
   597→   597→
   598→   598→                # Handle redirects that urllib exposes as errors.
   599→   599→                if exc.code in (301, 302, 303, 307, 308):
   600→   600→                    location = exc.headers.get("Location", "") if exc.headers else ""
   601→   601→                    if location:
   602→   602→                        next_url = urljoin(current_url, location)
   603→   603→                        redirect_chain.append({
   604→   604→                            "url": current_url,
   605→   605→                            "status": exc.code,
   606→   606→                            "location": next_url,
   607→   607→                            "response_time": elapsed,
   608→   608→                        })
   609→   609→                        current_url = next_url
   610→   610→                        if exc.code == 303:
   611→   611→                            method = "GET"
   612→   612→                            body = None
   613→   613→                        continue
   614→   614→
   615→   615→                resp_headers = (
   616→   616→                    dict(exc.headers.items()) if exc.headers else {}
   617→   617→                )
   618→   618→                return self._build_result(
   619→   619→                    ok=False,
   620→   620→                    status=exc.code,
   621→   621→                    reason=str(exc.reason),
   622→   622→                    headers=resp_headers,
   623→   623→                    body=body_str,
   624→   624→                    url=current_url,
   625→   625→                    redirect_chain=redirect_chain,
   626→   626→                    response_time=elapsed,
   627→   627→                    ssl_info={},
   628→   628→                )
   629→   629→            except Exception as exc:
   630→   630→                elapsed = time.monotonic() - t0
   631→   631→                return self._build_result(
   632→   632→                    ok=False, status=0, reason=str(exc),
   633→   633→                    headers={}, body="", url=current_url,
   634→   634→                    redirect_chain=redirect_chain,
   635→   635→                    response_time=elapsed, ssl_info={},
   636→   636→                )
   637→   637→
   638→   638→        return self._build_result(
   639→   639→            ok=False, status=0, reason="Too many redirects",
   640→   640→            headers={}, body="", url=current_url,
   641→   641→            redirect_chain=redirect_chain, response_time=0.0, ssl_info={},
   642→   642→        )

   692→   692→# ── async_probe ─────────────────────────────────────────────────────────
   693→   693→
   694→   694→
   695→   695→async def async_probe(
   696→   696→    url: str,
   697→   697→    method: str = "GET",
   698→   698→    body: Optional[bytes] = None,
   699→   699→    headers: Optional[Dict[str, str]] = None,
   700→   700→    timeout: int = 10,
   701→   701→    verify_ssl: bool = True,
   702→   702→    limiter: Optional[AdaptiveLimiter] = None,
   703→   703→    session: Optional[AsyncSession] = None,
   704→   704→) -> Dict[str, Any]:
   705→   705→    """Async equivalent of :func:`http.http_probe`.
   706→   706→
   707→   707→    Returns a dict compatible with the ``Finding`` creation pattern::
   708→   708→
   709→   709→        {
   710→   710→            "ok": bool,
   711→   711→            "status": int,
   712→   712→            "reason": str,
   713→   713→            "headers": dict,
   714→   714→            "body": str,
   715→   715→            "url": str,
   716→   716→            "redirect_chain": list,
   717→   717→            "response_time": float,
   718→   718→            "ssl_info": dict,
   719→   719→        }
   720→   720→
   721→   721→    Parameters
   722→   722→    ----------
   723→   723→    url : str
   724→   724→        Target URL.
   725→   725→    method : str
   726→   726→        HTTP method (GET, POST, HEAD, …).
   727→   727→    body : bytes | None
   728→   728→        Request body.
   729→   729→    headers : dict | None
   730→   730→        Extra headers merged onto the defaults.
   731→   731→    timeout : int
   732→   732→        Timeout in seconds.
   733→   733→    verify_ssl : bool
   734→   734→        Verify TLS certificates.
   735→   735→    limiter : AdaptiveLimiter | None
   736→   736→        Rate limiter instance.  Uses the module default if ``None``.
   737→   737→    session : AsyncSession | None
   738→   738→        Reusable session.  A fresh one is created and closed if ``None``.
   739→   739→    """
   740→   740→    _limiter = limiter or default_adaptive_limiter
   741→   741→    own_session = session is None
   742→   742→    _session = session or AsyncSession(
   743→   743→        verify_ssl=verify_ssl, timeout=timeout,
   744→   744→    )
   745→   745→
   746→   746→    try:
   747→   747→        await _limiter.acquire(url)
   748→   748→        result = await _session.request(
   749→   749→            method, url, headers=headers,
   750→   750→            body=body if method.upper() != "GET" else None,
   751→   751→        )
   752→   752→        # Feed response metrics back to the limiter.
   753→   753→        _limiter.record_response(
   754→   754→            url,
   755→   755→            status=result.get("status", 0),
   756→   756→            response_time=result.get("response_time", 0.0),
   757→   757→            timed_out=(result.get("reason") == "Timeout"),
   758→   758→        )
   759→   759→        return result
   760→   760→    except asyncio.TimeoutError:
   761→   761→        timed_out = True
   762→   762→        _limiter.record_response(url, status=0, response_time=float(timeout), timed_out=True)
   763→   763→        return {
   764→   764→            "ok": False, "status": 0, "reason": "Timeout",
   765→   765→            "headers": {}, "body": "", "url": url,
   766→   766→            "redirect_chain": [], "response_time": float(timeout),
   767→   767→            "ssl_info": {},
   768→   768→        }
   769→   769→    except Exception as exc:
   770→   770→        _limiter.record_response(url, status=0, response_time=0.0, timed_out=False)
   771→   771→        return {
   772→   772→            "ok": False, "status": 0, "reason": str(exc),
   773→   773→            "headers": {}, "body": "", "url": url,
   774→   774→            "redirect_chain": [], "response_time": 0.0, "ssl_info": {},
   775→   775→        }
   776→   776→    finally:
   777→   777→        if own_session:
   778→   778→            await _session.close()
   779→   779→

   781→   781→# ── probe_sync ──────────────────────────────────────────────────────────

   783→   783→
   784→   784→def probe_sync(
   785→   785→    url: str,
   786→   786→    method: str = "GET",
   787→   787→    body: Optional[bytes] = None,
   788→   788→    headers: Optional[Dict[str, str]] = None,
   789→   789→    timeout: int = 10,
   790→   790→    verify_ssl: bool = True,
   791→   791→    limiter: Optional[AdaptiveLimiter] = None,
   792→   792→) -> Dict[str, Any]:
   793→   793→    """Synchronous wrapper around :func:`async_probe`.
   794→   794→
   795→   795→    Creates (and closes) a fresh event loop for each call, making it
   796→   796→    safe to use from existing synchronous modules without any async
   797→   797→    boilerplate.
   798→   798→
   799→   799→    Optimisation: when not already inside an event loop, calls asyncio.run
   800→   800→    directly — the ThreadPoolExecutor fallback is only used when an
   801→   801→    existing loop is detected.
   802→   802→    """
   803→   803→    try:
   804→   804→        asyncio.get_running_loop()
   805→   805→        # Already inside a running loop — fall back to a thread.
   806→   806→        import concurrent.futures
   807→   807→        with concurrent.futures.ThreadPoolExecutor(max_workers=1) as pool:
   808→   808→            future = pool.submit(
   809→   809→                asyncio.run,
   810→   810→                async_probe(
   811→   811→                    url, method=method, body=body, headers=headers,
   812→   812→                    timeout=timeout, verify_ssl=verify_ssl,
   813→   813→                    limiter=limiter,
   814→   814→                ),
   815→   815→            )
   816→   816→            return future.result()
   817→   817→    except RuntimeError:
   818→   818→        # No running loop — safe to call asyncio.run directly.
   819→   819→        return asyncio.run(
   820→   820→            async_probe(
   821→   821→                url, method=method, body=body, headers=headers,
   822→   822→                timeout=timeout, verify_ssl=verify_ssl,
   823→   823→                limiter=limiter,
   824→   824→            ),
   825→   825→        )

   831→   831→async def batch_probe(
   832→   832→    urls: List[str],
   833→   833→    method: str = "GET",
   834→   834→    headers: Optional[Dict[str, str]] = None,
   835→   835→    timeout: int = 10,
   836→   836→    verify_ssl: bool = True,
   837→   837→    concurrency: int = 50,
   838→   838→    limiter: Optional[AdaptiveLimiter] = None,
   839→   839→) -> List[Dict[str, Any]]:
   840→   840→    """Probe a list of URLs concurrently with per-domain rate limiting.
   841→   841→
   842→   842→    Parameters
   843→   843→    ----------
   844→   844→    urls : list[str]
   845→   845→        URLs to probe.
   846→   846→    method : str
   847→   847→        HTTP method for all requests.
   848→   848→    headers : dict | None
   849→   849→        Extra headers.
   850→   850→    timeout : int
   851→   851→        Per-request timeout in seconds.
   852→   852→    verify_ssl : bool
   853→   853→        Verify TLS certificates.
   854→   854→    concurrency : int
   855→   855→        Maximum number of in-flight requests (semaphore bound).
   856→   856→    limiter : AdaptiveLimiter | None
   857→   857→        Shared rate limiter.  Uses the module default if ``None``.
   858→   858→
   859→   859→    Returns
   860→   860→    -------
   861→   861→    list[dict]
   862→   862→        Results in the **same order** as *urls*.
   863→   863→    """
   864→   864→    _limiter = limiter or default_adaptive_limiter
   865→   865→    semaphore = asyncio.Semaphore(concurrency)
   866→   866→
   867→   867→    async def _bounded_probe(url: str) -> Dict[str, Any]:
   868→   868→        """Probe a single URL under the concurrency semaphore."""
   869→   869→        async with semaphore:
   870→   870→            return await async_probe(
   871→   871→                url,
   872→   872→                method=method,
   873→   873→                headers=headers,
   874→   874→                timeout=timeout,
   875→   875→                verify_ssl=verify_ssl,
   876→   876→                limiter=_limiter,
   877→   877→            )
   878→   878→
   879→   879→    tasks = [_bounded_probe(u) for u in urls]
   880→   880→    return list(await asyncio.gather(*tasks, return_exceptions=False))
   881→   881→
   882→   882→
   883→   883→def batch_probe_sync(
   884→   884→    urls: List[str],
   885→   885→    method: str = "GET",
   886→   886→    headers: Optional[Dict[str, str]] = None,
   887→   887→    timeout: int = 10,
   888→   888→    verify_ssl: bool = True,
   889→   889→    concurrency: int = 50,
   890→   890→    limiter: Optional[AdaptiveLimiter] = None,
   891→   891→) -> List[Dict[str, Any]]:
   892→   892→    """Synchronous wrapper around :func:`batch_probe`.
   893→   893→
   894→   894→    Runs the full async batch inside a fresh event loop.
   895→   895→    """
   896→   896→    return asyncio.run(
   897→   897→        batch_probe(
   898→   898→            urls, method=method, headers=headers,
   899→   899→            timeout=timeout, verify_ssl=verify_ssl,
   900→   900→            concurrency=concurrency, limiter=limiter,
   901→   901→        ),
   902→   902→    )
