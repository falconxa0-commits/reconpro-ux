   186→function buildAttackPaths(findings: Finding[]): string {
   187→  const criticals = findings.filter(f => f.severity === 'critical');
   188→  const highs = findings.filter(f => f.severity === 'high');
   189→  const vulns = findings.filter(f => ['vulnerability', 'header', 'ssl', 'dns'].includes(f.category));
   190→
   191→  if (vulns.length === 0) return '';
   192→
   193→  let paths = '## Attack Path Analysis\n\n';
   194→  paths += 'Based on your attack surface, I\'ve identified these **chained attack paths** an adversary could exploit:\n\n';
   195→
   196→  // Path 1: Initial Access → Data Exfiltration
   197→  if (vulns.length > 0) {
   198→    paths += '### Path 1: External Breach → Data Exfiltration\n';
   199→    paths += '```\n';
   200→    paths += '[Internet] ──> ';
   201→    const steps: string[] = [];
   202→
   203→    const openPorts = findings.filter(f => f.category === 'port');
   204→    const sslIssues = findings.filter(f => f.category === 'ssl');
   205→    const headerIssues = findings.filter(f => f.category === 'header');
   206→    const vulnIssues = findings.filter(f => f.category === 'vulnerability');
   207→
   208→    if (openPorts.length > 0) steps.push(`Open Port (${openPorts[0].asset})`);
   209→    if (sslIssues.length > 0) steps.push(`SSL Bypass (${sslIssues[0].title})`);
   210→    if (headerIssues.length > 0) steps.push(`Header Exploit (${headerIssues[0].title})`);
   211→    if (vulnIssues.length > 0) steps.push(`Vuln Exploit (${vulnIssues[0].title})`);
   212→
   213→    steps.push('Internal Network');
   214→    steps.push('Database');
   215→    steps.push('DATA EXFILTRATION');
   216→
   217→    paths += steps.join(' ──> ');
   218→    paths += '\n```\n\n';
   219→
   220→    if (sslIssues.length > 0) {
   221→      paths += '**SSL issues** at the edge allow traffic interception, making all subsequent attacks easier. ';
   222→      paths += 'Fix SSL first to raise the attacker\'s cost of entry.\n\n';
   223→    }
   224→
   225→    if (vulnIssues.length > 0) {
   226→      paths += '**Application vulnerabilities** provide the actual code execution or data access. ';
   227→      paths += `${vulnIssues.filter(v => v.severity === 'critical').length} critical vulns need immediate patching.\n\n`;
   228→    }
   229→  }
   230→
   231→  // Path 2: Supply Chain
   232→  const subdomains = findings.filter(f => f.category === 'subdomain');
   233→  const techs = findings.filter(f => f.category === 'technology');
   234→
   235→  if (subdomains.length > 3) {
   236→    paths += '### Path 2: Subdomain Takeover → Pivot\n';
   237→    paths += '```\n';
   238→    paths += `[${subdomains[0].asset}] ──> DNS Hijack ──> Phishing Domain ──> Credential Harvest ──> SSO Bypass ──> FULL DOMAIN COMPROMISE\n`;
   239→    paths += '```\n\n';
   240→    paths += `With **${subdomains.length} discovered subdomains**, any dangling DNS record is a potential subdomain takeover. `;
   241→    paths += 'An attacker can host a phishing page on a legitimate subdomain, bypassing SPF/DKIM and EV certificates.\n\n';
   242→  }
   243→
   244→  // Path 3: Credential Stuffing
   245→  if (findings.some(f => f.title.toLowerCase().includes('ssh') || f.asset.includes(':22'))) {
   246→    paths += '### Path 3: SSH Brute Force → Lateral Movement\n';
   247→    paths += '```\n';
   248→    paths += '[SSH Port] ──> Credential Stuffing ──> Shell Access ──> Privilege Escalation ──> Pivot to DB/Cloud ──> RANSOMWARE\n';
   249→    paths += '```\n\n';
   250→    paths += 'Exposed SSH is the #1 initial access vector. Attackers use credential stuffing with leaked password databases. ';
   251→    paths += '**Mitigate**: Key-only auth, fail2ban, VPN-only access.\n\n';
   252→  }
   253→
   254→  return paths;
   255→}

   257→// ═══════════════════════════════════════════════════════════════════════
   258→// RISK INTELLIGENCE ENGINE
   259→// ═══════════════════════════════════════════════════════════════════════
   260→
   261→function generateFullAnalysis(findings: Finding[], domain: string): string {
   262→  const criticals = findings.filter(f => f.severity === 'critical');
   263→  const highs = findings.filter(f => f.severity === 'high');
   264→  const mediums = findings.filter(f => f.severity === 'medium');
   265→  const lows = findings.filter(f => f.severity === 'low');
   266→
   267→  const totalRisk = findings.length;
   268→  const riskScore = Math.min(100,
   269→    criticals.length * 25 + highs.length * 15 + mediums.length * 8 + lows.length * 3 + (findings.length - criticals.length - highs.length - mediums.length - lows.length)
   270→  );
   271→
   272→  let analysis = '';
   273→
   274→  // Executive summary
   275→  analysis += `# Security Intelligence Report: ${domain}\n\n`;
   276→  analysis += `**Scan ID**: RP-${Date.now().toString(36).toUpperCase()}\n`;
   277→  analysis += `**Total Contacts**: ${findings.length} | **Risk Score**: ${riskScore}/100\n`;
   278→  analysis += `**Threat Level**: ${riskScore > 70 ? '🔴 CRITICAL' : riskScore > 40 ? '🟠 HIGH' : riskScore > 20 ? '🟡 MODERATE' : '🟢 LOW'}\n\n`;
   279→
   280→  // Risk summary
   281→  analysis += '## Risk Summary\n\n';
   282→  analysis += `| Severity | Count | Impact |\n|----------|-------|--------|\n`;
   283→  analysis += `| Critical | ${criticals.length} | Immediate data breach risk |\n`;
   284→  analysis += `| High | ${highs.length} | Significant exposure window |\n`;
   285→  analysis += `| Medium | ${mediums.length} | Potential exploitation path |\n`;
   286→  analysis += `| Low | ${lows.length} | Minimal direct risk |\n\n`;
   287→
   288→  // Category breakdown
   289→  const categories: Record<string, number> = {};
   290→  for (const f of findings) { categories[f.category] = (categories[f.category] || 0) + 1; }
   291→
   292→  analysis += '## Attack Surface Composition\n\n';
   293→  for (const [cat, count] of Object.entries(categories).sort((a, b) => b[1] - a[1])) {
   294→    const catHighs = findings.filter(f => f.category === cat && ['critical', 'high'].includes(f.severity)).length;
   295→    analysis += `- **${cat}**: ${count} findings (${catHighs} high/critical)\n`;
   296→  }
   297→  analysis += '\n';
   298→
   299→  // Priority findings with remediation
   300→  const priorityFindings = [...findings]
   301→    .sort((a, b) => {
   302→      const order = { critical: 0, high: 1, medium: 2, low: 3, info: 4 };
   303→      return (order[a.severity] ?? 5) - (order[b.severity] ?? 5);
   304→    })
   305→    .slice(0, 5);
   306→
   307→  analysis += '## Priority Remediation Actions\n\n';
   308→  for (const f of priorityFindings) {
   309→    const rem = findRemediation(f);
   310→    analysis += `### ${f.severity.toUpperCase()}: ${f.title}\n`;
   311→    analysis += `**Asset**: \`${f.asset}\`\n`;
   312→    if (rem?.cve) analysis += `**CVE**: ${rem.cve} | **CVSS**: ${rem.cvss}\n`;
   313→    if (rem?.priority) analysis += `**Priority**: ${rem.priority}\n`;
   314→    analysis += `\n**Remediation**:\n${rem?.fix || 'Review the finding and apply appropriate security controls.'}\n\n`;
   315→    if (rem?.attackPath) {
   316→      analysis += `**Attack Path**: ${rem.attackPath}\n\n`;
   317→    }
   318→    if (rem?.frameworks && rem.frameworks.length > 0) {
   319→      analysis += `**Compliance**: ${rem.frameworks.join(', ')}\n\n`;
   320→    }
   321→  }
   322→
   323→  // Attack paths
   324→  const attackPaths = buildAttackPaths(findings);
   325→  if (attackPaths) {
   326→    analysis += attackPaths;
   327→  }
   328→
   329→  // Compliance overview
   330→  analysis += '## Compliance Impact Assessment\n\n';
   331→  const frameworkStatus = assessCompliance(findings);
   332→  for (const [fw, status] of Object.entries(frameworkStatus)) {
   333→    const icon = status.passed ? '✅' : '❌';
   334→    analysis += `- ${icon} **${fw}**: ${status.passed ? 'Compliant' : `${status.failures} failures`}\n`;
   335→    analysis += `  - ${status.details}\n`;
   336→  }
   337→  analysis += '\n';
   338→
   339→  // Recommendation
   340→  analysis += '## Recommended Action Plan\n\n';
   341→  if (riskScore > 70) {
   342→    analysis += '### ⚡ Immediate (24-48 hours)\n';
   343→    analysis += '1. Patch all **critical** vulnerabilities — these are actively exploitable\n';
   344→    analysis += '2. Fix SSL/TLS configuration — disable weak ciphers, renew expired certificates\n';
   345→    analysis += '3. Restrict exposed management ports (SSH, admin panels) via firewall\n';
   346→    analysis += '4. Enable WAF rules for SQL injection and XSS protection\n\n';
   347→    analysis += '### 📋 This Week\n';
   348→    analysis += '1. Deploy security headers (CSP, HSTS, X-Frame-Options)\n';
   349→    analysis += '2. Fix DNS configuration (SPF, DKIM, DMARC)\n';
   350→    analysis += '3. Update all software to latest versions\n';
   351→    analysis += '4. Implement incident response playbook\n\n';
   352→    analysis += '### 🔄 This Month\n';
   353→    analysis += '1. Establish continuous vulnerability scanning\n';
   354→    analysis += '2. Implement least-privilege access controls\n';
   355→    analysis += '3. Deploy network segmentation\n';
   356→    analysis += '4. Conduct penetration testing\n';
   357→  } else if (riskScore > 40) {
   358→    analysis += '### 📋 This Week\n';
   359→    analysis += '1. Address all high-severity findings\n';
   360→    analysis += '2. Harden SSL/TLS configuration\n';
   361→    analysis += '3. Add missing security headers\n';
   362→    analysis += '4. Review and restrict open ports\n\n';
   363→    analysis += '### 🔄 This Month\n';
   364→    analysis += '1. Implement automated vulnerability scanning in CI/CD\n';
   365→    analysis += '2. Set up SPF/DKIM/DMARC for email security\n';
   366→    analysis += '3. Conduct security awareness training\n';
   367→  } else {
   368→    analysis += 'Your attack surface is in relatively good shape. Focus on:\n';
   369→    analysis += '1. Continuous monitoring for new vulnerabilities\n';
   370→    analysis += '2. Keeping all dependencies updated\n';
   371→    analysis += '3. Periodic penetration testing\n';
   372→  }
   373→
   374→  return analysis;

   377→function findRemediation(finding: Finding) {
   378→  // Match by title keywords — updated for real reconnaissance findings
   379→  const title = finding.title.toLowerCase();
   380→  const asset = finding.asset.toLowerCase();
   381→  const combined = `${title} ${asset} ${finding.category}`;
   382→
   383→  // SSL/TLS findings (from real cert analysis)
   384→  if (combined.includes('expired'))
   385→    return REMEDIATION_DB['ssl-certificate-expired'];
   386→  if (combined.includes('ssl') && (combined.includes('expir') || combined.includes('cert') || combined.includes('validity') || combined.includes('lifespan')))
   387→    return REMEDIATION_DB['ssl-certificate-expired'];
   388→  if (combined.includes('cipher'))
   389→    return REMEDIATION_DB['ssl-weak-cipher'];
   390→  if (combined.includes('tls 1.0') || combined.includes('tls 1.1') || (combined.includes('tls') && combined.includes('deprecated')) || (combined.includes('weak tls')))
   391→    return REMEDIATION_DB['ssl-tls-v1.0'];
   392→  if (combined.includes('self-signed'))
   393→    return REMEDIATION_DB['ssl-self-signed'];
   394→
   395→  // HTTP Security Header findings (from real header analysis)
   396→  if (combined.includes('csp') || combined.includes('content-security-policy'))
   397→    return REMEDIATION_DB['missing-security-headers'];
   398→  if (combined.includes('hsts') || combined.includes('strict-transport'))
   399→    return REMEDIATION_DB['missing-hsts'];
   400→  if (combined.includes('frame') || combined.includes('clickjack'))
   401→    return REMEDIATION_DB['x-frame-options-missing'];
   402→  if (combined.includes('cors') || combined.includes('access-control-allow-origin'))
   403→    return REMEDIATION_DB['missing-security-headers'];
   404→  if (combined.includes('x-content-type') || combined.includes('nosniff') || combined.includes('referrer-policy') || combined.includes('permissions-policy'))
   405→    return REMEDIATION_DB['missing-security-headers'];
   406→  if (combined.includes('header'))
   407→    return REMEDIATION_DB['missing-security-headers'];
   408→
   409→  // DNS findings (from real DNS enumeration)
   410→  if (combined.includes('dns') && (combined.includes('zone') || combined.includes('transfer') || combined.includes('axfr')))
   411→    return REMEDIATION_DB['dns-zone-transfer'];
   412→  if (combined.includes('spf'))
   413→    return REMEDIATION_DB['dns-spf-missing'];
   414→  if (combined.includes('dmarc'))
   415→    return REMEDIATION_DB['dns-spf-missing'];
   416→  if (combined.includes('dkim'))
   417→    return REMEDIATION_DB['dns-spf-missing'];
   418→  if (combined.includes('dnssec'))
   419→    return REMEDIATION_DB['dns-zone-transfer'];
   420→  if (combined.includes('wildcard'))
   421→    return REMEDIATION_DB['dns-zone-transfer'];
   422→
   423→  // Port findings (from real port probing)
   424→  if (combined.includes('ssh') || (combined.includes('port') && combined.includes('22')))
   425→    return REMEDIATION_DB['open-ssh-version'];
   426→  if (combined.includes('port') && (combined.includes('open') || combined.includes('exposed') || combined.includes('accessible')))
   427→    return REMEDIATION_DB['open-port-ssh'];
   428→
   429→  // Subdomain findings (from real DNS resolution)
   430→  if (combined.includes('subdomain') && (combined.includes('sensitive') || combined.includes('admin') || combined.includes('internal') || combined.includes('staging')))
   431→    return REMEDIATION_DB['sensitive-data-exposure'];
   432→
   433→  // Technology/vulnerability findings
   434→  if (combined.includes('sql') || combined.includes('injection'))
   435→    return REMEDIATION_DB['sql-injection'];
   436→  if (combined.includes('xss') || combined.includes('cross-site'))
   437→    return REMEDIATION_DB['xss-vulnerability'];
   438→  if (combined.includes('version disclosure') || combined.includes('technology disclosure') || combined.includes('x-powered-by'))
   439→    return REMEDIATION_DB['outdated-server'];
   440→  if (combined.includes('traversal') || combined.includes('path') || combined.includes('directory'))
   441→    return REMEDIATION_DB['directory-traversal'];
   442→  if (combined.includes('sensitive') || combined.includes('expos') || combined.includes('.env') || combined.includes('.git'))
   443→    return REMEDIATION_DB['sensitive-data-exposure'];
   444→
   445→  // Generic fallback based on category
   446→  if (finding.category === 'ssl') return REMEDIATION_DB['ssl-certificate-expired'];
   447→  if (finding.category === 'header') return REMEDIATION_DB['missing-security-headers'];
   448→  if (finding.category === 'dns') return REMEDIATION_DB['dns-spf-missing'];
   449→  if (finding.category === 'port') return REMEDIATION_DB['open-port-ssh'];
   450→  if (finding.category === 'vulnerability') return REMEDIATION_DB['outdated-server'];
   451→  if (finding.category === 'subdomain') return REMEDIATION_DB['sensitive-data-exposure'];
   452→
   453→  return null;
   454→}

   456→function assessCompliance(findings: Finding[]): Record<string, { passed: boolean; failures: number; details: string }> {
   457→  const sslCrit = findings.some(f => f.category === 'ssl' && f.severity === 'critical');
   458→  const sslHigh = findings.some(f => f.category === 'ssl' && f.severity === 'high');
   459→  const missingHsts = findings.some(f => f.title.toLowerCase().includes('hsts') && f.severity !== 'info');
   460→  const missingCsp = findings.some(f => f.title.toLowerCase().includes('csp') && f.severity !== 'info');
   461→  const missingSPF = findings.some(f => f.title.toLowerCase().includes('spf') && f.severity !== 'info');
   462→  const missingDMARC = findings.some(f => f.title.toLowerCase().includes('dmarc') && f.severity !== 'info');
   463→  const hasCritVuln = findings.some(f => f.severity === 'critical');
   464→  const hasHighVuln = findings.some(f => f.severity === 'high');
   465→  const openCriticalPorts = findings.some(f => f.category === 'port' && f.severity === 'critical');
   466→  const corsOpen = findings.some(f => f.title.toLowerCase().includes('cors') && f.title.includes('*'));
   467→  const weakTls = findings.some(f => f.title.toLowerCase().includes('tls') && (f.title.includes('1.0') || f.title.includes('1.1')));
   468→
   469→  const countFailures = (...conds: boolean[]) => conds.filter(Boolean).length;
   470→
   471→  return {
   472→    'SOC2 Type II': {
   473→      passed: !hasCritVuln && !sslCrit && !missingHsts,
   474→      failures: countFailures(hasCritVuln, sslCrit, missingHsts, missingCsp, corsOpen),
   475→      details: hasCritVuln || sslCrit
   476→        ? `Critical issues: ${findings.filter(f => f.severity === 'critical').map(f => f.title).slice(0, 3).join(', ')}. CC6.1 (encryption) and CC7.1 (access controls) require immediate attention.`
   477→        : 'Encryption and access controls are properly configured for SOC2 CC requirements.',
   478→    },
   479→    'PCI-DSS v4.0': {
   480→      passed: !hasCritVuln && !sslCrit && !weakTls && !missingHsts,
   481→      failures: countFailures(sslCrit, weakTls, missingHsts, hasCritVuln, openCriticalPorts),
   482→      details: weakTls || sslCrit
   483→        ? `SSL/TLS does not meet PCI-DSS 4.1 requirements. ${weakTls ? 'TLS 1.0/1.1 must be disabled (Req 4.1).' : ''}${sslCrit ? 'Certificate issues detected.' : ''} Strong cryptography (Req 4.2.1) is mandatory.`
   484→        : 'Cardholder data environment encryption meets PCI-DSS requirements.',
   485→    },
   486→    'ISO 27001:2022': {
   487→      passed: !hasCritVuln && !openCriticalPorts,
   488→      failures: countFailures(hasCritVuln, openCriticalPorts, missingSPF, missingDMARC, missingHsts),
   489→      details: hasCritVuln
   490→        ? `Annex A controls need attention: ${findings.filter(f => f.severity === 'critical').map(f => f.title).slice(0, 3).join(', ')}. A.12.2 (data classification) and A.13.1 (network security) controls require review.`
   491→        : 'Information security management controls are adequate per Annex A requirements.',
   492→    },
   493→    'HIPAA': {
   494→      passed: !hasCritVuln && !sslCrit && !missingHsts,
   495→      failures: countFailures(sslCrit, missingHsts, hasCritVuln, missingCsp, corsOpen),
   496→      details: sslCrit || missingHsts
   497→        ? `ePHI transmission security (§164.312(e)) not met. ${missingHsts ? 'HSTS is required for HIPAA compliant HTTPS enforcement.' : ''}${sslCrit ? 'Certificate issues risk data in transit.' : ''}`
   498→        : 'Technical safeguards for protected health information (§164.312) appear adequate.',
   499→    },
   500→    'GDPR': {
   501→      passed: !hasCritVuln && !missingCsp && !corsOpen,
   502→      failures: countFailures(hasCritVuln, missingCsp, corsOpen, missingHsts, sslCrit),
   503→      details: missingCsp || corsOpen
   504→        ? `Data protection by design (Art. 25) requires proper security headers. ${missingCsp ? 'CSP is mandatory to prevent data injection.' : ''}${corsOpen ? 'Open CORS allows data exfiltration from any origin.' : ''}`
   505→        : 'Data protection measures (Art. 32) appear compliant.',
   506→    },
   507→    'NIST CSF 2.0': {
   508→      passed: !hasCritVuln,
   509→      failures: countFailures(hasCritVuln, openCriticalPorts, missingSPF, missingDMARC, weakTls),
   510→      details: hasCritVuln
   511→        ? `Identify (ID.RA) and Protect (PR.DS, PR.AC) functions need strengthening. ${findings.filter(f => f.severity === 'critical').length} critical findings require immediate risk management.`
   512→        : 'Core functions (Identify, Protect, Detect, Respond, Recover) are within acceptable risk tolerance.',
   513→    },
   514→  };
   515→}

   521→function answerQuestion(question: string, findings: Finding[], domain: string): string {
   522→  const q = question.toLowerCase();
   523→
   524→  // What should I fix first?
   525→  if (q.includes('fix') || q.includes('first') || q.includes('priority') || q.includes('urgent')) {
   526→    const crits = findings.filter(f => f.severity === 'critical');
   527→    const highs = findings.filter(f => f.severity === 'high');
   528→    if (crits.length === 0 && highs.length === 0) {
   529→      return `Good news — no critical or high severity findings for **${domain}**.\n\nYour current priorities should be:\n1. Address any medium-severity configuration issues\n2. Implement security headers if missing\n3. Set up continuous monitoring\n4. Schedule quarterly penetration tests`;
   530→    }
   531→    let resp = `## Priority Remediation Order for ${domain}\n\n`;
   532→    const all = [...crits, ...highs].slice(0, 8);
   533→    all.forEach((f, i) => {
   534→      const rem = findRemediation(f);
   535→      resp += `**${i + 1}. [${f.severity.toUpperCase()}] ${f.title}** — \`${f.asset}\`\n`;
   536→      if (rem?.priority) resp += `   Priority: ${rem.priority} | `;
   537→      if (rem?.cve) resp += `CVE: ${rem.cve} | `;
   538→      if (rem?.cvss) resp += `CVSS: ${rem.cvss}\n`;
   539→      resp += `   Quick Fix: ${(rem?.fix || '').split('\n')[0]}\n\n`;
   540→    });
   541→    return resp;
   542→  }
   543→
   544→  // Attack path questions
   545→  if (q.includes('attack') || q.includes('path') || q.includes('chain') || q.includes('exploit')) {
   546→    return buildAttackPaths(findings) + '\nWould you like me to elaborate on any specific attack path?';
   547→  }
   548→
   549→  // Compliance questions
   550→  if (q.includes('compliance') || q.includes('soc2') || q.includes('pci') || q.includes('hipaa') || q.includes('gdpr') || q.includes('iso') || q.includes('nist')) {
   551→    const frameworks = assessCompliance(findings);
   552→    let resp = '## Compliance Status Report\n\n';
   553→    for (const [fw, status] of Object.entries(frameworks)) {
   554→      const icon = status.passed ? '✅' : '❌';
   555→      resp += `### ${icon} ${fw}\n${status.details}\n\n`;
   556→    }
   557→    resp += '### Recommendation\n';
   558→    resp += 'Address all critical and high findings first — these impact every compliance framework. ';
   559→    resp += 'Then implement framework-specific controls as part of your compliance program.\n\n';
   560→    resp += 'Need details on a specific framework?';
   561→    return resp;
   562→  }
   563→
   564→  // SSL/TLS specific
   565→  if (q.includes('ssl') || q.includes('tls') || q.includes('certificate') || q.includes('https')) {
   566→    const sslFindings = findings.filter(f => f.category === 'ssl');
   567→    if (sslFindings.length === 0) {
   568→      return `No SSL/TLS issues detected for **${domain}**. Your HTTPS configuration appears secure.\n\n**Best practices to maintain:**\n- Keep certificates auto-renewed (certbot + cron)\n- Monitor with SSL Labs (aim for A+ rating)\n- Enforce TLS 1.2+ only\n- Enable HSTS preload`;
   569→    }
   570→    let resp = `## SSL/TLS Analysis for ${domain}\n\n`;
   571→    resp += `Found **${sslFindings.length} SSL/TLS findings**:\n\n`;
   572→    sslFindings.forEach(f => {
   573→      const rem = findRemediation(f);
   574→      resp += `**${f.severity.toUpperCase()}**: ${f.title}\n`;
   575→      if (rem) resp += `${rem.fix}\n\n`;
   576→    });
   577→    return resp;
   578→  }
   579→
   580→  // Port/infrastructure questions
   581→  if (q.includes('port') || q.includes('infrastructure') || q.includes('server') || q.includes('ssh')) {
   582→    const portFindings = findings.filter(f => f.category === 'port');
   583→    const techFindings = findings.filter(f => f.category === 'technology');
   584→    let resp = `## Infrastructure Analysis for ${domain}\n\n`;
   585→    if (portFindings.length > 0) {
   586→      resp += `### Open Ports (${portFindings.length})\n`;
   587→      portFindings.forEach(f => {
   588→        resp += `- \`${f.asset}\` — ${f.severity.toUpperCase()}: ${f.title}\n`;
   589→      });
   590→      resp += '\n';
   591→    }
   592→    if (techFindings.length > 0) {
   593→      resp += `### Detected Technologies (${techFindings.length})\n`;
   594→      techFindings.slice(0, 10).forEach(f => {
   595→        resp += `- ${f.asset}\n`;
   596→      });
   597→      resp += '\n';
   598→    }
   599→    resp += '### Hardening Recommendations\n';
   600→    resp += '1. Close unnecessary ports via firewall\n2. Use VPN for management access\n3. Implement network segmentation\n4. Keep all software updated\n5. Use infrastructure-as-code for reproducible security';
   601→    return resp;
   602→  }
   603→
   604→  // Subdomain questions
   605→  if (q.includes('subdomain') || q.includes('asset') || q.includes('surface') || q.includes('scope')) {
   606→    const subs = findings.filter(f => f.category === 'subdomain');
   607→    let resp = `## Attack Surface: ${domain}\n\n`;
   608→    resp += `Discovered **${subs.length} subdomains**:\n\n`;
   609→    subs.slice(0, 15).forEach(f => {
   610→      resp += `- \`${f.asset}\` — ${f.severity.toUpperCase()}\n`;
   611→    });
   612→    if (subs.length > 15) resp += `\n... and ${subs.length - 15} more\n`;
   613→    resp += '\n### Recommendations\n';
   614→    resp += '1. Audit all subdomains for necessity — remove unused ones\n';
   615→    resp += '2. Check for dangling DNS records (subdomain takeover)\n';
   616→    resp += '3. Ensure all subdomains have valid SSL certificates\n';
   617→    resp += '4. Implement consistent security headers across all subdomains\n';
   618→    resp += '5. Monitor for new subdomain additions with certificate transparency logs';
   619→    return resp;
   620→  }
   621→
   622→  // Risk score
   623→  if (q.includes('risk') || q.includes('score') || q.includes('rating') || q.includes('how bad')) {
   624→    const criticals = findings.filter(f => f.severity === 'critical').length;
   625→    const highs = findings.filter(f => f.severity === 'high').length;
   626→    const score = Math.min(100, criticals * 25 + highs * 15 + findings.filter(f => f.severity === 'medium').length * 8);
   627→    const level = score > 70 ? 'CRITICAL' : score > 40 ? 'HIGH' : score > 20 ? 'MODERATE' : 'LOW';
   628→
   629→    return `## Risk Assessment: ${domain}\n\n` +
   630→      `**Overall Risk Score**: ${score}/100 — **${level}**\n\n` +
   631→      `| Metric | Value |\n|--------|-------|\n` +
   632→      `| Critical | ${criticals} |\n` +
   633→      `| High | ${highs} |\n` +
   634→      `| Medium | ${findings.filter(f => f.severity === 'medium').length} |\n` +
   635→      `| Low/Info | ${findings.filter(f => ['low', 'info'].includes(f.severity)).length} |\n` +
   636→      `| Total Attack Surface | ${findings.length} findings |\n\n` +
   637→      (score > 70 ? 'This risk level requires **immediate executive attention**. You are actively vulnerable to data breaches.' :
   638→       score > 40 ? 'This risk level needs **prompt remediation**. Allocate security resources this sprint.' :
   639→       'Your risk level is manageable. Continue monitoring and hardening.');
   640→  }
   641→
   642→  // General / fallback
   643→  return `## Analysis for ${domain}\n\n` +
   644→    `I have **${findings.length} findings** for this target.\n\n` +
   645→    `**Ask me about:**\n` +
   646→    `- What to fix first (priorities)\n` +
   647→    `- Attack path analysis\n` +
   648→    `- Compliance status (SOC2, PCI-DSS, HIPAA, GDPR, ISO 27001)\n` +
   649→    `- SSL/TLS configuration\n` +
   650→    `- Open ports and infrastructure\n` +
   651→    `- Subdomain inventory\n` +
   652→    `- Risk score breakdown\n` +
   653→    `- Remediation steps for any finding\n\n` +
   654→    `Just type your question and I'll analyze your attack surface.`;
   655→}

   669→        response: 'No scan data available. Please run a scan first, then I can analyze your attack surface.\n\nClick **"New Scan"** to get started.',
   670→      });
   671→    }
   672→
   673→    // Simulate processing delay for realism
   674→    await new Promise(resolve => setTimeout(resolve, 400 + Math.random() * 800));
   675→
   676→    let response: string;
   677→
   678→    if (type === 'full-analysis') {
   679→      response = generateFullAnalysis(findings, domain);
   680→    } else if (type === 'attack-paths') {
   681→      response = buildAttackPaths(findings);
   682→    } else if (type === 'compliance') {
   683→      const frameworks = assessCompliance(findings);
   684→      response = '## Compliance Assessment\n\n';
   685→      for (const [fw, status] of Object.entries(frameworks)) {
   686→        const icon = status.passed ? '✅' : '❌';
   687→        response += `### ${icon} ${fw}\n${status.details}\n\n`;
   688→      }
   689→    } else if (question) {
   690→      response = answerQuestion(question, findings, domain);
   691→    } else {
   692→      response = generateFullAnalysis(findings, domain);
   693→    }
   694→
   695→    return NextResponse.json({ success: true, response });
   696→  } catch (error) {
   697→    console.error('AI Advisor error:', error);
   698→    return NextResponse.json({ success: false, error: 'Analysis failed' }, { status: 500 });
   699→  }
