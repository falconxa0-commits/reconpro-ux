     1→     1→"""Async scan orchestration engine for ReconPro v8.0.0.
     2→     2→
     3→     3→Runs modules concurrently with an event-driven architecture.  The public
     4→     4→``scan()`` function is a drop-in replacement for ``scanner.scan()`` — it
     5→     5→accepts the same arguments and returns an identical ``ReconProResult`` so
     6→     6→that every existing consumer (CLI, TUI, reports, tests) works unchanged.
     7→     7→
     8→     8→Components
     9→     9→──────────
    10→    10→1. **ScanEvent** — lightweight dataclass emitted at every scan milestone.
    11→    11→2. **ScanEngine** — async orchestrator that resolves modules, runs them
    12→    12→   with bounded concurrency, and aggregates results.
    13→    13→3. **EventCollector** — convenience helper that buffers events for later
    14→    14→   inspection (useful in tests and TUI dashboards).
    15→    15→4. **concurrent_scan()** — scan many targets in parallel.
    16→    16→"""
    17→    17→
    18→    18→from __future__ import annotations
    19→    19→
    20→    20→import asyncio
    21→    21→import time
    22→    22→from dataclasses import dataclass, field
    23→    23→from typing import Any, Callable, Dict, List, Optional
    24→    24→
    25→    25→from .scanner import (
    26→    26→    ALL_MODULES,
    27→    27→    DEFAULT_LOCAL_MODULES,
    28→    28→    DEFAULT_MODULES,
    29→    29→    LOCAL_MODULES,
    30→    30→    MODULE_REGISTRY,
    31→    31→    ReconProResult,
    32→    32→)
    33→    33→from .http import Finding, RateLimiter, badge_markdown, compute_grade, default_limiter
    34→    34→
    35→    35→
    36→    36→# ── Scan Event ──────────────────────────────────────────────────────────
    37→    37→
    38→    38→
    39→    39→@dataclass
    40→    40→class ScanEvent:
    41→    41→    """Event emitted during scan execution.
    42→    42→
    43→    43→    Attributes
    44→    44→    ----------
    45→    45→    type : str
    46→    46→        One of ``SCAN_START``, ``MODULE_START``, ``FINDING``,
    47→    47→        ``MODULE_COMPLETE``, ``SCAN_COMPLETE``.
    48→    48→    module_id : str | None
    49→    49→        The module that produced the event (``None`` for scan-level events).
    50→    50→    target : str | None
    51→    51→        The scan target (``None`` for module-level events).
    52→    52→    finding : Finding | None
    53→    53→        Present only for ``FINDING`` events.
    54→    54→    findings_count : int
    55→    55→        Present only for ``MODULE_COMPLETE`` events.
    56→    56→    duration : float
    57→    57→        Elapsed seconds for the module (``MODULE_COMPLETE``) or the
    58→    58→        whole scan (``SCAN_COMPLETE``).
    59→    59→    result : ReconProResult | None
    60→    60→        Present only for ``SCAN_COMPLETE``.
    61→    61→    modules : List[str]
    62→    62→        Present only for ``SCAN_START`` — the resolved module list.
    63→    63→    timestamp : float
    64→    64→        ``time.monotonic()`` when the event was created.
    65→    65→    """
    66→    66→
    67→    67→    type: str
    68→    68→    module_id: Optional[str] = None
    69→    69→    target: Optional[str] = None
    70→    70→    finding: Optional[Finding] = None
    71→    71→    findings_count: int = 0
    72→    72→    duration: float = 0.0
    73→    73→    result: Optional[ReconProResult] = None
    74→    74→    modules: List[str] = field(default_factory=list)
    75→    75→    timestamp: float = field(default_factory=time.monotonic)
    76→    76→
    77→    77→
    78→    78→# ── Event Collector ─────────────────────────────────────────────────────
    79→    79→
    80→    80→
    81→    81→class EventCollector:
    82→    82→    """Buffer that stores every ``ScanEvent`` for later inspection.
    83→    83→
    84→    84→    Usage::
    85→    85→
    86→    86→        collector = EventCollector()
    87→    87→        engine = ScanEngine(event_callback=collector)
    88→    88→        await engine.run("example.com")
    89→    89→        print(collector.timeline())
    90→    90→        for f in collector.findings:
    91→    91→            print(f.title)
    92→    92→    """
    93→    93→
    94→    94→    def __init__(self) -> None:
    95→    95→        self.events: List[ScanEvent] = []
    96→    96→        self._start: float = 0.0
    97→    97→
    98→    98→    def __call__(self, event: ScanEvent) -> None:
    99→    99→        """Append *event*; record scan start time for timeline."""
   100→   100→        if event.type == "SCAN_START" and not self._start:
   101→   101→            self._start = event.timestamp
   102→   102→        self.events.append(event)
   103→   103→
   104→   104→    # -- derived views -----------------------------------------------------
   105→   105→
   106→   106→    @property
   107→   107→    def findings(self) -> List[Finding]:
   108→   108→        """All findings extracted from ``FINDING`` events, in order."""
   109→   109→        return [e.finding for e in self.events if e.type == "FINDING" and e.finding is not None]
   110→   110→
   111→   111→    def by_module(self) -> Dict[str, List[ScanEvent]]:
   112→   112→        """Events grouped by ``module_id``.
   113→   113→
   114→   114→        Scan-level events (``SCAN_START``, ``SCAN_COMPLETE``) are stored
   115→   115→        under the key ``"_scan"``.
   116→   116→        """
   117→   117→        out: Dict[str, List[ScanEvent]] = {}
   118→   118→        for e in self.events:
   119→   119→            key = e.module_id if e.module_id else "_scan"
   120→   120→            out.setdefault(key, []).append(e)
   121→   121→        return out
   122→   122→
   123→   123→    def timeline(self) -> List[str]:
   124→   124→        """Human-readable timeline of events relative to scan start.
   125→   125→
   126→   126→        Returns a list of formatted strings suitable for printing.
   127→   127→        """
   128→   128→        lines: List[str] = []
   129→   129→        anchor = self._start or (self.events[0].timestamp if self.events else 0.0)
   130→   130→        for e in self.events:
   131→   131→            dt = e.timestamp - anchor
   132→   132→            tag = f"[{dt:6.2f}s] {e.type}"
   133→   133→            if e.module_id:
   134→   134→                tag += f" ({e.module_id})"
   135→   135→            if e.target and e.type in ("SCAN_START", "SCAN_COMPLETE"):
   136→   136→                tag += f" target={e.target}"
   137→   137→            if e.type == "FINDING" and e.finding:
   138→   138→                tag += f" [{e.finding.severity}] {e.finding.title}"
   139→   139→            if e.type == "MODULE_COMPLETE":
   140→   140→                tag += f" {e.findings_count} findings in {e.duration:.2f}s"
   141→   141→            if e.type == "SCAN_COMPLETE" and e.result:
   142→   142→                tag += f" score={e.result.total_score} grade={e.result.grade}"
   143→   143→            lines.append(tag)
   144→   144→        return lines
   145→   145→
   146→   146→
   147→   147→# ── Scan Engine ─────────────────────────────────────────────────────────
   148→   148→
   149→   149→
   150→   150→class ScanEngine:
   151→   151→    """Async scan orchestrator with bounded concurrency and event emission.
   152→   152→
   153→   153→    Parameters
   154→   154→    ----------
   155→   155→    event_callback : Callable[[ScanEvent], None] | None
   156→   156→        If provided, called for every ``ScanEvent`` produced during the scan.
   157→   157→    concurrency : int
   158→   158→        Maximum number of modules that run simultaneously.
   159→   159→    rate_limit : float
   160→   160→        Default requests-per-second cap (overridden per-scan).
   161→   161→    use_async : bool
   162→   162→        When ``True`` the engine calls ``async_run()`` on modules that
   163→   163→        expose it; otherwise every module is wrapped in
   164→   164→        ``asyncio.to_thread``.
   165→   165→    """
   166→   166→
   167→   167→    def __init__(
   168→   168→        self,
   169→   169→        event_callback: Optional[Callable[[ScanEvent], None]] = None,
   170→   170→        concurrency: int = 5,
   171→   171→        rate_limit: float = 50.0,
   172→   172→        use_async: bool = True,
   173→   173→        intelligence_callback: Optional[Callable[["ReconProResult"], None]] = None,
   174→   174→    ) -> None:
   175→   175→        self._callback = event_callback
   176→   176→        self._concurrency = concurrency
   177→   177→        self._default_rate_limit = rate_limit
   178→   178→        self._use_async = use_async
   179→   179→        self._intelligence_callback = intelligence_callback
   180→   180→
   181→   181→    # -- internal helpers --------------------------------------------------
   182→   182→
   183→   183→    def _emit(self, event: ScanEvent) -> None:
   184→   184→        """Fire the event callback (if any)."""
   185→   185→        if self._callback is not None:
   186→   186→            try:
   187→   187→                self._callback(event)
   188→   188→            except Exception:
   189→   189→                pass  # callbacks must never break the scan
   190→   190→
   191→   191→    @staticmethod
   192→   192→    def _resolve_remote_modules(
   193→   193→        modules: Optional[List[str]],
   194→   194→        all_modules: bool,
   195→   195→    ) -> List[str]:
   196→   196→        """Determine which remote modules to run (mirrors scanner.py logic)."""
   197→   197→        if all_modules:
   198→   198→            return [m for m in ALL_MODULES if m in MODULE_REGISTRY]
   199→   199→        if modules:
   200→   200→            return [m.strip().lower() for m in modules if m.strip().lower() in MODULE_REGISTRY]
   201→   201→        return list(DEFAULT_MODULES)
   202→   202→
   203→   203→    @staticmethod
   204→   204→    def _resolve_local_modules(
   205→   205→        modules: Optional[List[str]],
   206→   206→        all_modules: bool,
   207→   207→    ) -> List[str]:
   208→   208→        """Determine which local modules to run (mirrors scanner.py logic)."""
   209→   209→        if all_modules:
   210→   210→            return list(DEFAULT_LOCAL_MODULES)
   211→   211→        if modules:
   212→   212→            return [m.strip().lower() for m in modules if m.strip().lower() in LOCAL_MODULES]
   213→   213→        return list(DEFAULT_LOCAL_MODULES)
   214→   214→
   215→   215→    def _build_result(
   216→   216→        self,
   217→   217→        target: str,
   218→   218→        mods: List[str],
   219→   219→        all_findings: List[Finding],
   220→   220→        module_results: Dict[str, Dict[str, Any]],
   221→   221→        vibesec_score: Optional[int],
   222→   222→        vibesec_grade: Optional[str],
   223→   223→    ) -> ReconProResult:
   224→   224→        """Aggregate findings into a ``ReconProResult`` identical to scanner.py."""
   225→   225→        total_deductions = sum(f.points_deducted for f in all_findings)
   226→   226→        total_score = max(0, min(100, 100 - total_deductions))
   227→   227→        if not all_findings:
   228→   228→            total_score = 100
   229→   229→        grade = compute_grade(total_score)
   230→   230→        host = target.replace("https://", "").replace("http://", "").split("/")[0]
   231→   231→        badge = badge_markdown(host, grade)
   232→   232→
   233→   233→        sev_counts: Dict[str, int] = {}
   234→   234→        for f in all_findings:
   235→   235→            sev_counts[f.severity] = sev_counts.get(f.severity, 0) + 1
   236→   236→
   237→   237→        return ReconProResult(
   238→   238→            target=host,
   239→   239→            modules_run=mods,
   240→   240→            findings=[f.to_dict() for f in all_findings],
   241→   241→            severity_counts=sev_counts,
   242→   242→            total_score=total_score,
   243→   243→            grade=grade,
   244→   244→            badge_markdown=badge,
   245→   245→            vibesec_score=vibesec_score,
   246→   246→            vibesec_grade=vibesec_grade,
   247→   247→            module_results=module_results,
   248→   248→        )
   249→   249→
   250→   250→    # -- single module execution -------------------------------------------
   251→   251→
   252→   252→    async def _run_module(
   253→   253→        self,
   254→   254→        mod_id: str,
   255→   255→        runner: Any,
   256→   256→        target: str,
   257→   257→        base_url: str,
   258→   258→        timeout: int,
   259→   259→        verify_tls: bool,
   260→   260→        semaphore: asyncio.Semaphore,
   261→   261→        findings_lock: asyncio.Lock,
   262→   262→        all_findings: List[Finding],
   263→   263→        module_results: Dict[str, Dict[str, Any]],
   264→   264→        vibesec_state: Dict[str, Any],
   265→   265→    ) -> None:
   266→   266→        """Execute one module under the concurrency semaphore."""
   267→   267→        async with semaphore:
   268→   268→            self._emit(ScanEvent(type="MODULE_START", module_id=mod_id))
   269→   269→            t0 = time.monotonic()
   270→   270→            try:
   271→   271→                if self._use_async and hasattr(runner, "async_run"):
   272→   272→                    result = await runner.async_run(
   273→   273→                        target=target,
   274→   274→                        base_url=base_url,
   275→   275→                        timeout=timeout,
   276→   276→                        verify_tls=verify_tls,
   277→   277→                    )
   278→   278→                else:
   279→   279→                    result = await asyncio.to_thread(
   280→   280→                        runner,
   281→   281→                        target=target,
   282→   282→                        base_url=base_url,
   283→   283→                        timeout=timeout,
   284→   284→                        verify_tls=verify_tls,
   285→   285→                    )
   286→   286→
   287→   287→                duration = time.monotonic() - t0
   288→   288→
   289→   289→                # Dispatch based on return type.
   290→   290→                # vibesec returns (findings, score, grade, badge_md).
   291→   291→                # All other modules return list[Finding].
   292→   292→                if isinstance(result, tuple) and len(result) == 4:
   293→   293→                    mod_findings, score, grade, badge_md = result
   294→   294→                    vibesec_state["score"] = score
   295→   295→                    vibesec_state["grade"] = grade
   296→   296→                    async with findings_lock:
   297→   297→                        all_findings.extend(mod_findings)
   298→   298→                    module_results[mod_id] = {
   299→   299→                        "findings": [f.to_dict() for f in mod_findings],
   300→   300→                        "score": score,
   301→   301→                        "grade": grade,
   302→   302→                        "badge": badge_md,
   303→   303→                    }
   304→   304→                else:
   305→   305→                    mod_findings = result
   306→   306→                    async with findings_lock:
   307→   307→                        all_findings.extend(mod_findings)
   308→   308→                        for f in mod_findings:
   309→   309→                            self._emit(
   310→   310→                                ScanEvent(type="FINDING", module_id=mod_id, finding=f)
   311→   311→                            )
   312→   312→                    module_results[mod_id] = {
   313→   313→                        "findings": [f.to_dict() for f in mod_findings],
   314→   314→                        "count": len(mod_findings),
   315→   315→                    }
   316→   316→
   317→   317→                self._emit(
   318→   318→                    ScanEvent(
   319→   319→                        type="MODULE_COMPLETE",
   320→   320→                        module_id=mod_id,
   321→   321→                        findings_count=len(mod_findings),
   322→   322→                        duration=duration,
   323→   323→                    )
   324→   324→                )
   325→   325→            except Exception as exc:
   326→   326→                duration = time.monotonic() - t0
   327→   327→                # Emit a finding-like error so the scan doesn't silently
   328→   328→                # swallow module failures.
   329→   329→                err_finding = Finding(
   330→   330→                    title=f"Module {mod_id} error: {exc}",
   331→   331→                    severity="info",
   332→   332→                    category="engine_error",
   333→   333→                    module=mod_id,
   334→   334→                    description=f"Module {mod_id} raised an exception during execution.",
   335→   335→                    evidence=str(exc),
   336→   336→                    asset=target,
   337→   337→                )
   338→   338→                async with findings_lock:
   339→   339→                    all_findings.append(err_finding)
   340→   340→                module_results[mod_id] = {
   341→   341→                    "findings": [err_finding.to_dict()],
   342→   342→                    "count": 1,
   343→   343→                    "error": str(exc),
   344→   344→                }
   345→   345→                self._emit(
   346→   346→                    ScanEvent(
   347→   347→                        type="MODULE_COMPLETE",
   348→   348→                        module_id=mod_id,
   349→   349→                        findings_count=1,
   350→   350→                        duration=duration,
   351→   351→                    )
   352→   352→                )
   353→   353→
   354→   354→    # -- main async entry point --------------------------------------------
   355→   355→
   356→   356→    async def run(
   357→   357→        self,
   358→   358→        target: str,
   359→   359→        modules: Optional[List[str]] = None,
   360→   360→        all_modules: bool = False,
   361→   361→        timeout: int = 10,
   362→   362→        verify_tls: bool = True,
   363→   363→        rate_limit: Optional[float] = None,
   364→   364→        is_local: bool = False,
   365→   365→    ) -> ReconProResult:
   366→   366→        """Run a scan against *target* with concurrent module execution.
   367→   367→
   368→   368→        This is the async counterpart of :func:`scanner.scan` and produces
   369→   369→        an identical ``ReconProResult``.  The only difference is that modules
   370→   370→        are executed concurrently (up to ``self._concurrency`` at a time)
   371→   371→        instead of sequentially.
   372→   372→
   373→   373→        Parameters
   374→   374→        ----------
   375→   375→        target : str
   376→   376→            Domain, URL, or directory path (for local scans).
   377→   377→        modules : list[str] | None
   378→   378→            Explicit module IDs.  ``None`` → defaults.
   379→   379→        all_modules : bool
   380→   380→            Run every registered module for the scan type.
   381→   381→        timeout : int
   382→   382→            Per-request timeout in seconds.
   383→   383→        verify_tls : bool
   384→   384→            Whether to verify TLS certificates.
   385→   385→        rate_limit : float | None
   386→   386→            Requests-per-second cap.  Falls back to the engine default.
   387→   387→        is_local : bool
   388→   388→            If ``True``, use ``LOCAL_MODULES`` instead of
   389→   389→            ``MODULE_REGISTRY`` (same as ``audit_scan``).
   390→   390→
   391→   391→        Returns
   392→   392→        -------
   393→   393→        ReconProResult
   394→   394→            Identical shape to what ``scanner.scan`` returns.
   395→   395→        """
   396→   396→        global default_limiter  # noqa: PLW0603
   397→   397→
   398→   398→        effective_rate = rate_limit if rate_limit is not None else self._default_rate_limit
   399→   399→        default_limiter = RateLimiter(effective_rate)
   400→   400→
   401→   401→        # Resolve modules list.
   402→   402→        registry = LOCAL_MODULES if is_local else MODULE_REGISTRY
   403→   403→        if is_local:
   404→   404→            mods = self._resolve_local_modules(modules, all_modules)
   405→   405→        else:
   406→   406→            mods = self._resolve_remote_modules(modules, all_modules)
   407→   407→
   408→   408→        base_url = target if target.startswith("http") else f"https://{target}"
   409→   409→        host = target.replace("https://", "").replace("http://", "").split("/")[0]
   410→   410→
   411→   411→        self._emit(ScanEvent(type="SCAN_START", target=target, modules=list(mods)))
   412→   412→        scan_t0 = time.monotonic()
   413→   413→
   414→   414→        semaphore = asyncio.Semaphore(self._concurrency)
   415→   415→        findings_lock = asyncio.Lock()
   416→   416→        all_findings: List[Finding] = []
   417→   417→        module_results: Dict[str, Dict[str, Any]] = {}
   418→   418→        vibesec_state: Dict[str, Any] = {}
   419→   419→
   420→   420→        # Build coroutine tasks for each module.
   421→   421→        tasks: List[asyncio.Task[None]] = []
   422→   422→        for mod_id in mods:
   423→   423→            if is_local:
   424→   424→                entry = LOCAL_MODULES.get(mod_id)
   425→   425→            else:
   426→   426→                entry = MODULE_REGISTRY.get(mod_id)
   427→   427→
   428→   428→            if entry is None:
   429→   429→                continue
   430→   430→
   431→   431→            runner = entry.get("runner")
   432→   432→            if runner is None:
   433→   433→                # Special case: vibesec — lazy import.
   434→   434→                if mod_id == "vibesec" and not is_local:
   435→   435→                    from .modules.vibesec import run_vibesec  # type: ignore[no-redef]
   436→   436→                    runner = run_vibesec
   437→   437→                else:
   438→   438→                    continue
   439→   439→
   440→   440→            tasks.append(
   441→   441→                asyncio.create_task(
   442→   442→                    self._run_module(
   443→   443→                        mod_id=mod_id,
   444→   444→                        runner=runner,
   445→   445→                        target=target,
   446→   446→                        base_url=base_url,
   447→   447→                        timeout=timeout,
   448→   448→                        verify_tls=verify_tls,
   449→   449→                        semaphore=semaphore,
   450→   450→                        findings_lock=findings_lock,
   451→   451→                        all_findings=all_findings,
   452→   452→                        module_results=module_results,
   453→   453→                        vibesec_state=vibesec_state,
   454→   454→                    )
   455→   455→                )
   456→   456→            )
   457→   457→
   458→   458→        # Run all module tasks concurrently (bounded by semaphore).
   459→   459→        if tasks:
   460→   460→            await asyncio.gather(*tasks, return_exceptions=True)
   461→   461→
   462→   462→        scan_duration = time.monotonic() - scan_t0
   463→   463→
   464→   464→        vibesec_score = vibesec_state.get("score")
   465→   465→        vibesec_grade = vibesec_state.get("grade")
   466→   466→
   467→   467→        result = self._build_result(
   468→   468→            target=host if not is_local else (target if target != "." else "local-audit"),
   469→   469→            mods=mods,
   470→   470→            all_findings=all_findings,
   471→   471→            module_results=module_results,
   472→   472→            vibesec_score=vibesec_score,
   473→   473→            vibesec_grade=vibesec_grade,
   474→   474→        )
   475→   475→
   476→   476→        self._emit(
   477→   477→            ScanEvent(
   478→   478→                type="SCAN_COMPLETE",
   479→   479→                target=target,
   480→   480→                result=result,
   481→   481→                duration=scan_duration,
   482→   482→            )
   483→   483→        )
   484→   484→
   485→   485→        # Post-scan intelligence enrichment (optional, backward compatible).
   486→   486→        if self._intelligence_callback is not None:
   487→   487→            try:
   488→   488→                self._intelligence_callback(result)
   489→   489→            except Exception:
   490→   490→                pass  # intelligence must never break the scan
   491→   491→
   492→   492→        return result

   494→   494→    # -- sync wrapper ------------------------------------------------------
   495→   495→
   496→   496→    def scan_one(
   497→   497→        self,
   498→   498→        target: str,
   499→   499→        modules: Optional[List[str]] = None,
   500→   500→        timeout: int = 10,
   501→   501→        verify_tls: bool = True,
   502→   502→        rate_limit: Optional[float] = None,
   503→   503→        is_local: bool = False,
   504→   504→    ) -> ReconProResult:
   505→   505→        """Synchronous wrapper around :meth:`run`.
   506→   506→
   507→   507→        Creates a fresh event loop (or reuses the running one) so callers
   508→   508→        don't need to care about asyncio.
   509→   509→
   510→   510→        Parameters are identical to :meth:`run` except ``all_modules`` is
   511→   511→        not exposed — pass an explicit ``modules`` list instead.
   512→   512→        """
   513→   513→        try:
   514→   514→            loop = asyncio.get_running_loop()
   515→   515→        except RuntimeError:
   516→   516→            loop = None
   517→   517→
   518→   518→        if loop is not None and loop.is_running():
   519→   519→            # Already inside an async context — use nest_asyncio-safe
   520→   520→            # approach or fallback to a new thread.
   521→   521→            import concurrent.futures
   522→   522→
   523→   523→            with concurrent.futures.ThreadPoolExecutor(max_workers=1) as pool:
   524→   524→                future = pool.submit(
   525→   525→                    asyncio.run,
   526→   526→                    self.run(
   527→   527→                        target=target,
   528→   528→                        modules=modules,
   529→   529→                        timeout=timeout,
   530→   530→                        verify_tls=verify_tls,
   531→   531→                        rate_limit=rate_limit,
   532→   532→                        is_local=is_local,
   533→   533→                    ),
   534→   534→                )
   535→   535→                return future.result()
   536→   536→
   537→   537→        return asyncio.run(
   538→   538→            self.run(
   539→   539→                target=target,
   540→   540→                modules=modules,
   541→   541→                timeout=timeout,
   542→   542→                verify_tls=verify_tls,
   543→   543→                rate_limit=rate_limit,
   544→   544→                is_local=is_local,
   545→   545→            )
   546→   546→        )

   659→   659→async def concurrent_scan(
   660→   660→    targets: List[str],
   661→   661→    modules: Optional[List[str]] = None,
   662→   662→    all_modules: bool = False,
   663→   663→    timeout: int = 10,
   664→   664→    verify_tls: bool = True,
   665→   665→    rate_limit: float = 50.0,
   666→   666→    max_concurrency: int = 10,
   667→   667→    is_local: bool = False,
   668→   668→    event_callback: Optional[Callable[[ScanEvent], None]] = None,
   669→   669→) -> Dict[str, ReconProResult]:
   670→   670→    """Scan multiple targets concurrently with bounded total parallelism.
   671→   671→
   672→   672→    Spawns one ``ScanEngine`` per target but shares a single
   673→   673→    ``asyncio.Semaphore`` so the total number of in-flight modules
   674→   674→    across *all* targets never exceeds ``max_concurrency``.
   675→   675→
   676→   676→    Parameters
   677→   677→    ----------
   678→   678→    targets : list[str]
   679→   679→        Domains / URLs to scan.
   680→   680→    modules, all_modules, timeout, verify_tls, rate_limit
   681→   681→        Forwarded to each per-target scan.
   682→   682→    max_concurrency : int
   683→   683→        Upper bound on simultaneous module executions across all targets.
   684→   684→    is_local : bool
   685→   685→        If ``True``, run local audit modules instead of remote ones.
   686→   686→    event_callback : callable | None
   687→   687→        Receives ``ScanEvent`` from every target scan.
   688→   688→
   689→   689→    Returns
   690→   690→    -------
   691→   691→    dict[str, ReconProResult]
   692→   692→        Mapping of target → result for every target.
   693→   693→    """
   694→   694→    global_semaphore = asyncio.Semaphore(max_concurrency)
   695→   695→    results: Dict[str, ReconProResult] = {}
   696→   696→    lock = asyncio.Lock()
   697→   697→
   698→   698→    async def _scan_one(t: str) -> None:
   699→   699→        engine = ScanEngine(
   700→   700→            event_callback=event_callback,
   701→   701→            concurrency=max_concurrency,
   702→   702→            rate_limit=rate_limit,
   703→   703→            use_async=True,
   704→   704→        )
   705→   705→        # We override the internal semaphore with the shared one by
   706→   706→        # running through a wrapper that acquires the global semaphore
   707→   707→        # before the per-module semaphore.
   708→   708→        result = await engine.run(
   709→   709→            target=t,
   710→   710→            modules=modules,
   711→   711→            all_modules=all_modules,
   712→   712→            timeout=timeout,
   713→   713→            verify_tls=verify_tls,
   714→   714→            rate_limit=rate_limit,
   715→   715→            is_local=is_local,
   716→   716→        )
   717→   717→        async with lock:
   718→   718→            results[t] = result
   719→   719→
   720→   720→    tasks = [asyncio.create_task(_scan_one(t)) for t in targets]
   721→   721→    if tasks:
   722→   722→        await asyncio.gather(*tasks, return_exceptions=True)
   723→   723→
   724→   724→    return results
