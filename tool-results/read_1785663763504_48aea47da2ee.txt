   266→            chat.write("")

   272→            chat.write(f"[msg-nexus]{msg}[/]")
   273→        else:
   274→            chat.write(f"[boot-line]{msg} \u2714[/]")

   286→    def _process_command(self, text):
   287→        low = text.lower().strip()
   288→        chat = self.query_one("#chat-log", RichLog)
   289→        if low in ("quit", "exit", "q"):
   290→            self.quit_app(); return
   291→        if low in ("help", "?", "commands"):
   292→            self._show_help(); return
   293→        if low == "clear":
   294→            self.clear_chat(); return
   295→        if low.startswith("scan ") or ("." in low and " " not in low and not low.startswith(("agent", "blitz", "subdomain", "screenshot", "report", "history", "compare", "secrets", "ports", "audit", "dev", "doctor", "open ", "schedule", "plugin", "modules", "list"))) or low.startswith("http"):
   296→            target = low.replace("scan ", "").strip().split(" with ")[0].strip()
   297→            if not target or target == "scan":
   298→                chat.write("[msg-warning]Usage: scan <target> [with <modules>][/]"); return
   299→            modules = None
   300→            if "with " in low:
   301→                import re
   302→                m = re.search(r'with\s+([\w,\s]+)', low, re.IGNORECASE)
   303→                if m:
   304→                    modules = [x.strip().lower() for x in m.group(1).split(",") if x.strip()]
   305→            self._run_scan(target, modules); return
   306→        if low.startswith("audit"):
   307→            self._run_audit(); return
   308→        if low.startswith("dev"):
   309→            path = "."; parts = low.split()
   310→            if len(parts) > 1 and parts[1] not in ("scan", "check"): path = parts[1]
   311→            self._run_dev_scan(path); return
   312→        if low.startswith("doctor"):
   313→            self._run_doctor(); return
   314→        if low.startswith("blitz"):
   315→            import re
   316→            targets = re.findall(r'[\w\-]+\.[\w]{2,}', low)
   317→            if len(targets) < 2:
   318→                chat.write("[msg-warning]Usage: blitz t1.com t2.com t3.com[/]"); return
   319→            self._run_blitz(targets); return
   320→        if low.startswith("subdomain"):
   321→            parts = low.split()
   322→            if len(parts) < 2: chat.write("[msg-warning]Usage: subdomains <domain>[/]"); return
   323→            self._discover_subdomains(parts[1]); return
   324→        if low.startswith("agent "):
   325→            goal = text.split(" ", 1)[1]; self._run_agent_goal(goal); return
   326→        if low.startswith("report"):
   327→            self._generate_report(); return
   328→        if low.startswith("history"):
   329→            self._show_history(); return
   330→        if low.startswith("compare"):
   331→            self._compare_scans(); return
   332→        if low.startswith("secret"):
   333→            self._hunt_secrets(); return
   334→        if low.startswith("port"):
   335→            self._check_ports(); return
   336→        if low.startswith("screenshot"):
   337→            parts = low.split()
   338→            if len(parts) < 2: chat.write("[msg-warning]Usage: screenshot <url>[/]"); return
   339→            self._take_screenshot(parts[1]); return
   340→        if low.startswith("open "):
   341→            url = text.split(" ", 1)[1]; self._open_browser(url); return
   342→        if low == "score":
   343→            self._show_last_score(); return
   344→        if low.startswith("module") or low.startswith("list"):
   345→            self._list_modules(); return
   346→        self._run_agent_goal(text)

   465→    def _on_scan_complete(self, result):
   466→        self.scanning = False
   467→        chat = self.query_one("#chat-log", RichLog)
   468→        feed = self.query_one("#feed-log", RichLog)
   469→        mod_log = self.query_one("#module-log", RichLog)
   470→        self.score = result.total_score
   471→        self.grade = result.grade
   472→        for mod_id in result.modules_run:
   473→            count = len(result.module_results.get(mod_id, {}).get("findings", []))
   474→            self._update_module_status(mod_id, "done", count)
   475→        mod_log.clear()
   476→        mod_log.write("[bold #00ffcc]\u2550\u2550 SCAN COMPLETE[/]")

   606→            color = "msg-success" if change > 0 else ("msg-error" if change < 0 else "msg-system")
