    34→# ── Benchmark helpers ───────────────────────────────────────────────────
    35→
    36→@dataclass
    37→class BenchResult:
    38→    """Statistical summary of a benchmark run."""
    39→    name: str
    40→    iterations: int
    41→    total_ops: int
    42→    min_us: float
    43→    max_us: float
    44→    mean_us: float
    45→    median_us: float
    46→    ops_per_sec: float
    47→
    48→    def row(self) -> str:
    49→        return (
    50→            f"│ {self.name:<40s} │ {self.iterations:>3d} × {self.total_ops:>6d} │"
    51→            f" {self.min_us:>8.1f} │ {self.max_us:>8.1f} │"
    52→            f" {self.mean_us:>8.1f} │ {self.median_us:>8.1f} │"
    53→            f" {self.ops_per_sec:>12.0f} │"
    54→        )
    55→
    56→
    57→def run_benchmark(
    58→    name: str,
    59→    fn,
    60→    ops_per_iter: int = 1,
    61→    iterations: int = 5,
    62→    warmup: int = 1,
    63→) -> BenchResult:
    64→    """Run *fn* for *iterations*, return min/max/mean/median per-op timing.
    65→
    66→    Parameters
    67→    ----------
    68→    name : str
    69→        Human-readable benchmark name.
    70→    fn : callable
    71→        Function to benchmark. May accept no arguments.
    72→    ops_per_iter : int
    73→        Number of logical operations performed by one call to *fn*.
    74→    iterations : int
    75→        Number of timing iterations (after warmup).
    76→    warmup : int
    77→        Number of un-timed warmup iterations.
    78→    """
    79→    # Warmup
    80→    for _ in range(warmup):
    81→        fn()
    82→
    83→    times: List[float] = []
    84→    for _ in range(iterations):
    85→        t0 = time.perf_counter()
    86→        fn()
    87→        t1 = time.perf_counter()
    88→        times.append(t1 - t0)
    89→
    90→    per_op_us = [t / ops_per_iter * 1_000_000 for t in times]
    91→    total_ops = iterations * ops_per_iter
    92→    total_time = sum(times)
    93→    ops_sec = total_ops / total_time if total_time > 0 else float("inf")
    94→
    95→    return BenchResult(
    96→        name=name,
    97→        iterations=iterations,
    98→        total_ops=total_ops,
    99→        min_us=min(per_op_us),
   100→        max_us=max(per_op_us),
   101→        mean_us=statistics.mean(per_op_us),
   102→        median_us=statistics.median(per_op_us),
   103→        ops_per_sec=ops_sec,
   104→    )
   105→
   106→
   107→def _print_table(results: List[BenchResult], title: str) -> None:
   108→    """Print a formatted benchmark results table."""
   109→    sep = "├" + "─" * 42 + "┼" + "─" * 16 + "┼" + "─" * 11 + "┼" + "─" * 11 + "┼" + "─" * 11 + "┼" + "─" * 11 + "┼" + "─" * 15 + "┤"
   110→    top = "┌" + "─" * 42 + "┬" + "─" * 16 + "┬" + "─" * 11 + "┬" + "─" * 11 + "┬" + "─" * 11 + "┬" + "─" * 11 + "┬" + "─" * 15 + "┐"
   111→    bot = "└" + "─" * 42 + "┴" + "─" * 16 + "┴" + "─" * 11 + "┴" + "─" * 11 + "┴" + "─" * 11 + "┴" + "─" * 11 + "┴" + "─" * 15 + "┘"
   112→    hdr = "│ {:<40s} │ {:>14s} │ {:>9s} │ {:>9s} │ {:>9s} │ {:>9s} │ {:>13s} │".format(
   113→        title[:40], "Iterations", "Min (µs)", "Max (µs)", "Mean (µs)", "Med (µs)", "Ops/sec"
   114→    )
   115→
   116→    print()
   117→    print(top)
   118→    print(hdr)
   119→    print(sep)
   120→    for r in results:
   121→        print(r.row())
   122→    print(bot)
   123→    print()

   681→class TestParallelExecutionBenchmark(unittest.TestCase):
   682→    """Benchmark: ThreadPoolExecutor overhead for parallel scanning."""

   684→    def test_threadpool_overhead(self):
   685→        """Measure thread pool overhead with no-op tasks."""
   686→        results: List[BenchResult] = []
   687→
   688→        for n_workers in [1, 2, 4, 8]:
   689→            n_tasks = n_workers * 10
   690→
   691→            def bench_pool(nw=n_workers, nt=n_tasks):
   692→                def noop():
   693→                    pass
   694→                with concurrent.futures.ThreadPoolExecutor(max_workers=nw) as pool:
   695→                    futures = [pool.submit(noop) for _ in range(nt)]
   696→                    concurrent.futures.wait(futures)
   697→
   698→            results.append(run_benchmark(
   699→                f"ThreadPoolExecutor ({n_workers}w, {n_tasks} tasks) ×100",
   700→                bench_pool,
   701→                ops_per_iter=n_tasks,
   702→                iterations=5,
   703→            ))
   704→
   705→        _print_table(results, "ThreadPoolExecutor Overhead")
   706→
   707→    def test_threadpool_with_result_collection(self):
   708→        """Benchmark result collection patterns."""
   709→        results: List[BenchResult] = []
   710→
   711→        # Pattern A: as_completed loop (current parallel.py approach)
   712→        def bench_as_completed():
   713→            n = 40
   714→            with concurrent.futures.ThreadPoolExecutor(max_workers=4) as pool:
   715→                futures = {pool.submit(lambda: ("target", {"findings": []}, None)): i
   716→                           for i in range(n)}
   717→                collected = []
   718→                for f in concurrent.futures.as_completed(futures):
   719→                    collected.append(f.result())
   720→
   721→        results.append(run_benchmark(
   722→            "as_completed collection (40 tasks) ×100",
   723→            bench_as_completed,
   724→            ops_per_iter=40,
   725→            iterations=5,
   726→        ))
   727→
   728→        # Pattern B: callback-based collection
   729→        def bench_callback():
   730→            n = 40
   731→            collected_cb = []
   732→            lock = threading.Lock()
   733→
   734→            def on_done(f):
   735→                with lock:
   736→                    collected_cb.append(f.result())
   737→
   738→            with concurrent.futures.ThreadPoolExecutor(max_workers=4) as pool:
   739→                futures = [pool.submit(lambda: ("target", {"findings": []}, None))
   740→                           for _ in range(n)]
   741→                for f in futures:
   742→                    f.add_done_callback(on_done)
   743→                concurrent.futures.wait(futures)
   744→
   745→        results.append(run_benchmark(
   746→            "Callback collection (40 tasks) ×100",
   747→            bench_callback,
   748→            ops_per_iter=40,
   749→            iterations=5,
   750→        ))
   751→
   752→        # Pattern C: executor.map (simpler)
   753→        def bench_map():
   754→            n = 40
   755→            with concurrent.futures.ThreadPoolExecutor(max_workers=4) as pool:
   756→                list(pool.map(lambda _: ("target", {"findings": []}, None), range(n)))
   757→
   758→        results.append(run_benchmark(
   759→            "executor.map collection (40 tasks) ×100",
   760→            bench_map,
   761→            ops_per_iter=40,
   762→            iterations=5,
   763→        ))
   764→
   765→        _print_table(results, "Result Collection Patterns")
